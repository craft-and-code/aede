1use super::{Features, HOP, Moments, PROFILE, RATE, WINDOW, rhythm::Rhythm};
2use rustfft::{Fft, FftPlanner, num_complex::Complex};
3use std::sync::Arc;
4/// Resident FFT/scratch/window buffers; no allocation grows with source duration.
5pub struct Extractor {
6    channels: usize,
7    frames: u64,
8    windows: u64,
9    position: usize,
10    since_window: usize,
11    ring: [Vec<f32>; 2],
12    previous: [f32; 2],
13    crossings: u64,
14    squares: f64,
15    fft: Arc<dyn Fft<f32>>,
16    scratch: Vec<Complex<f32>>,
17    work: Vec<Complex<f32>>,
18    hann: Vec<f32>,
19    power: Vec<f64>,
20    previous_power: Vec<f64>,
21    centroid: Moments,
22    rolloff: Moments,
23    flatness: Moments,
24    chroma: [f64; 12],
25    chroma_bins: Vec<Option<usize>>,
26    rhythm: Rhythm,
27}
28impl Extractor {
29    /// Accept mono/stereo at RATE; a decoder/resampler belongs to the caller.
30    pub fn new(channels: usize) -> Result<Self, String> {
31        if !(1..=2).contains(&channels) {
32            return Err("acoustic profile requires mono or stereo".into());
33        }
34        let fft = FftPlanner::<f32>::new().plan_fft_forward(WINDOW);
35        let scratch = vec![Complex::default(); fft.get_inplace_scratch_len()];
36        Ok(Self {
37            channels,
38            frames: 0,
39            windows: 0,
40            position: 0,
41            since_window: 0,
42            ring: [vec![0.; WINDOW], vec![0.; WINDOW]],
43            previous: [0.; 2],
44            crossings: 0,
45            squares: 0.,
46            fft,
47            scratch,
48            work: vec![Complex::default(); WINDOW],
49            hann: (0..WINDOW)
50                .map(|n| 0.5 - 0.5 * (std::f32::consts::TAU * n as f32 / (WINDOW - 1) as f32).cos())
51                .collect(),
52            power: vec![0.; WINDOW / 2 + 1],
53            previous_power: vec![0.; WINDOW / 2 + 1],
54            centroid: Moments::default(),
55            rolloff: Moments::default(),
56            flatness: Moments::default(),
57            chroma: [0.; 12],
58            chroma_bins: (0..=WINDOW / 2)
59                .map(|bin| {
60                    let frequency = bin as f64 * f64::from(RATE) / WINDOW as f64;
61                    (40.0..=5000.).contains(&frequency).then(|| {
62                        let note = (69. + 12. * (frequency / 440.).log2()).round() as i32;
63                        note.rem_euclid(12) as usize
64                    })
65                })
66                .collect(),
67            rhythm: Rhythm::default(),
68        })
69    }
70    /// Check cancellation/foreground activity before ingesting a packet and each transform.
71    pub fn push(
72        &mut self,
73        samples: &[f32],
74        mut checkpoint: impl FnMut() -> Result<(), String>,
75    ) -> Result<(), String> {
76        checkpoint()?;
77        if !samples.len().is_multiple_of(self.channels)
78            || samples.iter().any(|x| !x.is_finite() || x.abs() > 16.)
79        {
80            return Err("invalid acoustic PCM packet".into());
81        }
82        for frame in samples.chunks_exact(self.channels) {
83            self.frames += 1;
84            if self.frames > u64::from(RATE) * 86400 {
85                return Err("acoustic duration budget exceeded".into());
86            }
87            for (channel, &sample) in frame.iter().enumerate() {
88                if self.frames > 1
89                    && ((self.previous[channel] < 0. && sample >= 0.)
90                        || (self.previous[channel] >= 0. && sample < 0.))
91                {
92                    self.crossings += 1;
93                }
94                self.previous[channel] = sample;
95                self.squares += f64::from(sample).powi(2);
96                self.ring[channel][self.position] = sample;
97            }
98            self.position = (self.position + 1) % WINDOW;
99            self.since_window += 1;
100            if self.frames >= WINDOW as u64 && self.since_window >= HOP {
101                checkpoint()?;
102                self.window();
103                self.since_window = 0;
104            }
105        }
106        Ok(())
107    }
108    fn window(&mut self) {
109        self.windows += 1;
110        self.power.fill(0.);
111        for channel in 0..self.channels {
112            for n in 0..WINDOW {
113                self.work[n] = Complex::new(
114                    self.ring[channel][(self.position + n) % WINDOW] * self.hann[n],
115                    0.,
116                );
117            }
118            self.fft
119                .process_with_scratch(&mut self.work, &mut self.scratch);
120            for (power, bin) in self.power.iter_mut().zip(&self.work) {
121                *power += f64::from(bin.norm_sqr()) / self.channels as f64;
122            }
123        }
124        let sum = self.power[1..].iter().sum::<f64>();
125        if sum < 1e-10 {
126            self.rhythm.push(0.);
127            self.previous_power.copy_from_slice(&self.power);
128            return;
129        }
130        let bin_hz = f64::from(RATE) / WINDOW as f64;
131        let centroid = self
132            .power
133            .iter()
134            .enumerate()
135            .skip(1)
136            .map(|(i, p)| i as f64 * bin_hz * p)
137            .sum::<f64>()
138            / sum;
139        let mut acc = 0.;
140        let mut rolloff = 0.;
141        for (bin, &value) in self.power.iter().enumerate().skip(1) {
142            acc += value;
143            if acc >= sum * 0.85 {
144                rolloff = bin as f64 * bin_hz;
145                break;
146            }
147        }
148        let floor = sum * 1e-12;
149        let geometric = (self.power[1..]
150            .iter()
151            .map(|p| p.max(floor).ln())
152            .sum::<f64>()
153            / (self.power.len() - 1) as f64)
154            .exp();
155        self.centroid.push(centroid);
156        self.rolloff.push(rolloff);
157        self.flatness
158            .push((geometric / (sum / (self.power.len() - 1) as f64)).clamp(0., 1.));
159        let flux = self
160            .power
161            .iter()
162            .zip(&self.previous_power)
163            .map(|(a, b)| (a - b).max(0.))
164            .sum::<f64>()
165            / sum;
166        self.rhythm
167            .push(if flux > 0.01 { flux.min(1.) as f32 } else { 0. });
168        self.previous_power.copy_from_slice(&self.power);
169        for (note, value) in self.chroma_bins.iter().zip(&self.power) {
170            if let Some(note) = note {
171                self.chroma[*note] += value / sum;
172            }
173        }
174    }
175    /// Complete one source, retaining unknown tempo and RMS for silence.
176    pub fn finish(mut self) -> Result<Features, String> {
177        if self.frames == 0 {
178            return Err("empty acoustic source".into());
179        }
180        if self.windows == 0 {
181            self.window();
182        }
183        let chroma_sum = self.chroma.iter().sum::<f64>();
184        let chroma = self.chroma.map(|value| {
185            if chroma_sum > 0. {
186                (value / chroma_sum) as f32
187            } else {
188                0.
189            }
190        });
191        let (tempo_bpm, periodicity) = self.rhythm.finish();
192        let features = Features {
193            profile: PROFILE,
194            frames: self.frames,
195            windows: self.windows,
196            centroid_hz: self.centroid.finish(),
197            rolloff_hz: self.rolloff.finish(),
198            flatness: self.flatness.finish(),
199            zero_crossing: self.crossings as f32 / (self.frames as f32 * self.channels as f32),
200            chroma,
201            tempo_bpm,
202            periodicity,
203            rms_dbfs: if self.squares > 1e-20 {
204                Some(
205                    (10. * (self.squares / (self.frames as f64 * self.channels as f64)).log10())
206                        .max(-240.) as f32,
207                )
208            } else {
209                None
210            },
211        };
212        if !features.valid() {
213            return Err("invalid acoustic descriptor".into());
214        }
215        Ok(features)
216    }
