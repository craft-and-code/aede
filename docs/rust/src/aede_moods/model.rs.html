1//! Request-local acoustic neighbors around explicit personal examples; no emotional labels are inferred.
2use crate::Features;
3const DIMENSIONS: usize = 7;
4fn vector(features: &Features) -> [Option<f32>; DIMENSIONS] {
5    let nyquist = crate::RATE as f32 / 2.;
6    [
7        Some(features.centroid_hz.mean / nyquist),
8        Some(features.centroid_hz.deviation / nyquist),
9        Some(features.rolloff_hz.mean / nyquist),
10        Some(features.flatness.mean),
11        Some(features.zero_crossing),
12        features.tempo_bpm.map(|bpm| (bpm / 120.).log2() / 3.),
13        features.periodicity,
14    ]
15}
16/// Bounded personal exemplars, calibrated with capped inverse variance per descriptor dimension.
17/// Values are normalized signal distances, not probabilities, moods or genre predictions.
18pub struct AcousticModel {
19    seeds: Vec<[Option<f32>; DIMENSIONS]>,
20    weights: [f32; DIMENSIONS],
21    radius: f32,
22}
23impl AcousticModel {
24    /// Require two to sixteen complete usable same-profile examples. Never silently fill missing tempo.
25    pub fn new(examples: &[&Features]) -> Option<Self> {
26        if !(2..=16).contains(&examples.len()) || examples.iter().any(|row| !row.usable()) {
27            return None;
28        }
29        let seeds = examples.iter().map(|row| vector(row)).collect::<Vec<_>>();
30        let base = [1., 0.6, 0.8, 0.7, 0.3, 0.8, 0.2];
31        let mut weights = [0.; DIMENSIONS];
32        for i in 0..DIMENSIONS {
33            let values = seeds.iter().filter_map(|row| row[i]).collect::<Vec<_>>();
34            if values.len() < 2 {
35                continue;
36            }
37            let mean = values.iter().sum::<f32>() / values.len() as f32;
38            let variance =
39                values.iter().map(|x| (x - mean).powi(2)).sum::<f32>() / values.len() as f32;
40            weights[i] = base[i] / (0.01 + variance);
41        }
42        let mut model = Self {
43            seeds,
44            weights,
45            radius: 0.,
46        };
47        let mut spread = 0_f32;
48        for i in 0..model.seeds.len() {
49            let nearest = model
50                .seeds
51                .iter()
52                .enumerate()
53                .filter(|(j, _)| *j != i)
54                .filter_map(|(_, other)| model.between(&model.seeds[i], other))
55                .fold(f32::INFINITY, f32::min);
56            spread = spread.max(nearest);
57        }
58        model.radius = (spread * 1.25 + 0.025).clamp(0.045, 0.18);
59        Some(model)
60    }
61    fn between(&self, a: &[Option<f32>; DIMENSIONS], b: &[Option<f32>; DIMENSIONS]) -> Option<f32> {
62        let mut sum = 0.;
63        let mut weight = 0.;
64        let mut known = 0;
65        for i in 0..DIMENSIONS {
66            if let (Some(a), Some(b)) = (a[i], b[i])
67                && self.weights[i] > 0.
68            {
69                sum += self.weights[i] * (a - b).powi(2);
70                weight += self.weights[i];
71                known += 1;
72            }
73        }
74        (known >= 4 && weight > 0.).then(|| (sum / weight).sqrt())
75    }
76    /// Nearest usable example distance; silent, invalid or insufficient shared evidence stays unknown.
77    pub fn distance(&self, features: &Features) -> Option<f32> {
78        if !features.usable() {
79            return None;
80        }
81        let current = vector(features);
82        self.seeds
83            .iter()
84            .filter_map(|seed| self.between(seed, &current))
85            .min_by(f32::total_cmp)
86    }
87    /// Capped acceptance radius calibrated from the personal examples, independently of mastering gain.
88    pub fn radius(&self) -> f32 {
89        self.radius
90    }
91    /// Number of usable personal examples admitted by this model.
92    pub fn examples(&self) -> usize {
93        self.seeds.len()
94    }
95}
