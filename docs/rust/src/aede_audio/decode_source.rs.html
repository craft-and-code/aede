1//! A metadata-free view of native FLAC for the playback demuxer.
2//!
3//! Playback needs STREAMINFO and audio, not artwork or textual metadata. The
4//! pinned demuxer's metadata readers can allocate from nested lengths before
5//! checking the containing block. Skip those bodies on the original descriptor
6//! and expose only a fixed-size synthetic header followed by untouched audio.
7
8use std::fs::File;
9use std::io::{self, Read, Seek, SeekFrom};

13use symphonia::core::io::{MediaSource, MediaSourceStream, ReadBytes, SeekBuffered};
14use symphonia::core::probe::{Instantiate, Probe, QueryDescriptor};
15use symphonia::default::formats::FlacReader;

19const PREFIX_BYTES: usize = 42;
20// Bound even zero-length blocks so hostile metadata cannot monopolize a
21// playback worker with endless seeks; artwork bodies remain uncapped.
22const MAX_METADATA_BLOCKS: usize = 65_536;
23const MAX_LEADING_ID3_BLOCKS: usize = 65_536;
24
25/// An opened FLAC-capable container selected by the protected source probe.
26pub struct FlacProbe {
27    /// Reader retaining the same descriptor supplied to [`FlacAudioSource::probe`].
28    pub format: Box<dyn FormatReader>,
29    /// Whether the selected marker was native FLAC rather than Ogg.
30    pub native_container: bool,
31}
32
33impl FlacAudioSource {
34    /// Probe a regular opened file for native or Ogg FLAC without decoding audio.
35    ///
36    /// Native FLAC uses the same metadata-free source as ordinary shared decoding,
37    /// including scanned prefixes and contiguous leading ID3 blocks. Other format
38    /// markers return `None` without constructing their reader. This supplies no
39    /// PCM or MD5 verdict; the caller selects and configures its codec validator.
40    pub fn probe(file: File) -> Result<Option<FlacProbe>, DecodeError> {
41        if !file.metadata().map_err(decode_error)?.is_file() {
42            return Err(decode_error("FLAC source is not a regular file"));
43        }
44        let input = prepare(file)?;
45        let mut source = MediaSourceStream::new(input, Default::default());
46        let file_length = source
47            .byte_len()
48            .ok_or_else(|| decode_error("FLAC source length is unknown"))?;
49        let probe = probe();
50        let mut id3_blocks = 0;
51        loop {
52            let selected = match probe.next(&mut source) {
53                Ok(selected) => selected,
54                Err(_) => return Ok(None),
55            };
56            let start = source.pos();
57            let marker = source.read_quad_bytes().map_err(decode_error)?;
58            source.seek_buffered_rev(4);
59            match selected {
60                Instantiate::Format(instantiate) => {
61                    let native_container = &marker == b"fLaC";
62                    if !native_container && &marker != b"OggS" {
63                        return Ok(None);
64                    }
65                    let format = match instantiate(source, &FormatOptions::default()) {
66                        Ok(format) => format,
67                        // A selected native marker must never fall through to an
68                        // ordinary decoder after its protected source was refused.
69                        Err(error) if native_container => return Err(decode_error(error)),
70                        Err(_) => return Ok(None),
71                    };
72                    return Ok(Some(FlacProbe {
73                        format,
74                        native_container,
75                    }));
76                }
77                Instantiate::Metadata(_) => {
78                    // Discovery needs only ID3's extent. Do not parse tag bodies
79                    // merely to decide which audio reader should own the source.
80                    if &marker[..3] != b"ID3" {
81                        return Ok(None);
82                    }
83                    if id3_blocks == MAX_LEADING_ID3_BLOCKS {
84                        return Err(decode_error(
85                            "leading ID3 metadata exceeds the header-count limit",
86                        ));
87                    }
88                    id3_blocks += 1;
89                    let mut header = [0; 10];
90                    ReadBytes::read_buf_exact(&mut source, &mut header).map_err(decode_error)?;
91                    let end = start
92                        .checked_add(id3_block_length(&header))
93                        .filter(|&end| end <= file_length)
94                        .ok_or_else(|| {
95                            decode_error("leading ID3 metadata exceeds the source length")
96                        })?;
97                    source.seek(SeekFrom::Start(end)).map_err(decode_error)?;
98                }
99            }
100        }
101    }
102}

104/// A seekable STREAMINFO-only native FLAC view followed by original audio bytes.
105///
106/// Optional metadata is skipped without parsing or allocating its bodies. The
107/// opened source is never modified, and reads retain its captured length.
108pub struct FlacAudioSource {
109    source: Box<dyn MediaSource>,
110    prefix: [u8; PREFIX_BYTES],
111    audio_offset: u64,
112    length: u64,
113    position: u64,
114}
115
116impl FlacAudioSource {
117    /// Inspect native metadata on the same opened regular file.
118    ///
119    /// A leading ID3v2 block, including its optional footer, is skipped from its
120    /// fixed-size header. `None` means its following bytes are not native FLAC.
121    /// The resulting virtual file consists of `fLaC`, one final STREAMINFO
122    /// block and original audio; it never changes the actual file or metadata.
123    pub fn new(mut file: File) -> Result<Option<Self>, DecodeError> {
124        let metadata = file.metadata().map_err(decode_error)?;
125        if !metadata.is_file() {
126            return Err(decode_error("FLAC source is not a regular file"));
127        }
128        let file_length = metadata.len();
129        let start = skip_leading_id3(&mut file)
130            .map_err(|error| decode_error(format!("input/output error: {error}")))?;
131        if start > file_length {
132            return Err(decode_error(
133                "leading ID3 metadata exceeds the source length",
134            ));
135        }
136        Self::from_source(Box::new(file), start, file_length)
137    }
138
139    fn from_source(
140        mut source: Box<dyn MediaSource>,
141        start: u64,
142        file_length: u64,
143    ) -> Result<Option<Self>, DecodeError> {
144        source.seek(SeekFrom::Start(start)).map_err(decode_error)?;
145        let mut magic = [0; 4];
146        match source.read_exact(&mut magic) {
147            Ok(()) => {}
148            Err(error) if error.kind() == io::ErrorKind::UnexpectedEof => return Ok(None),
149            Err(error) => return Err(decode_error(error)),
150        }
151        if &magic != b"fLaC" {
152            return Ok(None);
153        }
154
155        let mut prefix = [0; PREFIX_BYTES];
156        prefix[..8].copy_from_slice(b"fLaC\x80\0\0\x22");
157        let mut offset = start
158            .checked_add(4)
159            .filter(|&end| end <= file_length)
160            .ok_or_else(|| decode_error("FLAC marker exceeds the source length"))?;
161        let mut first = true;
162        let mut blocks = 0;
163        loop {
164            if blocks == MAX_METADATA_BLOCKS {
165                return Err(decode_error("FLAC metadata exceeds the header-count limit"));
166            }
167            blocks += 1;
168            let header_end = offset
169                .checked_add(4)
170                .filter(|&end| end <= file_length)
171                .ok_or_else(|| decode_error("FLAC metadata header is truncated"))?;
172            source.seek(SeekFrom::Start(offset)).map_err(decode_error)?;
173            let mut header = [0; 4];
174            source.read_exact(&mut header).map_err(decode_error)?;
175            let kind = header[0] & 0x7f;
176            let body_length = u64::from(u32::from_be_bytes([0, header[1], header[2], header[3]]));
177            let block_end = header_end
178                .checked_add(body_length)
179                .filter(|&end| end <= file_length)
180                .ok_or_else(|| decode_error("FLAC metadata block exceeds the source length"))?;
181            if first {
182                if kind != 0 || body_length != 34 {
183                    return Err(decode_error(
184                        "FLAC must begin with a 34-byte STREAMINFO block",
185                    ));
186                }
187                source.read_exact(&mut prefix[8..]).map_err(decode_error)?;
188                first = false;
189            } else if kind == 0 {
190                return Err(decode_error("FLAC has duplicate STREAMINFO metadata"));
191            }
192            if kind == 127 {
193                return Err(decode_error("FLAC has forbidden metadata type 127"));
194            }
195            offset = block_end;
196            if header[0] & 0x80 != 0 {
197                break;
198            }
199        }
200        source.seek(SeekFrom::Start(offset)).map_err(decode_error)?;
201        let length = (PREFIX_BYTES as u64)
202            .checked_add(file_length - offset)
203            .ok_or_else(|| decode_error("FLAC virtual source length is unrepresentable"))?;
204        Ok(Some(Self {
205            source,
206            prefix,
207            audio_offset: offset,
208            length,
209            position: 0,
210        }))
211    }
212}

214impl Read for FlacAudioSource {
215    fn read(&mut self, buffer: &mut [u8]) -> io::Result<usize> {
216        if buffer.is_empty() || self.position >= self.length {
217            return Ok(0);
218        }
219        if self.position < PREFIX_BYTES as u64 {
220            let start = self.position as usize;
221            let count = buffer.len().min(PREFIX_BYTES - start);
222            buffer[..count].copy_from_slice(&self.prefix[start..start + count]);
223            self.position += count as u64;
224            return Ok(count);
225        }
226        let remaining = usize::try_from(self.length - self.position).unwrap_or(usize::MAX);
227        let limit = buffer.len().min(remaining);
228        let count = self.source.read(&mut buffer[..limit])?;
229        self.position += count as u64;
230        Ok(count)
231    }
232}

234impl Seek for FlacAudioSource {
235    fn seek(&mut self, position: SeekFrom) -> io::Result<u64> {
236        let requested = match position {
237            SeekFrom::Start(position) => i128::from(position),
238            SeekFrom::End(offset) => i128::from(self.length) + i128::from(offset),
239            SeekFrom::Current(offset) => i128::from(self.position) + i128::from(offset),
240        };
241        let requested = u64::try_from(requested).map_err(|_| {
242            io::Error::new(io::ErrorKind::InvalidInput, "invalid FLAC virtual seek")
243        })?;
244        let audio_position = requested.saturating_sub(PREFIX_BYTES as u64);
245        let original = self
246            .audio_offset
247            .checked_add(audio_position)
248            .ok_or_else(|| {
249                io::Error::new(
250                    io::ErrorKind::InvalidInput,
251                    "FLAC virtual seek exceeds the source range",
252                )
253            })?;
254        self.source.seek(SeekFrom::Start(original))?;
255        self.position = requested;
256        Ok(requested)
257    }
258}

261    fn is_seekable(&self) -> bool {
262        true
263    }
264
265    fn byte_len(&self) -> Option<u64> {
266        Some(self.length)
267    }

270/// Bypass leading ID3 bodies only when they precede a native FLAC stream.
271pub(super) fn prepare(mut file: File) -> Result<Box<dyn MediaSource>, DecodeError> {
272    let file_length = file.metadata().map_err(decode_error)?.len();
273    let Some(start) = leading_id3_flac_start(&mut file, file_length)? else {
274        file.seek(SeekFrom::Start(0)).map_err(decode_error)?;
275        return Ok(Box::new(file));
276    };
277    let source = FlacAudioSource::from_source(Box::new(file), start, file_length)?
278        .ok_or_else(|| decode_error("FLAC marker was not found after leading ID3 metadata"))?;
279    Ok(Box::new(source))
280}

282fn leading_id3_flac_start(file: &mut File, file_length: u64) -> Result<Option<u64>, DecodeError> {
283    let mut start = 0_u64;
284    let mut blocks = 0;
285    loop {
286        file.seek(SeekFrom::Start(start)).map_err(decode_error)?;
287        let mut marker = [0; 4];
288        match file.read_exact(&mut marker) {
289            Ok(()) => {}
290            Err(error) if error.kind() == io::ErrorKind::UnexpectedEof => return Ok(None),
291            Err(error) => return Err(decode_error(error)),
292        }
293        if &marker == b"fLaC" {
294            return Ok((blocks != 0).then_some(start));
295        }
296        if &marker[..3] != b"ID3" {
297            return Ok(None);
298        }
299        if blocks == MAX_LEADING_ID3_BLOCKS {
300            return Err(decode_error(
301                "leading ID3 metadata exceeds the header-count limit",
302            ));
303        }
304        blocks += 1;
305        let mut header = [0; 10];
306        header[..4].copy_from_slice(&marker);
307        match file.read_exact(&mut header[4..]) {
308            Ok(()) => {}
309            Err(error) if error.kind() == io::ErrorKind::UnexpectedEof => return Ok(None),
310            Err(error) => return Err(decode_error(error)),
311        }
312        start = start
313            .checked_add(id3_block_length(&header))
314            .filter(|&end| end <= file_length)
315            .ok_or_else(|| decode_error("leading ID3 metadata exceeds the source length"))?;
316    }
317}

319// Preserve the catalogue tag helper's header-only ID3 size/footer semantics
320// without introducing an application or tag-parser dependency into decoding.
321fn skip_leading_id3(file: &mut File) -> io::Result<u64> {
322    file.seek(SeekFrom::Start(0))?;
323    let mut header = [0; 10];
324    match file.read_exact(&mut header) {
325        Ok(()) => {}
326        Err(error) if error.kind() == io::ErrorKind::UnexpectedEof => return Ok(0),
327        Err(error) => return Err(error),
328    }
329    if &header[..3] != b"ID3" {
330        return Ok(0);
331    }
332    Ok(id3_block_length(&header))
333}
334
335fn id3_block_length(header: &[u8; 10]) -> u64 {
336    let size = header[6..10]
337        .iter()
338        .fold(0_u64, |size, byte| (size << 7) | u64::from(byte & 0x7f));
339    10 + size + if header[5] & 0x10 != 0 { 10 } else { 0 }
340}

342/// Retain Symphonia's enabled registry and replace only native FLAC creation.
343pub(super) fn probe() -> Probe {
344    let mut probe = Probe::default();
345    for descriptor in FlacReader::query() {
346        let mut descriptor = *descriptor;
347        descriptor.inst = Instantiate::Format(flac_reader);
348        probe.register(&descriptor);
349    }
350    symphonia::default::register_enabled_formats(&mut probe);
351    probe
352}

354fn flac_reader(
355    source: MediaSourceStream,
356    options: &FormatOptions,
357) -> Result<Box<dyn FormatReader>, SymError> {
358    // The probe may have consumed leading tags or scanned unrelated bytes.
359    // Wrap its actual selected marker, preserving those established offsets.
360    let start = source.pos();
361    let file_length = source
362        .byte_len()
363        .ok_or(SymError::Unsupported("FLAC source length is unknown"))?;
364    let source = FlacAudioSource::from_source(Box::new(source), start, file_length)
365        .map_err(|error| SymError::IoError(io::Error::new(io::ErrorKind::InvalidData, error)))?
366        .ok_or(SymError::DecodeError(
367            "FLAC marker was not found at the probed position",
368        ))?;
369    let reader = FlacReader::try_new(
370        MediaSourceStream::new(Box::new(source), Default::default()),
371        options,
372    )?;
373    Ok(Box::new(reader))
374}
