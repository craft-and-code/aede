22/// A local decode failure, independent of any application's analysis errors.
23#[derive(Debug)]
24pub struct DecodeError(pub String);
25impl fmt::Display for DecodeError {
26    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
27        f.write_str(&self.0)
28    }
29}
30impl std::error::Error for DecodeError {}

32/// Packet-at-a-time PCM reader. No device, analyzer or complete-track allocation.
33pub struct PcmStreamDecoder {
34    format: Box<dyn FormatReader>,
35    decoder: Box<dyn Decoder>,
36    track_id: u32,
37    scratch: Option<SampleBuffer<f32>>,
38    capacity: u64,
39    failed: bool,
40    /// Source frames per second.
41    pub sample_rate: u32,
42    /// Interleaved samples per frame.
43    pub channels: usize,
44    /// Speaker mask from the same probed header, when available.
45    pub channel_mask: Option<u32>,
46}
47impl PcmStreamDecoder {
48    /// Open a local regular file and probe its default audio track with gapless trimming.
49    pub fn open(path: &Path) -> Result<Self, DecodeError> {
50        // Opening a FIFO can wait for a writer before descriptor metadata is available.
51        if !fs::metadata(path)
52            .map_err(|e| DecodeError(e.to_string()))?
53            .is_file()
54        {
55            return Err(DecodeError("audio source is not a regular file".into()));
56        }
57        let file = File::open(path).map_err(|e| DecodeError(e.to_string()))?;
58        if !file
59            .metadata()
60            .map_err(|e| DecodeError(e.to_string()))?
61            .is_file()
62        {
63            return Err(DecodeError("audio source is not a regular file".into()));
64        }
65        let mut hint = Hint::new();
66        if let Some(extension) = path.extension().and_then(|s| s.to_str()) {
67            hint.with_extension(extension);
68        }
69        let input = source::prepare(file)?;
70        let probed = catch_unwind(AssertUnwindSafe(|| {
71            source::probe().format(
72                &hint,
73                MediaSourceStream::new(input, Default::default()),
74                &FormatOptions {
75                    enable_gapless: true,
76                    ..Default::default()
77                },
78                &MetadataOptions::default(),
79            )
80        }))
81        .map_err(|_| DecodeError("audio probe panicked".into()))?
82        .map_err(|e| DecodeError(format!("probe failed: {e}")))?;
83        let format = probed.format;
84        let track = format
85            .default_track()
86            .ok_or_else(|| DecodeError("no default track".into()))?;
87        let sample_rate = track
88            .codec_params
89            .sample_rate
90            .filter(|v| *v > 0)
91            .ok_or_else(|| DecodeError("missing sample rate".into()))?;
92        let channels = track
93            .codec_params
94            .channels
95            .map(|v| v.count())
96            .filter(|v| *v > 0 && *v <= 32)
97            .ok_or_else(|| DecodeError("missing or unsupported channels".into()))?;
98        let channel_mask = track.codec_params.channels.map(|v| v.bits());
99        let track_id = track.id;
100        let decoder = catch_unwind(AssertUnwindSafe(|| {
101            symphonia::default::get_codecs().make(&track.codec_params, &DecoderOptions::default())
102        }))
103        .map_err(|_| DecodeError("audio decoder construction panicked".into()))?
104        .map_err(|e| DecodeError(format!("no decoder: {e}")))?;
105        Ok(Self {
106            format,
107            decoder,
108            track_id,
109            scratch: None,
110            capacity: 0,
111            failed: false,
112            sample_rate,
113            channels,
114            channel_mask,
115        })
116    }
117    /// Yield one complete finite PCM packet; retain scratch between calls.
118    /// As in the original preview path, recoverable corrupt packets are skipped.
119    pub fn next_chunk(&mut self) -> Result<Option<Vec<f32>>, DecodeError> {
120        self.next_chunk_admitted(|| Ok(()))
121    }
122    /// Retire packet reads (including skipped corrupt/non-audio packets) between I/O stages.
123    /// An already running regular-file read is not forcibly interrupted.
124    pub fn next_chunk_admitted(
125        &mut self,
126        mut current: impl FnMut() -> Result<(), DecodeError>,
127    ) -> Result<Option<Vec<f32>>, DecodeError> {
128        if self.failed {
129            return Err(DecodeError(
130                "audio decoder cannot continue after a panic".into(),
131            ));
132        }
133        loop {
134            current()?;
135            // A panicked reader or codec may have mutated its internal state. Retire it
136            // permanently rather than retrying or publishing a misleading EOF.
137            let packet = match catch_unwind(AssertUnwindSafe(|| self.format.next_packet())) {
138                Ok(result) => result,
139                Err(_) => {
140                    self.failed = true;
141                    return Err(DecodeError("audio packet reader panicked".into()));
142                }
143            };
144            let packet = match packet {
145                Ok(packet) => packet,
146                Err(SymError::IoError(e)) if e.kind() == std::io::ErrorKind::UnexpectedEof => {
147                    return Ok(None);
148                }
149                Err(e) => return Err(DecodeError(format!("packet error: {e}"))),
150            };
151            current()?;
152            if packet.track_id() != self.track_id {
153                continue;
154            }
155            let decoder = self.decoder.as_mut();
156            let decoded = match catch_unwind(AssertUnwindSafe(move || {
157                // Consume the borrowed codec so its decoded buffer can leave this closure.
158                let active_decoder = decoder;
159                active_decoder.decode(&packet)
160            })) {
161                Ok(result) => result,
162                Err(_) => {
163                    self.failed = true;
164                    return Err(DecodeError("audio codec panicked".into()));
165                }
166            };
167            match decoded {
168                Ok(decoded) => {
169                    current()?;
170                    if decoded.spec().rate != self.sample_rate
171                        || decoded.spec().channels.count() != self.channels
172                    {
173                        return Err(DecodeError("stream changed during decoding".into()));
174                    }
175                    let capacity = decoded.capacity() as u64;
176                    if capacity > 1_048_576 / self.channels as u64 {
177                        return Err(DecodeError("decoded packet exceeds PCM budget".into()));
178                    }
179                    if self.scratch.is_none() || capacity > self.capacity {
180                        self.scratch = Some(SampleBuffer::new(capacity, *decoded.spec()));
181                        self.capacity = capacity;
182                    }
183                    let scratch = self
184                        .scratch
185                        .as_mut()
186                        .ok_or_else(|| DecodeError("missing PCM scratch".into()))?;
187                    scratch.copy_interleaved_ref(decoded);
188                    let samples = scratch.samples();
189                    if !samples.len().is_multiple_of(self.channels)
190                        || samples.iter().any(|v| !v.is_finite())
191                    {
192                        return Err(DecodeError("incomplete or non-finite PCM sample".into()));
193                    }
194                    return Ok(Some(samples.to_vec()));
195                }
196                Err(SymError::DecodeError(_)) => continue,
197                Err(SymError::IoError(e)) if e.kind() == std::io::ErrorKind::UnexpectedEof => {
198                    return Ok(None);
199                }
200                Err(e) => return Err(DecodeError(format!("decode error: {e}"))),
201            }
202        }
203    }
204}
