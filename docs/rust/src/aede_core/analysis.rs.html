1//! FlacCompagnon analyses produced in-process or imported from a saved report.
2//!
3//! Aède describes what a file *contains* by reading its structure; it does not
4//! decode. A decoder answers questions no structural read can: is this FLAC a
5//! re-encoded MP3, was it upsampled, where does its spectrum stop, how loud is
6//! it really, and — the decisive one — does the audio still match the MD5 the
7//! encoder wrote in STREAMINFO.
8//!
9//! [FlacCompagnon](https://craft-and-code.github.io/FlacCompagnon/) does that
10//! pass through `aede analyze`; its saved reports can also be imported later.
11//!
12//! **Acoustic values are never merged into Aède's own.** They sit beside them,
13//! attributed to their source, because a verdict carries the method that
14//! produced it: overwriting `effective_bit_depth` — read from the wasted-bits
15//! counts of the frames — with a figure obtained by decoding would leave the
16//! catalog unable to say where the number came from, and unable to notice that
17//! the two disagree. Noticing is the point.
18

30/// One file, as FlacCompagnon measured it.
31///
32/// Every measurement is optional: a report may predate a field, omit one, or
33/// come from a tool that does not compute it. Nothing here is required for the
34/// catalog to work — this is entirely additional.
35#[derive(Debug, Clone, Default, PartialEq)]
36pub struct FileAnalysis {
37    /// Absolute path of the file this describes.
38    ///
39    /// The primary key is the **path**, not an identifier into the catalog. Two
40    /// reasons, and the second is the important one. Identifiers are positions
41    /// that a scan renumbers, so an identifier would have to be remapped after
42    /// every scan. And an analysis may perfectly well describe a file the
43    /// catalog does not hold *yet*: someone can run the analysis first and
44    /// build their library afterwards. Keying on the path lets such a record
45    /// wait, and attach itself the day the file is scanned. A whole-file MD5
46    /// can identify a renamed copy when the old path no longer exists.
47    pub path: String,
48    /// Tool that produced the measurements, `flaccompagnon` for now.
49    pub source: String,
50    /// Version of the report format, kept so an old import stays readable.
51    pub source_version: u32,
52    /// When it was imported, in seconds since the Unix epoch.
53    pub imported_at: u64,
54    /// Report modification time, or in-process measurement completion, in
55    /// nanoseconds since the Unix epoch. Zero means an older stored result
56    /// whose date is unknown. Distinct from the audio file's own mtime.
57    pub result_at_ns: u64,
58    /// Size the analysed file had, in bytes.
59    ///
60    /// With [`FileAnalysis::modified_unix`], this makes an exact-path or legacy
61    /// name-and-size match expire when the file changes. A moved file with a
62    /// matching whole-file MD5 can keep its analysis after a timestamp change.
63    pub size_bytes: u64,
64    /// Modification date the analysed file had.
65    pub modified_unix: u64,
66    /// MD5 of the entire file, including tags and artwork, when supplied.
67    /// Used to identify a moved file without trusting its name or timestamp.
68    pub file_md5: Option<String>,
69    /// Complete file entry from the source report, including measurements that
70    /// Aède does not interpret yet. It keeps the source path when a matching
71    /// file is later moved; [`FileAnalysis::path`] is the attached path.
72    pub source_data: Option<Json>,
73
74    /// Verdict on the MD5 stored in STREAMINFO: `Match`, `Mismatch`,
75    /// `NoSignature`, `Present`, `Error`.
76    ///
77    /// A *state*, not a hash — the tool compares and reports, it does not
78    /// publish the digest. `Match` is equivalent to a successful `flac -t`.
79    pub md5_state: Option<String>,
80    /// What went wrong, when the state is an error.
81    pub md5_detail: Option<String>,
82
83    /// Bit depth actually carried, as measured by decoding.
84    pub real_bit_depth: Option<u16>,
85    /// Proportion of samples that show requantisation.
86    pub requant_rate: Option<f64>,
87    /// Both channels hold the same signal.
88    pub fake_stereo: Option<bool>,
89    /// The extension does not match the container found inside.
90    pub ext_mismatch: Option<bool>,
91
92    /// The file was made from a lower-resolution source: `none`, `suspected`,
93    /// `detected`.
94    pub transcoding: Option<String>,
95    /// A lossless file built from a lossy one.
96    pub upscaling: Option<bool>,
97    /// Sample rate raised above what the content justifies.
98    pub upsampling: Option<bool>,
99    /// One-word verdict, as the tool words it.
100    pub summary: Option<String>,
101    /// The same verdict in a sentence, worth showing as it stands.
102    pub detail: Option<String>,
103
104    /// Frequency where the spectrum stops, in hertz. A cutoff well below half
105    /// the sample rate is the signature of a lossy ancestry.
106    pub cutoff_hz: Option<f64>,
107    /// That frequency as a fraction of the Nyquist limit.
108    pub cutoff_ratio: Option<f64>,
109
110    /// Dynamic range in decibels: the lower, the more compressed the master.
111    pub dr_db: Option<f64>,
112    /// Highest sample value, in dBFS.
113    pub peak_dbfs: Option<f64>,
114    /// True peak, in dBTP — what a converter will actually have to produce.
115    pub true_peak_dbtp: Option<f64>,
116    /// FlacCompagnon's measured integrated track loudness, in LUFS.
117    pub integrated_lufs: Option<f64>,
118    /// Samples sitting at full scale.
119    pub clipped_samples: Option<u64>,
120    /// Runs of consecutive clipped samples.
121    pub clip_events: Option<u64>,
122    /// The tool's own conclusion on clipping.
123    pub clipped: Option<bool>,
124
125    /// What stopped the analysis of this file, when something did.
126    pub error: Option<String>,
127}

129impl FileAnalysis {
130    /// `true` when the analysis still describes the file as it is now.
131    ///
132    /// Uses the source report's unchanged size and whole-second modification
133    /// date. Unlike the scanner's native cache identity, this contract has no
134    /// subsecond precision and cannot distinguish all same-second edits.
135    pub fn still_applies(&self, size: u64, mtime: u64) -> bool {
136        self.size_bytes == size && self.modified_unix == mtime
137    }
138
139    /// `true` when the decoded audio did not match the file's own MD5.
140    pub fn md5_failed(&self) -> bool {
141        self.md5_state.as_deref() == Some("Mismatch")
142    }
143
144    /// `true` when the file is suspected of being built from a lossy source.
145    pub fn suspect_encoding(&self) -> bool {
146        matches!(
147            self.transcoding.as_deref(),
148            Some("detected") | Some("suspected")
149        ) || self.upscaling == Some(true)
150            || self.upsampling == Some(true)
151    }
152
153    /// Last component of [`FileAnalysis::path`].
154    ///
155    /// With the size, this is what identifies a file that has moved since it
156    /// was analysed — a name and a byte count together are very nearly unique
157    /// in a music library.
158    pub fn file_name(&self) -> &str {
159        crate::text::file_name(&self.path)
160    }
161}

174/// Waiting and stale **folders** listed by [`Attachment`]; the rest are
175/// counted only.
176///
177/// Bounded on folders rather than on files because that is the unit a reader
178/// acts on: one album of fourteen tracks is one decision, and fourteen rows
179/// saying so push the next album off the screen.
180const FOLDERS_SHOWN: usize = 10;
181
182/// What became of a batch of records handed to [`merge_into`].
183#[derive(Debug, Default, Clone, PartialEq)]
184pub struct Attachment {
185    /// Records whose path the catalog already knew.
186    pub matched: usize,
187    /// Records attached to a file that has moved since it was analysed.
188    pub moved: usize,
189    /// Records about bytes that have changed since, and therefore dropped.
190    pub stale: usize,
191    /// Records kept although no file matches them yet.
192    pub waiting: usize,
193    /// Incoming records superseded by a more recent result for the same file
194    /// and source. Re-reading an old report must not undo a newer analysis.
195    pub older: usize,
196    /// The first few waiting **folders**, each with how many wait in it, for a
197    /// report that names rather than counts.
198    ///
199    /// Folders, not paths, and for the reason a reader would give: what one
200    /// does with a waiting record is scan the folder it names, or decide that
201    /// folder is gone. The file name inside it settles nothing, and listing
202    /// every file of an album spends ten rows to say one thing. It also
203    /// survives the column width — a path cut to fit is cut at the *tail*,
204    /// where the file name is, leaving the head that identifies the folder
205    /// exactly where it cannot be read.
206    pub waiting_folders: BTreeMap<String, usize>,
207    /// The first few **folders** holding a record dropped as stale, each with
208    /// how many, for the same reason `waiting_folders` names rather than
209    /// counts: "some files changed" answers how many, not which, and which is
210    /// the one thing a reader needs to know where to point FlacCompagnon again.
211    pub stale_folders: BTreeMap<String, usize>,
212}

214impl Attachment {
215    /// Records that found their file, one way or the other.
216    pub fn attached(&self) -> usize {
217        self.matched + self.moved
218    }

221/// Counts one more record against the folder it sits in, capping how many
222/// distinct folders are kept rather than how many records count against them
223/// — a folder already listed keeps counting past the cap, only a new one is
224/// refused room.
225fn note_folder(folders: &mut BTreeMap<String, usize>, path: &str) {
226    let folder = crate::text::folder(path);
227    if let Some(count) = folders.get_mut(folder) {
228        *count += 1;
229    } else if folders.len() < FOLDERS_SHOWN {
230        folders.insert(folder.to_string(), 1);
231    }
232}

234/// Stores records in the catalog, attaching each to the file it describes.
235///
236/// Three outcomes make the order of analysis and scanning irrelevant:
237///
238/// - the path is one the catalog holds — the usual case;
239/// - the path is unknown, but one file has the same size and whole-file MD5;
240///   older reports without a digest can use a unique name and size instead;
241///   the record is refiled under its current location;
242/// - nothing matches, and the record is kept as it is. The folder it names has
243///   most likely not been scanned yet.
244///
245/// Exact-path and legacy name-and-size matches also require the old size and
246/// modification date. A digest match proves the bytes agree, so a changed
247/// timestamp alone is harmless. Duplicate digests remain unresolved.
248pub fn merge_into(catalog: &mut Catalog, records: Vec<FileAnalysis>, now: u64) -> Attachment {
249    let mut out = Attachment::default();
250    let mut resolved: Vec<FileAnalysis> = Vec::with_capacity(records.len());
251
252    {
253        let known: BTreeMap<&str, (u64, u64)> = catalog
254            .files
255            .iter()
256            .map(|f| (f.path.as_str(), (f.size, f.mtime)))
257            .collect();
258        let by_name_size = name_size_index(catalog);
259        let wanted_sizes = records
260            .iter()
261            .filter(|record| !known.contains_key(record.path.as_str()))
262            .filter(|record| record.file_md5.is_some())
263            .map(|record| record.size_bytes)
264            .collect();
265        let by_digest = file_md5_index(catalog, &wanted_sizes);
266
267        for mut record in records {
268            match known.get(record.path.as_str()) {
269                Some(&(size, mtime)) => {
270                    if !record.still_applies(size, mtime) {
271                        out.stale += 1;
272                        note_folder(&mut out.stale_folders, &record.path);
273                        continue;
274                    }
275                    out.matched += 1;
276                }
277                None => {
278                    if let Some(digest) = &record.file_md5 {
279                        if let Some(&(path, mtime)) =
280                            by_digest.get(&(record.size_bytes, digest.clone()))
281                        {
282                            record.path = path.to_string();
283                            record.modified_unix = mtime;
284                            out.moved += 1;
285                        } else {
286                            out.waiting += 1;
287                            note_folder(&mut out.waiting_folders, &record.path);
288                        }
289                    } else {
290                        match by_name_size.get(&(record.file_name(), record.size_bytes)) {
291                            Some(&(path, mtime)) => {
292                                if !record.still_applies(record.size_bytes, mtime) {
293                                    out.stale += 1;
294                                    note_folder(&mut out.stale_folders, path);
295                                    continue;
296                                }
297                                record.path = path.to_string();
298                                out.moved += 1;
299                            }
300                            None => {
301                                out.waiting += 1;
302                                note_folder(&mut out.waiting_folders, &record.path);
303                            }
304                        }
305                    }
306                }
307            }
308            record.imported_at = now;
309            resolved.push(record);
310        }
311    }
312
313    for record in resolved {
314        if !store_one(catalog, record) {
315            out.older += 1;
316        }
317    }
318    sort_analyses(catalog);
319    out
320}

322/// Attaches the records that were still waiting for their file.
323///
324/// Called after a scan. A record waits because nothing matched it at import
325/// time, and the usual reason is that its folder had not been scanned yet — but
326/// also that the path it names is not the path the catalog settled on. Watched
327/// folders are stored canonical, so a report written against a symbolic link,
328/// or against `/var` where the system says `/private/var`, names the same file
329/// by another route. A whole-file MD5 resolves the path when available; older
330/// reports still use a unique name, size and modification date.
331///
332/// Returns how many were attached.
333pub fn reconcile(catalog: &mut Catalog) -> usize {
334    let mut moves: Vec<(usize, String, u64)> = Vec::new();
335    {
336        let known: BTreeSet<&str> = catalog.files.iter().map(|f| f.path.as_str()).collect();
337        let by_name_size = name_size_index(catalog);
338        let wanted_sizes = catalog
339            .analyses
340            .iter()
341            .filter(|record| !known.contains(record.path.as_str()))
342            .filter(|record| record.file_md5.is_some())
343            .map(|record| record.size_bytes)
344            .collect();
345        let by_digest = file_md5_index(catalog, &wanted_sizes);
346        let taken: BTreeSet<(&str, &str)> = catalog
347            .analyses
348            .iter()
349            .filter(|a| known.contains(a.path.as_str()))
350            .map(|a| (a.path.as_str(), a.source.as_str()))
351            .collect();
352
353        for (index, record) in catalog.analyses.iter().enumerate() {
354            if known.contains(record.path.as_str()) {
355                continue;
356            }
357            let candidate = if let Some(digest) = &record.file_md5 {
358                by_digest.get(&(record.size_bytes, digest.clone())).copied()
359            } else {
360                by_name_size
361                    .get(&(record.file_name(), record.size_bytes))
362                    .filter(|(_, mtime)| record.still_applies(record.size_bytes, *mtime))
363                    .copied()
364            };
365            let Some((path, mtime)) = candidate else {
366                continue;
367            };
368            // A record already filed under the real path wins: it was attached
369            // deliberately, this one is being guessed at.
370            if taken.contains(&(path, record.source.as_str())) {
371                continue;
372            }
373            moves.push((index, path.to_string(), mtime));
374        }
375    }
376    let attached = moves.len();
377    for (index, path, mtime) in moves {
378        catalog.analyses[index].path = path;
379        catalog.analyses[index].modified_unix = mtime;
380    }
381    if attached > 0 {
382        // Refiling two records onto the same path is possible when a report
383        // named the same file twice by two routes; the last one wins, as it
384        // does on import.
385        let mut seen: BTreeSet<(String, String)> = BTreeSet::new();
386        catalog.analyses.reverse();
387        catalog
388            .analyses
389            .retain(|a| seen.insert((a.path.clone(), a.source.clone())));
390        catalog.analyses.reverse();
391        sort_analyses(catalog);
392    }
393    attached
394}

396/// Files indexed by name and size, with ambiguous keys excluded.
397///
398/// If two albums contain the same name and byte count, guessing the first
399/// path would attach a measurement to the wrong music. The report can wait
400/// until an exact path is available instead.
401fn name_size_index(catalog: &Catalog) -> BTreeMap<(&str, u64), (&str, u64)> {
402    let mut index: BTreeMap<(&str, u64), (&str, u64)> = BTreeMap::new();
403    let mut ambiguous = BTreeSet::new();
404    for file in &catalog.files {
405        let key = (file.file_name(), file.size);
406        if ambiguous.contains(&key) {
407            continue;
408        }
409        if index
410            .insert(key, (file.path.as_str(), file.mtime))
411            .is_some()
412        {
413            index.remove(&key);
414            ambiguous.insert(key);
415        }
416    }
417    index
418}

420/// Find byte-identical files for reports whose old paths no longer exist.
421/// Duplicate hashes stay unresolved: identical copies still have distinct
422/// album placements, and the report cannot say which one it described.
423fn file_md5_index<'a>(
424    catalog: &'a Catalog,
425    wanted_sizes: &BTreeSet<u64>,
426) -> BTreeMap<(u64, String), (&'a str, u64)> {
427    let mut index = BTreeMap::new();
428    let mut ambiguous = BTreeSet::new();
429    for file in &catalog.files {
430        if !wanted_sizes.contains(&file.size) {
431            continue;
432        }
433        let Ok(digest) = file_md5(Path::new(&file.path)) else {
434            continue;
435        };
436        let key = (file.size, digest);
437        if ambiguous.contains(&key) {
438            continue;
439        }
440        if index
441            .insert(key.clone(), (file.path.as_str(), file.mtime))
442            .is_some()
443        {
444            index.remove(&key);
445            ambiguous.insert(key);
446        }
447    }
448    index
449}

451fn file_md5(path: &Path) -> std::io::Result<String> {
452    let mut file = std::fs::File::open(path)?;
453    let mut digest = Md5::new();
454    let mut buffer = [0u8; 64 * 1024];
455    loop {
456        let count = file.read(&mut buffer)?;
457        if count == 0 {
458            break;
459        }
460        digest.update(&buffer[..count]);
461    }
462    Ok(format!("{:x}", digest.finalize()))
463}

465/// Stores the latest dated result for one path and source. Equal or unknown
466/// dates retain the legacy import-order rule; records never accumulate.
467fn store_one(catalog: &mut Catalog, record: FileAnalysis) -> bool {
468    if record.result_at_ns != 0
469        && catalog.analyses.iter().any(|held| {
470            held.path == record.path
471                && held.source == record.source
472                && held.still_applies(record.size_bytes, record.modified_unix)
473                && held.result_at_ns > record.result_at_ns
474        })
475    {
476        return false;
477    }
478    catalog
479        .analyses
480        .retain(|a| !(a.path == record.path && a.source == record.source));
481    catalog.analyses.push(record);
482    true
483}
484
485fn sort_analyses(catalog: &mut Catalog) {
486    catalog
487        .analyses
488        .sort_by(|a, b| (&a.path, &a.source).cmp(&(&b.path, &b.source)));
489}

491/// Anything that can stop a report from being read.
492#[derive(Debug)]
493pub enum ImportError {
494    /// The file could not be opened or read.
495    Io(std::io::Error),
496    /// The text is not valid JSON.
497    Json(String),
498    /// Valid JSON, but not a report this reader knows.
499    NotAReport {
500        /// What the file claimed to be.
501        found: String,
502    },
503    /// A report of a version written after this build.
504    Version {
505        /// Version the file declares.
506        found: u32,
507    },
508}
509
510impl std::fmt::Display for ImportError {
511    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
512        match self {
513            ImportError::Io(e) => write!(f, "input/output error: {e}"),
514            ImportError::Json(e) => write!(f, "not valid JSON: {e}"),
515            ImportError::NotAReport { found } => write!(
516                f,
517                "not a FlacCompagnon report (format is \"{found}\", expected \"{FLACCOMPAGNON_FORMAT}\")"
518            ),
519            ImportError::Version { found } => write!(
520                f,
521                "report version {found} was written by a later FlacCompagnon; upgrade Aède"
522            ),
523        }
524    }
525}

527impl std::error::Error for ImportError {}

529impl From<std::io::Error> for ImportError {
530    fn from(e: std::io::Error) -> Self {
531        ImportError::Io(e)
532    }

535/// Highest report version this reader understands.
536const SUPPORTED_VERSION: u32 = 1;
537
538/// Reads a FlacCompagnon report.
539///
540/// Unknown fields are ignored rather than refused: the tool will grow new
541/// measurements, and a report carrying one must still import the rest.
542pub fn read_report(path: &Path) -> Result<Report, ImportError> {
543    let text = std::fs::read_to_string(path)?;
544    let mut report = parse_report(&text)?;
545    let result_at_ns = crate::clock::mtime_nanoseconds(&std::fs::metadata(path)?);
546    for file in &mut report.files {
547        file.result_at_ns = result_at_ns;
548    }
549    Ok(report)
550}

552/// Parse a report already held in memory, including a newly produced analysis.
553pub fn parse_report(text: &str) -> Result<Report, ImportError> {
554    let value = json::parse(text).map_err(|e| ImportError::Json(e.to_string()))?;
555
556    let format = value.field_str("format").unwrap_or_default();
557    if format != FLACCOMPAGNON_FORMAT {
558        return Err(ImportError::NotAReport { found: format });
559    }
560    let version = value.field_u32("version").unwrap_or(0);
561    if version > SUPPORTED_VERSION {
562        return Err(ImportError::Version { found: version });
563    }
564
565    let report = value.get("report");
566    let root = report.and_then(|r| r.field_str("root")).unwrap_or_default();
567    let entries = report
568        .and_then(|r| r.get("files"))
569        .and_then(|f| f.as_arr())
570        .unwrap_or(&[]);
571
572    // An entry without a path cannot be attached to anything, ever: it is
573    // dropped here rather than stored as a record about no file in particular.
574    let files = entries
575        .iter()
576        .filter(|item| item.field_str("path").is_some())
577        .map(|item| from_json(item, version))
578        .collect();
579
580    Ok(Report {
581        root,
582        version,
583        files,
584    })
585}

587/// Bytes read from a file to decide whether it is worth parsing.
588const SNIFF_BYTES: usize = 512;
589
590/// `true` when a file announces itself as a report, on a glance at its head.
591///
592/// A scan walks past every file in the library and must not parse each `.json`
593/// it meets — a cover-art sidecar, a player's own database, anything. The
594/// format marker sits in the first object of the file, so half a kilobyte
595/// settles it without reading a report that may hold thousands of entries.
596pub fn looks_like_a_report(path: &Path) -> bool {
597    use std::io::Read;
598    let Ok(mut file) = std::fs::File::open(path) else {
599        return false;
600    };
601    let mut head = [0u8; SNIFF_BYTES];
602    let Ok(read) = file.read(&mut head) else {
603        return false;
604    };
605    // Lossy on purpose: the read may well have cut a multi-byte character in
606    // half, and the marker being looked for is plain ASCII.
607    String::from_utf8_lossy(&head[..read]).contains(FLACCOMPAGNON_FORMAT)
608}

610/// Maps one entry of a report onto [`FileAnalysis`].
611fn from_json(item: &Json, version: u32) -> FileAnalysis {
612    let detections = item.get("detections");
613    let clipping = item.get("clipping");
614    let md5 = item.get("flac_md5");
615
616    FileAnalysis {
617        path: item.field_str("path").unwrap_or_default(),
618        source: "flaccompagnon".to_string(),
619        source_version: version,
620        imported_at: 0,
621        result_at_ns: 0,
622        size_bytes: item.field_u64("size_bytes").unwrap_or(0),
