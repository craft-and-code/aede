1//! A query language over the catalog and over what the user wrote about it.
2//!
3//! Options compose by AND and by nothing else. That is the ceiling no number of
4//! new flags ever raises: there is no `--genre metal OR --genre jazz`, no
5//! "everything except this label", no "between 1990 and 1999". A grammar has
6//! all three for free, and having one is what turns a saved query into a smart
7//! collection — which, since a selection is already what `--csv`, `--m3u` and
8//! M3's queue consume, is playable the day it is written.
9//!
10//! ```text
11//! genre:metal year:1990..1999 rating:>=4 -label:earache
12//! (artist:ozzy OR artist:dio) loved
13//! album.rating:5 played:0
14//! work:"War Pigs" instrument:guitar
15//! guest:"Zakk Wylde" with:"Ozzy Osbourne"
16//! ```
17//!
18//! **It is an interface, not a storage engine.** Defined on its own it works
19//! today over the vectors in memory and tomorrow over SQL. Defined as "whatever
20//! the database makes easy" it would arrive late and shaped by the wrong
21//! concerns.
22//!
23//! Everything evaluates against a **track**, because a track is the finest
24//! grain and every coarser answer is a fold of it: the albums matching a query
25//! are the albums of the tracks matching it. One evaluator, not five.
26

33/// A parsed query, ready to be run against any catalog.
34#[derive(Debug, Clone, PartialEq)]
35pub enum Query {
36    /// Matches everything, which is what an empty expression means.
37    All,
38    /// Every part must match.
39    And(Vec<Query>),
40    /// At least one part must match.
41    Or(Vec<Query>),
42    /// The part must not match.
43    Not(Box<Query>),
44    /// One condition on one field.
45    Term(Term),
46}
47
48/// One condition: a field, and what it is being asked.
49#[derive(Debug, Clone, PartialEq)]
50pub struct Term {
51    /// Which field, already resolved to something known.
52    pub field: Field,
53    /// What is being asked of it.
54    pub test: Test,
55}
56
57/// What a term asks of a value.
58#[derive(Debug, Clone, PartialEq)]
59pub enum Test {
60    /// The text contains this, compared normalized.
61    Contains(String),
62    /// The text is exactly this, compared normalized.
63    Is(String),
64    /// The number satisfies the comparison.
65    Compare(Compare, f64),
66    /// The number falls in the range, either end open.
67    Between(Option<f64>, Option<f64>),
68    /// The flag is set.
69    Set,
70    /// The flag is not set, which `lossless:false` and `loved:no` ask for.
71    Unset,
72}
73
74/// How two numbers are compared.
75#[derive(Debug, Clone, Copy, PartialEq, Eq)]
76pub enum Compare {
77    /// `=`
78    Equal,
79    /// `>`
80    Greater,
81    /// `>=`
82    AtLeast,
83    /// `<`
84    Less,
85    /// `<=`
86    AtMost,
87}

89/// Which value of a track a term reads.
90///
91/// The names are the ones typed, and the dotted ones say **where** an opinion
92/// was written: `rating` is the track's own, `album.rating` the album's,
93/// `artist.rating` the artist's. Without that distinction "rated five stars"
94/// would be a different claim depending on where the user happened to put it,
95/// and no message could say which was meant.
96#[derive(Debug, Clone, PartialEq, Eq)]
97pub enum Field {
98    /// Track title.
99    Title,
100    /// Any artist credited on the track, in any role.
101    Artist,
102    /// Album title.
103    Album,
104    /// The canonical recorded performance behind the local track.
105    Recording,
106    /// An identified composition, its explicit parent, or this file's WORK tag.
107    Work,
108    /// A movement title or number written in the local file's tags.
109    Movement,
110    /// The release group shared by this edition and its remasters.
111    ReleaseGroup,
112    /// The album's own artist.
113    AlbumArtist,
114    /// A genre carried by the track or by its album.
115    Genre,
116    /// A label the album came out on.
117    Label,
118    /// The comment tag, as the tagger wrote it.
119    Comment,
120    /// The words: from the tag that carries them, or from the `.lrc` beside
121    /// the file.
122    Lyrics,
123    /// The file's path.
124    Path,
125    /// The codec: flac, mp3, opus…
126    Codec,
127    /// Year of the album.
128    Year,
129    /// Playing time, in milliseconds.
130    Duration,
131    /// Size on disk, in bytes.
132    Size,
133    /// Bitrate in kbps.
134    Bitrate,
135    /// Sample rate in Hz.
136    SampleRate,
137    /// `true` for a lossless codec.
138    Lossless,
139    /// `true` when the album is one several artists share.
140    Compilation,
141    /// Stars given, on the entity named by the scope.
142    Rating(Scope),
143    /// A favourite, on the entity named by the scope.
144    Loved(Scope),
145    /// A favourite, wherever it was actually written: the track itself, or
146    /// failing that its album, or failing that its artist.
147    ///
148    /// What a bare `loved` asks. Unlike [`Field::Rating`], a favourite is a
149    /// blunter signal — closer to "this matters to me" than to a precise
150    /// score — so loving a whole album should not have to be repeated one
151    /// track at a time. `track.loved` still reaches the precise, track-only
152    /// question, exactly as `album.loved` and `artist.loved` already do for
153    /// theirs.
154    LovedAnywhere,
155    /// A free label, on the entity named by the scope.
156    Tag(Scope),
157    /// A note was written, and contains this text.
158    Note(Scope),
159    /// How many times it was played.
160    Played,
161    /// An artist audible on the track, in any performing role.
162    ///
163    /// The class `model::is_performing_role` draws: singing one guest verse
164    /// counts, having written the words does not. `artist --with` asks exactly
165    /// this question, and no pile of role fields ORed together would say it as
166    /// plainly.
167    Performing,
168    /// An artist credited in one named role.
169    ///
170    /// `artist:` matches any credit, in any role, which is what makes
171    /// `artist:ozzy artist:"zakk wylde"` already mean "both are on it". This
172    /// asks the finer question the graph was built for: *who did what*.
173    Credit(&'static str),
174    /// An instrument or relationship attribute on a credit.
175    Instrument,
176    /// A performing artist appearing on another artist's non-compilation release.
177    Guest,
178    /// A performing artist appearing on a compilation.
179    CompilationArtist,
180    /// An artist with a non-performing contribution such as composer or producer.
181    Contributor,
182    /// A performing artist sharing the track with at least one other performer.
183    Collaborator,
184    /// The whole track, for a bare word with no field.
185    Anything,
186}

190pub enum Scope {

192    Track,
193    /// The album it belongs to.
194    Album,

199/// Why a query could not be read.
200#[derive(Debug, Clone, PartialEq)]
201pub struct QueryError {
202    /// What went wrong, worded for the person who typed it.
203    pub message: String,
204}
205
206impl std::fmt::Display for QueryError {
207    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
208        write!(f, "{}", self.message)
209    }
210}
211
212impl std::error::Error for QueryError {}

214fn error(message: impl Into<String>) -> QueryError {
215    QueryError {
216        message: message.into(),
217    }

224/// Splits a query into its words, keeping quoted runs whole.
225///
226/// Parentheses are words of their own so that `(a OR b)` needs no spaces around
227/// them, which nobody would remember to type.
228fn tokenize(input: &str) -> Result<Vec<String>, QueryError> {
229    let mut out: Vec<String> = Vec::new();
230    let mut current = String::new();
231    let mut quote: Option<char> = None;
232    for ch in input.chars() {
233        match quote {
234            Some(q) if ch == q => quote = None,
235            Some(_) => current.push(ch),
236            None if ch == '"' || ch == '\'' => quote = Some(ch),
237            None if ch.is_whitespace() => {
238                if !current.is_empty() {
239                    out.push(std::mem::take(&mut current));
240                }
241            }
242            None if ch == '(' || ch == ')' => {
243                if !current.is_empty() {
244                    out.push(std::mem::take(&mut current));
245                }
246                out.push(ch.to_string());
247            }
248            None => current.push(ch),
249        }
250    }
251    if quote.is_some() {
252        return Err(error("a quotation mark is left open"));
253    }
254    if !current.is_empty() {
255        out.push(current);
256    }
257    Ok(out)
258}

260/// Reads a query.
261pub fn parse(input: &str) -> Result<Query, QueryError> {
262    let words = tokenize(input)?;
263    if words.is_empty() {
264        return Ok(Query::All);
265    }
266    let mut at = 0usize;
267    let query = parse_or(&words, &mut at)?;
268    if at < words.len() {
269        return Err(error(format!("\"{}\" is one bracket too many", words[at])));
270    }
271    Ok(query)
272}
273
274fn parse_or(words: &[String], at: &mut usize) -> Result<Query, QueryError> {
275    let mut parts = vec![parse_and(words, at)?];
276    while *at < words.len() && is_or(&words[*at]) {
277        *at += 1;
278        parts.push(parse_and(words, at)?);
279    }
280    Ok(if parts.len() == 1 {
281        parts.remove(0)
282    } else {
283        Query::Or(parts)
284    })
285}
286
287fn is_or(word: &str) -> bool {
288    word.eq_ignore_ascii_case("or") || word == "|" || word == "||"
289}
290
291fn parse_and(words: &[String], at: &mut usize) -> Result<Query, QueryError> {
292    let mut parts: Vec<Query> = Vec::new();
293    while *at < words.len() && words[*at] != ")" && !is_or(&words[*at]) {
294        // `AND` may be written, and is what juxtaposition already means.
295        if words[*at].eq_ignore_ascii_case("and") {
296            *at += 1;
297            continue;
298        }
299        parts.push(parse_unary(words, at)?);
300    }
301    if parts.is_empty() {
302        return Err(error("something is missing between the brackets"));
303    }
304    Ok(if parts.len() == 1 {
305        parts.remove(0)
306    } else {
307        Query::And(parts)
308    })
309}
310
311fn parse_unary(words: &[String], at: &mut usize) -> Result<Query, QueryError> {
312    let word = &words[*at];
313    if word == "-" || word.eq_ignore_ascii_case("not") {
314        *at += 1;
315        if *at >= words.len() {
316            return Err(error("nothing follows the minus sign"));
317        }
318        return Ok(Query::Not(Box::new(parse_unary(words, at)?)));
319    }
320    if let Some(rest) = word.strip_prefix('-')
321        && !rest.is_empty()
322    {
323        // `-genre:metal`, the common spelling, with no space after the sign.
324        let mut inner = vec![rest.to_string()];
325        inner.extend_from_slice(&words[*at + 1..]);
326        let mut inner_at = 0usize;
327        let negated = parse_unary(&inner, &mut inner_at)?;
328        *at += inner_at;
329        return Ok(Query::Not(Box::new(negated)));
330    }
331    if word == "(" {
332        *at += 1;
333        let inside = parse_or(words, at)?;
334        if *at >= words.len() || words[*at] != ")" {
335            return Err(error("a bracket is left open"));
336        }
337        *at += 1;
338        return Ok(inside);
339    }
340    if word == ")" {
341        return Err(error("a closing bracket has nothing to close"));
342    }
343    let term = parse_term(word)?;
344    *at += 1;
345    Ok(Query::Term(term))
346}

348fn parse_term(word: &str) -> Result<Term, QueryError> {
349    let Some((name, value)) = word.split_once(':') else {
350        // A bare word is a question when it names a field that can be asked
351        // one, and a search otherwise.
352        if let Some(field) = field_named(word)
353            && asks_whether_it_holds_anything(&field)
354        {
355            return Ok(Term {
356                field,
357                test: Test::Set,
358            });
359        }
360        return Ok(Term {
361            field: Field::Anything,
362            test: Test::Contains(word.to_string()),
363        });
364    };
365    let Some(field) = field_named(name) else {
366        return Err(error(format!(
367            "\"{name}\" is not a field.\nFields: {}",
368            FIELD_NAMES
369                .iter()
370                .map(|(n, _)| *n)
371                .collect::<Vec<_>>()
372                .join(", ")
373        )));
374    };
375    let test = parse_test(&field, value)?;
376    Ok(Term { field, test })
377}
378
379fn parse_test(field: &Field, value: &str) -> Result<Test, QueryError> {
380    if value.is_empty() {
381        return Err(error("a field needs something after the colon"));
382    }
383    // A flag asked with a word: `lossless:false` reads better than `-lossless`
384    // and means the same, so both are accepted rather than one being a trap.
385    if is_flag(field) {
386        match value.to_lowercase().as_str() {
387            "true" | "yes" | "1" => return Ok(Test::Set),
388            "false" | "no" | "0" => return Ok(Test::Unset),
389            other => {
390                return Err(error(format!(
391                    "\"{other}\" is not a yes or a no: try true or false"
392                )));
393            }
394        }
395    }
396    if is_numeric(field) {
397        // A range, either end of which may be left open: `1990..`, `..1999`.
398        if let Some((low, high)) = value.split_once("..") {
399            let low = parse_number(field, low)?;
400            let high = parse_number(field, high)?;
401            return Ok(Test::Between(low, high));
402        }
403        for (prefix, compare) in [
404            (">=", Compare::AtLeast),
405            ("<=", Compare::AtMost),
406            (">", Compare::Greater),
407            ("<", Compare::Less),
408            ("=", Compare::Equal),
409        ] {
410            if let Some(rest) = value.strip_prefix(prefix) {
411                let Some(number) = parse_number(field, rest)? else {
412                    return Err(error(format!("\"{rest}\" is not a number")));
413                };
414                return Ok(Test::Compare(compare, number));
415            }
416        }
417        let Some(number) = parse_number(field, value)? else {
418            return Err(error(format!("\"{value}\" is not a number")));
419        };
420        return Ok(Test::Compare(Compare::Equal, number));
421    }
422    if let Some(exact) = value.strip_prefix('=') {
423        return Ok(Test::Is(exact.to_string()));
424    }
425    Ok(Test::Contains(value.to_string()))
426}

428/// Reads a number, accepting `3:45` wherever a duration is expected.
429fn parse_number(field: &Field, raw: &str) -> Result<Option<f64>, QueryError> {
430    let raw = raw.trim();
431    if raw.is_empty() {
432        return Ok(None);
433    }
434    if *field == Field::Duration
435        && let Some((minutes, seconds)) = raw.split_once(':')
436    {
437        let minutes: f64 = minutes
438            .parse()
439            .map_err(|_| error(format!("\"{raw}\" is not a length")))?;
440        let seconds: f64 = seconds
441            .parse()
442            .map_err(|_| error(format!("\"{raw}\" is not a length")))?;
443        return Ok(Some((minutes * 60.0 + seconds) * 1000.0));
444    }
445    let number: f64 = raw
446        .parse()
447        .map_err(|_| error(format!("\"{raw}\" is not a number")))?;
448    // Durations are stored in milliseconds and typed in seconds.
449    Ok(Some(if *field == Field::Duration {
450        number * 1000.0
451    } else {
452        number
453    }))
454}

552/// The same question, asked of the album or the artist instead of the track.
553///
554/// Every field the *user* writes carries a scope, and a bare `rating`, `tag`
555/// or `note` means the track's own. That is deliberate — five stars on an
556/// artist is not five stars on a track — but it makes one answer badly
557/// misleading: somebody who rated an **album** types `rating`, is told
558/// nothing matches, and concludes the feature is broken. It is not; they
559/// asked a different question from the one they meant. (`loved` is the one
560/// exception: see [`Field::LovedAnywhere`], which already looks past the
561/// track, so there is nothing here for it to rescope.)
562///
563/// So a caller that gets an empty result can ask the same question again at
564/// another scope, and — if *that* answers — say which scope holds what was
565/// written. The suggestion is only ever a suggestion: the query itself keeps
566/// meaning exactly what it says.
567///
568/// Fields that carry no scope are left alone, so a mixed expression such as
569/// `genre:metal rating` is rescoped only where rescoping means something.
570pub fn rescoped(query: &Query, scope: Scope) -> Query {
571    match query {
572        Query::All => Query::All,
573        Query::And(parts) => Query::And(parts.iter().map(|p| rescoped(p, scope)).collect()),
574        Query::Or(parts) => Query::Or(parts.iter().map(|p| rescoped(p, scope)).collect()),
575        Query::Not(inner) => Query::Not(Box::new(rescoped(inner, scope))),
576        Query::Term(term) => Query::Term(Term {
577            field: match &term.field {
578                Field::Rating(_) => Field::Rating(scope),
579                Field::Loved(_) => Field::Loved(scope),
580                Field::Tag(_) => Field::Tag(scope),
581                Field::Note(_) => Field::Note(scope),
582                other => other.clone(),
583            },
584            test: term.test.clone(),
585        }),
586    }
587}

589/// `true` when the expression asks about something the user wrote, at the
590/// track's own scope — the case where [`rescoped`] has anything to offer.
591pub fn asks_about_the_track_itself(query: &Query) -> bool {
592    match query {
593        Query::All => false,
594        Query::And(parts) | Query::Or(parts) => parts.iter().any(asks_about_the_track_itself),
595        Query::Not(inner) => asks_about_the_track_itself(inner),
596        Query::Term(term) => matches!(
597            term.field,
598            Field::Rating(Scope::Track)
599                | Field::Loved(Scope::Track)
600                | Field::Tag(Scope::Track)
601                | Field::Note(Scope::Track)
602        ),
603    }
604}

606/// Fields a bare mention can ask about: "is there one at all?"
607///
608/// Wider than [`is_flag`], and the two were one predicate until that turned out
609/// to answer two different questions with one answer. `is_flag` says whether
610/// `field:true` means a yes or a no; this says whether the field's *name*,
611/// written alone, is a question. They coincide for `lossless` and `loved`, and
612/// come apart exactly where it matters:
613///
614/// - `note:vinyle` searches inside the note, so `note` is not a yes/no field;
615/// - `note` alone can only mean "the ones I have written a note on", because
616///   nobody searches a music library for the word "note".
617///
618/// Before they were separated there was **no way at all** to ask which things
619/// carried a note, a tag or a rating: a bare `note` fell through to a text
620/// search for the word, `note:true` searched for the word "true", and the
621/// fallback in [`flag_of`] that exists precisely to answer this — "any other
622/// field used as a bare flag asks whether it holds anything" — was unreachable.
623///
624/// The cost is that a bare `note`, `tag` or `rating` can no longer be a text
625/// search for those three words. Written with a field they still are:
626/// `title:note` finds the word.
627fn asks_whether_it_holds_anything(field: &Field) -> bool {
628    is_flag(field) || matches!(field, Field::Rating(_) | Field::Note(_) | Field::Tag(_))
629}

631/// Fields whose values come from a closed list the library holds.
632///
633/// Asking for a genre that exists and holds nothing, and asking for a genre
634/// nobody ever heard of, are two different questions and deserve two different
635/// answers — the same distinction `artists --role` already draws. A grammar
636/// that answered "nothing matches" to both would be a step backwards from the
637/// options it is meant to replace.
638fn closed_vocabulary(field: &Field) -> Option<&'static str> {
639    Some(match field {
640        Field::Genre => "genre",
641        Field::Label => "label",
642        Field::Credit(_)
643        | Field::Performing
644        | Field::Guest
645        | Field::CompilationArtist
646        | Field::Contributor
647        | Field::Collaborator => "artist",
648        _ => return None,
649    })
650}

652/// Values a query names that the library has never heard of.
653///
654/// Returned rather than raised, so the caller decides: a command refuses, and
655/// a saved collection listing shows the row anyway.
656pub fn unknown_values(query: &Query, context: &Context) -> Vec<(String, String)> {
657    let mut found = Vec::new();
658    collect_unknown(query, context, &mut found);
659    found
660}
661
662fn collect_unknown(query: &Query, context: &Context, out: &mut Vec<(String, String)>) {
663    match query {
664        Query::All => {}
665        Query::And(parts) | Query::Or(parts) => {
666            for part in parts {
667                collect_unknown(part, context, out);
668            }
669        }
670        Query::Not(inner) => collect_unknown(inner, context, out),
671        Query::Term(term) => {
672            let Some(what) = closed_vocabulary(&term.field) else {
673                return;
674            };
675            let wanted = match &term.test {
676                Test::Contains(value) | Test::Is(value) => text::normalize(value),
677                _ => return,
678            };
679            if wanted.is_empty() {
680                return;
681            }
682            let known = match &term.field {
683                Field::Genre => context
684                    .catalog
685                    .genres
686                    .iter()
687                    .any(|g| g.key.contains(&wanted)),
688                Field::Label => context.catalog.labels.iter().any(|l| {
689                    l.key.contains(&wanted)
690                        || l.mbid
691                            .as_deref()
692                            .is_some_and(|mbid| text::normalize(mbid).contains(&wanted))
693                        || context
694                            .sourced
695                            .label_mbids
696                            .get(&l.id)
697                            .is_some_and(|mbid| text::normalize(mbid).contains(&wanted))
698                }),
699                // A role field names a person, and a person who is in the
700                // library but never credited that way is an empty result, not
701                // a misunderstanding.
702                Field::Credit(_)
703                | Field::Performing
704                | Field::Guest
705                | Field::CompilationArtist
706                | Field::Contributor
707                | Field::Collaborator => {
708                    context.catalog.artists.iter().any(|a| {
709                        a.key.contains(&wanted)
710                            || a.mbid
711                                .as_deref()
712                                .is_some_and(|mbid| text::normalize(mbid) == wanted)
713                    }) || context
714                        .sourced
715                        .credits
716                        .values()
717                        .chain(context.sourced.edition_credits.values())
718                        .flatten()
719                        .any(|credit| {
720                            text::normalize(&credit.artist_name).contains(&wanted)
721                                || text::normalize(&credit.artist_mbid) == wanted
722                                || credit
723                                    .credited_as
724                                    .as_deref()
725                                    .is_some_and(|name| text::normalize(name).contains(&wanted))
726                        })
727                }
728                _ => true,
729            };
730            if !known {
731                out.push((what.to_string(), wanted));
732            }
733        }
734    }
735}

746/// How a result is ordered.
747///
748/// A query with no order is a query whose second page means nothing: paging is
749/// only meaningful while the order is the same on every run. Catalog order is
750/// the default because it is deterministic; everything else is asked for.
751#[derive(Debug, Clone, Copy, PartialEq, Eq)]
752pub struct Sort {
753    /// What to order on.
754    pub key: SortKey,
755    /// `true` to put the largest, latest or highest first.
756    pub descending: bool,
757}
758
759/// What a result can be ordered on.
760#[derive(Debug, Clone, Copy, PartialEq, Eq)]
761pub enum SortKey {
762    /// The order the catalog was built in, which groups an album together.
763    Catalog,
764    /// Track title.
765    Title,
766    /// Main artist, by filing name.
767    Artist,
768    /// Album title, then disc and track number.
769    Album,
770    /// Year of the album.
771    Year,
772    /// Playing time.
773    Duration,
774    /// Size on disk.
775    Size,
776    /// Stars given to the track.
777    Rating,
778    /// How many times it was played.
779    Played,
780}
781
782/// The sort keys, by the name they are typed under.
783const SORT_KEYS: &[(&str, SortKey)] = &[
784    ("catalog", SortKey::Catalog),
785    ("title", SortKey::Title),
786    ("artist", SortKey::Artist),
787    ("album", SortKey::Album),
788    ("year", SortKey::Year),
789    ("duration", SortKey::Duration),
790    ("length", SortKey::Duration),
791    ("size", SortKey::Size),
792    ("rating", SortKey::Rating),
793    ("played", SortKey::Played),
794];
795
796impl Sort {
797    /// Reads `year`, `year-` or `-year`; the sign is the direction.
798    pub fn parse(input: &str) -> Result<Sort, QueryError> {
799        let raw = input.trim();
800        let (name, descending) = match raw.strip_suffix('-').or_else(|| raw.strip_prefix('-')) {
801            Some(rest) => (rest, true),
802            None => (raw.trim_end_matches('+').trim_start_matches('+'), false),
803        };
804        let wanted = name.trim().to_lowercase();
805        let Some((_, key)) = SORT_KEYS.iter().find(|(n, _)| *n == wanted) else {
806            return Err(error(format!(
807                "\"{name}\" is not something to sort on.\nTry: {}",
808                sort_key_names().join(", ")
809            )));
810        };
811        Ok(Sort {
812            key: *key,
813            descending,
814        })
815    }
816}
817
818/// The sort keys, for a help message.
819pub fn sort_key_names() -> Vec<&'static str> {
820    SORT_KEYS.iter().map(|(name, _)| *name).collect()
821}

823/// Puts a result in order.
824///
825/// Ties fall back on catalog order, so the same query gives the same rows in
826/// the same places twice running — without which `--offset` would show a track
827/// twice and hide another.
828pub fn sort(tracks: &mut [Id], sort: Sort, context: &Context) {
829    if sort.key == SortKey::Catalog {
830        if sort.descending {
831            tracks.reverse();
832        }
833        return;
834    }
835    let position: std::collections::BTreeMap<Id, usize> = tracks
836        .iter()
837        .enumerate()
838        .map(|(at, &id)| (id, at))
839        .collect();
840    tracks.sort_by(|&a, &b| {
841        use std::cmp::Ordering;
842        // "Unknown" is not "smallest", and it is not "largest" either: a track
843        // with nothing to compare goes last **whichever way round the sort was
844        // asked**, which is why this sits outside the reversal. Sorting by year
845        // must not open with everything nobody ever tagged.
846        let order = match (missing(sort.key, context, a), missing(sort.key, context, b)) {
847            (true, true) => Ordering::Equal,
848            (true, false) => Ordering::Greater,
849            (false, true) => Ordering::Less,
850            (false, false) => {
851                let order = compare(sort.key, context, a, b);
852                if sort.descending {
853                    order.reverse()
854                } else {
855                    order
856                }
857            }
858        };
859        order.then_with(|| position.get(&a).cmp(&position.get(&b)))
860    });
861}
862
863/// `true` when a track has no value for this key, and therefore belongs at the
864/// end rather than at either extreme.
865fn missing(key: SortKey, context: &Context, track: Id) -> bool {
866    match sort_field(key) {
867        Some(field) => number_of(&field, context, track).is_none(),
868        None => false,
869    }
870}
871
872/// The field a numeric sort key reads, if it is a numeric one.
873fn sort_field(key: SortKey) -> Option<Field> {
874    Some(match key {
875        SortKey::Year => Field::Year,
876        SortKey::Duration => Field::Duration,
877        SortKey::Size => Field::Size,
878        SortKey::Rating => Field::Rating(Scope::Track),
879        SortKey::Played => Field::Played,
880        _ => return None,
881    })
882}
883
884fn compare(key: SortKey, context: &Context, a: Id, b: Id) -> std::cmp::Ordering {
885    use std::cmp::Ordering;
886    let catalog = context.catalog;
887    match key {
888        SortKey::Catalog => Ordering::Equal,
889        SortKey::Title => catalog
890            .track(a)
891            .map(|t| text::normalize(&t.title))
892            .cmp(&catalog.track(b).map(|t| text::normalize(&t.title))),
893        SortKey::Artist => sort_name(context, a).cmp(&sort_name(context, b)),
894        SortKey::Album => album_position(context, a).cmp(&album_position(context, b)),
895        SortKey::Year | SortKey::Duration | SortKey::Size | SortKey::Rating | SortKey::Played => {
896            let Some(field) = sort_field(key) else {
897                return Ordering::Equal;
898            };
899            let left = number_of(&field, context, a);
900            let right = number_of(&field, context, b);
901            match (left, right) {
902                (Some(l), Some(r)) => l.partial_cmp(&r).unwrap_or(Ordering::Equal),
903                _ => Ordering::Equal,
904            }
905        }
906    }
907}
908
909fn sort_name(context: &Context, track: Id) -> Option<String> {
910    context
911        .catalog
912        .credits_on(EntityKind::Track, track)
913        .into_iter()
914        .find(|(_, role)| *role == "main")
915        .map(|(artist, _)| artist.sort_name.clone())
916}

918/// Album title, then where the track sits in it: sorting by album and getting
919/// the tracks shuffled inside it would be half an answer.
920fn album_position(context: &Context, track: Id) -> (String, u32, u32) {
921    let catalog = context.catalog;
922    let Some(row) = catalog.track(track) else {
923        return (String::new(), 0, 0);
924    };
925    let title = row
926        .release_id
927        .and_then(|r| catalog.release(r))
928        .map(|r| text::normalize(&r.title))
929        .unwrap_or_default();
930    (title, row.disc_no.unwrap_or(1), row.track_no.unwrap_or(0))
931}

933/// Everything needed to answer, gathered once rather than per track.
934pub struct Context<'a> {
935    /// The library.
936    pub catalog: &'a Catalog,
937    /// What the user wrote about it.
938    pub data: &'a UserData,
939    /// Whose opinions count.
940    pub owner: &'a str,
941    /// Certain external relationships, indexed once for this evaluation.
942    sourced: SourcedRelations,
943}
944
945impl<'a> Context<'a> {
946    /// Creates a query context over local tags and user data.
947    pub fn new(catalog: &'a Catalog, data: &'a UserData, owner: &'a str) -> Context<'a> {
948        Context {
949            catalog,
950            data,
951            owner,
952            sourced: SourcedRelations::default(),
953        }
954    }
955
956    /// Adds externally sourced relationships that are certain enough to
957    /// traverse. Approximate matches remain evidence and never silently become
958    /// query results.
959    pub fn with_sources(mut self, sources: &crate::sources::Sources) -> Context<'a> {
960        self.sourced = SourcedRelations::new(self.catalog, sources);
961        self
962    }
963}

965#[derive(Default)]
966struct SourcedRelations {
967    works: BTreeMap<Id, Vec<crate::sources::WorkLink>>,
968    credits: BTreeMap<Id, Vec<crate::sources::CreditLink>>,
969    edition_credits: BTreeMap<Id, Vec<crate::sources::CreditLink>>,
970    label_mbids: BTreeMap<Id, String>,
971}
972
973impl SourcedRelations {
974    fn new(catalog: &Catalog, sources: &crate::sources::Sources) -> SourcedRelations {
975        let mut indexed = SourcedRelations::default();
976        for link in sources
977            .work_links(catalog)
978            .into_iter()
979            .filter(|link| link.trusted)
980        {
981            let works = indexed.works.entry(link.recording_id).or_default();
982            if !works.contains(&link.work) {
983                works.push(link.work);
984            }
985        }
986        for link in sources
987            .credit_links(catalog)
988            .into_iter()
989            .filter(|link| link.trusted)
990        {
991            let credits = indexed.credits.entry(link.recording_id).or_default();
992            if !credits.contains(&link.credit) {
993                credits.push(link.credit);
994            }
995        }
996        for link in sources
997            .edition_credit_links(catalog)
998            .into_iter()
999            .filter(|link| link.trusted)
1000        {
1001            let credits = indexed.edition_credits.entry(link.release_id).or_default();
1002            if !credits.contains(&link.credit) {
1003                credits.push(link.credit);
1004            }
1005        }
1006        for label in &catalog.labels {
1007            let Some(identity) = sources.label_identity(catalog, label.id) else {
1008                continue;
1009            };
1010            let mbid = match identity {
1011                crate::sources::LabelIdentityResolution::Confirmed { mbid, .. }
1012                | crate::sources::LabelIdentityResolution::Agrees { mbid, .. }
1013                | crate::sources::LabelIdentityResolution::Accepted { mbid, .. } => mbid,
1014                _ => continue,
1015            };
1016            indexed.label_mbids.insert(label.id, mbid);
1017        }
1018        indexed
1019    }
1020}

1022/// The tracks a query matches, in catalog order.
1023///
1024/// Order matters because everything downstream pages through it, and paging is
1025/// only meaningful while the order is the same on every run.
1026pub fn run(query: &Query, context: &Context) -> Vec<Id> {
1027    context
1028        .catalog
1029        .tracks
1030        .iter()
1031        .filter(|track| matches(query, context, track.id))
1032        .map(|track| track.id)
1033        .collect()
1034}
1035
1036/// Whether one track satisfies a query.
1037pub fn matches(query: &Query, context: &Context, track: Id) -> bool {
1038    match query {
1039        Query::All => true,
1040        Query::And(parts) => parts.iter().all(|p| matches(p, context, track)),
1041        Query::Or(parts) => parts.iter().any(|p| matches(p, context, track)),
1042        Query::Not(inner) => !matches(inner, context, track),
1043        Query::Term(term) => term_matches(term, context, track),
1044    }
1045}
1046
1047fn term_matches(term: &Term, context: &Context, track: Id) -> bool {
1048    match &term.test {
1049        Test::Set => flag_of(&term.field, context, track),
1050        Test::Unset => !flag_of(&term.field, context, track),
1051        Test::Compare(compare, wanted) => match number_of(&term.field, context, track) {
1052            // A track with no year cannot satisfy a question about years. It is
1053            // absent from the answer rather than counted as zero, which would
1054            // put every untagged file in "before 1970".
1055            None => false,
1056            Some(value) => match compare {
1057                Compare::Equal => (value - wanted).abs() < f64::EPSILON,
1058                Compare::Greater => value > *wanted,
1059                Compare::AtLeast => value >= *wanted,
1060                Compare::Less => value < *wanted,
1061                Compare::AtMost => value <= *wanted,
1062            },
1063        },
1064        Test::Between(low, high) => match number_of(&term.field, context, track) {
1065            None => false,
1066            Some(value) => {
1067                low.map(|l| value >= l).unwrap_or(true) && high.map(|h| value <= h).unwrap_or(true)
1068            }
1069        },
1070        Test::Contains(wanted) => {
1071            let wanted = text::normalize(wanted);
1072            texts_of(&term.field, context, track)
1073                .iter()
1074                .any(|value| text::normalize(value).contains(&wanted))
1075        }
1076        Test::Is(wanted) => {
1077            let wanted = text::normalize(wanted);
1078            texts_of(&term.field, context, track)
1079                .iter()
1080                .any(|value| text::normalize(value) == wanted)
1081        }
1082    }
1083}

1085/// The reference for the entity a scope names, starting from a track.
1086fn scoped(scope: Scope, context: &Context, track: Id) -> Option<EntityRef> {
1087    let catalog = context.catalog;
1088    match scope {
1089        Scope::Track => EntityRef::of(catalog, EntityKind::Track, track),
1090        Scope::Album => catalog
1091            .track(track)
1092            .and_then(|t| t.release_id)
1093            .and_then(|r| EntityRef::of(catalog, EntityKind::Release, r)),
1094        Scope::Artist => catalog
1095            .credits_on(EntityKind::Track, track)
1096            .into_iter()
1097            .find(|(_, role)| *role == "main")
1098            .and_then(|(artist, _)| EntityRef::of(catalog, EntityKind::Artist, artist.id)),
1099    }
1100}

1111fn flag_of(field: &Field, context: &Context, track: Id) -> bool {
1112    match field {
1113        Field::Lossless => context
1114            .catalog
1115            .track(track)
1116            .and_then(|t| context.catalog.file(t.file_id))
1117            .map(|f| f.properties.lossless)
1118            .unwrap_or(false),
1119        Field::Compilation => context
1120            .catalog
1121            .track(track)
1122            .and_then(|t| t.release_id)
1123            .and_then(|r| context.catalog.release(r))
1124            .map(|r| r.is_compilation)
1125            .unwrap_or(false),
1126        Field::Loved(scope) => annotation(*scope, context, track)
1127            .map(|a| a.loved)
1128            .unwrap_or(false),
1129        Field::LovedAnywhere => [Scope::Track, Scope::Album, Scope::Artist]
1130            .into_iter()
1131            .any(|scope| annotation(scope, context, track).is_some_and(|a| a.loved)),
1132        // Any other field used as a bare flag asks whether it holds anything.
1133        _ => {
1134            !texts_of(field, context, track).is_empty()
1135                || number_of(field, context, track).is_some()
1136        }
1137    }
1138}
1139
1140fn number_of(field: &Field, context: &Context, track: Id) -> Option<f64> {
1141    let catalog = context.catalog;
1142    let track_row = catalog.track(track)?;
1143    let file = catalog.file(track_row.file_id);
1144    match field {
1145        Field::Year => track_row
1146            .release_id
1147            .and_then(|r| catalog.release(r))
1148            .and_then(|r| r.year)
1149            .map(f64::from),
1150        Field::Duration => track_row.duration_ms.map(|d| d as f64),
1151        Field::Size => file.map(|f| f.size as f64),
1152        Field::Bitrate => file.and_then(|f| f.properties.bitrate_kbps).map(f64::from),
1153        Field::SampleRate => file.and_then(|f| f.properties.sample_rate).map(f64::from),
1154        Field::Played => {
1155            let reference = scoped(Scope::Track, context, track)?;
1156            Some(f64::from(
1157                context.data.play_count(context.owner, &reference),
1158            ))
1159        }
1160        Field::Rating(scope) => annotation(*scope, context, track)
1161            .and_then(|a| a.rating)
1162            .map(f64::from),
1163        _ => None,
1164    }
1165}

1167fn texts_of(field: &Field, context: &Context, track: Id) -> Vec<String> {
1168    let catalog = context.catalog;
1169    let Some(track_row) = catalog.track(track) else {
1170        return Vec::new();
1171    };
1172    let file = catalog.file(track_row.file_id);
1173    let release = track_row.release_id.and_then(|r| catalog.release(r));
1174    match field {
1175        Field::Title => vec![track_row.title.clone()],
1176        Field::Artist => {
1177            let mut values = Vec::new();
1178            for credit in catalog
1179                .credits
1180                .iter()
1181                .filter(|c| c.entity_kind == EntityKind::Track && c.entity_id == track)
1182            {
1183                extend_artist_values(catalog, credit.artist_id, &mut values);
1184                if let Some(credited_as) = &credit.credited_as {
1185                    push_unique(&mut values, credited_as.clone());
1186                }
1187            }
1188            if let Some(release_id) = track_row.release_id {
1189                for credit in catalog.credits.iter().filter(|credit| {
1190                    credit.entity_kind == EntityKind::Release && credit.entity_id == release_id
1191                }) {
1192                    extend_artist_values(catalog, credit.artist_id, &mut values);
1193                }
1194            }
1195            for credit in sourced_credits(context, track) {
1196                extend_sourced_credit_values(credit, &mut values);
1197            }
1198            values
1199        }
1200        Field::Album => release
1201            .map(|r| {
1202                let mut values = vec![r.title.clone()];
1203                if let Some(mbid) = &r.mbid {
1204                    values.push(mbid.clone());
1205                }
1206                values
1207            })
1208            .unwrap_or_default(),
1209        Field::Recording => catalog
1210            .recording(track_row.recording_id)
1211            .map(|recording| {
1212                let mut values = vec![recording.title.clone()];
1213                if let Some(mbid) = &recording.mbid {
1214                    values.push(mbid.clone());
1215                }
1216                values
1217            })
1218            .unwrap_or_default(),
1219        Field::Work => {
1220            let mut values = catalog
1221                .recording(track_row.recording_id)
1222                .map(|recording| {
1223                    recording
1224                        .work_ids
1225                        .iter()
1226                        .filter_map(|&id| catalog.work(id))
1227                        .flat_map(|work| vec![work.title.clone(), work.mbid.clone()])
1228                        .collect()
1229                })
1230                .unwrap_or_default();
1231            // A local WORK/grouping tag is searchable even without an MBID.
1232            // This is text on one file, not a new canonical work identity.
1233            if let Some(work_tag) = catalog
1234                .file(track_row.file_id)
1235                .and_then(|file| file.first_tag("grouping"))
1236            {
1237                push_unique(&mut values, work_tag.to_string());
1238            }
1239            if let Some(works) = context.sourced.works.get(&track_row.recording_id) {
1240                for work in works {
1241                    push_unique(&mut values, work.title.clone());
1242                    push_unique(&mut values, work.mbid.clone());
1243                    for parent in &work.parents {
1244                        push_unique(&mut values, parent.title.clone());
1245                        push_unique(&mut values, parent.mbid.clone());
1246                    }
1247                }
1248            }
1249            values
1250        }
1251        Field::Movement => catalog
1252            .file(track_row.file_id)
1253            .map(|file| {
1254                ["movement", "movementnumber"]
1255                    .into_iter()
1256                    .filter_map(|tag| file.first_tag(tag).map(str::to_string))
1257                    .collect()
1258            })
1259            .unwrap_or_default(),
1260        Field::ReleaseGroup => release
1261            .and_then(|r| r.release_group_id)
1262            .and_then(|id| catalog.release_group(id))
1263            .map(|group| vec![group.title.clone(), group.mbid.clone()])
1264            .unwrap_or_default(),
1265        Field::AlbumArtist => release
1266            .and_then(|r| r.album_artist_id)
1267            .map(|a| artist_values(catalog, a))
1268            .unwrap_or_default(),
1269        Field::Genre => {
1270            let mut names: Vec<String> = catalog
1271                .genres_of(EntityKind::Track, track)
1272                .into_iter()
1273                .map(|g| g.name.clone())
1274                .collect();
1275            if let Some(release) = release {
1276                names.extend(
1277                    catalog
1278                        .genres_of(EntityKind::Release, release.id)
1279                        .into_iter()
1280                        .map(|g| g.name.clone()),
1281                );
1282            }
1283            names
1284        }
1285        Field::Label => release
1286            .map(|r| {
1287                r.label_ids.iter().fold(Vec::new(), |mut values, &id| {
1288                    if let Some(label) = catalog.label(id) {
1289                        values.push(label.name.clone());
1290                        if let Some(mbid) = &label.mbid {
1291                            values.push(mbid.clone());
1292                        }
1293                        if let Some(mbid) = context.sourced.label_mbids.get(&id) {
1294                            push_unique(&mut values, mbid.clone());
1295                        }
1296                    }
1297                    values
1298                })
1299            })
1300            .unwrap_or_default(),
1301        Field::Comment => file
1302            .and_then(|f| f.tags.get("comment"))
1303            .cloned()
1304            .unwrap_or_default(),
1305        // The tag first, because it costs nothing — raw tags are in the
1306        // catalog. The sidecar is only opened when the tag holds nothing, and
1307        // only for the tracks that have one, which is what keeps a search
1308        // across a library from being ten thousand file reads.
1309        Field::Lyrics => catalog
1310            .lyrics_of_track(track)
1311            .map(|lyrics| vec![lyrics.text()])
1312            .unwrap_or_default(),
1313        Field::Path => file.map(|f| vec![f.path.clone()]).unwrap_or_default(),
1314        Field::Codec => file
1315            .map(|f| {
1316                vec![
1317                    f.properties.codec.clone(),
1318                    f.properties.container.clone(),
1319                    f.properties.quality_label(),
1320                ]
1321            })
1322            .unwrap_or_default(),
1323        Field::Performing => credits_as_text(context, track, |role, _| {
1324            crate::model::is_performing_role(role)
1325        }),
1326        Field::Credit(role) => credits_as_text(context, track, |credited, attributes| {
1327            credited == *role
1328                || (*role == "orchestra" && credited == "performing orchestra")
1329                || (*role == "soloist"
1330                    && matches!(credited, "performer" | "instrument" | "vocal")
1331                    && attributes.iter().any(|attribute| attribute.name == "solo"))
1332        }),
1333        Field::Instrument => {
1334            let mut values = Vec::new();
1335            for credit in catalog
1336                .credits
1337                .iter()
1338                .filter(|c| c.entity_kind == EntityKind::Track && c.entity_id == track)
1339            {
1340                for attribute in &credit.attributes {
1341                    push_unique(&mut values, attribute.name.clone());
1342                    if let Some(value) = &attribute.value {
1343                        push_unique(&mut values, value.clone());
1344                    }
1345                    if let Some(credited_as) = &attribute.credited_as {
1346                        push_unique(&mut values, credited_as.clone());
1347                    }
1348                }
1349            }
1350            for credit in sourced_credits(context, track) {
1351                for attribute in &credit.attributes {
1352                    push_unique(&mut values, attribute.name.clone());
1353                    if let Some(value) = &attribute.value {
1354                        push_unique(&mut values, value.clone());
1355                    }
1356                    if let Some(credited_as) = &attribute.credited_as {
1357                        push_unique(&mut values, credited_as.clone());
1358                    }
1359                }
1360            }
1361            values
1362        }
1363        Field::Guest | Field::CompilationArtist | Field::Contributor | Field::Collaborator => {
1364            participation_values(field, context, track, release)
1365        }
1366        Field::Tag(scope) => annotation(*scope, context, track)
1367            .map(|a| a.tags.iter().cloned().collect())
1368            .unwrap_or_default(),
1369        Field::Note(scope) => annotation(*scope, context, track)
1370            .and_then(|a| a.note.clone())
1371            .map(|n| vec![n])
1372            .unwrap_or_default(),
1373        // A bare word searches where a person would expect it to.
1374        Field::Anything => {
1375            let mut all = vec![track_row.title.clone()];
1376            all.extend(texts_of(&Field::Artist, context, track));
1377            all.extend(texts_of(&Field::Album, context, track));
1378            all
1379        }
1380        _ => Vec::new(),
1381    }
1382}

1384fn push_unique(values: &mut Vec<String>, value: String) {
1385    if !values.iter().any(|existing| existing == &value) {
1386        values.push(value);
1387    }
1388}

1390fn artist_values(catalog: &Catalog, artist_id: Id) -> Vec<String> {
1391    let Some(artist) = catalog.artist(artist_id) else {
1392        return Vec::new();
1393    };
1394    let mut values = vec![artist.name.clone()];
1395    if let Some(mbid) = &artist.mbid {
1396        values.push(mbid.clone());
1397    }
1398    values.extend(artist.aliases.iter().cloned());
1399    values
1400}
1401
1402fn extend_artist_values(catalog: &Catalog, artist_id: Id, values: &mut Vec<String>) {
1403    for value in artist_values(catalog, artist_id) {
1404        push_unique(values, value);
1405    }
1406}

1408fn sourced_credits<'a>(
1409    context: &'a Context<'_>,
1410    track_id: Id,
1411) -> Vec<&'a crate::sources::CreditLink> {
1412    let Some(track) = context.catalog.track(track_id) else {
1413        return Vec::new();
1414    };
1415    context
1416        .sourced
1417        .credits
1418        .get(&track.recording_id)
1419        .into_iter()
1420        .flat_map(|credits| credits.iter())
1421        .chain(
1422            track
1423                .release_id
1424                .and_then(|id| context.sourced.edition_credits.get(&id))
1425                .into_iter()
1426                .flat_map(|credits| credits.iter()),
1427        )
1428        .collect()
1429}

1431fn extend_sourced_credit_values(credit: &crate::sources::CreditLink, values: &mut Vec<String>) {
1432    push_unique(values, credit.artist_name.clone());
1433    push_unique(values, credit.artist_mbid.clone());
1434    if let Some(credited_as) = &credit.credited_as {
1435        push_unique(values, credited_as.clone());
1436    }
1437}

1439fn credits_as_text(
1440    context: &Context,
1441    track: Id,
1442    wanted: impl Fn(&str, &[crate::model::CreditAttribute]) -> bool,
1443) -> Vec<String> {
1444    let catalog = context.catalog;
1445    let mut values = Vec::new();
1446    for credit in catalog
1447        .credits
1448        .iter()
1449        .filter(|credit| {
1450            (credit.entity_kind == EntityKind::Track && credit.entity_id == track)
1451                || (credit.entity_kind == EntityKind::Release
1452                    && catalog.track(track).and_then(|row| row.release_id)
1453                        == Some(credit.entity_id))
1454        })
1455        .filter(|credit| wanted(&credit.role, &credit.attributes))
1456    {
1457        extend_artist_values(catalog, credit.artist_id, &mut values);
1458        if let Some(credited_as) = &credit.credited_as {
1459            push_unique(&mut values, credited_as.clone());
1460        }
1461    }
1462    for credit in sourced_credits(context, track)
1463        .into_iter()
1464        .filter(|credit| wanted(&credit.role, &credit.attributes))
1465    {
1466        extend_sourced_credit_values(credit, &mut values);
1467    }
1468    values
1469}

1471fn participation_values(
1472    field: &Field,
1473    context: &Context,
1474    track: Id,
1475    release: Option<&crate::model::Release>,
1476) -> Vec<String> {
1477    let catalog = context.catalog;
1478    if catalog.track(track).is_none() {
1479        return Vec::new();
1480    }
1481    let performing: Vec<&crate::model::Credit> = catalog
1482        .credits
1483        .iter()
1484        .filter(|credit| {
1485            credit.entity_kind == EntityKind::Track
1486                && credit.entity_id == track
1487                && crate::model::is_performing_role(&credit.role)
1488        })
1489        .collect();
1490    let sourced_performing: Vec<&crate::sources::CreditLink> = sourced_credits(context, track)
1491        .into_iter()
1492        .filter(|credit| crate::model::is_performing_role(&credit.role))
1493        .collect();
1494    let mut distinct_performers: BTreeSet<String> = performing
1495        .iter()
1496        .map(|credit| local_artist_key(catalog, credit.artist_id))
1497        .collect();
1498    distinct_performers.extend(
1499        sourced_performing
1500            .iter()
1501            .map(|credit| sourced_artist_key(catalog, credit)),
1502    );
1503    if matches!(field, Field::Contributor) {
1504        let mut values = Vec::new();
1505        for credit in catalog.credits.iter().filter(|credit| {
1506            ((credit.entity_kind == EntityKind::Track && credit.entity_id == track)
1507                || (credit.entity_kind == EntityKind::Release
1508                    && release.is_some_and(|release| release.id == credit.entity_id)))
1509                && !crate::model::is_performing_role(&credit.role)
1510        }) {
1511            extend_artist_values(catalog, credit.artist_id, &mut values);
1512        }
1513        for credit in sourced_credits(context, track)
1514            .into_iter()
1515            .filter(|credit| !crate::model::is_performing_role(&credit.role))
1516        {
1517            extend_sourced_credit_values(credit, &mut values);
1518        }
1519        return values;
1520    }
1521    let Some(release) = release else {
1522        return if matches!(field, Field::Collaborator) && distinct_performers.len() > 1 {
1523            performer_values(catalog, &performing, &sourced_performing)
1524        } else {
1525            Vec::new()
1526        };
1527    };
1528    let mut values = Vec::new();
1529    match field {
1530        Field::Guest => {
1531            if !release.is_compilation {
1532                for credit in performing {
1533                    if release.album_artist_id != Some(credit.artist_id) {
1534                        extend_artist_values(catalog, credit.artist_id, &mut values);
1535                    }
1536                }
1537                for credit in sourced_performing {
1538                    if !is_album_artist(catalog, release, credit) {
1539                        extend_sourced_credit_values(credit, &mut values);
1540                    }
1541                }
1542            }
1543        }
1544        Field::CompilationArtist => {
1545            if release.is_compilation {
1546                for credit in performing {
1547                    if release.album_artist_id != Some(credit.artist_id) {
1548                        extend_artist_values(catalog, credit.artist_id, &mut values);
1549                    }
1550                }
1551                for credit in sourced_performing {
1552                    if !is_album_artist(catalog, release, credit) {
1553                        extend_sourced_credit_values(credit, &mut values);
1554                    }
1555                }
1556            }
1557        }
1558        Field::Contributor => unreachable!("contributors are handled before release lookup"),
1559        Field::Collaborator if distinct_performers.len() > 1 => {
1560            return performer_values(catalog, &performing, &sourced_performing);
1561        }
1562        _ => {}
1563    }
1564    values
1565}

1567fn local_artist_key(catalog: &Catalog, artist_id: Id) -> String {
1568    catalog
1569        .artist(artist_id)
1570        .map(|artist| {
1571            artist
1572                .mbid
1573                .clone()
1574                .unwrap_or_else(|| format!("name:{}", artist.key))
1575        })
1576        .unwrap_or_else(|| format!("local:{artist_id}"))
1577}
1578
1579fn sourced_artist_key(catalog: &Catalog, credit: &crate::sources::CreditLink) -> String {
1580    let credited_key = credit.credited_as.as_deref().map(text::normalize);
1581    if let Some(artist) = catalog.artists.iter().find(|artist| {
1582        artist.mbid.as_deref() == Some(credit.artist_mbid.as_str())
1583            || artist.key == text::normalize(&credit.artist_name)
1584            || credited_key.as_deref() == Some(artist.key.as_str())
1585    }) {
1586        return local_artist_key(catalog, artist.id);
1587    }
1588    if credit.artist_mbid.trim().is_empty() {
1589        format!("name:{}", text::normalize(&credit.artist_name))
1590    } else {
1591        credit.artist_mbid.clone()
1592    }
1593}

1595fn is_album_artist(
1596    catalog: &Catalog,
1597    release: &crate::model::Release,
1598    credit: &crate::sources::CreditLink,
1599) -> bool {
1600    let Some(artist) = release.album_artist_id.and_then(|id| catalog.artist(id)) else {
1601        return false;
1602    };
1603    artist.mbid.as_deref() == Some(credit.artist_mbid.as_str())
1604        || artist.key == text::normalize(&credit.artist_name)
1605        || credit
1606            .credited_as
1607            .as_deref()
1608            .is_some_and(|name| artist.key == text::normalize(name))
1609}

1611fn performer_values(
1612    catalog: &Catalog,
1613    local: &[&crate::model::Credit],
1614    sourced: &[&crate::sources::CreditLink],
1615) -> Vec<String> {
1616    let mut values = Vec::new();
1617    let mut seen = BTreeSet::new();
1618    for credit in local {
1619        if seen.insert(credit.artist_id) {
1620            extend_artist_values(catalog, credit.artist_id, &mut values);
1621        }
1622    }
1623    for credit in sourced {
1624        extend_sourced_credit_values(credit, &mut values);
1625    }
1626    values
1627}
