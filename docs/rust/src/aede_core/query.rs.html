1//! A query language over the catalog and over what the user wrote about it.
2//!
3//! Options compose by AND and by nothing else. That is the ceiling no number of
4//! new flags ever raises: there is no `--genre metal OR --genre jazz`, no
5//! "everything except this label", no "between 1990 and 1999". A grammar has
6//! all three for free, and having one is what turns a saved query into a smart
7//! collection — which, since a selection is already what `--csv`, `--m3u` and
8//! M3's queue consume, is playable the day it is written.
9//!
10//! ```text
11//! genre:metal year:1990..1999 rating:>=4 -label:earache
12//! (artist:ozzy OR artist:dio) loved
13//! album.rating:5 played:0
14//! work:"War Pigs" instrument:guitar
15//! guest:"Zakk Wylde" with:"Ozzy Osbourne"
16//! ```
17//!
18//! **It is an interface, not a storage engine.** Defined on its own it works
19//! today over the vectors in memory and tomorrow over SQL. Defined as "whatever
20//! the database makes easy" it would arrive late and shaped by the wrong
21//! concerns.
22//!
23//! Everything evaluates against a **track**, because a track is the finest
24//! grain and every coarser answer is a fold of it: the albums matching a query
25//! are the albums of the tracks matching it. One evaluator, not five.
26

33/// A parsed query, ready to be run against any catalog.
34#[derive(Debug, Clone, PartialEq)]
35pub enum Query {
36    /// Matches everything, which is what an empty expression means.
37    All,
38    /// Every part must match.
39    And(Vec<Query>),
40    /// At least one part must match.
41    Or(Vec<Query>),
42    /// The part must not match.
43    Not(Box<Query>),
44    /// One condition on one field.
45    Term(Term),
46}
47
48/// One condition: a field, and what it is being asked.
49#[derive(Debug, Clone, PartialEq)]
50pub struct Term {
51    /// Which field, already resolved to something known.
52    pub field: Field,
53    /// What is being asked of it.
54    pub test: Test,
55}
56
57/// What a term asks of a value.
58#[derive(Debug, Clone, PartialEq)]
59pub enum Test {
60    /// The text contains this, compared normalized.
61    Contains(String),
62    /// The text is exactly this, compared normalized.
63    Is(String),
64    /// The number satisfies the comparison.
65    Compare(Compare, f64),
66    /// The number falls in the range, either end open.
67    Between(Option<f64>, Option<f64>),
68    /// The flag is set.
69    Set,
70    /// The flag is not set, which `lossless:false` and `loved:no` ask for.
71    Unset,
72}
73
74/// How two numbers are compared.
75#[derive(Debug, Clone, Copy, PartialEq, Eq)]
76pub enum Compare {
77    /// `=`
78    Equal,
79    /// `>`
80    Greater,
81    /// `>=`
82    AtLeast,
83    /// `<`
84    Less,
85    /// `<=`
86    AtMost,
87}

89/// Which value of a track a term reads.
90///
91/// The names are the ones typed, and the dotted ones say **where** an opinion
92/// was written: `rating` is the track's own, `album.rating` the album's,
93/// `artist.rating` the artist's. Without that distinction "rated five stars"
94/// would be a different claim depending on where the user happened to put it,
95/// and no message could say which was meant.
96#[derive(Debug, Clone, PartialEq, Eq)]
97pub enum Field {
98    /// Track title.
99    Title,
100    /// Any artist credited on the track, in any role.
101    Artist,
102    /// Album title.
103    Album,
104    /// The canonical recorded performance behind the local track.
105    Recording,
106    /// An identified composition, its explicit parent, or this file's WORK tag.
107    Work,
108    /// A movement title or number written in the local file's tags.
109    Movement,
110    /// The release group shared by this edition and its remasters.
111    ReleaseGroup,
112    /// The album's own artist.
113    AlbumArtist,
114    /// A genre carried by the track or by its album.
115    Genre,
116    /// A label the album came out on.
117    Label,
118    /// The comment tag, as the tagger wrote it.
119    Comment,
120    /// The words: from the tag that carries them, or from the `.lrc` beside
121    /// the file.
122    Lyrics,
123    /// The file's path.
124    Path,
125    /// The codec: flac, mp3, opus…
126    Codec,
127    /// Year of the album.
128    Year,
129    /// Playing time, in milliseconds.
130    Duration,
131    /// Size on disk, in bytes.
132    Size,
133    /// Bitrate in kbps.
134    Bitrate,
135    /// Sample rate in Hz.
136    SampleRate,
137    /// `true` for a lossless codec.
138    Lossless,
139    /// `true` when the album is one several artists share.
140    Compilation,
141    /// Stars given, on the entity named by the scope.
142    Rating(Scope),
143    /// A favourite, on the entity named by the scope.
144    Loved(Scope),
145    /// A favourite, wherever it was actually written: the track itself, or
146    /// failing that its album, or failing that its artist.
147    ///
148    /// What a bare `loved` asks. Unlike [`Field::Rating`], a favourite is a
149    /// blunter signal — closer to "this matters to me" than to a precise
150    /// score — so loving a whole album should not have to be repeated one
151    /// track at a time. `track.loved` still reaches the precise, track-only
152    /// question, exactly as `album.loved` and `artist.loved` already do for
153    /// theirs.
154    LovedAnywhere,
155    /// A free label, on the entity named by the scope.
156    Tag(Scope),
157    /// A note was written, and contains this text.
158    Note(Scope),
159    /// How many times it was played.
160    Played,
161    /// An artist audible on the track, in any performing role.
162    ///
163    /// The class `model::is_performing_role` draws: singing one guest verse
164    /// counts, having written the words does not. `artist --with` asks exactly
165    /// this question, and no pile of role fields ORed together would say it as
166    /// plainly.
167    Performing,
168    /// An artist credited in one named role.
169    ///
170    /// `artist:` matches any credit, in any role, which is what makes
171    /// `artist:ozzy artist:"zakk wylde"` already mean "both are on it". This
172    /// asks the finer question the graph was built for: *who did what*.
173    Credit(&'static str),
174    /// An instrument or relationship attribute on a credit.
175    Instrument,
176    /// A performing artist appearing on another artist's non-compilation release.
177    Guest,
178    /// A performing artist appearing on a compilation.
179    CompilationArtist,
180    /// An artist with a non-performing contribution such as composer or producer.
181    Contributor,
182    /// A performing artist sharing the track with at least one other performer.
183    Collaborator,
184    /// The whole track, for a bare word with no field.
185    Anything,
186}
187
188/// Which entity an opinion was written on.
189#[derive(Debug, Clone, Copy, PartialEq, Eq)]
190pub enum Scope {
191    /// The track itself.
192    Track,
193    /// The album it belongs to.
194    Album,
195    /// Its main artist.
196    Artist,
197}

319/// The same question, asked of the album or the artist instead of the track.
320///
321/// Every field the *user* writes carries a scope, and a bare `rating`, `tag`
322/// or `note` means the track's own. That is deliberate — five stars on an
323/// artist is not five stars on a track — but it makes one answer badly
324/// misleading: somebody who rated an **album** types `rating`, is told
325/// nothing matches, and concludes the feature is broken. It is not; they
326/// asked a different question from the one they meant. (`loved` is the one
327/// exception: see [`Field::LovedAnywhere`], which already looks past the
328/// track, so there is nothing here for it to rescope.)
329///
330/// So a caller that gets an empty result can ask the same question again at
331/// another scope, and — if *that* answers — say which scope holds what was
332/// written. The suggestion is only ever a suggestion: the query itself keeps
333/// meaning exactly what it says.
334///
335/// Fields that carry no scope are left alone, so a mixed expression such as
336/// `genre:metal rating` is rescoped only where rescoping means something.
337pub fn rescoped(query: &Query, scope: Scope) -> Query {
338    match query {
339        Query::All => Query::All,
340        Query::And(parts) => Query::And(parts.iter().map(|p| rescoped(p, scope)).collect()),
341        Query::Or(parts) => Query::Or(parts.iter().map(|p| rescoped(p, scope)).collect()),
342        Query::Not(inner) => Query::Not(Box::new(rescoped(inner, scope))),
343        Query::Term(term) => Query::Term(Term {
344            field: match &term.field {
345                Field::Rating(_) => Field::Rating(scope),
346                Field::Loved(_) => Field::Loved(scope),
347                Field::Tag(_) => Field::Tag(scope),
348                Field::Note(_) => Field::Note(scope),
349                other => other.clone(),
350            },
351            test: term.test.clone(),
352        }),
353    }
354}

356/// `true` when the expression asks about something the user wrote, at the
357/// track's own scope — the case where [`rescoped`] has anything to offer.
358pub fn asks_about_the_track_itself(query: &Query) -> bool {
359    match query {
360        Query::All => false,
361        Query::And(parts) | Query::Or(parts) => parts.iter().any(asks_about_the_track_itself),
362        Query::Not(inner) => asks_about_the_track_itself(inner),
363        Query::Term(term) => matches!(
364            term.field,
365            Field::Rating(Scope::Track)
366                | Field::Loved(Scope::Track)
367                | Field::Tag(Scope::Track)
368                | Field::Note(Scope::Track)
369        ),
370    }
371}

373/// Fields a bare mention can ask about: "is there one at all?"
374///
375/// Wider than [`is_flag`], and the two were one predicate until that turned out
376/// to answer two different questions with one answer. `is_flag` says whether
377/// `field:true` means a yes or a no; this says whether the field's *name*,
378/// written alone, is a question. They coincide for `lossless` and `loved`, and
379/// come apart exactly where it matters:
380///
381/// - `note:vinyle` searches inside the note, so `note` is not a yes/no field;
382/// - `note` alone can only mean "the ones I have written a note on", because
383///   nobody searches a music library for the word "note".
384///
385/// Before they were separated there was **no way at all** to ask which things
386/// carried a note, a tag or a rating: a bare `note` fell through to a text
387/// search for the word, `note:true` searched for the word "true", and the
388/// fallback in [`flag_of`] that exists precisely to answer this — "any other
389/// field used as a bare flag asks whether it holds anything" — was unreachable.
390///
391/// The cost is that a bare `note`, `tag` or `rating` can no longer be a text
392/// search for those three words. Written with a field they still are:
393/// `title:note` finds the word.
394fn asks_whether_it_holds_anything(field: &Field) -> bool {
395    is_flag(field) || matches!(field, Field::Rating(_) | Field::Note(_) | Field::Tag(_))
396}

398/// Fields whose values come from a closed list the library holds.
399///
400/// Asking for a genre that exists and holds nothing, and asking for a genre
401/// nobody ever heard of, are two different questions and deserve two different
402/// answers — the same distinction `artists --role` already draws. A grammar
403/// that answered "nothing matches" to both would be a step backwards from the
404/// options it is meant to replace.
405fn closed_vocabulary(field: &Field) -> Option<&'static str> {
406    Some(match field {
407        Field::Genre => "genre",
408        Field::Label => "label",
409        Field::Credit(_)
410        | Field::Performing
411        | Field::Guest
412        | Field::CompilationArtist
413        | Field::Contributor
414        | Field::Collaborator => "artist",
415        _ => return None,
416    })
417}

419/// Values a query names that the library has never heard of.
420///
421/// Returned rather than raised, so the caller decides: a command refuses, and
422/// a saved collection listing shows the row anyway.
423pub fn unknown_values(query: &Query, context: &Context) -> Vec<(String, String)> {
424    let mut found = Vec::new();
425    collect_unknown(query, context, &mut found);
426    found
427}

429fn collect_unknown(query: &Query, context: &Context, out: &mut Vec<(String, String)>) {
430    match query {
431        Query::All => {}
432        Query::And(parts) | Query::Or(parts) => {
433            for part in parts {
434                collect_unknown(part, context, out);
435            }
436        }
437        Query::Not(inner) => collect_unknown(inner, context, out),
438        Query::Term(term) => {
439            let Some(what) = closed_vocabulary(&term.field) else {
440                return;
441            };
442            let wanted = match &term.test {
443                Test::Contains(value) | Test::Is(value) => text::normalize(value),
444                _ => return,
445            };
446            if wanted.is_empty() {
447                return;
448            }
449            let known = match &term.field {
450                Field::Genre => context
451                    .catalog
452                    .genres
453                    .iter()
454                    .any(|g| g.key.contains(&wanted)),
455                Field::Label => context.catalog.labels.iter().any(|l| {
456                    l.key.contains(&wanted)
457                        || l.mbid
458                            .as_deref()
459                            .is_some_and(|mbid| text::normalize(mbid).contains(&wanted))
460                        || context
461                            .sourced
462                            .label_mbids
463                            .get(&l.id)
464                            .is_some_and(|mbid| text::normalize(mbid).contains(&wanted))
465                }),
466                // A role field names a person, and a person who is in the
467                // library but never credited that way is an empty result, not
468                // a misunderstanding.
469                Field::Credit(_)
470                | Field::Performing
471                | Field::Guest
472                | Field::CompilationArtist
473                | Field::Contributor
474                | Field::Collaborator => {
475                    context.catalog.artists.iter().any(|a| {
476                        a.key.contains(&wanted)
477                            || a.mbid
478                                .as_deref()
479                                .is_some_and(|mbid| text::normalize(mbid) == wanted)
480                    }) || context
481                        .sourced
482                        .credits
483                        .values()
484                        .chain(context.sourced.edition_credits.values())
485                        .flatten()
486                        .any(|credit| {
487                            text::normalize(&credit.artist_name).contains(&wanted)
488                                || text::normalize(&credit.artist_mbid) == wanted
489                                || credit
490                                    .credited_as
491                                    .as_deref()
492                                    .is_some_and(|name| text::normalize(name).contains(&wanted))
493                        })
494                }
495                _ => true,
496            };
497            if !known {
498                out.push((what.to_string(), wanted));
499            }
500        }
501    }
502}

517/// Everything needed to answer, gathered once rather than per track.
518pub struct Context<'a> {
519    /// The library.
520    pub catalog: &'a Catalog,
521    /// What the user wrote about it.
522    pub data: &'a UserData,
523    /// Whose opinions count.
524    pub owner: &'a str,
525    /// Certain external relationships, indexed once for this evaluation.
526    sourced: SourcedRelations,
527    local: indexes::LocalIndexes<'a>,
528}
529
530impl<'a> Context<'a> {
531    /// Creates a query context over local tags and user data.
532    pub fn new(catalog: &'a Catalog, data: &'a UserData, owner: &'a str) -> Context<'a> {
533        Context {
534            catalog,
535            data,
536            owner,
537            sourced: SourcedRelations::default(),
538            local: indexes::LocalIndexes::new(catalog, data, owner),
539        }
540    }
541
542    /// Adds externally sourced relationships that are certain enough to
543    /// traverse. Approximate matches remain evidence and never silently become
544    /// query results.
545    pub fn with_sources(mut self, sources: &crate::sources::Sources) -> Context<'a> {
546        self.sourced = SourcedRelations::new(self.catalog, sources);
547        self
548    }
549}

551#[derive(Default)]
552struct SourcedRelations {
553    works: BTreeMap<Id, Vec<crate::sources::WorkLink>>,
554    credits: BTreeMap<Id, Vec<crate::sources::CreditLink>>,
555    edition_credits: BTreeMap<Id, Vec<crate::sources::CreditLink>>,
556    label_mbids: BTreeMap<Id, String>,
557}
558
559impl SourcedRelations {
560    fn new(catalog: &Catalog, sources: &crate::sources::Sources) -> SourcedRelations {
561        let mut indexed = SourcedRelations::default();
562        for link in sources
563            .work_links(catalog)
564            .into_iter()
565            .filter(|link| link.trusted)
566        {
567            let works = indexed.works.entry(link.recording_id).or_default();
568            if !works.contains(&link.work) {
569                works.push(link.work);
570            }
571        }
572        for link in sources
573            .credit_links(catalog)
574            .into_iter()
575            .filter(|link| link.trusted)
576        {
577            let credits = indexed.credits.entry(link.recording_id).or_default();
578            if !credits.contains(&link.credit) {
579                credits.push(link.credit);
580            }
581        }
582        for link in sources
583            .edition_credit_links(catalog)
584            .into_iter()
585            .filter(|link| link.trusted)
586        {
587            let credits = indexed.edition_credits.entry(link.release_id).or_default();
588            if !credits.contains(&link.credit) {
589                credits.push(link.credit);
590            }
591        }
592        for label in &catalog.labels {
593            let Some(identity) = sources.label_identity(catalog, label.id) else {
594                continue;
595            };
596            let mbid = match identity {
597                crate::sources::LabelIdentityResolution::Confirmed { mbid, .. }
598                | crate::sources::LabelIdentityResolution::Agrees { mbid, .. }
599                | crate::sources::LabelIdentityResolution::Accepted { mbid, .. } => mbid,
600                _ => continue,
601            };
602            indexed.label_mbids.insert(label.id, mbid);
603        }
604        indexed
605    }
606}

608/// The tracks a query matches, in catalog order.
609///
610/// Order matters because everything downstream pages through it, and paging is
611/// only meaningful while the order is the same on every run.
612pub fn run(query: &Query, context: &Context) -> Vec<Id> {
613    context
614        .catalog
615        .tracks
616        .iter()
617        .filter(|track| matches(query, context, track.id))
618        .map(|track| track.id)
619        .collect()
620}
621
622/// Whether one track satisfies a query.
623pub fn matches(query: &Query, context: &Context, track: Id) -> bool {
624    match query {
625        Query::All => true,
626        Query::And(parts) => parts.iter().all(|p| matches(p, context, track)),
627        Query::Or(parts) => parts.iter().any(|p| matches(p, context, track)),
628        Query::Not(inner) => !matches(inner, context, track),
629        Query::Term(term) => term_matches(term, context, track),
630    }
631}
632
633fn term_matches(term: &Term, context: &Context, track: Id) -> bool {
634    match &term.test {
635        Test::Set => flag_of(&term.field, context, track),
636        Test::Unset => !flag_of(&term.field, context, track),
637        Test::Compare(compare, wanted) => match number_of(&term.field, context, track) {
638            // A track with no year cannot satisfy a question about years. It is
639            // absent from the answer rather than counted as zero, which would
640            // put every untagged file in "before 1970".
641            None => false,
642            Some(value) => match compare {
643                Compare::Equal => (value - wanted).abs() < f64::EPSILON,
644                Compare::Greater => value > *wanted,
645                Compare::AtLeast => value >= *wanted,
646                Compare::Less => value < *wanted,
647                Compare::AtMost => value <= *wanted,
648            },
649        },
650        Test::Between(low, high) => match number_of(&term.field, context, track) {
651            None => false,
652            Some(value) => {
653                low.map(|l| value >= l).unwrap_or(true) && high.map(|h| value <= h).unwrap_or(true)
654            }
655        },
656        Test::Contains(wanted) => {
657            let wanted = text::normalize(wanted);
658            texts_of(&term.field, context, track)
659                .iter()
660                .any(|value| text::normalize(value).contains(&wanted))
661        }
662        Test::Is(wanted) => {
663            let wanted = text::normalize(wanted);
664            texts_of(&term.field, context, track)
665                .iter()
666                .any(|value| text::normalize(value) == wanted)
667        }
668    }
669}

671/// The reference for the entity a scope names, starting from a track.
672fn scoped(scope: Scope, context: &Context, track: Id) -> Option<EntityRef> {
673    let catalog = context.catalog;
674    match scope {
675        Scope::Track => EntityRef::of(catalog, EntityKind::Track, track),
676        Scope::Album => catalog
677            .track(track)
678            .and_then(|t| t.release_id)
679            .and_then(|r| EntityRef::of(catalog, EntityKind::Release, r)),
680        Scope::Artist => main_artist(context, track)
681            .and_then(|artist| EntityRef::of(catalog, EntityKind::Artist, artist.id)),
682    }
683}

687    context: &'a Context,

690    let reference = scoped(scope, context, track)?;
691    context

720fn flag_of(field: &Field, context: &Context, track: Id) -> bool {
721    match field {
722        Field::Lossless => context
723            .catalog
724            .track(track)
725            .and_then(|t| context.catalog.file(t.file_id))
726            .map(|f| f.properties.lossless)
727            .unwrap_or(false),
728        Field::Compilation => context
729            .catalog
730            .track(track)
731            .and_then(|t| t.release_id)
732            .and_then(|r| context.catalog.release(r))
733            .map(|r| r.is_compilation)
734            .unwrap_or(false),
735        Field::Loved(scope) => annotation(*scope, context, track)
736            .map(|a| a.loved)
737            .unwrap_or(false),
738        Field::LovedAnywhere => [Scope::Track, Scope::Album, Scope::Artist]
739            .into_iter()
740            .any(|scope| annotation(scope, context, track).is_some_and(|a| a.loved)),
741        // Any other field used as a bare flag asks whether it holds anything.
742        _ => {
743            !texts_of(field, context, track).is_empty()
744                || number_of(field, context, track).is_some()
745        }
746    }
747}

749fn number_of(field: &Field, context: &Context, track: Id) -> Option<f64> {
750    let catalog = context.catalog;
751    let track_row = catalog.track(track)?;
752    let file = catalog.file(track_row.file_id);
753    match field {
754        Field::Year => track_row
755            .release_id
756            .and_then(|r| catalog.release(r))
757            .and_then(|r| r.year)
758            .map(f64::from),
759        Field::Duration => track_row.duration_ms.map(|d| d as f64),
760        Field::Size => file.map(|f| f.size as f64),
761        Field::Bitrate => file.and_then(|f| f.properties.bitrate_kbps).map(f64::from),
762        Field::SampleRate => file.and_then(|f| f.properties.sample_rate).map(f64::from),
763        Field::Played => {
764            let reference = scoped(Scope::Track, context, track)?;
765            Some(f64::from(context.local.play_count(
766                context.data,
767                context.owner,
768                &reference,
769            )))
770        }
771        Field::Rating(scope) => annotation(*scope, context, track)
772            .and_then(|a| a.rating)
773            .map(f64::from),
774        _ => None,
775    }
776}

778fn texts_of(field: &Field, context: &Context, track: Id) -> Vec<String> {
779    let catalog = context.catalog;
780    let Some(track_row) = catalog.track(track) else {
781        return Vec::new();
782    };
783    let file = catalog.file(track_row.file_id);
784    let release = track_row.release_id.and_then(|r| catalog.release(r));
785    match field {
786        Field::Title => vec![track_row.title.clone()],
787        Field::Artist => {
788            let mut values = Vec::new();
789            for credit in context
790                .local
791                .credits(context.catalog, EntityKind::Track, track)
792                .iter()
793            {
794                extend_artist_values(catalog, credit.artist_id, &mut values);
795                if let Some(credited_as) = &credit.credited_as {
796                    push_unique(&mut values, credited_as.clone());
797                }
798            }
799            if let Some(release_id) = track_row.release_id {
800                for credit in context
801                    .local
802                    .credits(context.catalog, EntityKind::Release, release_id)
803                    .iter()
804                {
805                    extend_artist_values(catalog, credit.artist_id, &mut values);
806                }
807            }
808            for credit in sourced_credits(context, track) {
809                extend_sourced_credit_values(credit, &mut values);
810            }
811            values
812        }
813        Field::Album => release
814            .map(|r| {
815                let mut values = vec![r.title.clone()];
816                if let Some(mbid) = &r.mbid {
817                    values.push(mbid.clone());
818                }
819                values
820            })
821            .unwrap_or_default(),
822        Field::Recording => catalog
823            .recording(track_row.recording_id)
824            .map(|recording| {
825                let mut values = vec![recording.title.clone()];
826                if let Some(mbid) = &recording.mbid {
827                    values.push(mbid.clone());
828                }
829                values
830            })
831            .unwrap_or_default(),
832        Field::Work => {
833            let mut values = catalog
834                .recording(track_row.recording_id)
835                .map(|recording| {
836                    recording
837                        .work_ids
838                        .iter()
839                        .filter_map(|&id| catalog.work(id))
840                        .flat_map(|work| vec![work.title.clone(), work.mbid.clone()])
841                        .collect()
842                })
843                .unwrap_or_default();
844            // A local WORK/grouping tag is searchable even without an MBID.
845            // This is text on one file, not a new canonical work identity.
846            if let Some(work_tag) = catalog
847                .file(track_row.file_id)
848                .and_then(|file| file.first_tag("grouping"))
849            {
850                push_unique(&mut values, work_tag.to_string());
851            }
852            if let Some(works) = context.sourced.works.get(&track_row.recording_id) {
853                for work in works {
854                    push_unique(&mut values, work.title.clone());
855                    push_unique(&mut values, work.mbid.clone());
856                    for parent in &work.parents {
857                        push_unique(&mut values, parent.title.clone());
858                        push_unique(&mut values, parent.mbid.clone());
859                    }
860                }
861            }
862            values
863        }
864        Field::Movement => catalog
865            .file(track_row.file_id)
866            .map(|file| {
867                ["movement", "movementnumber"]
868                    .into_iter()
869                    .filter_map(|tag| file.first_tag(tag).map(str::to_string))
870                    .collect()
871            })
872            .unwrap_or_default(),
873        Field::ReleaseGroup => release
874            .and_then(|r| r.release_group_id)
875            .and_then(|id| catalog.release_group(id))
876            .map(|group| vec![group.title.clone(), group.mbid.clone()])
877            .unwrap_or_default(),
878        Field::AlbumArtist => release
879            .and_then(|r| r.album_artist_id)
880            .map(|a| artist_values(catalog, a))
881            .unwrap_or_default(),
882        Field::Genre => {
883            let mut names: Vec<String> = context
884                .local
885                .genres(context.catalog, EntityKind::Track, track)
886                .iter()
887                .map(|g| g.name.clone())
888                .collect();
889            if let Some(release) = release {
890                names.extend(
891                    context
892                        .local
893                        .genres(context.catalog, EntityKind::Release, release.id)
894                        .iter()
895                        .map(|g| g.name.clone()),
896                );
897            }
898            names
899        }
900        Field::Label => release
901            .map(|r| {
902                r.label_ids.iter().fold(Vec::new(), |mut values, &id| {
903                    if let Some(label) = catalog.label(id) {
904                        values.push(label.name.clone());
905                        if let Some(mbid) = &label.mbid {
906                            values.push(mbid.clone());
907                        }
908                        if let Some(mbid) = context.sourced.label_mbids.get(&id) {
909                            push_unique(&mut values, mbid.clone());
910                        }
911                    }
912                    values
913                })
914            })
915            .unwrap_or_default(),
916        Field::Comment => file
917            .and_then(|f| f.tags.get("comment"))
918            .cloned()
919            .unwrap_or_default(),
920        // The tag first, because it costs nothing — raw tags are in the
921        // catalog. The sidecar is only opened when the tag holds nothing, and
922        // only for the tracks that have one, which is what keeps a search
923        // across a library from being ten thousand file reads.
924        Field::Lyrics => catalog
925            .lyrics_of_track(track)
926            .map(|lyrics| vec![lyrics.text()])
927            .unwrap_or_default(),
928        Field::Path => file.map(|f| vec![f.path.clone()]).unwrap_or_default(),
929        Field::Codec => file
930            .map(|f| {
931                vec![
932                    f.properties.codec.clone(),
933                    f.properties.container.clone(),
934                    f.properties.quality_label(),
935                ]
936            })
937            .unwrap_or_default(),
938        Field::Performing => credits_as_text(context, track, |role, _| {
939            crate::model::is_performing_role(role)
940        }),
941        Field::Credit(role) => credits_as_text(context, track, |credited, attributes| {
942            credited == *role
943                || (*role == "orchestra" && credited == "performing orchestra")
944                || (*role == "soloist"
945                    && matches!(credited, "performer" | "instrument" | "vocal")
946                    && attributes.iter().any(|attribute| attribute.name == "solo"))
947        }),
948        Field::Instrument => {
949            let mut values = Vec::new();
950            for credit in context
951                .local
952                .credits(context.catalog, EntityKind::Track, track)
953                .iter()
954            {
955                for attribute in &credit.attributes {
956                    push_unique(&mut values, attribute.name.clone());
957                    if let Some(value) = &attribute.value {
958                        push_unique(&mut values, value.clone());
959                    }
960                    if let Some(credited_as) = &attribute.credited_as {
961                        push_unique(&mut values, credited_as.clone());
962                    }
963                }
964            }
965            for credit in sourced_credits(context, track) {
966                for attribute in &credit.attributes {
967                    push_unique(&mut values, attribute.name.clone());
968                    if let Some(value) = &attribute.value {
969                        push_unique(&mut values, value.clone());
970                    }
971                    if let Some(credited_as) = &attribute.credited_as {
972                        push_unique(&mut values, credited_as.clone());
973                    }
974                }
975            }
976            values
977        }
978        Field::Guest | Field::CompilationArtist | Field::Contributor | Field::Collaborator => {
979            participation_values(field, context, track, release)
980        }
981        Field::Tag(scope) => annotation(*scope, context, track)
982            .map(|a| a.tags.iter().cloned().collect())
983            .unwrap_or_default(),
984        Field::Note(scope) => annotation(*scope, context, track)
985            .and_then(|a| a.note.clone())
986            .map(|n| vec![n])
987            .unwrap_or_default(),
988        // A bare word searches where a person would expect it to.
989        Field::Anything => {
990            let mut all = vec![track_row.title.clone()];
991            all.extend(texts_of(&Field::Artist, context, track));
992            all.extend(texts_of(&Field::Album, context, track));
993            all
994        }
995        _ => Vec::new(),
996    }
997}

1055    context: &Context,
1056    track: Id,
1057    wanted: impl Fn(&str, &[crate::model::CreditAttribute]) -> bool,

1069    }

1071        .into_iter()

1073    {
1074        extend_sourced_credit_values(credit, &mut values);
1075    }

1079fn participation_values(
1080    field: &Field,
1081    context: &Context,
1082    track: Id,
1083    release: Option<&crate::model::Release>,
1084) -> Vec<String> {
1085    let catalog = context.catalog;
1086    if catalog.track(track).is_none() {
1087        return Vec::new();
1088    }
1089    let performing: Vec<&crate::model::Credit> = context
1090        .local
1091        .credits(context.catalog, EntityKind::Track, track)
1092        .iter()
1093        .copied()
1094        .filter(|credit| crate::model::is_performing_role(&credit.role))
1095        .collect();
1096    let sourced_performing: Vec<&crate::sources::CreditLink> = sourced_credits(context, track)
1097        .into_iter()
1098        .filter(|credit| crate::model::is_performing_role(&credit.role))
1099        .collect();
1100    let mut distinct_performers: BTreeSet<String> = performing
1101        .iter()
1102        .map(|credit| local_artist_key(catalog, credit.artist_id))
1103        .collect();
1104    distinct_performers.extend(
1105        sourced_performing
1106            .iter()
1107            .map(|credit| sourced_artist_key(catalog, credit)),
1108    );
1109    if matches!(field, Field::Contributor) {
1110        let mut values = Vec::new();
1111        for credit in local_credits(context, track)
1112            .into_iter()
1113            .filter(|credit| !crate::model::is_performing_role(&credit.role))
1114        {
1115            extend_artist_values(catalog, credit.artist_id, &mut values);
1116        }
1117        for credit in sourced_credits(context, track)
1118            .into_iter()
1119            .filter(|credit| !crate::model::is_performing_role(&credit.role))
1120        {
1121            extend_sourced_credit_values(credit, &mut values);
1122        }
1123        return values;
1124    }
1125    let Some(release) = release else {
1126        return if matches!(field, Field::Collaborator) && distinct_performers.len() > 1 {
1127            performer_values(catalog, &performing, &sourced_performing)
1128        } else {
1129            Vec::new()
1130        };
1131    };
1132    let mut values = Vec::new();
1133    match field {
1134        Field::Guest => {
1135            if !release.is_compilation {
1136                for credit in performing {
1137                    if release.album_artist_id != Some(credit.artist_id) {
1138                        extend_artist_values(catalog, credit.artist_id, &mut values);
1139                    }
1140                }
1141                for credit in sourced_performing {
1142                    if !is_album_artist(catalog, release, credit) {
1143                        extend_sourced_credit_values(credit, &mut values);
1144                    }
1145                }
1146            }
1147        }
1148        Field::CompilationArtist => {
1149            if release.is_compilation {
1150                for credit in performing {
1151                    if release.album_artist_id != Some(credit.artist_id) {
1152                        extend_artist_values(catalog, credit.artist_id, &mut values);
1153                    }
1154                }
1155                for credit in sourced_performing {
1156                    if !is_album_artist(catalog, release, credit) {
1157                        extend_sourced_credit_values(credit, &mut values);
1158                    }
1159                }
1160            }
1161        }
1162        Field::Contributor => unreachable!("contributors are handled before release lookup"),
1163        Field::Collaborator if distinct_performers.len() > 1 => {
1164            return performer_values(catalog, &performing, &sourced_performing);
1165        }
1166        _ => {}
1167    }
1168    values
1169}

1171fn local_artist_key(catalog: &Catalog, artist_id: Id) -> String {
1172    catalog
1173        .artist(artist_id)
1174        .map(|artist| {
1175            artist
1176                .mbid
1177                .clone()
1178                .unwrap_or_else(|| format!("name:{}", artist.key))
1179        })
1180        .unwrap_or_else(|| format!("local:{artist_id}"))
1181}

1183fn sourced_artist_key(catalog: &Catalog, credit: &crate::sources::CreditLink) -> String {
1184    let credited_key = credit.credited_as.as_deref().map(text::normalize);
1185    if let Some(artist) = catalog.artists.iter().find(|artist| {
1186        artist.mbid.as_deref() == Some(credit.artist_mbid.as_str())
1187            || artist.key == text::normalize(&credit.artist_name)
1188            || credited_key.as_deref() == Some(artist.key.as_str())
1189    }) {
1190        return local_artist_key(catalog, artist.id);
1191    }
1192    if credit.artist_mbid.trim().is_empty() {
1193        format!("name:{}", text::normalize(&credit.artist_name))
1194    } else {
1195        credit.artist_mbid.clone()
1196    }
1197}

1199fn is_album_artist(
1200    catalog: &Catalog,
1201    release: &crate::model::Release,
1202    credit: &crate::sources::CreditLink,
1203) -> bool {
1204    let Some(artist) = release.album_artist_id.and_then(|id| catalog.artist(id)) else {
1205        return false;
1206    };
1207    artist.mbid.as_deref() == Some(credit.artist_mbid.as_str())
1208        || artist.key == text::normalize(&credit.artist_name)
1209        || credit
1210            .credited_as
1211            .as_deref()
1212            .is_some_and(|name| artist.key == text::normalize(name))
1213}

1215fn performer_values(
1216    catalog: &Catalog,
1217    local: &[&crate::model::Credit],
1218    sourced: &[&crate::sources::CreditLink],
1219) -> Vec<String> {
1220    let mut values = Vec::new();
1221    let mut seen = BTreeSet::new();
1222    for credit in local {
1223        if seen.insert(credit.artist_id) {
1224            extend_artist_values(catalog, credit.artist_id, &mut values);
1225        }
1226    }
1227    for credit in sourced {
1228        extend_sourced_credit_values(credit, &mut values);
1229    }
1230    values
1231}
