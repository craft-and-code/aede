1//! The Cover Art Archive: the front image of a record, when it has one.
2//!
3//! Deliberately **no network**, like [`crate::musicbrainz`]: this module turns
4//! an index somebody else fetched into an address to download, and decides
5//! which image is the front one. Reaching the address is the client's job.
6//!
7//! # The index says what exists, so nothing here guesses
8//!
9//! Asking `coverartarchive.org` about a release answers with a small JSON
10//! document — not the image — listing every image it holds:
11//!
12//! ```text
13//! { "images": [ { "front": true, "types": ["Front"], "approved": true,
14//!                 "image": "…/12345.jpg",
15//!                 "thumbnails": { "250": "…", "500": "…", "1200": "…" } } ],
16//!   "release": "https://musicbrainz.org/release/…" }
17//! ```
18//!
19//! The sizes are **named in the answer**, so a caller asking for 1200 px is
20//! told whether one exists rather than being sent to an address that might
21//! `404`. That is why the index is fetched first: one small request decides
22//! everything, including whether there is anything to download at all.
23
24use crate::json::Json;

26/// The name records from this service carry in `sources.json`.
27pub const SOURCE: &str = "coverartarchive";
28
29/// Base address of the service.
30pub const WEB_SERVICE: &str = "https://coverartarchive.org";
31
32/// Which image to take, of the ones the archive offers.
33#[derive(Debug, Clone, Copy, PartialEq, Eq)]
34pub enum Size {
35    /// The file as it was uploaded: often 1500–3000 px and several megabytes.
36    Original,
37    /// A thumbnail of this width, when the archive has generated one.
38    Thumbnail(u32),
39}
40
41impl Size {
42    /// Reads what a reader typed after `--size`.
43    ///
44    /// Only the widths the archive actually generates are accepted. A number it
45    /// has never heard of would be asked for, not found, and silently fall back
46    /// to something else — so it is refused here, where the message can list
47    /// what there is.
48    pub fn parse(text: &str) -> Option<Size> {
49        match text.trim().to_ascii_lowercase().as_str() {
50            "original" | "full" => Some(Size::Original),
51            "250" => Some(Size::Thumbnail(250)),
52            "500" => Some(Size::Thumbnail(500)),
53            "1200" => Some(Size::Thumbnail(1200)),
54            _ => None,
55        }
56    }
57
58    /// How this size reads in a message.
59    pub fn as_str(self) -> String {
60        match self {
61            Size::Original => "original".to_string(),
62            Size::Thumbnail(px) => format!("{px} px"),
63        }
64    }
65}

67/// Where to ask what images a record has.
68///
69/// A **release group** rather than a release: the catalog's own identifier for
70/// an album is the group's, and the archive answers a group by redirecting to
71/// whichever release its editors chose to represent it — which is the right
72/// answer for "the cover of this album" and saves picking a pressing here.
73pub fn index_url(release_group: &str) -> String {
74    format!("{WEB_SERVICE}/release-group/{release_group}")
75}
76
77/// Where to ask about one precise edition, when the tags name one.
78///
79/// Preferred over the group when it is available: the cover of the pressing on
80/// the shelf is a better answer than the cover of the pressing somebody else
81/// thought representative — a reissue often has different artwork.
82pub fn release_index_url(release: &str) -> String {
83    format!("{WEB_SERVICE}/release/{release}")
84}

86/// The front image of a record, at the size asked for.
87#[derive(Debug, Clone, PartialEq, Eq)]
88pub struct Front {
89    /// Where to download it.
90    pub url: String,
91    /// The size actually served, which may not be the one asked for.
92    pub size: Size,
93}
94
95/// Picks the front image out of an index, at the size asked for.
96///
97/// "Front" is decided by the `front` flag first and by the `types` list after:
98/// both are filled by the archive's editors, and an image marked in one and not
99/// the other is common enough that reading only one leaves records looking as
100/// though they have no cover.
101///
102/// An **approved** image wins over one that is not, when there is a choice.
103/// Anyone may upload; approval is an editor having looked. Where nothing is
104/// approved the unapproved image is still taken, because a cover somebody
105/// uploaded is a better answer than an empty folder.
106///
107/// The size falls back to the original when the thumbnail asked for is not in
108/// the answer. The archive generates them for everything it holds today, but a
109/// caller that got nothing back because a width was missing would look exactly
110/// like a record with no artwork, which is a different thing entirely.
111pub fn front(response: &Json, size: Size) -> Option<Front> {
112    let images = response.get("images")?.as_arr()?;
113    let is_front = |image: &&Json| -> bool {
114        image.field_bool("front")
115            || image
116                .get("types")
117                .and_then(Json::as_arr)
118                .is_some_and(|types| {
119                    types
120                        .iter()
121                        .filter_map(Json::as_str)
122                        .any(|t| t.eq_ignore_ascii_case("front"))
123                })
124    };
125    let fronts: Vec<&Json> = images.iter().filter(is_front).collect();
126    let chosen = fronts
127        .iter()
128        .find(|image| image.field_bool("approved"))
129        .or_else(|| fronts.first())?;
130
131    let original = chosen.field_str("image");
132    if let Size::Thumbnail(px) = size
133        && let Some(url) = chosen
134            .get("thumbnails")
135            .and_then(|t| t.field_str(&px.to_string()))
136        && !url.is_empty()
137    {
138        return Some(Front { url, size });
139    }
140    Some(Front {
141        url: original.filter(|u| !u.is_empty())?,
142        size: Size::Original,
143    })
144}

146/// What image format these bytes are, if this program should write them.
147///
148/// **Sniffed, never taken from the address.** A download that went wrong — a
149/// redirect to a login page, an error document, a truncated transfer — arrives
150/// as bytes like any other, and writing those into a music folder under the
151/// name `cover.jpg` is silent corruption that only surfaces months later when
152/// something tries to display it. An extension read off a URL would not catch
153/// any of it.
154///
155/// `None` means: do not write this.
156pub fn image_kind(bytes: &[u8]) -> Option<&'static str> {
157    match bytes {
158        // JPEG: SOI marker, then the start of the first segment.
159        [0xFF, 0xD8, 0xFF, ..] => Some("jpg"),
160        [0x89, b'P', b'N', b'G', 0x0D, 0x0A, 0x1A, 0x0A, ..] => Some("png"),
161        // The archive serves neither today, and accepting a format nothing in
162        // this program can read would only move the failure somewhere quieter.
163        _ => None,
164    }
165}

167/// What a picture is of, for the few kinds worth telling apart.
168///
169/// The archive and the tag formats both classify their images, and the
170/// classification is the only thing that makes a second image worth keeping: a
171/// folder of `image-1.jpg` … `image-9.jpg` is not more useful than none.
172#[derive(Debug, Clone, Copy, PartialEq, Eq)]
173pub enum Kind {
174    /// The front of the sleeve: the album cover.
175    Front,
176    /// The back of the sleeve, usually the track listing.
177    Back,
178    /// A page of the booklet or insert.
179    Booklet,
180    /// The disc itself — the label printed on a CD or a record.
181    Media,
182    /// A photograph of the artist.
183    Artist,
184    /// An artist's logo — not a photograph of them, a mark that stands for
185    /// them. See [`crate::fanarttv::logo_url`], its only source today.
186    Logo,
187    /// A wide background image for the artist — a header or hero image, not
188    /// the small mark [`Kind::Logo`] is. See [`crate::fanarttv::banner_url`],
189    /// its only source today.
190    Banner,
191    /// A full-screen artist background. Fanart.tv's 4K pool is preferred.
192    Background,
193    /// Anything else the source classifies, or nothing at all.
194    Other,
195}
196
197impl Kind {
198    /// How this program spells the kind, in a name and in a message.
199    ///
200    /// `cover` for the front, because that is the name every player and file
201    /// manager looks for and this program's own scanner looks for first — see
202    /// [`file_name`]. The rest are named for what they are.
203    pub fn stem(self) -> &'static str {
204        match self {
205            Kind::Front => "cover",
206            Kind::Back => "back",
207            Kind::Booklet => "booklet",
208            Kind::Media => "media",
209            Kind::Artist => "artist",
210            Kind::Logo => "logo",
211            Kind::Banner => "banner",
212            Kind::Background => "background",
213            Kind::Other => "image",
214        }
215    }
216
217    /// Reads the word a source uses for a kind.
218    ///
219    /// Unknown words become [`Kind::Other`] rather than being dropped: an image
220    /// nobody classified is still an image, and losing it because the label was
221    /// unfamiliar would be a strange way to fail.
222    pub fn parse(text: &str) -> Kind {
223        match text.trim().to_ascii_lowercase().as_str() {
224            "front" => Kind::Front,
225            "back" => Kind::Back,
226            "booklet" => Kind::Booklet,
227            "medium" | "media" => Kind::Media,
228            "artist" => Kind::Artist,
229            _ => Kind::Other,
230        }
231    }
232}

234/// What to call the file written beside the music.
235///
236/// `cover` is the first name [`crate::scan`] looks for, so the next scan finds
237/// it without being told — the file is not registered anywhere, it is simply
238/// discovered like one the user put there themselves.
239pub fn file_name(kind: &str) -> String {
240    format!("cover.{kind}")
241}
242
243/// What to call one image of a given kind, when several may share it.
244///
245/// `back.jpg` for the only back, `booklet-01.jpg` and `booklet-02.jpg` for the
246/// pages of an insert. The number starts at one and is only added when there is
247/// more than one of a kind, because `booklet-01.jpg` alone reads as a page torn
248/// out of something.
249pub fn image_name(kind: Kind, format: &str, index: usize, of: usize) -> String {
250    match of {
251        0 | 1 => format!("{}.{format}", kind.stem()),
252        _ => format!("{}-{:02}.{format}", kind.stem(), index + 1),
253    }
254}

256/// The subfolder everything that is not the cover is written into.
257///
258/// Not beside the music, and the reason is [`crate::scan::cover_in`]: the
259/// scanner takes *any* image in a folder as a candidate cover when none is
260/// named `cover`. A `back.jpg` sitting next to the tracks would therefore
261/// become the album's artwork in this program and in a good many players —
262/// which is precisely the wrong picture, and one nothing would ever look at
263/// again. So the front stays beside the music, under the name everything looks
264/// for, and the rest go one level down.
265///
266/// `artwork/` alongside the `spectrograms/` that `aede spectrum` already
267/// writes: a folder named for what is in it.
268pub const EXTRAS: &str = "artwork";
269
270/// Where the images that are not the cover belong, for one album folder.
271pub fn extras_in(folder: &std::path::Path) -> std::path::PathBuf {
272    folder.join(EXTRAS)
273}

275/// Where each image sits among the others of its own kind.
276///
277/// Answers, for every image in the order given, `(index, of)` — its place
278/// among the images of that same kind, and how many there are. That pair is
279/// what [`image_name`] needs, and computing it here rather than in each caller
280/// is what keeps a downloaded booklet and an extracted one named alike.
281pub fn positions(kinds: &[Kind]) -> Vec<(usize, usize)> {
282    let mut so_far: Vec<(Kind, usize)> = Vec::new();
283    let mut out = Vec::with_capacity(kinds.len());
284    for kind in kinds {
285        let index = match so_far.iter_mut().find(|(k, _)| k == kind) {
286            Some((_, seen)) => {
287                *seen += 1;
288                *seen - 1
289            }
290            None => {
291                so_far.push((*kind, 1));
292                0
293            }
294        };
295        let of = kinds.iter().filter(|k| *k == kind).count();
296        out.push((index, of));
297    }
298    out
299}

301/// Writes an image beside music, or says why it did not.
302///
303/// **The one place anything in this program puts a picture into somebody's
304/// library**, whether it was downloaded from the archive or pulled out of their
305/// own files. One function rather than two, because it carries two guards that
306/// would otherwise be written twice and eventually differ:
307///
308/// - the bytes are **sniffed**, and anything that is not a JPEG or a PNG is
309///   refused — see [`image_kind`];
310/// - the target is checked **immediately before writing**, not when the caller
311///   chose it. A catalog is a snapshot and a disk is not, and this is the one
312///   moment in the program where being out of date destroys something the user
313///   made.
314///
315/// Never overwrites. There is no flag to make it.
316///
317/// **And never adds a second cover to a folder that has one**, whatever it is
318/// called. Checking only the name about to be written is not enough: an
319/// extracted `cover.png` and a downloaded `cover.jpg` are two different names
320/// and both rank as the album's cover, so a folder can end up holding two, with
321/// no way to say which a player will show — and the one that wins may be the
322/// 1200 px download rather than the full-size picture that was in the files.
323/// [`crate::scan::cover_in`] asks the question the scanner itself asks, at the
324/// moment of writing, which is the only moment that can answer it.
325pub fn write_beside(folder: &std::path::Path, bytes: &[u8]) -> Result<std::path::PathBuf, String> {
326    if let Some(already) = crate::scan::cover_in(folder) {
327        return Err(format!(
328            "{} is already the cover of this folder",
329            already.display()
330        ));
331    }
332    match write_image(folder, Kind::Front, (0, 1), bytes)? {
333        Written::New(path) => Ok(path),
334        Written::Already(path) => Err(format!("{} already exists", path.display())),
335    }
336}

338/// What became of one image.
339///
340/// Two outcomes rather than one, because a file that is already there is not a
341/// failure and must not be counted or reported as one: running `--images` twice
342/// over a library would otherwise print a page of errors for having done its
343/// job the first time.
344#[derive(Debug, Clone, PartialEq, Eq)]
345pub enum Written {
346    /// Written, at this path.
347    New(std::path::PathBuf),
348    /// A file of that name was already there, and nothing overwrites.
349    Already(std::path::PathBuf),
350}
351
352/// Whether an image of this kind is already sitting beside the music, under
353/// either extension [`write_image`] can produce.
354///
355/// Fanart.tv's auxiliary artist images are tracked by disk presence, so a pass
356/// needs to ask "is there one already" before downloading it. Checks both extensions
357/// because the writer picks jpg or png from the bytes it was given, never
358/// from what the caller asked for, so a caller checking ahead of the
359/// download cannot know which one a previous run wrote.
360pub fn exists_beside(folder: &std::path::Path, kind: Kind) -> bool {
361    folder.join(format!("{}.jpg", kind.stem())).exists()
362        || folder.join(format!("{}.png", kind.stem())).exists()
363}

365/// Writes one image of a known kind, under the name that says what it is.
366///
367/// The general form of [`write_beside`], carrying the same two guards — the
368/// bytes are sniffed, and nothing is ever written over. `where_` is the pair
369/// [`positions`] gives for this image.
370///
371/// The folder is created if it is not there, because the images that are not
372/// the cover go into a subfolder that will not exist the first time.
373///
374/// `Err` is for the two things that are actually wrong: bytes that are not an
375/// image, and a write that failed.
376pub fn write_image(
377    folder: &std::path::Path,
378    kind: Kind,
379    where_: (usize, usize),
380    bytes: &[u8],
381) -> Result<Written, String> {
382    let Some(format) = image_kind(bytes) else {
383        return Err(format!(
384            "what came back is not a JPEG or a PNG ({} bytes), so nothing was written",
385            bytes.len()
386        ));
387    };
388    let (index, of) = where_;
389    if !folder.exists() {
390        std::fs::create_dir_all(folder).map_err(|e| format!("{}: {e}", folder.display()))?;
391    }
392    let path = folder.join(image_name(kind, format, index, of));
393    if path.exists() {
394        return Ok(Written::Already(path));
395    }
396    match crate::store::write_new_atomic(&path, bytes) {
397        Ok(true) => Ok(Written::New(path)),
398        Ok(false) => Ok(Written::Already(path)),
399        Err(error) => Err(format!("{}: {error}", path.display())),
400    }
401}

403/// Every image an index holds, with what each one is of.
404///
405/// The front, when there is one, keeps its place as the album's cover; the rest
406/// are what `--images` is for. Ordered front first and then as the archive
407/// lists them, so a booklet's pages keep the order somebody scanned them in.
408pub fn images(response: &Json, size: Size) -> Vec<(Kind, Front)> {
409    let Some(rows) = response.get("images").and_then(Json::as_arr) else {
410        return Vec::new();
411    };
412    let mut out: Vec<(Kind, Front)> = Vec::new();
413    for row in rows {
414        let kind = match row.field_bool("front") {
415            true => Kind::Front,
416            false => row
417                .get("types")
418                .and_then(Json::as_arr)
419                .and_then(|types| types.iter().find_map(Json::as_str).map(Kind::parse))
420                .unwrap_or(Kind::Other),
421        };
422        if let Some(found) = at_size(row, size) {
423            out.push((kind, found));
424        }
425    }
426    out.sort_by_key(|(kind, _)| u8::from(*kind != Kind::Front));
427    out
428}

430/// One image's address at the size asked for, falling back to the original.
431fn at_size(image: &Json, size: Size) -> Option<Front> {
432    if let Size::Thumbnail(px) = size
433        && let Some(url) = image
434            .get("thumbnails")
435            .and_then(|t| t.field_str(&px.to_string()))
436        && !url.is_empty()
437    {
438        return Some(Front { url, size });
439    }
440    Some(Front {
441        url: image.field_str("image").filter(|u| !u.is_empty())?,
442        size: Size::Original,
443    })
444}
