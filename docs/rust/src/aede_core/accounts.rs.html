1//! Stable personal owners, account administration and protected credentials.
2//!
3//! The first administrator owns `local`, so existing personal data stays put.
4//! Other owners are generated independently of their changeable login names.
5//! Callers hold the data-directory writer lock for read/modify/save operations.
6//! Sessions belong to the server and are never persisted in this store.
7//! Optional persistent client keys retain only salted secret verifiers and
8//! are independently revocable; account-generation changes purge them.
9

24/// Account-store format, independent of the catalog and personal stores.
25pub const FORMAT_VERSION: u32 = 1;
26/// Account store inside the data directory.
27pub const ACCOUNTS_FILE: &str = "accounts.json";
28/// Maximum number of accounts in one installation.
29pub const MAX_ACCOUNTS: usize = 256;
30/// Maximum accepted UTF-8 password length in bytes.
31pub const MAX_PASSWORD_BYTES: usize = 1024;
32const MAX_FILE_BYTES: u64 = 1024 * 1024;
33const MAX_REVISION: u64 = (1 << 53) - 1;
34const MEMORY_KIB: u32 = 19 * 1024;
35// Unknown names still pay the same bounded Argon2 work. This dummy verifier
36// does not authenticate anyone, regardless of the verification result.
37const DUMMY_HASH: &str = "$argon2id$v=19$m=19456,t=2,p=1$YWFhYWFhYWFhYWFhYWFhYQ$AAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAA";

39/// Installation privileges; every account may access only its own opinions.
40#[derive(Debug, Clone, Copy, PartialEq, Eq)]
41pub enum Role {
42    /// May manage accounts and run administrative installation operations.
43    Administrator,
44    /// May browse the shared catalog and manage their own personal data.
45    User,
46    /// May browse the shared catalog and read their own personal data only.
47    Auditor,
48}
49
50impl Role {
51    /// Stable spelling used in persisted and HTTP documents.
52    pub fn as_str(self) -> &'static str {
53        match self {
54            Self::Administrator => "admin",
55            Self::User => "user",
56            Self::Auditor => "auditor",
57        }
58    }
59
60    /// Parse an explicit role without silently selecting a default.
61    pub fn parse(value: &str) -> Option<Self> {
62        match value {
63            "admin" => Some(Self::Administrator),
64            "user" => Some(Self::User),
65            "auditor" => Some(Self::Auditor),
66            _ => None,
67        }
68    }
69}

71/// An account with an immutable personal-data owner and private verifier.
72#[derive(Clone, PartialEq, Eq)]
73pub struct Account {
74    /// Stable owner of annotations, history and collections.
75    pub id: String,
76    /// Canonical lowercase login name, independent of ownership.
77    pub username: String,
78    /// Installation privileges.
79    pub role: Role,
80    /// Disabled accounts retain their personal data and cannot authenticate.
81    pub enabled: bool,
82    /// Creation time in Unix seconds.
83    pub created_at: u64,
84    /// Most recent administration time in Unix seconds.
85    pub updated_at: u64,
86    revision: u64,
87    password_hash: String,
88}

109/// Independently versioned account store; an empty store is never published.
110#[derive(Debug, Clone, PartialEq, Eq)]
111pub struct Accounts {
112    epoch: String,
113    records: Vec<Account>,
114    api_keys: Vec<ApiKey>,
115}

117impl Accounts {
118    /// Create the first administrator, retaining the existing `local` owner.
119    pub fn bootstrap(username: &str, password: &str, now: u64) -> Result<Self, String> {
120        Ok(Self {
121            epoch: random_token()?,
122            records: vec![Account {
123                id: crate::user::LOCAL_USER.into(),
124                username: login_name(username)?,
125                role: Role::Administrator,
126                enabled: true,
127                created_at: now,
128                updated_at: now,
129                revision: 1,
130                password_hash: hash_password(password)?,
131            }],
132            api_keys: Vec::new(),
133        })
134    }
135
136    /// Account metadata in creation order; verifiers remain private.
137    pub fn all(&self) -> &[Account] {
138        &self.records
139    }
140
141    /// Epoch captured by every session; restoring a backup rotates it.
142    pub fn epoch(&self) -> &str {
143        &self.epoch
144    }
145
146    /// Find a canonical login name, accepting an uppercase spelling.
147    pub fn find(&self, username: &str) -> Option<&Account> {
148        let name = login_name(username).ok()?;
149        self.records.iter().find(|account| account.username == name)
150    }
151
152    /// Authenticate without distinguishing missing, disabled or wrong-password accounts.
153    /// Password work must be scheduled on a bounded blocking worker by HTTP callers.
154    pub fn authenticate(&self, username: &str, password: &str) -> Option<&Account> {
155        if password.len() > MAX_PASSWORD_BYTES {
156            return None;
157        }
158        let account = self.find(username);
159        let hash = account.map_or(DUMMY_HASH, |account| account.password_hash.as_str());
160        let parsed = PasswordHash::new(hash).ok()?;
161        let matches = hasher()
162            .ok()?
163            .verify_password(password.as_bytes(), &parsed)
164            .is_ok();
165        account.filter(|account| account.enabled && matches)
166    }
167
168    /// Validate the owner and generation captured by a session after a reload.
169    pub fn session_account(&self, owner: &str, epoch: &str, revision: u64) -> Option<&Account> {
170        if self.epoch != epoch {
171            return None;
172        }
173        self.records
174            .iter()
175            .find(|account| account.id == owner && account.enabled && account.revision == revision)
176    }
177
178    /// Add an enabled account with a generated, immutable personal owner.
179    pub fn create(
180        &mut self,
181        username: &str,
182        password: &str,
183        role: Role,
184        now: u64,
185    ) -> Result<(), String> {
186        let username = login_name(username)?;
187        if self.records.len() >= MAX_ACCOUNTS {
188            return Err("the installation already has 256 accounts".into());
189        }
190        if self.find(&username).is_some() {
191            return Err("this login name is already in use".into());
192        }
193        let id = format!("account-{}", random_token()?);
194        if self.records.iter().any(|account| account.id == id) {
195            return Err("could not generate a unique account owner; retry".into());
196        }
197        self.records.push(Account {
198            id,
199            username,
200            role,
201            enabled: true,
202            created_at: now,
203            updated_at: now,
204            revision: 1,
205            password_hash: hash_password(password)?,
206        });
207        Ok(())
208    }
209
210    /// Change privileges while retaining at least one enabled administrator.
211    pub fn set_role(&mut self, username: &str, role: Role, now: u64) -> Result<(), String> {
212        self.edit(username, now, |account| account.role = role)
213    }
214
215    /// Disable or enable access, preserving every personal record.
216    pub fn set_enabled(&mut self, username: &str, enabled: bool, now: u64) -> Result<(), String> {
217        self.edit(username, now, |account| account.enabled = enabled)
218    }
219
220    /// Rename a login without moving personal data.
221    pub fn rename(&mut self, username: &str, replacement: &str, now: u64) -> Result<(), String> {
222        let replacement = login_name(replacement)?;
223        if self
224            .find(&replacement)
225            .is_some_and(|other| !other.username.eq_ignore_ascii_case(username))
226        {
227            return Err("this login name is already in use".into());
228        }
229        self.edit(username, now, |account| account.username = replacement)
230    }
231
232    /// Replace a salted password verifier and invalidate this account's sessions and API keys.
233    pub fn set_password(&mut self, username: &str, password: &str, now: u64) -> Result<(), String> {
234        if self.find(username).is_none() {
235            return Err("no account has this login name".into());
236        }
237        let hash = hash_password(password)?;
238        self.edit(username, now, |account| account.password_hash = hash)
239    }
240
241    /// Revoke one account's sessions and API keys without changing its password or opinions.
242    pub fn revoke(&mut self, username: &str, now: u64) -> Result<(), String> {
243        let account = self
244            .records
245            .iter_mut()
246            .find(|account| account.username.eq_ignore_ascii_case(username))
247            .ok_or("no account has this login name")?;
248        account.revision = next_revision(account.revision)?;
249        account.updated_at = now;
250        let owner = account.id.clone();
251        self.purge_api_keys(&owner);
252        Ok(())
253    }
254
255    /// Invalidate all sessions and API keys, including credentials from before a backup restore.
256    pub fn revoke_all(&mut self) -> Result<(), String> {
257        self.epoch = random_token()?;
258        self.api_keys.clear();
259        Ok(())
260    }
261
262    fn edit(
263        &mut self,
264        username: &str,
265        now: u64,
266        edit: impl FnOnce(&mut Account),
267    ) -> Result<(), String> {
268        let index = self
269            .records
270            .iter()
271            .position(|account| account.username.eq_ignore_ascii_case(username))
272            .ok_or("no account has this login name")?;
273        let mut candidate = self.records[index].clone();
274        edit(&mut candidate);
275        if candidate == self.records[index] {
276            return Ok(());
277        }
278        if !(candidate.enabled && candidate.role == Role::Administrator)
279            && !self.records.iter().enumerate().any(|(other, account)| {
280                other != index && account.enabled && account.role == Role::Administrator
281            })
282        {
283            return Err("keep at least one enabled administrator".into());
284        }
285        candidate.revision = next_revision(candidate.revision)?;
286        candidate.updated_at = now;
287        let owner = candidate.id.clone();
288        self.records[index] = candidate;
289        self.purge_api_keys(&owner);
290        Ok(())
291    }
292}

294fn next_revision(previous: u64) -> Result<u64, String> {
295    previous
296        .checked_add(1)
297        .filter(|next| *next <= MAX_REVISION)
298        .ok_or_else(|| "account generation is exhausted".into())
299}

301/// Validate and canonicalize a login name without treating it as a path or owner.
302/// Accept 1–64 ASCII letters, digits, dots, underscores or hyphens, with at
303/// least one letter or digit. The canonical spelling is lowercase.
304pub fn login_name(name: &str) -> Result<String, String> {
305    if name.is_empty()
306        || name.len() > 64
307        || !name
308            .bytes()
309            .all(|byte| byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'-'))
310        || !name.bytes().any(|byte| byte.is_ascii_alphanumeric())
311    {
312        return Err(
313            "login names need 1–64 ASCII letters, digits, dots, underscores or hyphens, including at least one letter or digit".into(),
314        );
315    }
316    Ok(name.to_ascii_lowercase())
317}

319fn hasher() -> Result<Argon2<'static>, String> {
320    let params =
321        Params::new(MEMORY_KIB, 2, 1, Some(32)).map_err(|_| "invalid password hash policy")?;
322    Ok(Argon2::new(Algorithm::Argon2id, Version::V0x13, params))
323}
324
325fn hash_password(password: &str) -> Result<String, String> {
326    if password.chars().count() < 15
327        || password.len() > MAX_PASSWORD_BYTES
328        || password.contains('\0')
329    {
330        return Err(
331            "passwords need at least 15 characters and at most 1024 UTF-8 bytes, without NUL"
332                .into(),
333        );
334    }
335    let mut bytes = [0_u8; 16];
336    OsRng
337        .try_fill_bytes(&mut bytes)
338        .map_err(|_| "secure random generation failed")?;
339    let salt = SaltString::encode_b64(&bytes).map_err(|_| "could not encode password salt")?;
340    hasher()?
341        .hash_password(password.as_bytes(), &salt)
342        .map(|hash| hash.to_string())
343        .map_err(|_| "password hashing failed".into())
344}

346/// Generate an opaque 256-bit secret from the operating system's secure RNG.
347/// Failures are reported; there is no timestamp or deterministic fallback.
348pub fn random_token() -> Result<String, String> {
349    use std::fmt::Write;
350    let mut bytes = [0_u8; 32];
351    OsRng
352        .try_fill_bytes(&mut bytes)
353        .map_err(|_| "secure random generation failed")?;
354    let mut token = String::with_capacity(64);
355    for byte in bytes {
356        write!(&mut token, "{byte:02x}").map_err(|_| "could not encode random token")?;
357    }
358    Ok(token)
359}

366/// Refuse links, special files and Unix credentials readable by other users.
367/// An absent destination is accepted for a new atomic write.
368pub fn check_private_file(path: &Path) -> Result<(), StoreError> {
369    inspect_private_file(path).map(|_| ())
370}
371
372fn inspect_private_file(path: &Path) -> Result<Option<std::fs::Metadata>, StoreError> {
373    match std::fs::symlink_metadata(path) {
374        Ok(metadata) => {
375            check_private_metadata(&metadata)?;
376            Ok(Some(metadata))
377        }
378        Err(error) if error.kind() == std::io::ErrorKind::NotFound => Ok(None),
379        Err(error) => Err(error.into()),
380    }
381}
382
383fn check_private_metadata(metadata: &std::fs::Metadata) -> Result<(), StoreError> {
384    if !metadata.is_file() {
385        return Err(StoreError::AccountsInvalid(
386            "credentials require a regular file, never a symbolic link",
387        ));
388    }
389    #[cfg(unix)]
390    {
391        use std::os::unix::fs::PermissionsExt;
392        if metadata.permissions().mode() & 0o077 != 0 {
393            return Err(StoreError::AccountsInvalid(
394                "credential files must be private (chmod 600)",
395            ));
396        }
397    }
398    Ok(())
399}

401/// Validate the inspected source and the descriptor that supplied credentials.
402/// On Unix their device/inode identity must also match, so an intervening
403/// replacement cannot supply a different account store or private backup.
404pub(crate) fn validate_private_read(
405    inspected: &std::fs::Metadata,
406    opened: &std::fs::Metadata,
407) -> Result<(), StoreError> {
408    check_private_metadata(inspected)?;
409    check_private_metadata(opened)?;
410    #[cfg(unix)]
411    {
412        use std::os::unix::fs::MetadataExt;
413        if inspected.dev() != opened.dev() || inspected.ino() != opened.ino() {
414            return Err(StoreError::AccountsInvalid(
415                "credential source changed while it was opened; retry",
416            ));
417        }
418    }
419    Ok(())
420}

422/// Read a bounded private account file; initial absence is distinct from malformed data.
423/// The opened descriptor is rechecked for regular-file type and private Unix
424/// permissions, including device/inode identity on Unix. Disappearance after
425/// inspection is an error, so it cannot silently disable authentication.
426pub fn load(path: &Path) -> Result<Option<Accounts>, StoreError> {
427    load_with_open(path, |path| std::fs::File::open(path))
428}
429
430fn load_with_open(
431    path: &Path,
432    open: impl FnOnce(&Path) -> std::io::Result<std::fs::File>,
433) -> Result<Option<Accounts>, StoreError> {
434    let Some(inspected) = inspect_private_file(path)? else {
435        return Ok(None);
436    };
437    let file = open(path)?;
438    let opened = file.metadata()?;
439    validate_private_read(&inspected, &opened)?;
440    if opened.len() > MAX_FILE_BYTES {
441        return Err(StoreError::AccountsInvalid("account file exceeds 1 MiB"));
442    }
443    let mut text = String::new();
444    file.take(MAX_FILE_BYTES + 1).read_to_string(&mut text)?;
445    if text.len() as u64 > MAX_FILE_BYTES {
446        return Err(StoreError::AccountsInvalid("account file exceeds 1 MiB"));
447    }
448    from_json(&json::parse(&text).map_err(StoreError::AccountsParse)?).map(Some)
449}

451/// Save credentials atomically, preserving private permissions and validating all rows.
452/// The caller must hold the existing data-directory writer lock.
453pub fn save(accounts: &Accounts, path: &Path) -> Result<(), StoreError> {
454    check_private_file(path)?;
455    let document = to_json(accounts);
456    from_json(&document)?;
457    crate::atomic_file::write(path, document.to_string_compact().as_bytes())?;
458    Ok(())
459}

461/// Serialize the credential store for protected persistence and full backups only.
462/// Never expose this document as an account-list or catalog API response.
463pub fn to_json(accounts: &Accounts) -> Json {
464    let mut root = Json::obj();
465    root.set("format_version", FORMAT_VERSION.into());
466    root.set("epoch", accounts.epoch.clone().into());
467    root.set(
468        "account",
469        Json::Arr(
470            accounts
471                .records
472                .iter()
473                .map(|account| {
474                    let mut row = Json::obj();
475                    row.set("id", account.id.clone().into());
476                    row.set("username", account.username.clone().into());
477                    row.set("role", account.role.as_str().into());
478                    row.set("enabled", account.enabled.into());
479                    row.set("created_at", account.created_at.into());
480                    row.set("updated_at", account.updated_at.into());
481                    row.set("revision", account.revision.into());
482                    row.set("password_hash", account.password_hash.clone().into());
483                    row
484                })
485                .collect(),
486        ),
487    );
488    if !accounts.api_keys.is_empty() {
489        root.set("api_keys", keys::to_json(&accounts.api_keys));
490    }
491    root
492}

494fn valid_token(value: &str) -> bool {
495    value.len() == 64
496        && value
497            .bytes()
498            .all(|byte| byte.is_ascii_hexdigit() && !byte.is_ascii_uppercase())
499}
500
501fn valid_hash(hash: &str) -> bool {
502    if hash.len() > 256 {
503        return false;
504    }
505    let Ok(parsed) = PasswordHash::new(hash) else {
506        return false;
507    };
508    let Ok(params) = Params::try_from(&parsed) else {
509        return false;
510    };
511    parsed.algorithm.as_str() == "argon2id"
512        && parsed.version == Some(19)
513        && params.m_cost() == MEMORY_KIB
514        && params.t_cost() == 2
515        && params.p_cost() == 1
516        && parsed.hash.is_some_and(|hash| hash.len() == 32)
517        && parsed.salt.is_some_and(|salt| salt.len() == 22)
518        && parsed.params.iter().count() == 3
519}

521/// Refuse malformed, duplicate or administrator-less credentials as one whole store.
522/// Stored hash parameters are fixed to the bounded policy supported by this format.
523pub fn from_json(root: &Json) -> Result<Accounts, StoreError> {
524    let invalid = || StoreError::AccountsInvalid("malformed account store or unsupported version");
525    if root.field_u32("format_version") != Some(FORMAT_VERSION) {
526        return Err(invalid());
527    }
528    let epoch = root
529        .get("epoch")
530        .and_then(Json::as_str)
531        .filter(|epoch| valid_token(epoch))
532        .ok_or_else(invalid)?;
533    let rows = root
534        .get("account")
535        .and_then(Json::as_arr)
536        .filter(|rows| !rows.is_empty() && rows.len() <= MAX_ACCOUNTS)
537        .ok_or_else(invalid)?;
538    let mut records = Vec::with_capacity(rows.len());
539    let mut names = BTreeSet::new();
540    let mut owners = BTreeSet::new();
541    for row in rows {
542        let string = |key| row.get(key).and_then(Json::as_str).ok_or_else(invalid);
543        let number = |key| {
544            row.get(key)
545                .and_then(Json::as_u64)
546                .filter(|n| *n <= MAX_REVISION)
547                .ok_or_else(invalid)
548        };
549        let id = string("id")?;
550        if id != crate::user::LOCAL_USER && !id.strip_prefix("account-").is_some_and(valid_token) {
551            return Err(invalid());
552        }
553        let username = string("username")?;
554        if login_name(username).ok().as_deref() != Some(username)
555            || !names.insert(username)
556            || !owners.insert(id)
557        {
558            return Err(invalid());
559        }
560        let hash = string("password_hash")?;
561        if !valid_hash(hash) {
562            return Err(invalid());
563        }
564        let revision = number("revision")?;
565        if revision == 0 {
566            return Err(invalid());
567        }
568        records.push(Account {
569            id: id.into(),
570            username: username.into(),
571            role: Role::parse(string("role")?).ok_or_else(invalid)?,
572            enabled: row
573                .get("enabled")
574                .and_then(Json::as_bool)
575                .ok_or_else(invalid)?,
576            created_at: number("created_at")?,
577            updated_at: number("updated_at")?,
578            revision,
579            password_hash: hash.into(),
580        });
581    }
582    if !records
583        .iter()
584        .any(|account| account.enabled && account.role == Role::Administrator)
585    {
586        return Err(invalid());
587    }
588    let api_keys = keys::from_json(root.get("api_keys"), &records)?;
589    Ok(Accounts {
590        epoch: epoch.into(),
591        records,
592        api_keys,
593    })
594}
