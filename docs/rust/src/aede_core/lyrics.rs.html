1//! Lyrics that are already there: in the tags, or in an `.lrc` beside the file.
2//!
3//! Nothing here touches the network. Reading what a library already holds is
4//! the first half of the job and the cheap one — the parsers walk past `USLT`,
5//! `LYRICS` and `©lyr` today, and a `.lrc` sitting next to a track is a text
6//! file anyone can open. Fetching what is *missing* is another matter
7//! entirely: it needs a network, a source whose terms allow it, and an explicit
8//! choice, because lyrics are the composition's copyright and owning a FLAC
9//! grants no rights in it. See `docs/design/lyrics.md`.
10//!
11//! **Where they live follows the rule the rest of the catalog follows.** Tag
12//! lyrics are already in it, because raw tags are kept per file; a sidecar is
13//! not in the file, so pretending it were a tag would make the catalog lie
14//! about what the file says. The catalog therefore stores the sidecar's
15//! **path**, and the text is read from it when somebody asks — it is one small
16//! file, sitting right next to the music it belongs to.
17//!
18//! [`Timeline`] indexes the parsed timings for terminal and graphical players.
19//! The player supplies its decoded-track position; this module does not run a
20//! playback clock or attach lyrics to audio packets.
21

36/// At most this much of a lyrics file is read.
37///
38/// Lyrics are a page of text. A ten-megabyte `.lrc` is not a song, and reading
39/// it whole to show a track page would be paying for somebody's mistake.
40const LIMIT: usize = 256 * 1024;
41/// Text storage after repeating lines for their timestamps.
42const MIN_TEXT_BUDGET: usize = 1024 * 1024;
43
44/// One line, with the moment it is sung when the file says so.
45#[derive(Debug, Clone, PartialEq)]
46pub struct Line {
47    /// Milliseconds from the start of the track, or `None` on a plain line.
48    ///
49    /// Offsets in LRC input have already been applied. A player uses its
50    /// decoded-track position, including a seek offset, to follow these times.
51    pub at_ms: Option<u64>,
52    /// The line as written, without its timestamp when it could be expanded safely.
53    /// Invalid or excessive timestamps remain literal text on an untimed line.
54    pub text: String,
55}

57/// Where a track's lyrics were found.
58#[derive(Debug, Clone, Copy, PartialEq, Eq)]
59pub enum Source {
60    /// A tag inside the audio file itself.
61    Tag,
62    /// A `.lrc` file beside it.
63    Sidecar,
64}
65
66/// The lyrics of one track, as read.
67#[derive(Debug, Clone, PartialEq)]
68pub struct Lyrics {
69    /// Where they came from, which the page names so the reader knows.
70    pub source: Source,
71    /// The file they were read from: the audio itself, or the sidecar.
72    pub origin: String,
73    /// Every line, in order.
74    pub lines: Vec<Line>,
75}
76
77impl Lyrics {
78    /// `true` when at least one line carries a time.
79    ///
80    /// Not "every line": real `.lrc` files carry untimed headers, blank lines
81    /// and the occasional stray, and a file that is 95% timed is a synced file.
82    pub fn synced(&self) -> bool {
83        self.lines.iter().any(|line| line.at_ms.is_some())
84    }
85
86    /// The whole text, timestamps dropped — what a search looks through.
87    pub fn text(&self) -> String {
88        self.lines
89            .iter()
90            .map(|line| line.text.as_str())
91            .collect::<Vec<_>>()
92            .join("\n")
93    }
94}

106/// Reads a sidecar, bounded, and never fatally.
107///
108/// Invalid UTF-8 is replaced rather than refused. A `.lrc` written on a Windows
109/// machine in 2003 is Latin-1 as often as not, and refusing to show a song
110/// because of one accent would be the wrong trade. Reads at most 256 KiB;
111/// expanded line text stays within 1 MiB, including after lossy decoding.
112pub fn read(path: &Path) -> Option<Lyrics> {
113    let file = std::fs::File::open(path).ok()?;
114    let text = read_text(file).ok()?;
115    let lines = parse_with_budget(&text, MIN_TEXT_BUDGET);
116    match lines.is_empty() {
117        true => None,
118        false => Some(Lyrics {
119            source: Source::Sidecar,
120            origin: path.to_string_lossy().to_string(),
121            lines,
122        }),
123    }
124}

126fn read_text(reader: impl Read) -> std::io::Result<String> {
127    let mut raw = Vec::new();
128    reader.take(LIMIT as u64).read_to_end(&mut raw)?;
129    Ok(String::from_utf8_lossy(&raw).into_owned())
130}

136        true => None,
137        false => Some(Lyrics {

139            origin: origin.to_string(),
140            lines,
141        }),
142    }

145/// Reads lyrics text, timed or not.
146///
147/// One parser for both, because a tag can perfectly well hold LRC — plenty of
148/// taggers write the synced text straight into `LYRICS`, and a reader that only
149/// understood plain text would show a page of `[00:12.34]` to the user.
150///
151/// The `.lrc` metadata headers (`[ar:…]`, `[ti:…]`, `[by:…]`) are dropped: they
152/// repeat what the tags already say, and this file is not where the artist's
153/// name is settled. `[offset:…]` is applied, since it exists precisely to shift
154/// a file that was timed against another encoding. An unrepresentable timestamp
155/// remains literal, untimed text rather than hiding the line or overflowing.
156/// Timed blank lines are kept, including at the end, so a player can clear the
157/// preceding verse. Only untimed blank edges and an initial UTF-8 BOM are removed.
158///
159/// Repeated timestamps share a text budget of `max(1 MiB, 4 * text.len())`
160/// bytes across the entire result. A line that cannot fit its expanded copies
161/// remains one untimed literal line, preserving its words and timestamps.
162pub fn parse(text: &str) -> Vec<Line> {
163    parse_with_budget(text, MIN_TEXT_BUDGET.max(text.len().saturating_mul(4)))
164}

166/// A complete lyrics result would exceed the permitted UTF-8 text budget.
167#[derive(Debug, Clone, Copy, PartialEq, Eq)]
168pub struct ParseLimitExceeded;
169
170impl std::fmt::Display for ParseLimitExceeded {
171    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
172        formatter.write_str("complete lyrics exceed the text budget")
173    }
174}
175
176impl std::error::Error for ParseLimitExceeded {}

178/// Parses lyrics completely, refusing excessive timestamp expansion.
179///
180/// Unlike [`parse`], this does not turn an over-budget timed chorus into
181/// literal untimed text. This distinction lets an API promise complete timing
182/// rather than silently downgrade it. Malformed timestamps remain literal
183/// text under the same rules as the ordinary parser.
184///
185/// `text_budget` limits the combined UTF-8 bytes in all returned line texts,
186/// including every expanded repetition; it does not include vector storage,
187/// line separators or a transport's serialization overhead. Callers reading
188/// untrusted sources must also bound their input and encoded response size.
189pub fn parse_complete(text: &str, text_budget: usize) -> Result<Vec<Line>, ParseLimitExceeded> {
190    let (lines, exceeded) = parse_bounded(text, text_budget, true);
191    if exceeded {
192        Err(ParseLimitExceeded)
193    } else {
194        Ok(lines)
195    }
196}

198fn parse_with_budget(text: &str, budget: usize) -> Vec<Line> {
199    parse_bounded(text, budget, false).0
200}

202fn parse_bounded(text: &str, budget: usize, complete: bool) -> (Vec<Line>, bool) {
203    let text = text.strip_prefix('\u{feff}').unwrap_or(text);
204    let mut offset_ms: i64 = 0;
205    let mut lines = Vec::new();
206    let mut used_text = 0usize;
207    // Reserve the raw spelling of later lines, so an accepted chorus cannot
208    // consume the space needed to retain subsequent words and timestamps.
209    let mut remaining_raw = text
210        .lines()
211        .fold(0usize, |bytes, line| bytes.saturating_add(line.len()));
212    for raw in text.lines() {
213        remaining_raw = remaining_raw.saturating_sub(raw.len());
214        let raw = raw.trim_end_matches('\r');
215        if let Some(value) = header(raw, "offset") {
216            offset_ms = value.trim().parse().unwrap_or(0);
217            continue;
218        }
219        if is_header(raw) {
220            continue;
221        }
222        let (times, rest) = timestamps(raw);
223        if times.is_empty() {
224            let text = rest.trim_end().to_string();
225            // A blank line inside lyrics is a verse break and worth keeping;
226            // one at the very start is just the file's shape.
227            if !text.is_empty() || !lines.is_empty() {
228                used_text = used_text.saturating_add(text.len());
229                if complete && used_text > budget {
230                    return (Vec::new(), true);
231                }
232                lines.push(Line { at_ms: None, text });
233            }
234            continue;
235        }
236        // `[00:12.00][01:44.00] the chorus again` is one line sung twice, and
237        // the file means both: a chorus that only appeared once would leave a
238        // player silent at its second turn.
239        let text = rest.trim();
240        let expanded = text.len().saturating_mul(times.len());
241        let available = match complete {
242            true => budget.saturating_sub(used_text),
243            false => budget
244                .saturating_sub(used_text)
245                .saturating_sub(remaining_raw),
246        };
247        if expanded > available {
248            if complete {
249                return (Vec::new(), true);
250            }
251            used_text = used_text.saturating_add(raw.len());
252            lines.push(Line {
253                at_ms: None,
254                text: raw.to_owned(),
255            });
256            continue;
257        }
258        used_text = used_text.saturating_add(expanded);
259        for at in times {
260            lines.push(Line {
261                at_ms: Some(at.saturating_add(offset_ms).max(0) as u64),
262                text: text.to_owned(),
263            });
264        }
265    }
266    while lines
267        .last()
268        .is_some_and(|line| line.at_ms.is_none() && line.text.is_empty())
269    {
270        lines.pop();
271    }
272    (lines, false)
273}

275/// The value of an `.lrc` header such as `[offset:+250]`, if this is one.
276fn header<'a>(line: &'a str, name: &str) -> Option<&'a str> {
277    let inner = line.trim().strip_prefix('[')?.strip_suffix(']')?;
278    let (key, value) = inner.split_once(':')?;
279    (key.trim().eq_ignore_ascii_case(name)).then_some(value)
280}
281
282/// `true` for the metadata lines of an `.lrc`: `[ar:…]`, `[ti:…]`, `[length:…]`.
283fn is_header(line: &str) -> bool {
284    let Some(inner) = line
285        .trim()
286        .strip_prefix('[')
287        .and_then(|rest| rest.strip_suffix(']'))
288    else {
289        return false;
290    };
291    let Some((key, _)) = inner.split_once(':') else {
292        return false;
293    };
294    // A timestamp is `[mm:ss…]`, so a key of digits is not a header.
295    !key.is_empty() && key.chars().all(|c| c.is_ascii_alphabetic())
296}

298/// Every timestamp at the head of a line, and what follows them.
299fn timestamps(line: &str) -> (Vec<i64>, &str) {
300    let mut rest = line;
301    let mut found = Vec::new();
302    loop {
303        let trimmed = rest.trim_start();
304        let Some(close) = trimmed.find(']') else {
305            break;
306        };
307        if !trimmed.starts_with('[') {
308            break;
309        }
310        let Some(at) = moment(&trimmed[1..close]) else {
311            break;
312        };
313        found.push(at);
314        rest = &trimmed[close + 1..];
315    }
316    (found, rest)
317}

319/// `mm:ss`, `mm:ss.cc` or `mm:ss.mmm`, in milliseconds.
320fn moment(inner: &str) -> Option<i64> {
321    let (minutes, rest) = inner.split_once(':')?;
322    let minutes: i64 = minutes.trim().parse().ok()?;
323    let (seconds, fraction) = match rest.split_once(['.', ':']) {
324        Some((seconds, fraction)) => (seconds, fraction),
325        None => (rest, ""),
326    };
327    let seconds: i64 = seconds.trim().parse().ok()?;
328    // Two digits are hundredths, three are milliseconds — both are written.
329    let digits: String = fraction
330        .chars()
331        .take_while(|c| c.is_ascii_digit())
332        .collect();
333    let fraction: i64 = match digits.len() {
334        0 => 0,
335        1 => digits.parse::<i64>().ok()? * 100,
336        2 => digits.parse::<i64>().ok()? * 10,
337        _ => digits[..3].parse().ok()?,
338    };
339    minutes
340        .checked_mul(60_000)?
341        .checked_add(seconds.checked_mul(1_000)?)?
342        .checked_add(fraction)
343}
