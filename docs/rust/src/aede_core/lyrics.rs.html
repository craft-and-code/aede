1//! Lyrics that are already there: in the tags, or in an `.lrc` beside the file.
2//!
3//! Nothing here touches the network. Reading what a library already holds is
4//! the first half of the job and the cheap one — the parsers walk past `USLT`,
5//! `LYRICS` and `©lyr` today, and a `.lrc` sitting next to a track is a text
6//! file anyone can open. Fetching what is *missing* is another matter
7//! entirely: it needs a network, a source whose terms allow it, and an explicit
8//! choice, because lyrics are the composition's copyright and owning a FLAC
9//! grants no rights in it. See `docs/design/lyrics.md`.
10//!
11//! **Where they live follows the rule the rest of the catalog follows.** Tag
12//! lyrics are already in it, because raw tags are kept per file; a sidecar is
13//! not in the file, so pretending it were a tag would make the catalog lie
14//! about what the file says. The catalog therefore stores the sidecar's
15//! **path**, and the text is read from it when somebody asks — it is one small
16//! file, sitting right next to the music it belongs to.
17
18use std::path::{Path, PathBuf};
19
20/// Extension of a lyrics file beside a track.
21pub const EXTENSION: &str = "lrc";
22
23/// At most this much of a lyrics file is read.
24///
25/// Lyrics are a page of text. A ten-megabyte `.lrc` is not a song, and reading
26/// it whole to show a track page would be paying for somebody's mistake.
27const LIMIT: usize = 256 * 1024;
28
29/// One line, with the moment it is sung when the file says so.
30#[derive(Debug, Clone, PartialEq)]
31pub struct Line {
32    /// Milliseconds from the start of the track, or `None` on a plain line.
33    ///
34    /// Kept at read time rather than recovered later: M3 needs a playhead to
35    /// follow, and asking it to re-read every file would be asking twice for
36    /// something already in hand.
37    pub at_ms: Option<u64>,
38    /// The line as written, without its timestamp.
39    pub text: String,
40}
41
42/// Where a track's lyrics were found.
43#[derive(Debug, Clone, Copy, PartialEq, Eq)]
44pub enum Source {
45    /// A tag inside the audio file itself.
46    Tag,
47    /// A `.lrc` file beside it.
48    Sidecar,
49}
50
51/// The lyrics of one track, as read.
52#[derive(Debug, Clone, PartialEq)]
53pub struct Lyrics {
54    /// Where they came from, which the page names so the reader knows.
55    pub source: Source,
56    /// The file they were read from: the audio itself, or the sidecar.
57    pub origin: String,
58    /// Every line, in order.
59    pub lines: Vec<Line>,
60}
61
62impl Lyrics {
63    /// `true` when at least one line carries a time.
64    ///
65    /// Not "every line": real `.lrc` files carry untimed headers, blank lines
66    /// and the occasional stray, and a file that is 95% timed is a synced file.
67    pub fn synced(&self) -> bool {
68        self.lines.iter().any(|line| line.at_ms.is_some())
69    }
70
71    /// The whole text, timestamps dropped — what a search looks through.
72    pub fn text(&self) -> String {
73        self.lines
74            .iter()
75            .map(|line| line.text.as_str())
76            .collect::<Vec<_>>()
77            .join("\n")
78    }
79}

81/// Where the sidecar of an audio file would be: the same name, `.lrc`.
82pub fn sidecar_of(audio: &Path) -> PathBuf {
83    audio.with_extension(EXTENSION)
84}
85
86/// `true` when a file name is a lyrics sidecar, whatever its case.
87pub fn is_sidecar(name: &str) -> bool {
88    name.to_ascii_lowercase().ends_with(".lrc")
89}

91/// Reads a sidecar, bounded, and never fatally.
92///
93/// Invalid UTF-8 is replaced rather than refused. A `.lrc` written on a Windows
94/// machine in 2003 is Latin-1 as often as not, and refusing to show a song
95/// because of one accent would be the wrong trade.
96pub fn read(path: &Path) -> Option<Lyrics> {
97    let raw = std::fs::read(path).ok()?;
98    let text = String::from_utf8_lossy(&raw[..raw.len().min(LIMIT)]).to_string();
99    let lines = parse(&text);
100    match lines.is_empty() {
101        true => None,
102        false => Some(Lyrics {
103            source: Source::Sidecar,
104            origin: path.to_string_lossy().to_string(),
105            lines,
106        }),
107    }
108}

110/// The lyrics a tag carries, if it carries any.
111pub fn from_tag(origin: &str, tag: &str) -> Option<Lyrics> {
112    let lines = parse(tag);
113    match lines.is_empty() {
114        true => None,
115        false => Some(Lyrics {
116            source: Source::Tag,
117            origin: origin.to_string(),
118            lines,
119        }),
120    }
121}

123/// Reads lyrics text, timed or not.
124///
125/// One parser for both, because a tag can perfectly well hold LRC — plenty of
126/// taggers write the synced text straight into `LYRICS`, and a reader that only
127/// understood plain text would show a page of `[00:12.34]` to the user.
128///
129/// The `.lrc` metadata headers (`[ar:…]`, `[ti:…]`, `[by:…]`) are dropped: they
130/// repeat what the tags already say, and this file is not where the artist's
131/// name is settled. `[offset:…]` is applied, since it exists precisely to shift
132/// a file that was timed against another encoding.
133pub fn parse(text: &str) -> Vec<Line> {
134    let mut offset_ms: i64 = 0;
135    let mut lines = Vec::new();
136    for raw in text.lines() {
137        let raw = raw.trim_end_matches('\r');
138        if let Some(value) = header(raw, "offset") {
139            offset_ms = value.trim().parse().unwrap_or(0);
140            continue;
141        }
142        if is_header(raw) {
143            continue;
144        }
145        let (times, rest) = timestamps(raw);
146        if times.is_empty() {
147            let text = rest.trim_end().to_string();
148            // A blank line inside lyrics is a verse break and worth keeping;
149            // one at the very start is just the file's shape.
150            if !text.is_empty() || !lines.is_empty() {
151                lines.push(Line { at_ms: None, text });
152            }
153            continue;
154        }
155        // `[00:12.00][01:44.00] the chorus again` is one line sung twice, and
156        // the file means both: a chorus that only appeared once would leave a
157        // player silent at its second turn.
158        for at in times {
159            lines.push(Line {
160                at_ms: Some(at.saturating_add(offset_ms).max(0) as u64),
161                text: rest.trim().to_string(),
162            });
163        }
164    }
165    while lines.last().is_some_and(|l| l.text.is_empty()) {
166        lines.pop();
167    }
168    lines
169}

171/// The value of an `.lrc` header such as `[offset:+250]`, if this is one.
172fn header<'a>(line: &'a str, name: &str) -> Option<&'a str> {
173    let inner = line.trim().strip_prefix('[')?.strip_suffix(']')?;
174    let (key, value) = inner.split_once(':')?;
175    (key.trim().eq_ignore_ascii_case(name)).then_some(value)
176}
177
178/// `true` for the metadata lines of an `.lrc`: `[ar:…]`, `[ti:…]`, `[length:…]`.
179fn is_header(line: &str) -> bool {
180    let Some(inner) = line
181        .trim()
182        .strip_prefix('[')
183        .and_then(|rest| rest.strip_suffix(']'))
184    else {
185        return false;
186    };
187    let Some((key, _)) = inner.split_once(':') else {
188        return false;
189    };
190    // A timestamp is `[mm:ss…]`, so a key of digits is not a header.
191    !key.is_empty() && key.chars().all(|c| c.is_ascii_alphabetic())
192}

194/// Every timestamp at the head of a line, and what follows them.
195fn timestamps(line: &str) -> (Vec<i64>, &str) {
196    let mut rest = line;
197    let mut found = Vec::new();
198    loop {
199        let trimmed = rest.trim_start();
200        let Some(close) = trimmed.find(']') else {
201            break;
202        };
203        if !trimmed.starts_with('[') {
204            break;
205        }
206        let Some(at) = moment(&trimmed[1..close]) else {
207            break;
208        };
209        found.push(at);
210        rest = &trimmed[close + 1..];
211    }
212    (found, rest)
213}

215/// `mm:ss`, `mm:ss.cc` or `mm:ss.mmm`, in milliseconds.
216fn moment(inner: &str) -> Option<i64> {
217    let (minutes, rest) = inner.split_once(':')?;
218    let minutes: i64 = minutes.trim().parse().ok()?;
219    let (seconds, fraction) = match rest.split_once(['.', ':']) {
220        Some((seconds, fraction)) => (seconds, fraction),
221        None => (rest, ""),
222    };
223    let seconds: i64 = seconds.trim().parse().ok()?;
224    // Two digits are hundredths, three are milliseconds — both are written.
225    let digits: String = fraction
226        .chars()
227        .take_while(|c| c.is_ascii_digit())
228        .collect();
229    let fraction: i64 = match digits.len() {
230        0 => 0,
231        1 => digits.parse::<i64>().ok()? * 100,
232        2 => digits.parse::<i64>().ok()? * 10,
233        _ => digits[..3].parse().ok()?,
234    };
235    Some(minutes * 60_000 + seconds * 1_000 + fraction)
236}
