1//! Walks the FLAC frame structure without decoding any audio.
2//!
3//! Two claims a file makes about itself cannot be checked from its header
4//! alone: how many bits of resolution it really carries, and whether its two
5//! channels hold different sound. Both answers are already written in the
6//! encoded stream, and reading them costs far less than decoding.
7//!
8//! **Resolution.** When the low bits of every sample are zero — the signature
9//! of 16-bit audio re-encoded as 24-bit — the encoder does not store them. It
10//! records a *wasted bits* count in each subframe instead. A file claiming 24
11//! bits with eight wasted bits everywhere carries 16 bits of real music.
12//!
13//! **Channels.** For stereo, the encoder picks the cheapest of four
14//! correlations per frame. When both channels are identical the difference
15//! channel is constant zero, which shows up as a `CONSTANT` subframe.
16//!
17//! Neither check can accuse a file wrongly. An encoder that skipped these
18//! optimisations yields "unknown", never a false positive.
19//!
20//! Reference: <https://xiph.org/flac/format.html>
21

28/// How much of a file the walk is allowed to read.
29///
30/// Walking costs about 80 MB/s, so an unbounded pass over a large library is
31/// out of the question. Padding is a property of the whole file and shows up
32/// in the first frames; identical channels, in principle, might not.
33#[derive(Debug, Clone, Copy, PartialEq, Eq)]
34pub struct Limits {
35    /// Bytes of encoded audio to read at most.
36    pub max_bytes: usize,
37    /// Frames to walk at most.
38    pub max_frames: usize,
39}
40
41impl Limits {
42    /// Enough to answer confidently while staying cheap over a whole library:
43    /// 512 frames is roughly 47 seconds of CD-quality audio.
44    pub fn quick() -> Limits {
45        Limits {
46            max_bytes: 8 * 1024 * 1024,
47            max_frames: 512,
48        }
49    }
50
51    /// Reads the file to its end. For inspecting one file on demand.
52    pub fn thorough() -> Limits {
53        Limits {
54            max_bytes: usize::MAX,
55            max_frames: usize::MAX,
56        }
57    }
58}
59
60impl Default for Limits {
61    fn default() -> Limits {
62        Limits::quick()
63    }
64}

66/// What the frame structure says about the audio, beyond what the header
67/// claims.
68#[derive(Debug, Clone, PartialEq)]
69pub struct FlacAudit {
70    /// Bit depth written in STREAMINFO.
71    pub declared_bit_depth: u16,
72    /// Bit depth actually carried, once the wasted low bits are removed.
73    ///
74    /// Lower than `declared_bit_depth` means the file was padded: 24-bit
75    /// packaging around 16-bit music.
76    pub effective_bit_depth: u16,
77    /// Channel count written in STREAMINFO.
78    pub channels: u16,
79    /// What the two channels of a stereo file actually hold.
80    pub stereo: StereoContent,
81    /// Whether every subframe examined was constant, which means silence.
82    pub digital_silence: bool,
83    /// Number of frames the walk covered.
84    pub frames_examined: usize,
85    /// `true` when a limit stopped the walk before the end of the file.
86    ///
87    /// The verdict then describes the portion examined, not the whole file.
88    pub truncated: bool,
89}
90
91impl FlacAudit {
92    /// `true` when the file claims more resolution than it carries.
93    pub fn is_padded(&self) -> bool {
94        self.effective_bit_depth < self.declared_bit_depth
95    }
96}

98/// What a stereo file's two channels hold relative to one another.
99#[derive(Debug, Clone, Copy, PartialEq, Eq)]
100pub enum StereoContent {
101    /// Not a two-channel file, or the encoder gave nothing away.
102    Unknown,
103    /// The channels differ: real stereo.
104    Independent,
105    /// Both channels are identical: mono in a stereo container.
106    Duplicated,
107}
108
109/// Walks the frames of a FLAC file and reports what they reveal.
110pub fn audit(path: &Path, limits: Limits) -> Result<FlacAudit, TagError> {
111    let mut file = std::fs::File::open(path)?;
112    let (stream_info, audio_start) = read_stream_info(&mut file)?;
113
114    let budget = match limits.max_bytes {
115        usize::MAX => file.metadata()?.len().saturating_sub(audio_start) as usize,
116        bytes => bytes,
117    };
118    let data = read_at_most(&mut file, audio_start, budget)?;
119    let reached_end_of_file = audio_start + data.len() as u64 >= file.metadata()?.len();
120    let mut reader = BitReader::new(&data);
121
122    let mut min_wasted = u32::MAX;
123    let mut frames = 0usize;
124    let mut all_constant = true;
125    let mut duplicated_frames = 0usize;
126    let mut stereo_frames = 0usize;
127
128    while let Some(frame) = read_frame(&mut reader, &stream_info) {
129        frames += 1;
130        min_wasted = min_wasted.min(frame.min_wasted);
131        all_constant &= frame.all_constant;
132        if frame.is_stereo {
133            stereo_frames += 1;
134            if frame.channels_identical {
135                duplicated_frames += 1;
136            }
137        }
138        // Frames are byte aligned; the next sync code starts on a boundary.
139        reader.align_to_byte();
140        if reader.is_exhausted() || frames >= limits.max_frames {
141            break;
142        }
143    }
144
145    let wasted = if frames == 0 || min_wasted == u32::MAX {
146        0
147    } else {
148        min_wasted
149    };
150    let effective = (stream_info.bits_per_sample as u32).saturating_sub(wasted);
151
152    let stereo = if stream_info.channels != 2 || stereo_frames == 0 {
153        StereoContent::Unknown
154    } else if duplicated_frames == stereo_frames {
155        StereoContent::Duplicated
156    } else {
157        StereoContent::Independent
158    };
159
160    Ok(FlacAudit {
161        declared_bit_depth: stream_info.bits_per_sample,
162        effective_bit_depth: effective.max(1) as u16,
163        channels: stream_info.channels,
164        stereo,
165        digital_silence: frames > 0 && all_constant,
166        frames_examined: frames,
167        truncated: !reached_end_of_file || frames >= limits.max_frames,
168    })
169}

171/// The few STREAMINFO fields the frame walk needs.
172pub(crate) struct StreamInfo {
173    pub(crate) bits_per_sample: u16,
174    pub(crate) channels: u16,
175}

177/// Reads STREAMINFO and returns it with the offset of the first frame.
178pub(crate) fn read_stream_info(file: &mut std::fs::File) -> Result<(StreamInfo, u64), TagError> {
179    let start = crate::tags::id3::skip_id3v2(file)?;
180    let header = read_at_most(file, start, 4)?;
181    if header.len() < 4 || &header[..4] != b"fLaC" {
182        return Err(TagError::UnrecognizedFormat);
183    }
184
185    let mut offset = start + 4;
186    let mut info = None;
187    loop {
188        let head = read_at_most(file, offset, 4)?;
189        if head.len() < 4 {
190            return Err(TagError::Malformed("metadata block header truncated"));
191        }
192        let last = head[0] & 0x80 != 0;
193        let kind = head[0] & 0x7F;
194        let length = ((head[1] as u64) << 16) | ((head[2] as u64) << 8) | head[3] as u64;
195        offset += 4;
196
197        if kind == 0 {
198            let body = read_at_most(file, offset, length.min(34) as usize)?;
199            if body.len() < 18 {
200                return Err(TagError::Malformed("STREAMINFO truncated"));
201            }
202            let packed = u64::from_be_bytes([
203                body[10], body[11], body[12], body[13], body[14], body[15], body[16], body[17],
204            ]);
205            info = Some(StreamInfo {
206                channels: (((packed >> 41) & 0x07) as u16) + 1,
207                bits_per_sample: (((packed >> 36) & 0x1F) as u16) + 1,
208            });
209        }
210        offset += length;
211        if last {
212            break;
213        }
214    }
215
216    match info {
217        Some(info) => Ok((info, offset)),
218        None => Err(TagError::Malformed("FLAC stream without STREAMINFO")),
219    }
220}

222/// What one frame contributes to the audit.
223pub(crate) struct FrameAudit {
224    min_wasted: u32,
225    all_constant: bool,
226    is_stereo: bool,
227    channels_identical: bool,
228    /// Byte range of the frame within the slice being walked, and where its
229    /// header stops — what a checksum needs, and what the resolution audit
230    /// ignores.
231    pub(crate) span: FrameSpan,
232}

234/// Where a frame sits in the stream, and the checksums it carries.
235pub(crate) struct FrameSpan {
236    /// Offset of the sync code.
237    pub(crate) start: usize,
238    /// Offset just past the frame's own CRC-16.
239    pub(crate) end: usize,
240    /// Offset of the header's CRC-8, which is also where the header stops.
241    pub(crate) header_end: usize,
242    /// The CRC-8 the header announces.
243    pub(crate) stored_crc8: u8,
244    /// The CRC-16 the frame announces.
245    pub(crate) stored_crc16: u16,
246}

248/// Reads one frame header and walks its subframes.
249///
250/// Returns `None` at the end of the stream or on anything unexpected, which
251/// ends the walk without failing the audit.
252pub(crate) fn read_frame(reader: &mut BitReader<'_>, info: &StreamInfo) -> Option<FrameAudit> {
253    let start = reader.bit_pos() / 8;
254    if reader.bits(14)? != 0b11_1111_1111_1110 {
255        return None;
256    }
257    reader.bits(1)?; // reserved
258    reader.bits(1)?; // blocking strategy
259
260    let block_size_bits = reader.bits(4)? as u32;
261    let sample_rate_bits = reader.bits(4)? as u32;
262    let channel_assignment = reader.bits(4)? as u32;
263    let sample_size_bits = reader.bits(3)? as u32;
264    reader.bits(1)?; // reserved
265
266    // Coded frame or sample number, in a UTF-8-like variable length encoding.
267    let first = reader.bits(8)? as u8;
268    let extra = match first {
269        0x00..=0x7F => 0,
270        0xC0..=0xDF => 1,
271        0xE0..=0xEF => 2,
272        0xF0..=0xF7 => 3,
273        0xF8..=0xFB => 4,
274        0xFC..=0xFD => 5,
275        0xFE => 6,
276        _ => return None,
277    };
278    reader.skip(extra * 8)?;
279
280    let block_size = match block_size_bits {
281        0 => return None,
282        1 => 192,
283        2..=5 => 576 << (block_size_bits - 2),
284        6 => reader.bits(8)? as u32 + 1,
285        7 => reader.bits(16)? as u32 + 1,
286        _ => 256 << (block_size_bits - 8),
287    };
288    match sample_rate_bits {
289        12 => {
290            reader.bits(8)?;
291        }
292        13 | 14 => {
293            reader.bits(16)?;
294        }
295        15 => return None,
296        _ => {}
297    }
298    let header_end = reader.bit_pos() / 8;
299    let stored_crc8 = reader.bits(8)? as u8;
300
301    let bits_per_sample = match sample_size_bits {
302        0 => info.bits_per_sample as u32,
303        1 => 8,
304        2 => 12,
305        4 => 16,
306        5 => 20,
307        6 => 24,
308        7 => 32,
309        _ => return None,
310    };
311
312    // Channel assignments 8, 9 and 10 store a difference channel, which needs
313    // one extra bit and tells us whether the channels are identical.
314    let (channel_count, side_index) = match channel_assignment {
315        0..=7 => (channel_assignment + 1, None),
316        8 => (2, Some(1)),
317        9 => (2, Some(0)),
318        10 => (2, Some(1)),
319        _ => return None,
320    };
321    // Mid/side stores (left + right) >> 1, and that shift eats one of the
322    // trailing zero bits. Without putting it back, a padded file looks one bit
323    // richer than it is.
324    let mid_index = if channel_assignment == 10 {
325        Some(0)
326    } else {
327        None
328    };
329
330    let mut min_wasted = u32::MAX;
331    let mut all_constant = true;
332    let mut side_is_zero = false;
333
334    for channel in 0..channel_count {
335        let extra_bit = side_index == Some(channel);
336        let bps = bits_per_sample + if extra_bit { 1 } else { 0 };
337        let sub = read_subframe(reader, block_size, bps)?;
338        // A constant subframe says nothing about resolution: the encoder has
339        // no reason to record wasted bits on a value it stores once. Counting
340        // it would drag the minimum to zero and hide a padded file.
341        if sub.constant_value.is_none() {
342            let corrected = sub.wasted + if mid_index == Some(channel) { 1 } else { 0 };
343            min_wasted = min_wasted.min(corrected);
344        }
345        all_constant &= sub.constant_value == Some(0);
346        if extra_bit && sub.constant_value == Some(0) {
347            side_is_zero = true;
348        }
349    }
350
351    reader.align_to_byte();
352    let stored_crc16 = reader.bits(16)? as u16;
353
354    Some(FrameAudit {
355        min_wasted,
356        all_constant,
357        is_stereo: side_index.is_some(),
358        channels_identical: side_is_zero,
359        span: FrameSpan {
360            start,
361            end: reader.bit_pos() / 8,
362            header_end,
363            stored_crc8,
364            stored_crc16,
365        },
366    })
367}

369/// What one subframe contributes.
370struct SubframeAudit {
371    wasted: u32,
372    /// The value, when the subframe is a constant one.
373    constant_value: Option<i64>,
374}
375
376/// Reads a subframe header and steps over its data.
377///
378/// The residual is parsed but not reconstructed: Rice codes are
379/// variable-length, so they must be read to be skipped, but nothing needs to
380/// be computed from them.
381fn read_subframe(
382    reader: &mut BitReader<'_>,
383    block_size: u32,
384    bits_per_sample: u32,
385) -> Option<SubframeAudit> {
386    if reader.bits(1)? != 0 {
387        return None; // padding bit must be zero
388    }
389    let kind = reader.bits(6)? as u32;
390    let has_wasted = reader.bits(1)? == 1;
391    let wasted = if has_wasted { reader.unary()? + 1 } else { 0 };
392
393    let bps = bits_per_sample.checked_sub(wasted)?;
394    if bps == 0 || bps > 33 {
395        return None;
396    }
397
398    let mut constant_value = None;
399    match kind {
400        0 => {
401            constant_value = Some(reader.signed_bits(bps)?);
402        }
403        1 => {
404            reader.skip(block_size as usize * bps as usize)?;
405        }
406        8..=12 => {
407            let order = kind - 8;
408            reader.skip(order as usize * bps as usize)?;
409            skip_residual(reader, block_size, order)?;
410        }
411        32..=63 => {
412            let order = kind - 31;
413            reader.skip(order as usize * bps as usize)?;
414            let precision = reader.bits(4)? as u32 + 1;
415            if precision > 15 {
416                return None; // 0b1111 is invalid
417            }
418            reader.bits(5)?; // quantisation shift
419            reader.skip(order as usize * precision as usize)?;
420            skip_residual(reader, block_size, order)?;
421        }
422        _ => return None, // reserved
423    }
424
425    Some(SubframeAudit {
426        wasted,
427        constant_value,
428    })
429}

431/// Steps over a residual, partition by partition.
432fn skip_residual(reader: &mut BitReader<'_>, block_size: u32, order: u32) -> Option<()> {
433    let method = reader.bits(2)?;
434    let param_bits = match method {
435        0 => 4,
436        1 => 5,
437        _ => return None, // reserved
438    };
439    let escape = (1u64 << param_bits) - 1;
440
441    let partition_order = reader.bits(4)? as u32;
442    let partitions = 1u32 << partition_order;
443    if !block_size.is_multiple_of(partitions) {
444        return None;
445    }
446
447    for partition in 0..partitions {
448        let mut samples = block_size >> partition_order;
449        if partition == 0 {
450            samples = samples.checked_sub(order)?;
451        }
452        let parameter = reader.bits(param_bits)?;
453        if parameter == escape {
454            let raw_bits = reader.bits(5)? as usize;
455            reader.skip(samples as usize * raw_bits)?;
456        } else {
457            for _ in 0..samples {
458                reader.unary()?;
459                reader.bits(parameter as u32)?;
460            }
461        }
462    }
463    Some(())
