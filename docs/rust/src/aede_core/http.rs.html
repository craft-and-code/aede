1//! The one place this program reaches the network.
2//!
3//! Behind the `fetch` feature, so that a build without it has no way to make a
4//! request at all — a property the compiler enforces rather than a promise a
5//! README makes.
6//!
7//! It does three things and refuses to do more: it waits its turn, it names
8//! itself, and it hands back parsed JSON. Everything about *what* to ask and
9//! *what to do with the answer* lives in [`crate::musicbrainz`], which has no
10//! socket and is therefore testable.
11//!
12//! **Waiting its turn is not politeness, it is the contract.** MusicBrainz
13//! allows one request per second per address and answers `503` to everything
14//! once that is exceeded — not just to the request that went over. A client
15//! without a throttle does not run slightly too fast; it stops working, and it
16//! does so for every other program on the same address.
17
18use crate::json::Json;
19use std::time::{Duration, Instant};
20
21/// How long to wait for one answer before giving up on it.
22const TIMEOUT: Duration = Duration::from_secs(25);
23
24/// The largest body this client will read into memory.
25///
26/// Cover art is the only thing here that is not a small JSON document, and the
27/// original upload of one is a few megabytes at most. A ceiling turns a wrong
28/// address — a redirect gone astray, a service handing back something else
29/// entirely — into a refusal rather than a machine filling its memory.
30const MAX_BODY: u64 = 32 * 1024 * 1024;
31
32/// What can go wrong, kept apart because the answers differ.
33#[derive(Debug, Clone, PartialEq, Eq)]
34pub enum Error {
35    /// The service asked us to slow down. Not a failure of the request: a
36    /// statement about the rate, and the only sane response is to stop.
37    RateLimited,
38    /// The service answered, with something other than success.
39    Status(u16),
40    /// Nothing came back: no route, no name, a timeout.
41    Network(String),
42    /// Something came back, and it was not the JSON that was asked for.
43    NotJson(String),
44}
45
46impl std::fmt::Display for Error {
47    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
48        match self {
49            Error::RateLimited => write!(
50                f,
51                "the service is refusing requests because too many were sent \
52                 (one per second is the limit); nothing was lost, try later"
53            ),
54            Error::Status(code) => write!(f, "the service answered {code}"),
55            Error::Network(detail) => write!(f, "could not reach the service: {detail}"),
56            Error::NotJson(detail) => write!(f, "the answer was not readable: {detail}"),
57        }
58    }
59}
60
61impl std::error::Error for Error {}
62
63/// A client that waits its turn and says who it is.
64pub struct Client {
65    agent: ureq::Agent,
66    user_agent: String,
67    interval: Duration,
68    last: Option<Instant>,
69}
70
71impl Client {
72    /// A client for one service.
73    ///
74    /// `user_agent` is not decoration: MusicBrainz **requires** a descriptive
75    /// one carrying a way to contact whoever wrote the program, and throttles
76    /// anonymous or generic callers as a single shared pool — so a library
77    /// that sent `ureq/3.4` would be sharing a rate limit with every other
78    /// program that could not be bothered either. Build it with
79    /// [`Client::identify`].
80    pub fn new(user_agent: impl Into<String>, interval: Duration) -> Client {
81        let config = ureq::Agent::config_builder()
82            .timeout_global(Some(TIMEOUT))
83            .build();
84        Client {
85            agent: config.into(),
86            user_agent: user_agent.into(),
87            interval,
88            last: None,
89        }
90    }
91
92    /// The `User-Agent` MusicBrainz asks for: `name/version ( contact )`.
93    pub fn identify(name: &str, version: &str, contact: &str) -> String {
94        format!("{name}/{version} ( {contact} )")
95    }
96
97    /// Fetches a URL and parses the answer as JSON.
98    ///
99    /// Blocks for as long as the rate requires before sending. That is the
100    /// whole point: a caller that could forget to wait is a caller that will.
101    pub fn get_json(&mut self, url: &str) -> Result<Json, Error> {
102        self.wait_turn();
103        let text = self
104            .agent
105            .get(url)
106            .header("User-Agent", &self.user_agent)
107            .header("Accept", "application/json")
108            .call()
109            .map_err(from_ureq)?
110            .body_mut()
111            .read_to_string()
112            .map_err(|e| Error::Network(e.to_string()))?;
113        crate::json::parse(&text).map_err(|e| Error::NotJson(e.to_string()))
114    }
115
116    /// Fetches a URL and hands back the body as it came.
117    ///
118    /// The one thing this client downloads that is not an answer to a question:
119    /// an image. It waits its turn like everything else.
120    ///
121    /// The size ceiling is the client library's own — it refuses an oversized
122    /// body rather than reading it — and the ceiling below is a second line
123    /// behind it, checked on what actually arrived. Two guards rather than one
124    /// because the first belongs to a dependency whose defaults may change, and
125    /// the failure it prevents is a machine filling its memory from a wrong
126    /// address.
127    pub fn get_bytes(&mut self, url: &str) -> Result<Vec<u8>, Error> {
128        self.wait_turn();
129        let bytes = self
130            .agent
131            .get(url)
132            .header("User-Agent", &self.user_agent)
133            .call()
134            .map_err(from_ureq)?
135            .body_mut()
136            .read_to_vec()
137            .map_err(|e| Error::Network(e.to_string()))?;
138        if bytes.len() as u64 > MAX_BODY {
139            return Err(Error::Network(format!(
140                "the answer was larger than {} MB, so it was not kept",
141                MAX_BODY / (1024 * 1024)
142            )));
143        }
144        Ok(bytes)
145    }
146
147    /// Sleeps until the next request is allowed.
148    fn wait_turn(&mut self) {
149        if let Some(last) = self.last {
150            let since = last.elapsed();
151            if since < self.interval {
152                std::thread::sleep(self.interval - since);
153            }
154        }
155        self.last = Some(Instant::now());
156    }
157
158    /// How long the next [`Client::get_json`] would wait before sending.
159    ///
160    /// Exposed so a command can tell the user that asking about six hundred
161    /// artists will take ten minutes *before* it starts, rather than leaving
162    /// them to work it out from a progress line.
163    pub fn interval(&self) -> Duration {
164        self.interval
165    }
166}
167
168/// Translates the client library's errors into this module's four cases.
169fn from_ureq(error: ureq::Error) -> Error {
170    match error {
171        // 503 is what MusicBrainz answers when the rate is exceeded, and it
172        // answers it to everything until the rate drops — so it has to be told
173        // apart from an ordinary failure, which a retry could survive.
174        ureq::Error::StatusCode(503) => Error::RateLimited,
175        ureq::Error::StatusCode(code) => Error::Status(code),
176        other => Error::Network(other.to_string()),
177    }
178}
