1//! Bounded validation before artwork is published.
2//!
3//! The codecs decode pixels; the container guards require a complete, single
4//! image and bound their work before allocation. JPEG has no integrity checksum:
5//! a changed stream that remains decodable, including some streams repaired
6//! with an early EOI marker, cannot be certified as the original image.
7
8use std::io::{self, BufRead, Cursor, Read, Seek, SeekFrom};
9use std::time::{Duration, Instant};
10
11const MAX_BYTES: usize = 32 * 1024 * 1024;
12const MAX_AXIS: usize = 8192;
13const MAX_PIXELS: usize = 16_000_000;
14const MAX_OUTPUT: usize = 128 * 1024 * 1024;
15const MAX_PARTS: usize = 16_384;
16const MAX_SCANS: usize = 64;
17const TIME_BUDGET: Duration = Duration::from_secs(5);

23/// Validates all image rows before any filesystem mutation.
24///
25/// PNG checks include every chunk CRC, IDAT Adler-32 and the final IEND.
26/// Animated PNG is refused rather than validating only its default frame.
27/// JPEG uses the codec's strict mode and requires the final EOI. Input size,
28/// dimensions, pixel count and scan/chunk counts bound the work. PNG's internal
29/// allocation limit is best effort; the output budget is not a total RSS cap.
30/// The deadline is cooperative at reads and PNG rows, with a final check before
31/// publication; it cannot interrupt a codec's computation between those points.
32pub(crate) fn validate(bytes: &[u8], format: &str) -> Result<(), String> {
33    validate_with_budget(bytes, format, TIME_BUDGET)
34}
35
36fn validate_with_budget(bytes: &[u8], format: &str, budget: Duration) -> Result<(), String> {
37    if bytes.len() > MAX_BYTES {
38        return Err("image exceeds the 32 MiB download limit".into());
39    }
40    let started = Instant::now();
41    check_time(started, budget)?;
42    match format {
43        "jpg" => validate_jpeg(bytes, started, budget)?,
44        "png" => validate_png(bytes, started, budget)?,
45        _ => return Err("unsupported image format".into()),
46    }
47    check_time(started, budget)
48}
49
50fn check_time(started: Instant, budget: Duration) -> Result<(), String> {
51    if started.elapsed() >= budget {
52        Err("image validation exceeded its time budget".into())
53    } else {
54        Ok(())
55    }
56}

58fn dimensions(width: usize, height: usize) -> Result<(), String> {
59    if width == 0
60        || height == 0
61        || width > MAX_AXIS
62        || height > MAX_AXIS
63        || width.checked_mul(height).is_none_or(|n| n > MAX_PIXELS)
64    {
65        Err("image exceeds the limits of 8192 pixels per axis and 16 million pixels; choose a smaller Cover Art Archive --size if needed".into())
66    } else {
67        Ok(())
68    }
69}

71fn png_container(bytes: &[u8], started: Instant, budget: Duration) -> Result<(), String> {
72    let mut at = 8;
73    for part in 0..MAX_PARTS {
74        check_time(started, budget)?;
75        let header = bytes.get(at..at + 8).ok_or("truncated PNG chunk")?;
76        let length = u32::from_be_bytes([header[0], header[1], header[2], header[3]]) as usize;
77        let end = at
78            .checked_add(12)
79            .and_then(|at| at.checked_add(length))
80            .filter(|end| *end <= bytes.len())
81            .ok_or("truncated PNG chunk")?;
82        let kind = &header[4..8];
83        if part == 0 {
84            if kind != b"IHDR" || length != 13 {
85                return Err("PNG must begin with an IHDR chunk".into());
86            }
87            let data = &bytes[at + 8..end - 4];
88            let width = u32::from_be_bytes([data[0], data[1], data[2], data[3]]) as usize;
89            let height = u32::from_be_bytes([data[4], data[5], data[6], data[7]]) as usize;
90            dimensions(width, height)?;
91        }
92        if matches!(kind, b"acTL" | b"fcTL" | b"fdAT") {
93            return Err("animated PNG artwork is not supported".into());
94        }
95        if kind == b"IEND" {
96            return if length == 0 && end == bytes.len() {
97                Ok(())
98            } else {
99                Err("invalid PNG end or data after IEND".into())
100            };
101        }
102        at = end;
103    }
104    Err("PNG exceeds the chunk-count limit".into())
105}

107fn validate_png(bytes: &[u8], started: Instant, budget: Duration) -> Result<(), String> {
108    png_container(bytes, started, budget)?;
109    let mut options = png::DecodeOptions::default();
110    options.set_ignore_checksums(false);
111    options.set_skip_ancillary_crc_failures(false);
112    let mut decoder =
113        png::Decoder::new_with_options(TimedCursor::new(bytes, started, budget), options);
114    decoder.set_limits(png::Limits {
115        bytes: 16 * 1024 * 1024,
116    });
117    // Compressed ancillary metadata is not needed for pixel validation and
118    // must not turn a small picture into an unbounded metadata expansion.
119    decoder.set_ignore_text_chunk(true);
120    decoder.set_ignore_iccp_chunk(true);
121    let mut reader = decoder
122        .read_info()
123        .map_err(|e| format!("invalid PNG: {e}"))?;
124    let output = reader
125        .output_buffer_size()
126        .ok_or("PNG output size overflow")?;
127    if output > MAX_OUTPUT {
128        return Err("PNG exceeds the decoded-output budget".into());
129    }
130    if reader.info().color_type == png::ColorType::Indexed {
131        let info = reader.info();
132        let width = info.width as usize;
133        let depth = info.bit_depth as usize;
134        let palette_bytes = info.palette.as_ref().map_or(0, |p| p.len());
135        let entries = palette_bytes / 3;
136        if !matches!(depth, 1 | 2 | 4 | 8)
137            || palette_bytes % 3 != 0
138            || entries == 0
139            || entries > (1 << depth)
140        {
141            return Err("invalid PNG palette size".into());
142        }
143        if info
144            .trns
145            .as_ref()
146            .is_some_and(|alpha| alpha.len() > entries)
147        {
148            return Err("PNG transparency exceeds its palette".into());
149        }
150        // EXPAND fills missing palette entries with black. Validate the raw
151        // indices instead, after the codec has reassembled any Adam7 passes.
152        let mut indices = Vec::new();
153        indices
154            .try_reserve_exact(output)
155            .map_err(|_| "cannot allocate PNG validation buffer")?;
156        indices.resize(output, 0);
157        reader
158            .next_frame(&mut indices)
159            .map_err(|e| format!("invalid PNG: {e}"))?;
160        let row_bytes = (width * depth).div_ceil(8);
161        let mask = (1u16 << depth) - 1;
162        for row in indices.chunks_exact(row_bytes) {
163            check_time(started, budget)?;
164            for pixel in 0..width {
165                let bit = pixel * depth;
166                let index = (u16::from(row[bit / 8]) >> (8 - depth - bit % 8)) & mask;
167                if usize::from(index) >= entries {
168                    return Err("PNG pixel refers to a missing palette entry".into());
169                }
170            }
171        }
172    } else {
173        while reader
174            .next_interlaced_row()
175            .map_err(|e| format!("invalid PNG: {e}"))?
176            .is_some()
177        {
178            check_time(started, budget)?;
179        }
180    }
181    reader.finish().map_err(|e| format!("invalid PNG: {e}"))
182}

184fn jpeg_container(bytes: &[u8], started: Instant, budget: Duration) -> Result<(), String> {
185    if !bytes.starts_with(&[0xff, 0xd8]) {
186        return Err("missing JPEG start".into());
187    }
188    let mut at = 2;
189    let mut scans = 0;
190    for _ in 0..MAX_PARTS {
191        check_time(started, budget)?;
192        if bytes.get(at) != Some(&0xff) {
193            return Err("invalid JPEG marker".into());
194        }
195        while bytes.get(at) == Some(&0xff) {
196            at += 1;
197        }
198        let marker = *bytes.get(at).ok_or("truncated JPEG marker")?;
199        at += 1;
200        if marker == 0xd9 {
201            return if scans > 0 && at == bytes.len() {
202                Ok(())
203            } else {
204                Err("invalid JPEG end or data after EOI".into())
205            };
206        }
207        if matches!(marker, 0x00 | 0xd8 | 0xd0..=0xd7 | 0x01) {
208            return Err("unexpected JPEG marker outside a scan".into());
209        }
210        let length = bytes.get(at..at + 2).ok_or("truncated JPEG segment")?;
211        let length = usize::from(u16::from_be_bytes([length[0], length[1]]));
212        let end = at
213            .checked_add(length)
214            .filter(|end| length >= 2 && *end <= bytes.len())
215            .ok_or("truncated JPEG segment")?;
216        if matches!(marker, 0xc0..=0xc3 | 0xc5..=0xc7 | 0xc9..=0xcb | 0xcd..=0xcf) {
217            let frame = bytes
218                .get(at + 2..end)
219                .filter(|f| f.len() >= 6)
220                .ok_or("truncated JPEG frame")?;
221            let height = usize::from(u16::from_be_bytes([frame[1], frame[2]]));
222            let width = usize::from(u16::from_be_bytes([frame[3], frame[4]]));
223            dimensions(width, height)?;
224            let components = usize::from(frame[5]);
225            if !matches!(components, 1 | 3 | 4) || frame.len() != 6 + components * 3 {
226                return Err("invalid JPEG component count".into());
227            }
228            let mut blocks = 0;
229            for component in frame[6..].as_chunks::<3>().0 {
230                let h = component[1] >> 4;
231                let v = component[1] & 15;
232                if !(1..=4).contains(&h) || !(1..=4).contains(&v) {
233                    return Err("invalid JPEG sampling factors".into());
234                }
235                blocks += usize::from(h) * usize::from(v);
236            }
237            if blocks > 10 {
238                return Err("JPEG exceeds the sampling-block limit".into());
239            }
240        }
241        at = end;
242        if marker == 0xda {
243            scans += 1;
244            if scans > MAX_SCANS {
245                return Err("JPEG exceeds the 64-scan limit".into());
246            }
247            // This only locates entropy boundaries; the codec validates Huffman
248            // data and reconstructs pixels. FF00 is stuffing, FFD0..D7 restart.
249            loop {
250                check_time(started, budget)?;
251                let offset = bytes
252                    .get(at..)
253                    .and_then(|b| b.iter().position(|b| *b == 0xff))
254                    .ok_or("JPEG is missing its final EOI marker")?;
255                at += offset;
256                let mut next = at + 1;
257                while bytes.get(next) == Some(&0xff) {
258                    next += 1;
259                }
260                match bytes.get(next) {
261                    Some(0x00 | 0xd0..=0xd7) => at = next + 1,
262                    Some(_) => break,
263                    None => return Err("truncated JPEG entropy data".into()),
264                }
265            }
266        }
267    }
268    Err("JPEG exceeds the marker-count limit".into())
269}

271fn validate_jpeg(bytes: &[u8], started: Instant, budget: Duration) -> Result<(), String> {
272    use zune_jpeg::zune_core::options::DecoderOptions;
273    jpeg_container(bytes, started, budget)?;
274    let options = DecoderOptions::default()
275        .set_strict_mode(true)
276        .set_use_unsafe(false)
277        .set_max_width(MAX_AXIS)
278        .set_max_height(MAX_AXIS)
279        .jpeg_set_max_scans(MAX_SCANS);
280    let mut decoder =
281        zune_jpeg::JpegDecoder::new_with_options(TimedCursor::new(bytes, started, budget), options);
282    decoder
283        .decode_headers()
284        .map_err(|e| format!("invalid JPEG: {e}"))?;
285    let size = decoder
286        .output_buffer_size()
287        .filter(|n| *n <= MAX_OUTPUT)
288        .ok_or("JPEG exceeds the decoded-output budget")?;
289    let mut output = Vec::new();
290    output
291        .try_reserve_exact(size)
292        .map_err(|_| "cannot allocate JPEG validation buffer")?;
293    output.resize(size, 0);
294    decoder
295        .decode_into(&mut output)
296        .map_err(|e| format!("invalid JPEG: {e}"))
297}

299struct TimedCursor<'a> {
300    cursor: Cursor<&'a [u8]>,
301    started: Instant,
302    budget: Duration,
303}
304
305impl<'a> TimedCursor<'a> {
306    fn new(bytes: &'a [u8], started: Instant, budget: Duration) -> Self {
307        Self {
308            cursor: Cursor::new(bytes),
309            started,
310            budget,
311        }
312    }
313    fn check(&self) -> io::Result<()> {
314        check_time(self.started, self.budget)
315            .map_err(|e| io::Error::new(io::ErrorKind::TimedOut, e))
316    }
317}
318
319impl Read for TimedCursor<'_> {
320    fn read(&mut self, buffer: &mut [u8]) -> io::Result<usize> {
321        self.check()?;
322        self.cursor.read(buffer)
323    }
324}
325
326impl BufRead for TimedCursor<'_> {
327    fn fill_buf(&mut self) -> io::Result<&[u8]> {
328        self.check()?;
329        self.cursor.fill_buf()
330    }
331    fn consume(&mut self, amount: usize) {
332        self.cursor.consume(amount);
333    }
334}
335
336impl Seek for TimedCursor<'_> {
337    fn seek(&mut self, position: SeekFrom) -> io::Result<u64> {
338        self.check()?;
339        self.cursor.seek(position)
340    }
341}
