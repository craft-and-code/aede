1//! Versioned original-audio features beside the catalogue, never inside source files or personal opinions.
2use crate::playback::stream::PcmTrack;
3use aede_moods::{Extractor, Features, RATE};
4use std::path::Path;
5use std::time::{Duration, Instant};
6mod cache;

8/// Decode/resample with existing Aède and FlacCompagnon primitives into bounded descriptor state.
9/// Checkpoints may wait for foreground playback, or refuse cancellation/source replacement.
10/// The active computation budget excludes checkpoint waiting; blocked OS reads retain platform limits.
11pub fn extract(
12    path: &Path,
13    mut checkpoint: impl FnMut() -> Result<(), String>,
14) -> Result<Features, String> {
15    checkpoint()?;
16    let mut track = PcmTrack::open_native(path).map_err(|e| e.to_string())?;
17    let format = track.source_format();
18    if !(1..=2).contains(&format.channels()) || !(8000..=192000).contains(&format.sample_rate()) {
19        return Err("source outside native mono/stereo acoustic profile".into());
20    }
21    track.set_output_rate(RATE).map_err(|e| e.to_string())?;
22    let origin = Instant::now();
23    extract_source(
24        &mut track,
25        usize::from(format.channels()),
26        checkpoint,
27        || origin.elapsed(),
28    )
29}

31trait Source {
32    fn read_samples(&mut self) -> Result<Option<&[f32]>, String>;
33}
34impl Source for PcmTrack {
35    fn read_samples(&mut self) -> Result<Option<&[f32]>, String> {
36        self.read_block::<std::convert::Infallible>(|_| Ok(()))
37            .map(|block| block.map(|block| block.samples))
38            .map_err(|error| error.to_string())
39    }
40}
41
42fn check_budget(active: Duration) -> Result<(), String> {
43    if active > Duration::from_secs(180) {
44        Err("acoustic active computation budget exceeded".into())
45    } else {
46        Ok(())
47    }
48}
49
50// Keep budget accounting on the same path for native decoding and controlled clocks.
51fn extract_source(
52    track: &mut impl Source,
53    channels: usize,
54    mut checkpoint: impl FnMut() -> Result<(), String>,
55    mut now: impl FnMut() -> Duration,
56) -> Result<Features, String> {
57    let mut extractor = Extractor::new(channels)?;
58    let mut active = Duration::ZERO;
59    loop {
60        checkpoint()?;
61        let step = now();
62        let samples = track.read_samples()?;
63        active += now().saturating_sub(step);
64        check_budget(active)?;
65        let Some(samples) = samples else {
66            break;
67        };
68        let mut parked = Duration::ZERO;
69        let step = now();
70        extractor.push(samples, || {
71            let before = now();
72            let result = checkpoint();
73            parked += now().saturating_sub(before);
74            result
75        })?;
76        active += now().saturating_sub(step).saturating_sub(parked);
77        check_budget(active)?;
78    }
79    checkpoint()?;
80    extractor.finish()
81}
