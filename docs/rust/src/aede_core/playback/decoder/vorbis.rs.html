34impl VorbisStream {
35    pub fn open(path: &Path) -> Result<Option<Self>, Error> {
36        let mut file = File::open(path).map_err(decode_error)?;
37        let Some(initial) = first_audio_page(&mut file)? else {
38            return Ok(None);
39        };
40        file.seek(SeekFrom::Start(0)).map_err(decode_error)?;
41        let source = MediaSourceStream::new(Box::new(file), Default::default());
42        let mut hint = Hint::new();
43        hint.with_extension("ogg");
44        let probed = symphonia::default::get_probe()
45            .format(
46                &hint,
47                source,
48                &FormatOptions {
49                    enable_gapless: true,
50                    ..Default::default()
51                },
52                &MetadataOptions::default(),
53            )
54            .map_err(decode_error)?;
55        let track = probed.format.default_track().ok_or(Error::InvalidFormat)?;
56        let params = &track.codec_params;
57        if track.id != initial.serial || params.codec != CODEC_TYPE_VORBIS {
58            return Err(decode_error("multiplexed Vorbis playback is unsupported"));
59        }
60        let sample_rate = params.sample_rate.ok_or(Error::InvalidFormat)?;
61        let speaker_mask = params.channels.ok_or(Error::InvalidFormat)?;
62        let channels = u16::try_from(speaker_mask.count()).map_err(|_| Error::InvalidFormat)?;
63        let layout =
64            ChannelLayout::from_mask(speaker_mask.bits()).map_err(|_| Error::InvalidFormat)?;
65        // n_frames is populated from a validated EOS page by the Ogg reader.
66        // Without that bound, exact completion cannot be established.
67        let remaining_frames = params
68            .n_frames
69            .ok_or_else(|| decode_error("Vorbis stream has no valid end-of-stream granule"))?;
70        // An EOS first audio page can describe tail padding as `delay` in
71        // Symphonia 0.5.5. A non-EOS first page instead describes a real crop.
72        let skip_frames = if initial.is_final {
73            0
74        } else {
75            u64::from(params.delay.unwrap_or(0))
76        };
77        let decoder = symphonia::default::get_codecs()
78            .make(params, &DecoderOptions::default())
79            .map_err(decode_error)?;
80        Ok(Some(Self {
81            format: probed.format,
82            decoder,
83            track_id: initial.serial,
84            spec: SignalSpec::new(sample_rate, speaker_mask),
85            buffer: None,
86            skip_frames,
87            remaining_frames,
88            discarded_tail: 0,
89            last_packet_frames: 0,
90            sample_rate,
91            channels,
92            layout,
93        }))
94    }
95
96    pub fn next_chunk(&mut self) -> Result<Option<Vec<f32>>, Error> {
97        loop {
98            let mut packet = match self.format.next_packet() {
99                Ok(packet) => packet,
100                Err(SymError::IoError(error))
101                    if error.kind() == std::io::ErrorKind::UnexpectedEof =>
102                {
103                    if self.remaining_frames != 0 || self.skip_frames != 0 {
104                        return Err(decode_error(
105                            "Vorbis ended before its declared playable frames",
106                        ));
107                    }
108                    if self.discarded_tail > self.last_packet_frames {
109                        return Err(decode_error(
110                            "Vorbis final granule trims more than its final packet",
111                        ));
112                    }
113                    return Ok(None);
114                }
115                Err(SymError::ResetRequired) => {
116                    return Err(decode_error("chained Vorbis playback is unsupported"));
117                }
118                Err(error) => return Err(decode_error(error)),
119            };
120            if packet.track_id() != self.track_id {
121                return Err(decode_error("multiplexed Vorbis playback is unsupported"));
122            }
123            // The initial page was buffered before Symphonia probed its bounds.
124            // Apply both bounds ourselves so no packet is missed or trimmed twice.
125            packet.trim_start = 0;
126            packet.trim_end = 0;
127            let decoded = self.decoder.decode(&packet).map_err(decode_error)?;
128            if *decoded.spec() != self.spec {
129                return Err(Error::InvalidFormat);
130            }
131            let capacity = decoded.capacity() * usize::from(self.channels);
132            if self
133                .buffer
134                .as_ref()
135                .is_none_or(|buffer| buffer.capacity() < capacity)
136            {
137                self.buffer = Some(SampleBuffer::new(decoded.capacity() as u64, self.spec));
138            }
139            let buffer = self.buffer.as_mut().ok_or(Error::InvalidBuffer)?;
140            buffer.copy_interleaved_ref(decoded);
141            let channels = usize::from(self.channels);
142            let frames = buffer.samples().len() / channels;
143            self.last_packet_frames = frames;
144            let skip = self.skip_frames.min(frames as u64) as usize;
145            self.skip_frames -= skip as u64;
146            let take = self.remaining_frames.min((frames - skip) as u64) as usize;
147            self.remaining_frames -= take as u64;
148            self.discarded_tail = self.discarded_tail.saturating_add(frames - skip - take);
149            if take != 0 {
150                return Ok(Some(
151                    buffer.samples()[skip * channels..(skip + take) * channels].to_vec(),
152                ));
153            }
154            // Continue through the physical EOF even after the PCM bound: a
155            // chained stream or decoder error must not become successful EOF.
156        }
157    }

165/// Inspect bounded header pages on the same handle that the decoder will own.
166/// The page flag is needed to distinguish a prefix crop from a one-page tail.
167fn first_audio_page(file: &mut File) -> Result<Option<InitialPage>, Error> {
168    const MAX_HEADER_BYTES: usize = 8 * 1024 * 1024;
169    let mut serial = None;
170    let mut headers = 0;
171    let mut total_bytes = 0;
172    loop {
173        let mut header = [0u8; 27];
174        match file.read_exact(&mut header) {
175            Ok(()) => {}
176            Err(error) if serial.is_none() && error.kind() == std::io::ErrorKind::UnexpectedEof => {
177                return Ok(None);
178            }
179            Err(error) => return Err(decode_error(error)),
180        }
181        if header.get(..4) != Some(b"OggS") {
182            if serial.is_none() {
183                return Ok(None);
184            }
185            return Err(decode_error("invalid Vorbis Ogg page capture"));
186        }
187        let segment_count = usize::from(*header.get(26).ok_or(Error::InvalidFormat)?);
188        let mut lacing = vec![0; segment_count];
189        file.read_exact(&mut lacing).map_err(decode_error)?;
190        let body_len = lacing
191            .iter()
192            .map(|&length| usize::from(length))
193            .sum::<usize>();
194        let mut body = vec![0; body_len];
195        file.read_exact(&mut body).map_err(decode_error)?;
196        let mut page = Vec::with_capacity(header.len() + segment_count + body_len);
197        page.extend_from_slice(&header);
198        page.extend_from_slice(&lacing);
199        page.extend_from_slice(&body);
200        total_bytes += page.len();
201        if total_bytes > MAX_HEADER_BYTES {
202            return Err(decode_error(
203                "Vorbis headers exceed the 8 MiB playback limit",
204            ));
205        }
206        let flags = *header.get(5).ok_or(Error::InvalidFormat)?;
207        if serial.is_none() && body.get(..7) != Some(b"\x01vorbis") {
208            return Ok(None);
209        }
210        let expected_crc = read_u32(&header, 22)?;
211        page.get_mut(22..26).ok_or(Error::InvalidFormat)?.fill(0);
212        if header.get(4) != Some(&0)
213            || flags & !7 != 0
214            || crate::audit::crc::crc32_ogg(&page) != expected_crc
215        {
216            return Err(decode_error("invalid Vorbis Ogg header page or checksum"));
217        }
218        let page_serial = read_u32(&header, 14)?;
219        let granule: [u8; 8] = header
220            .get(6..14)
221            .ok_or(Error::InvalidFormat)?
222            .try_into()
223            .map_err(|_| Error::InvalidFormat)?;
224        if u64::from_le_bytes(granule) > i64::MAX as u64
225            && lacing.iter().any(|&length| length < 255)
226        {
227            return Err(decode_error(
228                "Vorbis completed packets have an invalid granule",
229            ));
230        }
231        let selected = *serial.get_or_insert(page_serial);
232        if selected != page_serial {
233            return Err(decode_error("multiplexed Vorbis playback is unsupported"));
234        }
235        for length in lacing {
236            if headers == 3 {
237                return Ok(Some(InitialPage {
238                    serial: selected,
239                    is_final: flags & 4 != 0,
240                }));
241            }
242            if length < 255 {
243                headers += 1;
244            }
245        }
246    }
247}

249fn read_u32(bytes: &[u8], offset: usize) -> Result<u32, Error> {
250    let value = bytes.get(offset..offset + 4).ok_or(Error::InvalidFormat)?;
251    let value: [u8; 4] = value.try_into().map_err(|_| Error::InvalidFormat)?;
252    Ok(u32::from_le_bytes(value))
253}

255fn decode_error(error: impl std::fmt::Display) -> Error {
256    Error::Decode(AnalysisError::Decode(error.to_string()))
