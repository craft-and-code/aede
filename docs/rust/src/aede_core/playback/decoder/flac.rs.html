1//! Progressive FLAC decoding with verification of the original integer PCM.
2//!
3//! The FlacCompagnon preview API does not expose decoder verification options.
4//! Use the already shared Symphonia codec here: its validator hashes samples
5//! before conversion to float, with FLAC's signed little-endian interleaving.

47impl<S: ConvertibleSample> FlacDecoder<S> {
48    pub fn open(path: &Path) -> Result<Option<Self>, Error> {
49        catch_unwind(AssertUnwindSafe(|| Self::open_inner(path)))
50            .unwrap_or_else(|_| Err(decode_error("FLAC decoder rejected a malformed header")))
51    }
52
53    fn open_inner(path: &Path) -> Result<Option<Self>, Error> {
54        let file = File::open(path).map_err(decode_error)?;
55        Self::open_file_inner(file, path.extension().and_then(|value| value.to_str()))
56    }
57
58    fn open_file_inner(mut file: File, extension: Option<&str>) -> Result<Option<Self>, Error> {
59        let mut magic = [0; 4];
60        match file.read_exact(&mut magic) {
61            Ok(()) => {}
62            Err(error) if error.kind() == std::io::ErrorKind::UnexpectedEof => return Ok(None),
63            Err(error) => return Err(decode_error(error)),
64        }
65        let native_container = &magic == b"fLaC" || &magic[..3] == b"ID3";
66        if !native_container && &magic != b"OggS" {
67            return Ok(None);
68        }
69        file.seek(SeekFrom::Start(0)).map_err(decode_error)?;
70        let input: Box<dyn MediaSource> = if native_container {
71            match super::flac_source::FlacAudioSource::new(file)? {
72                Some(source) => Box::new(source),
73                None => return Ok(None),
74            }
75        } else {
76            Box::new(file)
77        };
78        let source = MediaSourceStream::new(input, Default::default());
79        let mut hint = Hint::new();
80        if let Some(extension) = extension {
81            hint.with_extension(extension);
82        }
83        let probed = match symphonia::default::get_probe().format(
84            &hint,
85            source,
86            &FormatOptions::default(),
87            &MetadataOptions::default(),
88        ) {
89            Ok(probed) => probed,
90            // The bounded source already established native FLAC, including
91            // after ID3. Never retry that recognized source without verification.
92            Err(error) if native_container => return Err(decode_error(error)),
93            Err(_) => return Ok(None),
94        };
95        let track = probed.format.default_track().ok_or(Error::InvalidFormat)?;
96        if track.codec_params.codec != CODEC_TYPE_FLAC {
97            return Ok(None);
98        }
99        if probed.format.tracks().len() != 1 {
100            return Err(decode_error("multiplexed FLAC playback is unsupported"));
101        }
102        // Native and Ogg readers supply the 34-byte STREAMINFO as extra data.
103        // An all-zero signature is absent: avoid meaningless hash work.
104        let has_signature = track
105            .codec_params
106            .extra_data
107            .as_deref()
108            .and_then(|info| info.get(18..34))
109            .is_some_and(|digest| digest.iter().any(|&byte| byte != 0));
110        let decoder = symphonia::default::get_codecs()
111            .make(
112                &track.codec_params,
113                &DecoderOptions {
114                    verify: has_signature,
115                },
116            )
117            .map_err(decode_error)?;
118        let params = decoder.codec_params();
119        let sample_rate = params.sample_rate.ok_or(Error::InvalidFormat)?;
120        let mask = params.channels.ok_or(Error::InvalidFormat)?;
121        let channels = u16::try_from(mask.count()).map_err(|_| Error::InvalidFormat)?;
122        let layout = ChannelLayout::from_mask(mask.bits()).map_err(|_| Error::InvalidFormat)?;
123        let bits_per_sample = params.bits_per_sample.ok_or(Error::InvalidFormat)?;
124        if sample_rate == 0 || channels == 0 || !(4..=32).contains(&bits_per_sample) {
125            return Err(Error::InvalidFormat);
126        }
127        let declared_frames = params.n_frames.filter(|&frames| frames != 0);
128        let track_id = track.id;
129        Ok(Some(Self {
130            format: probed.format,
131            decoder,
132            track_id,
133            spec: SignalSpec::new(sample_rate, mask),
134            buffer: None,
135            declared_frames,
136            decoded_frames: 0,
137            bits_per_sample,
138            native_container,
139            integer_output: false,
140            status: if has_signature {
141                FlacMd5Status::Pending
142            } else {
143                FlacMd5Status::NoSignature
144            },
145            finished: false,
146            failure: None,
147            sample_rate,
148            channels,
149            layout,
150        }))
151    }
152
153    pub fn md5_status(&self) -> FlacMd5Status {
154        self.status
155    }
156
157    pub fn next_chunk(&mut self) -> Result<Option<Vec<S>>, Error> {
158        if self.status == FlacMd5Status::Mismatch {
159            return Err(Error::FlacMd5Mismatch);
160        }
161        if let Some(failure) = &self.failure {
162            return Err(decode_error(failure));
163        }
164        if self.finished {
165            return Ok(None);
166        }
167        // The pinned codec has unchecked arithmetic/assertions on malformed
168        // frames. Contain those failures here and never expose clean EOF after
169        // a partially advanced decoder has failed. No panic recovery resumes it.
170        let result =
171            catch_unwind(AssertUnwindSafe(|| self.next_chunk_inner())).unwrap_or_else(|_| {
172                Err(decode_error(
173                    "FLAC decoder rejected an unsupported or malformed frame",
174                ))
175            });
176        if let Err(error) = &result {
177            self.failure = Some(error.to_string());
178        }
179        result
180    }
181
182    fn next_chunk_inner(&mut self) -> Result<Option<Vec<S>>, Error> {
183        let packet = match self.format.next_packet() {
184            Ok(packet) => packet,
185            Err(SymError::IoError(error)) if error.kind() == std::io::ErrorKind::UnexpectedEof => {
186                return self.finish();
187            }
188            Err(error) => return Err(decode_error(error)),
189        };
190        if packet.track_id() != self.track_id {
191            return Err(decode_error("multiplexed FLAC playback is unsupported"));
192        }
193        if self.native_container && packet.ts != self.decoded_frames {
194            return Err(decode_error(
195                "FLAC frames are missing, repeated or out of order",
196            ));
197        }
198        let header = *packet.data.get(3).ok_or(Error::IncompleteFrame)?;
199        let frame_bits = match (header >> 1) & 7 {
200            0 => self.bits_per_sample,
201            1 => 8,
202            2 => 12,
203            4 => 16,
204            5 => 20,
205            6 => 24,
206            7 => 32,
207            _ => return Err(Error::InvalidFormat),
208        };
209        let assignment = header >> 4;
210        let frame_channels = match assignment {
211            0..=7 => u16::from(assignment) + 1,
212            8..=10 => 2,
213            _ => return Err(Error::InvalidFormat),
214        };
215        if frame_bits != self.bits_per_sample
216            || frame_channels != self.channels
217            || frame_sample_rate(&packet.data, self.sample_rate) != Some(self.sample_rate)
218        {
219            return Err(Error::InvalidFormat);
220        }
221        // A correlated 32-bit stereo frame needs a 33-bit side channel, which
222        // Symphonia 0.5.5 cannot decode. Independent 32-bit channels remain valid.
223        if frame_bits == 32 && assignment >= 8 {
224            return Err(decode_error(
225                "32-bit correlated FLAC stereo is unsupported by the playback decoder",
226            ));
227        }
228        let decoded = self.decoder.decode(&packet).map_err(decode_error)?;
229        if *decoded.spec() != self.spec {
230            return Err(Error::InvalidFormat);
231        }
232        // The pinned FLAC codec emits S32 with valid bits left-aligned. Never
233        // let its integer adapter silently quantize a future floating result.
234        if self.integer_output {
235            let AudioBufferRef::S32(samples) = &decoded else {
236                return Err(Error::InvalidFormat);
237            };
238            use symphonia::core::audio::Signal;
239            let padding_mask = (1_u32 << (32 - self.bits_per_sample)) - 1;
240            for channel in 0..usize::from(self.channels) {
241                if samples
242                    .chan(channel)
243                    .iter()
244                    .any(|sample| (*sample as u32) & padding_mask != 0)
245                {
246                    return Err(Error::InvalidFormat);
247                }
248            }
249        }
250        let frames = decoded.frames() as u64;
251        if frames == 0 || packet.dur != frames || packet.trim_start != 0 || packet.trim_end != 0 {
252            return Err(decode_error(
253                "FLAC packet disagrees with its decoded frame count",
254            ));
255        }
256        self.decoded_frames = self
257            .decoded_frames
258            .checked_add(frames)
259            .ok_or(Error::InvalidFormat)?;
260        if self
261            .declared_frames
262            .is_some_and(|expected| self.decoded_frames > expected)
263        {
264            return Err(decode_error("FLAC exceeds its declared source-frame count"));
265        }
266        let capacity = decoded.capacity() * usize::from(self.channels);
267        if self
268            .buffer
269            .as_ref()
270            .is_none_or(|buffer| buffer.capacity() < capacity)
271        {
272            self.buffer = Some(SampleBuffer::new(decoded.capacity() as u64, self.spec));
273        }
274        let buffer = self.buffer.as_mut().ok_or(Error::InvalidBuffer)?;
275        buffer.copy_interleaved_ref(decoded);
276        Ok(Some(buffer.samples().to_vec()))
277    }
278
279    fn finish(&mut self) -> Result<Option<Vec<S>>, Error> {
280        if self
281            .declared_frames
282            .is_some_and(|expected| self.decoded_frames != expected)
283        {
284            return Err(decode_error(
285                "FLAC ended before its declared source-frame count",
286            ));
287        }
288        if self.status == FlacMd5Status::Pending {
289            match self.decoder.finalize().verify_ok {
290                Some(true) => self.status = FlacMd5Status::Verified,
291                Some(false) => {
292                    self.status = FlacMd5Status::Mismatch;
293                    return Err(Error::FlacMd5Mismatch);
294                }
295                None => {
296                    return Err(decode_error(
297                        "FLAC decoder did not provide MD5 verification",
298                    ));
299                }
300            }
301        }
302        self.finished = true;
303        Ok(None)
304    }
305}

307impl IntegerFlacStream {
308    /// Reuse native FLAC validation on the same descriptor used for admission.
309    pub(super) fn open_integer(file: File) -> Result<Self, Error> {
310        let stream = catch_unwind(AssertUnwindSafe(|| Self::open_file_inner(file, None)))
311            .unwrap_or_else(|_| Err(decode_error("FLAC decoder rejected a malformed header")))?;
312        let mut stream = stream.ok_or(Error::UnsupportedIntegerSource(
313            "integer playback requires native FLAC or PCM WAV",
314        ))?;
315        if !stream.native_container
316            || !matches!(stream.bits_per_sample, 16 | 24)
317            || !matches!(stream.layout, ChannelLayout::MONO | ChannelLayout::STEREO)
318        {
319            return Err(Error::UnsupportedIntegerSource(
320                "integer FLAC playback requires native 16/24-bit mono or stereo",
321            ));
322        }
323        stream.integer_output = true;
324        Ok(stream)
325    }
326
327    pub(super) fn bits_per_sample(&self) -> u32 {
328        self.bits_per_sample
329    }
330}

332// Ogg's FLAC mapper checks frame CRCs, but unlike the native reader it does
333// not check format declarations against STREAMINFO. Inspect only the small
334// header; the validated UTF-8 counter and block-size suffix precede the rate.
335fn frame_sample_rate(data: &[u8], stream_rate: u32) -> Option<u32> {
336    let codes = [
337        0, 88_200, 176_400, 192_000, 8_000, 16_000, 22_050, 24_000, 32_000, 44_100, 48_000, 96_000,
338    ];
339    let code = data.get(2)? & 15;
340    match code {
341        0 => Some(stream_rate),
342        1..=11 => Some(codes[usize::from(code)]),
343        12..=14 => {
344            let counter_bytes = match data.get(4)?.leading_ones() {
345                0 => 1,
346                size @ 2..=7 => size as usize,
347                _ => return None,
348            };
349            let block_bytes = match data[2] >> 4 {
350                6 => 1,
351                7 => 2,
352                _ => 0,
353            };
354            let offset = 4 + counter_bytes + block_bytes;
355            if code == 12 {
356                Some(u32::from(*data.get(offset)?) * 1000)
357            } else {
358                let bytes: [u8; 2] = data.get(offset..offset + 2)?.try_into().ok()?;
359                Some(u32::from(u16::from_be_bytes(bytes)) * if code == 14 { 10 } else { 1 })
360            }
361        }
362        _ => None,
363    }
364}
