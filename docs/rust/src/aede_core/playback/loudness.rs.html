50/// Read the identity used by both the imported-analysis and playback caches.
51pub fn identity(path: &Path) -> Result<ProgrammeFile, std::io::Error> {
52    let metadata = std::fs::metadata(path)?;
53    Ok(ProgrammeFile {
54        path: path.to_string_lossy().into_owned(),
55        size: metadata.len(),
56        mtime: clock::mtime_seconds(&metadata),
57    })
58}

60/// Reuse an attributed analysis only while it describes these exact bytes.
61/// A missing true peak stays unknown so the output stage can choose its policy.
62pub fn from_flaccompagnon(analyses: &[FileAnalysis], file: &ProgrammeFile) -> Option<Measurement> {
63    analyses.iter().rev().find_map(|analysis| {
64        if analysis.path != file.path
65            || analysis.source != "flaccompagnon"
66            || analysis.error.is_some()
67            || !analysis.still_applies(file.size, file.mtime)
68        {
69            return None;
70        }
71        let loudness = analysis.integrated_lufs?;
72        if !loudness.is_finite() || !(-100.0..=20.0).contains(&loudness) {
73            return None;
74        }
75        let peak = analysis
76            .true_peak_dbtp
77            .filter(|value| value.is_finite())
78            .map(|db| 10f64.powf(db / 20.0) as f32)
79            .filter(|peak| peak.is_finite() && *peak >= 0.0);
80        Some(Measurement {
81            integrated_lufs: loudness as f32,
82            true_peak: peak,
83        })
84    })
85}

87fn decode_into(
88    decoder: &mut FileDecoder,
89    meter: &mut LoudnessProgramme,
90) -> Result<(), Box<dyn Error>> {
91    let channels = usize::from(decoder.channels());
92    let format = PcmFormat::new(decoder.sample_rate(), decoder.channels())?;
93    let mut samples = vec![0.0; 4096 * channels];
94    loop {
95        let frames = decoder.read_frames(&mut samples)?;
96        if frames == 0 {
97            break;
98        }
99        meter.push(format, &samples[..frames * channels])?;
100    }
101    Ok(())
102}

104/// Measure a track once before output, so playback never has to wait mid-track.
105pub fn measure_track(path: &Path) -> Result<Option<Measurement>, Box<dyn Error>> {
106    let mut decoder = FileDecoder::open(path)?;
107    if !matches!(decoder.channels(), 1 | 2) {
108        return Ok(None);
109    }
110    let mut meter = LoudnessProgramme::new();
111    decode_into(&mut decoder, &mut meter)?;
112    Ok(meter.measurement()?)
113}
114
115/// Measure an ordered album programme. One meter spans matching formats;
116/// different formats retain independent meters, then share one gated result.
117pub fn measure_programme(paths: &[&Path]) -> Result<Option<Measurement>, Box<dyn Error>> {
118    if paths.is_empty() {
119        return Ok(None);
120    }
121    let mut meter = LoudnessProgramme::new();
122    for path in paths {
123        let mut decoder = FileDecoder::open(path)?;
124        if !matches!(decoder.channels(), 1 | 2) {
125            return Ok(None);
126        }
127        decode_into(&mut decoder, &mut meter)?;
128    }
129    Ok(meter.measurement()?)
130}
