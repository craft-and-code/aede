20/// A completed source measurement, saved outside the audio decoding loop.
21pub struct LoudnessUpdate {
22    track: Option<(ProgrammeFile, CachedTrack)>,
23    programme: Option<CachedProgramme>,
24    legacy_seed: Option<Arc<Conclusions>>,
25}
26
27impl LoudnessUpdate {
28    /// Merge into the latest store under its writer lock. A source changed
29    /// after decoding is rejected instead of publishing a stale measurement.
30    pub fn save(&self, directory: &Path) -> Result<(), Box<dyn Error>> {
31        let _lock = StoreLock::acquire(directory)?;
32        let files = self
33            .track
34            .as_ref()
35            .map(|(file, _)| std::slice::from_ref(file))
36            .or_else(|| self.programme.as_ref().map(|item| item.files.as_slice()))
37            .unwrap_or_default();
38        for file in files {
39            if loudness::identity(Path::new(&file.path))? != *file {
40                return Err("audio file changed before its loudness could be cached".into());
41            }
42        }
43        let path = conclusions::conclusions_path(directory);
44        let mut current = conclusions::load(&path)?
45            .unwrap_or_else(|| self.legacy_seed.as_deref().cloned().unwrap_or_default());
46        if let Some((file, track)) = &self.track {
47            current
48                .loudness_tracks
49                .insert(file.path.clone(), track.clone());
50        }
51        if let Some(programme) = &self.programme {
52            replace_programme(&mut current, programme.clone());
53        }
54        conclusions::save(&current, &path)?;
55        Ok(())
56    }
57}
58
59fn replace_programme(cache: &mut Conclusions, item: CachedProgramme) {
60    cache.loudness_programmes.retain(|cached| {
61        cached
62            .files
63            .iter()
64            .map(|file| &file.path)
65            .ne(item.files.iter().map(|file| &file.path))
66    });
67    cache.loudness_programmes.push(item);
68}

77/// A lazy normalization session shared by CLI and native PCM playback.
78///
79/// Only the current track (or current album's tags/identities) is inspected.
80/// The CLI can measure missing loudness from source PCM already being decoded;
81/// native PCM playback only reuses existing gains and does not enable capture.
82/// An album decision is frozen for the session, never changed midway through it.
83pub struct ReadyNormalization<'a> {
84    paths: &'a [PathBuf],
85    catalog: Option<&'a Catalog>,
86    mode: NormalizationMode,
87    cache: Conclusions,
88    legacy_seed: Option<Arc<Conclusions>>,
89    groups: Vec<Vec<usize>>,
90    prepared: Vec<bool>,
91    learnable_album_starts: Vec<bool>,
92    plans: Vec<Option<GainPlan>>,
93    capture: Option<Capture>,
94}
95
96impl<'a> ReadyNormalization<'a> {
97    /// Load available measurements without reading audio files or their tags.
98    pub fn new(
99        paths: &'a [PathBuf],
100        is_album: bool,
101        catalog: Option<&'a Catalog>,
102        directory: &Path,
103        mode: NormalizationMode,
104    ) -> Result<Self, Box<dyn Error>> {
105        let loaded = if mode == NormalizationMode::Off {
106            Some(Conclusions::default())
107        } else {
108            conclusions::load(&conclusions::conclusions_path(directory))?
109        };
110        let legacy_seed = if loaded.is_none() {
111            catalog.map(|catalog| Arc::new(Conclusions::from_catalog(catalog)))
112        } else {
113            None
114        };
115        let cache = loaded.unwrap_or_else(|| legacy_seed.as_deref().cloned().unwrap_or_default());
116        Ok(Self {
117            paths,
118            catalog,
119            mode,
120            cache,
121            legacy_seed,
122            groups: if mode == NormalizationMode::Album {
123                album_groups(paths, is_album, catalog)
124            } else {
125                Vec::new()
126            },
127            prepared: vec![false; paths.len()],
128            learnable_album_starts: vec![false; paths.len()],
129            plans: vec![None; paths.len()],
130            capture: None,
131        })
132    }
133
134    /// Choose an already available gain for the current track or album.
135    /// Missing source measurements are captured later through `observe_source`.
136    pub fn prepare(&mut self, index: usize) -> Result<Option<GainPlan>, Box<dyn Error>> {
137        let path = self
138            .paths
139            .get(index)
140            .ok_or("playback track index is out of range")?;
141        if self
142            .capture
143            .as_ref()
144            .is_some_and(|capture| capture.indices.get(capture.next) != Some(&index))
145        {
146            self.capture = None;
147        }
148        if self.mode == NormalizationMode::Off {
149            return Ok(None);
150        }
151        if self.mode == NormalizationMode::Album {
152            if !self.prepared[index] {
153                self.prepare_album(index)?;
154            } else if self.capture.is_none() && self.learnable_album_starts[index] {
155                let indices = self
156                    .groups
157                    .iter()
158                    .find(|group| group.first() == Some(&index))
159                    .ok_or("playback album group is missing")?
160                    .clone();
161                let files = indices
162                    .iter()
163                    .map(|&index| loudness::identity(&self.paths[index]))
164                    .collect::<Result<Vec<_>, _>>()?;
165                // Restart learning only from the album's beginning; its gain
166                // decision remains frozen even after a complete measurement.
167                self.capture = Some(Capture {
168                    indices,
169                    files,
170                    next: 0,
171                    meter: Some(LoudnessProgramme::new()),
172                });
173            }
174            return Ok(self.plans[index]);
175        }
176        let file = loudness::identity(path)?;
177        let raw = tags::read(path)?;
178        let metadata =
179            normalization::select_raw(&raw, self.mode, normalization::DEFAULT_TARGET_LUFS)?;
180        if let Some(gain) = metadata.filter(|gain| is_requested_scope(gain.source, self.mode)) {
181            self.capture = None;
182            return Ok(Some(metadata_plan(gain)));
183        }
184        let imported = loudness::from_flaccompagnon(&self.cache.analyses, &file).or_else(|| {
185            self.catalog
186                .and_then(|catalog| loudness::from_flaccompagnon(&catalog.analyses, &file))
187        });
188        if let Some(measurement) = imported {
189            self.capture = None;
190            return Ok(Some(measured_plan(measurement, false, true)));
191        }
192        if let Some(cached) = self
193            .cache
194            .loudness_tracks
195            .get(&file.path)
196            .filter(|cached| cached.size == file.size && cached.mtime == file.mtime)
197        {
198            self.capture = None;
199            return Ok(cached
200                .measurement
201                .map(|value| measured_plan(value, false, false))
202                .or_else(|| metadata.map(metadata_plan)));
203        }
204        self.capture = Some(Capture {
205            indices: vec![index],
206            files: vec![file],
207            next: 0,
208            meter: Some(LoudnessProgramme::new()),
209        });
210        Ok(metadata.map(metadata_plan))
211    }

213    fn prepare_album(&mut self, index: usize) -> Result<(), Box<dyn Error>> {
214        let indices = self
215            .groups
216            .iter()
217            .find(|group| group.contains(&index))
218            .ok_or("playback album group is missing")?
219            .clone();
220        let files = indices
221            .iter()
222            .map(|&index| loudness::identity(&self.paths[index]))
223            .collect::<Result<Vec<_>, _>>()?;
224        let metadata = indices
225            .iter()
226            .map(|&index| {
227                let raw = tags::read(&self.paths[index])?;
228                normalization::select_raw(&raw, self.mode, normalization::DEFAULT_TARGET_LUFS)
229                    .map_err(|error| Box::new(error) as Box<dyn Error>)
230            })
231            .collect::<Result<Vec<_>, Box<dyn Error>>>()?;
232        if metadata
233            .iter()
234            .all(|gain| gain.is_some_and(|gain| is_requested_scope(gain.source, self.mode)))
235        {
236            for (&index, gain) in indices.iter().zip(metadata) {
237                self.plans[index] = gain.map(metadata_plan);
238            }
239        } else if let Some(cached) = self
240            .cache
241            .loudness_programmes
242            .iter()
243            .find(|cached| cached.files == files)
244        {
245            let plan = cached
246                .measurement
247                .map(|value| measured_plan(value, true, false));
248            for &index in &indices {
249                self.plans[index] = plan;
250            }
251        } else if let Some(&first) = indices.first() {
252            // Incomplete album metadata must not turn album mode into a series
253            // of per-track adjustments. Learn one programme for a later play.
254            self.learnable_album_starts[first] = true;
255            if first == index {
256                self.capture = Some(Capture {
257                    indices: indices.clone(),
258                    files,
259                    next: 0,
260                    meter: Some(LoudnessProgramme::new()),
261                });
262            }
263        }
264        for index in indices {
265            self.prepared[index] = true;
266        }
267        Ok(())
268    }

270    /// Whether an active meter is capturing source PCM for a later decision.
271    pub fn is_measuring(&self) -> bool {
272        self.capture
273            .as_ref()
274            .is_some_and(|capture| capture.meter.is_some())
275    }

277    /// Observe decoded source PCM before downmix, rate conversion and effects.
278    /// Unsupported layouts retain an unavailable result only after a complete
279    /// decode. Other meter failures abandon the optional capture.
280    pub fn observe_source(
281        &mut self,
282        format: PcmFormat,
283        samples: &[f32],
284    ) -> Result<(), Box<dyn Error>> {
285        if let Some(capture) = &mut self.capture
286            && let Some(meter) = &mut capture.meter
287            && let Err(error) = meter.push(format, samples)
288        {
289            if matches!(error, LoudnessError::UnsupportedLayout) {
290                capture.meter = None;
291            } else {
292                self.capture = None;
293            }
294            return Err(error.into());
295        }
296        Ok(())
297    }

299    /// Publish only fully decoded tracks/programmes with unchanged sources.
300    pub fn finish_track(
301        &mut self,
302        index: usize,
303        complete: bool,
304    ) -> Result<Option<LoudnessUpdate>, Box<dyn Error>> {
305        let Some(mut capture) = self.capture.take() else {
306            return Ok(None);
307        };
308        if !complete || capture.indices.get(capture.next) != Some(&index) {
309            return Ok(None);
310        }
311        let file = &capture.files[capture.next];
312        if loudness::identity(Path::new(&file.path))? != *file {
313            return Err("audio file changed while its loudness was measured".into());
314        }
315        capture.next += 1;
316        if capture.next < capture.indices.len() {
317            self.capture = Some(capture);
318            return Ok(None);
319        }
320        let measurement = capture
321            .meter
322            .as_ref()
323            .map(LoudnessProgramme::measurement)
324            .transpose()?
325            .flatten();
326        let mut update = LoudnessUpdate {
327            track: None,
328            programme: None,
329            legacy_seed: self.legacy_seed.clone(),
330        };
331        if self.mode == NormalizationMode::Track {
332            let file = capture.files.remove(0);
333            let cached = CachedTrack {
334                size: file.size,
335                mtime: file.mtime,
336                measurement,
337            };
338            self.cache
339                .loudness_tracks
340                .insert(file.path.clone(), cached.clone());
341            update.track = Some((file, cached));
342        } else {
343            if let Some(&first) = capture.indices.first() {
344                self.learnable_album_starts[first] = false;
345            }
346            let cached = CachedProgramme {
347                files: capture.files,
348                measurement,
349            };
350            replace_programme(&mut self.cache, cached.clone());
351            update.programme = Some(cached);
352        }
353        Ok(Some(update))
354    }
