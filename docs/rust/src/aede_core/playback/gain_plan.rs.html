18/// The gain chosen before the DSP enforces output headroom.
19#[derive(Clone, Copy)]
20pub struct GainPlan {
21    /// Requested gain before output headroom is applied.
22    pub gain_db: f32,
23    /// Linear peak of the unamplified source, when known.
24    pub source_peak: Option<f32>,
25    /// Origin of the loudness decision.
26    pub label: &'static str,
27    /// Origin of the peak used for headroom.
28    pub peak_label: &'static str,
29}
30
31fn metadata_plan(selection: normalization::Selection) -> GainPlan {
32    let (label, peak_label) = match selection.source {
33        Source::ReplayGainTrack => ("ReplayGain track", "ReplayGain peak"),
34        Source::ReplayGainAlbum => ("ReplayGain album", "ReplayGain peak"),
35        Source::OpusR128Track => ("Opus R128 track", "assumed full-scale peak"),
36        Source::OpusR128Album => ("Opus R128 album", "assumed full-scale peak"),
37    };
38    GainPlan {
39        gain_db: selection.gain_db,
40        source_peak: selection.source_peak,
41        label,
42        peak_label,
43    }
44}

46fn measured_plan(measurement: Measurement, album: bool, imported: bool) -> GainPlan {
47    GainPlan {
48        gain_db: normalization::DEFAULT_TARGET_LUFS - measurement.integrated_lufs,
49        source_peak: measurement.true_peak,
50        label: if album {
51            "measured album"
52        } else if imported {
53            "FlacCompagnon track"
54        } else {
55            "measured track"
56        },
57        peak_label: if measurement.true_peak.is_some() {
58            "measured true peak"
59        } else {
60            "assumed full-scale peak"
61        },
62    }
63}

65fn is_requested_scope(source: Source, mode: NormalizationMode) -> bool {
66    matches!(
67        (source, mode),
68        (
69            Source::ReplayGainTrack | Source::OpusR128Track,
70            NormalizationMode::Track
71        ) | (
72            Source::ReplayGainAlbum | Source::OpusR128Album,
73            NormalizationMode::Album
74        )
75    )
76}

78fn album_groups(paths: &[PathBuf], is_album: bool, catalog: Option<&Catalog>) -> Vec<Vec<usize>> {
79    if !is_album {
80        return vec![(0..paths.len()).collect()];
81    }
82    let mut groups: Vec<Vec<usize>> = Vec::new();
83    let mut previous = None;
84    for (index, path) in paths.iter().enumerate() {
85        let release = catalog.and_then(|catalog| {
86            catalog.tracks.iter().find_map(|track| {
87                (catalog.file(track.file_id)?.path == path.to_string_lossy())
88                    .then_some(track.release_id)
89                    .flatten()
90            })
91        });
92        if groups.is_empty() || (index > 0 && release != previous) {
93            groups.push(Vec::new());
94        }
95        if let Some(group) = groups.last_mut() {
96            group.push(index);
97        }
98        previous = release;
99    }
100    groups
101}

103/// Explicitly premeasure and cache an ordered playback selection.
104/// This blocking preparation can decode every missing track or whole album.
105/// Interactive CLI and future remote playback use [`ReadyNormalization`].
106pub fn plan_normalization(
107    paths: &[PathBuf],
108    is_album: bool,
109    catalog: Option<&Catalog>,
110    directory: &Path,
111    mode: NormalizationMode,
112) -> Result<Vec<Option<GainPlan>>, Box<dyn Error>> {
113    let mut plans = vec![None; paths.len()];
114    if mode == NormalizationMode::Off {
115        return Ok(plans);
116    }
117    let identities = paths
118        .iter()
119        .map(|path| loudness::identity(path))
120        .collect::<Result<Vec<_>, _>>()?;
121    let tags = paths
122        .iter()
123        .map(|path| tags::read(path))
124        .collect::<Result<Vec<_>, _>>()?;
125    let metadata = tags
126        .iter()
127        .map(|raw| normalization::select_raw(raw, mode, normalization::DEFAULT_TARGET_LUFS))
128        .collect::<Result<Vec<_>, _>>()?;
129    let cache_path = conclusions::conclusions_path(directory);
130    let cache = conclusions::load(&cache_path)?.unwrap_or_default();
131    let mut new_tracks: BTreeMap<String, CachedTrack> = BTreeMap::new();
132    let mut new_programmes = Vec::new();
133    if mode == NormalizationMode::Track {
134        for (index, file) in identities.iter().enumerate() {
135            if let Some(gain) = metadata[index].filter(|gain| is_requested_scope(gain.source, mode))
136            {
137                plans[index] = Some(metadata_plan(gain));
138                continue;
139            }
140            let imported = loudness::from_flaccompagnon(&cache.analyses, file).or_else(|| {
141                catalog.and_then(|catalog| loudness::from_flaccompagnon(&catalog.analyses, file))
142            });
143            let measurement = if let Some(measurement) = imported {
144                Some(measurement)
145            } else if let Some(cached) = new_tracks
146                .get(&file.path)
147                .or_else(|| cache.loudness_tracks.get(&file.path))
148                .filter(|cached| cached.size == file.size && cached.mtime == file.mtime)
149            {
150                cached.measurement
151            } else {
152                let measured = loudness::measure_track(&paths[index])?;
153                new_tracks.insert(
154                    file.path.clone(),
155                    CachedTrack {
156                        size: file.size,
157                        mtime: file.mtime,
158                        measurement: measured,
159                    },
160                );
161                measured
162            };
163            plans[index] = measurement
164                .map(|value| measured_plan(value, false, imported.is_some()))
165                .or_else(|| metadata[index].map(metadata_plan));
166        }
167    } else {
168        for group in album_groups(paths, is_album, catalog) {
169            let has_complete_album_tags = group.iter().all(|&index| {
170                metadata[index].is_some_and(|gain| is_requested_scope(gain.source, mode))
171            });
172            if has_complete_album_tags {
173                for &index in &group {
174                    plans[index] = metadata[index].map(metadata_plan);
175                }
176                continue;
177            }
178            let files = group
179                .iter()
180                .map(|&index| identities[index].clone())
181                .collect::<Vec<ProgrammeFile>>();
182            let measurement = if let Some(cached) = cache
183                .loudness_programmes
184                .iter()
185                .find(|cached| cached.files == files)
186            {
187                cached.measurement
188            } else {
189                let paths = group
190                    .iter()
191                    .map(|&index| paths[index].as_path())
192                    .collect::<Vec<_>>();
193                let measured = loudness::measure_programme(&paths)?;
194                new_programmes.push(CachedProgramme {
195                    files,
196                    measurement: measured,
197                });
198                measured
199            };
200            for &index in &group {
201                plans[index] = measurement
202                    .map(|value| measured_plan(value, true, false))
203                    .or_else(|| metadata[index].map(metadata_plan));
204            }
205        }
206    }
207    if !new_tracks.is_empty() || !new_programmes.is_empty() {
208        let _lock = StoreLock::acquire(directory)?;
209        let mut current = conclusions::load(&cache_path)?
210            .unwrap_or_else(|| catalog.map(Conclusions::from_catalog).unwrap_or_default());
211        current.loudness_tracks.extend(new_tracks);
212        for item in new_programmes {
213            current.loudness_programmes.retain(|cached| {
214                cached
215                    .files
216                    .iter()
217                    .map(|file| &file.path)
218                    .ne(item.files.iter().map(|file| &file.path))
219            });
220            current.loudness_programmes.push(item);
221        }
222        conclusions::save(&current, &cache_path)?;
223    }
224    Ok(plans)
225}
