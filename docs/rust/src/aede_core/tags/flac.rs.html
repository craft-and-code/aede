1//! Reading of FLAC files: STREAMINFO block (properties) and VORBIS_COMMENT
2//! block (tags).
3//!
4//! Reference: <https://xiph.org/flac/format.html>
5
6use std::fs::File;
7
8use super::bytes::{Cursor, read_at};
9use super::{RawTags, TagError};
10
11const BLOCK_STREAMINFO: u8 = 0;
12const BLOCK_VORBIS_COMMENT: u8 = 4;
13const BLOCK_PICTURE: u8 = 6;
14
15/// Reads the tags and the audio properties of an open FLAC file.
16///
17/// `file_size` serves to estimate the bitrate from the length of the audio
18/// data, which the format never states. An ID3v2 tag placed ahead of the
19/// `fLaC` signature by an old encoder is skipped; if no signature turns up
20/// there either, the call fails with [`TagError::UnrecognizedFormat`].
21pub fn read(file: &mut File, file_size: u64) -> Result<RawTags, TagError> {
22    let mut tags = RawTags::default();
23    tags.properties.container = "flac".into();
24    tags.properties.codec = "flac".into();
25    tags.properties.lossless = true;
26
27    // Some encoders place an ID3v2 tag before the fLaC signature.
28    let start = super::id3::skip_id3v2(file)?;
29
30    let header = read_at(file, start, 4)?;
31    if &header[..4] != b"fLaC" {
32        return Err(TagError::UnrecognizedFormat);
33    }
34
35    let mut offset = start + 4;
36    let mut audio_start = offset;
37
38    loop {
39        if offset + 4 > file_size {
40            break;
41        }
42        let head = read_at(file, offset, 4)?;
43        let is_last = head[0] & 0x80 != 0;
44        let block_type = head[0] & 0x7F;
45        let length = ((head[1] as u64) << 16) | ((head[2] as u64) << 8) | head[3] as u64;
46        offset += 4;
47
48        if offset + length > file_size {
49            return Err(TagError::Malformed("truncated FLAC metadata block"));
50        }
51
52        match block_type {
53            BLOCK_STREAMINFO => {
54                let body = read_at(file, offset, length.min(34) as usize)?;
55                read_streaminfo(&body, &mut tags)?;
56            }
57            BLOCK_VORBIS_COMMENT => {
58                // Guard: an oversized comment block is suspect.
59                let capped = length.min(8 * 1024 * 1024) as usize;
60                let body = read_at(file, offset, capped)?;
61                parse_vorbis_comment(&body, &mut tags);
62            }
63            BLOCK_PICTURE => tags.has_embedded_art = true,
64            _ => {}
65        }
66
67        offset += length;
68        audio_start = offset;
69        if is_last {
70            break;
71        }
72    }
73
74    // Actual bitrate: size of the audio alone, excluding metadata.
75    if let Some(ms) = tags.properties.duration_ms
76        && ms > 0
77        && file_size > audio_start
78    {
79        let audio_bytes = file_size - audio_start;
80        tags.properties.bitrate_kbps = Some(((audio_bytes * 8) / ms) as u32);
81    }
82
83    Ok(tags)
84}

86/// STREAMINFO: 34 bytes including a 64-bit field that packs sample rate
87/// (20 b), channels (3 b), bit depth (5 b) and the total number of samples
88/// (36 b).
89fn read_streaminfo(body: &[u8], tags: &mut RawTags) -> Result<(), TagError> {
90    let mut c = Cursor::new(body);
91    c.skip(2 + 2 + 3 + 3); // blocksize min/max, framesize min/max
92    let packed = c
93        .u64_be()
94        .ok_or(TagError::Malformed("STREAMINFO too short"))?;
95
96    let sample_rate = (packed >> 44) as u32 & 0x0F_FFFF;
97    let channels = ((packed >> 41) & 0x07) as u16 + 1;
98    let bit_depth = ((packed >> 36) & 0x1F) as u16 + 1;
99    let total_samples = packed & 0x0F_FFFF_FFFF;
100
101    if sample_rate == 0 {
102        return Err(TagError::Malformed("zero sample rate"));
103    }
104    tags.properties.sample_rate = Some(sample_rate);
105    tags.properties.channels = Some(channels);
106    tags.properties.bit_depth = Some(bit_depth);
107    if total_samples > 0 {
108        tags.properties.duration_ms = Some(total_samples * 1000 / sample_rate as u64);
109    }
110    Ok(())
111}

113/// Vorbis comment block, shared by FLAC, Ogg Vorbis and Opus.
114///
115/// Format: `vendor_len` (u32 LE) + vendor, then `count` (u32 LE) followed by
116/// that many entries of `len` (u32 LE) + `KEY=value` in UTF-8.
117pub(crate) fn parse_vorbis_comment(body: &[u8], tags: &mut RawTags) {
118    let mut c = Cursor::new(body);
119    let Some(vendor_len) = c.u32_le() else { return };
120    c.skip(vendor_len as usize);
121    let Some(count) = c.u32_le() else { return };
122
123    // An absurd count signals a corrupted block: stop cleanly.
124    let count = count.min(10_000);
125    for _ in 0..count {
126        let Some(len) = c.u32_le() else { return };
127        if len as usize > c.remaining() {
128            return;
129        }
130        let Some(entry) = c.utf8(len as usize) else {
131            return;
132        };
133        if let Some((key, value)) = entry.split_once('=') {
134            if key.eq_ignore_ascii_case("metadata_block_picture") {
135                tags.has_embedded_art = true;
136                continue;
137            }
138            tags.insert(key, value);
139        }
140    }
141}
