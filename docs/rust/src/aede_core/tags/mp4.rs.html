1//! MP4 / M4A container: walk over ISO-BMFF boxes, iTunes atoms.
2//!
3//! It carries AAC (lossy) as well as ALAC (lossless); the actual codec is read
4//! from the `stsd` sample description.
5//!
6//! Reference: ISO/IEC 14496-12.
7
8use std::fs::File;
9
10use super::bytes::{Cursor, read_at, read_at_most};
11use super::{RawTags, TagError};
12
13/// Maximum nesting depth: a guard against malicious or circular files.
14const MAX_DEPTH: u8 = 12;
15/// Maximum size of a chunk loaded into memory (cover art included).
16const MAX_BOX_LOAD: u64 = 32 * 1024 * 1024;
17
18/// Reads the tags and the audio properties of an open MP4 / M4A file.
19///
20/// `file_size` bounds the box walk, since a top-level box may declare a length
21/// running past the end of the file. When the `stsd` description names no
22/// codec, the stream falls back to `aac`, the overwhelmingly common case.
23pub fn read(file: &mut File, file_size: u64) -> Result<RawTags, TagError> {
24    let mut tags = RawTags::default();
25    tags.properties.container = "mp4".into();
26
27    walk(file, 0, file_size, 0, &mut tags)?;
28
29    if tags.properties.codec.is_empty() {
30        tags.properties.codec = "aac".into();
31    }
32    Ok(tags)
33}
34
35/// Walks the boxes of a `[start, end)` range.
36fn walk(
37    file: &mut File,
38    start: u64,
39    end: u64,
40    depth: u8,
41    tags: &mut RawTags,
42) -> Result<(), TagError> {
43    if depth > MAX_DEPTH {
44        return Ok(());
45    }
46    let mut offset = start;
47
48    while offset + 8 <= end {
49        let header = read_at(file, offset, 8)?;
50        let mut size = u32::from_be_bytes([header[0], header[1], header[2], header[3]]) as u64;
51        let kind = [header[4], header[5], header[6], header[7]];
52        let mut body = offset + 8;
53
54        if size == 1 {
55            // Size extended to 64 bits.
56            let large = read_at(file, offset + 8, 8)?;
57            size = u64::from_be_bytes([
58                large[0], large[1], large[2], large[3], large[4], large[5], large[6], large[7],
59            ]);
60            body = offset + 16;
61        } else if size == 0 {
62            size = end - offset; // the box runs to the end
63        }
64
65        if size < 8 || offset + size > end {
66            break; // inconsistent structure: stop without failing
67        }
68        let body_end = offset + size;
69
70        match &kind {
71            // Pure containers: descend straight into them.
72            b"moov" | b"trak" | b"mdia" | b"minf" | b"stbl" | b"udta" => {
73                walk(file, body, body_end, depth + 1, tags)?;
74            }
75            // `meta` carries 4 bytes of version/flags before its children.
76            b"meta" => {
77                walk(file, body + 4, body_end, depth + 1, tags)?;
78            }
79            b"mvhd" => {
80                let data = read_at_most(file, body, (body_end - body).min(120) as usize)?;
81                read_mvhd(&data, tags);
82            }
83            b"stsd" => {
84                let len = (body_end - body).min(4096) as usize;
85                let data = read_at_most(file, body, len)?;
86                read_stsd(&data, tags);
87            }
88            b"ilst" if body_end - body <= MAX_BOX_LOAD => {
89                let data = read_at_most(file, body, (body_end - body) as usize)?;
90                read_ilst(&data, tags);
91            }
92            _ => {}
93        }
94
95        offset = body_end;
96    }
97    Ok(())
98}

100fn read_mvhd(data: &[u8], tags: &mut RawTags) {
101    let mut c = Cursor::new(data);
102    let Some(version) = c.u8() else { return };
103    c.skip(3); // flags
104    let (timescale, duration) = if version == 1 {
105        c.skip(16);
106        match (c.u32_be(), c.u64_be()) {
107            (Some(t), Some(d)) => (t, d),
108            _ => return,
109        }
110    } else {
111        c.skip(8);
112        match (c.u32_be(), c.u32_be()) {
113            (Some(t), Some(d)) => (t, d as u64),
114            _ => return,
115        }
116    };
117    if timescale > 0 && duration > 0 {
118        tags.properties.duration_ms = Some(duration * 1000 / timescale as u64);
119    }
120}

122/// Sample description: gives the codec, the channels, the sample rate and, for
123/// ALAC, the real bit depth.
124fn read_stsd(data: &[u8], tags: &mut RawTags) {
125    let mut c = Cursor::new(data);
126    c.skip(4); // version + flags
127    let Some(entries) = c.u32_be() else { return };
128    if entries == 0 {
129        return;
130    }
131    let Some(entry_size) = c.u32_be() else { return };
132    let Some(format) = c.take(4) else { return };
133    let format = [format[0], format[1], format[2], format[3]];
134
135    match &format {
136        b"alac" => {
137            tags.properties.codec = "alac".into();
138            tags.properties.lossless = true;
139        }
140        b"mp4a" => {
141            tags.properties.codec = "aac".into();
142            tags.properties.lossless = false;
143        }
144        b"Opus" => {
145            tags.properties.codec = "opus".into();
146            tags.properties.lossless = false;
147        }
148        b"fLaC" => {
149            tags.properties.codec = "flac".into();
150            tags.properties.lossless = true;
151        }
152        other => {
153            tags.properties.codec = String::from_utf8_lossy(other).trim().to_lowercase();
154        }
155    }
156
157    // AudioSampleEntry
158    c.skip(6); // reserved
159    c.skip(2); // data reference index
160    let Some(version) = c.u16_be() else { return };
161    c.skip(2 + 4); // revision, vendor
162    let channels = c.u16_be();
163    let sample_size = c.u16_be();
164    c.skip(2 + 2); // compression identifier, packet size
165    let sample_rate_fixed = c.u32_be();
166
167    if let Some(ch) = channels
168        && ch > 0
169        && ch <= 64
170    {
171        tags.properties.channels = Some(ch);
172    }
173    if let Some(rate) = sample_rate_fixed {
174        // 16.16 fixed point: the integer part is enough.
175        let rate = rate >> 16;
176        if rate > 0 {
177            tags.properties.sample_rate = Some(rate);
178        }
179    }
180    if let Some(bits) = sample_size
181        && tags.properties.lossless
182        && (8..=32).contains(&bits)
183    {
184        tags.properties.bit_depth = Some(bits);
185    }
186    if version == 1 {
187        c.skip(16);
188    }
189
190    // The ALAC "magic cookie" gives the exact bit depth and sample rate, which
191    // are more reliable than the AudioSampleEntry: it often saturates at
192    // 16 bits / 44.1 kHz.
193    let limit = (entry_size as usize).min(c.remaining() + c.position());
194    while c.position() + 8 <= limit {
195        let Some(size) = c.u32_be() else { break };
196        let Some(kind) = c.take(4) else { break };
197        if size < 8 {
198            break;
199        }
200        let payload_len = (size as usize - 8).min(c.remaining());
201        if kind == b"alac" {
202            let Some(cookie) = c.take(payload_len) else {
203                break;
204            };
205            read_alac_cookie(cookie, tags);
206            break;
207        }
208        c.skip(payload_len);
209    }
210}

212fn read_alac_cookie(data: &[u8], tags: &mut RawTags) {
213    let mut c = Cursor::new(data);
214    c.skip(4); // version/flags
215    c.skip(4); // frameLength
216    c.skip(1); // compatibleVersion
217    let Some(bit_depth) = c.u8() else { return };
218    c.skip(3); // pb, mb, kb
219    let Some(channels) = c.u8() else { return };
220    c.skip(2); // maxRun
221    c.skip(4); // maxFrameBytes
222    let avg_bitrate = c.u32_be();
223    let sample_rate = c.u32_be();
224
225    if (8..=32).contains(&bit_depth) {
226        tags.properties.bit_depth = Some(bit_depth as u16);
227    }
228    if channels > 0 {
229        tags.properties.channels = Some(channels as u16);
230    }
231    if let Some(rate) = sample_rate
232        && rate > 0
233    {
234        tags.properties.sample_rate = Some(rate);
235    }
236    if let Some(bits) = avg_bitrate
237        && bits > 0
238    {
239        tags.properties.bitrate_kbps = Some(bits / 1000);
240    }
241}

243/// iTunes atom list: each entry carries one or more `data` sub-chunks.
244fn read_ilst(data: &[u8], tags: &mut RawTags) {
245    let mut c = Cursor::new(data);
246    while c.remaining() >= 8 {
247        let Some(size) = c.u32_be() else { break };
248        let Some(name) = c.take(4) else { break };
249        let name = [name[0], name[1], name[2], name[3]];
250        if size < 8 {
251            break;
252        }
253        let payload_len = (size as usize - 8).min(c.remaining());
254        let Some(payload) = c.take(payload_len) else {
255            break;
256        };
257
258        if &name == b"----" {
259            read_freeform(payload, tags);
260            continue;
261        }
262        if &name == b"covr" {
263            tags.has_embedded_art = true;
264            continue;
265        }
266        let Some(key) = atom_key(&name) else { continue };
267        for (data_type, value) in data_atoms(payload) {
268            apply_atom(key, data_type, value, tags);
269        }
270    }
271}

273/// Extracts the `data` sub-chunks of an atom: `(type, content)`.
274fn data_atoms(payload: &[u8]) -> Vec<(u32, &[u8])> {
275    let mut out = Vec::new();
276    let mut c = Cursor::new(payload);
277    while c.remaining() >= 8 {
278        let Some(size) = c.u32_be() else { break };
279        let Some(kind) = c.take(4) else { break };
280        if size < 16 {
281            break;
282        }
283        let inner_len = (size as usize - 8).min(c.remaining());
284        let start = c.position();
285        c.skip(inner_len);
286        if kind != b"data" {
287            continue;
288        }
289        let inner = &payload[start..start + inner_len];
290        if inner.len() < 8 {
291            continue;
292        }
293        let data_type = u32::from_be_bytes([inner[0], inner[1], inner[2], inner[3]]) & 0x00FF_FFFF;
294        out.push((data_type, &inner[8..]));
295    }
296    out
297}

299fn apply_atom(key: &str, data_type: u32, value: &[u8], tags: &mut RawTags) {
300    match key {
301        "tracknumber" | "discnumber" => {
302            // 2 reserved bytes, then number and total on 2 bytes each.
303            if value.len() >= 6 {
304                let number = u16::from_be_bytes([value[2], value[3]]);
305                let total = u16::from_be_bytes([value[4], value[5]]);
306                if number > 0 {
307                    tags.insert(key, number.to_string());
308                }
309                if total > 0 {
310                    let total_key = if key == "tracknumber" {
311                        "tracktotal"
312                    } else {
313                        "disctotal"
314                    };
315                    tags.insert(total_key, total.to_string());
316                }
317            }
318        }
319        "gnre" => {
320            // Numeric genre inherited from ID3v1 (index shifted by 1).
321            if value.len() >= 2 {
322                let index = u16::from_be_bytes([value[0], value[1]]);
323                if index > 0
324                    && let Some(name) = id3v1_genre(index - 1)
325                {
326                    tags.insert("genre", name);
327                }
328            }
329        }
330        "compilation" => {
331            let flag = value.first().copied().unwrap_or(0);
332            if flag != 0 {
333                tags.insert("compilation", "1");
334            }
335        }
336        _ => match data_type {
337            1 => tags.insert(key, String::from_utf8_lossy(value).into_owned()),
338            21 | 22 => {
339                let mut number: u64 = 0;
340                for &b in value.iter().take(8) {
341                    number = (number << 8) | b as u64;
342                }
343                tags.insert(key, number.to_string());
344            }
345            13 | 14 => tags.has_embedded_art = true,
346            _ => tags.insert(key, String::from_utf8_lossy(value).into_owned()),
347        },
348    }
349}

351/// `----` atom: free-form key, shaped as `mean` (domain) + `name` (key).
352fn read_freeform(payload: &[u8], tags: &mut RawTags) {
353    let mut c = Cursor::new(payload);
354    let mut key: Option<String> = None;
355
356    while c.remaining() >= 8 {
357        let Some(size) = c.u32_be() else { break };
358        let Some(kind) = c.take(4) else { break };
359        if size < 8 {
360            break;
361        }
362        let inner_len = (size as usize - 8).min(c.remaining());
363        let start = c.position();
364        c.skip(inner_len);
365        let inner = &payload[start..start + inner_len];
366
367        match kind {
368            b"name" if inner.len() > 4 => {
369                key = Some(String::from_utf8_lossy(&inner[4..]).trim().to_string());
370            }
371            b"data" if inner.len() > 8 => {
372                if let Some(ref k) = key {
373                    let canonical = freeform_key(k);
374                    tags.insert(
375                        &canonical,
376                        String::from_utf8_lossy(&inner[8..]).into_owned(),
377                    );
378                }
379            }
380            _ => {}
381        }
382    }
383}

385fn freeform_key(name: &str) -> String {
386    let lower = name.trim().to_lowercase();
387    match lower.as_str() {
388        "musicbrainz album id" => "musicbrainz_albumid".into(),
389        "musicbrainz artist id" => "musicbrainz_artistid".into(),
390        "musicbrainz album artist id" => "musicbrainz_albumartistid".into(),
391        "musicbrainz release group id" => "musicbrainz_releasegroupid".into(),
392        "musicbrainz track id" => "musicbrainz_recordingid".into(),
393        "musicbrainz work id" => "musicbrainz_workid".into(),
394        "musicbrainz label id" => "musicbrainz_labelid".into(),
395        "acoustid id" => "acoustid_id".into(),
396        "catalognumber" => "catalognumber".into(),
397        "label" => "label".into(),
398        "isrc" => "isrc".into(),
399        "barcode" => "barcode".into(),
400        other => super::canonical_key(other),
401    }
402}

404fn atom_key(name: &[u8; 4]) -> Option<&'static str> {
405    // The © prefix is 0xA9 in ISO-8859-1.
406    Some(match name {
407        b"\xa9nam" => "title",
408        b"\xa9ART" => "artist",
409        b"aART" => "albumartist",
410        b"\xa9alb" => "album",
411        b"\xa9day" => "date",
412        b"\xa9gen" => "genre",
413        b"gnre" => "gnre",
414        b"trkn" => "tracknumber",
415        b"disk" => "discnumber",
416        b"\xa9wrt" => "composer",
417        b"\xa9cmt" => "comment",
418        b"\xa9lyr" => "lyrics",
419        b"\xa9grp" => "grouping",
420        b"cprt" => "copyright",
421        b"cpil" => "compilation",
422        b"tmpo" => "bpm",
423        b"soar" => "artistsort",
424        b"soaa" => "albumartistsort",
425        b"soal" => "albumsort",
426        b"\xa9too" => "encodedby",
427        b"\xa9wrk" => "work",
428        b"\xa9mvn" => "movement",
429        _ => return None,
430    })
431}

433fn id3v1_genre(index: u16) -> Option<&'static str> {
434    // The ID3 table is reused rather than duplicated here: delegating keeps a
435    // single source of truth for the genre names.
436    super::id3::genre_name_public(index)
437}
