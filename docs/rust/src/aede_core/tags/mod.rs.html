1//! Reading of metadata and audio properties.
2//!
3//! Each format has its own submodule. They all converge on [`RawTags`], whose
4//! keys are normalised on the Vorbis Comment vocabulary (`title`, `artist`,
5//! `albumartist`, `date`…): it is the cleanest common denominator, and the one
6//! MusicBrainz uses.
7//!
8//! The formats a library is actually made of are parsed here, from the
9//! specifications: it is what allows the encoder delay, the ALAC magic cookie
10//! and the Opus pre-skip to be extracted, none of which a general-purpose
11//! library exposes. Anything else falls through to [`foreign`].
12

25/// Technical properties of the audio stream.
26#[derive(Debug, Clone, Default, PartialEq)]
27pub struct AudioProperties {
28    /// Stream codec: `flac`, `mp3`, `alac`, `aac`, `vorbis`, `opus`, `pcm`.
29    pub codec: String,
30    /// Container: `flac`, `mp3`, `mp4`, `ogg`, `wav`, `aiff`.
31    pub container: String,
32    /// Sampling frequency in hertz, as declared by the stream header.
33    pub sample_rate: Option<u32>,
34    /// Bits per sample. Always absent for a lossy codec, which has no such
35    /// notion.
36    pub bit_depth: Option<u16>,
37    /// Number of channels: 1 for mono, 2 for stereo, more for multichannel.
38    pub channels: Option<u16>,
39    /// Playing time in milliseconds. Absent when the file is truncated or its
40    /// header is unusable.
41    pub duration_ms: Option<u64>,
42    /// Bitrate in kbit/s: measured when the container states it, estimated from
43    /// the size of the audio data otherwise.
44    pub bitrate_kbps: Option<u32>,
45    /// Whether the encoding preserves the original signal. Decides which
46    /// quality tier the file lands in.
47    pub lossless: bool,
48}
49
50impl AudioProperties {
51    /// `true` if the stream goes beyond CD format (44.1 kHz / 16 bits).
52    pub fn is_hi_res(&self) -> bool {
53        self.lossless
54            && (self.sample_rate.unwrap_or(0) > 48_000 || self.bit_depth.unwrap_or(0) > 16)
55    }
56
57    /// Readable label: `FLAC 24/96`, `MP3 320`, `AAC`…
58    pub fn quality_label(&self) -> String {
59        let codec = self.codec.to_uppercase();
60        match (self.lossless, self.bit_depth, self.sample_rate) {
61            (true, Some(bits), Some(rate)) => {
62                let khz = rate as f64 / 1000.0;
63                if khz.fract() == 0.0 {
64                    format!("{codec} {bits}/{}", khz as u32)
65                } else {
66                    format!("{codec} {bits}/{khz:.1}")
67                }
68            }
69            _ => match self.bitrate_kbps {
70                Some(kbps) => format!("{codec} {kbps}"),
71                None => codec,
72            },
73        }
74    }
75}

77/// Raw metadata read from a file, before any entity matching.
78#[derive(Debug, Clone, Default)]
79pub struct RawTags {
80    /// Normalised fields; a key may carry several values
81    /// (several artists, several genres…).
82    pub fields: BTreeMap<String, Vec<String>>,
83    /// Technical characteristics of the stream, filled in even when the file
84    /// carries no tag at all.
85    pub properties: AudioProperties,
86    /// Is cover art embedded in the file?
87    pub has_embedded_art: bool,
88}
89
90impl RawTags {
91    /// Adds a value under `key`, after normalising the key to the Vorbis
92    /// Comment vocabulary.
93    ///
94    /// The value is trimmed of whitespace and of the null bytes that some
95    /// taggers leave behind; if nothing remains, or if the same value is
96    /// already present under that key, the call is a no-op.
97    pub fn insert(&mut self, key: &str, value: impl Into<String>) {
98        let value: String = value.into();
99        let value = value
100            .trim_matches(|c: char| c == '\0' || c.is_whitespace())
101            .to_string();
102        if value.is_empty() {
103            return;
104        }
105        let entry = self.fields.entry(canonical_key(key)).or_default();
106        if !entry.contains(&value) {
107            entry.push(value);
108        }
109    }
110
111    /// First value of a field.
112    pub fn first(&self, key: &str) -> Option<&str> {
113        self.fields
114            .get(key)
115            .and_then(|v| v.first())
116            .map(|s| s.as_str())
117    }
118
119    /// All values of a field.
120    pub fn all(&self, key: &str) -> &[String] {
121        self.fields.get(key).map(|v| v.as_slice()).unwrap_or(&[])
122    }
123
124    /// First value among several candidate keys, in the given order.
125    pub fn first_of(&self, keys: &[&str]) -> Option<&str> {
126        keys.iter().find_map(|k| self.first(k))
127    }
128
129    /// `true` when no textual field was found. The audio properties may still
130    /// be filled in: an untagged file is not an unreadable one.
131    pub fn is_empty(&self) -> bool {
132        self.fields.is_empty()
133    }
134}

136/// Error while reading an audio file.
137#[derive(Debug)]
138pub enum TagError {
139    /// The file could not be opened or a read failed part-way through; the
140    /// content itself is not implicated.
141    Io(std::io::Error),
142    /// The file does not carry the signature expected for its extension.
143    UnrecognizedFormat,
144    /// Inconsistent internal structure (truncated or corrupted file).
145    Malformed(&'static str),
146}
147
148impl std::fmt::Display for TagError {
149    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
150        match self {
151            TagError::Io(e) => write!(f, "input/output error: {e}"),
152            TagError::UnrecognizedFormat => write!(f, "unrecognised file signature"),
153            TagError::Malformed(what) => write!(f, "malformed file: {what}"),
154        }
155    }
156}
157
158impl std::error::Error for TagError {}
159
160impl From<std::io::Error> for TagError {
161    fn from(e: std::io::Error) -> Self {
162        TagError::Io(e)
163    }
164}

166/// Extensions treated as audio tracks.
167pub const AUDIO_EXTENSIONS: &[&str] = &[
168    // Parsed here.
169    "flac", "mp3", "m4a", "m4b", "mp4", "alac", "ogg", "oga", "opus", "wav", "wave", "aif", "aiff",
170    "aifc", // Handed to `lofty`.
171    "aac", "ape", "wv", "mpc", "mp+", "mpp", "spx",
172];

174/// `true` if the path extension is one of the supported audio formats.
175pub fn is_audio_path(path: &Path) -> bool {
176    match path.extension().and_then(|e| e.to_str()) {
177        Some(ext) => {
178            let ext = ext.to_ascii_lowercase();
179            AUDIO_EXTENSIONS.contains(&ext.as_str())
180        }
181        None => false,
182    }
183}

185/// Reads the tags and properties of an audio file.
186///
187/// Dispatch is driven first by the **actual signature** of the file, and only
188/// then by the extension: a real library always contains a few `.mp3` files
189/// that are in fact FLAC.
190pub fn read(path: &Path) -> Result<RawTags, TagError> {
191    let mut file = std::fs::File::open(path)?;
192    let size = file.metadata()?.len();
193    let magic = read_magic(&mut file)?;
194
195    let native = match detect(&magic) {
196        Some(Container::Flac) => flac::read(&mut file, size),
197        Some(Container::Ogg) => ogg::read(&mut file, size),
198        Some(Container::Mp4) => mp4::read(&mut file, size),
199        Some(Container::Riff) => riff::read_wav(&mut file, size),
200        Some(Container::Aiff) => riff::read_aiff(&mut file, size),
201        Some(Container::Mp3) => mp3::read(&mut file, size),
202        None => Err(TagError::UnrecognizedFormat),
203    };
204
205    // A signature none of the parsers claims is not necessarily an unreadable
206    // file: it may simply be one of the formats left to `lofty`. A parser that
207    // recognised the file and then failed keeps its own diagnosis, which is
208    // more precise than anything the fallback would say.
209    let mut tags = match native {
210        Err(TagError::UnrecognizedFormat) => foreign::read(path)?,
211        other => other?,
212    };
213
214    // Bitrate approximated from the total size, for the formats that do not
215    // announce it themselves.
216    if tags.properties.bitrate_kbps.is_none() {
217        tags.properties.bitrate_kbps = tags
218            .properties
219            .duration_ms
220            .and_then(|ms| (size * 8).checked_div(ms))
221            .map(|kbps| kbps as u32);
222    }
223    Ok(tags)
224}

226enum Container {
227    Flac,
228    Ogg,
229    Mp4,
230    Riff,
231    Aiff,
232    Mp3,
233}
234
235fn detect(magic: &[u8; 16]) -> Option<Container> {
236    if &magic[0..4] == b"fLaC" {
237        return Some(Container::Flac);
238    }
239    if &magic[0..4] == b"OggS" {
240        return Some(Container::Ogg);
241    }
242    if &magic[4..8] == b"ftyp" {
243        return Some(Container::Mp4);
244    }
245    if &magic[0..4] == b"RIFF" && &magic[8..12] == b"WAVE" {
246        return Some(Container::Riff);
247    }
248    if &magic[0..4] == b"FORM" && (&magic[8..12] == b"AIFF" || &magic[8..12] == b"AIFC") {
249        return Some(Container::Aiff);
250    }
251    // A leading ID3v2 tag may precede MP3 as well as FLAC; mp3::read handles
252    // both cases by resynchronising after the tag.
253    if &magic[0..3] == b"ID3" {
254        return Some(Container::Mp3);
255    }
256    // An MPEG frame and an ADTS frame share the same eleven-bit sync word and
257    // are told apart by the layer field, which ADTS leaves at zero — a value
258    // MPEG reserves. Without this, raw AAC would be handed to the MP3 parser.
259    if magic[0] == 0xFF && (magic[1] & 0xE0) == 0xE0 && (magic[1] & 0x06) != 0 {
260        return Some(Container::Mp3);
261    }
262    None
263}

265fn read_magic(file: &mut std::fs::File) -> Result<[u8; 16], TagError> {
266    use std::io::{Read, Seek, SeekFrom};
267    let mut magic = [0u8; 16];
268    let read = read_up_to(file, &mut magic)?;
269    if read < 12 {
270        return Err(TagError::UnrecognizedFormat);
271    }
272    file.seek(SeekFrom::Start(0))?;
273    return Ok(magic);
274
275    fn read_up_to(file: &mut std::fs::File, buf: &mut [u8]) -> std::io::Result<usize> {
276        let mut total = 0;
277        while total < buf.len() {
278            match file.read(&mut buf[total..])? {
279                0 => break,
280                n => total += n,
281            }
282        }
283        Ok(total)
284    }
285}
