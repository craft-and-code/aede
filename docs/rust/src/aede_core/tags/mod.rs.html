1//! Reading of metadata and audio properties.
2//!
3//! Each format has its own submodule. They all converge on [`RawTags`], whose
4//! keys are normalised on the Vorbis Comment vocabulary (`title`, `artist`,
5//! `albumartist`, `date`…): it is the cleanest common denominator, and the one
6//! MusicBrainz uses.
7//!
8//! The formats a library is actually made of are parsed here, from the
9//! specifications: it is what allows the encoder delay, the ALAC magic cookie
10//! and the Opus pre-skip to be extracted, none of which a general-purpose
11//! library exposes. Anything else falls through to [`foreign`].
12

29/// Technical properties of the audio stream.
30#[derive(Debug, Clone, Default, PartialEq)]
31pub struct AudioProperties {
32    /// Stream codec: `flac`, `mp3`, `alac`, `aac`, `vorbis`, `opus`, `pcm`.
33    pub codec: String,
34    /// Container: `flac`, `mp3`, `mp4`, `ogg`, `wav`, `aiff`.
35    pub container: String,
36    /// Sampling frequency in hertz, as declared by the stream header.
37    pub sample_rate: Option<u32>,
38    /// Bits per sample. Always absent for a lossy codec, which has no such
39    /// notion.
40    pub bit_depth: Option<u16>,
41    /// Number of channels: 1 for mono, 2 for stereo, more for multichannel.
42    pub channels: Option<u16>,
43    /// Playing time in milliseconds. Absent when the file is truncated or its
44    /// header is unusable.
45    pub duration_ms: Option<u64>,
46    /// Bitrate in kbit/s: measured when the container states it, estimated from
47    /// the size of the audio data otherwise.
48    pub bitrate_kbps: Option<u32>,
49    /// Whether the encoding preserves the original signal. Decides which
50    /// quality tier the file lands in.
51    pub lossless: bool,
52}
53
54impl AudioProperties {
55    /// `true` if the stream goes beyond CD format (44.1 kHz / 16 bits).
56    pub fn is_hi_res(&self) -> bool {
57        self.lossless
58            && (self.sample_rate.unwrap_or(0) > 48_000 || self.bit_depth.unwrap_or(0) > 16)
59    }
60
61    /// Readable label: `FLAC 24/96`, `MP3 320`, `AAC`…
62    pub fn quality_label(&self) -> String {
63        let codec = self.codec.to_uppercase();
64        match (self.lossless, self.bit_depth, self.sample_rate) {
65            (true, Some(bits), Some(rate)) => {
66                let khz = rate as f64 / 1000.0;
67                if khz.fract() == 0.0 {
68                    format!("{codec} {bits}/{}", khz as u32)
69                } else {
70                    format!("{codec} {bits}/{khz:.1}")
71                }
72            }
73            _ => match self.bitrate_kbps {
74                Some(kbps) => format!("{codec} {kbps}"),
75                None => codec,
76            },
77        }
78    }
79}

81/// Raw metadata read from a file, before any entity matching.
82#[derive(Debug, Clone, Default)]
83pub struct RawTags {
84    /// Normalised fields; a key may carry several values
85    /// (several artists, several genres…).
86    pub fields: BTreeMap<String, Vec<String>>,
87    /// Technical characteristics of the stream, filled in even when the file
88    /// carries no tag at all.
89    pub properties: AudioProperties,
90    /// Is cover art embedded in the file?
91    pub has_embedded_art: bool,
92}
93
94impl RawTags {
95    /// Adds a value under `key`, after normalising the key to the Vorbis
96    /// Comment vocabulary.
97    ///
98    /// The value is trimmed of whitespace and of the null bytes that some
99    /// taggers leave behind; if nothing remains, or if the same value is
100    /// already present under that key, the call is a no-op.
101    pub fn insert(&mut self, key: &str, value: impl Into<String>) {
102        let value: String = value.into();
103        let value = value
104            .trim_matches(|c: char| c == '\0' || c.is_whitespace())
105            .to_string();
106        if value.is_empty() {
107            return;
108        }
109        let entry = self.fields.entry(canonical_key(key)).or_default();
110        if !entry.contains(&value) {
111            entry.push(value);
112        }
113    }
114
115    /// First value of a field.
116    pub fn first(&self, key: &str) -> Option<&str> {
117        self.fields
118            .get(key)
119            .and_then(|v| v.first())
120            .map(|s| s.as_str())
121    }
122
123    /// All values of a field.
124    pub fn all(&self, key: &str) -> &[String] {
125        self.fields.get(key).map(|v| v.as_slice()).unwrap_or(&[])
126    }
127
128    /// First value among several candidate keys, in the given order.
129    pub fn first_of(&self, keys: &[&str]) -> Option<&str> {
130        keys.iter().find_map(|k| self.first(k))
131    }
132
133    /// `true` when no textual field was found. The audio properties may still
134    /// be filled in: an untagged file is not an unreadable one.
135    pub fn is_empty(&self) -> bool {
136        self.fields.is_empty()
137    }
138}

140/// Error while reading an audio file.
141#[derive(Debug)]
142pub enum TagError {
143    /// The file could not be opened or a read failed part-way through; the
144    /// content itself is not implicated.
145    Io(std::io::Error),
146    /// The file does not carry the signature expected for its extension.
147    UnrecognizedFormat,
148    /// Inconsistent internal structure (truncated or corrupted file).
149    Malformed(&'static str),
150}
151
152impl std::fmt::Display for TagError {
153    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
154        match self {
155            TagError::Io(e) => write!(f, "input/output error: {e}"),
156            TagError::UnrecognizedFormat => write!(f, "unrecognised file signature"),
157            TagError::Malformed(what) => write!(f, "malformed file: {what}"),
158        }
159    }
160}
161
162impl std::error::Error for TagError {}
163
164impl From<std::io::Error> for TagError {
165    fn from(e: std::io::Error) -> Self {
166        TagError::Io(e)
167    }
168}

178/// `true` if the path extension is one of the supported audio formats.
179pub fn is_audio_path(path: &Path) -> bool {
180    match path.extension().and_then(|e| e.to_str()) {
181        Some(ext) => {
182            let ext = ext.to_ascii_lowercase();
183            AUDIO_EXTENSIONS.contains(&ext.as_str())
184        }
185        None => false,
186    }
187}

189/// Reads the tags and properties of an audio file.
190///
191/// Dispatch is driven first by the **actual signature** of the file, and only
192/// then by the extension: a real library always contains a few `.mp3` files
193/// that are in fact FLAC.
194pub fn read(path: &Path) -> Result<RawTags, TagError> {
195    let mut file = std::fs::File::open(path)?;
196    read_opened(&mut file, |_| foreign::read(path))
197}
198
199/// Read metadata from an already opened regular file without reopening a path.
200///
201/// The descriptor is rewound first; native and fallback parsers inspect that
202/// same descriptor by signature. This lets callers retain their source identity
203/// checks across path replacement. It does not acquire or validate a filesystem
204/// identity for the caller, and leaves the descriptor at the parser's position.
205pub fn read_from_file(file: &mut std::fs::File) -> Result<RawTags, TagError> {
206    use std::io::{Seek, SeekFrom};
207    if !file.metadata()?.is_file() {
208        return Err(TagError::Io(std::io::Error::new(
209            std::io::ErrorKind::InvalidInput,
210            "metadata descriptor must be a regular file",
211        )));
212    }
213    file.seek(SeekFrom::Start(0))?;
214    read_opened(file, foreign::read_from_file)
215}
216
217fn read_opened(
218    file: &mut std::fs::File,
219    fallback: impl FnOnce(&mut std::fs::File) -> Result<RawTags, TagError>,
220) -> Result<RawTags, TagError> {
221    let size = file.metadata()?.len();
222    let magic = read_magic(file)?;
223
224    let native = match detect(&magic) {
225        Some(Container::Flac) => flac::read(file, size),
226        Some(Container::Ogg) => ogg::read(file, size),
227        Some(Container::Mp4) => mp4::read(file, size),
228        Some(Container::Riff) => riff::read_wav(file, size),
229        Some(Container::Aiff) => riff::read_aiff(file, size),
230        Some(Container::Mp3) => mp3::read(file, size),
231        None => Err(TagError::UnrecognizedFormat),
232    };
233
234    // A signature none of the parsers claims is not necessarily an unreadable
235    // file: it may simply be one of the formats left to `lofty`. A parser that
236    // recognised the file and then failed keeps its own diagnosis, which is
237    // more precise than anything the fallback would say.
238    let mut tags = match native {
239        Err(TagError::UnrecognizedFormat) => fallback(file)?,
240        other => other?,
241    };
242
243    // Bitrate approximated from the total size, for the formats that do not
244    // announce it themselves.
245    if tags.properties.bitrate_kbps.is_none() {
246        tags.properties.bitrate_kbps = tags
247            .properties
248            .duration_ms
249            .and_then(|ms| (size * 8).checked_div(ms))
250            .map(|kbps| kbps as u32);
251    }
252    Ok(tags)
253}
254
255enum Container {
256    Flac,
257    Ogg,
258    Mp4,
259    Riff,
260    Aiff,
261    Mp3,
262}
263
264fn detect(magic: &[u8; 16]) -> Option<Container> {
265    if &magic[0..4] == b"fLaC" {
266        return Some(Container::Flac);
267    }
268    if &magic[0..4] == b"OggS" {
269        return Some(Container::Ogg);
270    }
271    if &magic[4..8] == b"ftyp" {
272        return Some(Container::Mp4);
273    }
274    if &magic[0..4] == b"RIFF" && &magic[8..12] == b"WAVE" {
275        return Some(Container::Riff);
276    }
277    if &magic[0..4] == b"FORM" && (&magic[8..12] == b"AIFF" || &magic[8..12] == b"AIFC") {
278        return Some(Container::Aiff);
279    }
280    // A leading ID3v2 tag may precede MP3 as well as FLAC; mp3::read handles
281    // both cases by resynchronising after the tag.
282    if &magic[0..3] == b"ID3" {
283        return Some(Container::Mp3);
284    }
285    // An MPEG frame and an ADTS frame share the same eleven-bit sync word and
286    // are told apart by the layer field, which ADTS leaves at zero — a value
287    // MPEG reserves. Without this, raw AAC would be handed to the MP3 parser.
288    if magic[0] == 0xFF && (magic[1] & 0xE0) == 0xE0 && (magic[1] & 0x06) != 0 {
289        return Some(Container::Mp3);
290    }
291    None
292}

294fn read_magic(file: &mut std::fs::File) -> Result<[u8; 16], TagError> {
295    use std::io::{Read, Seek, SeekFrom};
296    let mut magic = [0u8; 16];
297    let read = read_up_to(file, &mut magic)?;
298    if read < 12 {
299        return Err(TagError::UnrecognizedFormat);
300    }
301    file.seek(SeekFrom::Start(0))?;
302    return Ok(magic);
303
304    fn read_up_to(file: &mut std::fs::File, buf: &mut [u8]) -> std::io::Result<usize> {
305        let mut total = 0;
306        while total < buf.len() {
307            match file.read(&mut buf[total..])? {
308                0 => break,
309                n => total += n,
310            }
311        }
312        Ok(total)
313    }
314}
