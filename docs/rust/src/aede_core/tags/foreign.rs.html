1//! Containers Aede has no parser of its own for, read through `lofty`.
2//!
3//! The hand-written parsers cover the formats a music library is actually made
4//! of, and they extract things a general-purpose library does not expose — the
5//! LAME encoder delay, the ALAC magic cookie, the Opus pre-skip. They stay in
6//! charge of those formats.
7//!
8//! What is left over is the long tail: WavPack, Monkey's Audio, Musepack,
9//! Speex, raw AAC. Writing a parser for each would cost weeks and would be
10//! exercised by a handful of files. This module hands them to `lofty` instead,
11//! and maps what comes back onto [`RawTags`], so the rest of the program never
12//! learns that two different readers exist.
13//!
14//! The fallback is only reached when the signature matches none of the native
15//! parsers, so it can never take a format away from them.
16

28/// Reads a file through `lofty` and translates the result into [`RawTags`].
29pub fn read(path: &Path) -> Result<RawTags, TagError> {
30    let probe = Probe::open(path).map_err(|_| TagError::UnrecognizedFormat)?;
31    read_probe(probe)
32}
33
34pub(super) fn read_from_file(file: &mut std::fs::File) -> Result<RawTags, TagError> {
35    use std::io::{Seek, SeekFrom};
36    file.seek(SeekFrom::Start(0))?;
37    read_probe(Probe::new(file))
38}
39
40fn read_probe<R: std::io::Read + std::io::Seek>(probe: Probe<R>) -> Result<RawTags, TagError> {
41    let probe = probe
42        .guess_file_type()
43        .map_err(|_| TagError::UnrecognizedFormat)?;
44    if probe.file_type() == Some(FileType::WavPack) {
45        let mut reader = probe.into_inner();
46        // The first audio block's FLOAT_DATA flag is not exposed by Lofty.
47        let flags = wavpack_audio_flags(&mut reader)?;
48        reader
49            .seek(std::io::SeekFrom::Start(0))
50            .map_err(|_| TagError::UnrecognizedFormat)?;
51        let wavpack = WavPackFile::read_from(&mut reader, ParseOptions::default())
52            .map_err(|_| TagError::UnrecognizedFormat)?;
53        // Generic FileProperties drops WavPack's hybrid/lossless distinction.
54        let lossless = wavpack.properties().is_lossless();
55        let mut tags = translate(wavpack.into());
56        tags.properties.lossless = lossless;
57        // Shared properties cannot describe compressed floating-point PCM;
58        // unknown precision makes lossless exporters refuse quantization.
59        if !lossless || flags.is_none_or(|flags| flags & 0x80 != 0) {
60            tags.properties.bit_depth = None;
61        }
62        return Ok(tags);
63    }
64    let file = probe.read().map_err(|_| TagError::UnrecognizedFormat)?;
65    Ok(translate(file))
66}

68/// Reads only headers, bounding non-audio prefixes without loading their bodies.
69/// Unknown flags must not be treated as evidence of integer PCM precision.
70fn wavpack_audio_flags<R: std::io::Read + std::io::Seek>(
71    reader: &mut R,
72) -> Result<Option<u32>, TagError> {
73    let mut offset = 0u64;
74    for _ in 0..64 {
75        reader.seek(std::io::SeekFrom::Start(offset))?;
76        let mut bytes = [0; 32];
77        match reader.read_exact(&mut bytes) {
78            Ok(()) => {}
79            Err(error) if error.kind() == std::io::ErrorKind::UnexpectedEof => return Ok(None),
80            Err(error) => return Err(error.into()),
81        }
82        let mut header = super::bytes::Cursor::new(&bytes);
83        if header.take(4) != Some(b"wvpk".as_slice()) {
84            return Ok(None);
85        }
86        let Some(size @ 24..=1_048_576) = header.u32_le() else {
87            return Ok(None);
88        };
89        header.skip(12);
90        let (Some(samples), Some(flags)) = (header.u32_le(), header.u32_le()) else {
91            return Ok(None);
92        };
93        if samples > 0 {
94            return Ok(Some(flags));
95        }
96        offset = offset
97            .checked_add(u64::from(size) + 8)
98            .ok_or(TagError::Malformed("WavPack block offset overflow"))?;
99    }
100    Ok(None)
101}

103fn translate(file: TaggedFile) -> RawTags {
104    let mut tags = RawTags::default();
105
106    describe_stream(file.file_type(), &mut tags);
107    let properties = file.properties();
108    tags.properties.sample_rate = properties.sample_rate();
109    tags.properties.channels = properties.channels().map(u16::from);
110    if tags.properties.lossless {
111        tags.properties.bit_depth = properties.bit_depth().map(u16::from);
112    }
113    let duration = properties.duration().as_millis();
114    if duration > 0 {
115        tags.properties.duration_ms = Some(duration as u64);
116    }
117    tags.properties.bitrate_kbps = properties.audio_bitrate().or(properties.overall_bitrate());
118
119    // Catalogue readers take the first value. Lofty does not always list the
120    // format's primary tag first; keep it ahead of legacy alternatives while
121    // retaining their unique values and fields missing from the primary tag.
122    let primary = file.primary_tag_type();
123    for tag in file
124        .tags()
125        .iter()
126        .filter(|tag| tag.tag_type() == primary)
127        .chain(file.tags().iter().filter(|tag| tag.tag_type() != primary))
128    {
129        collect(tag, &mut tags);
130    }
131    tags
132}

134/// Fills in codec, container and losslessness from the detected file type.
135///
136/// Names follow the ones the native parsers use, since they end up in the same
137/// statistics and in the same `codec` column of the catalog.
138fn describe_stream(file_type: FileType, tags: &mut RawTags) {
139    let (codec, container, lossless) = match file_type {
140        FileType::Aac => ("aac", "adts", false),
141        FileType::Ape => ("ape", "ape", true),
142        FileType::Mpc => ("musepack", "musepack", false),
143        FileType::WavPack => ("wavpack", "wavpack", true),
144        FileType::Speex => ("speex", "ogg", false),
145        FileType::Flac => ("flac", "flac", true),
146        FileType::Mpeg => ("mp3", "mp3", false),
147        FileType::Mp4 => ("aac", "mp4", false),
148        FileType::Opus => ("opus", "ogg", false),
149        FileType::Vorbis => ("vorbis", "ogg", false),
150        FileType::Wav => ("pcm", "wav", true),
151        FileType::Aiff => ("pcm", "aiff", true),
152        // A resolver registered by a third party. Nothing sensible to say
153        // about the stream, but its tags are still worth keeping.
154        _ => ("", "", false),
155    };
156    tags.properties.codec = codec.to_string();
157    tags.properties.container = container.to_string();
158    tags.properties.lossless = lossless;
159}

161/// Copies the textual items and the presence of cover art out of one tag.
162fn collect(tag: &Tag, tags: &mut RawTags) {
163    if tag.picture_count() > 0 {
164        tags.has_embedded_art = true;
165    }
166    for item in tag.items() {
167        let Some(text) = item.value().text() else {
168            continue;
169        };
170        if let Some(key) = field_name(item.key()) {
171            tags.insert(key, text);
172        }
173    }
174}

176/// Spelling of a key in the vocabulary the rest of the program speaks.
177///
178/// `lofty` normalises every format onto its own [`ItemKey`] enumeration; asking
179/// it for the Vorbis Comment spelling brings the value back into the vocabulary
180/// [`super::canonical_key`] already understands. `ItemKey` carries no variant
181/// for a key it does not know, so a tag `lofty` did not recognise is lost here —
182/// the alternative being a file that cannot be read at all.
183fn field_name(key: ItemKey) -> Option<&'static str> {
184    key.map_key(TagType::VorbisComments)
185        .or_else(|| key.map_key(TagType::Ape))
186}
