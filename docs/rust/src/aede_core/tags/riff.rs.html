1//! WAV (RIFF) and AIFF (IFF).
2//!
3//! Two very similar chunk containers, one little-endian, the other
4//! big-endian. Both can carry a full ID3v2 chunk on top of their own fields,
5//! which are historically poor.
6
7use std::fs::File;
8
9use super::bytes::{Cursor, extended80_to_f64, read_at_most};
10use super::{RawTags, TagError};
11
12/// Maximum size of a metadata chunk loaded into memory.
13const MAX_META_CHUNK: usize = 16 * 1024 * 1024;
14
15/// Reads the tags and the audio properties of an open WAV file.
16///
17/// `file_size` bounds the chunk walk rather than the `RIFF` header, which is
18/// routinely wrong on files written by a recorder that never came back to fix
19/// it. Tags are gathered from the `LIST`/`INFO` chunk and from an embedded
20/// ID3v2 chunk alike; both contribute to the same fields.
21pub fn read_wav(file: &mut File, file_size: u64) -> Result<RawTags, TagError> {
22    let mut tags = RawTags::default();
23    tags.properties.container = "wav".into();
24    tags.properties.codec = "pcm".into();
25    tags.properties.lossless = true;
26
27    let mut byte_rate = 0u32;
28    let mut offset = 12u64; // "RIFF" + size + "WAVE"
29
30    while offset + 8 <= file_size {
31        let header = read_at_most(file, offset, 8)?;
32        if header.len() < 8 {
33            break;
34        }
35        let id = [header[0], header[1], header[2], header[3]];
36        let size = u32::from_le_bytes([header[4], header[5], header[6], header[7]]) as u64;
37        let body = offset + 8;
38        if body + size > file_size + 1 {
39            break;
40        }
41
42        match &id {
43            b"fmt " => {
44                let data = read_at_most(file, body, size.min(40) as usize)?;
45                let mut c = Cursor::new(&data);
46                let format = c.u16_le().unwrap_or(1);
47                let channels = c.u16_le().unwrap_or(0);
48                let rate = c.u32_le().unwrap_or(0);
49                byte_rate = c.u32_le().unwrap_or(0);
50                c.skip(2); // block alignment
51                let bits = c.u16_le().unwrap_or(0);
52
53                if channels > 0 {
54                    tags.properties.channels = Some(channels);
55                }
56                if rate > 0 {
57                    tags.properties.sample_rate = Some(rate);
58                }
59                if bits > 0 {
60                    tags.properties.bit_depth = Some(bits);
61                }
62                tags.properties.codec = match format {
63                    1 => "pcm".into(),
64                    3 => "pcm_float".into(),
65                    0xFFFE => "pcm".into(),
66                    other => format!("wav_{other:#x}"),
67                };
68                tags.properties.lossless = matches!(format, 1 | 3 | 0xFFFE);
69            }
70            b"data" if byte_rate > 0 => {
71                tags.properties.duration_ms = Some(size * 1000 / byte_rate as u64);
72            }
73            b"LIST" => {
74                let capped = size.min(MAX_META_CHUNK as u64) as usize;
75                let data = read_at_most(file, body, capped)?;
76                if data.len() >= 4 && &data[..4] == b"INFO" {
77                    read_info_list(&data[4..], &mut tags);
78                }
79            }
80            b"id3 " | b"ID3 " => {
81                let capped = size.min(MAX_META_CHUNK as u64) as usize;
82                let data = read_at_most(file, body, capped)?;
83                super::id3::read_id3v2_buffer(&data, &mut tags);
84            }
85            _ => {}
86        }
87
88        // RIFF chunks are aligned on an even number of bytes.
89        offset = body + size + (size % 2);
90    }
91
92    if let Some(rate) = byte_rate.checked_mul(8)
93        && rate > 0
94    {
95        tags.properties.bitrate_kbps = Some(rate / 1000);
96    }
97    Ok(tags)
98}

100/// `LIST/INFO` chunk: the handful of standard WAV fields.
101fn read_info_list(data: &[u8], tags: &mut RawTags) {
102    let mut c = Cursor::new(data);
103    while c.remaining() >= 8 {
104        let Some(id) = c.take(4) else { break };
105        let id = [id[0], id[1], id[2], id[3]];
106        let Some(size) = c.u32_le() else { break };
107        let size = size as usize;
108        if size > c.remaining() {
109            break;
110        }
111        let Some(value) = c.utf8(size) else { break };
112        if size % 2 == 1 {
113            c.skip(1);
114        }
115        let value = value.trim_end_matches('\0').trim().to_string();
116        let key = match &id {
117            b"INAM" => "title",
118            b"IART" => "artist",
119            b"IPRD" => "album",
120            b"ICRD" => "date",
121            b"IGNR" => "genre",
122            b"ITRK" => "tracknumber",
123            b"ICMT" => "comment",
124            b"IPUB" => "label",
125            b"ICOP" => "copyright",
126            b"IENG" => "engineer",
127            b"ISFT" => "encodedby",
128            _ => continue,
129        };
130        tags.insert(key, value);
131    }
132}

134/// Reads the tags and the audio properties of an open AIFF or AIFC file.
135///
136/// `file_size` bounds the chunk walk, the `FORM` header being no more reliable
137/// here than in WAV. The native `NAME`, `AUTH` and `ANNO` chunks carry very
138/// little, so most real files are described by their `ID3 ` chunk. An AIFC
139/// compression name in `COMM` overrides the `pcm` default and marks the stream
140/// as lossy.
141pub fn read_aiff(file: &mut File, file_size: u64) -> Result<RawTags, TagError> {
142    let mut tags = RawTags::default();
143    tags.properties.container = "aiff".into();
144    tags.properties.codec = "pcm".into();
145    tags.properties.lossless = true;
146
147    let mut offset = 12u64; // "FORM" + size + "AIFF"
148
149    while offset + 8 <= file_size {
150        let header = read_at_most(file, offset, 8)?;
151        if header.len() < 8 {
152            break;
153        }
154        let id = [header[0], header[1], header[2], header[3]];
155        let size = u32::from_be_bytes([header[4], header[5], header[6], header[7]]) as u64;
156        let body = offset + 8;
157        if body + size > file_size + 1 {
158            break;
159        }
160
161        match &id {
162            b"COMM" => {
163                let data = read_at_most(file, body, size.min(64) as usize)?;
164                let mut c = Cursor::new(&data);
165                let channels = c.u16_be().unwrap_or(0);
166                let frames = c.u32_be().unwrap_or(0);
167                let bits = c.u16_be().unwrap_or(0);
168                let rate = c.take(10).and_then(extended80_to_f64).unwrap_or(0.0);
169
170                if channels > 0 {
171                    tags.properties.channels = Some(channels);
172                }
173                if bits > 0 {
174                    tags.properties.bit_depth = Some(bits);
175                }
176                if rate > 0.0 {
177                    tags.properties.sample_rate = Some(rate.round() as u32);
178                    tags.properties.duration_ms = Some((frames as f64 * 1000.0 / rate) as u64);
179                    let bitrate = rate * channels as f64 * bits as f64 / 1000.0;
180                    tags.properties.bitrate_kbps = Some(bitrate as u32);
181                }
182                // AIFC: the real codec follows, on 4 bytes.
183                if let Some(codec) = c.take(4) {
184                    let name = String::from_utf8_lossy(codec).trim().to_lowercase();
185                    if !name.is_empty() && name != "none" && name != "sowt" {
186                        tags.properties.codec = name;
187                        tags.properties.lossless = false;
188                    }
189                }
190            }
191            b"NAME" => insert_iff_text(file, body, size, "title", &mut tags)?,
192            b"AUTH" => insert_iff_text(file, body, size, "artist", &mut tags)?,
193            b"ANNO" => insert_iff_text(file, body, size, "comment", &mut tags)?,
194            b"(c) " => insert_iff_text(file, body, size, "copyright", &mut tags)?,
195            b"ID3 " | b"id3 " => {
196                let capped = size.min(MAX_META_CHUNK as u64) as usize;
197                let data = read_at_most(file, body, capped)?;
198                super::id3::read_id3v2_buffer(&data, &mut tags);
199            }
200            _ => {}
201        }
202
203        offset = body + size + (size % 2);
204    }
205    Ok(tags)
206}
207
208fn insert_iff_text(
209    file: &mut File,
210    offset: u64,
211    size: u64,
212    key: &str,
213    tags: &mut RawTags,
214) -> Result<(), TagError> {
215    let data = read_at_most(file, offset, size.min(4096) as usize)?;
216    tags.insert(key, String::from_utf8_lossy(&data).into_owned());
217    Ok(())
218}
