1//! WAV (RIFF) and AIFF (IFF).
2//!
3//! Two very similar chunk containers, one little-endian, the other
4//! big-endian. Both can carry a full ID3v2 chunk on top of their own fields,
5//! which are historically poor.
6
7use std::fs::File;

9use super::bytes::{Cursor, extended80_to_f64, read_at_most};
10use super::{RawTags, TagError};

12/// Maximum size of a metadata chunk loaded into memory.
13const MAX_META_CHUNK: usize = 16 * 1024 * 1024;
14
15/// Reads the tags and the audio properties of an open WAV file.
16///
17/// `file_size` bounds the chunk walk rather than the `RIFF` header, which is
18/// routinely wrong on files written by a recorder that never came back to fix
19/// it. Tags are gathered from the `LIST`/`INFO` chunk and from an embedded
20/// ID3v2 chunk alike; both contribute to the same fields.
21pub fn read_wav(file: &mut File, file_size: u64) -> Result<RawTags, TagError> {
22    let mut tags = RawTags::default();
23    tags.properties.container = "wav".into();
24    tags.properties.codec = "pcm".into();
25    tags.properties.lossless = true;
26
27    let mut byte_rate = 0u32;
28    let mut offset = 12u64; // "RIFF" + size + "WAVE"
29
30    while offset + 8 <= file_size {
31        let header = read_at_most(file, offset, 8)?;
32        if header.len() < 8 {
33            break;
34        }
35        let id = [header[0], header[1], header[2], header[3]];
36        let size = u32::from_le_bytes([header[4], header[5], header[6], header[7]]) as u64;
37        let body = offset + 8;
38        if body + size > file_size + 1 {
39            break;
40        }
41
42        match &id {
43            b"fmt " => {
44                let data = read_at_most(file, body, size.min(40) as usize)?;
45                let mut c = Cursor::new(&data);
46                let truncated = || TagError::Malformed("truncated WAV format chunk");
47                let mut format = c.u16_le().ok_or_else(truncated)?;
48                let channels = c.u16_le().ok_or_else(truncated)?;
49                let rate = c.u32_le().ok_or_else(truncated)?;
50                byte_rate = c.u32_le().ok_or_else(truncated)?;
51                c.u16_le().ok_or_else(truncated)?; // block alignment
52                let bits = c.u16_le().ok_or_else(truncated)?;
53                if format == 0xfffe {
54                    format = extensible_subtype(&mut c)?;
55                }
56
57                if channels > 0 {
58                    tags.properties.channels = Some(channels);
59                }
60                if rate > 0 {
61                    tags.properties.sample_rate = Some(rate);
62                }
63                if bits > 0 {
64                    tags.properties.bit_depth = Some(bits);
65                }
66                tags.properties.codec = match format {
67                    1 => "pcm".into(),
68                    3 => "pcm_float".into(),
69                    other => format!("wav_{other:#x}"),
70                };
71                tags.properties.lossless = matches!(format, 1 | 3);
72            }
73            b"data" if byte_rate > 0 => {
74                tags.properties.duration_ms = Some(size * 1000 / byte_rate as u64);
75            }
76            b"LIST" => {
77                let capped = size.min(MAX_META_CHUNK as u64) as usize;
78                let data = read_at_most(file, body, capped)?;
79                if data.len() >= 4 && &data[..4] == b"INFO" {
80                    read_info_list(&data[4..], &mut tags);
81                }
82            }
83            b"id3 " | b"ID3 " => {
84                let capped = size.min(MAX_META_CHUNK as u64) as usize;
85                let data = read_at_most(file, body, capped)?;
86                super::id3::read_id3v2_buffer(&data, &mut tags);
87            }
88            _ => {}
89        }
90
91        // RIFF chunks are aligned on an even number of bytes.
92        offset = body + size + (size % 2);
93    }
94
95    if let Some(rate) = byte_rate.checked_mul(8)
96        && rate > 0
97    {
98        tags.properties.bitrate_kbps = Some(rate / 1000);
99    }
100    Ok(tags)
101}

103/// Reads the WAVEFORMATEXTENSIBLE subformat rather than assuming integer PCM.
104///
105/// The GUID can also name compressed or vendor-specific formats. Only the
106/// standard WAVEFORMATEX suffix permits interpreting its first word as a
107/// format tag; an unknown GUID remains unknown and is never called lossless.
108/// See <https://learn.microsoft.com/en-us/windows/win32/api/mmreg/ns-mmreg-waveformatextensible>.
109fn extensible_subtype(cursor: &mut Cursor<'_>) -> Result<u16, TagError> {
110    let truncated = || TagError::Malformed("truncated extensible WAV format");
111    let extension_size = cursor.u16_le().ok_or_else(truncated)?;
112    if extension_size < 22 {
113        return Err(TagError::Malformed("undersized extensible WAV format"));
114    }
115    cursor.u16_le().ok_or_else(truncated)?; // valid bits per sample
116    cursor.u32_le().ok_or_else(truncated)?; // channel mask
117    let subtype = cursor.u32_le().ok_or_else(truncated)?;
118    let suffix = cursor.take(12).ok_or_else(truncated)?;
119    const WAVEFORMATEX_SUFFIX: &[u8] = &[0, 0, 0x10, 0, 0x80, 0, 0, 0xaa, 0, 0x38, 0x9b, 0x71];
120    Ok(if suffix == WAVEFORMATEX_SUFFIX {
121        u16::try_from(subtype).unwrap_or(0xfffe)
122    } else {
123        0xfffe
124    })
125}

127/// `LIST/INFO` chunk: the handful of standard WAV fields.
128fn read_info_list(data: &[u8], tags: &mut RawTags) {
129    let mut c = Cursor::new(data);
130    while c.remaining() >= 8 {
131        let Some(id) = c.take(4) else { break };
132        let id = [id[0], id[1], id[2], id[3]];
133        let Some(size) = c.u32_le() else { break };
134        let size = size as usize;
135        if size > c.remaining() {
136            break;
137        }
138        let Some(value) = c.utf8(size) else { break };
139        if size % 2 == 1 {
140            c.skip(1);
141        }
142        let value = value.trim_end_matches('\0').trim().to_string();
143        let key = match &id {
144            b"INAM" => "title",
145            b"IART" => "artist",
146            b"IPRD" => "album",
147            b"ICRD" => "date",
148            b"IGNR" => "genre",
149            b"ITRK" => "tracknumber",
150            b"ICMT" => "comment",
151            b"IPUB" => "label",
152            b"ICOP" => "copyright",
153            b"IENG" => "engineer",
154            b"ISFT" => "encodedby",
155            _ => continue,
156        };
157        tags.insert(key, value);
158    }
159}

161/// Reads the tags and the audio properties of an open AIFF or AIFC file.
162///
163/// `file_size` bounds the chunk walk, the `FORM` header being no more reliable
164/// here than in WAV. The native `NAME`, `AUTH` and `ANNO` chunks carry very
165/// little, so most real files are described by their `ID3 ` chunk. An AIFC
166/// compression name in `COMM` overrides the `pcm` default. Integer PCM and the
167/// known `fl32`/`fl64` floating PCM encodings remain lossless; other encodings
168/// are not assumed to preserve samples.
169pub fn read_aiff(file: &mut File, file_size: u64) -> Result<RawTags, TagError> {
170    let mut tags = RawTags::default();
171    tags.properties.container = "aiff".into();
172    tags.properties.codec = "pcm".into();
173    tags.properties.lossless = true;
174
175    let mut offset = 12u64; // "FORM" + size + "AIFF"
176
177    while offset + 8 <= file_size {
178        let header = read_at_most(file, offset, 8)?;
179        if header.len() < 8 {
180            break;
181        }
182        let id = [header[0], header[1], header[2], header[3]];
183        let size = u32::from_be_bytes([header[4], header[5], header[6], header[7]]) as u64;
184        let body = offset + 8;
185        if body + size > file_size + 1 {
186            break;
187        }
188
189        match &id {
190            b"COMM" => {
191                let data = read_at_most(file, body, size.min(64) as usize)?;
192                let mut c = Cursor::new(&data);
193                let channels = c.u16_be().unwrap_or(0);
194                let frames = c.u32_be().unwrap_or(0);
195                let bits = c.u16_be().unwrap_or(0);
196                let rate = c.take(10).and_then(extended80_to_f64).unwrap_or(0.0);
197
198                if channels > 0 {
199                    tags.properties.channels = Some(channels);
200                }
201                if bits > 0 {
202                    tags.properties.bit_depth = Some(bits);
203                }
204                if rate > 0.0 {
205                    tags.properties.sample_rate = Some(rate.round() as u32);
206                    tags.properties.duration_ms = Some((frames as f64 * 1000.0 / rate) as u64);
207                    let bitrate = rate * channels as f64 * bits as f64 / 1000.0;
208                    tags.properties.bitrate_kbps = Some(bitrate as u32);
209                }
210                // AIFC: the real codec follows, on 4 bytes.
211                if let Some(codec) = c.take(4) {
212                    let name = String::from_utf8_lossy(codec).trim().to_lowercase();
213                    if !name.is_empty() && name != "none" && name != "sowt" {
214                        tags.properties.lossless = matches!(name.as_str(), "fl32" | "fl64");
215                        tags.properties.codec = name;
216                    }
217                }
218            }
219            b"NAME" => insert_iff_text(file, body, size, "title", &mut tags)?,
220            b"AUTH" => insert_iff_text(file, body, size, "artist", &mut tags)?,
221            b"ANNO" => insert_iff_text(file, body, size, "comment", &mut tags)?,
222            b"(c) " => insert_iff_text(file, body, size, "copyright", &mut tags)?,
223            b"ID3 " | b"id3 " => {
224                let capped = size.min(MAX_META_CHUNK as u64) as usize;
225                let data = read_at_most(file, body, capped)?;
226                super::id3::read_id3v2_buffer(&data, &mut tags);
227            }
228            _ => {}
229        }
230
231        offset = body + size + (size % 2);
232    }
233    Ok(tags)
234}
235
236fn insert_iff_text(
237    file: &mut File,
238    offset: u64,
239    size: u64,
240    key: &str,
241    tags: &mut RawTags,
242) -> Result<(), TagError> {
243    let data = read_at_most(file, offset, size.min(4096) as usize)?;
244    tags.insert(key, String::from_utf8_lossy(&data).into_owned());
245    Ok(())
246}
