1//! Missing studio albums are derived from trusted discographies, never catalog entities.
2use crate::model::Catalog;
3use crate::sources::{self, Facts, KnownRelease};
4use crate::text;
5
6/// One record credited to an artist that this catalog does not hold.
7pub struct Absent<'a> {
8    /// The artist, as the catalog spells them.
9    pub artist: String,
10    /// Why this row is not normally on the report, and `None` when it is.
11    ///
12    /// Two quite different reasons, in one field because they answer one
13    /// question — *why am I not being shown this* — and because a record can
14    /// be both: a live album the reader also set aside. The words for the
15    /// first are MusicBrainz's own; see [`KnownRelease::stated_type`].
16    pub held_back: Option<String>,
17    /// `true` when the reader set this one aside and asked to see it anyway.
18    pub set_aside: bool,
19    /// Which artist, for a caller asking about one rather than about all.
20    pub artist_id: crate::model::Id,
21    /// The record MusicBrainz credits to them.
22    pub known: &'a KnownRelease,
23}
24
25/// What is credited to your artists and not on your shelf.
26///
27/// **Derived, never stored.** The comparison happens here, at the moment of
28/// reading, so the day an album is added to the library it stops being listed —
29/// with nothing to update and nothing to go stale.
30///
31/// Two ways of recognising a record you already have, in this order:
32///
33/// 1. the **release-group identifier**, when your tags carry one. Exact, and
34///    the only comparison that cannot be wrong.
35/// 2. the **normalised title**, otherwise. `text::normalize` is the same
36///    function that decides two spellings are one name everywhere else in this
37///    program, so a shelf that says "Kind Of Blue" is not told it is missing
38///    "Kind of Blue".
39///
40/// Two things are held back, and `everything` lifts both.
41///
42/// Only studio albums are reported — see [`KnownRelease::is_studio_album`] for
43/// why a complete discography would be a true and useless answer — and only
44/// records the reader has not set aside. Those are the report's two filters,
45/// and `--all` is one word for "hold nothing back", so it lifts both and every
46/// row that came back this way carries the reason it would not normally be
47/// here. **A filter is only honest when the reader can turn it off**: saying
48/// what is left out and offering no way to see it is a locked door with a
49/// label on it.
50pub fn absent<'a>(
51    catalog: &Catalog,
52    held: &'a sources::Sources,
53    aside: &[crate::user::SetAside],
54    everything: bool,
55) -> Vec<Absent<'a>> {
56    let mut out: Vec<Absent<'a>> = Vec::new();
57    let ids: std::collections::BTreeSet<&str> = catalog
58        .releases
59        .iter()
60        .filter_map(|release| release.release_group_mbid.as_deref())
61        .filter(|id| !id.trim().is_empty())
62        .collect();
63    for record in &held.records {
64        if record.source != sources::MUSICBRAINZ {
65            continue;
66        }
67        if !held.is_trusted(catalog, record) {
68            continue;
69        }
70        let Facts::Artist(facts) = &record.facts else {
71            continue;
72        };
73        if facts.discography.is_empty() {
74            continue;
75        }
76        let entity = record.entity();
77        // An artist the catalog cannot place has no shelf to be missing from.
78        let Some(artist_id) = entity.resolve(catalog) else {
79            continue;
80        };
81        let Some(artist) = catalog.artist(artist_id) else {
82            continue;
83        };
84
85        // Exact identities apply across the catalog; a title fallback belongs
86        // only to this artist's albums without a known release-group identity.
87        let mut has_shelf = false;
88        let mut titles: Vec<String> = Vec::new();
89        for release in &catalog.releases {
90            if release.album_artist_id != Some(artist_id) {
91                continue;
92            }
93            has_shelf = true;
94            if release
95                .release_group_mbid
96                .as_deref()
97                .is_none_or(|id| id.trim().is_empty())
98            {
99                titles.push(text::normalize(&release.title));
100            }
101        }
102
103        // **No album of their own, no shelf to have gaps in.** The catalog
104        // holds an artist for every credit it reads — a guest on one track, a
105        // composer, one name on a compilation — and being in the catalog is
106        // not the same as having a place in the library. The first version
107        // missed that distinction and answered a question nobody asked: one
108        // Rolling Stones track on a compilation produced their entire studio
109        // discography as "missing", which is true and worthless, and it did
110        // so for every passing credit at once until the report was mostly
111        // that. What this reports is an *incomplete* discography, which means
112        // one that was started.
113        //
114        // The same rule decides who is browsed at all — see the discography fetch selection.
115        if !has_shelf {
116            continue;
117        }
118
119        for known in &facts.discography {
120            // Set aside by hand. Filtered here rather than by dropping it from
121            // the layer, because the two are different claims and this program
122            // keeps them apart everywhere else: MusicBrainz still says this is
123            // an album, and the user still says it is not one they want listed.
124            // Deleting the record would lose the first to record the second.
125            let put_aside = aside.iter().any(|a| a.release_group == known.mbid);
126
127            // Both reasons, in the order a reader meets them: what it is, then
128            // what they decided about it. A record can be both, and naming only
129            // one would answer half the question.
130            let mut why: Vec<String> = Vec::new();
131            if !known.is_studio_album() {
132                why.push(known.stated_type());
133            }
134            if put_aside {
135                why.push("set aside".to_string());
136            }
137            let held_back = (!why.is_empty()).then(|| why.join(", "));
138            if held_back.is_some() && !everything {
139                continue;
140            }
141
142            let have = ids.contains(known.mbid.as_str())
143                || titles.contains(&text::normalize(&known.title));
144            if !have {
145                out.push(Absent {
146                    artist: artist.name.clone(),
147                    held_back,
148                    set_aside: put_aside,
149                    artist_id,
150                    known,
151                });
152            }
153        }
154    }
155    // Oldest first within an artist, so a discography reads as one.
156    out.sort_by(|a, b| {
157        a.artist
158            .cmp(&b.artist)
159            .then_with(|| a.known.first_released.cmp(&b.known.first_released))
160    });
161    out
162}

164/// How many records one artist has that this shelf does not.
165///
166/// For a page to say so at the moment somebody is looking at that artist —
167/// which is the moment they wonder. `aede missing` was in the help, in the
168/// README and in the manual, and still could not be found, because it was
169/// named nowhere near the question it answers. The same fault `aede extract`
170/// had, and the same fix.
171///
172/// Derived like everything else here: nothing is stored, so the number falls
173/// the day an album is added.
174pub fn absent_for(
175    catalog: &Catalog,
176    held: &sources::Sources,
177    aside: &[crate::user::SetAside],
178    artist_id: crate::model::Id,
179) -> usize {
180    absent(catalog, held, aside, false)
181        .iter()
182        .filter(|record| record.artist_id == artist_id)
183        .count()
184}
