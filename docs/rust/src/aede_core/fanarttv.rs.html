1//! Fanart.tv music artwork: artist portraits, logos, banners and backgrounds,
2//! album covers and disc art, plus record-label logos.
3//!
4//! Deliberately **no network**, like [`crate::acoustid`] and
5//! [`crate::musicbrainz`]: this module builds the address to ask and reads
6//! the answer somebody else fetched.
7//!
8//! # Second for a portrait, only, for a logo
9//!
10//! [`crate::wikipedia::portrait_file`] is asked before this for a portrait —
11//! a Commons image is reached through the identifier MusicBrainz already
12//! gave, and Commons hosts nothing whose licence is not stated. Fanart.tv is
13//! user-submitted: `fetch --portraits` uses it only where Wikidata has
14//! nothing, and the record it leaves says which service actually answered, so
15//! a reader who cares about that distinction can tell — see [`SOURCE`].
16//!
17//! A logo has no such alternative to try first: Wikidata states no logo claim
18//! for an artist, so `fetch --logos` asks Fanart.tv directly, under its own
19//! record name — see [`LOGO_SOURCE`]. A banner rides the very same request:
20//! `fetch --logos --banners` reads [`banner_url`] out of the answer already
21//! fetched for the logo, rather than asking twice for one artist.
22//!
23//! # Indexed by the identifier that never needed a search
24//!
25//! Unlike AcoustID, which is shown a fingerprint and has to guess which
26//! recording it is, Fanart.tv is asked about an artist **by MusicBrainz
27//! identifier** — the same one every artist record in `sources.json` already
28//! carries. There is nothing to score and nothing to arbitrate: the answer is
29//! either this artist's images or none.
30

40/// The name a logo is stored under — not [`SOURCE`], though the same request
41/// answers both.
42///
43/// A portrait and a logo are two different claims about the same artist,
44/// fetched by `fetch --portraits` and `fetch --logos` on their own schedules,
45/// and a record that carried both could not be re-fetched for one without
46/// silently dropping the other — `Sources::set` replaces a record whole, it
47/// does not merge fields into it. Keeping them as two records under two names
48/// is what lets each pass redo its own work without undoing the other's — the
49/// same reasoning, and the same fix, as [`crate::wikipedia::PORTRAIT_SOURCE`].
50pub const LOGO_SOURCE: &str = "fanarttv-logo";

75/// The environment variable holding the application key.
76///
77/// Free to register, and not shipped with Aède for the reason
78/// [`crate::acoustid::KEY_VARIABLE`] gives: a key baked into an open-source
79/// binary is a key every copy shares, and the quota with it.
80pub const KEY_VARIABLE: &str = "AEDE_FANARTTV_KEY";
81
82/// The key this machine holds, when it holds one.
83///
84/// Read once, at the edge, and carried in — the same discipline
85/// [`crate::acoustid::key`] follows, for the same reason: a value read from
86/// the process environment inside the pass that needs it cannot be handed a
87/// different answer by a test.
88pub fn key() -> Option<String> {
89    std::env::var(KEY_VARIABLE)
90        .ok()
91        .map(|value| value.trim().to_string())
92        .filter(|value| !value.is_empty())
93}

102/// Where to ask about one artist.

106        encode_component(mbid),

120/// Refuses a response whose identity does not match the artist requested.
121///
122/// Fanart.tv v3.2 guarantees `mbid_id` on artist answers. Without this guard,
123/// an error object or a changed response shape reads indistinguishably from
124/// "this artist has no logo".
125pub fn artist_response(response: &Json, mbid: &str) -> Result<(), String> {
126    match response.field_str("mbid_id") {
127        Some(found) if found == mbid => Ok(()),
128        Some(found) => Err(format!(
129            "Fanart.tv answered for MusicBrainz id {found}, not {mbid}"
130        )),
131        None => Err(
132            "Fanart.tv returned no artist identifier; the response was not an artist answer"
133                .to_string(),
134        ),
135    }
136}
137
138/// Refuses a response whose identity does not match the label requested.
139pub fn label_response(response: &Json, mbid: &str) -> Result<(), String> {
140    match response.field_str("id") {
141        Some(found) if found == mbid => Ok(()),
142        Some(found) => Err(format!(
143            "Fanart.tv answered for MusicBrainz label id {found}, not {mbid}"
144        )),
145        None => Err(
146            "Fanart.tv returned no label identifier; the response was not a label answer"
147                .to_string(),
148        ),
149    }
150}

152/// One image the service knows of, and how well liked it is.
153#[derive(Debug, Clone, PartialEq, Eq)]
154struct Thumb {
155    url: String,
156    likes: u32,
157}
158
159/// The most liked image under `field`, or `None` when it is empty or absent.
160///
161/// Ranked by `likes`, which is the only signal the service gives about which
162/// of several submissions is the one people chose — the same role a score
163/// plays for a search result, and unlike a search result these are not scored
164/// against the question at all, only against each other.
165fn best(response: &Json, field: &str) -> Option<Thumb> {
166    response
167        .get(field)
168        .and_then(Json::as_arr)
169        .unwrap_or(&[])
170        .iter()
171        .filter_map(|row| {
172            let url = row.field_str("url").filter(|u| !u.is_empty())?;
173            let likes = row
174                .field_str("likes")
175                .and_then(|l| l.parse::<u32>().ok())
176                .unwrap_or(0);
177            Some(Thumb { url, likes })
178        })
179        .min_by(|a, b| b.likes.cmp(&a.likes).then_with(|| a.url.cmp(&b.url)))
180}
181
182/// The most liked portrait the service holds for this artist, or `None`.
183///
184/// **`artistthumb` only.** Fanart.tv also serves banners, logos and backdrops
185/// for the same artist, and those answer a different question — the wallpaper
186/// behind a "now playing" screen, not "who is this". Bringing them in under
187/// one function would ask a caller wanting a portrait to filter what should
188/// never have been mixed together.
189fn best_thumb(response: &Json) -> Option<Thumb> {
190    best(response, "artistthumb")
191}
192
193/// The image address to download, or `None` when the service has nothing.
194pub fn portrait_url(response: &Json) -> Option<String> {
195    best_thumb(response).map(|thumb| thumb.url)
196}
197
198/// The artist's logo address to download, or `None` when the service has
199/// none.
200///
201/// **`hdmusiclogo`, then `musiclogo`.** The two are the same submissions at
202/// two resolutions, not two different pools to rank against each other — an
203/// HD entry always wins over a standard one, however many likes the standard
204/// one has, and `musiclogo` is only consulted when `hdmusiclogo` is empty.
205pub fn logo_url(response: &Json) -> Option<String> {
206    best(response, "hdmusiclogo")
207        .or_else(|| best(response, "musiclogo"))
208        .map(|thumb| thumb.url)
209}
210
211/// The artist's banner address to download, or `None` when the service has
212/// none.
213///
214/// **`musicbanner`.** Unlike the logo, the service serves this at one
215/// resolution only — there is no `hdmusicbanner` to prefer first.
216pub fn banner_url(response: &Json) -> Option<String> {
217    best(response, "musicbanner").map(|thumb| thumb.url)
218}
219
220/// The best artist background, preferring the native 4K pool.
221///
222/// Fanart.tv exposes 3840×2160 images separately from its 1920×1080
223/// backgrounds.  They are alternative resolutions for the same use, so any
224/// 4K image wins; likes only choose within one resolution.
225pub fn background_url(response: &Json) -> Option<String> {
226    best(response, "artist4kbackground")
227        .or_else(|| best(response, "artistbackground"))
228        .map(|thumb| thumb.url)
229}

231/// Fanart.tv artwork belonging to one MusicBrainz release group.
232#[derive(Debug, Clone, PartialEq, Eq)]
233pub struct AlbumArtwork {
234    /// The most liked album cover, when present.
235    pub cover: Option<String>,
236    /// The best disc image for each disc number, in disc order.
237    pub discs: Vec<String>,
238}
239
240/// Reads the album artwork embedded in an artist response from API v3.2.
241///
242/// Covers are interchangeable alternatives, hence one most-liked image.
243/// Disc art is not: disc one and disc two depict different objects, so the
244/// most-liked image for every distinct `disc` value is retained.
245pub fn album_artwork(response: &Json, release_group: &str) -> Option<AlbumArtwork> {
246    let album = response
247        .get("albums")?
248        .as_arr()?
249        .iter()
250        .find(|album| album.field_str("release_group_id").as_deref() == Some(release_group))?;
251    let cover = best(album, "albumcover").map(|thumb| thumb.url);
252
253    let mut discs = std::collections::BTreeMap::<String, Thumb>::new();
254    for row in album.get("cdart").and_then(Json::as_arr).unwrap_or(&[]) {
255        let Some(url) = row.field_str("url").filter(|url| !url.is_empty()) else {
256            continue;
257        };
258        let disc = row.field_str("disc").unwrap_or_else(|| "1".to_string());
259        let likes = row
260            .field_str("likes")
261            .and_then(|likes| likes.parse::<u32>().ok())
262            .unwrap_or(0);
263        let candidate = Thumb { url, likes };
264        match discs.get_mut(&disc) {
265            Some(current)
266                if candidate.likes > current.likes
267                    || (candidate.likes == current.likes && candidate.url < current.url) =>
268            {
269                *current = candidate;
270            }
271            Some(_) => {}
272            None => {
273                discs.insert(disc, candidate);
274            }
275        }
276    }
277    let mut discs: Vec<_> = discs.into_iter().collect();
278    discs.sort_by(|(a, _), (b, _)| {
279        a.parse::<u32>()
280            .unwrap_or(u32::MAX)
281            .cmp(&b.parse::<u32>().unwrap_or(u32::MAX))
282            .then_with(|| a.cmp(b))
283    });
284    Some(AlbumArtwork {
285        cover,
286        discs: discs.into_iter().map(|(_, image)| image.url).collect(),
287    })
288}

295/// What to tell somebody who has no application key.
296///
297/// Named separately from [`crate::acoustid::no_key`] rather than shared: the
298/// two services, the two registration pages and the two variables are not the
299/// same fact stated twice, and a reader following one must not be sent to
300/// register at the other's address by an error message that forgot which it
301/// was talking about.
302///
303/// States only the key itself, never who is affected: `--portraits` can fall
304/// back to Wikidata and `--logos` cannot, so which artists a missing key
305/// leaves untouched is a different sentence for each — one this function does
306/// not know and must not guess at — and each pass says it in the line before
307/// this one is printed.
308pub fn no_key() -> String {
309    format!(
310        "\
311Fanart.tv images need an application key, and none is set.
312
313  1. Register at https://fanart.tv/get-an-api-key/
314  2. export {KEY_VARIABLE}=<the key it gives you>"
315    )
316}
