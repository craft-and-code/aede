1//! IANA civil boundaries; durations remain UTC intervals across clock changes.
2use super::Policy;
3use jiff::{Timestamp, civil::Date, tz::TimeZone};
4
5pub(super) struct Clock {
6    pub(super) day: usize,
7    pub(super) day_start: u64,
8    pub(super) day_end: u64,
9    pub(super) week_start: u64,
10    pub(super) week_end: u64,
11    pub(super) scheduled: bool,
12    pub(super) next: u64,
13}
14fn stamp(value: Timestamp) -> Result<u64, String> {
15    u64::try_from(value.as_second())
16        .map_err(|_| "parental calendar is outside supported dates".into())
17}
18fn midnight(date: Date, zone: &TimeZone) -> Result<u64, String> {
19    stamp(
20        date.to_zoned(zone.clone())
21            .map_err(|_| "invalid parental civil date")?
22            .timestamp(),
23    )
24}
25fn scheduled(policy: &Policy, now: u64, zone: &TimeZone) -> Result<bool, String> {
26    let time = Timestamp::from_second(i64::try_from(now).map_err(|_| "invalid parental clock")?)
27        .map_err(|_| "invalid parental clock")?
28        .to_zoned(zone.clone());
29    let minute = u16::try_from(time.hour()).map_err(|_| "invalid parental hour")? * 60
30        + u16::try_from(time.minute()).map_err(|_| "invalid parental minute")?;
31    Ok(policy.days[time.weekday().to_monday_zero_offset() as usize]
32        .windows
33        .iter()
34        .any(|window| window.start_minute <= minute && minute < window.end_minute))
35}
36pub(super) fn clock(policy: &Policy, now: u64) -> Result<Clock, String> {
37    let zone = TimeZone::get(&policy.timezone).map_err(|_| "invalid parental time zone")?;
38    let timestamp =
39        Timestamp::from_second(i64::try_from(now).map_err(|_| "invalid parental clock")?)
40            .map_err(|_| "invalid parental clock")?;
41    let local = timestamp.to_zoned(zone.clone());
42    let date = local.date();
43    let day = local.weekday().to_monday_zero_offset() as usize;
44    let day_start = midnight(date, &zone)?;
45    let day_end = midnight(
46        date.tomorrow().map_err(|_| "invalid parental civil date")?,
47        &zone,
48    )?;
49    let mut monday = date;
50    for _ in 0..day {
51        monday = monday.yesterday().map_err(|_| "invalid parental week")?;
52    }
53    let week_start = midnight(monday, &zone)?;
54    for _ in 0..7 {
55        monday = monday.tomorrow().map_err(|_| "invalid parental week")?;
56    }
57    let week_end = midnight(monday, &zone)?;
58    let mut next = day_end;
59    // Both sides of a fold matter. A missing wall-clock boundary is only
60    // admitted if actual clock membership changes there; zone transitions
61    // independently cover jumps into/out of a window in the middle of a gap.
62    for window in &policy.days[day].windows {
63        for minute in [window.start_minute, window.end_minute] {
64            let (date, minute) = if minute == 1440 {
65                (date.tomorrow().map_err(|_| "invalid parental day")?, 0)
66            } else {
67                (date, minute)
68            };
69            let value = date.at((minute / 60) as i8, (minute % 60) as i8, 0, 0);
70            for candidate in [
71                zone.to_ambiguous_timestamp(value).earlier(),
72                zone.to_ambiguous_timestamp(value).later(),
73            ]
74            .into_iter()
75            .flatten()
76            {
77                let candidate = stamp(candidate)?;
78                if now < candidate
79                    && candidate < next
80                    && scheduled(policy, candidate - 1, &zone)?
81                        != scheduled(policy, candidate, &zone)?
82                {
83                    next = candidate;
84                }
85            }
86        }
87    }
88    if let Some(transition) = zone.following(timestamp).next() {
89        let candidate = stamp(transition.timestamp())?;
90        if now < candidate
91            && candidate < next
92            && scheduled(policy, candidate - 1, &zone)? != scheduled(policy, candidate, &zone)?
93        {
94            next = candidate;
95        }
96    }
97    Ok(Clock {
98        day,
99        day_start,
100        day_end,
101        week_start,
102        week_end,
103        scheduled: scheduled(policy, now, &zone)?,
104        next,
105    })
106}
