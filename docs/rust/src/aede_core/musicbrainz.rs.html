1//! Reading what MusicBrainz answers, and deciding whether it is about us.
2//!
3//! Deliberately **no network**. This module turns a response that somebody
4//! else fetched into [`crate::sources`] records, and decides how firmly one of
5//! them is attached to an entity of the catalog. Those two jobs are where the
6//! mistakes live — a misread field, a wrong album picked confidently — and
7//! neither of them has any reason to need a socket to be tested.
8//!
9//! The field spellings come from the MusicBrainz documentation and are quoted
10//! exactly: `sort-name`, `life-span.begin`, `primary-type`, `secondary-types`,
11//! `first-release-date`, `label-info[].label.name`. Hyphens, not underscores —
12//! guessing one of those wrong produces a record that is silently empty.
13

22/// Base address of the web service, kept here so the client has nothing to
23/// decide.
24pub const WEB_SERVICE: &str = "https://musicbrainz.org/ws/2";

26/// The gap the service requires between two requests, with a small margin.
27///
28/// One request per second per address is the documented limit, and going over
29/// it does not slow a run down — it makes the service answer `503` to
30/// *everything* from that address until the rate drops, for every program
31/// sharing it. The margin is there because the clock that decides is the
32/// server's, not ours.
33///
34/// It lives here rather than in the client because it is a property of
35/// MusicBrainz, not of how we talk to it: a build without the network feature
36/// still needs it to say how long a run would take.
37pub const REQUEST_INTERVAL: std::time::Duration = std::time::Duration::from_millis(1100);

39/// What to ask for alongside an artist, in one request.
40///
41/// The whole point of a lookup over a search: these ride on the request
42/// already being made, so genres, other names and the links an artist has —
43/// Wikidata among them — cost nothing extra at a service that allows one
44/// request per second.
45///
46/// `artist-rels` carries the dated memberships — who played in a band and
47/// between which years — read by `memberships` below. It rides on the same request
48/// as the rest, so a line-up costs nothing beyond the lookup already being
49/// made, and the dates are what let an album page name the band as it stood
50/// the year that record came out.
51pub const ARTIST_INCLUDES: &str = "genres+tags+aliases+url-rels+artist-rels";
52
53/// URL relationships include the Wikidata link used by label summaries.
54pub const LABEL_INCLUDES: &str = "url-rels";
55
56/// What a recording lookup needs for performance, production and composition
57/// credits in a single request.
58///
59/// `work-level-rels` is a switch: `work-rels` first attaches each work and
60/// `artist-rels` then asks for artist relationships both on the recording and
61/// on those linked works. `work-rels` also exposes explicit parent-work
62/// relationships on linked works, including classical movements.
63pub const RECORDING_INCLUDES: &str = "artist-rels+work-rels+work-level-rels";
64
65/// What to ask for alongside a *release*, in one request.
66///
67/// A release is one edition; a release group is the album every edition of it
68/// belongs to. The two facts a library wants sit on opposite sides of that
69/// line — the **label** is the edition's, the **type and first release date**
70/// are the album's — and asking for them separately would be two requests per
71/// album at one request per second.
72///
73/// `release-groups` folds the album into the edition's answer, so a library
74/// tagged by Picard, which writes the edition identifier, gets the whole
75/// record for the price of one lookup.
76pub const RELEASE_INCLUDES: &str = "labels+release-groups+artist-rels";

78/// One answer among the several a search returns.
79#[derive(Debug, Clone, PartialEq, Eq)]
80pub struct Candidate<F> {
81    /// The MusicBrainz identifier, which is what makes a later fetch an update
82    /// rather than a second opinion.
83    pub mbid: String,
84    /// Name or title as MusicBrainz spells it, for the reader to judge.
85    pub name: String,
86    /// Whatever the search decided about relevance, 0 to 100.
87    ///
88    /// **Not a probability, and not a confidence.** It says how well the query
89    /// matched the index, so a library holding one album by an obscure band
90    /// gets a 100 for the only thing that answered. It is one input to
91    /// [`best_match`], never the verdict.
92    pub score: u8,
93    /// What was said, ready to be stored.
94    pub facts: F,
95}
96
97/// Why nothing was attached, which is worth saying rather than returning
98/// `None` and letting the caller invent a reason.
99#[derive(Debug, Clone, PartialEq, Eq)]
100pub enum NoMatch {
101    /// The search came back with nothing at all.
102    Nothing,
103    /// Several answers are equally good, and choosing between them would be
104    /// arbitrary. The names are carried so the report can show them.
105    Ambiguous(Vec<String>),
106    /// The best answer is not close enough to the name asked about.
107    TooWeak {
108        /// The best candidate's name, so the reader can see what was rejected.
109        best: String,
110        /// Its search score.
111        score: u8,
112    },
113}
114
115/// Below this, an answer is not worth attaching to anything.
116///
117/// A floor rather than a preference: MusicBrainz answers *something* for
118/// almost any query, and an unfiltered "best" result is how a library ends up
119/// with a Beatles record filed under a bar band of the same initials.
120const FLOOR: u8 = 70;
121
122/// Picks the answer that describes the thing asked about, or says why none does.
123///
124/// Two rules, both taken from behaviour this codebase already has:
125///
126/// - **An exact name match wins, and only widens when nothing matches
127///   exactly** — the rule `Catalog::find_releases` follows. The comparison is
128///   [`text::normalize`], the same function that decides two spellings are one
129///   name everywhere else.
130/// - **Several equally good answers are refused, not arbitrated.** Returning
131///   the first of them is an arbitrary answer given without saying so, which
132///   is the fault `find_releases` and `moved_to` were both fixed for.
133///
134/// The confidence returned is [`Confidence::Identified`] only when the caller
135/// asked by identifier; a search can never produce more than
136/// [`Confidence::Matched`], however sure it looks.
137pub fn best_match<F: Clone>(
138    candidates: &[Candidate<F>],
139    wanted: &str,
140) -> Result<(Candidate<F>, Confidence), NoMatch> {
141    if candidates.is_empty() {
142        return Err(NoMatch::Nothing);
143    }
144    let wanted = text::normalize(wanted);
145
146    let exact: Vec<&Candidate<F>> = candidates
147        .iter()
148        .filter(|c| text::normalize(&c.name) == wanted)
149        .collect();
150
151    let pool: Vec<&Candidate<F>> = match exact.is_empty() {
152        false => exact,
153        true => candidates.iter().collect(),
154    };
155
156    let best = pool
157        .iter()
158        .max_by_key(|c| c.score)
159        .expect("a non-empty pool");
160    if best.score < FLOOR {
161        return Err(NoMatch::TooWeak {
162            best: best.name.clone(),
163            score: best.score,
164        });
165    }
166
167    // Two answers of the same quality are two answers. Names that normalise to
168    // the same thing are not a tie — a reissue and its original often share a
169    // title, and both being "right" is not the same as the program having to
170    // choose.
171    let tied: Vec<&&Candidate<F>> = pool
172        .iter()
173        .filter(|c| {
174            c.score == best.score && text::normalize(&c.name) != text::normalize(&best.name)
175        })
176        .collect();
177    if !tied.is_empty() {
178        let mut names: Vec<String> = std::iter::once(best.name.clone())
179            .chain(tied.iter().map(|c| c.name.clone()))
180            .collect();
181        names.sort();
182        return Err(NoMatch::Ambiguous(names));
183    }
184
185    // The score MusicBrainz gives is about its index, not about us. A name
186    // that matches exactly deserves more than one that merely ranked well, and
187    // neither is ever a certainty.
188    let confidence = match text::normalize(&best.name) == wanted {
189        true => Confidence::matched(best.score.min(95)),
190        false => Confidence::matched(best.score.saturating_sub(25)),
191    };
192    Ok(((*best).clone(), confidence))
193}

195/// Escapes the characters Lucene reads as syntax.
196///
197/// The search server parses the query, so a name carrying `/`, `:`, `(` or `-`
198/// is not a name to it but an expression — and an expression that does not
199/// parse is not a polite "nothing found": the request fails. MusicBrainz
200/// documents this and gives the same example, `ac/dc` sent as `ac\/dc`, and
201/// the escaping happens **before** URL-encoding, never instead of it.
202///
203/// This is what a whole-library run trips over on its first awkward name,
204/// while a run naming one well-behaved artist works perfectly — which is
205/// exactly how it was found.
206pub fn escape_query(name: &str) -> String {
207    const SPECIAL: [char; 22] = [
208        '\\', '+', '-', '&', '|', '!', '(', ')', '{', '}', '[', ']', '^', '"', '~', '*', '?', ':',
209        '/', '<', '>', '=',
210    ];
211    let mut out = String::with_capacity(name.len() + 8);
212    for c in name.chars() {
213        if SPECIAL.contains(&c) {
214            out.push('\\');
215        }
216        out.push(c);
217    }
218    out
219}

225/// A string field, absent rather than empty.
226fn field(value: &Json, key: &str) -> Option<String> {
227    value
228        .field_str(key)
229        .map(|s| s.trim().to_string())
230        .filter(|s| !s.is_empty())
231}

233/// The search score, which MusicBrainz gives as a number 0–100.
234fn score_of(value: &Json) -> u8 {
235    value
236        .field_u32("score")
237        .unwrap_or(0)
238        .min(100)
239        .try_into()
240        .unwrap_or(100)
241}

243/// Artists, as `/ws/2/artist/?query=…&fmt=json` returns them.
244///
245/// An entry with no identifier is skipped: without it a later fetch could not
246/// tell an update from a second opinion, which is the one thing the identifier
247/// is stored for.
248/// The entity, read the same way whether it came from a search or a lookup.
249///
250/// One extractor rather than two: the two answers carry the same artist, and
251/// reading it in two places is how one of them quietly stops keeping a field.
252fn artist_facts(row: &Json) -> ArtistFacts {
253    let life = row.get("life-span");
254    ArtistFacts {
255        // `country` is a code, `area.name` is the name a reader wants. Both
256        // are kept: the name is what a listing shows, the code is what
257        // somebody types. Preferring one and discarding the other threw away
258        // a fact the source had stated — and it was the short one, which is
259        // the half a reader reaches for.
260        area: row
261            .get("area")
262            .and_then(|a| field(a, "name"))
263            .or_else(|| field(row, "country")),
264        country_code: field(row, "country"),
265        began: life.and_then(|l| field(l, "begin")),
266        // An artist that has not ended has no end date, and a `life-span` may
267        // carry `ended: false` with no `end`.
268        ended: life.and_then(|l| field(l, "end")),
269        // `ended: false` is an answer — "still going" — and it is the one a
270        // reader looking at a band wants most.
271        active: life
272            .and_then(|l| l.field_optional_bool("ended"))
273            .map(|e| !e),
274        kind: field(row, "type"),
275        disambiguation: field(row, "disambiguation"),
276        // `genres` and `tags` have the same shape; genres are the curated
277        // half, so they come first and tags fill in only when there are none.
278        genres: voted(row, "genres")
279            .or_else(|| voted(row, "tags"))
280            .unwrap_or_default(),
281        aliases: row
282            .get("aliases")
283            .and_then(Json::as_arr)
284            .map(|a| a.iter().filter_map(|x| field(x, "name")).collect())
285            .unwrap_or_default(),
286        wikidata: linked(row, "wikidata"),
287        discogs: linked(row, "discogs"),
288        // MusicBrainz spells this relationship "official homepage".
289        homepage: linked(row, "official homepage"),
290        // Filled by its own pass, never by this one: a discography is a
291        // browse over release groups, not a field of an artist lookup, and
292        // reading it here would quietly empty it on every ordinary fetch.
293        discography: Vec::new(),
294        members: memberships(row),
295        // MusicBrainz holds no prose about an artist: an annotation there is
296        // an editorial note about the data, not a description of the
297        // musician. The summary comes from Wikipedia, reached through the
298        // `wikidata` link above, and carries its own licence with it.
299        summary: None,
300        // Likewise filled by its own pass: MusicBrainz holds no picture of an
301        // artist either, and `fetch --portraits` reaches one the same way
302        // Wikipedia is reached, through the `wikidata` link above.
303        portrait: None,
304        // Same story again: `fetch --logos` is the only pass that fills this.
305        logo: None,
306    }
307}

309/// Band memberships, from the `artist-rels` a lookup was asked for.
310///
311/// # The direction, measured rather than assumed
312///
313/// MusicBrainz states **one** relation between two artists and returns it on
314/// both, distinguished by `direction`. Reading that field wrongly does not lose
315/// data, it inverts it — Black Sabbath would appear in the list of Ozzy
316/// Osbourne's members — so it was checked against live answers for a person and
317/// for a band rather than reasoned about.
318///
319/// Both answers carry `"direction": "backward"`, and that settles it: these
320/// relationships are defined **from the musician towards the group**, so a
321/// backward one is being read *from the group's end* and the artist it names is
322/// the player. The first version of this function had it the other way round,
323/// on the reasonable-sounding assumption that a person's own record would read
324/// forward. It does not.
325///
326/// # Which relationships count
327///
328/// `member of band` is the obvious one and it is not enough. Ozzy Osbourne's
329/// own record holds no members at all — a solo artist is not a band — and every
330/// musician who played on his records is there as an
331/// `instrumental supporting musician`: Randy Rhoads on guitar, Bob Daisley on
332/// bass. A line-up that showed nothing for him while MusicBrainz plainly holds
333/// his band would be a worse answer than no feature.
334///
335/// The two are kept apart rather than merged, because a founding member and a
336/// guitarist hired for one tour are both on the record and only one of them was
337/// in the band. [`Membership::kind`] carries the source's own phrase.
338///
339/// **The list is what has been seen in a real answer**, and nothing else: a
340/// relationship type nobody has checked is a guess with a `const` around it.
341fn memberships(row: &Json) -> Vec<Membership> {
342    let Some(relations) = row.get("relations").and_then(Json::as_arr) else {
343        return Vec::new();
344    };
345    relations
346        .iter()
347        .filter(|r| {
348            r.field_str("type")
349                .is_some_and(|t| MEMBER_RELATIONS.contains(&t.as_str()))
350        })
351        .filter_map(|r| {
352            let artist = r.get("artist")?;
353            Some(Membership {
354                relation_id: field(r, "id"),
355                role_id: field(r, "type-id"),
356                direction: field(r, "direction"),
357                // No identifier, no membership: a name alone cannot be
358                // followed to the artist it names, and the whole value of this
359                // list is that each row leads somewhere.
360                mbid: field(artist, "id")?,
361                name: field(artist, "name").unwrap_or_default(),
362                credited_as: field(r, "target-credit").filter(|value| !value.is_empty()),
363                kind: field(r, "type").unwrap_or_default(),
364                side: match r.field_str("direction").as_deref() {
365                    Some("backward") => Side::Player,
366                    _ => Side::Group,
367                },
368                attributes: r
369                    .get("attributes")
370                    .and_then(Json::as_arr)
371                    .map(|a| a.iter().filter_map(Json::as_string).collect())
372                    .unwrap_or_default(),
373                // `begin` may be a year, `1979-11`, or a full date, and every
374                // one of those is answered by taking the first four characters
375                // where a year is wanted.
376                began: field(r, "begin"),
377                ended: field(r, "end"),
378                // `ended: false` is an answer — "still in the band" — and it is
379                // the one a reader looking at a line-up wants most. It arrives
380                // beside `"end": null`, so the two must be read separately.
381                over: r.field_optional_bool("ended"),
382            })
383        })
384        .collect()
385}

387/// The relationships this reads, spelt as MusicBrainz spells them.
388///
389/// One place, because the strings are the whole filter: a typo here empties
390/// every line-up in the program and nothing else goes wrong. Each was seen in
391/// a live answer — `member of band` on Judas Priest, `instrumental supporting
392/// musician` on Ozzy Osbourne — and a type nobody has checked does not belong
393/// on this list, however plausible its name.
394const MEMBER_RELATIONS: [&str; 2] = ["member of band", "instrumental supporting musician"];
395
396/// Names from a `genres` or `tags` list, most agreed first.
397///
398/// `None` when the list is absent — which is not the same as an artist nobody
399/// tagged, and only the first of those means "we did not ask for it".
400fn voted(row: &Json, key: &str) -> Option<Vec<String>> {
401    let rows = row.get(key)?.as_arr()?;
402    let mut scored: Vec<(u32, String)> = rows
403        .iter()
404        .filter_map(|t| Some((t.field_u32("count").unwrap_or(0), field(t, "name")?)))
405        .collect();
406    // Most agreed first, then alphabetically so two runs give the same order.
407    scored.sort_by(|a, b| b.0.cmp(&a.0).then_with(|| a.1.cmp(&b.1)));
408    Some(scored.into_iter().map(|(_, name)| name).collect())
409}

411/// The URL behind a relationship of the given type.
412///
413/// `relations[].url.resource` is the shape a URL relationship takes, and the
414/// one spelling here I could not confirm against a live answer — so it is
415/// written to come back empty rather than to break, and the tests below pin
416/// what it expects.
417fn linked(row: &Json, relation: &str) -> Option<String> {
418    row.get("relations")?.as_arr()?.iter().find_map(|r| {
419        (r.field_str("type").as_deref() == Some(relation))
420            .then(|| r.get("url").and_then(|u| field(u, "resource")))
421            .flatten()
422    })
423}

425/// One artist, as `/ws/2/artist/{mbid}?fmt=json` returns it.
426///
427/// A **lookup**, not a search, and the difference is the whole reason this
428/// exists beside [`artists`]. A search answers with an abbreviated entry and a
429/// relevance score, because it has to guess which artist was meant. A lookup
430/// is asked about one identifier and answers with the entity itself —
431/// `life-span.ended` and `disambiguation` among the fields a search result
432/// leaves out.
433///
434/// Which matters because a library tagged with Picard already carries those
435/// identifiers. Searching by name for an artist whose MBID sits in the tags is
436/// asking a question that has already been answered, and accepting a guess in
437/// place of a certainty.
438pub fn artist(response: &Json) -> Option<Candidate<ArtistFacts>> {
439    let mbid = field(response, "id")?;
440    Some(Candidate {
441        mbid,
442        name: field(response, "name").unwrap_or_default(),
443        // Nothing was ranked: the service was asked about this one thing.
444        score: 100,
445        facts: artist_facts(response),
446    })
447}

449/// One recording, as a lookup with [`RECORDING_INCLUDES`] returns it.
450///
451/// A lookup is evidence about the identifier we asked for, never a title
452/// match. Its recording credits, work relationships and work credits remain
453/// source facts: reconciliation decides later whether and how they join the
454/// local canonical graph.
455pub fn recording(response: &Json) -> Option<Candidate<TrackFacts>> {
456    let mbid = field(response, "id")?;
457    let credits = relationship_credits(response);
458    let works = response
459        .get("relations")
460        .and_then(Json::as_arr)
461        .map(|relations| {
462            relations
463                .iter()
464                .filter(|relation| relation.field_str("target-type").as_deref() == Some("work"))
465                .filter_map(|relation| {
466                    let work = relation.get("work")?;
467                    Some(WorkLink {
468                        mbid: field(work, "id")?,
469                        title: field(work, "title").unwrap_or_default(),
470                        relation_id: field(relation, "id"),
471                        relation_type: field(relation, "type"),
472                        relation_type_id: field(relation, "type-id"),
473                        direction: field(relation, "direction"),
474                        attributes: relationship_attributes(relation),
475                        credits: relationship_credits(work),
476                        parents: work_parent_links(work),
477                    })
478                })
479                .collect()
480        })
481        .unwrap_or_default();
482    Some(Candidate {
483        mbid: mbid.clone(),
484        name: field(response, "title").unwrap_or_default(),
485        score: 100,
486        facts: TrackFacts {
487            recording: Some(mbid),
488            title: field(response, "title"),
489            works,
490            credits,
491            relationships_complete: true,
492            ..Default::default()
493        },
494    })
495}

497/// A child work may have several containing works. Only the backward
498/// MusicBrainz "parts" relationship says the queried work is a part; a
499/// forward relationship names one of its children, and a derivative is not a
500/// container at all.
501fn work_parent_links(work: &Json) -> Vec<crate::sources::WorkParentLink> {
502    const PARTS_TYPE_ID: &str = "ca8d3642-ce5f-49f8-91f2-125d72524e6a";
503    work.get("relations")
504        .and_then(Json::as_arr)
505        .into_iter()
506        .flatten()
507        .filter(|relation| relation.field_str("target-type").as_deref() == Some("work"))
508        .filter(|relation| relation.field_str("direction").as_deref() == Some("backward"))
509        .filter(|relation| match relation.field_str("type-id") {
510            Some(id) => id == PARTS_TYPE_ID,
511            None => relation.field_str("type").as_deref() == Some("parts"),
512        })
513        .filter_map(|relation| {
514            let parent = relation.get("work")?;
515            Some(crate::sources::WorkParentLink {
516                mbid: field(parent, "id")?,
517                title: field(parent, "title").unwrap_or_default(),
518                relation_id: field(relation, "id"),
519                relation_type_id: field(relation, "type-id"),
520                direction: field(relation, "direction"),
521                attributes: relationship_attributes(relation),
522                order: relation.field_u32("ordering-key"),
523            })
524        })
525        .collect()
526}

528/// Artist relationships carried by one recording or work response.
529fn relationship_credits(entity: &Json) -> Vec<CreditLink> {
530    entity
531        .get("relations")
532        .and_then(Json::as_arr)
533        .map(|relations| {
534            relations
535                .iter()
536                .filter(|relation| relation.field_str("target-type").as_deref() == Some("artist"))
537                .filter_map(|relation| {
538                    let artist = relation.get("artist")?;
539                    let credited_as = field(relation, "target-credit").filter(|s| !s.is_empty());
540                    Some(CreditLink {
541                        relation_id: field(relation, "id"),
542                        role_id: field(relation, "type-id"),
543                        role: field(relation, "type")?,
544                        direction: field(relation, "direction"),
545                        artist_mbid: field(artist, "id")?,
546                        artist_name: field(artist, "name").unwrap_or_default(),
547                        credited_as,
548                        attributes: relationship_attributes(relation),
549                        began: field(relation, "begin"),
550                        ended: field(relation, "end"),
551                        over: relation.field_optional_bool("ended"),
552                        order: relation.field_u32("ordering-key"),
553                    })
554                })
555                .collect()
556        })
557        .unwrap_or_default()
558}

560/// Attributes of one relationship, retaining values and credited spellings.
561fn relationship_attributes(relation: &Json) -> Vec<CreditAttribute> {
562    relation
563        .get("attributes")
564        .and_then(Json::as_arr)
565        .map(|attributes| {
566            attributes
567                .iter()
568                .filter_map(Json::as_string)
569                .map(|name| CreditAttribute {
570                    id: relation
571                        .get("attribute-ids")
572                        .and_then(|ids| ids.get(&name))
573                        .and_then(Json::as_string),
574                    value: relation
575                        .get("attribute-values")
576                        .and_then(|values| values.get(&name))
577                        .and_then(Json::as_string),
578                    credited_as: relation
579                        .get("attribute-credits")
580                        .and_then(|credits| credits.get(&name))
581                        .and_then(Json::as_string)
582                        .filter(|value| !value.is_empty()),
583                    name,
584                })
585                .collect()
586        })
587        .unwrap_or_default()
588}

590/// Artists, as `/ws/2/artist/?query=…&fmt=json` returns them.
591///
592/// A **search**: the answers are ranked guesses at what was meant, and each
593/// entry is abbreviated. Prefer [`artist`] whenever an identifier is known.
594///
595/// An entry with no identifier is skipped: without it a later fetch could not
596/// tell an update from a second opinion, which is the one thing the identifier
597/// is stored for.
598pub fn artists(response: &Json) -> Vec<Candidate<ArtistFacts>> {
599    let rows = response
600        .get("artists")
601        .and_then(Json::as_arr)
602        .unwrap_or(&[]);
603    rows.iter()
604        .filter_map(|row| {
605            let mbid = field(row, "id")?;
606            Some(Candidate {
607                mbid,
608                name: field(row, "name").unwrap_or_default(),
609                score: score_of(row),
610                facts: artist_facts(row),
611            })
612        })
613        .collect()
614}

616/// Release groups, as `/ws/2/release-group/?query=…&fmt=json` returns them.
617///
618/// Groups rather than releases, deliberately: a release group is *the album*,
619/// and every remaster and reissue of it is a release inside that group. Asking
620/// about releases would answer with one edition among twenty and call it the
621/// album's date.
622pub fn release_groups(response: &Json) -> Vec<Candidate<ReleaseFacts>> {
623    let rows = response
624        .get("release-groups")
625        .and_then(Json::as_arr)
626        .unwrap_or(&[]);
627    rows.iter()
628        .filter_map(|row| {
629            let mbid = field(row, "id")?;
630            Some(Candidate {
631                mbid,
632                name: field(row, "title").unwrap_or_default(),
633                score: score_of(row),
634                facts: release_facts(row),
635            })
636        })
637        .collect()
638}

640/// The album, read the same way wherever the group came from.
641///
642/// One extractor for the search result, the group lookup and the group folded
643/// into a release lookup — for the reason [`artist_facts`] is one function:
644/// reading the same entity in three places is how two of them quietly stop
645/// keeping a field.
646///
647/// The label is deliberately **not** read here. A group has none; a release
648/// does. Filling it from whichever edition answered first would attribute one
649/// pressing's label to the album itself.
650fn release_facts(group: &Json) -> ReleaseFacts {
651    ReleaseFacts {
652        primary_type: field(group, "primary-type"),
653        secondary_types: group
654            .get("secondary-types")
655            .and_then(Json::as_arr)
656            .map(|a| a.iter().filter_map(Json::as_string).collect())
657            .unwrap_or_default(),
658        first_released: field(group, "first-release-date"),
659        label: None,
660        label_mbid: None,
661        // MusicBrainz holds no artwork: the images live at the Cover Art
662        // Archive, which is a different service with its own answer — see
663        // [`crate::coverart`].
664        cover_art: None,
665        edition_mbid: None,
666        credits: Vec::new(),
667        relationships_complete: false,
668    }
669}

671/// One release group, as a lookup answers: `/ws/2/release-group/<mbid>`.
672///
673/// The counterpart of [`artist`]: an answer about the identifier that was
674/// asked for is a certainty, where a search result is a guess with a score.
675/// Used when the tags carry `MUSICBRAINZ_RELEASEGROUPID` but no edition.
676pub fn release_group(response: &Json) -> Option<Candidate<ReleaseFacts>> {
677    let mbid = field(response, "id")?;
678    Some(Candidate {
679        mbid,
680        name: field(response, "title").unwrap_or_default(),
681        score: 100,
682        facts: release_facts(response),
683    })
684}
685
686/// One release, as a lookup with [`RELEASE_INCLUDES`] answers.
687///
688/// The richest answer this program can get about an album, and the cheapest:
689/// the edition supplies the label, the folded release group supplies the type
690/// and the date the album first appeared — which is the fact a reissue's `DATE`
691/// tag most often contradicts.
692///
693/// The identifier kept is the **release group's**, not the edition's. Two
694/// pressings of one album are one album, and keying the record on the edition
695/// would file a second copy of the same answer the day the user replaces their
696/// CD rip with a vinyl one. The edition is what was asked; the album is what
697/// was learnt.
698pub fn release(response: &Json) -> Option<Candidate<ReleaseFacts>> {
699    let group = response.get("release-group");
700    let mut facts = group.map(release_facts).unwrap_or_default();
701    facts.edition_mbid = field(response, "id");
702    facts.credits = relationship_credits(response);
703    facts.relationships_complete = true;
704    if let Some(label) = label_of_release(response) {
705        facts.label = Some(label.name);
706        facts.label_mbid = label.mbid;
707    }
708    // Falling back to the edition's own identifier: an answer with no group is
709    // not one this program has seen, but storing it under the edition is
710    // better than dropping a lookup that succeeded.
711    let mbid = group
712        .and_then(|g| field(g, "id"))
713        .or_else(|| field(response, "id"))?;
714    Some(Candidate {
715        mbid,
716        name: group
717            .and_then(|g| field(g, "title"))
718            .or_else(|| field(response, "title"))
719            .unwrap_or_default(),
720        score: 100,
721        facts,
722    })
723}

725/// How many release groups one browse request may return.
726///
727/// The service's own ceiling. A prolific artist has more than this, so the
728/// caller pages — which is the one place in this program where a single entity
729/// can cost more than one request, and the reason [`discography`] hands back
730/// the total alongside the page.
731pub const BROWSE_LIMIT: usize = 100;
732
733/// Where to ask for everything credited to an artist.
734///
735/// A **browse**, not a search: browse answers "what does this identifier have",
736/// which is a question with one right answer, where a search answers "what
737/// resembles these words". Nothing here is scored or matched, because nothing
738/// here is guessed.
739///
740/// `type=album` narrows what travels. It is a saving and not the filter that
741/// matters — [`crate::sources::KnownRelease::is_studio_album`] decides what a
742/// reader is shown, so a parameter the service ignores costs bandwidth and
743/// never correctness.
744pub fn discography_url(artist_mbid: &str, offset: usize) -> String {
745    format!(
746        "{WEB_SERVICE}/release-group?artist={artist_mbid}&type=album\
747         &fmt=json&limit={BROWSE_LIMIT}&offset={offset}"
748    )
749}

751/// One page of an artist's discography, and how many there are in all.
752///
753/// The total comes back as `release-group-count` and is what tells the caller
754/// whether to ask for another page. Absent, it is taken to be what arrived:
755/// stopping is the safe reading, since paging forever on a field that was never
756/// there would be a request per second with no end.
757pub fn discography(response: &Json) -> (Vec<crate::sources::KnownRelease>, usize) {
758    use crate::sources::KnownRelease;
759    let rows = response
760        .get("release-groups")
761        .and_then(Json::as_arr)
762        .unwrap_or(&[]);
763    let page: Vec<KnownRelease> = rows
764        .iter()
765        .filter_map(|row| {
766            // No identifier, no row: the only way to tell "you own this one"
767            // from "you are missing it" without an identifier is the title, and
768            // two records share a title often enough that a wish list built on
769            // titles alone is wrong.
770            Some(KnownRelease {
771                mbid: field(row, "id")?,
772                title: field(row, "title").unwrap_or_default(),
773                first_released: field(row, "first-release-date"),
774                primary_type: field(row, "primary-type"),
775                secondary_types: row
776                    .get("secondary-types")
777                    .and_then(Json::as_arr)
778                    .map(|a| a.iter().filter_map(Json::as_string).collect())
779                    .unwrap_or_default(),
780            })
781        })
782        .collect();
783    let total = response
784        .field_u32("release-group-count")
785        .map(|n| n as usize)
786        .unwrap_or(page.len());
787    (page, total)
788}

790/// A label credited on a release: its name and, when the same entry carried
791/// one, its MusicBrainz identifier.
792pub struct LabelOfRelease {
793    /// The name, as the release lookup spells it.
794    pub name: String,
795    /// The label's own MusicBrainz identifier, when the same entry named
796    /// one.
797    pub mbid: Option<String>,
798}

800/// The label of a release, as a release lookup returns it.
801///
802/// Separate from [`release_groups`] because it comes from a different request,
803/// and because it describes an edition rather than the album.
804///
805/// The name and the identifier are read from **the same** `label-info` entry
806/// — the first one that names a label — never a name from one row and an
807/// identifier from another: a release crediting several labels would
808/// otherwise risk pairing the name of one with the address of a different
809/// one. No tag carries a label's MusicBrainz identifier the way
810/// `MUSICBRAINZ_ARTISTID` does an artist's, so this is the only place one is
811/// ever read from.
812pub fn label_of_release(response: &Json) -> Option<LabelOfRelease> {
813    response
814        .get("label-info")
815        .and_then(Json::as_arr)?
816        .iter()
817        .find_map(|info| {
818            let label = info.get("label")?;
819            let name = field(label, "name")?;
820            Some(LabelOfRelease {
821                name,
822                mbid: field(label, "id"),
823            })
824        })
825}

827/// One label, as `/ws/2/label/{mbid}?fmt=json` returns it.
828///
829/// A **lookup**, asked only once an identifier is already known — see
830/// `crate::sources::ReleaseFacts::label_mbid` and the `fetch --labels`
831/// module doc for where that comes from: a release this catalog looked up
832/// sometimes already names its label's own identifier, and a lookup then
833/// turns that into a certainty instead of guessing by name all over again.
834/// No tag ever carries it directly, so unlike [`artist`] this is never asked
835/// with a value read out of the files.
836pub fn label(response: &Json) -> Option<Candidate<LabelFacts>> {
837    let mbid = field(response, "id")?;
838    Some(Candidate {
839        mbid,
840        name: field(response, "name").unwrap_or_default(),
841        // Nothing was ranked: the service was asked about this one thing.
842        score: 100,
843        facts: LabelFacts {
844            wikidata: linked(response, "wikidata"),
845            discogs: linked(response, "discogs"),
846            ..Default::default()
847        },
848    })
849}

866        })

868}
