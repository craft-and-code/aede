1//! Asking LRCLIB for the words, and reading what it answers.
2//!
3//! No socket here, exactly as in [`crate::musicbrainz`]: this builds addresses
4//! and reads answers, and everything that waits its turn lives in
5//! [`crate::http`]. That is what makes the whole of it testable against canned
6//! text.
7//!
8//! # Why this service and not another
9//!
10//! Lyrics are the **composition's** copyright, and owning a FLAC grants no
11//! rights in it — they are legally a different object from the recording. Of
12//! the free sources, LRCLIB is the only one an open-source program can query
13//! without breaking an API's own terms: Genius forbids the scraping its lyrics
14//! require, and Musixmatch's free tier is non-commercial and truncated. The
15//! reasoning in full is in `docs/design/lyrics.md`.
16//!
17//! It is also the gentlest network path this program has. No key, no account, a
18//! JSON document of a few fields, and a failure that costs nothing: no lyrics
19//! is not a broken catalog, where a bad MusicBrainz match attaches the wrong
20//! record to an album.
21//!
22//! # Identify yourself
23//!
24//! LRCLIB drops connections from agents it has come to distrust — Jellyfin's
25//! server agent is refused outright, with no status code at all, the connection
26//! simply closing. This program sends the same honest `User-Agent` it sends
27//! MusicBrainz, and shares its throttle: being slower than a free service
28//! requires is never the wrong mistake.
29
30use crate::json::Json;
31
32/// Where the service lives.
33pub const WEB_SERVICE: &str = "https://lrclib.net/api";
34
35/// What the service had for one track.
36#[derive(Debug, Clone, PartialEq, Eq)]
37pub enum Found {
38    /// Words, timed or not.
39    Words(Words),
40    /// The service knows this recording and says it has no words at all.
41    ///
42    /// **Not the same as not knowing**, and the distinction is the reason this
43    /// is a variant rather than an empty [`Words`]: an instrumental is a
44    /// finished question, and telling a reader "no lyrics found" about one
45    /// would send them looking for a fault that is not there.
46    Instrumental,
47    /// The service has nothing for this track.
48    Nothing,
49}
50
51/// The words themselves.
52#[derive(Debug, Clone, PartialEq, Eq)]
53pub struct Words {
54    /// The text, in the form it will be written to disk.
55    pub text: String,
56    /// Whether the text carries timings.
57    ///
58    /// The service holds both forms and they are two different things to a
59    /// reader: one is a page to read, the other is a file a player can follow.
60    /// The synced form is preferred where there is one, and this says which was
61    /// taken so the run can report it rather than leave it to be discovered.
62    pub synced: bool,
63}
64
65/// The address that asks about one track.
66///
67/// **Four criteria, and only two of them are the service's requirement.** A
68/// live answer to `artist_name` and `track_name` alone came back `200`, so the
69/// album and the length are *this program's* insistence rather than LRCLIB's:
70/// a song has a studio take, a live one and three covers, they share a title,
71/// and the length is what tells them apart. Both are things this catalog holds
72/// exactly — the duration is measured on the stream rather than read from a
73/// tag — so asking the narrow question costs nothing and answers about this
74/// recording.
75///
76/// An empty album is **left out of the address** rather than sent empty. The
77/// two are different questions: an omitted parameter asks with one criterion
78/// fewer, where `album_name=` states that the album is called nothing. Only the
79/// first has been seen to work, and inventing the behaviour of the second is
80/// exactly what a checked answer is for.
81pub fn get_url(artist: &str, title: &str, album: &str, duration_secs: u64) -> String {
82    let mut url = format!(
83        "{WEB_SERVICE}/get?artist_name={}&track_name={}",
84        encode(artist),
85        encode(title)
86    );
87    if !album.trim().is_empty() {
88        url.push_str(&format!("&album_name={}", encode(album)));
89    }
90    url.push_str(&format!("&duration={duration_secs}"));
91    url
92}

94/// Percent-encodes a query value.
95///
96/// Local to this module rather than shared with the MusicBrainz client: that
97/// one escapes a *search query* first, in a syntax with its own reserved
98/// characters, and folding the two would mean one caller quietly getting the
99/// other's rules.
100fn encode(value: &str) -> String {
101    let mut out = String::with_capacity(value.len());
102    for byte in value.as_bytes() {
103        match byte {
104            b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'_' | b'.' | b'~' => {
105                out.push(*byte as char)
106            }
107            other => out.push_str(&format!("%{other:02X}")),
108        }
109    }
110    out
111}

113/// Reads what the service answered.
114///
115/// The synced form wins where there is one, because it is the richer of the
116/// two: [`crate::lyrics::parse`] reads it into lines that carry their timings,
117/// and a reader who only wants the words gets them from the same lines with the
118/// timings dropped. Taking the plain form when a timed one exists would throw
119/// away something that cannot be recovered.
120///
121/// An answer whose text is empty is [`Found::Nothing`], whatever else it says:
122/// a record with no words in it is not an answer about the words.
123pub fn read(value: &Json) -> Found {
124    // Asked before the text, because it is an answer *about* the absence of
125    // text and would otherwise be read as a service that failed to have any.
126    if value.field_optional_bool("instrumental") == Some(true) {
127        return Found::Instrumental;
128    }
129    let synced = value
130        .field_str("syncedLyrics")
131        .filter(|t| !t.trim().is_empty());
132    let plain = value
133        .field_str("plainLyrics")
134        .filter(|t| !t.trim().is_empty());
135    match (synced, plain) {
136        (Some(text), _) => Found::Words(Words { text, synced: true }),
137        (None, Some(text)) => Found::Words(Words {
138            text,
139            synced: false,
140        }),
141        (None, None) => Found::Nothing,
142    }
143}
