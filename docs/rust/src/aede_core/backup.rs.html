1//! One file holding everything Aède knows, so that a disk failure costs a scan
2//! and not a year of listening.
3//!
4//! Four independently versioned stores are nested whole: the mostly derived
5//! catalog (which also holds watched roots), costly byte-level conclusions,
6//! irreplaceable user statements, and slowly re-fetchable external sources.
7//! A version-1 envelope is still accepted: its embedded legacy conclusions
8//! are extracted into the fourth store before restoration.
9//!
10//! # A document of documents, and why that matters
11//!
12//! The four are nested **as they are**, each keeping its own `format_version`
13//! and read back by its own `from_json`. Nothing here re-encodes a catalog, so
14//! a field added to the catalog tomorrow is in the backup tomorrow, with no
15//! second writer to forget it — the fault that would otherwise be discovered
16//! the day somebody restored one.
17//!
18//! The consequence worth stating: **each store is refused on its own.** A
19//! backup written by an Aède whose catalog format has since moved still
20//! restores the notes and the fetched facts, because the catalog's version
21//! check has nothing to say about `user.json`. A single check at the top would
22//! have thrown away irreplaceable data to protect the rebuildable catalog.
23//!
24//! # It stores, it never fetches or scans
25//!
26//! Reading a backup produces documents; writing one consumes them. Nothing
27//! here touches the library, and — as everywhere in this program — nothing
28//! here touches an audio file.
29

39/// The version of the wrapper, which is not the version of what it wraps.
40///
41/// It changes when the *shape of the envelope* changes — a fourth store, a
42/// renamed field — and not when a store inside it moves, because each of those
43/// carries its own version and answers for itself.
44pub const BACKUP_FORMAT_VERSION: u32 = 2;
45
46/// One store inside a backup, and what this build can do with it.
47///
48/// Three states rather than an `Option`, because "the backup holds no
49/// `sources.json`" and "the backup holds one this build refuses" call for
50/// opposite things to be said and opposite things to be done. Collapsed into
51/// one, the second would restore as though the file had never existed — which
52/// is the quietest possible way to lose a year of fetching.
53#[derive(Debug, Clone)]
54pub enum Part<T> {
55    /// Held, and readable by this build.
56    Held(T),
57    /// There was no such store when the backup was made.
58    Empty,
59    /// Present, and refused by that store's own rule. Carries what it said.
60    Unreadable(String),
61}
62
63impl<T> Part<T> {
64    /// The store, when there is one this build can use.
65    pub fn held(&self) -> Option<&T> {
66        match self {
67            Part::Held(value) => Some(value),
68            _ => None,
69        }
70    }
71
72    /// Reads one nested document, keeping the three states apart.
73    ///
74    /// A missing key and a `null` are both "there was none": a store that did
75    /// not exist is written as `null` rather than left out, so that a reader
76    /// can tell a complete backup from a truncated one, and both spellings are
77    /// accepted because only one of them is this program's own.
78    fn read(
79        value: &Json,
80        key: &str,
81        parse: impl Fn(&Json) -> Result<T, StoreError>,
82    ) -> Result<Part<T>, StoreError> {
83        match value.get(key) {
84            None | Some(Json::Null) => Ok(Part::Empty),
85            Some(document) => match parse(document) {
86                Ok(store) => Ok(Part::Held(store)),
87                Err(why) => Ok(Part::Unreadable(why.to_string())),
88            },
89        }
90    }
91}

93/// Everything Aède knows, at one moment.
94///
95/// Not comparable, and not by omission: a `Catalog` is not `Eq` — two catalogs
96/// of the same library are equal in no useful sense — so a test that wanted to
97/// prove a round trip compares what the documents say rather than the structs.
98#[derive(Debug, Clone)]
99pub struct Backup {
100    /// When it was written, in seconds since the epoch.
101    pub made_at: u64,
102    /// The version of Aède that wrote it.
103    ///
104    /// For a reader, not for the program: nothing branches on it, because the
105    /// four stores each state their own format and a version string is a poor
106    /// substitute for that. What it answers is "which build made this", which
107    /// is the first question asked of a file that will not restore.
108    pub made_by: String,
109    /// The scanned graph and watched roots.
110    pub catalog: Part<Catalog>,
111    /// Byte-level verdicts and imported measurements, independently restorable.
112    pub conclusions: Part<Conclusions>,
113    /// What the user said. The part nothing can rebuild.
114    pub user: Part<UserData>,
115    /// What other sources said. Re-fetchable, slowly.
116    pub sources: Part<Sources>,
117}
118
119impl Backup {
120    /// `true` when there is nothing in here worth writing.
121    ///
122    /// A backup of an empty data folder would restore an empty data folder, and
123    /// reporting success for it teaches the reader that the command worked —
124    /// which is exactly the belief that costs them the library later.
125    pub fn is_empty(&self) -> bool {
126        self.catalog.held().is_none()
127            && self.conclusions.held().is_none()
128            && self.user.held().is_none()
129            && self.sources.held().is_none()
130    }
131}

133/// The envelope, with the four documents nested unchanged.
134pub fn to_json(backup: &Backup) -> Json {
135    let mut root = Json::obj();
136    root.set("format_version", BACKUP_FORMAT_VERSION.into());
137    root.set("made_at", backup.made_at.into());
138    root.set("made_by", backup.made_by.clone().into());
139    // Written even when there is nothing to write, as `null`. A key that is
140    // simply absent cannot be told from a file that was cut short.
141    root.set(
142        "catalog",
143        match backup.catalog.held() {
144            Some(catalog) => crate::store::to_json(catalog),
145            None => Json::Null,
146        },
147    );
148    root.set(
149        "conclusions",
150        match backup.conclusions.held() {
151            Some(conclusions) => crate::conclusions::to_json(conclusions),
152            None => Json::Null,
153        },
154    );
155    root.set(
156        "user",
157        match backup.user.held() {
158            Some(user) => crate::user::to_json(user),
159            None => Json::Null,
160        },
161    );
162    root.set(
163        "sources",
164        match backup.sources.held() {
165            Some(sources) => crate::sources::to_json(sources),
166            None => Json::Null,
167        },
168    );
169    root
170}

172/// Reads back what [`to_json`] wrote.
173///
174/// Versions 1 and 2 are understood; other envelope shapes are refused rather
175/// than guessed at. A *store* inside a readable
176/// envelope is never fatal: it comes back as [`Part::Unreadable`] with what its
177/// own reader said, and the other stores remain restorable.
178pub fn from_json(value: &Json) -> Result<Backup, StoreError> {
179    let found = value.field_u32("format_version").unwrap_or(0);
180    if found != BACKUP_FORMAT_VERSION && found != 1 {
181        return Err(StoreError::Version {
182            found,
183            expected: BACKUP_FORMAT_VERSION,
184        });
185    }
186    let catalog = Part::read(value, "catalog", crate::store::from_json)?;
187    let conclusions = if found == 1 {
188        match catalog.held() {
189            Some(catalog) => {
190                let gathered = Conclusions::from_catalog(catalog);
191                if gathered.files.is_empty() && gathered.analyses.is_empty() {
192                    Part::Empty
193                } else {
194                    Part::Held(gathered)
195                }
196            }
197            None => Part::Empty,
198        }
199    } else {
200        Part::read(value, "conclusions", crate::conclusions::from_json)?
201    };
202    Ok(Backup {
203        made_at: value.field_u64("made_at").unwrap_or(0),
204        made_by: value.field_str("made_by").unwrap_or_default(),
205        catalog,
206        conclusions,
207        user: Part::read(value, "user", crate::user::from_json)?,
208        sources: Part::read(value, "sources", crate::sources::from_json)?,
209    })
210}

212/// Writes a backup where it was asked to go.
213pub fn write(backup: &Backup, path: &Path) -> Result<(), StoreError> {
214    // Pretty rather than compact, and deliberately: this file exists to be
215    // opened by a worried person, and half of what it is for is that they can
216    // see their notes are in there.
217    std::fs::write(path, to_json(backup).to_string_pretty())?;
218    Ok(())
219}
220
221/// Reads a backup from disk.
222pub fn read(path: &Path) -> Result<Backup, StoreError> {
223    let text = std::fs::read_to_string(path)?;
224    let value = crate::json::parse(&text).map_err(StoreError::Parse)?;
225    from_json(&value)
226}
