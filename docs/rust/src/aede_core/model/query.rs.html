1//! Reading the graph.
2//!
3//! Every question a command asks of the catalog lands here, and nothing in this
4//! file changes anything: the whole module takes `&self`. That is the point of
5//! keeping it apart from `build` — a query that had to mutate would be a design
6//! mistake, and here it would not even compile.
7//!
8//! Two rules run through all of it. A lookup that can match several things
9//! returns all of them together with how it matched ([`TitleMatch`]), because a
10//! command must never pick one answer out of several in silence. And a figure
11//! about an artist says which class of role it counts: being audible on a
12//! record and having written it are not the same fact.
13

68    /// Recordings whose title or MusicBrainz identifier matches the query.
69    pub fn find_recordings(&self, query: &str) -> Vec<&Recording> {
70        // A local recording can share its title with many other takes. Its
71        // first placement's path is the stable fallback identity printed by
72        // navigation when no MusicBrainz recording ID exists.
73        if let Some(path) = query.strip_prefix("local:") {
74            return self
75                .recordings
76                .iter()
77                .filter(|recording| {
78                    recording.track_ids.iter().any(|&track_id| {
79                        self.track(track_id)
80                            .and_then(|track| self.file(track.file_id))
81                            .is_some_and(|file| file.path == path)
82                    })
83                })
84                .collect();
85        }
86        let key = text::normalize(query);
87        self.recordings
88            .iter()
89            .filter(|recording| {
90                recording.key == key
91                    || recording.key.contains(&key)
92                    || recording.mbid.as_deref() == Some(query)
93            })
94            .collect()
95    }

251    /// Releases the artist wrote or produced **and is not audible on**.
252    ///
253    /// Not a measure of anything — a display set. It exists so the artist page
254    /// can show the records that would otherwise never appear on it, without
255    /// repeating the discography above. Never label it "writing": that is
256    /// [`Catalog::releases_with_writing_credit`].
257    pub fn releases_written_without_performing(&self, artist_id: Id) -> Vec<Id> {
258        let heard: BTreeSet<Id> = self
259            .releases_as_album_artist(artist_id)
260            .into_iter()
261            .chain(self.guest_appearances(artist_id))
262            .collect();
263        self.releases_for(artist_id, false)
264            .into_iter()
265            .filter(|id| !heard.contains(id))
266            .collect()
267    }

269    /// Releases reached through the artist's credits, keeping either the
270    /// performing roles or the writing ones.
271    fn releases_for(&self, artist_id: Id, performing: bool) -> Vec<Id> {
272        let mut out = BTreeSet::new();
273        for credit in self.credits.iter().filter(|c| c.artist_id == artist_id) {
274            if is_performing_role(&credit.role) != performing {
275                continue;
276            }
277            let release_id = match credit.entity_kind {
278                EntityKind::Release => Some(credit.entity_id),
279                EntityKind::Track => self.track(credit.entity_id).and_then(|t| t.release_id),
280                _ => None,
281            };
282            if let Some(id) = release_id {
283                out.insert(id);
284            }
285        }
286        out.into_iter().collect()
287    }

304    ///
305    /// The display counterpart of [`Catalog::performed_tracks_of_artist`]:
306    /// together they cover every track credit without counting one twice, a
307    /// track where someone both plays and composes going to the first. For

322                set.insert(credit.entity_id);

324        }

328    /// Every release the artist appears on, whatever their role.
329    pub fn releases_of_artist(&self, artist_id: Id) -> Vec<Id> {
330        let mut set = BTreeSet::new();
331        for credit in self.credits.iter().filter(|c| c.artist_id == artist_id) {
332            match credit.entity_kind {
333                EntityKind::Release => {
334                    set.insert(credit.entity_id);
335                }
336                EntityKind::Track => {
337                    if let Some(release_id) =
338                        self.track(credit.entity_id).and_then(|t| t.release_id)
339                    {
340                        set.insert(release_id);
341                    }
342                }
343                _ => {}
344            }
345        }
346        set.into_iter().collect()
347    }

372    /// An artist's neighbours in the graph, from strongest link to weakest.
373    pub fn neighbours_of_artist(&self, artist_id: Id) -> Vec<(&Artist, u32, &str)> {
374        let mut out: Vec<(&Artist, u32, &str)> = self
375            .relations
376            .iter()
377            .filter(|r| {
378                r.source_kind == EntityKind::Artist
379                    && r.source_id == artist_id
380                    && r.target_kind == EntityKind::Artist
381                    && r.kind == "collaborated"
382            })
383            .filter_map(|r| {
384                self.artist(r.target_id)
385                    .map(|a| (a, r.weight, r.kind.as_str()))
386            })
387            .collect();
388        out.sort_by(|a, b| b.1.cmp(&a.1).then_with(|| a.0.name.cmp(&b.0.name)));
389        out
390    }

392    /// Tracks on which two artists are both credited in a performing role.
393    ///
394    /// This is what the weight of a `collaborated` relation counts, recomputed
395    /// on demand rather than stored: the `credit` table already holds the
396    /// answer, and a second copy is a second thing that can fall out of step.
397    pub fn tracks_in_common(&self, a: Id, b: Id) -> Vec<Id> {
398        let performing_on = |artist: Id| -> BTreeSet<Id> {
399            self.credits
400                .iter()
401                .filter(|c| {
402                    c.entity_kind == EntityKind::Track
403                        && c.artist_id == artist
404                        && is_performing_role(&c.role)
405                })
406                .map(|c| c.entity_id)
407                .collect()
408        };
409        let (left, right) = (performing_on(a), performing_on(b));
410        left.intersection(&right).copied().collect()
411    }

439    /// Case- and accent-insensitive search, across every entity.
440    pub fn search(&self, query: &str, limit: usize) -> Vec<SearchHit> {
441        let needle = text::normalize(query);
442        if needle.is_empty() {
443            return Vec::new();
444        }
445        let mut hits: Vec<SearchHit> = Vec::new();
446
447        let mut push = |kind: EntityKind, id: Id, name: &str, key: &str, detail: String| {
448            let score = match () {
449                _ if key == needle => 0u8,
450                _ if key.starts_with(&needle) => 1,
451                _ if key.contains(&needle) => 2,
452                _ => return,
453            };
454            hits.push(SearchHit {
455                kind,
456                id,
457                name: name.to_string(),
458                detail,
459                score,
460            });
461        };
462
463        for a in &self.artists {
464            push(EntityKind::Artist, a.id, &a.name, &a.key, String::new());
465        }
466        for r in &self.releases {
467            let detail = r
468                .album_artist_id
469                .and_then(|id| self.artist(id))
470                .map(|a| a.name.clone())
471                .unwrap_or_else(|| "Various Artists".to_string());
472            push(EntityKind::Release, r.id, &r.title, &r.key, detail);
473        }
474        for t in &self.tracks {
475            let key = text::normalize(&t.title);
476            let detail = t
477                .release_id
478                .and_then(|id| self.release(id))
479                .map(|r| r.title.clone())
480                .unwrap_or_default();
481            push(EntityKind::Track, t.id, &t.title, &key, detail);
482        }
483        for recording in &self.recordings {
484            push(
485                EntityKind::Recording,
486                recording.id,
487                &recording.title,
488                &recording.key,
489                ui_count(recording.track_ids.len(), "local placement"),
490            );
491        }
492        for work in &self.works {
493            push(
494                EntityKind::Work,
495                work.id,
496                &work.title,
497                &work.key,
498                ui_count(work.recording_ids.len(), "recording"),
499            );
500        }
501        for group in &self.release_groups {
502            push(
503                EntityKind::ReleaseGroup,
504                group.id,
505                &group.title,
506                &group.key,
507                ui_count(group.release_ids.len(), "local edition"),
508            );
509        }
510        for l in &self.labels {
511            push(EntityKind::Label, l.id, &l.name, &l.key, String::new());
512        }
513
514        hits.sort_by(|a, b| {
515            a.score
516                .cmp(&b.score)
517                .then_with(|| a.kind.cmp(&b.kind))
518                .then_with(|| a.name.len().cmp(&b.name.len()))
519                .then_with(|| a.name.cmp(&b.name))
520        });
521        hits.truncate(limit);
522        hits
523    }

531    /// Every artist whose name matches, exactly or failing that partially.
532    ///
533    /// **The rule [`Catalog::find_releases`] and [`Catalog::find_tracks`]
534    /// already followed, and that this one did not.** `find_artist` matched
535    /// exactly and nothing else, so every caller that wanted `osbourne` to
536    /// reach Ozzy bolted a fuzzy search underneath and took its **first hit**
537    /// — an arbitrary answer among several, given without a word. On a shelf
538    /// holding both `Ozzy Osbourne` and `O. Osbourne` that is not a nicety:
539    /// `aede artist osbourne` and `aede artist ozzy` answered about two
540    /// different people, one of them with a biography and thirteen albums and
541    /// the other with seven.
542    ///
543    /// So: exact first, and only widen when nothing was exact — the same shape,
544    /// returning the same [`TitleMatch`], so a caller can say which of the two
545    /// happened and refuse to arbitrate between several.
546    pub fn find_artists(&self, name: &str) -> (Vec<&Artist>, TitleMatch) {
547        let key = text::normalize(name);
548        if key.is_empty() {
549            return (Vec::new(), TitleMatch::Exact);
550        }
551        let exact: Vec<&Artist> = self.artists.iter().filter(|a| a.key == key).collect();
552        if !exact.is_empty() {
553            return (exact, TitleMatch::Exact);
554        }
555        let partial = self
556            .artists
557            .iter()
558            .filter(|a| a.key.contains(&key))
559            .collect();
560        (partial, TitleMatch::Partial)
561    }

563    /// One release by MusicBrainz identity, or every title match.
564    ///
565    /// Same rule as [`Catalog::find_tracks`], and for the same reason: a
566    /// command must not pick one answer out of several without saying so.
567    /// An identifier wins before names. Unlike tracks, two matching albums are
568    /// two *different* editions — a shared prefix is not an ambiguity — which
569    /// is why an exact title usually ends the name search without choosing one.
570    pub fn find_releases(&self, title: &str) -> (Vec<&Release>, TitleMatch) {
571        let identified: Vec<&Release> = self
572            .releases
573            .iter()
574            .filter(|release| release.mbid.as_deref() == Some(title))
575            .collect();
576        if !identified.is_empty() {
577            return (identified, TitleMatch::Exact);
578        }
579        let key = text::normalize(title);
580        if key.is_empty() {
581            return (Vec::new(), TitleMatch::Exact);
582        }
583        let exact: Vec<&Release> = self.releases.iter().filter(|r| r.key == key).collect();
584        if !exact.is_empty() {
585            return (exact, TitleMatch::Exact);
586        }
587        let partial = self
588            .releases
589            .iter()
590            .filter(|r| r.key.contains(&key))
591            .collect();
592        (partial, TitleMatch::Partial)
593    }

595    /// Every track carrying this title, up to normalization.
596    ///
597    /// A title is not an identifier: the same one legitimately comes back on
598    /// the album, on a single and on a live record, and those are different
599    /// recordings. All of them are returned, in catalog order.
600    ///
601    /// Exact matches win. Only when there is none does the search widen to the
602    /// titles containing the text, so that a half-remembered title still leads
603    /// somewhere; [`TitleMatch`] says which of the two happened.
604    pub fn find_tracks(&self, title: &str) -> (Vec<&Track>, TitleMatch) {
605        let key = text::normalize(title);
606        if key.is_empty() {
607            return (Vec::new(), TitleMatch::Exact);
608        }
609        let exact: Vec<&Track> = self
610            .tracks
611            .iter()
612            .filter(|t| text::normalize(&t.title) == key)
613            .collect();
614        if !exact.is_empty() {
615            return (exact, TitleMatch::Exact);
616        }
617        let partial = self
618            .tracks
619            .iter()
620            .filter(|t| text::normalize(&t.title).contains(&key))
621            .collect();
622        (partial, TitleMatch::Partial)
623    }

625    /// Every genre whose name matches, exactly or failing that partially.
626    ///
627    /// Same rule as [`Catalog::find_releases`]: "metal" must reach the genre
628    /// spelled exactly that way without hiding "Black Metal" and "Doom Metal"
629    /// when there is no exact one.
630    pub fn find_genres(&self, name: &str) -> (Vec<&Genre>, TitleMatch) {
631        let key = text::normalize(name);
632        if key.is_empty() {
633            return (Vec::new(), TitleMatch::Exact);
634        }
635        let exact: Vec<&Genre> = self.genres.iter().filter(|g| g.key == key).collect();
636        if !exact.is_empty() {
637            return (exact, TitleMatch::Exact);
638        }
639        (
640            self.genres
641                .iter()
642                .filter(|g| g.key.contains(&key))
643                .collect(),
644            TitleMatch::Partial,
645        )
646    }

648    /// Every label whose name matches, exactly or failing that partially.
649    pub fn find_labels(&self, name: &str) -> (Vec<&Label>, TitleMatch) {
650        let key = text::normalize(name);
651        if key.is_empty() {
652            return (Vec::new(), TitleMatch::Exact);
653        }
654        let exact: Vec<&Label> = self.labels.iter().filter(|l| l.key == key).collect();
655        if !exact.is_empty() {
656            return (exact, TitleMatch::Exact);
657        }
658        (
659            self.labels
660                .iter()
661                .filter(|l| l.key.contains(&key))
662                .collect(),
663            TitleMatch::Partial,
664        )
665    }

667    /// Tracks carrying a genre, directly or through their release.
668    ///
669    /// A genre attached to an album is a genre of every track on it: tags put
670    /// it in either place, and a listener does not think of the difference.
671    pub fn tracks_of_genre(&self, genre_id: Id) -> Vec<Id> {
672        let mut tracks: BTreeSet<Id> = BTreeSet::new();
673        for link in &self.genre_links {
674            if link.genre_id != genre_id {
675                continue;
676            }
677            match link.entity_kind {
678                EntityKind::Track => {
679                    tracks.insert(link.entity_id);
680                }
681                EntityKind::Release => {
682                    if let Some(release) = self.release(link.entity_id) {
683                        tracks.extend(release.track_ids.iter().copied());
684                    }
685                }
686                EntityKind::Artist
687                | EntityKind::Recording
688                | EntityKind::Work
689                | EntityKind::ReleaseGroup
690                | EntityKind::Label
691                | EntityKind::Genre => {}
692            }
693        }
694        tracks.into_iter().collect()
695    }

697    /// Releases holding at least one track of a genre, or carrying it whole.
698    pub fn releases_of_genre(&self, genre_id: Id) -> Vec<Id> {
699        let tracks: BTreeSet<Id> = self.tracks_of_genre(genre_id).into_iter().collect();
700        let mut releases: BTreeSet<Id> = BTreeSet::new();
701        for track in self.tracks.iter().filter(|t| tracks.contains(&t.id)) {
702            if let Some(id) = track.release_id {
703                releases.insert(id);
704            }
705        }
706        releases.into_iter().collect()
707    }

727    /// Everyone credited in a role, with how many credits each holds.
728    ///
729    /// This is the inverse of the artist page: that one answers "what did this
730    /// person do", this one answers "who does this in my library". The whole
731    /// point of storing roles rather than a bare artist column is that the
732    /// question can be asked in both directions.
733    ///
734    /// Sorted by number of credits, then by sort name, so the answer is stable
735    /// and the most present come first.
736    pub fn artists_in_role(&self, role: &str) -> Vec<(Id, usize)> {
737        let mut counts: BTreeMap<Id, usize> = BTreeMap::new();
738        for credit in self.credits.iter().filter(|c| c.role == role) {
739            *counts.entry(credit.artist_id).or_insert(0) += 1;
740        }
741        let mut rows: Vec<(Id, usize)> = counts.into_iter().collect();
742        rows.sort_by(|a, b| {
743            b.1.cmp(&a.1).then_with(|| {
744                self.artist(a.0)
745                    .map(|x| x.sort_name.as_str())
746                    .cmp(&self.artist(b.0).map(|x| x.sort_name.as_str()))
747            })
748        });
749        rows
750    }

752    /// Tracks on which an artist holds one particular role.
753    ///
754    /// The other direction of [`Catalog::artists_in_role`], one person at a
755    /// time: not "who produces here" but "what did this person produce". The
756    /// artist page separates performing from writing; this goes one step
757    /// finer, down to the single role.
758    ///
759    /// `album` is credited on the **release**, not on its tracks, so asking
760    /// for it yields every track of the releases they sign — otherwise the
761    /// answer would be empty for the one role every album artist holds.
762    pub fn tracks_of_artist_in_role(&self, artist_id: Id, role: &str) -> Vec<Id> {
763        let mut tracks: BTreeSet<Id> = BTreeSet::new();
764        for credit in self
765            .credits
766            .iter()
767            .filter(|c| c.artist_id == artist_id && c.role == role)
768        {
769            match credit.entity_kind {
770                EntityKind::Track => {
771                    tracks.insert(credit.entity_id);
772                }
773                EntityKind::Release => {
774                    if let Some(release) = self.release(credit.entity_id) {
775                        tracks.extend(release.track_ids.iter().copied());
776                    }
777                }
778                EntityKind::Artist
779                | EntityKind::Recording
780                | EntityKind::Work
781                | EntityKind::ReleaseGroup
782                | EntityKind::Label
783                | EntityKind::Genre => {}
784            }
785        }
786        tracks.into_iter().collect()
787    }

789    /// The roles one artist holds, with how many credits each.
790    ///
791    /// What makes an empty answer readable: told that nobody is credited a
792    /// given way, the user has to be able to see what this person *is*
793    /// credited as.
794    pub fn roles_of_artist(&self, artist_id: Id) -> Vec<(&str, usize)> {
795        let mut counts: BTreeMap<&str, usize> = BTreeMap::new();
796        for credit in self.credits.iter().filter(|c| c.artist_id == artist_id) {
797            *counts.entry(credit.role.as_str()).or_insert(0) += 1;
798        }
799        let mut rows: Vec<(&str, usize)> = counts.into_iter().collect();
800        rows.sort_by(|a, b| b.1.cmp(&a.1).then_with(|| a.0.cmp(b.0)));
801        rows
802    }

804    /// Every role actually used in this catalog, in the order they are shown.
805    ///
806    /// Read from the credits rather than from a fixed list: a role that came
807    /// from a tag nobody anticipated is still a role, and it must be offerable.
808    pub fn roles_in_use(&self) -> Vec<&str> {
809        let mut roles: BTreeSet<&str> = BTreeSet::new();
810        for credit in &self.credits {
811            roles.insert(credit.role.as_str());
812        }
813        roles.into_iter().collect()
814    }

816    /// Tracks whose file carries a comment containing this text.
817    ///
818    /// The comment belongs to the **file**, not to the track: it is free text a
819    /// user wrote, and it survives in the raw tags exactly as it was typed.
820    /// Matching is done on the normalized form, so accents and case are no
821    /// obstacle — a comment is prose, and prose is typed carelessly.
822    pub fn tracks_with_comment(&self, text: &str) -> Vec<Id> {
823        let needle = text::normalize(text);
824        if needle.is_empty() {
825            return Vec::new();
826        }
827        self.tracks
828            .iter()
829            .filter(|t| {
830                self.file(t.file_id)
831                    .and_then(|f| f.first_tag("comment"))
832                    .map(|c| text::normalize(c).contains(&needle))
833                    .unwrap_or(false)
834            })
835            .map(|t| t.id)
836            .collect()
837    }
838
839    /// The words of a track: the tag that carries them, or the `.lrc` beside
840    /// the file.
841    ///
842    /// **The one place that decides which of the two wins.** The tag is
843    /// preferred, and costs nothing since raw tags are in the catalog; the
844    /// sidecar is opened only when the tag holds none, and only for the tracks
845    /// that have one. Three callers ask this question — the track page, the
846    /// `lyrics:` field of the grammar, `search --lyrics` — and three answers
847    /// that could disagree about precedence is two too many.
848    pub fn lyrics_of_track(&self, track_id: Id) -> Option<crate::lyrics::Lyrics> {
849        let file = self.file(self.track(track_id)?.file_id)?;
850        file.first_tag("lyrics")
851            .and_then(|text| crate::lyrics::from_tag(&file.path, text))
852            .or_else(|| crate::lyrics::read(std::path::Path::new(file.lyrics_path.as_ref()?)))
853    }
854
855    /// The tracks whose words carry this text, with the line that carries it.
856    ///
857    /// The **line**, not the song: a table cell holding four hundred lines is a
858    /// table nobody can read, and what somebody searching for "all aboard"
859    /// wants to see is the line they half-remembered, in the shape they
860    /// half-remembered it.
861    pub fn tracks_with_lyric(&self, text: &str) -> Vec<(Id, String)> {
862        let needle = text::normalize(text);
863        if needle.is_empty() {
864            return Vec::new();
865        }
866        self.tracks
867            .iter()
868            .filter_map(|track| {
869                let lyrics = self.lyrics_of_track(track.id)?;
870                let line = lyrics
871                    .lines
872                    .iter()
873                    .find(|line| text::normalize(&line.text).contains(&needle))?;
874                Some((track.id, line.text.clone()))
875            })
876            .collect()
877    }

879    /// The comment carried by the file behind a track, when there is one.
880    pub fn comment_of_track(&self, track_id: Id) -> Option<&str> {
881        let track = self.track(track_id)?;
882        self.file(track.file_id)?.first_tag("comment")
883    }

890/// How [`Catalog::find_tracks`] reached its results.
