1//! Turning scanned files into a graph of entities.
2//!
3//! Construction is **deterministic**: the files are sorted by path before
4//! anything else happens, so two scans of the same library produce exactly the
5//! same identifiers. Without that there is neither a readable diff between two
6//! catalogs nor a reproducible test.
7//!
8//! The work is one pass per file, with every name interned as it is met, and a
9//! `finalize` step at the end for what can only be known once everything is in:
10//! an album's year, its cover, the order of its tracks.
11

21};

27/// A file read by the scanner, before entity matching.
28#[derive(Debug, Clone)]
29pub struct ScannedFile {
30    /// Absolute path of the file that was read.
31    pub path: String,
32    /// Size in bytes at the moment of the scan.
33    pub size: u64,
34    /// Modification date in seconds since the Unix epoch. The scanner stores
35    /// its nanosecond fraction separately on [`Catalog::file_mtime_subseconds`]
36    /// after building the graph; both parts and `size` decide cache reuse.
37    pub mtime: u64,
38    /// Tags and stream properties, exactly as the decoder handed them over.
39    pub tags: RawTags,
40    /// Cover art found in the folder (`cover.jpg`, `folder.png`…).
41    pub folder_cover: Option<String>,
42    /// A `.lrc` file sitting beside this one, when the walk saw one.
43    pub sidecar: Option<String>,
44    /// Integrity verdict carried over from the previous catalog, when the file
45    /// was reused unchanged. A freshly read file has none: it has to be checked
46    /// again.
47    pub integrity: Option<IntegrityRecord>,
48    /// Fingerprint carried over the same way, and dropped for the same reason:
49    /// a file that changed is different audio, and its old fingerprint
50    /// describes something that is no longer there.
51    pub fingerprint: Option<crate::fingerprint::Fingerprint>,
52}
53
54/// Assembles the graph from the files that were read.
55///
56/// Processing order is deterministic — files are sorted by path first — so two
57/// scans of the same library produce exactly the same identifiers. Without
58/// that, neither a readable diff nor a reproducible test is possible.
59pub fn build(
60    mut scanned: Vec<ScannedFile>,
61    roots: Vec<String>,
62    scanned_at: u64,
63    chosen: &[super::identity::Chosen],
64) -> Catalog {
65    scanned.sort_by(|a, b| a.path.cmp(&b.path));
66
67    // **Read before anything is interned.** Two spellings under one MusicBrainz
68    // identifier are one artist, and which spelling survives is decided by the
69    // library as a whole — the most frequent one — so it cannot be settled file
70    // by file as the walk goes. `chosen` is what the owner of the disk said
71    // about the files no identifier can answer for, and it is resolved in the
72    // same pass rather than applied over it. See [`super::identity`].
73    let aliases = super::identity::aliases(scanned.iter().flat_map(named_once), chosen);
74
75    let mut builder = Builder::new(roots, scanned_at, aliases);
76    for item in &scanned {
77        builder.add_file(item);
78    }
79    builder.finish()
80}
81
82/// Interning state shared by every step of the construction.
83///
84/// Each map answers "have I already seen this name?" in constant time. Doing
85/// it with a linear scan made construction quadratic on real libraries.
86struct Builder {
87    catalog: Catalog,
88    /// Spellings that are somebody else's, worked out before the walk.
89    aliases: super::identity::Aliases,
90    artists: HashMap<String, Id>,
91    labels: HashMap<String, Id>,
92    genres: HashMap<String, Id>,
93    releases: HashMap<String, Id>,
94    release_groups: HashMap<String, Id>,
95    recordings: HashMap<String, Id>,
96    works: HashMap<String, Id>,
97    credits: HashSet<CreditKey>,
98    release_genres: HashSet<(Id, Id)>,
99}

101/// Everything one file contributes, resolved before the entities are wired up.
102struct FileEntities {
103    file_id: Id,
104    artist_ids: Vec<Id>,
105    album_artist_ids: Vec<Id>,
106    release_id: Option<Id>,

108    /// The disc the file's own folder announces, when it announces one, so
109    /// that a rip which put each disc in its own folder and left the tag
110    /// empty still knows which disc a track is on.

112}

135    /// Turns one scanned file into a file row, a track, and all the entities
136    /// and links its tags imply.
137    fn add_file(&mut self, item: &ScannedFile) {
138        let entities = self.add_entities(item);
139        let track_id = self.add_track(item, &entities);
140        let recording_id = self.catalog.tracks[track_id as usize].recording_id;
141        self.add_works(item, recording_id);
142        self.add_credits(item, &entities, track_id);
143        self.add_labels(item, &entities);
144        self.add_genres(item, &entities, track_id);
145    }

147    /// Records the physical file and resolves the artists and the release.
148    fn add_entities(&mut self, item: &ScannedFile) -> FileEntities {
149        let tags = &item.tags;
150        let file_id = self.catalog.files.len() as Id;
151        let file = AudioFile {
152            id: file_id,
153            path: item.path.clone(),
154            size: item.size,
155            mtime: item.mtime,
156            properties: tags.properties.clone(),
157            has_embedded_art: tags.has_embedded_art,
158            tags: tags.fields.clone(),
159            lyrics_path: item.sidecar.clone(),
160            integrity: item.integrity.clone(),
161            fingerprint: item.fingerprint.clone(),
162        };
163        // Where the *release* lives, which is not always where the file does.
164        // A box set laid out as `Album/Disc 1`, `Album/Disc 2` is one album:
165        // the disc folder is a subdivision of the release, not another edition
166        // of it, and keying the release on it split one soundtrack into two
167        // albums of the same name each numbering its tracks from one.
168        let file_folder = file.folder().to_string();
169        let disc_from_folder = text::disc_folder(text::file_name(&file_folder));
170        let folder = match disc_from_folder {
171            Some(_) => text::folder(&file_folder).to_string(),
172            None => file_folder,
173        };
174        self.catalog.files.push(file);
175
176        let artist_names = credited(tags, "artists", "artist");
177        let album_artist_names = credited(tags, "albumartists", "albumartist");
178
179        let artist_ids: Vec<Id> = artist_names
180            .iter()
181            .map(|n| {
182                let id = self.intern_artist(n);
183                self.name_artist(id, n);
184                id
185            })
186            .collect();
187        let album_artist_ids: Vec<Id> = album_artist_names
188            .iter()
189            .filter(|n| !is_various_artists(n))
190            .map(|n| {
191                let id = self.intern_artist(n);
192                self.name_artist(id, n);
193                id
194            })
195            .collect();
196
197        // MBIDs encountered along the way enrich the artist they name — both
198        // of them, since an album artist's identifier used to be read by
199        // nothing at all and is exactly as good as a track artist's.
200        for (ids, tag) in [
201            (&artist_ids, "musicbrainz_artistid"),
202            (&album_artist_ids, "musicbrainz_albumartistid"),
203        ] {
204            // One name and one identifier, or nothing — the same rule
205            // `named_once` applies, and for the same reason: with two of
206            // either, position is not evidence of which belongs to which.
207            let (Some(mbid), [id]) = (only_one(tags.all(tag)), ids.as_slice()) else {
208                continue;
209            };
210            if let Some(artist) = self.catalog.artists.get_mut(*id as usize) {
211                artist.mbid.get_or_insert_with(|| mbid.to_string());
212            }
213        }
214
215        let is_compilation = tags.first("compilation").is_some_and(|value| {
216            matches!(
217                value.trim().to_ascii_lowercase().as_str(),
218                "1" | "true" | "yes"
219            )
220        }) || album_artist_names.iter().any(|n| is_various_artists(n));
221        let release_id = self.release_for(
222            item,
223            &folder,
224            &artist_ids,
225            &album_artist_ids,
226            is_compilation,
227        );
228
229        FileEntities {
230            file_id,
231            artist_ids,
232            album_artist_ids,
233            release_id,
234            is_compilation,
235            disc_from_folder,
236        }
237    }

239    /// Finds or creates the release this file belongs to.
240    fn release_for(
241        &mut self,
242        item: &ScannedFile,
243        folder: &str,
244        artist_ids: &[Id],
245        album_artist_ids: &[Id],
246        is_compilation: bool,
247    ) -> Option<Id> {
248        let tags = &item.tags;
249        let title = tags.first("album").map(|s| s.trim().to_string())?;
250
251        let album_artist = if is_compilation {
252            // A compilation has no album artist: that is what distinguishes it
253            // from an ordinary album.
254            None
255        } else {
256            // With no declared album artist, fall back to the track artist.
257            album_artist_ids
258                .first()
259                .copied()
260                .or_else(|| artist_ids.first().copied())
261        };
262
263        let owner = if is_compilation {
264            "__va__".to_string()
265        } else {
266            album_artist
267                .and_then(|id| self.catalog.artist(id).map(|a| a.key.clone()))
268                .unwrap_or_default()
269        };
270        // The folder separates two same-named albums by the same artist, for
271        // instance two editions of one record.
272        let key = format!("{}|{}|{}", text::normalize(&title), owner, folder);
273
274        if let Some(&id) = self.releases.get(&key) {
275            // A field missing from the first track may show up on another one.
276            let missing_group = self
277                .catalog
278                .releases
279                .get(id as usize)
280                .filter(|release| release.release_group_id.is_none())
281                .map(|release| release.title.clone());
282            if let Some(release) = self.catalog.releases.get_mut(id as usize) {
283                if release.cover_path.is_none() {
284                    release.cover_path = item.folder_cover.clone();
285                }
286                if release.catalog_number.is_none() {
287                    release.catalog_number = tags.first("catalognumber").map(String::from);
288                }
289                if release.mbid.is_none() {
290                    release.mbid = tags.first("musicbrainz_albumid").map(String::from);
291                }
292            }
293            if let Some(title) = missing_group
294                && let Some(group_id) =
295                    self.release_group_for(&title, tags.first("musicbrainz_releasegroupid"))
296            {
297                self.catalog.releases[id as usize].release_group_id = Some(group_id);
298                self.catalog.release_groups[group_id as usize]
299                    .release_ids
300                    .push(id);
301            }
302            return Some(id);
303        }
304
305        let id = self.catalog.releases.len() as Id;
306        let release_group_mbid = tags.first("musicbrainz_releasegroupid").map(String::from);
307        let release_group_id = self.release_group_for(&title, release_group_mbid.as_deref());
308        self.catalog.releases.push(Release {
309            id,
310            key: text::normalize(&title),
311            title,
312            album_artist_id: album_artist,
313            date: None,
314            year: None,
315            label_ids: Vec::new(),
316            catalog_number: tags.first("catalognumber").map(String::from),
317            barcode: tags.first("barcode").map(String::from),
318            media: tags.first("media").map(String::from),
319            mbid: tags.first("musicbrainz_albumid").map(String::from),
320            release_group_mbid,
321            release_group_id,
322            is_compilation,
323            folder: folder.to_string(),
324            cover_path: item.folder_cover.clone(),
325            track_ids: Vec::new(),
326        });
327        if let Some(group_id) = release_group_id {
328            self.catalog.release_groups[group_id as usize]
329                .release_ids
330                .push(id);
331        }
332        self.releases.insert(key, id);
333        Some(id)
334    }

336    /// Resolves a release group only from its explicit MusicBrainz identifier.
337    fn release_group_for(&mut self, title: &str, mbid: Option<&str>) -> Option<Id> {
338        let mbid = mbid?.trim();
339        if mbid.is_empty() {
340            return None;
341        }
342        if let Some(&id) = self.release_groups.get(mbid) {
343            return Some(id);
344        }
345        let id = self.catalog.release_groups.len() as Id;
346        self.catalog.release_groups.push(ReleaseGroup {
347            id,
348            title: title.to_string(),
349            key: text::normalize(title),
350            mbid: mbid.to_string(),
351            release_ids: Vec::new(),
352        });
353        self.release_groups.insert(mbid.to_string(), id);
354        Some(id)
355    }

357    /// Creates the track and attaches it to its release.
358    fn add_track(&mut self, item: &ScannedFile, entities: &FileEntities) -> Id {
359        let tags = &item.tags;
360        let title = tags
361            .first("title")
362            .map(|s| s.trim().to_string())
363            .unwrap_or_else(|| title_from_filename(&item.path));
364
365        let (track_no, _) = tags
366            .first("tracknumber")
367            .map(text::parse_track_number)
368            .unwrap_or((None, None));
369        let (disc_no, _) = tags
370            .first("discnumber")
371            .map(text::parse_track_number)
372            .unwrap_or((None, None));
373        // A rip that put each disc in its own folder and left the tag empty is
374        // common, and without this the two discs would both be disc one — two
375        // track 1s in one release, which is worse than the split it replaces.
376        // The tag still wins where there is one: it is what the person who
377        // made the file said.
378        let disc_no = disc_no.or(entities.disc_from_folder);
379
380        let recording_id = self.recording_for(item, &title);
381        let track_id = self.catalog.tracks.len() as Id;
382        self.catalog.tracks.push(Track {
383            id: track_id,
384            file_id: entities.file_id,
385            release_id: entities.release_id,
386            recording_id,
387            title,
388            disc_no,
389            track_no: track_no.or_else(|| track_from_filename(&item.path)),
390            duration_ms: tags.properties.duration_ms,
391            isrc: tags.first("isrc").map(String::from),
392            mbid: tags.first("musicbrainz_recordingid").map(String::from),
393        });
394        self.catalog.recordings[recording_id as usize]
395            .track_ids
396            .push(track_id);
397
398        if let Some(rid) = entities.release_id
399            && let Some(release) = self.catalog.releases.get_mut(rid as usize)
400        {
401            release.track_ids.push(track_id);
402        }
403        track_id
404    }

406    /// Finds the recorded performance this placement carries, or gives an
407    /// un-identified placement a recording of its own. A shared title is not
408    /// enough to merge music: only an explicit MusicBrainz recording ID is.
409    fn recording_for(&mut self, item: &ScannedFile, title: &str) -> Id {
410        let mbid = item
411            .tags
412            .first("musicbrainz_recordingid")
413            .map(str::to_owned);
414        let key = match &mbid {
415            Some(id) => format!("mbid:{id}"),
416            None => format!("local:{}", item.path),
417        };
418        if let Some(&id) = self.recordings.get(&key) {
419            return id;
420        }
421        let id = self.catalog.recordings.len() as Id;
422        self.catalog.recordings.push(Recording {
423            id,
424            title: title.to_string(),
425            key: text::normalize(title),
426            isrc: item.tags.first("isrc").map(String::from),
427            mbid,
428            track_ids: Vec::new(),
429            work_ids: Vec::new(),
430        });
431        self.recordings.insert(key, id);
432        id
433    }

435    /// Connects one recording to every explicitly identified work its tags
436    /// name. Several works are allowed: a medley is not forced into one title.
437    fn add_works(&mut self, item: &ScannedFile, recording_id: Id) {
438        for (position, mbid) in item.tags.all("musicbrainz_workid").iter().enumerate() {
439            let mbid = mbid.trim();
440            if mbid.is_empty() {
441                continue;
442            }
443            let work_id = match self.works.get(mbid) {
444                Some(&id) => id,
445                None => {
446                    let title = item
447                        .tags
448                        .all("grouping")
449                        .get(position)
450                        .map(String::as_str)
451                        .or_else(|| item.tags.first("grouping"))
452                        .unwrap_or(&self.catalog.recordings[recording_id as usize].title);
453                    let id = self.catalog.works.len() as Id;
454                    self.catalog.works.push(Work {
455                        id,
456                        title: title.to_string(),
457                        key: text::normalize(title),
458                        mbid: mbid.to_string(),
459                        recording_ids: Vec::new(),
460                    });
461                    self.works.insert(mbid.to_string(), id);
462                    id
463                }
464            };
465            let recording = &mut self.catalog.recordings[recording_id as usize];
466            if !recording.work_ids.contains(&work_id) {
467                recording.work_ids.push(work_id);
468                self.catalog.works[work_id as usize]
469                    .recording_ids
470                    .push(recording_id);
471            }
472        }
473    }

475    /// Records who did what on this track, and who signs the album.
476    fn add_credits(&mut self, item: &ScannedFile, entities: &FileEntities, track_id: Id) {
477        for &artist_id in &entities.artist_ids {
478            self.push_credit(artist_id, EntityKind::Track, track_id, "main");
479        }
480        for role in ROLE_TAGS {
481            // The whole tag at once, not one value at a time: whether a value
482            // restates the others can only be asked of the list it belongs to,
483            // and `PERFORMER` is where a collaboration is habitually written
484            // out twice — once per musician, then once as the joint credit.
485            for name in credited_under(&item.tags, role) {
486                let id = self.intern_artist(&name);
487                self.push_credit(id, EntityKind::Track, track_id, role);
488            }
489        }
490        // Picard and the Vorbis Comment convention encode instruments in the
491        // field name itself: `PERFORMER:guitar=Name`. Preserve the instrument
492        // on the relationship instead of flattening every such row into an
493        // indistinguishable performer credit.
494        for key in item
495            .tags
496            .fields
497            .keys()
498            .filter(|key| key.starts_with("performer:") && key.len() > "performer:".len())
499        {
500            let instrument = key["performer:".len()..].trim();
501            for (order, name) in credited_under(&item.tags, key).into_iter().enumerate() {
502                let id = self.intern_artist(&name);
503                let credited_as = self
504                    .catalog
505                    .artist(id)
506                    .filter(|artist| artist.name != name)
507                    .map(|_| name.clone());
508                self.push_credit_details(
509                    id,
510                    EntityKind::Track,
511                    track_id,
512                    "performer",
513                    credited_as,
514                    vec![super::CreditAttribute {
515                        id: None,
516                        name: instrument.to_string(),
517                        value: None,
518                        credited_as: None,
519                    }],
520                    Some(order as u32),
521                );
522            }
523        }
524        if let Some(rid) = entities.release_id
525            && !entities.is_compilation
526        {
527            for &artist_id in &entities.album_artist_ids {
528                self.push_credit(artist_id, EntityKind::Release, rid, "album");
529            }
530        }
531        if let Some(rid) = entities.release_id {
532            for &(tag, role) in ALBUM_ROLE_TAGS {
533                for name in credited_under(&item.tags, tag) {
534                    let id = self.intern_artist(&name);
535                    self.push_credit(id, EntityKind::Release, rid, role);
536                }
537            }
538        }
539    }

541    fn add_labels(&mut self, item: &ScannedFile, entities: &FileEntities) {
542        let Some(rid) = entities.release_id else {
543            return;
544        };
545        for (position, value) in item.tags.all("label").iter().enumerate() {
546            let mbid = item
547                .tags
548                .all("musicbrainz_labelid")
549                .get(position)
550                .map(String::as_str)
551                .or_else(|| item.tags.first("musicbrainz_labelid"));
552            let label_id = self.intern_label(value, mbid);
553            if let Some(release) = self.catalog.releases.get_mut(rid as usize)
554                && !release.label_ids.contains(&label_id)
555            {
556                release.label_ids.push(label_id);
557            }
558        }
559    }

561    /// Attaches genres to the track, and to its release the first time round.
562    fn add_genres(&mut self, item: &ScannedFile, entities: &FileEntities, track_id: Id) {
563        let mut track_genres = HashSet::new();
564        for value in item.tags.all("genre") {
565            for name in value.split(';').map(str::trim).filter(|s| !s.is_empty()) {
566                let genre_id = self.intern_genre(name);
567                if !track_genres.insert(genre_id) {
568                    continue;
569                }
570                self.catalog.genre_links.push(GenreLink {
571                    genre_id,
572                    entity_kind: EntityKind::Track,
573                    entity_id: track_id,
574                });
575                if let Some(rid) = entities.release_id
576                    && self.release_genres.insert((genre_id, rid))
577                {
578                    self.catalog.genre_links.push(GenreLink {
579                        genre_id,
580                        entity_kind: EntityKind::Release,
581                        entity_id: rid,
582                    });
583                }
584            }
585        }
586    }

588    /// The artist this spelling belongs to, creating them the first time.
589    ///
590    /// A spelling MusicBrainz places under another one is filed there, and the
591    /// row is **named for the spelling it is filed under** rather than for
592    /// whichever file was read first: the key and the displayed name have to
593    /// agree, or a listing shows `O. Osbourne` and a query for `ozzy osbourne`
594    /// finds it, which is worse than either alone.
595    fn intern_artist(&mut self, name: &str) -> Id {
596        let spelling = text::normalize(name);
597        let key = super::identity::filed_as(&self.aliases, &spelling).to_string();
598        if let Some(&id) = self.artists.get(&key) {
599            return id;
600        }
601        // `name` is one of the spellings; the surviving key is another. Naming
602        // the row from the key would print a normalised string — lower case,
603        // no punctuation — where a reader expects a name, so the *first file
604        // carrying the surviving spelling* supplies it. Until one is read, the
605        // spelling in hand stands in, and is replaced below.
606        let display = match key == spelling {
607            true => name.trim().to_string(),
608            false => key.clone(),
609        };
610        // The other spellings that were filed here, kept so the merge can be
611        // seen: two rows becoming one is a decision, and a reader who cannot
612        // see it cannot tell it from a folder that was never scanned.
613        let mut aliases: Vec<String> = self
614            .aliases
615            .iter()
616            .filter(|(_, under)| **under == key)
617            .map(|(spelling, _)| spelling.clone())
618            .collect();
619        aliases.sort();
620        let id = self.catalog.artists.len() as Id;
621        self.catalog.artists.push(Artist {
622            id,
623            name: display.clone(),
624            sort_name: text::sort_name(&display),
625            key: key.clone(),
626            mbid: None,
627            aliases,
628        });
629        self.artists.insert(key, id);
630        id
631    }
632
633    /// Gives an artist the spelling their surviving key was taken from.
634    ///
635    /// Called with every name read, and it only ever acts once: the row keeps
636    /// the first *unnormalised* form of the winning spelling that the walk
637    /// meets. Without it a merged artist would be displayed by their key, which
638    /// is a matching form and not a name.
639    fn name_artist(&mut self, id: Id, name: &str) {
640        let Some(artist) = self.catalog.artists.get_mut(id as usize) else {
641            return;
642        };
643        if artist.name != artist.key || text::normalize(name) != artist.key {
644            return;
645        }
646        artist.name = name.trim().to_string();
647        artist.sort_name = text::sort_name(name.trim());
648    }
649
650    fn intern_label(&mut self, name: &str, mbid: Option<&str>) -> Id {
651        let key = text::normalize(name);
652        if let Some(&id) = self.labels.get(&key) {
653            if self.catalog.labels[id as usize].mbid.is_none() {
654                self.catalog.labels[id as usize].mbid = mbid.map(str::to_owned);
655            }
656            return id;
657        }
658        let id = self.catalog.labels.len() as Id;
659        self.catalog.labels.push(Label {
660            id,
661            name: name.trim().to_string(),
662            key: key.clone(),
663            mbid: mbid.map(str::to_owned),
664        });
665        self.labels.insert(key, id);
666        id
667    }
668
669    fn intern_genre(&mut self, name: &str) -> Id {
670        let key = text::normalize(name);
671        if let Some(&id) = self.genres.get(&key) {
672            return id;
673        }
674        let id = self.catalog.genres.len() as Id;
675        self.catalog.genres.push(Genre {
676            id,
677            name: name.trim().to_string(),
678            key: key.clone(),
679        });
680        self.genres.insert(key, id);
681        id
682    }
683
684    /// Adds a credit unless the same one is already recorded.
685    fn push_credit(&mut self, artist_id: Id, kind: EntityKind, id: Id, role: &str) {
686        self.push_credit_details(artist_id, kind, id, role, None, Vec::new(), None);
687    }
688
689    fn push_credit_details(
690        &mut self,
691        artist_id: Id,
692        kind: EntityKind,
693        id: Id,
694        role: &str,
695        credited_as: Option<String>,
696        attributes: Vec<super::CreditAttribute>,
697        order: Option<u32>,
698    ) {
699        let attribute_key = attributes
700            .iter()
701            .map(|attribute| {
702                format!(
703                    "{}\u{1f}{}\u{1f}{}",
704                    attribute.name,
705                    attribute.value.as_deref().unwrap_or(""),
706                    attribute.credited_as.as_deref().unwrap_or("")
707                )
708            })
709            .collect();
710        if self
711            .credits
712            .insert((artist_id, kind, id, role.to_string(), attribute_key))
713        {
714            self.catalog.credits.push(Credit {
715                artist_id,
716                entity_kind: kind,
717                entity_id: id,
718                role: role.to_string(),
719                credited_as,
720                attributes,
721                began: None,
722                ended: None,
723                order,
724                source: "tags".to_string(),
725                source_id: None,
726            });
727        }
728    }
729
730    fn finish(mut self) -> Catalog {
731        finalize(&mut self.catalog);
732        self.catalog
733    }
734}

774/// Key of a credit, used to reject duplicates in constant time.
775type CreditKey = (Id, EntityKind, Id, String, Vec<String>);

777/// The unambiguous (identifier, name) pairs one file states.
778///
779/// A file speaks only where it names **exactly one** artist and carries
780/// **exactly one** identifier for them. A tag naming two artists arrives as one
781/// string that `split_artists` cuts in two, while the identifiers arrive as
782/// their own list in an order nothing guarantees to match — pairing those by
783/// position would file an identifier against whichever name sorted first. Both
784/// the artist and the album artist are read, since either is evidence.
785fn named_once(item: &ScannedFile) -> Vec<super::identity::Said> {
786    let mut said = Vec::new();
787    for (plural, single, tag) in [
788        ("artists", "artist", "musicbrainz_artistid"),
789        ("albumartists", "albumartist", "musicbrainz_albumartistid"),
790    ] {
791        // Resolved exactly as the credit itself is, or the two would disagree
792        // about how many artists a file names — and this pre-pass would call a
793        // collaboration one artist while the walk called it two.
794        let names = credited(&item.tags, plural, single);
795        let (Some(mbid), [name]) = (only_one(item.tags.all(tag)), names.as_slice()) else {
796            continue;
797        };
798        if is_various_artists(name) {
799            continue;
800        }
801        said.push(super::identity::Said {
802            mbid: mbid.to_string(),
803            name: name.clone(),
804        });
805    }
806    said
807}

809/// The artists a file credits, from the tag written for the purpose.
810///
811/// **`ARTISTS` before `ARTIST`, and it is not a nicety.** `ARTIST` is one
812/// string holding however many names, and the conventions for joining them do
813/// not survive contact with a real library: `Rob Zombie & Ozzy Osbourne` is two
814/// artists, `Simon & Garfunkel` is one, and nothing in either string says
815/// which. `ARTISTS` is the tag taggers write with **one value per artist** for
816/// exactly this reason, and a file that carries it has already answered the
817/// question — no separator to guess at, no band name to shatter.
818///
819/// Each value still goes through [`text::split_artists`], because a tagger that
820/// writes `ARTISTS` as one joined value is not rare either, and splitting a
821/// value that holds one name gives that name back.
822fn credited(tags: &RawTags, plural: &str, single: &str) -> Vec<String> {
823    let named = credited_under(tags, plural);
824    match named.is_empty() {
825        false => named,
826        true => credited_under(tags, single),
827    }
828}
829
830/// Every artist one tag names, split and freed of the values that only restate
831/// the others.
832///
833/// The scope matters and is the whole point: [`text::without_restatements`]
834/// judges a name against the *rest of its own tag*, so the list has to be
835/// gathered before anything is dropped from it. See that function for why a
836/// joint credit sitting beside its members is a duplicate rather than a band.
837fn credited_under(tags: &RawTags, tag: &str) -> Vec<String> {
838    text::without_restatements(split_all(tags.all(tag)))
839}

841/// The only value there is, when there is exactly one.
842fn only_one(values: &[String]) -> Option<&str> {
843    match values {
844        [one] if !one.trim().is_empty() => Some(one.trim()),
845        _ => None,
846    }
847}
848
849fn split_all(values: &[String]) -> Vec<String> {
850    values.iter().flat_map(|v| text::split_artists(v)).collect()
851}

853/// `true` for the placeholder names that stand for "no album artist".
854///
855/// Recording them as artists would pollute every count and every listing.
856fn is_various_artists(name: &str) -> bool {
857    matches!(
858        text::normalize(name).as_str(),
859        "various artists" | "various" | "va" | "artistes divers" | "divers" | "multi interpretes"
860    )
861}

863/// Final pass: track ordering, inferred release dates, collaboration graph.
864fn finalize(catalog: &mut Catalog) {
865    // Order the tracks of each release.
866    let order: Vec<(Id, Vec<Id>)> = catalog
867        .releases
868        .iter()
869        .map(|release| {
870            let mut ids = release.track_ids.clone();
871            ids.sort_by_key(|&id| {
872                let t = &catalog.tracks[id as usize];
873                (
874                    t.disc_no.unwrap_or(1),
875                    t.track_no.unwrap_or(u32::MAX),
876                    t.title.clone(),
877                )
878            });
879            (release.id, ids)
880        })
881        .collect();
882    for (id, ids) in order {
883        catalog.releases[id as usize].track_ids = ids;
884    }
885
886    // Release year: the earliest one announced by its tracks.
887    let mut years: HashMap<Id, (Option<String>, Option<u32>)> = HashMap::new();
888    for track in &catalog.tracks {
889        let Some(rid) = track.release_id else {
890            continue;
891        };
892        let file = &catalog.files[track.file_id as usize];
893        let raw = file
894            .first_tag("originaldate")
895            .or_else(|| file.first_tag("date"))
896            .unwrap_or("");
897        if raw.is_empty() {
898            continue;
899        }
900        let year = text::extract_year(raw);
901        let entry = years.entry(rid).or_insert((None, None));
902        if entry.1.is_none() || (year.is_some() && year < entry.1) {
903            *entry = (Some(raw.to_string()), year);
904        }
905    }
906    for (rid, (date, year)) in years {
907        if let Some(release) = catalog.releases.get_mut(rid as usize) {
908            release.date = date;
909            release.year = year;
910        }
911    }
912
913    rebuild_relations(catalog);
914}

916/// Without a `title` tag, fall back to the file name, stripped of its leading
917/// track number and its extension.
918fn title_from_filename(path: &str) -> String {
919    let name = text::file_name(path);
920    let stem = name.rsplit_once('.').map(|(s, _)| s).unwrap_or(name);
921    let trimmed = stem
922        .trim_start_matches(|c: char| c.is_ascii_digit())
923        .trim_start_matches([' ', '-', '.', '_'])
924        .trim();
925    if trimmed.is_empty() {
926        stem.to_string()
927    } else {
928        trimmed.to_string()
929    }
930}

932/// Track number inferred from a file name that starts with digits.
933fn track_from_filename(path: &str) -> Option<u32> {
934    let name = text::file_name(path);
935    let digits: String = name.chars().take_while(|c| c.is_ascii_digit()).collect();
936    if digits.is_empty() || digits.len() > 3 {
937        return None;
938    }
939    digits.parse().ok().filter(|&n| n > 0)
940}
