7/// Last observed identity evidence for a file named by personal data.
8///
9/// A basename alone never proves a move. Size, stream characteristics and
10/// identifying tags must agree, as must an acoustic fingerprint when one was
11/// available. This is a conservative relocation heuristic, not a byte checksum.
12#[derive(Debug, Clone, PartialEq, Eq)]
13pub struct TrackIdentity {
14    /// Path at which the evidence was observed.
15    pub path: String,
16    /// File size in bytes.
17    pub size: u64,
18    /// Codec declared by the stream.
19    pub codec: String,
20    /// Duration measured from the stream.
21    pub duration_ms: Option<u64>,
22    /// Sampling frequency in hertz.
23    pub sample_rate: Option<u32>,
24    /// Stream precision when meaningful.
25    pub bit_depth: Option<u16>,
26    /// Number of channels.
27    pub channels: Option<u16>,
28    /// Identifying tags, without notes, lyrics or other large text fields.
29    pub metadata: BTreeMap<String, Vec<String>>,
30    /// Acoustic fingerprint, when already known without reading the file.
31    pub fingerprint: Option<String>,
32}
33
34impl TrackIdentity {
35    fn of(file: &AudioFile) -> Self {
36        let metadata = [
37            "title",
38            "artist",
39            "albumartist",
40            "album",
41            "tracknumber",
42            "discnumber",
43            "musicbrainz_recordingid",
44            "musicbrainz_albumid",
45            "isrc",
46        ]
47        .into_iter()
48        .filter_map(|key| {
49            file.tags
50                .get(key)
51                .map(|values| (key.into(), values.clone()))
52        })
53        .collect();
54        Self {
55            path: file.path.clone(),
56            size: file.size,
57            codec: file.properties.codec.clone(),
58            duration_ms: file.properties.duration_ms,
59            sample_rate: file.properties.sample_rate,
60            bit_depth: file.properties.bit_depth,
61            channels: file.properties.channels,
62            metadata,
63            fingerprint: file.fingerprint.as_ref().map(|fp| fp.data.clone()),
64        }
65    }
66
67    fn matches(&self, file: &AudioFile) -> bool {
68        let current = Self::of(file);
69        self.size == current.size
70            && self.codec == current.codec
71            && self.duration_ms == current.duration_ms
72            && self.sample_rate == current.sample_rate
73            && self.bit_depth == current.bit_depth
74            && self.channels == current.channels
75            && self.metadata == current.metadata
76            && self
77                .fingerprint
78                .as_ref()
79                .is_none_or(|fp| current.fingerprint.as_ref() == Some(fp))
80    }
81
82    pub(super) fn to_json(&self) -> Json {
83        let mut row = Json::obj();
84        row.set("path", self.path.as_str().into());
85        row.set("size", self.size.into());
86        row.set("codec", self.codec.as_str().into());
87        if let Some(value) = self.duration_ms {
88            row.set("duration_ms", value.into());
89        }
90        if let Some(value) = self.sample_rate {
91            row.set("sample_rate", value.into());
92        }
93        if let Some(value) = self.bit_depth {
94            row.set("bit_depth", u32::from(value).into());
95        }
96        if let Some(value) = self.channels {
97            row.set("channels", u32::from(value).into());
98        }
99        if let Some(value) = &self.fingerprint {
100            row.set("fingerprint", value.as_str().into());
101        }
102        let mut metadata = Json::obj();
103        for (key, values) in &self.metadata {
104            metadata.set(
105                key,
106                Json::Arr(values.iter().map(|value| value.as_str().into()).collect()),
107            );
108        }
109        row.set("metadata", metadata);
110        row
111    }
112
113    pub(super) fn from_json(row: &Json) -> Option<Self> {
114        let Json::Obj(fields) = row.get("metadata")? else {
115            return None;
116        };
117        let metadata = fields
118            .iter()
119            .map(|(key, values)| {
120                Some((
121                    key.clone(),
122                    values
123                        .as_arr()?
124                        .iter()
125                        .map(Json::as_string)
126                        .collect::<Option<Vec<_>>>()?,
127                ))
128            })
129            .collect::<Option<BTreeMap<_, _>>>()?;
130        Some(Self {
131            path: row.field_str("path")?,
132            size: row.field_u64("size")?,
133            codec: row.field_str("codec")?,
134            duration_ms: row.field_u64("duration_ms"),
135            sample_rate: row.field_u32("sample_rate"),
136            bit_depth: row.field_u32("bit_depth").and_then(|n| n.try_into().ok()),
137            channels: row.field_u32("channels").and_then(|n| n.try_into().ok()),
138            metadata,
139            fingerprint: row.field_str("fingerprint"),
140        })
141    }
142}

144/// Counts personal records attached, waiting, or relocated during reconciliation.
145#[derive(Debug, Clone, Copy, Default, PartialEq)]
146pub struct Attachment {
147    /// Records whose target is in the catalog.
148    pub attached: usize,
149    /// Records whose target is absent and retained unchanged.
150    pub waiting: usize,
151    /// Records whose reference was migrated or whose file moved.
152    pub moved: usize,
153}

155pub(super) fn resolve_release(reference: &EntityRef, catalog: &Catalog) -> Option<Id> {
156    if let Some(release) = catalog.releases.iter().find(|release| {
157        EntityRef::of(catalog, EntityKind::Release, release.id).as_ref() == Some(reference)
158    }) {
159        return Some(release.id);
160    }
161    let (prefix, folder) = reference.key.rsplit_once(RELEASE_KEY_SEPARATOR)?;
162    let legacy_parent = text::disc_folder(text::file_name(folder)).map(|_| text::folder(folder));
163    let mut matches = catalog.releases.iter().filter(|release| {
164        EntityRef::of(catalog, EntityKind::Release, release.id).is_some_and(|current| {
165            let Some((current_prefix, current_folder)) =
166                current.key.rsplit_once(RELEASE_KEY_SEPARATOR)
167            else {
168                return false;
169            };
170            current_prefix == prefix
171                && (current_folder == folder || legacy_parent == Some(current_folder))
172        })
173    });
174    let found = matches.next()?;
175    matches.next().is_none().then_some(found.id)
176}

178/// Resolution for a reconciliation pass, including legacy recording aliases.
179/// Building once keeps a referenced library from being walked for every record.
180struct ReferenceIndex {
181    current: BTreeMap<EntityRef, (Id, usize)>,
182}
183
184impl ReferenceIndex {
185    fn new(catalog: &Catalog, kinds: BTreeSet<EntityKind>) -> Self {
186        let mut current: BTreeMap<EntityRef, (Id, usize)> = BTreeMap::new();
187        for kind in kinds {
188            let ids: Vec<_> = match kind {
189                EntityKind::Track => catalog.tracks.iter().map(|row| row.id).collect(),
190                EntityKind::Release => catalog.releases.iter().map(|row| row.id).collect(),
191                EntityKind::Artist => catalog.artists.iter().map(|row| row.id).collect(),
192                EntityKind::Recording => catalog.recordings.iter().map(|row| row.id).collect(),
193                EntityKind::Work => catalog.works.iter().map(|row| row.id).collect(),
194                EntityKind::ReleaseGroup => {
195                    catalog.release_groups.iter().map(|row| row.id).collect()
196                }
197                EntityKind::Label => catalog.labels.iter().map(|row| row.id).collect(),
198                EntityKind::Genre => catalog.genres.iter().map(|row| row.id).collect(),
199            };
200            for id in ids {
201                if let Some(reference) = EntityRef::of(catalog, kind, id) {
202                    current
203                        .entry(reference)
204                        .and_modify(|value| value.1 += 1)
205                        .or_insert((id, 1));
206                }
207            }
208            if kind == EntityKind::Recording {
209                for recording in &catalog.recordings {
210                    for track in &recording.track_ids {
211                        if let Some(file) = catalog
212                            .track(*track)
213                            .and_then(|track| catalog.file(track.file_id))
214                        {
215                            current
216                                .entry(EntityRef::new(kind, format!("local:{}", file.path)))
217                                .or_insert((recording.id, 1));
218                        }
219                    }
220                }
221            }
222        }
223        Self { current }
224    }
225
226    fn resolve(&self, target: &EntityRef) -> Option<Id> {
227        if let Some((id, _)) = self.current.get(target) {
228            return Some(*id);
229        }
230        if target.kind != EntityKind::Release {
231            return None;
232        }
233        let (prefix, folder) = target.key.rsplit_once(RELEASE_KEY_SEPARATOR)?;
234        text::disc_folder(text::file_name(folder))?;
235        let parent = EntityRef::new(
236            EntityKind::Release,
237            format!("{prefix}{RELEASE_KEY_SEPARATOR}{}", text::folder(folder)),
238        );
239        self.current
240            .get(&parent)
241            .and_then(|(id, count)| (*count == 1).then_some(*id))
242    }
243}

245/// Reconciles personal references without guessing from a filename alone.
246///
247/// Existing references capture optional file evidence. Missing tracks move only
248/// to one matching basename, size, metadata and stream identity, with no personal
249/// data already at the destination for that owner. Legacy references lacking
250/// evidence remain waiting. Album references using a former disc folder migrate
251/// to the release's canonical folder only when their resolution is unambiguous.
252/// No annotation, play or counter is deleted.
253pub fn reconcile(data: &mut UserData, catalog: &Catalog) -> Attachment {
254    let mut report = Attachment::default();
255    let mut files_by_name: BTreeMap<&str, Vec<&AudioFile>> = BTreeMap::new();
256    for file in &catalog.files {
257        files_by_name
258            .entry(file.file_name())
259            .or_default()
260            .push(file);
261    }
262    let identities: BTreeMap<_, _> = data
263        .track_identities
264        .iter()
265        .map(|identity| (identity.path.as_str(), identity))
266        .collect();
267    let existing: BTreeSet<_> = data
268        .annotations
269        .iter()
270        .map(|a| (a.owner.clone(), a.target.clone()))
271        .chain(
272            data.plays
273                .iter()
274                .map(|p| (p.owner.clone(), p.track.clone())),
275        )
276        .chain(
277            data.counts
278                .iter()
279                .map(|c| (c.owner.clone(), c.track.clone())),
280        )
281        .chain(
282            data.scrobbles
283                .iter()
284                .map(|p| (p.owner.clone(), p.track.clone())),
285        )
286        .chain(data.playlists.iter().flat_map(|p| {
287            p.tracks
288                .iter()
289                .map(|track| (p.owner.clone(), track.clone()))
290        }))
291        .chain(data.relation_annotations.iter().flat_map(|a| {
292            [
293                (a.owner.clone(), a.relation.source.clone()),
294                (a.owner.clone(), a.relation.target.clone()),
295            ]
296        }))
297        .collect();
298    let references = ReferenceIndex::new(
299        catalog,
300        existing.iter().map(|(_, target)| target.kind).collect(),
301    );
302    let undone: BTreeSet<_> = data
303        .relinks
304        .iter()
305        .filter(|event| event.undone_at.is_some())
306        .map(|event| (&event.owner, &event.from))
307        .collect();
308    let mut moves = BTreeMap::new();
309    for (owner, target) in &existing {
310        let next = if let Some(id) = references.resolve(target) {
311            EntityRef::of(catalog, target.kind, id)
312        } else if target.kind == EntityKind::Track {
313            if undone.contains(&(owner, target)) {
314                continue;
315            }
316            identities.get(target.key.as_str()).and_then(|identity| {
317                let mut matches = files_by_name
318                    .get(text::file_name(&target.key))?
319                    .iter()
320                    .filter(|file| identity.matches(file));
321                let candidate = matches.next()?;
322                matches
323                    .next()
324                    .is_none()
325                    .then(|| EntityRef::new(EntityKind::Track, &candidate.path))
326            })
327        } else {
328            None
329        };
330        if let Some(next) = next
331            && next != *target
332            && !existing.contains(&(owner.clone(), next.clone()))
333        {
334            moves.insert((owner.clone(), target.clone()), next);
335        }
336    }
337    // Two missing references converging on one file are just as ambiguous as
338    // two candidate files. Do not manufacture duplicate owner/target records.
339    let mut destinations = BTreeMap::new();
340    for ((owner, _), to) in &moves {
341        *destinations
342            .entry((owner.clone(), to.clone()))
343            .or_insert(0usize) += 1;
344    }
345    moves.retain(|(owner, _), to| destinations.get(&(owner.clone(), to.clone())) == Some(&1));
346    for (owner, target) in data
347        .annotations
348        .iter_mut()
349        .map(|a| (&a.owner, &mut a.target))
350        .chain(data.plays.iter_mut().map(|p| (&p.owner, &mut p.track)))
351        .chain(data.counts.iter_mut().map(|c| (&c.owner, &mut c.track)))
352        .chain(data.scrobbles.iter_mut().map(|p| (&p.owner, &mut p.track)))
353        .chain(
354            data.playlists
355                .iter_mut()
356                .flat_map(|p| p.tracks.iter_mut().map(|track| (&p.owner, track))),
357        )
358    {
359        if let Some(next) = moves.get(&(owner.clone(), target.clone())) {
360            *target = next.clone();
361            report.moved += 1;
362        }
363        if references.resolve(target).is_some() {
364            report.attached += 1;
365        } else {
366            report.waiting += 1;
367        }
368    }
369    for annotation in &mut data.relation_annotations {
370        for endpoint in [
371            &mut annotation.relation.source,
372            &mut annotation.relation.target,
373        ] {
374            if let Some(next) = moves.get(&(annotation.owner.clone(), endpoint.clone())) {
375                *endpoint = next.clone();
376            }
377        }
378    }
379    let paths: BTreeSet<_> = data
380        .annotations
381        .iter()
382        .map(|a| &a.target)
383        .chain(data.plays.iter().map(|p| &p.track))
384        .chain(data.counts.iter().map(|c| &c.track))
385        .chain(data.scrobbles.iter().map(|p| &p.track))
386        .chain(data.playlists.iter().flat_map(|p| &p.tracks))
387        .chain(
388            data.relation_annotations
389                .iter()
390                .flat_map(|a| [&a.relation.source, &a.relation.target]),
391        )
392        .filter(|target| target.kind == EntityKind::Track)
393        .map(|target| target.key.as_str())
394        .collect();
395    let mut identities: BTreeMap<_, _> = std::mem::take(&mut data.track_identities)
396        .into_iter()
397        .map(|identity| (identity.path.clone(), identity))
398        .collect();
399    for file in catalog
400        .files
401        .iter()
402        .filter(|file| paths.contains(file.path.as_str()))
403    {
404        identities.insert(file.path.clone(), TrackIdentity::of(file));
405    }
406    let relink_paths: BTreeSet<_> = data
407        .relinks
408        .iter()
409        .filter(|event| event.from.kind == EntityKind::Track)
410        .flat_map(|event| [&event.from.key, &event.to.key])
411        .collect();
412    data.track_identities = identities
413        .into_values()
414        .filter(|identity| {
415            paths.contains(identity.path.as_str()) || relink_paths.contains(&identity.path)
416        })
417        .collect();
418    report
419}
