1//! Destination playlists retain only selected audio and use its planned paths.
2
3use std::collections::{BTreeMap, BTreeSet};
4use std::path::{Path, PathBuf};
5
6use super::{Item, ItemKind, Plan};
7use crate::model::{Catalog, Id};
8
9impl Plan {
10    /// Rewrites selected M3U/M3U8 companions and adds a selection M3U8.
11    ///
12    /// Source playlists are read as UTF-8 and left untouched. Entries not in
13    /// this plan (including remote URLs) are omitted; the returned count makes
14    /// that narrowing visible. Paths refer to adapted names and converted
15    /// extensions, relative to each destination playlist. The generated
16    /// selection follows the caller's track order.
17    pub fn prepare_playlists(&mut self, catalog: &Catalog, tracks: &[Id]) -> Result<usize, String> {
18        let mapping: BTreeMap<String, PathBuf> = self
19            .items
20            .iter()
21            .filter(|item| item.kind == ItemKind::Audio)
22            .map(|item| {
23                (
24                    path_key(&item.source.to_string_lossy()),
25                    item.relative.clone(),
26                )
27            })
28            .collect();
29        let mut omitted = 0;
30        for item in &mut self.items {
31            if item.kind == ItemKind::Audio || !is_playlist(&item.source) {
32                continue;
33            }
34            let text = std::fs::read_to_string(&item.source).map_err(|error| {
35                format!(
36                    "{}: cannot read playlist as UTF-8: {error}",
37                    item.source.display()
38                )
39            })?;
40            let (rendered, removed) = remap(&text, Some(&item.source), &item.relative, &mapping)?;
41            omitted += removed;
42            item.size = rendered.len() as u64;
43            item.contents = Some(rendered);
44        }
45        let mut root_names: BTreeSet<String> = self
46            .items
47            .iter()
48            .filter_map(|item| item.relative.components().next())
49            .map(|component| super::names::portable_key(&component.as_os_str().to_string_lossy()))
50            .collect();
51        let relative = PathBuf::from(super::names::make_unique_portable(
52            "aede-selection.m3u8",
53            &mut root_names,
54        ));
55        let mut paths = BTreeMap::new();
56        for &id in tracks {
57            let Some(file) = catalog
58                .track(id)
59                .and_then(|track| catalog.file(track.file_id))
60            else {
61                continue;
62            };
63            let Some(path) = mapping.get(&path_key(&file.path)) else {
64                continue;
65            };
66            let path = relative_path(Path::new(""), path);
67            if path.contains(['\n', '\r']) {
68                return Err("selected output path cannot be represented in M3U".into());
69            }
70            let path = if path.starts_with('#') {
71                format!("./{path}")
72            } else {
73                path
74            };
75            paths.insert(id, path);
76        }
77        let contents = crate::playlist::render_with_paths(
78            catalog,
79            tracks,
80            crate::playlist::Style::Extended,
81            |id| paths.get(&id).cloned(),
82        );
83        self.items.push(Item {
84            source: PathBuf::from("selected tracks"),
85            relative,
86            size: contents.len() as u64,
87            kind: ItemKind::Other,
88            convert: None,
89            contents: Some(contents),
90        });
91        Ok(omitted)
92    }
93}

95fn is_playlist(path: &Path) -> bool {
96    path.extension().is_some_and(|extension| {
97        let extension = extension.to_string_lossy();
98        extension.eq_ignore_ascii_case("m3u") || extension.eq_ignore_ascii_case("m3u8")
99    })
100}

102fn remap(
103    text: &str,
104    source: Option<&Path>,
105    output: &Path,
106    mapping: &BTreeMap<String, PathBuf>,
107) -> Result<(String, usize), String> {
108    let mut rendered = String::new();
109    let mut pending = Vec::new();
110    let mut omitted = 0;
111    for line in text.trim_start_matches('\u{feff}').lines() {
112        let line = line.trim_end_matches('\r');
113        if line.is_empty() {
114            continue;
115        }
116        if line == "#EXTM3U" {
117            rendered.push_str("#EXTM3U\n");
118            continue;
119        }
120        if line.starts_with('#') {
121            pending.push(line);
122            continue;
123        }
124        let key = match source {
125            Some(source) if !is_absolute(line) => path_key(&format!(
126                "{}/{}",
127                crate::text::folder(&source.to_string_lossy()),
128                line
129            )),
130            _ => path_key(line),
131        };
132        let Some(target) = mapping.get(&key) else {
133            pending.clear();
134            omitted += 1;
135            continue;
136        };
137        for comment in pending.drain(..) {
138            rendered.push_str(comment);
139            rendered.push('\n');
140        }
141        let path = relative_path(output.parent().unwrap_or_else(|| Path::new("")), target);
142        if path.contains(['\n', '\r']) {
143            return Err(format!(
144                "{}: output path cannot be represented in M3U",
145                target.display()
146            ));
147        }
148        if path.starts_with('#') {
149            rendered.push_str("./");
150        }
151        rendered.push_str(&path);
152        rendered.push('\n');
153    }
154    Ok((rendered, omitted))
155}

157fn is_absolute(path: &str) -> bool {
158    path.starts_with('/')
159        || path.starts_with("\\\\")
160        || (path.as_bytes().get(1) == Some(&b':')
161            && path
162                .as_bytes()
163                .get(2)
164                .is_some_and(|separator| matches!(separator, b'/' | b'\\')))
165}

167/// Lexical normalization also supports Windows source playlists on Unix test
168/// hosts. Backslashes are separators only for a Windows absolute spelling.
169fn path_key(path: &str) -> String {
170    let windows = path.starts_with("\\\\")
171        || (path.as_bytes().get(1) == Some(&b':')
172            && path
173                .as_bytes()
174                .get(2)
175                .is_some_and(|separator| matches!(separator, b'/' | b'\\')));
176    let normalized = if windows {
177        path.replace('\\', "/")
178    } else {
179        path.to_string()
180    };
181    let mut parts = Vec::new();
182    for part in normalized.split('/') {
183        match part {
184            "" | "." => {}
185            ".." => {
186                parts.pop();
187            }
188            _ => parts.push(part),
189        }
190    }
191    let key = format!(
192        "{}{}",
193        if normalized.starts_with('/') { "/" } else { "" },
194        parts.join("/")
195    );
196    if windows { key.to_lowercase() } else { key }
197}

199fn relative_path(base: &Path, target: &Path) -> String {
200    let base: Vec<_> = base.components().collect();
201    let target: Vec<_> = target.components().collect();
202    let common = base
203        .iter()
204        .zip(&target)
205        .take_while(|(left, right)| left == right)
206        .count();
207    let mut out = vec!["..".to_string(); base.len() - common];
208    out.extend(
209        target[common..]
210            .iter()
211            .map(|component| component.as_os_str().to_string_lossy().into_owned()),
212    );
213    out.join("/")
214}
