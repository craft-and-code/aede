1//! Converting audio on the way out, by driving ffmpeg.
2//!
3//! **ffmpeg is an external program, not a dependency.** It is looked for, run,
4//! and its absence reported in a sentence rather than as a mysterious failure.
5//! The rule in `CLAUDE.md` about adding crates is not being bent here: nothing
6//! is linked, nothing is vendored, and a build without ffmpeg installed is a
7//! build where every other command still works.
8//!
9//! This is the first thing in the project that **writes audio**, and the line is
10//! worth guarding: it writes new files, under a folder the user named, and it
11//! never opens a source file for writing. Writing tags into a file produced here
12//! is not the tag-rewriting the project refuses — that refusal protects the
13//! user's own files, whose modification date, integrity verdict and scan state
14//! all depend on not being touched. A file that did not exist a second ago has
15//! none of those.
16
17use std::collections::BTreeMap;
18use std::path::Path;
19use std::process::Command;

21/// A format a copy can be converted into.
22///
23/// Deliberately short. Every entry here is a format a portable player is likely
24/// to want, and each one costs a decoder somebody has to have: adding a target
25/// nobody asked for is a promise to keep it working.
26#[derive(Debug, Clone, Copy, PartialEq, Eq)]
27pub enum Target {
28    /// The one every device plays.
29    Mp3,
30    /// Best quality per byte at low bitrates, and what a modern player wants.
31    Opus,
32    /// What Apple devices prefer, in an MP4 container.
33    Aac,
34    /// The free lossy format Ogg carries.
35    Vorbis,
36    /// Lossless, and much smaller than WAV: the reason to convert a WAV rip.
37    Flac,
38    /// Uncompressed, for a device that plays nothing else.
39    Wav,
40}
41
42/// One row per target: how it is typed, what it produces, and how it is asked
43/// for on the command line.
44///
45/// A table rather than a `match` in four places, for the reason the role
46/// vocabulary is one: a format added in one place and forgotten in another is
47/// a target that half-works.
48const TARGETS: &[(&str, Target, &str, &str, bool)] = &[
49    // typed, target, extension, ffmpeg codec, lossless
50    ("mp3", Target::Mp3, "mp3", "libmp3lame", false),
51    ("opus", Target::Opus, "opus", "libopus", false),
52    ("aac", Target::Aac, "m4a", "aac", false),
53    ("m4a", Target::Aac, "m4a", "aac", false),
54    ("vorbis", Target::Vorbis, "ogg", "libvorbis", false),
55    ("ogg", Target::Vorbis, "ogg", "libvorbis", false),
56    ("flac", Target::Flac, "flac", "flac", true),
57    ("wav", Target::Wav, "wav", "pcm", true),
58];
59
60impl Target {
61    /// The target a typed word names, or `None` for a word that names none.
62    pub fn parse(word: &str) -> Option<Target> {
63        let wanted = crate::text::normalize(word);
64        TARGETS
65            .iter()
66            .find(|(name, ..)| *name == wanted)
67            .map(|(_, target, ..)| *target)
68    }
69
70    /// Every spelling accepted, for a message that offers what it refuses.
71    pub fn names() -> String {
72        let mut names: Vec<&str> = TARGETS.iter().map(|(name, ..)| *name).collect();
73        names.dedup();
74        names.join(", ")
75    }
76
77    /// The targets that have a quality setting at all.
78    ///
79    /// Read from the same table as everything else, so a format added tomorrow
80    /// appears here without anybody remembering to add it — which is the whole
81    /// reason the table exists.
82    pub fn lossy_names() -> String {
83        TARGETS
84            .iter()
85            .filter(|(_, _, _, _, lossless)| !lossless)
86            .map(|(name, ..)| *name)
87            .collect::<Vec<_>>()
88            .join(", ")
89    }
90
91    /// Accepted quality settings for this encoder, for actionable refusals.
92    pub fn quality_forms(self) -> &'static str {
93        match self {
94            Target::Mp3 => "V0…V9 or a supported MP3 bitrate (8…320k, e.g. 192k)",
95            Target::Vorbis => "q0…q10 or a bitrate (32…500k, e.g. 192k)",
96            Target::Opus => "a bitrate (8…510k, e.g. 128k)",
97            Target::Aac => "a bitrate (8…2048k, e.g. 192k)",
98            Target::Flac | Target::Wav => "no quality setting",
99        }
100    }
101
102    fn row(self) -> (&'static str, &'static str, bool) {
103        TARGETS
104            .iter()
105            .find(|(_, target, ..)| *target == self)
106            .map(|(_, _, extension, codec, lossless)| (*extension, *codec, *lossless))
107            .unwrap_or(("mp3", "libmp3lame", false))
108    }
109
110    /// The file extension a converted file carries, without its dot.
111    pub fn extension(self) -> &'static str {
112        self.row().0
113    }
114
115    /// `true` when the target keeps every sample it was given.
116    pub fn lossless(self) -> bool {
117        self.row().2
118    }
119
120    /// Whether a cover embedded in the source can travel into this target at
121    /// all. `false` here is not this code declining to carry it — ffmpeg
122    /// itself refuses to mux a picture stream into `wav`, `opus` or `vorbis`
123    /// (verified directly: mapping a video stream into any of the three fails
124    /// outright and produces no file), so the loss happens at the encoder
125    /// whatever is asked of it. [`convert`] reads this same rule, once.
126    pub fn keeps_embedded_art(self) -> bool {
127        matches!(self, Target::Mp3 | Target::Aac | Target::Flac)
128    }
129
130    /// Canonical tag keys known to survive a conversion into this target, or
131    /// `None` when every key does.
132    ///
133    /// Only `wav` has a real answer here. ffmpeg's WAV muxer writes the
134    /// legacy RIFF `LIST`/`INFO` chunk, and that chunk's vocabulary is exactly
135    /// this short — verified against a real encode carrying ten canonical
136    /// tags, of which precisely these six arrived and no others. Every other
137    /// target's tag format takes an arbitrary key, so there is nothing here
138    /// worth naming.
139    fn known_tag_vocabulary(self) -> Option<&'static [&'static str]> {
140        match self {
141            Target::Wav => Some(&["title", "artist", "album", "genre", "date", "tracknumber"]),
142            _ => None,
143        }
144    }
145
146    /// Canonical tag keys `present` holds that this target is known to drop.
147    ///
148    /// Empty for every target but `wav`, and empty for `wav` too when nothing
149    /// present falls outside its vocabulary — a file tagged with only title,
150    /// artist and album loses nothing by becoming a WAV.
151    pub fn tags_it_would_drop(self, present: &BTreeMap<String, Vec<String>>) -> Vec<String> {
152        let Some(kept) = self.known_tag_vocabulary() else {
153            return Vec::new();
154        };
155        present
156            .keys()
157            .filter(|key| !kept.contains(&key.as_str()))
158            .cloned()
159            .collect()
160    }
161
162    /// Typical bitrate in kbps, used only to estimate a size before the work is
163    /// done. `None` for a lossless target, where the figure would be a fiction.
164    fn nominal_kbps(self, quality: Option<Quality>) -> Option<u32> {
165        if self.lossless() {
166            return None;
167        }
168        if let Some(Quality::Bitrate(kbps)) = quality {
169            return Some(kbps);
170        }
171        Some(match self {
172            // LAME's VBR scale is only a planning estimate, never a fixed rate.
173            Target::Mp3 => match quality {
174                Some(Quality::Variable(level)) => [245, 225, 190, 175, 165, 130, 115, 100, 85, 65]
175                    .get(level as usize)
176                    .copied()
177                    .unwrap_or(245),
178                _ => 245,
179            },
180            Target::Opus => 128,
181            Target::Aac => 192,
182            Target::Vorbis => 192,
183            _ => 192,
184        })
185    }
186}

188/// How hard the encoder should try, as the user asked for it.
189#[derive(Debug, Clone, Copy, PartialEq, Eq)]
190pub enum Quality {
191    /// A variable-quality setting, `V0`–`V9` for MP3 or `q0`–`q10` for Vorbis:
192    /// lower is better for MP3, higher is better for Vorbis, which is the
193    /// encoders' own convention and not something to paper over.
194    Variable(u32),
195    /// A constant bitrate in kbps, written `192k`.
196    Bitrate(u32),
197}
198
199impl Quality {
200    /// Reads `V0`, `q6` or `192k`, and nothing else.
201    ///
202    /// A value that parses as none of the three is refused rather than quietly
203    /// replaced by the default: an encoder silently running at the wrong
204    /// setting produces files that are wrong in a way nobody notices until the
205    /// card is full.
206    pub fn parse(word: &str) -> Option<Quality> {
207        let word = word.trim();
208        if let Some(digits) = word
209            .strip_prefix(['V', 'v', 'Q', 'q'])
210            .and_then(|rest| rest.parse::<u32>().ok())
211            && digits <= 10
212        {
213            return Some(Quality::Variable(digits));
214        }
215        let bitrate = word.strip_suffix(['k', 'K']).unwrap_or(word);
216        bitrate
217            .parse::<u32>()
218            .ok()
219            .filter(|kbps| (8..=2048).contains(kbps))
220            .map(Quality::Bitrate)
221    }
222
223    /// Reads and validates a quality setting for one encoder.
224    ///
225    /// MP3 accepts only its V scale and supported constant bitrates; Vorbis
226    /// accepts its q scale or a bitrate. Opus and AAC have bitrate settings.
227    pub fn parse_for(target: Target, word: &str) -> Option<Quality> {
228        let word = word.trim();
229        let quality = Self::parse(word)?;
230        let valid = match quality {
231            Quality::Variable(level) => match target {
232                Target::Mp3 => word.starts_with(['V', 'v']) && level <= 9,
233                Target::Vorbis => word.starts_with(['Q', 'q']) && level <= 10,
234                _ => false,
235            },
236            Quality::Bitrate(kbps) => match target {
237                Target::Mp3 => [
238                    8, 16, 24, 32, 40, 48, 56, 64, 80, 96, 112, 128, 144, 160, 192, 224, 256, 320,
239                ]
240                .contains(&kbps),
241                Target::Opus => (8..=510).contains(&kbps),
242                Target::Aac => (8..=2048).contains(&kbps),
243                Target::Vorbis => (32..=500).contains(&kbps),
244                _ => false,
245            },
246        };
247        valid.then_some(quality)
248    }
249
250    /// How the setting is written, for a message that repeats what was typed.
251    pub const FORMS: &'static str = "V0…V9 for MP3, q0…q10 for Vorbis, or a bitrate like 192k";
252}

254/// What ffmpeg should be told about quality, given the target.
255fn quality_arguments(target: Target, quality: Option<Quality>) -> Vec<String> {
256    match (target, quality) {
257        // A lossless target has no quality knob worth exposing: FLAC's
258        // compression level trades encoding time for a few percent, and PCM
259        // has none at all.
260        (t, _) if t.lossless() => Vec::new(),
261        (_, Some(Quality::Bitrate(kbps))) => vec!["-b:a".into(), format!("{kbps}k")],
262        (_, Some(Quality::Variable(level))) => vec!["-q:a".into(), level.to_string()],
263        // The defaults are the ones each encoder is actually good at, not a
264        // single number applied to all of them.
265        (Target::Mp3, None) => vec!["-q:a".into(), "0".into()],
266        (Target::Vorbis, None) => vec!["-q:a".into(), "6".into()],
267        (Target::Opus, None) => vec!["-b:a".into(), "128k".into()],
268        (Target::Aac, None) => vec!["-b:a".into(), "192k".into()],
269        (_, None) => Vec::new(),
270    }
271}

273/// Roughly what a converted file will weigh, for a report given before the work.
274///
275/// An estimate, and said to be one wherever it is shown. The alternative is to
276/// answer "unknown" to "will this fit on my card", which is the one question
277/// somebody filling a card is asking.
278pub fn estimated_size(
279    target: Target,
280    quality: Option<Quality>,
281    duration_ms: u64,
282    source: u64,
283) -> u64 {
284    match target.nominal_kbps(quality) {
285        // kbps is kilobits per second, so a millisecond holds kbps/8 bytes:
286        // ms × kbps ÷ 8. Dividing by 8000 instead — seconds and bits confused
287        // in one step — made every estimate a thousand times too small, and a
288        // card that could "obviously" hold the whole library. A test caught it.
289        Some(kbps) => duration_ms.saturating_mul(kbps as u64) / 8,
290        // Lossless from lossless: FLAC lands around 60 % of PCM on real music,
291        // and PCM from FLAC around the inverse.
292        None if target == Target::Flac => source.saturating_mul(3) / 5,
293        None => source.saturating_mul(2),
294    }
295}

313/// Refuses lossless conversions that cannot represent the source samples.
314///
315/// FLAC stores integer PCM; arbitrary floating-point PCM must remain in a
316/// floating-capable container such as WAV rather than being quantized silently.
317pub fn validate_conversion(
318    properties: &crate::tags::AudioProperties,
319    target: Target,
320) -> Result<(), String> {
321    if target == Target::Flac && matches!(properties.codec.as_str(), "pcm_float" | "fl32" | "fl64")
322    {
323        return Err("FLAC cannot preserve arbitrary floating-point PCM; use WAV or copy the source unchanged".into());
324    }
325    Ok(())
326}

328/// Converts one file, and reports what went wrong in words rather than in an
329/// exit status.
330///
331/// The metadata travels: `-map_metadata 0` carries the tags across, and the
332/// embedded cover is copied as a picture stream where the container takes one.
333/// Neither is perfect — no two tag formats hold quite the same fields, and a
334/// Vorbis comment with three values for one field cannot be an ID3 frame that
335/// holds one — but losing the artist and the title on the way to a player is
336/// not a trade anybody would accept, and ffmpeg's mapping is what beets relies
337/// on for the same reason.
338///
339/// Sources must resolve to ordinary local files. Output is encoded in an
340/// isolated temporary directory, then atomically replaces an ordinary target;
341/// final symlinks and the source itself are refused. A different hard-link
342/// alias is replaced without writing through to the original source inode.
343pub fn convert(
344    ffmpeg: &str,
345    source: &Path,
346    destination: &Path,
347    target: Target,
348    quality: Option<Quality>,
349) -> Result<(), String> {
350    if let Some(quality) = quality {
351        let valid = match quality {
352            Quality::Variable(level) => {
353                matches!(target, Target::Mp3) && level <= 9
354                    || matches!(target, Target::Vorbis) && level <= 10
355            }
356            Quality::Bitrate(kbps) => Quality::parse_for(target, &format!("{kbps}k")).is_some(),
357        };
358        if !valid {
359            return Err("quality is not supported by the selected encoder".into());
360        }
361    }
362    super::validate_source(source)?;
363    // Absolute input prevents a literal filename such as `pipe:123` from
364    // becoming an ffmpeg protocol, and keeps local conversion local.
365    let source = source.canonicalize().map_err(|error| error.to_string())?;
366    if destination.canonicalize().is_ok_and(|path| path == source) {
367        return Err("source and destination are the same file".into());
368    }
369    let temporary = super::TemporaryOutput::new(destination)?;
370    encode(ffmpeg, &source, temporary.path(), target, quality)?;
371    temporary.publish()
372}

374fn encode(
375    ffmpeg: &str,
376    source: &Path,
377    destination: &Path,
378    target: Target,
379    quality: Option<Quality>,
380) -> Result<(), String> {
381    let high_precision_flac = if target == Target::Flac {
382        let properties = crate::tags::read(source)
383            .map_err(|error| format!("cannot determine source PCM representation: {error}"))?
384            .properties;
385        validate_conversion(&properties, target)?;
386        properties.bit_depth.is_some_and(|bits| bits > 24)
387    } else {
388        false
389    };
390    let (_, default_codec, _) = target.row();
391    let codec = if target == Target::Wav {
392        wav_codec(source)?
393    } else {
394        default_codec
395    };
396    let mut command = Command::new(ffmpeg);
397    command
398        // Without this a run in a terminal can stop dead waiting for an answer
399        // to a question nobody saw, in the middle of nine hundred files.
400        .arg("-nostdin")
401        .args(["-loglevel", "error"])
402        .arg("-y")
403        .arg("-i")
404        .arg(source)
405        .args(["-map", "0:a"]);
406
407    // The cover, where the container holds one. `?` makes the stream optional,
408    // so a file without art is converted rather than refused.
409    if target.keeps_embedded_art() {
410        command.args([
411            "-map",
412            "0:v?",
413            "-c:v",
414            "copy",
415            "-disposition:v:0",
416            "attached_pic",
417        ]);
418    }
419    if high_precision_flac {
420        // FFmpeg otherwise quantizes s32 input to 24 bits. Versions with
421        // 32-bit FLAC support require this explicit experimental opt-in;
422        // older versions must fail instead of publishing lower precision.
423        command.args([
424            "-sample_fmt",
425            "s32",
426            "-strict",
427            "experimental",
428            "-bits_per_raw_sample",
429            "32",
430        ]);
431    }
432    command
433        .args(["-map_metadata", "0"])
434        .args(["-c:a", codec])
435        .args(quality_arguments(target, quality))
436        .arg(destination);
437
438    let output = command
439        .output()
440        .map_err(|e| format!("ffmpeg could not be run: {e}"))?;
441    if output.status.success() {
442        if target.lossless() {
443            let source = crate::tags::read(source)
444                .map_err(|error| format!("cannot read source precision: {error}"))?
445                .properties;
446            let written = crate::tags::read(destination)
447                .map_err(|error| format!("cannot read encoded precision: {error}"))?
448                .properties;
449            verify_lossless_properties(&source, &written)?;
450        }
451        return Ok(());
452    }
453    // ffmpeg's own words, trimmed to the last few lines: the first ones are
454    // usually the banner, and the reason is at the end.
455    let said = String::from_utf8_lossy(&output.stderr);
456    let reason: Vec<&str> = said.lines().filter(|l| !l.trim().is_empty()).collect();
457    let tail = reason
458        .iter()
459        .rev()
460        .take(2)
461        .rev()
462        .copied()
463        .collect::<Vec<_>>()
464        .join("; ");
465    Err(match tail.is_empty() {
466        true => "ffmpeg failed without saying why".to_string(),
467        false => format!("ffmpeg: {tail}"),
468    })
469}

471/// A successful encoder must still retain the source's PCM representation.
472fn verify_lossless_properties(
473    source: &crate::tags::AudioProperties,
474    written: &crate::tags::AudioProperties,
475) -> Result<(), String> {
476    if !written.lossless
477        || source
478            .bit_depth
479            .is_some_and(|bits| written.bit_depth.is_none_or(|output| output < bits))
480        || source
481            .sample_rate
482            .is_some_and(|rate| written.sample_rate != Some(rate))
483        || source
484            .channels
485            .is_some_and(|channels| written.channels != Some(channels))
486    {
487        return Err(
488            "lossless encoder cannot preserve the source precision, sample rate or channel count"
489                .into(),
490        );
491    }
492    if matches!(source.codec.as_str(), "pcm_float" | "fl32" | "fl64")
493        && !matches!(written.codec.as_str(), "pcm_float" | "fl32" | "fl64")
494    {
495        return Err("lossless encoder quantized floating-point PCM to integer samples".into());
496    }
497    Ok(())
498}

500/// Uses enough PCM precision for the source rather than forcing CD precision.
501/// Unsupported/unknown precision is refused instead of silently quantized.
502fn wav_codec(source: &Path) -> Result<&'static str, String> {
503    let properties = crate::tags::read(source)
504        .map_err(|error| format!("cannot determine source PCM precision: {error}"))?
505        .properties;
506    if matches!(properties.codec.as_str(), "pcm_float" | "fl32" | "fl64") {
507        return match properties.bit_depth {
508            Some(32) => Ok("pcm_f32le"),
509            Some(64) => Ok("pcm_f64le"),
510            _ => Err("cannot preserve unsupported floating-point WAV precision".into()),
511        };
512    }
513    match properties.bit_depth {
514        Some(1..=16) => Ok("pcm_s16le"),
515        Some(17..=24) => Ok("pcm_s24le"),
516        Some(25..=32) => Ok("pcm_s32le"),
517        _ => Err("cannot preserve unknown or unsupported source PCM precision in WAV".into()),
518    }
519}

521/// Longest a converted file's playing time may differ from its source.
522///
523/// Encoders pad and trim: an MP3 frame is 1152 samples and the encoder delay is
524/// real, so an exact match would fail on correct output. Two seconds is far
525/// larger than any of that and far smaller than a truncated encode.
526const DURATION_TOLERANCE_MS: u64 = 2_000;
527
528/// Checks a converted file the only way a converted file can be checked.
529///
530/// Comparing checksums is meaningless here: the bytes are different by
531/// construction, which is the whole point. What can be checked is that the
532/// result is a file this program can read, that it holds audio, and that the
533/// audio lasts as long as what went in — which is what catches the failure that
534/// actually happens, an encode cut short by a full disk or a killed process.
535///
536/// It reads the output with Aède's own parsers rather than asking ffmpeg again:
537/// a verification that trusts the tool being verified is not one.
538pub fn verify(destination: &Path, expected_ms: Option<u64>) -> Result<(), String> {
539    let tags = crate::tags::read(destination)
540        .map_err(|e| format!("what was written cannot be read back: {e}"))?;
541    let Some(written) = tags.properties.duration_ms else {
542        return Err("what was written has no readable duration".into());
543    };
544    if written == 0 {
545        return Err("what was written holds no audio".into());
546    }
547    let Some(expected) = expected_ms.filter(|ms| *ms > 0) else {
548        return Ok(());
549    };
550    if written.abs_diff(expected) > DURATION_TOLERANCE_MS {
551        return Err(format!(
552            "the result lasts {} where the source lasts {}: the encode was cut short",
553            crate::text::format_duration(written),
554            crate::text::format_duration(expected)
555        ));
556    }
557    Ok(())
558}
