1//! Converting audio on the way out, by driving ffmpeg.
2//!
3//! **ffmpeg is an external program, not a dependency.** It is looked for, run,
4//! and its absence reported in a sentence rather than as a mysterious failure.
5//! The rule in `CLAUDE.md` about adding crates is not being bent here: nothing
6//! is linked, nothing is vendored, and a build without ffmpeg installed is a
7//! build where every other command still works.
8//!
9//! This is the first thing in the project that **writes audio**, and the line is
10//! worth guarding: it writes new files, under a folder the user named, and it
11//! never opens a source file for writing. Writing tags into a file produced here
12//! is not the tag-rewriting the project refuses — that refusal protects the
13//! user's own files, whose modification date, integrity verdict and scan state
14//! all depend on not being touched. A file that did not exist a second ago has
15//! none of those.
16
17use std::collections::BTreeMap;
18use std::path::Path;
19use std::process::Command;

21/// A format a copy can be converted into.
22///
23/// Deliberately short. Every entry here is a format a portable player is likely
24/// to want, and each one costs a decoder somebody has to have: adding a target
25/// nobody asked for is a promise to keep it working.
26#[derive(Debug, Clone, Copy, PartialEq, Eq)]
27pub enum Target {
28    /// The one every device plays.
29    Mp3,
30    /// Best quality per byte at low bitrates, and what a modern player wants.
31    Opus,
32    /// What Apple devices prefer, in an MP4 container.
33    Aac,
34    /// The free lossy format Ogg carries.
35    Vorbis,
36    /// Lossless, and much smaller than WAV: the reason to convert a WAV rip.
37    Flac,
38    /// Uncompressed, for a device that plays nothing else.
39    Wav,
40}
41
42/// One row per target: how it is typed, what it produces, and how it is asked
43/// for on the command line.
44///
45/// A table rather than a `match` in four places, for the reason the role
46/// vocabulary is one: a format added in one place and forgotten in another is
47/// a target that half-works.
48const TARGETS: &[(&str, Target, &str, &str, bool)] = &[
49    // typed, target, extension, ffmpeg codec, lossless
50    ("mp3", Target::Mp3, "mp3", "libmp3lame", false),
51    ("opus", Target::Opus, "opus", "libopus", false),
52    ("aac", Target::Aac, "m4a", "aac", false),
53    ("m4a", Target::Aac, "m4a", "aac", false),
54    ("vorbis", Target::Vorbis, "ogg", "libvorbis", false),
55    ("ogg", Target::Vorbis, "ogg", "libvorbis", false),
56    ("flac", Target::Flac, "flac", "flac", true),
57    ("wav", Target::Wav, "wav", "pcm_s16le", true),
58];
59
60impl Target {
61    /// The target a typed word names, or `None` for a word that names none.
62    pub fn parse(word: &str) -> Option<Target> {
63        let wanted = crate::text::normalize(word);
64        TARGETS
65            .iter()
66            .find(|(name, ..)| *name == wanted)
67            .map(|(_, target, ..)| *target)
68    }
69
70    /// Every spelling accepted, for a message that offers what it refuses.
71    pub fn names() -> String {
72        let mut names: Vec<&str> = TARGETS.iter().map(|(name, ..)| *name).collect();
73        names.dedup();
74        names.join(", ")
75    }
76
77    /// The targets that have a quality setting at all.
78    ///
79    /// Read from the same table as everything else, so a format added tomorrow
80    /// appears here without anybody remembering to add it — which is the whole
81    /// reason the table exists.
82    pub fn lossy_names() -> String {
83        TARGETS
84            .iter()
85            .filter(|(_, _, _, _, lossless)| !lossless)
86            .map(|(name, ..)| *name)
87            .collect::<Vec<_>>()
88            .join(", ")
89    }
90
91    fn row(self) -> (&'static str, &'static str, bool) {
92        TARGETS
93            .iter()
94            .find(|(_, target, ..)| *target == self)
95            .map(|(_, _, extension, codec, lossless)| (*extension, *codec, *lossless))
96            .unwrap_or(("mp3", "libmp3lame", false))
97    }
98
99    /// The file extension a converted file carries, without its dot.
100    pub fn extension(self) -> &'static str {
101        self.row().0
102    }
103
104    /// `true` when the target keeps every sample it was given.
105    pub fn lossless(self) -> bool {
106        self.row().2
107    }
108
109    /// Whether a cover embedded in the source can travel into this target at
110    /// all. `false` here is not this code declining to carry it — ffmpeg
111    /// itself refuses to mux a picture stream into `wav`, `opus` or `vorbis`
112    /// (verified directly: mapping a video stream into any of the three fails
113    /// outright and produces no file), so the loss happens at the encoder
114    /// whatever is asked of it. [`convert`] reads this same rule, once.
115    pub fn keeps_embedded_art(self) -> bool {
116        matches!(self, Target::Mp3 | Target::Aac | Target::Flac)
117    }
118
119    /// Canonical tag keys known to survive a conversion into this target, or
120    /// `None` when every key does.
121    ///
122    /// Only `wav` has a real answer here. ffmpeg's WAV muxer writes the
123    /// legacy RIFF `LIST`/`INFO` chunk, and that chunk's vocabulary is exactly
124    /// this short — verified against a real encode carrying ten canonical
125    /// tags, of which precisely these six arrived and no others. Every other
126    /// target's tag format takes an arbitrary key, so there is nothing here
127    /// worth naming.
128    fn known_tag_vocabulary(self) -> Option<&'static [&'static str]> {
129        match self {
130            Target::Wav => Some(&["title", "artist", "album", "genre", "date", "tracknumber"]),
131            _ => None,
132        }
133    }
134
135    /// Canonical tag keys `present` holds that this target is known to drop.
136    ///
137    /// Empty for every target but `wav`, and empty for `wav` too when nothing
138    /// present falls outside its vocabulary — a file tagged with only title,
139    /// artist and album loses nothing by becoming a WAV.
140    pub fn tags_it_would_drop(self, present: &BTreeMap<String, Vec<String>>) -> Vec<String> {
141        let Some(kept) = self.known_tag_vocabulary() else {
142            return Vec::new();
143        };
144        present
145            .keys()
146            .filter(|key| !kept.contains(&key.as_str()))
147            .cloned()
148            .collect()
149    }
150
151    /// Typical bitrate in kbps, used only to estimate a size before the work is
152    /// done. `None` for a lossless target, where the figure would be a fiction.
153    fn nominal_kbps(self, quality: Option<Quality>) -> Option<u32> {
154        if self.lossless() {
155            return None;
156        }
157        if let Some(Quality::Bitrate(kbps)) = quality {
158            return Some(kbps);
159        }
160        Some(match self {
161            // V0 averages around 245 kbps on real music.
162            Target::Mp3 => 245,
163            Target::Opus => 128,
164            Target::Aac => 192,
165            Target::Vorbis => 192,
166            _ => 192,
167        })
168    }
169}

171/// How hard the encoder should try, as the user asked for it.
172#[derive(Debug, Clone, Copy, PartialEq, Eq)]
173pub enum Quality {
174    /// A variable-quality setting, `V0`–`V9` for MP3 or `q0`–`q10` for Vorbis:
175    /// lower is better for MP3, higher is better for Vorbis, which is the
176    /// encoders' own convention and not something to paper over.
177    Variable(u32),
178    /// A constant bitrate in kbps, written `192k`.
179    Bitrate(u32),
180}
181
182impl Quality {
183    /// Reads `V0`, `q6` or `192k`, and nothing else.
184    ///
185    /// A value that parses as none of the three is refused rather than quietly
186    /// replaced by the default: an encoder silently running at the wrong
187    /// setting produces files that are wrong in a way nobody notices until the
188    /// card is full.
189    pub fn parse(word: &str) -> Option<Quality> {
190        let word = word.trim();
191        if let Some(digits) = word
192            .strip_prefix(['V', 'v', 'Q', 'q'])
193            .and_then(|rest| rest.parse::<u32>().ok())
194            && digits <= 10
195        {
196            return Some(Quality::Variable(digits));
197        }
198        let bitrate = word.strip_suffix(['k', 'K']).unwrap_or(word);
199        bitrate
200            .parse::<u32>()
201            .ok()
202            .filter(|kbps| (8..=2048).contains(kbps))
203            .map(Quality::Bitrate)
204    }
205
206    /// How the setting is written, for a message that repeats what was typed.
207    pub const FORMS: &'static str = "V0…V9 for MP3, q0…q10 for Vorbis, or a bitrate like 192k";
208}

210/// What ffmpeg should be told about quality, given the target.
211fn quality_arguments(target: Target, quality: Option<Quality>) -> Vec<String> {
212    match (target, quality) {
213        // A lossless target has no quality knob worth exposing: FLAC's
214        // compression level trades encoding time for a few percent, and PCM
215        // has none at all.
216        (t, _) if t.lossless() => Vec::new(),
217        (_, Some(Quality::Bitrate(kbps))) => vec!["-b:a".into(), format!("{kbps}k")],
218        (_, Some(Quality::Variable(level))) => vec!["-q:a".into(), level.to_string()],
219        // The defaults are the ones each encoder is actually good at, not a
220        // single number applied to all of them.
221        (Target::Mp3, None) => vec!["-q:a".into(), "0".into()],
222        (Target::Vorbis, None) => vec!["-q:a".into(), "6".into()],
223        (Target::Opus, None) => vec!["-b:a".into(), "128k".into()],
224        (Target::Aac, None) => vec!["-b:a".into(), "192k".into()],
225        (_, None) => Vec::new(),
226    }
227}

229/// Roughly what a converted file will weigh, for a report given before the work.
230///
231/// An estimate, and said to be one wherever it is shown. The alternative is to
232/// answer "unknown" to "will this fit on my card", which is the one question
233/// somebody filling a card is asking.
234pub fn estimated_size(
235    target: Target,
236    quality: Option<Quality>,
237    duration_ms: u64,
238    source: u64,
239) -> u64 {
240    match target.nominal_kbps(quality) {
241        // kbps is kilobits per second, so a millisecond holds kbps/8 bytes:
242        // ms × kbps ÷ 8. Dividing by 8000 instead — seconds and bits confused
243        // in one step — made every estimate a thousand times too small, and a
244        // card that could "obviously" hold the whole library. A test caught it.
245        Some(kbps) => duration_ms.saturating_mul(kbps as u64) / 8,
246        // Lossless from lossless: FLAC lands around 60 % of PCM on real music,
247        // and PCM from FLAC around the inverse.
248        None if target == Target::Flac => source * 3 / 5,
249        None => source * 2,
250    }
251}

269/// Converts one file, and reports what went wrong in words rather than in an
270/// exit status.
271///
272/// The metadata travels: `-map_metadata 0` carries the tags across, and the
273/// embedded cover is copied as a picture stream where the container takes one.
274/// Neither is perfect — no two tag formats hold quite the same fields, and a
275/// Vorbis comment with three values for one field cannot be an ID3 frame that
276/// holds one — but losing the artist and the title on the way to a player is
277/// not a trade anybody would accept, and ffmpeg's mapping is what beets relies
278/// on for the same reason.
279pub fn convert(
280    ffmpeg: &str,
281    source: &Path,
282    destination: &Path,
283    target: Target,
284    quality: Option<Quality>,
285) -> Result<(), String> {
286    let (_, codec, _) = target.row();
287    let mut command = Command::new(ffmpeg);
288    command
289        // Without this a run in a terminal can stop dead waiting for an answer
290        // to a question nobody saw, in the middle of nine hundred files.
291        .arg("-nostdin")
292        .args(["-loglevel", "error"])
293        .arg("-y")
294        .arg("-i")
295        .arg(source)
296        .args(["-map", "0:a"]);
297
298    // The cover, where the container holds one. `?` makes the stream optional,
299    // so a file without art is converted rather than refused.
300    if target.keeps_embedded_art() {
301        command.args([
302            "-map",
303            "0:v?",
304            "-c:v",
305            "copy",
306            "-disposition:v:0",
307            "attached_pic",
308        ]);
309    }
310    command
311        .args(["-map_metadata", "0"])
312        .args(["-c:a", codec])
313        .args(quality_arguments(target, quality))
314        .arg(destination);
315
316    let output = command
317        .output()
318        .map_err(|e| format!("ffmpeg could not be run: {e}"))?;
319    if output.status.success() {
320        return Ok(());
321    }
322    // ffmpeg's own words, trimmed to the last few lines: the first ones are
323    // usually the banner, and the reason is at the end.
324    let said = String::from_utf8_lossy(&output.stderr);
325    let reason: Vec<&str> = said.lines().filter(|l| !l.trim().is_empty()).collect();
326    let tail = reason
327        .iter()
328        .rev()
329        .take(2)
330        .rev()
331        .copied()
332        .collect::<Vec<_>>()
333        .join("; ");
334    Err(match tail.is_empty() {
335        true => "ffmpeg failed without saying why".to_string(),
336        false => format!("ffmpeg: {tail}"),
337    })
338}

340/// Longest a converted file's playing time may differ from its source.
341///
342/// Encoders pad and trim: an MP3 frame is 1152 samples and the encoder delay is
343/// real, so an exact match would fail on correct output. Two seconds is far
344/// larger than any of that and far smaller than a truncated encode.
345const DURATION_TOLERANCE_MS: u64 = 2_000;
346
347/// Checks a converted file the only way a converted file can be checked.
348///
349/// Comparing checksums is meaningless here: the bytes are different by
350/// construction, which is the whole point. What can be checked is that the
351/// result is a file this program can read, that it holds audio, and that the
352/// audio lasts as long as what went in — which is what catches the failure that
353/// actually happens, an encode cut short by a full disk or a killed process.
354///
355/// It reads the output with Aède's own parsers rather than asking ffmpeg again:
356/// a verification that trusts the tool being verified is not one.
357pub fn verify(destination: &Path, expected_ms: Option<u64>) -> Result<(), String> {
358    let tags = crate::tags::read(destination)
359        .map_err(|e| format!("what was written cannot be read back: {e}"))?;
360    let Some(written) = tags.properties.duration_ms else {
361        return Err("what was written has no readable duration".into());
362    };
363    if written == 0 {
364        return Err("what was written holds no audio".into());
365    }
366    let Some(expected) = expected_ms.filter(|ms| *ms > 0) else {
367        return Ok(());
368    };
369    if written.abs_diff(expected) > DURATION_TOLERANCE_MS {
370        return Err(format!(
371            "the result lasts {} where the source lasts {}: the encode was cut short",
372            crate::text::format_duration(written),
373            crate::text::format_duration(expected)
374        ));
375    }
376    Ok(())
377}
