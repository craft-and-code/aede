1//! Copying a selection out of the library, keeping the tree it sits in.
2//!
3//! What this is for: filling a portable player, a card, an external drive —
4//! somewhere that is not a library and will never be scanned. The copy is a
5//! **derived artifact**, not a second catalog, and nothing here writes to the
6//! catalog or to the files it reads.
7//!
8//! Two halves, and the split is the point. [`plan`] decides everything and
9//! touches nothing: which files, where each one lands, what had to be renamed,
10//! what could not be placed at all. Only then does the caller write. A run that
11//! can say what it is about to do before doing it is a run that can be shown to
12//! the user first — which is what `--dry-run` is — and, more importantly, one
13//! whose decisions can be tested without a filesystem.
14//!
15//! The tree is kept **relative to the watched root that holds the file**, so a
16//! track at `/Volumes/Music/Ozzy/1980 Blizzard/01.flac`, scanned under
17//! `/Volumes/Music`, lands at `<destination>/Ozzy/1980 Blizzard/01.flac`. The
18//! alternative — inventing a layout from the tags — is a different feature
19//! (organising), and one this project has not decided it wants.
20

30/// Which files beside the audio travel with it.
31///
32/// The ladder exists because "images" is the wrong question. A rip folder holds
33/// the cover *and* the spectrograms *and* the scans of the booklet, all of them
34/// PNG or JPEG, and a player wants exactly one of the three. The catalog
35/// already knows which: the scan picked a cover by rank and stored it on the
36/// release. [`Extras::Cover`] is therefore an exact answer where
37/// [`Extras::Images`] can only be a heuristic.
38#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
39pub enum Extras {
40    /// Audio only. Cover art embedded in the tags still travels: it is inside
41    /// the file.
42    None,
43    /// The one cover the catalog identified for the release.
44    #[default]
45    Cover,
46    /// Every image in the folder, spectrograms and booklet scans included.
47    Images,
48    /// Everything sitting beside the audio: logs, cue sheets, reports.
49    All,
50}
51
52impl Extras {
53    /// The keyword as it is typed, or `None` for a word that names none.
54    pub fn parse(word: &str) -> Option<Extras> {
55        match text::normalize(word).as_str() {
56            "none" => Some(Extras::None),
57            "cover" | "covers" => Some(Extras::Cover),
58            "image" | "images" => Some(Extras::Images),
59            "all" => Some(Extras::All),
60            _ => None,
61        }
62    }
63
64    /// Every spelling accepted, for a message that offers what it refuses.
65    pub const NAMES: &'static str = "none, cover, images, all";
66}

71/// What a file is doing in the plan, which is what lets a report say "12
72/// albums and their covers" rather than a single undifferentiated count.
73#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
74pub enum ItemKind {
75    /// A track that was selected.

77    /// The cover art of a release holding a selected track.

80    Other,

85pub struct Item {

112/// Everything a copy would do, decided before anything is written.
113#[derive(Debug, Clone, Default, PartialEq, Eq)]
114pub struct Plan {
115    /// The files to write, in a stable order.
116    pub items: Vec<Item>,
117    /// Every component the destination filesystem forced a change to.
118    pub renamed: Vec<Renamed>,
119    /// Selected files sitting under no watched root, which therefore have no
120    /// tree to keep. Reported rather than dropped, and rather than invented a
121    /// place for.
122    pub rootless: Vec<PathBuf>,
123    /// Audio files being encoded whose embedded cover cannot follow them into
124    /// the target format, because ffmpeg refuses to mux a picture stream into
125    /// it — see [`transcode::Target::keeps_embedded_art`]. Counted rather
126    /// than named: the loss is the same whichever file it happens to.
127    pub covers_the_target_cannot_hold: usize,
128    /// Audio files being encoded into a target whose tag format has no room
129    /// for one of the tags they carry — see
130    /// [`transcode::Target::tags_it_would_drop`]. In practice this is `wav`
131    /// and its fixed legacy vocabulary; every other target takes an
132    /// arbitrary key and never adds to this count.
133    pub tags_the_target_cannot_hold: usize,
134}

158    pub extras: Extras,

169/// Whether a file should be encoded on the way out, and what into.
170///
171/// **One rule, and it falls out of what conversion is for.** A file is encoded
172/// only when it is lossless *and* not already in the target format. Everything
173/// else is copied as it stands, which covers three cases that would otherwise
174/// each need their own argument:
175///
176/// - it is already an MP3 and MP3 was asked for — re-encoding would lose
177///   quality to produce the same thing;
178/// - it is an MP3 and Opus was asked for — a second lossy pass over a first one
179///   is audible, and the file is already small, which was the point;
180/// - it is an MP3 and FLAC was asked for — the result would be *larger* than
181///   the source and no better: lossless in name, lossy in substance, which is
182///   the one thing nobody rips on purpose. Producing it deliberately would be
183///   absurd.
184///
185/// So a mixed library converted for a phone comes out with its lossless half
186/// encoded and its lossy half untouched, which is what somebody filling a phone
187/// wants and never has to ask for.
188fn conversion_for(
189    file: &crate::model::AudioFile,
190    target: Option<transcode::Target>,
191) -> Option<transcode::Target> {
192    let target = target?;
193    if !file.properties.lossless {
194        return None;
195    }
196    let already = crate::text::file_name(&file.path)
197        .rsplit_once('.')
198        .is_some_and(|(_, extension)| extension.eq_ignore_ascii_case(target.extension()));
199    (!already).then_some(target)
200}

202/// Works out what copying these tracks would mean. Writes nothing, and reads
203/// the disk only to list what sits beside the audio: every other answer comes
204/// from the catalog.
205pub fn plan(catalog: &Catalog, tracks: &[Id], recipe: &Recipe) -> Plan {
206    let (extras, restrict_names) = (recipe.extras, recipe.restrict_names);
207    let mut out = Plan::default();
208    // Two tracks can share a file — the same path selected twice through two
209    // routes — and a folder's extras are gathered once however many of its
210    // tracks were picked.
211    let mut sources: BTreeSet<&str> = BTreeSet::new();
212    let mut folders: BTreeSet<&str> = BTreeSet::new();
213    let mut releases: BTreeSet<Id> = BTreeSet::new();
214
215    for &id in tracks {
216        let Some(track) = catalog.track(id) else {
217            continue;
218        };
219        let Some(file) = catalog.file(track.file_id) else {
220            continue;
221        };
222        if sources.insert(file.path.as_str()) {
223            folders.insert(text::folder(&file.path));
224        }
225        if let Some(release_id) = track.release_id {
226            releases.insert(release_id);
227        }
228    }
229
230    let mut wanted: Vec<(String, ItemKind)> = sources
231        .iter()
232        .map(|path| ((*path).to_string(), ItemKind::Audio))
233        .collect();
234
235    // The cover the catalog settled on, which is the whole reason `Cover` can
236    // be exact where an extension filter cannot.
237    if extras == Extras::Cover {
238        for release_id in &releases {
239            if let Some(cover) = catalog
240                .release(*release_id)
241                .and_then(|r| r.cover_path.as_deref())
242            {
243                wanted.push((cover.to_string(), ItemKind::Cover));
244            }
245        }
246    }
247    if matches!(extras, Extras::Images | Extras::All) {
248        for folder in &folders {
249            for path in beside(folder, extras) {
250                wanted.push((path, ItemKind::Other));
251            }
252        }
253    }
254
255    // Sorted and deduplicated before any path is decided, so that the same
256    // catalog and the same selection produce the same plan every time — the
257    // renaming below depends on the order names are met in. `Audio` sorts
258    // before `Cover` and `Other`, so a file wanted for two reasons is kept as
259    // the more specific one.
260    wanted.sort();
261    wanted.dedup_by(|a, b| a.0 == b.0);
262
263    let mut taken: BTreeSet<PathBuf> = BTreeSet::new();
264    for (path, kind) in wanted {
265        let Some(relative) = under_a_root(catalog, &path) else {
266            out.rootless.push(PathBuf::from(path));
267            continue;
268        };
269        let file = catalog.files.iter().find(|f| f.path == path);
270        // Only the audio is ever encoded: a cover is a cover on any device.
271        let convert = match kind {
272            ItemKind::Audio => file.and_then(|f| conversion_for(f, recipe.convert)),
273            _ => None,
274        };
275        // Said once, before anything is written, rather than left for a
276        // player to discover: an encode that quietly drops a cover or a tag
277        // is a wrong answer standing in for a missing one.
278        if let (Some(target), Some(f)) = (convert, file) {
279            if !target.keeps_embedded_art() && f.has_embedded_art {
280                out.covers_the_target_cannot_hold += 1;
281            }
282            if !target.tags_it_would_drop(&f.tags).is_empty() {
283                out.tags_the_target_cannot_hold += 1;
284            }
285        }
286        // The extension changes **before** the name is placed, so that two
287        // sources landing on one name — `01.flac` and `01.wav` both becoming
288        // `01.mp3` — are seen as the collision they are rather than one file
289        // written over the other.
290        let renamed_relative = match convert {
291            Some(target) => with_extension(&relative, target.extension()),
292            None => relative.clone(),
293        };
294        let placed = place(&renamed_relative, restrict_names, &mut taken);
295        if placed != Path::new(&relative) {
296            out.renamed.push(Renamed {
297                from: PathBuf::from(&relative),
298                to: placed.clone(),
299            });
300        }
301        let size = match convert {
302            Some(target) => transcode::estimated_size(
303                target,
304                recipe.quality,
305                file.and_then(|f| f.properties.duration_ms).unwrap_or(0),
306                file.map(|f| f.size).unwrap_or(0),
307            ),
308            None => size_of(catalog, &path),
309        };
310        out.items.push(Item {
311            size,
312            source: PathBuf::from(path),
313            relative: placed,
314            kind,
315            convert,
316        });
317    }
318    out
319}

321/// The same relative path with another extension on its last component.
322///
323/// Written out rather than `Path::set_extension`, which takes everything after
324/// the **first** dot of the file name to be an extension on some platforms and
325/// would turn `Vol. 2 - Live.flac` into `Vol.mp3`.
326fn with_extension(relative: &str, extension: &str) -> String {
327    let (folder, name) = match relative.rfind('/') {
328        Some(slash) => (&relative[..=slash], &relative[slash + 1..]),
329        None => ("", relative),
330    };
331    let stem = match name.rfind('.') {
332        Some(dot) if dot > 0 => &name[..dot],
333        _ => name,
334    };
335    format!("{folder}{stem}.{extension}")
336}

338/// How many files the plan would encode rather than copy.
339impl Plan {
340    /// Items that go through an encoder, and those that are copied as they are.
341    ///
342    /// Reported separately because the difference is the one thing a user needs
343    /// to see before a conversion starts: a library that is half MP3 already
344    /// comes out half untouched, and a count that hid that would look like the
345    /// conversion had silently skipped things.
346    pub fn converted(&self) -> usize {
347        self.items
348            .iter()
349            .filter(|item| item.convert.is_some())
350            .count()
351    }
352
353    /// `true` when any size in the plan is an encoder's output, and therefore
354    /// a guess rather than a measurement.
355    pub fn size_is_estimated(&self) -> bool {
356        self.converted() > 0
357    }
358}

360/// The files sitting beside the audio that this level of `extras` asks for.
361///
362/// The one place in this module that reads the disk, and it has to: the catalog
363/// holds what the scan recognised as audio, and everything asked for here is by
364/// definition what it did not. Sorted, so two runs agree.
365///
366/// An unreadable folder yields nothing rather than stopping the plan — the
367/// audio is what was asked for, and the extras are extra.
368fn beside(folder: &str, extras: Extras) -> Vec<String> {
369    let folder = Path::new(folder);
370    let mut found: Vec<String> = Vec::new();
371    add_files_of(folder, extras, &mut found);
372    // Aède writes what it draws or keeps beside a track into a subfolder of
373    // its own rather than into the album folder directly — `spectrograms/`
374    // today, and both `Extras::Images` and `Extras::All` are documented to
375    // reach a spectrogram, not just the two of them that happen to be
376    // sitting loose next to the audio. One level down and no further: a
377    // folder inside `spectrograms/` would not be Aède's, and guessing at its
378    // shape is not this function's business.
379    if let Ok(entries) = std::fs::read_dir(folder) {
380        for entry in entries.flatten() {
381            if entry.file_type().is_ok_and(|kind| kind.is_dir()) {
382                add_files_of(&entry.path(), extras, &mut found);
383            }
384        }
385    }
386    found.sort();
387    found
388}

390/// Every plain file directly inside `folder` that this level of `extras`
391/// wants, appended as an absolute path.
392///
393/// Shared between the album folder itself and each subfolder [`beside`] steps
394/// into, so a dotfile or an audio file is refused the same way wherever it is
395/// found.
396fn add_files_of(folder: &Path, extras: Extras, found: &mut Vec<String>) {
397    let Ok(entries) = std::fs::read_dir(folder) else {
398        return;
399    };
400    for entry in entries.flatten() {
401        if !entry.file_type().is_ok_and(|kind| kind.is_file()) {
402            continue;
403        }
404        let path = entry.path();
405        let name = entry.file_name().to_string_lossy().to_string();
406        // A dotfile beside an album is the operating system's business —
407        // `.DS_Store`, thumbnail caches — and never something to carry onto a
408        // player.
409        if name.starts_with('.') {
410            continue;
411        }
412        // Audio is already in the plan through the catalog, with its size and
413        // its kind; picking it up again here would put it in twice under a
414        // vaguer name.
415        if crate::tags::is_audio_path(&path) {
416            continue;
417        }
418        if extras == Extras::Images && !is_image(&name) {
419            continue;
420        }
421        found.push(path.to_string_lossy().to_string());
422    }
423}

425/// `true` when a file name ends in an extension pictures use.
426fn is_image(name: &str) -> bool {
427    name.rsplit_once('.').is_some_and(|(_, extension)| {
428        let extension = extension.to_ascii_lowercase();
429        IMAGE_EXTENSIONS.contains(&extension.as_str())
430    })
431}

433/// Bytes the catalog last read for this path, or zero for a file it does not

435fn size_of(catalog: &Catalog, path: &str) -> u64 {
436    catalog
437        .files
438        .iter()
439        .find(|file| file.path == path)

441        .unwrap_or(0)

444/// The path of a file relative to the watched root that holds it.
445///
446/// `None` when no root does, which is not a fault to hide: the file has no tree
447/// to keep, and putting it at the destination's top level would silently mix it
448/// in with the folders that do.
449fn under_a_root(catalog: &Catalog, path: &str) -> Option<String> {
450    // The longest matching root wins. Nesting one watched folder inside another
451    // is legal, and taking the shorter one would carry the intermediate
452    // folders into a destination the user asked to be rid of.
453    catalog
454        .roots
455        .iter()
456        .filter(|root| text::is_under(path, root))
457        .max_by_key(|root| root.len())
458        .and_then(|root| text::relative_under(path, root).filter(|rest| !rest.is_empty()))
459}

461/// Turns a relative path into one the destination will accept, component by
462/// component, keeping it distinct from everything already placed.
463fn place(relative: &str, restrict_names: bool, taken: &mut BTreeSet<PathBuf>) -> PathBuf {
464    let parts: Vec<&str> = relative.split('/').filter(|p| !p.is_empty()).collect();
465    let adapt = |part: &str| match restrict_names {
466        true => names::adapt(part).unwrap_or_else(|| part.to_string()),
467        false => part.to_string(),
468    };
469    let Some((file, folders)) = parts.split_last() else {
470        return PathBuf::new();
471    };
472
473    // Folders are *meant* to be met again — every other track of the album
474    // lands in the same one — so only the file name is forced unique, and only
475    // against the names already placed in that same folder. Uniqueness across
476    // the whole tree would rename half a library, where two albums legitimately
477    // both hold an `01 Intro.flac`.
478    let directory: PathBuf = folders.iter().map(|part| adapt(part)).collect();
479    let mut here: BTreeSet<String> = taken
480        .iter()
481        .filter(|path| path.parent() == Some(directory.as_path()))
482        .filter_map(|path| path.file_name())
483        .map(|name| name.to_string_lossy().to_string())
484        .collect();
485
486    let placed = directory.join(names::make_unique(&adapt(file), &mut here));
487    taken.insert(placed.clone());
488    placed
489}

491/// `true` when a destination sits inside a folder the catalog watches.
492///
493/// Copying a library into itself is not a mistake to warn about afterwards: the
494/// next scan reads the copies as new files, the catalog doubles, and `doctor`
495/// reports every album as a duplicate of itself. Refused before anything is
496/// written.
497pub fn inside_a_watched_root(catalog: &Catalog, destination: &Path) -> Option<String> {
498    let destination = destination.to_string_lossy().to_string();
499    catalog
500        .roots
501        .iter()
502        .find(|root| text::is_under(&destination, root) || text::is_under(root, &destination))
503        .cloned()
504}

506/// The name a file is written under before it is moved into place.
507///
508/// **The extension is kept**, and not as a nicety: ffmpeg chooses its muxer
509/// from it, so a file written as `01 Crazy Train.aede-partial` made it answer
510/// "Invalid argument" and the whole conversion failed on every track. The
511/// marker therefore goes *before* the extension rather than replacing it.
512///
513/// One helper for the copy and the conversion alike, so the two cannot disagree
514/// about what a half-written file is called — which is what the resume test
515/// depends on.
516pub fn partial_path(destination: &Path) -> PathBuf {
517    let name = destination
518        .file_name()
519        .map(|n| n.to_string_lossy().to_string())
520        .unwrap_or_default();
521    let (stem, extension) = match name.rfind('.') {
522        Some(dot) if dot > 0 => (&name[..dot], &name[dot..]),
523        _ => (name.as_str(), ""),
524    };
525    destination.with_file_name(format!("{stem}.aede-partial{extension}"))
526}

547/// Copies one file, creating the folders above it.
548///
549/// `verify` re-reads what was written and compares a checksum against what was
550/// read. Two honest limits, both worth knowing rather than glossing over:
551///
552/// - the file is flushed to the device before being read back, but a read can
553///   still be served from the kernel's cache, so this proves the bytes made it
554///   through the program and the filesystem rather than onto the platter. It
555///   catches a truncated write, a full disk and a bad cable; it is not a
556///   substitute for reading the card back on another machine;
557/// - the checksum is a CRC-32, which detects accidental corruption and is not
558///   meant to resist anybody deliberately producing a collision. Nothing here
559///   is a security boundary — the question is whether a transfer went wrong.
560pub fn copy_one(
561    source: &Path,
562    destination: &Path,
563    expected_size: u64,
564    verify: bool,
565    replace: bool,
566) -> Result<Wrote, String> {
567    use std::io::Read;
568
569    // Already there and the right length: the cheap half of verification, and
570    // it runs on every file whether `--verify` was asked for or not. It costs
571    // one metadata read and catches the failure that actually happens — a run
572    // interrupted mid-file, or a disk that filled up.
573    if !replace
574        && let Ok(existing) = std::fs::metadata(destination)
575        && existing.is_file()
576        && existing.len() == expected_size
577        && expected_size > 0
578    {
579        return Ok(Wrote::Skipped);
580    }
581
582    if let Some(parent) = destination.parent() {
583        std::fs::create_dir_all(parent).map_err(|e| format!("{}: {e}", parent.display()))?;
584    }
585
586    // Written under a temporary name and moved into place, so that an
587    // interrupted run never leaves a half-file wearing the name of a whole
588    // one — the next run would see the right name, and only a wrong length
589    // would give it away.
590    let partial = partial_path(destination);
591    let written = std::fs::copy(source, &partial)
592        .map_err(|e| format!("{} → {}: {e}", source.display(), destination.display()))?;
593
594    let checked = verify
595        .then(|| {
596            // Windows requires a write-capable handle for sync_all. A copied
597            // read-only source also makes the partial read-only, so that case
598            // can only be checked by reading it back after fs::copy closes.
599            #[cfg(windows)]
600            if !std::fs::metadata(&partial)
601                .map_err(|e| e.to_string())?
602                .permissions()
603                .readonly()
604            {
605                let file = std::fs::OpenOptions::new()
606                    .write(true)
607                    .open(&partial)
608                    .map_err(|e| e.to_string())?;
609                file.sync_all().map_err(|e| e.to_string())?;
610            }
611            #[cfg(not(windows))]
612            {
613                let file = std::fs::File::open(&partial).map_err(|e| e.to_string())?;
614                file.sync_all().map_err(|e| e.to_string())?;
615            }
616            let read = |path: &Path| -> Result<u32, String> {
617                let mut file = std::fs::File::open(path).map_err(|e| e.to_string())?;
618                let mut buffer = vec![0u8; 1 << 20];
619                let mut all = Vec::new();
620                loop {
621                    let n = file.read(&mut buffer).map_err(|e| e.to_string())?;
622                    if n == 0 {
623                        break;
624                    }
625                    all.extend_from_slice(&buffer[..n]);
626                }
627                Ok(crate::audit::crc::crc32_ogg(&all))
628            };
629            Ok::<(u32, u32), String>((read(source)?, read(&partial)?))
630        })
631        .transpose()
632        .map_err(|e| format!("{}: {e}", destination.display()))?;
633
634    if let Some((from, to)) = checked
635        && from != to
636    {
637        let _ = std::fs::remove_file(&partial);
638        return Err(format!(
639            "{}: what was written back does not match what was read",
640            destination.display()
641        ));
642    }
643    if written != expected_size && expected_size > 0 {
644        let _ = std::fs::remove_file(&partial);
645        return Err(format!(
646            "{}: {written} bytes written where {expected_size} were expected",
647            destination.display()
648        ));
649    }
650
651    std::fs::rename(&partial, destination)
652        .map_err(|e| format!("{}: {e}", destination.display()))?;
653    Ok(Wrote::Copied)
654}
