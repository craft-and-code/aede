1//! Copying a selection out of the library, keeping the tree it sits in.
2//!
3//! What this is for: filling a portable player, a card, an external drive —
4//! somewhere that is not a library and will never be scanned. The copy is a
5//! **derived artifact**, not a second catalog, and nothing here writes to the
6//! catalog or to the files it reads.
7//!
8//! Two halves, and the split is the point. [`plan`] decides everything and
9//! touches nothing: which files, where each one lands, what had to be renamed,
10//! what could not be placed at all. Only then does the caller write. A run that
11//! can say what it is about to do before doing it is a run that can be shown to
12//! the user first — which is what `--dry-run` is — and, more importantly, one
13//! whose decisions can be tested without a filesystem.
14//!
15//! The tree is kept **relative to the watched root that holds the file**, so a
16//! track at `/Volumes/Music/Ozzy/1980 Blizzard/01.flac`, scanned under
17//! `/Volumes/Music`, lands at `<destination>/Ozzy/1980 Blizzard/01.flac`. The
18//! alternative — inventing a layout from the tags — is a different feature
19//! (organising), and one this project has not decided it wants.

36/// Which files beside the audio travel with it.
37///
38/// The ladder exists because "images" is the wrong question. A rip folder holds
39/// the cover *and* the spectrograms *and* the scans of the booklet, all of them
40/// PNG or JPEG, and a player wants exactly one of the three. The catalog
41/// already knows which: the scan picked a cover by rank and stored it on the
42/// release. [`Extras::Cover`] is therefore an exact answer where
43/// [`Extras::Images`] can only be a heuristic.
44#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
45pub enum Extras {
46    /// Audio only. Cover art embedded in the tags still travels: it is inside
47    /// the file.
48    None,
49    /// The one cover the catalog identified for the release.
50    #[default]
51    Cover,
52    /// Every image in the folder, spectrograms and booklet scans included.
53    Images,
54    /// Everything sitting beside the audio: logs, cue sheets, reports.
55    All,
56}
57
58impl Extras {
59    /// The keyword as it is typed, or `None` for a word that names none.
60    pub fn parse(word: &str) -> Option<Extras> {
61        match text::normalize(word).as_str() {
62            "none" => Some(Extras::None),
63            "cover" | "covers" => Some(Extras::Cover),
64            "image" | "images" => Some(Extras::Images),
65            "all" => Some(Extras::All),
66            _ => None,
67        }
68    }
69
70    /// Every spelling accepted, for a message that offers what it refuses.
71    pub const NAMES: &'static str = "none, cover, images, all";
72}

81    /// A track that was selected.

121/// Everything a copy would do, decided before anything is written.
122#[derive(Debug, Clone, Default, PartialEq, Eq)]
123pub struct Plan {
124    /// The files to write, in a stable order.
125    pub items: Vec<Item>,
126    /// Every component the destination filesystem forced a change to.
127    pub renamed: Vec<Renamed>,
128    /// Selected files sitting under no watched root, which therefore have no
129    /// tree to keep. Reported rather than dropped, and rather than invented a
130    /// place for.
131    pub rootless: Vec<PathBuf>,
132    /// Sources or requested companion folders that cannot be read or converted
133    /// as requested. The caller must refuse these before writing.
134    pub rejected: Vec<Failed>,
135    /// Audio files being encoded whose embedded cover cannot follow them into
136    /// the target format, because ffmpeg refuses to mux a picture stream into
137    /// it — see [`transcode::Target::keeps_embedded_art`]. Counted rather
138    /// than named: the loss is the same whichever file it happens to.
139    pub covers_the_target_cannot_hold: usize,
140    /// Audio files being encoded into a target whose tag format has no room
141    /// for one of the tags they carry — see
142    /// [`transcode::Target::tags_it_would_drop`]. In practice this is `wav`
143    /// and its fixed legacy vocabulary; every other target takes an
144    /// arbitrary key and never adds to this count.
145    pub tags_the_target_cannot_hold: usize,
146}

149    /// Bytes the copy would write.
150    pub fn total_bytes(&self) -> u64 {
151        self.items
152            .iter()
153            .fold(0, |total, item| total.saturating_add(item.size))
154    }

156    /// How many files of each kind, for a report that says what it is copying.
157    pub fn counts(&self) -> BTreeMap<ItemKind, usize> {
158        let mut out = BTreeMap::new();

161        }
162        out
163    }

183/// Whether a file should be encoded on the way out, and what into.
184///
185/// **One rule, and it falls out of what conversion is for.** A file is encoded
186/// only when it is lossless *and* not already in the target format. Everything
187/// else is copied as it stands, which covers three cases that would otherwise
188/// each need their own argument:
189///
190/// - it is already an MP3 and MP3 was asked for — re-encoding would lose
191///   quality to produce the same thing;
192/// - it is an MP3 and Opus was asked for — a second lossy pass over a first one
193///   is audible, and the file is already small, which was the point;
194/// - it is an MP3 and FLAC was asked for — the result would be *larger* than
195///   the source and no better: lossless in name, lossy in substance, which is
196///   the one thing nobody rips on purpose. Producing it deliberately would be
197///   absurd.
198///
199/// So a mixed library converted for a phone comes out with its lossless half
200/// encoded and its lossy half untouched, which is what somebody filling a phone
201/// wants and never has to ask for.
202fn conversion_for(
203    file: &crate::model::AudioFile,
204    target: Option<transcode::Target>,
205) -> Option<transcode::Target> {
206    let target = target?;
207    if !file.properties.lossless {
208        return None;
209    }
210    // A container extension does not identify its codec: lossless ALAC in
211    // M4A must still be encoded when AAC is requested. Container aliases such
212    // as .wave already hold WAV and can be copied without re-encoding.
213    let already = match target {
214        transcode::Target::Flac => {
215            file.properties.codec.eq_ignore_ascii_case("flac")
216                && (file.properties.container.eq_ignore_ascii_case("flac")
217                    || (file.properties.container.is_empty()
218                        && Path::new(&file.path)
219                            .extension()
220                            .is_some_and(|extension| extension.eq_ignore_ascii_case("flac"))))
221        }
222        transcode::Target::Wav => file.properties.container.eq_ignore_ascii_case("wav"),
223        transcode::Target::Mp3 => file.properties.codec.eq_ignore_ascii_case("mp3"),
224        transcode::Target::Opus => file.properties.codec.eq_ignore_ascii_case("opus"),
225        transcode::Target::Aac => file.properties.codec.eq_ignore_ascii_case("aac"),
226        transcode::Target::Vorbis => file.properties.codec.eq_ignore_ascii_case("vorbis"),
227    };
228    (!already).then_some(target)
229}

231/// Works out what copying these tracks would mean. Writes nothing, and reads
232/// the disk only to list what sits beside the audio: every other answer comes
233/// from the catalog.
234pub fn plan(catalog: &Catalog, tracks: &[Id], recipe: &Recipe) -> Plan {
235    let (extras, restrict_names) = (recipe.extras, recipe.restrict_names);
236    let mut out = Plan::default();
237    // Two tracks can share a file — the same path selected twice through two
238    // routes — and a folder's extras are gathered once however many of its
239    // tracks were picked.
240    let mut sources: BTreeSet<&str> = BTreeSet::new();
241    let mut folders: BTreeSet<&str> = BTreeSet::new();
242    let mut releases: BTreeSet<Id> = BTreeSet::new();
243
244    for &id in tracks {
245        let Some(track) = catalog.track(id) else {
246            continue;
247        };
248        let Some(file) = catalog.file(track.file_id) else {
249            continue;
250        };
251        if sources.insert(file.path.as_str()) {
252            folders.insert(text::folder(&file.path));
253        }
254        if let Some(release_id) = track.release_id {
255            releases.insert(release_id);
256        }
257    }
258
259    let mut wanted: Vec<(String, ItemKind)> = sources
260        .iter()
261        .map(|path| ((*path).to_string(), ItemKind::Audio))
262        .collect();
263
264    // The cover the catalog settled on, which is the whole reason `Cover` can
265    // be exact where an extension filter cannot.
266    if extras == Extras::Cover {
267        for release_id in &releases {
268            if let Some(cover) = catalog
269                .release(*release_id)
270                .and_then(|r| r.cover_path.as_deref())
271            {
272                wanted.push((cover.to_string(), ItemKind::Cover));
273            }
274        }
275    }
276    if matches!(extras, Extras::Images | Extras::All) {
277        for folder in &folders {
278            match beside(folder, extras) {
279                Ok(paths) => wanted.extend(paths.into_iter().map(|path| (path, ItemKind::Other))),
280                Err(failure) => out.rejected.push(failure),
281            }
282        }
283    }
284
285    // Sorted and deduplicated before any path is decided, so that the same
286    // catalog and the same selection produce the same plan every time — the
287    // renaming below depends on the order names are met in. `Audio` sorts
288    // before `Cover` and `Other`, so a file wanted for two reasons is kept as
289    // the more specific one.
290    wanted.sort();
291    wanted.dedup_by(|a, b| a.0 == b.0);
292
293    let files: BTreeMap<&str, &crate::model::AudioFile> = catalog
294        .files
295        .iter()
296        .map(|file| (file.path.as_str(), file))
297        .collect();
298    let mut placement = Placement::default();
299    for (path, kind) in wanted {
300        let Some(relative) = under_a_root(catalog, &path) else {
301            out.rootless.push(PathBuf::from(path));
302            continue;
303        };
304        let file = files.get(path.as_str()).copied();
305        // Only the audio is ever encoded: a cover is a cover on any device.
306        let convert = match kind {
307            ItemKind::Audio => file.and_then(|f| conversion_for(f, recipe.convert)),
308            _ => None,
309        };
310        // Said once, before anything is written, rather than left for a
311        // player to discover: an encode that quietly drops a cover or a tag
312        // is a wrong answer standing in for a missing one.
313        if let (Some(target), Some(f)) = (convert, file) {
314            if let Err(reason) = transcode::validate_conversion(&f.properties, target) {
315                out.rejected.push(Failed {
316                    source: PathBuf::from(&path),
317                    reason,
318                });
319                continue;
320            }
321            if !target.keeps_embedded_art() && f.has_embedded_art {
322                out.covers_the_target_cannot_hold += 1;
323            }
324            if !target.tags_it_would_drop(&f.tags).is_empty() {
325                out.tags_the_target_cannot_hold += 1;
326            }
327        }
328        // The extension changes **before** the name is placed, so that two
329        // sources landing on one name — `01.flac` and `01.wav` both becoming
330        // `01.mp3` — are seen as the collision they are rather than one file
331        // written over the other.
332        let renamed_relative = match convert {
333            Some(target) => with_extension(&relative, target.extension()),
334            None => relative.clone(),
335        };
336        let placed = placement.place(&renamed_relative, &path, restrict_names);
337        if placed != Path::new(&relative) {
338            out.renamed.push(Renamed {
339                from: PathBuf::from(&relative),
340                to: placed.clone(),
341            });
342        }
343        let size = match convert {
344            Some(target) => transcode::estimated_size(
345                target,
346                recipe.quality,
347                file.and_then(|f| f.properties.duration_ms).unwrap_or(0),
348                file.map(|f| f.size).unwrap_or(0),
349            ),
350            None => file
351                .map(|f| f.size)
352                .unwrap_or_else(|| std::fs::metadata(&path).map(|meta| meta.len()).unwrap_or(0)),
353        };
354        out.items.push(Item {
355            size,
356            source: PathBuf::from(path),
357            relative: placed,
358            kind,
359            convert,
360            contents: None,
361        });
362    }
363    out
364}

366/// The same relative path with another extension on its last component.

370    path.to_string_lossy().into_owned()

373/// How many files the plan would encode rather than copy.
374impl Plan {
375    /// Items that go through an encoder, and those that are copied as they are.
376    ///
377    /// Reported separately because the difference is the one thing a user needs
378    /// to see before a conversion starts: a library that is half MP3 already

382        self.items

386    }

388    /// `true` when any size in the plan is an encoder's output, and therefore

391        self.converted() > 0
392    }

395/// The files sitting beside the audio that this level of `extras` asks for.
396///
397/// The one place in this module that reads the disk, and it has to: the catalog
398/// holds what the scan recognised as audio, and everything asked for here is by
399/// definition what it did not. Sorted, so two runs agree.
400///
401/// An unreadable requested folder is an error: reporting a complete copy while
402/// silently omitting selected companions would not honour `extras`.
403fn beside(folder: &str, extras: Extras) -> Result<Vec<String>, Failed> {
404    let folder = Path::new(folder);
405    let mut found: Vec<String> = Vec::new();
406    let folders = add_files_of(folder, extras, &mut found)?;
407    // Aède writes what it draws or keeps beside a track into a subfolder of
408    // its own rather than into the album folder directly — `spectrograms/`
409    // today, and both `Extras::Images` and `Extras::All` are documented to
410    // reach a spectrogram, not just the two of them that happen to be
411    // sitting loose next to the audio. One level down and no further: a
412    // folder inside `spectrograms/` would not be Aède's, and guessing at its
413    // shape is not this function's business.
414    for folder in folders {
415        add_files_of(&folder, extras, &mut found)?;
416    }
417    found.sort();
418    Ok(found)
419}

421/// Every plain file directly inside `folder` that this level of `extras`
422/// wants, appended as an absolute path.
423///
424/// Shared between the album folder itself and each subfolder [`beside`] steps
425/// into, so a dotfile or an audio file is refused the same way wherever it is
426/// found.
427fn add_files_of(
428    folder: &Path,
429    extras: Extras,
430    found: &mut Vec<String>,
431) -> Result<Vec<PathBuf>, Failed> {
432    let entries = std::fs::read_dir(folder).map_err(|error| extra_error(folder, error))?;
433    let mut folders = Vec::new();
434    for entry in entries {
435        let entry = entry.map_err(|error| extra_error(folder, error))?;
436        let kind = entry
437            .file_type()
438            .map_err(|error| extra_error(&entry.path(), error))?;
439        if kind.is_dir() {
440            folders.push(entry.path());
441            continue;
442        }
443        if !kind.is_file() {
444            continue;
445        }
446        let path = entry.path();
447        let name = entry.file_name().to_string_lossy().to_string();
448        // A dotfile beside an album is the operating system's business —
449        // `.DS_Store`, thumbnail caches — and never something to carry onto a
450        // player.
451        if name.starts_with('.') {
452            continue;
453        }
454        // Audio is already in the plan through the catalog, with its size and
455        // its kind; picking it up again here would put it in twice under a
456        // vaguer name.
457        if crate::tags::is_audio_path(&path) {
458            continue;
459        }
460        if extras == Extras::Images && !is_image(&name) {
461            continue;
462        }
463        found.push(path.to_string_lossy().to_string());
464    }
465    folders.sort();
466    Ok(folders)
467}

469fn extra_error(path: &Path, error: std::io::Error) -> Failed {
470    Failed {
471        source: path.to_path_buf(),
472        reason: format!("cannot read requested companions: {error}"),
473    }
474}

476/// `true` when a file name ends in an extension pictures use.
477fn is_image(name: &str) -> bool {
478    name.rsplit_once('.').is_some_and(|(_, extension)| {
479        let extension = extension.to_ascii_lowercase();
480        IMAGE_EXTENSIONS.contains(&extension.as_str())
481    })

484/// The path of a file relative to the watched root that holds it.
485///
486/// `None` when no root does, which is not a fault to hide: the file has no tree
487/// to keep, and putting it at the destination's top level would silently mix it
488/// in with the folders that do.
489fn under_a_root(catalog: &Catalog, path: &str) -> Option<String> {
490    // The longest matching root wins. Nesting one watched folder inside another
491    // is legal, and taking the shorter one would carry the intermediate
492    // folders into a destination the user asked to be rid of.
493    catalog
494        .roots
495        .iter()
496        .filter(|root| text::is_under(path, root))
497        .max_by_key(|root| root.len())
498        .and_then(|root| text::relative_under(path, root).filter(|rest| !rest.is_empty()))
499}

501/// Original directories have stable placements; each output parent has its
502/// own namespace shared by files and directories. Conservative matching keys make
503/// the plan portable without allowing a case variant to silently merge albums.
504#[derive(Default)]
505struct Placement {
506    directories: BTreeMap<String, PathBuf>,
507    names: BTreeMap<PathBuf, BTreeSet<String>>,
508}
509
510impl Placement {
511    fn place(&mut self, relative: &str, source: &str, restrict: bool) -> PathBuf {
512        let parts: Vec<&str> = relative
513            .split('/')
514            .filter(|part| !part.is_empty())
515            .collect();
516        let Some((file, folders)) = parts.split_last() else {
517            return PathBuf::new();
518        };
519        let adapt = |part: &str| {
520            if restrict {
521                names::adapt(part).unwrap_or_else(|| part.to_string())
522            } else {
523                part.to_string()
524            }
525        };
526        let mut source_root = text::folder(source);
527        for _ in folders {
528            source_root = text::folder(source_root);
529        }
530        let mut original = format!("{source_root}/");
531        let mut parent = PathBuf::new();
532        for folder in folders {
533            original.push_str(folder);
534            original.push('/');
535            parent = match self.directories.get(&original) {
536                Some(held) => held.clone(),
537                None => {
538                    let name = names::make_unique_portable(
539                        &adapt(folder),
540                        self.names.entry(parent.clone()).or_default(),
541                    );
542                    let placed = parent.join(name);
543                    self.directories.insert(original.clone(), placed.clone());
544                    placed
545                }
546            };
547        }
548        let name = names::make_unique_portable(
549            &adapt(file),
550            self.names.entry(parent.clone()).or_default(),
551        );
552        parent.join(name)
553    }
554}

556/// `true` when a destination sits inside a folder the catalog watches.
557///
558/// Copying a library into itself is not a mistake to warn about afterwards: the
559/// next scan reads the copies as new files, the catalog doubles, and `doctor`
560/// reports every album as a duplicate of itself. Refused before anything is
561/// written.
562pub fn inside_a_watched_root(catalog: &Catalog, destination: &Path) -> Option<String> {
563    let destination = destination.to_string_lossy().to_string();
564    catalog
565        .roots
566        .iter()
567        .find(|root| text::is_under(&destination, root) || text::is_under(root, &destination))
568        .cloned()
569}

571/// Recognizes the fixed temporary pathname used by older versions.
572///
573/// New writes use [`TemporaryOutput`] with an exclusive private directory;
574/// this helper remains available for callers identifying abandoned old files.
575pub fn partial_path(destination: &Path) -> PathBuf {
576    let name = destination
577        .file_name()
578        .map(|n| n.to_string_lossy().to_string())
579        .unwrap_or_default();
580    let (stem, extension) = match name.rfind('.') {
581        Some(dot) if dot > 0 => (&name[..dot], &name[dot..]),
582        _ => (name.as_str(), ""),
583    };
584    destination.with_file_name(format!("{stem}.aede-partial{extension}"))
585}

623/// Copies a file, optionally requiring matching bytes for an existing output.
624///
625/// A mismatching existing output is retained and reported as an error; explicit
626/// replacement is required to refresh it. Verification has fixed memory use.
627pub fn copy_one_checked(
628    source: &Path,
629    destination: &Path,
630    expected_size: u64,
631    verify: bool,
632    replace: bool,
633    verify_existing: bool,
634) -> Result<Wrote, String> {
635    validate_source(source)?;
636    if let (Ok(source_path), Ok(destination_path)) =
637        (source.canonicalize(), destination.canonicalize())
638        && source_path == destination_path
639    {
640        return Err(format!(
641            "{}: source and destination are the same file",
642            destination.display()
643        ));
644    }
645    if !replace && let Some(existing) = output::regular_destination(destination)? {
646        if verify_existing {
647            return match files_match(source, destination)? {
648                true => Ok(Wrote::Skipped),
649                false => Err(format!(
650                    "{}: existing content does not match the source; use --replace to refresh it",
651                    destination.display()
652                )),
653            };
654        }
655        if existing.len() == expected_size && expected_size > 0 {
656            return Ok(Wrote::Skipped);
657        }
658        return Err(format!(
659            "{}: existing size differs from the source; use --replace to refresh it",
660            destination.display()
661        ));
662    }
663    let temporary = TemporaryOutput::new(destination)?;
664    let written = std::fs::copy(source, temporary.path())
665        .map_err(|error| format!("{} → {}: {error}", source.display(), destination.display()))?;
666    if written != expected_size && expected_size > 0 {
667        return Err(format!(
668            "{}: {written} bytes written where {expected_size} were expected",
669            destination.display()
670        ));
671    }
672    if verify {
673        #[cfg(windows)]
674        if !std::fs::metadata(temporary.path())
675            .map_err(|error| error.to_string())?
676            .permissions()
677            .readonly()
678        {
679            std::fs::OpenOptions::new()
680                .write(true)
681                .open(temporary.path())
682                .and_then(|file| file.sync_all())
683                .map_err(|error| error.to_string())?;
684        }
685        #[cfg(not(windows))]
686        std::fs::File::open(temporary.path())
687            .and_then(|file| file.sync_all())
688            .map_err(|error| error.to_string())?;
689        if !files_match(source, temporary.path())? {
690            return Err(format!(
691                "{}: what was written back does not match what was read",
692                destination.display()
693            ));
694        }
695    }
696    if replace {
697        temporary.publish()?;
698    } else {
699        temporary.publish_new()?;
700    }
701    Ok(Wrote::Copied)
702}
