1//! What the user writes: favourites, ratings, notes, tags, listening history.
2//!
3//! This is the only data in the program that cannot be recovered. Lose the
4//! catalog and a scan rebuilds it in a minute; lose this and it is gone. Two
5//! consequences run through the whole module.
6//!
7//! **It lives in its own file**, never in the catalog. The catalog is derived
8//! from the disk and written whole; a rating that changes on a keystroke has no
9//! business rewriting a library.
10//!
11//! **It is never keyed by a catalog identifier.** Those are positions in a
12//! vector that every scan renumbers — the lesson the imported analyses already
13//! taught, at the cost of a rewrite. A [`EntityRef`] names a thing the way the
14//! thing names itself, and an annotation whose target is missing is **kept
15//! waiting, never dropped**: a folder renamed between two scans must not cost
16//! anybody a note they wrote.
17//!
18//! Favourites, ratings, notes and tags look like four features and are one:
19//! something a person said about an entity. One record per target, so copying
20//! everything said about one album onto another is a record copy rather than a
21//! loop over four kinds.
22

69/// A stable way to name an entity, independent of any scan.
70///
71/// The key is what the thing calls itself rather than where it currently sits:
72/// a path for a track, the release key for an album, the normalized name for
73/// name-based entities. Recordings, works and release groups use an explicit
74/// MusicBrainz identifier when the catalog has one. Source-only artists and
75/// canonical credit exclusions may also use `mbid:<identifier>`.
76#[derive(Debug, Clone, PartialEq, Eq, PartialOrd, Ord)]
77pub struct EntityRef {
78    /// Which kind of thing is named.
79    pub kind: EntityKind,
80    /// Its stable key, in the spelling [`EntityRef`] builds.
81    pub key: String,
82}
83
84impl EntityRef {
85    /// Builds a reference by hand, for a key already in the right shape.
86    pub fn new(kind: EntityKind, key: impl Into<String>) -> EntityRef {
87        EntityRef {
88            kind,
89            key: key.into(),
90        }
91    }
92
93    /// How the reference is written in one token: `track:/music/a.flac`.
94    ///
95    /// The kind comes first and the key may itself contain colons — a path
96    /// does — so the split is on the **first** one only.
97    pub fn to_token(&self) -> String {
98        format!("{}:{}", self.kind.as_str(), self.key)
99    }
100
101    /// Reads back what [`EntityRef::to_token`] wrote.
102    pub fn parse_token(text: &str) -> Option<EntityRef> {
103        let (kind, key) = text.split_once(':')?;
104        Some(EntityRef {
105            kind: EntityKind::parse_kind(kind)?,
106            key: key.to_string(),
107        })
108    }
109
110    /// The reference naming an entity of a catalog, if that entity exists.
111    pub fn of(catalog: &Catalog, kind: EntityKind, id: Id) -> Option<EntityRef> {
112        let key = match kind {
113            // The path is what the file calls itself, and it survives every
114            // rescan that leaves the file where it is.
115            EntityKind::Track => catalog
116                .track(id)
117                .and_then(|t| catalog.file(t.file_id))
118                .map(|f| f.path.clone())?,
119            // The same three parts the release is built on, so a release
120            // recognised as itself by a scan is recognised here too.
121            EntityKind::Release => {
122                let release = catalog.release(id)?;
123                let artist = release
124                    .album_artist_id
125                    .and_then(|a| catalog.artist(a))
126                    .map(|a| a.key.clone())
127                    .unwrap_or_default();
128                let folder = &release.folder;
129                format!(
130                    "{artist}{RELEASE_KEY_SEPARATOR}{}{RELEASE_KEY_SEPARATOR}{folder}",
131                    text::normalize(&release.title)
132                )
133            }
134            EntityKind::Artist => catalog.artist(id).map(|a| a.key.clone())?,
135            EntityKind::Recording => {
136                let recording = catalog.recording(id)?;
137                recording.mbid.clone().or_else(|| {
138                    recording
139                        .track_ids
140                        .first()
141                        .and_then(|track| catalog.track(*track))
142                        .and_then(|track| catalog.file(track.file_id))
143                        .map(|file| format!("local:{}", file.path))
144                })?
145            }
146            EntityKind::Work => catalog.work(id).map(|w| w.mbid.clone())?,
147            EntityKind::ReleaseGroup => catalog.release_group(id).map(|g| g.mbid.clone())?,
148            EntityKind::Label => catalog.label(id).map(|l| l.key.clone())?,
149            EntityKind::Genre => catalog.genres.get(id as usize).map(|g| g.key.clone())?,
150        };
151        Some(EntityRef { kind, key })
152    }
153
154    /// The entity a reference names in this catalog, if it holds one.
155    pub fn resolve(&self, catalog: &Catalog) -> Option<Id> {
156        match self.kind {
157            EntityKind::Track => catalog
158                .tracks
159                .iter()
160                .find(|t| catalog.file(t.file_id).is_some_and(|f| f.path == self.key))
161                .map(|t| t.id),
162            EntityKind::Release => identity::resolve_release(self, catalog),
163            EntityKind::Artist => catalog
164                .artists
165                .iter()
166                .find(|a| a.key == self.key)
167                .or_else(|| {
168                    let mbid = self.key.strip_prefix("mbid:")?;
169                    let mut matched = catalog
170                        .artists
171                        .iter()
172                        .filter(|artist| artist.mbid.as_deref() == Some(mbid));
173                    let artist = matched.next()?;
174                    matched.next().is_none().then_some(artist)
175                })
176                .map(|a| a.id),
177            EntityKind::Recording => catalog
178                .recordings
179                .iter()
180                .find(|recording| {
181                    recording.mbid.as_deref() == Some(&self.key)
182                        || self.key.strip_prefix("local:").is_some_and(|path| {
183                            recording.track_ids.iter().any(|track| {
184                                catalog
185                                    .track(*track)
186                                    .and_then(|track| catalog.file(track.file_id))
187                                    .is_some_and(|file| file.path == path)
188                            })
189                        })
190                })
191                .map(|recording| recording.id),
192            EntityKind::Work => catalog
193                .works
194                .iter()
195                .find(|work| work.mbid == self.key)
196                .map(|work| work.id),
197            EntityKind::ReleaseGroup => catalog
198                .release_groups
199                .iter()
200                .find(|group| group.mbid == self.key)
201                .map(|group| group.id),
202            EntityKind::Label => catalog
203                .labels
204                .iter()
205                .find(|l| l.key == self.key)
206                .map(|l| l.id),
207            EntityKind::Genre => catalog
208                .genres
209                .iter()
210                .find(|g| g.key == self.key)
211                .map(|g| g.id),
212        }
213    }
214
215    /// How the thing is called on screen, or the key when it is gone.
216    pub fn display_name(&self, catalog: &Catalog) -> String {
217        let named = self.resolve(catalog).and_then(|id| match self.kind {
218            EntityKind::Track => catalog.track(id).map(|t| t.title.clone()),
219            EntityKind::Release => catalog.release(id).map(|r| r.title.clone()),
220            EntityKind::Artist => catalog.artist(id).map(|a| a.name.clone()),
221            EntityKind::Recording => catalog.recording(id).map(|r| r.title.clone()),
222            EntityKind::Work => catalog.work(id).map(|w| w.title.clone()),
223            EntityKind::ReleaseGroup => catalog.release_group(id).map(|g| g.title.clone()),
224            EntityKind::Label => catalog.label(id).map(|l| l.name.clone()),
225            EntityKind::Genre => catalog.genres.get(id as usize).map(|g| g.name.clone()),
226        });
227        named.unwrap_or_else(|| match self.kind {
228            // A path is unreadable in a column; its file name is not.
229            EntityKind::Track => text::file_name(&self.key).to_string(),
230            _ => self.key.clone(),
231        })
232    }

235/// Everything one person said about one entity.
236///
237/// One record per target rather than one per fact: a favourite, a rating, a
238/// note and a set of tags are four ways of having an opinion, not four
239/// features, and keeping them together is what makes copying what was said
240/// about an album onto another album a single operation.
241#[derive(Debug, Clone, Default, PartialEq)]
242pub struct Annotation {
243    /// Whose opinion it is.
244    pub owner: UserRef,
245    /// What it is about.
246    pub target: EntityRef,
247    /// A favourite.
248    pub loved: bool,
249    /// One to five stars, or nothing said.
250    pub rating: Option<u8>,
251    /// Free text, as long as the user cares to make it.
252    pub note: Option<String>,
253    /// Free labels: "to rip again", "vinyl", "for the car".
254    pub tags: BTreeSet<String>,
255    /// When it was first written (Unix epoch, seconds).
256    pub created_at: u64,
257    /// When it was last touched.
258    pub updated_at: u64,
259}
260
261impl Default for EntityRef {
262    fn default() -> EntityRef {
263        EntityRef {
264            kind: EntityKind::Track,
265            key: String::new(),
266        }
267    }
268}

270impl Annotation {
271    /// `true` when the record says nothing at all and can be dropped.
272    ///
273    /// Removing the last thing said about an album must remove the record, not
274    /// leave an empty one that a listing would then have to filter out — and
275    /// that an export would carry for ever.
276    pub fn is_empty(&self) -> bool {
277        // A note of `Some("")` says nothing, the same as `None` — the
278        // difference between the two is an implementation detail (typed
279        // empty text, an empty `--file`, a round trip through export and
280        // import) that nobody looking at this record should have to know
281        // about. Treating it as content would keep a record alive, and
282        // therefore visible, for having said nothing at all.
283        !self.loved
284            && self.rating.is_none()
285            && self.note.as_deref().is_none_or(|n| n.trim().is_empty())
286            && self.tags.is_empty()
287    }
288}

290/// A personal note and free-form labels attached to one relationship.
291///
292/// Relations are not entities: the opinion belongs to the edge itself, not to
293/// either endpoint. Keeping it in the user layer also preserves the invariant
294/// that the shared catalog contains facts while each person's notes remain
295/// private.
296#[derive(Debug, Clone, PartialEq, Eq)]
297pub struct RelationAnnotation {
298    /// Whose opinion it is.
299    pub owner: UserRef,
300    /// Stable relationship being described.
301    pub relation: crate::graph::RelationRef,
302    /// Free text about this exact relationship.
303    pub note: Option<String>,
304    /// Personal labels such as `dubious`, `favourite collaboration`…
305    pub tags: BTreeSet<String>,
306    /// When the annotation was first written.
307    pub created_at: u64,
308    /// When it was last changed.
309    pub updated_at: u64,
310}
311
312impl RelationAnnotation {
313    /// Whether the annotation says nothing and can be removed.
314    pub fn is_empty(&self) -> bool {
315        self.note
316            .as_deref()
317            .is_none_or(|note| note.trim().is_empty())
318            && self.tags.is_empty()
319    }

322/// One listening event.
323///
324/// An annotation is a statement; a play is an event, and events accumulate.
325#[derive(Debug, Clone, PartialEq)]
326pub struct Play {

328    pub owner: UserRef,
329    /// What was played.
330    pub track: EntityRef,

360/// A query saved under a name: a smart collection.
361///
362/// It holds the expression rather than the result, so it answers with what the
363/// library holds *now* — which is the whole difference between a smart
364/// collection and a playlist. And since running one produces a selection, and a
365/// selection is what `--csv`, `--m3u` and M3's queue consume, a saved query is
366/// playable the day it is written, with nothing built for it.
367#[derive(Debug, Clone, PartialEq)]
368pub struct Collection {
369    /// Whose collection it is.
370    pub owner: UserRef,
371    /// What it is called, as typed.
372    pub name: String,
373    /// The expression, kept as written rather than as parsed: a grammar that
374    /// gains a field must not silently change what an old collection means.
375    pub expression: String,
376    /// When it was first saved.
377    pub created_at: u64,
378    /// When it was last changed.
379    pub updated_at: u64,
380}

389/// A record the user has taken off a report, and why that is allowed.
390///
391/// **The one thing in this file that is not keyed on an entity of the
392/// catalog.** Everything else here describes something the library holds; this
393/// describes something it deliberately does *not*. `aede missing` lists albums a
394/// source credits to an artist and the shelf lacks, and sometimes the source is
395/// simply wrong about what an album is — a demo, a compilation and a single all
396/// arrive typed `Album` until somebody says otherwise on MusicBrainz.
397///
398/// Aède will not overrule a source: the whole attributed layer exists so that
399/// what somebody else said stays what they said, correctable only by them. But
400/// it will record that **you** disagree, which is what this file has always
401/// been for, and stop putting the record in front of you.
402#[derive(Debug, Clone, PartialEq, Eq)]
403pub struct SetAside {
404    /// Whose decision it is.
405    pub owner: UserRef,
406    /// The MusicBrainz release-group identifier, which is the key.
407    ///
408    /// Globally unique, so no artist is needed to tell two apart, and stable
409    /// across a re-fetch and a rescan — where a title is neither.
410    pub release_group: String,
411    /// The title as it read when it was set aside.
412    ///
413    /// Kept so a listing can show something a person recognises. Without it the
414    /// list is a column of identifiers, and a decision nobody can read is one
415    /// nobody can undo.
416    pub title: String,
417    /// When it was set aside.
418    pub created_at: u64,
419}

421/// Two spellings the owner of the disk says are one musician.
422///
423/// The second half of artist identity, and the half no authority can settle.
424/// `MUSICBRAINZ_ARTISTID` answers for the files that met Picard; for an old
425/// rip, a download, a friend's drive, **nobody on earth knows that a particular
426/// `O. Osbourne` is Ozzy except the person whose shelf it is**. Comparing the
427/// strings cannot stand in for them — a fragment match merges Angus Young with
428/// Neil Young — so the program asks instead of guessing, and keeps the answer
429/// here, with everything else that was said rather than derived.
430///
431/// **Both keys are normalised**, because that is the form the merge is decided
432/// on everywhere else in the program: a statement about `Ozzy  Osbourne` must
433/// still be about `ozzy osbourne` on the next scan. They are readable as they
434/// stand — unlike a [`SetAside`], whose key is an identifier and which
435/// therefore has to carry a title beside it.
436///
437/// Nothing in an audio file changes, ever. This is a statement about how the
438/// shelf is *read*, and `aede merge --forget` takes it back.
439#[derive(Debug, Clone, PartialEq, Eq)]
440pub struct SameArtist {
441    /// Whose statement it is.
442    pub owner: UserRef,
443    /// The spelling that gives way, normalised.
444    pub spelling: String,
445    /// The spelling it is filed under, normalised.
446    pub filed_as: String,
447    /// When it was said.
448    pub created_at: u64,
449}

480impl UserData {
481    /// The record for one owner and target, if there is one.
482    pub fn find(&self, owner: &str, target: &EntityRef) -> Option<&Annotation> {
483        self.annotations
484            .iter()
485            .find(|a| a.owner == owner && &a.target == target)
486    }

488    /// The record for one owner and target, opening one if needed.
489    ///
490    /// `now` stamps a record that did not exist; an existing one keeps its
491    /// creation date and has its `updated_at` moved by the caller.
492    pub fn entry(&mut self, owner: &str, target: &EntityRef, now: u64) -> &mut Annotation {
493        let found = self
494            .annotations
495            .iter()
496            .position(|a| a.owner == owner && &a.target == target);
497        let at = match found {
498            Some(at) => at,
499            None => {
500                self.annotations.push(Annotation {
501                    owner: owner.to_string(),
502                    target: target.clone(),
503                    created_at: now,
504                    updated_at: now,
505                    ..Default::default()
506                });
507                self.annotations.len() - 1
508            }
509        };
510        &mut self.annotations[at]
511    }

513    /// Drops the records that no longer say anything.
514    ///
515    /// Called after every change, so that un-loving the one album somebody ever
516    /// loved leaves nothing behind.
517    pub fn forget_empty(&mut self) {
518        self.annotations.retain(|a| !a.is_empty());
519        self.relation_annotations.retain(|a| !a.is_empty());
520    }

533    /// Opens the personal annotation for one relation.
534    pub fn relation_entry(
535        &mut self,
536        owner: &str,
537        relation: &crate::graph::RelationRef,
538        now: u64,
539    ) -> &mut RelationAnnotation {
540        let found = self
541            .relation_annotations
542            .iter()
543            .position(|annotation| annotation.owner == owner && &annotation.relation == relation);
544        let index = match found {
545            Some(index) => index,
546            None => {
547                self.relation_annotations.push(RelationAnnotation {
548                    owner: owner.to_string(),
549                    relation: relation.clone(),
550                    note: None,
551                    tags: BTreeSet::new(),
552                    created_at: now,
553                    updated_at: now,
554                });
555                self.relation_annotations.len() - 1
556            }
557        };
558        &mut self.relation_annotations[index]
559    }

561    /// Copies everything said about one target onto another.
562    ///
563    /// Returns `false` when there was nothing to copy — which is worth telling
564    /// the user, since a silent success there means they typed the wrong name
565    /// and will not find out until they look.
566    pub fn copy(&mut self, owner: &str, from: &EntityRef, to: &EntityRef, now: u64) -> bool {
567        let Some(source) = self.find(owner, from).cloned() else {
568            return false;
569        };
570        let target = self.entry(owner, to, now);
571        target.loved = source.loved;
572        target.rating = source.rating;
573        target.note = source.note.clone();
574        target.tags = source.tags;
575        target.updated_at = now;
576        true
577    }
578
579    /// Takes back the most recent play of a track, log and counter together.
580    ///
581    /// The counter is not a summary of the log — the log is bounded at
582    /// [`HISTORY_LIMIT`] and the counter is not — so a removal that touched one
583    /// and not the other would leave a track played "three times" with two
584    /// plays behind it, and nothing on screen to say which was right. They are
585    /// written together by [`UserData::record_play`] and they are taken back
586    /// together here.
587    ///
588    /// Returns `false` when there was nothing to take back, which the caller
589    /// says rather than reporting a removal that did not happen.
590    pub fn forget_last_play(&mut self, owner: &str, track: &EntityRef) -> bool {
591        let Some(index) = self
592            .plays
593            .iter()
594            .rposition(|p| p.owner == owner && &p.track == track)
595        else {
596            // The log may have rolled the play off its front while the counter
597            // still holds it. Decrementing on a play nobody can point at would
598            // be guessing, so it is refused.
599            return false;
600        };
601        self.plays.remove(index);
602        if let Some(counter) = self
603            .counts
604            .iter_mut()
605            .find(|c| c.owner == owner && &c.track == track)
606        {
607            counter.count = counter.count.saturating_sub(1);
608            // The last-played date now belongs to whatever play is newest of
609            // those left, and to nothing at all when none is.
610            counter.last_played = self
611                .plays
612                .iter()
613                .filter(|p| p.owner == owner && &p.track == track)
614                .map(|p| p.at)
615                .chain(
616                    self.scrobbles
617                        .iter()
618                        .filter(|p| p.owner == owner && &p.track == track)
619                        .map(|p| p.at_ms / 1000),
620                )
621                .max()
622                .unwrap_or(0);
623        }
624        self.counts.retain(|c| c.count > 0);
625        true
626    }

628    /// Forgets everything an owner ever played.
629    ///
630    /// Native events, declared scrobbles and counters are cleared together.
631    /// Returns how many events/declarations and how many counters went,
632    /// because "your history is cleared" is a claim nobody
633    /// can check and a number is.
634    pub fn forget_history(&mut self, owner: &str) -> (usize, usize) {
635        let plays = self.plays.iter().filter(|p| p.owner == owner).count()
636            + self.scrobbles.iter().filter(|p| p.owner == owner).count();
637        let counts = self.counts.iter().filter(|c| c.owner == owner).count();
638        self.plays.retain(|p| p.owner != owner);
639        self.scrobbles.retain(|p| p.owner != owner);
640        self.counts.retain(|c| c.owner != owner);
641        (plays, counts)
642    }
643
644    /// Records one listen: the bounded log and the all-time counter together.
645    ///
646    /// Both, always — see [`UserData::forget_last_play`] for why the two must
647    /// never be written apart.
648    pub fn record_play(&mut self, play: Play) {
649        match self
650            .counts
651            .iter_mut()
652            .find(|c| c.owner == play.owner && c.track == play.track)
653        {
654            Some(counter) => {
655                counter.count = counter.count.saturating_add(1);
656                counter.last_played = counter.last_played.max(play.at);
657            }
658            None => self.counts.push(PlayCount {
659                owner: play.owner.clone(),
660                track: play.track.clone(),
661                count: 1,
662                last_played: play.at,
663            }),
664        }
665        self.plays.push(play);
666        // Offline clients can submit older listens after newer ones. Stable
667        // ordering keeps equal-time events in arrival order without deduping.
668        self.plays.sort_by_key(|play| play.at);
669        // Oldest first, so the excess falls off the front.
670        if self.plays.len() > HISTORY_LIMIT {
671            let excess = self.plays.len() - HISTORY_LIMIT;
672            self.plays.drain(..excess);
673        }
674    }
675
676    /// A saved query by name, compared without regard to case or accents:
677    /// somebody typing `aede collection Metal` means the one they saved as
678    /// "metal".
679    pub fn collection(&self, owner: &str, name: &str) -> Option<&Collection> {
680        let wanted = text::normalize(name);
681        self.collections
682            .iter()
683            .find(|c| c.owner == owner && text::normalize(&c.name) == wanted)
684    }
685
686    /// Saves a query under a name, replacing one of the same name.
687    ///
688    /// Returns `true` when a collection was replaced, which the caller says out
689    /// loud: overwriting somebody's saved query in silence is how they find out
690    /// a week later.
691    pub fn save_collection(&mut self, owner: &str, name: &str, expression: &str, now: u64) -> bool {
692        let wanted = text::normalize(name);
693        if let Some(existing) = self
694            .collections
695            .iter_mut()
696            .find(|c| c.owner == owner && text::normalize(&c.name) == wanted)
697        {
698            existing.expression = expression.to_string();
699            existing.updated_at = now;
700            return true;
701        }
702        self.collections.push(Collection {
703            owner: owner.to_string(),
704            name: name.to_string(),
705            expression: expression.to_string(),
706            created_at: now,
707            updated_at: now,
708        });
709        false
710    }
711
712    /// Drops a saved query; `false` when there was none by that name.
713    pub fn forget_collection(&mut self, owner: &str, name: &str) -> bool {
714        let wanted = text::normalize(name);
715        let before = self.collections.len();
716        self.collections
717            .retain(|c| !(c.owner == owner && text::normalize(&c.name) == wanted));
718        self.collections.len() != before
719    }
720
721    /// How many times one person played a track, all-time.
722    pub fn play_count(&self, owner: &str, track: &EntityRef) -> u32 {
723        self.counts
724            .iter()
725            .find(|c| c.owner == owner && &c.track == track)
726            .map(|c| c.count)
727            .unwrap_or(0)
728    }
729}

933/// Reads the document back.
934///
935/// A legacy row whose target cannot be parsed is **skipped rather than fatal**: one
936/// corrupted line must not cost the user every note they ever wrote, which is
937/// the opposite of the catalog's rule, where a broken row means the graph does
938/// not hold together and refusing is the safe answer. Here there is no graph —
939/// only statements, each standing alone.
940/// Missing optional tables remain compatible with old stores. A present table
941/// of the wrong JSON type is refused before a later write can empty it.
942/// New static-playlist and scrobble rows are strictly validated: malformed or
943/// oversized rows fail the load, preserving the original file for recovery.
944pub fn from_json(value: &crate::json::Json) -> Result<UserData, crate::store::StoreError> {
945    use crate::store::StoreError;
946    let found = value.field_u32("format_version").unwrap_or(0);
947    if !matches!(found, 1 | USER_FORMAT_VERSION) {
948        return Err(StoreError::Version {
949            found,
950            expected: USER_FORMAT_VERSION,
951        });
952    }
953    for (table, message) in [
954        ("annotations", "user annotations must be an array"),
955        (
956            "relation_annotations",
957            "user relation annotations must be an array",
958        ),
959        ("plays", "user plays must be an array"),
960        ("counts", "user counts must be an array"),
961        ("collections", "user collections must be an array"),
962        ("playlists", "user playlists must be an array"),
963        ("scrobbles", "user scrobbles must be an array"),
964        ("playback_states", "user playback states must be an array"),
965        ("set_aside", "user set-aside decisions must be an array"),
966        ("same_artist", "user artist decisions must be an array"),
967        ("track_identities", "user track identities must be an array"),
968        ("relinks", "user relink history must be an array"),
969    ] {
970        if value
971            .get(table)
972            .is_some_and(|value| value.as_arr().is_none())
973        {
974            return Err(StoreError::Invalid(message));
975        }
976    }
977    let mut data = UserData {
978        playlists: playlists::read_playlists(value)?,
979        scrobbles: playlists::read_scrobbles(value)?,
980        playback_states: playback::read_states(value)?,
981        track_identities: value
982            .get("track_identities")
983            .and_then(crate::json::Json::as_arr)
984            .unwrap_or(&[])
985            .iter()
986            .filter_map(TrackIdentity::from_json)
987            .collect(),
988        relinks: value
989            .get("relinks")
990            .and_then(crate::json::Json::as_arr)
991            .unwrap_or(&[])
992            .iter()
993            .filter_map(Relink::from_json)
994            .collect(),
995        ..Default::default()
996    };
997
998    for row in value
999        .get("annotations")
1000        .and_then(|v| v.as_arr())
1001        .unwrap_or(&[])
1002    {
1003        let Some(target) = row
1004            .field_str("target")
1005            .and_then(|t| EntityRef::parse_token(&t))
1006        else {
1007            continue;
1008        };
1009        data.annotations.push(Annotation {
1010            owner: row.field_str("owner").unwrap_or_else(|| LOCAL_USER.into()),
1011            target,
1012            loved: row.field_bool("loved"),
1013            rating: row.field_u32("rating").map(|r| r.clamp(1, 5) as u8),
1014            note: row.field_str("note"),
1015            tags: row
1016                .get("tags")
1017                .and_then(|v| v.as_arr())
1018                .map(|list| list.iter().filter_map(|t| t.as_string()).collect())
1019                .unwrap_or_default(),
1020            created_at: row.field_u64("created_at").unwrap_or(0),
1021            updated_at: row.field_u64("updated_at").unwrap_or(0),
1022        });
1023    }
1024
1025    for row in value
1026        .get("relation_annotations")
1027        .and_then(|value| value.as_arr())
1028        .unwrap_or(&[])
1029    {
1030        let (Some(source), Some(kind), Some(target), Some(provenance)) = (
1031            row.field_str("source")
1032                .and_then(|value| EntityRef::parse_token(&value)),
1033            row.field_str("relation"),
1034            row.field_str("target")
1035                .and_then(|value| EntityRef::parse_token(&value)),
1036            row.field_str("provenance"),
1037        ) else {
1038            continue;
1039        };
1040        let annotation = RelationAnnotation {
1041            owner: row.field_str("owner").unwrap_or_else(|| LOCAL_USER.into()),
1042            relation: crate::graph::RelationRef {
1043                source,
1044                kind,
1045                target,
1046                provenance,
1047                source_id: row.field_str("source_id"),
1048            },
1049            note: row.field_str("note"),
1050            tags: row
1051                .get("tags")
1052                .and_then(|value| value.as_arr())
1053                .map(|values| {
1054                    values
1055                        .iter()
1056                        .filter_map(|value| value.as_string())
1057                        .collect()
1058                })
1059                .unwrap_or_default(),
1060            created_at: row.field_u64("created_at").unwrap_or(0),
1061            updated_at: row.field_u64("updated_at").unwrap_or(0),
1062        };
1063        if !annotation.is_empty() {
1064            data.relation_annotations.push(annotation);
1065        }
1066    }
1067
1068    for row in value.get("plays").and_then(|v| v.as_arr()).unwrap_or(&[]) {
1069        let Some(track) = row
1070            .field_str("track")
1071            .and_then(|t| EntityRef::parse_token(&t))
1072        else {
1073            continue;
1074        };
1075        data.plays.push(Play {
1076            owner: row.field_str("owner").unwrap_or_else(|| LOCAL_USER.into()),
1077            track,
1078            at: row.field_u64("at").unwrap_or(0),
1079            ms_played: row.field_u64("ms_played").unwrap_or(0),
1080            completed: row.field_bool("completed"),
1081        });
1082    }
1083
1084    for row in value.get("counts").and_then(|v| v.as_arr()).unwrap_or(&[]) {
1085        let Some(track) = row
1086            .field_str("track")
1087            .and_then(|t| EntityRef::parse_token(&t))
1088        else {
1089            continue;
1090        };
1091        data.counts.push(PlayCount {
1092            owner: row.field_str("owner").unwrap_or_else(|| LOCAL_USER.into()),
1093            track,
1094            count: row.field_u32("count").unwrap_or(0),
1095            last_played: row.field_u64("last_played").unwrap_or(0),
1096        });
1097    }
1098    for row in value
1099        .get("collections")
1100        .and_then(|v| v.as_arr())
1101        .unwrap_or(&[])
1102    {
1103        let (Some(name), Some(expression)) = (row.field_str("name"), row.field_str("query")) else {
1104            continue;
1105        };
1106        data.collections.push(Collection {
1107            owner: row.field_str("owner").unwrap_or_else(|| LOCAL_USER.into()),
1108            name,
1109            expression,
1110            created_at: row.field_u64("created_at").unwrap_or(0),
1111            updated_at: row.field_u64("updated_at").unwrap_or(0),
1112        });
1113    }
1114    for row in value
1115        .get("set_aside")
1116        .and_then(crate::json::Json::as_arr)
1117        .unwrap_or(&[])
1118    {
1119        // No identifier, no decision: the title alone cannot say which record
1120        // was meant, and a wish list quietly shortened by one is worse than
1121        // one item too long.
1122        let Some(release_group) = row.field_str("release_group") else {
1123            continue;
1124        };
1125        data.set_aside.push(SetAside {
1126            owner: row
1127                .field_str("owner")
1128                .unwrap_or_else(|| LOCAL_USER.to_string()),
1129            release_group,
1130            title: row.field_str("title").unwrap_or_default(),
1131            created_at: row.field_u64("created_at").unwrap_or(0),
1132        });
1133    }
1134    for row in value
1135        .get("same_artist")
1136        .and_then(crate::json::Json::as_arr)
1137        .unwrap_or(&[])
1138    {
1139        // Half a statement is not a statement: a row naming one spelling says
1140        // nothing about which artist it gives way to, and filing it under the
1141        // empty key would collect every broken row in the file under one
1142        // artist nobody named.
1143        let (Some(spelling), Some(filed_as)) =
1144            (row.field_str("spelling"), row.field_str("filed_as"))
1145        else {
1146            continue;
1147        };
1148        // Normalised on the way in as well as on the way out. A file edited by
1149        // hand — which is a thing this format invites, being readable — must
1150        // not be able to state a merge the program can never match.
1151        let (spelling, filed_as) = (text::normalize(&spelling), text::normalize(&filed_as));
1152        if spelling.is_empty() || filed_as.is_empty() || spelling == filed_as {
1153            continue;
1154        }
1155        data.same_artist.push(SameArtist {
1156            owner: row
1157                .field_str("owner")
1158                .unwrap_or_else(|| LOCAL_USER.to_string()),
1159            spelling,
1160            filed_as,
1161            created_at: row.field_u64("created_at").unwrap_or(0),
1162        });
1163    }
1164    // Older writers stored arrival order. Normalize in memory so readers and
1165    // undo operations use event time without changing counts or losing events.
1166    data.plays.sort_by_key(|play| play.at);
1167    Ok(data)
1168}

1170/// What a merge did, so that an import can say it rather than be trusted.
1171#[derive(Debug, Clone, Copy, Default, PartialEq)]
1172pub struct Merge {
1173    /// Records that did not exist here.
1174    pub added: usize,
1175    /// Records replaced because the incoming one was written later.
1176    pub updated: usize,
1177    /// Records left alone because what is here is newer.
1178    pub kept: usize,
1179    /// Native listening events or client declarations newly added to their logs.
1180    pub plays: usize,
1181    /// Saved queries taken in.
1182    pub collections: usize,
1183}
1184
1185/// Folds one set of user data into another.
1186///
1187/// **Merging, never replacing.** Someone restoring half a backup wants their
1188/// two halves, not the older one; and an import that emptied what was already
1189/// there would be the one operation in this program capable of losing
1190/// everything at once.
1191///
1192/// The rule is the same everywhere: the record written **last** wins, and the
1193/// one that lost is reported rather than dropped in silence. Play counters take
1194/// the larger of the two, since a count is a total and neither side ever
1195/// counted the other's listens.
1196/// Recent listening events are merged as a multiset of their complete values:
1197/// simultaneous identical events keep their multiplicity, and reimporting a
1198/// backup does not append them again. Relink history remains an audit trail;
1199/// importing it never replays reattachment or undo commands.
1200/// Native-player profiles are imported only when absent locally: an import
1201/// never rewinds an existing device checkpoint, even if its timestamp is older.
1202pub fn merge(into: &mut UserData, incoming: UserData) -> Merge {
1203    let mut report = Merge::default();
1204    playlists::merge(into, incoming.playlists, incoming.scrobbles, &mut report);
1205    playback::merge(into, incoming.playback_states, &mut report);
1206
1207    // These are evidence and an audit trail, not commands to replay. Preserve
1208    // local evidence on conflict; imported history never moves current data.
1209    for identity in incoming.track_identities {
1210        if !into
1211            .track_identities
1212            .iter()
1213            .any(|known| known.path == identity.path)
1214        {
1215            into.track_identities.push(identity);
1216        }
1217    }
1218    for mut event in incoming.relinks {
1219        let known = into.relinks.iter().any(|known| {
1220            known.origin_id == event.origin_id
1221                && known.owner == event.owner
1222                && known.from == event.from
1223                && known.to == event.to
1224                && known.at == event.at
1225                && known.before.annotations == event.before.annotations
1226                && known.before.plays == event.before.plays
1227                && known.before.counts == event.before.counts
1228                && known.before.relation_annotations == event.before.relation_annotations
1229                && known.before.playlists == event.before.playlists
1230                && known.before.scrobbles == event.before.scrobbles
1231        });
1232        if known {
1233            continue;
1234        }
1235        if into.relinks.iter().any(|known| known.id == event.id) {
1236            let Some(next) = into
1237                .relinks
1238                .iter()
1239                .map(|known| known.id)
1240                .max()
1241                .unwrap_or(0)
1242                .checked_add(1)
1243            else {
1244                continue;
1245            };
1246            event.id = next;
1247        }
1248        into.relinks.push(event);
1249    }
1250
1251    for annotation in incoming.annotations {
1252        match into
1253            .annotations
1254            .iter_mut()
1255            .find(|a| a.owner == annotation.owner && a.target == annotation.target)
1256        {
1257            Some(existing) if existing.updated_at >= annotation.updated_at => report.kept += 1,
1258            Some(existing) => {
1259                *existing = annotation;
1260                report.updated += 1;
1261            }
1262            None => {
1263                into.annotations.push(annotation);
1264                report.added += 1;
1265            }
1266        }
1267    }
1268
1269    for annotation in incoming.relation_annotations {
1270        match into.relation_annotations.iter_mut().find(|existing| {
1271            existing.owner == annotation.owner && existing.relation == annotation.relation
1272        }) {
1273            Some(existing) if existing.updated_at >= annotation.updated_at => report.kept += 1,
1274            Some(existing) => {
1275                *existing = annotation;
1276                report.updated += 1;
1277            }
1278            None => {
1279                into.relation_annotations.push(annotation);
1280                report.added += 1;
1281            }
1282        }
1283    }
1284
1285    // Equal-time listens can be distinct, including repeated identical events.
1286    // Consume only the multiplicity already present, so restoring a backup
1287    // keeps every listen while importing that same backup again is idempotent.
1288    let play_key = |play: &Play| {
1289        (
1290            play.owner.clone(),
1291            play.track.clone(),
1292            play.at,
1293            play.ms_played,
1294            play.completed,
1295        )
1296    };
1297    let mut available = BTreeMap::new();
1298    for play in &into.plays {
1299        *available.entry(play_key(play)).or_insert(0usize) += 1;
1300    }
1301    for play in incoming.plays {
1302        let copies = available.entry(play_key(&play)).or_default();
1303        if *copies > 0 {
1304            *copies -= 1;
1305        } else {
1306            into.plays.push(play);
1307            report.plays += 1;
1308        }
1309    }
1310    into.plays.sort_by_key(|p| p.at);
1311    if into.plays.len() > HISTORY_LIMIT {
1312        let excess = into.plays.len() - HISTORY_LIMIT;
1313        into.plays.drain(..excess);
1314    }
1315
1316    for count in incoming.counts {
1317        match into
1318            .counts
1319            .iter_mut()
1320            .find(|c| c.owner == count.owner && c.track == count.track)
1321        {
1322            Some(existing) => {
1323                existing.count = existing.count.max(count.count);
1324                existing.last_played = existing.last_played.max(count.last_played);
1325            }
1326            None => into.counts.push(count),
1327        }
1328    }
1329
1330    for collection in incoming.collections {
1331        let owner = collection.owner.clone();
1332        match into.collections.iter_mut().find(|c| {
1333            c.owner == owner && text::normalize(&c.name) == text::normalize(&collection.name)
1334        }) {
1335            Some(existing) if existing.updated_at >= collection.updated_at => report.kept += 1,
1336            Some(existing) => {
1337                *existing = collection;
1338                report.collections += 1;
1339            }
1340            None => {
1341                into.collections.push(collection);
1342                report.collections += 1;
1343            }
1344        }
1345    }
1346    // A decision has no versions to arbitrate: it was taken or it was not, so
1347    // importing the same backup twice must not file it twice.
1348    for aside in incoming.set_aside {
1349        let known = into
1350            .set_aside
1351            .iter()
1352            .any(|a| a.owner == aside.owner && a.release_group == aside.release_group);
1353        if !known {
1354            into.set_aside.push(aside);
1355            report.added += 1;
1356        }
1357    }
1358    // Same rule, and the same reason: a merge was stated or it was not.
1359    // Keyed on the spelling that gives way alone, not on the pair — two rows
1360    // filing `o osbourne` under two different artists is not a statement said
1361    // twice, it is a contradiction, and the one already on this disk wins
1362    // because taking it back is one command and finding it is not.
1363    for same in incoming.same_artist {
1364        let known = into
1365            .same_artist
1366            .iter()
1367            .any(|m| m.owner == same.owner && m.spelling == same.spelling);
1368        if !known {
1369            into.same_artist.push(same);
1370            report.added += 1;
1371        }
1372    }
1373    report
1374}
