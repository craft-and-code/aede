1//! M3U playlists: the one place that knows how to write one.
2//!
3//! Two callers with two purposes. `--m3u` hands whatever is on screen to a
4//! player, so it writes **absolute** paths to a file that may end up anywhere.
5//! `aede playlist` writes into the album folder itself, so it writes
6//! **relative** ones: a playlist beside its music travels with it, survives the
7//! folder being moved or copied to a card, and is the same file on another
8//! machine. One renderer for both, because two would agree about `#EXTINF`
9//! today and disagree about it in six months.
10
11use std::path::Path;
12
13use crate::model::{Catalog, EntityKind, Id};
14
15/// Which of the two M3U dialects to write.
16#[derive(Clone, Copy, PartialEq)]
17pub enum Style {
18    /// `#EXTM3U` with an `#EXTINF` line per track: duration and title, which
19    /// is what lets a player show a name rather than a file name.
20    Extended,
21    /// Paths and nothing else. Older hardware players — car head units, some
22    /// DAPs — stop at the first `#` they do not understand.
23    Simple,
24}
25
26/// Renders a playlist.
27///
28/// `base` is the folder the file will be written into: paths under it are
29/// written relative to it, anything else absolute. A playlist that silently
30/// dropped the tracks it could not make relative would be worse than one that
31/// names them the long way.
32pub fn render(catalog: &Catalog, tracks: &[Id], base: Option<&Path>, style: Style) -> String {
33    let mut out = String::new();
34    if style == Style::Extended {
35        out.push_str("#EXTM3U\n");
36    }
37    for &id in tracks {
38        let Some(track) = catalog.track(id) else {
39            continue;
40        };
41        let Some(file) = catalog.file(track.file_id) else {
42            continue;
43        };
44        if style == Style::Extended {
45            let artist = catalog
46                .credits_on(EntityKind::Track, id)
47                .into_iter()
48                .find(|(_, role)| *role == "main")
49                .map(|(a, _)| a.name.clone())
50                .unwrap_or_default();
51            // Seconds, rounded, and -1 when unknown: that is what the format
52            // says, and a player reading 0 would show a track of no length.
53            let seconds = match track.duration_ms {
54                Some(ms) => ((ms + 500) / 1000) as i64,
55                None => -1,
56            };
57            match artist.is_empty() {
58                true => out.push_str(&format!("#EXTINF:{seconds},{}\n", track.title)),
59                false => out.push_str(&format!("#EXTINF:{seconds},{artist} - {}\n", track.title)),
60            }
61        }
62        out.push_str(&relative_to(&file.path, base));
63        out.push('\n');
64    }
65    out
66}
67
68/// The path as the playlist should carry it.
69fn relative_to(path: &str, base: Option<&Path>) -> String {
70    let Some(base) = base.and_then(|b| b.to_str()) else {
71        return path.to_string();
72    };
73    crate::text::relative_under(path, base)
74        .filter(|relative| !relative.is_empty())
75        .unwrap_or_else(|| path.to_string())
76}

78/// The name of the playlist a folder should hold: the folder's own name.
79///
80/// Not the album title, which is tempting and wrong twice over: two folders can
81/// hold the same title (a rip and a remaster), and a title carries `/` and `:`
82/// on records that were named by people rather than by filesystems. The folder
83/// name is unique where the file goes, is already legal there, and is what the
84/// user recognises in a player's list.
85pub fn file_name(folder: &Path) -> String {
86    let stem = folder
87        .file_name()
88        .and_then(|n| n.to_str())
89        .unwrap_or("playlist");
90    format!("{stem}.m3u")
91}
92
93/// `true` when the file already holds exactly this, byte for byte.
94///
95/// Content rather than a modification date, because a playlist is derived from
96/// the *set* of tracks and not from their bytes: adding a track to an album
97/// changes what the playlist should say without touching any file the playlist
98/// already names. Comparing the text answers both questions at once, and
99/// leaves the file's date alone when nothing changed — which matters to
100/// whatever syncs the folder afterwards.
101pub fn already_says(path: &Path, content: &str) -> bool {
102    std::fs::read_to_string(path).is_ok_and(|held| held == content)
103}
