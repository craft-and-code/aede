1//! Persistent collection identities and compatibility with name-keyed stores.
2use super::*;
3use crate::{json::Json, store::StoreError};

5/// Recognize opaque random IDs and deterministic IDs assigned to legacy rows.
6pub fn valid_collection_id(id: &str) -> bool {
7    let digits = id
8        .strip_prefix("collection-")
9        .filter(|s| s.len() == 64)
10        .or_else(|| {
11            id.strip_prefix("collection-legacy-")
12                .filter(|s| s.len() == 32)
13        });
14    digits.is_some_and(|s| {
15        s.bytes()
16            .all(|b| b.is_ascii_digit() || (b'a'..=b'f').contains(&b))
17    })
18}

20impl UserData {
21    /// Resolve an immutable collection ID only within its owner's namespace.
22    pub fn collection_by_id(&self, owner: &str, id: &str) -> Option<&Collection> {
23        self.collections
24            .iter()
25            .find(|row| row.owner == owner && row.id == id)
26    }
27}

29pub(super) fn validate(data: &UserData) -> Result<(), StoreError> {
30    let mut ids = BTreeSet::new();
31    let mut names = BTreeSet::new();
32    for row in &data.collections {
33        if !valid_collection_id(&row.id)
34            || !ids.insert(&row.id)
35            || !names.insert((&row.owner, text::normalize(&row.name)))
36        {
37            return Err(StoreError::Invalid(
38                "invalid or duplicate collection identity",
39            ));
40        }
41    }
42    Ok(())
43}

45pub(super) fn read(root: &Json) -> Result<Vec<Collection>, StoreError> {
46    use md5::{Digest, Md5};
47    let mut data = UserData::default();
48    for row in root
49        .get("collections")
50        .and_then(Json::as_arr)
51        .unwrap_or(&[])
52    {
53        let (Some(name), Some(expression)) = (row.field_str("name"), row.field_str("query")) else {
54            return Err(StoreError::Invalid("invalid collection row"));
55        };
56        let owner = read_owner(row)?;
57        let created_at = row.field_u64("created_at").unwrap_or(0);
58        let id = if row.get("id").is_some() {
59            row.field_str("id")
60                .ok_or(StoreError::Invalid("invalid collection identity"))?
61        } else {
62            if root
63                .field_u32("format_version")
64                .is_some_and(|version| version >= 3)
65            {
66                return Err(StoreError::Invalid("missing collection identity"));
67            }
68            // Length-delimited JSON avoids concatenation ambiguities. This is an
69            // identity migration, never an authentication or integrity hash.
70            let key = serde_json::to_vec(&(&owner, &name, created_at))
71                .map_err(|_| StoreError::Invalid("invalid legacy collection"))?;
72            format!("collection-legacy-{:x}", Md5::digest(key))
73        };
74        data.collections.push(Collection {
75            id,
76            owner,
77            name,
78            expression,
79            created_at,
80            updated_at: row.field_u64("updated_at").unwrap_or(0),
81        });
82    }
83    validate(&data)?;
84    Ok(data.collections)
85}

87pub(super) fn merge(into: &mut UserData, incoming: Vec<Collection>, report: &mut Merge) {
88    for mut row in incoming {
89        // Existing local identity wins a name collision. A renamed backup row
90        // matches by ID first, but cannot overwrite another collection's name.
91        let same_id = into.collections.iter().position(|c| c.id == row.id);
92        let same_name = into.collections.iter().position(|c| {
93            c.owner == row.owner && text::normalize(&c.name) == text::normalize(&row.name)
94        });
95        if same_id.is_some_and(|i| into.collections[i].owner != row.owner)
96            || (same_id.is_some() && same_name.is_some() && same_id != same_name)
97        {
98            report.kept += 1;
99            continue;
100        }
101        match same_id.or(same_name) {
102            Some(i) if into.collections[i].updated_at >= row.updated_at => report.kept += 1,
103            Some(i) => {
104                row.id = into.collections[i].id.clone();
105                into.collections[i] = row;
106                report.collections += 1;
107            }
108            None => {
109                into.collections.push(row);
110                report.collections += 1;
111            }
112        }
113    }
114}
