1//! The one place that knows where ffmpeg is, and what to say when it is not.
2//!
3//! **ffmpeg is an external program, not a dependency.** Nothing is linked,
4//! nothing is vendored, and a checkout without ffmpeg installed builds and
5//! passes its tests — the features that need it say so and stop. Copying,
6//! spectrum rendering and playback share these install instructions: a fact
7//! hand-copied into several places will eventually be wrong in one of them.
8//!
9//! The search is deliberately plain — `ffmpeg`, on the `PATH`. A GUI
10//! application has to hunt through `/opt/homebrew/bin` and the rest because it
11//! is launched by Finder and inherits no shell environment; a command run from
12//! a terminal inherits the user's `PATH` by construction, and looking anywhere
13//! else would only find a *different* ffmpeg from the one they get when they
14//! type the name themselves. Playback uses it for formats that the native
15//! decoder cannot open and for exact Vorbis end trimming.
16
17use std::process::{Command, Stdio};
18
19/// Where ffmpeg is, or `None` when it is not installed.
20///
21/// Callers processing many files can retain this answer for their run.
22pub fn find() -> Option<String> {
23    let name = "ffmpeg";
24    Command::new(name)
25        .arg("-version")
26        .stdout(Stdio::null())
27        .stderr(Stdio::null())
28        .status()
29        .is_ok_and(|status| status.success())
30        .then(|| name.to_string())
31}
32
33/// What to tell somebody who has no ffmpeg, worded so they can act on it.
34///
35/// `what` names the feature that wanted it, because "ffmpeg was not found" on
36/// its own leaves the reader wondering what they have lost.
37pub fn missing(what: &str) -> String {
38    format!(
39        "\
40{what} needs ffmpeg, and it was not found.
41
42ffmpeg is an external program, not something Aède ships: it does the decoding
43and the drawing, Aède decides what to hand it. Everything else works without
44it.
45
46  macOS          brew install ffmpeg
47  Debian/Ubuntu  sudo apt install ffmpeg
48  Arch           sudo pacman -S ffmpeg
49  Fedora         sudo dnf install ffmpeg"
50    )
51}
