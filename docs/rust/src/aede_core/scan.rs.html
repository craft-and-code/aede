1//! Directory traversal and parallel file reading.
2//!
3//! The scan is **incremental**: a file whose path, size and modification date
4//! have not changed is not read again, its tags are taken from the previous
5//! catalog. On a library of 50,000 titles, that is the difference between
6//! several minutes and a few seconds.
7

19/// Image file names recognised as album cover art, in order of preference.
20const COVER_NAMES: &[&str] = &["cover", "folder", "front", "albumart", "album", "artwork"];
21const COVER_EXTENSIONS: &[&str] = &["jpg", "jpeg", "png", "webp", "bmp"];
22
23/// Knobs governing a traversal. [`Default`] gives the settings suited to a
24/// personal library: automatic parallelism, no symlink following, hidden
25/// entries left alone.
26#[derive(Debug, Clone)]
27pub struct ScanOptions {
28    /// Read every accessible file again while retaining the previous catalog
29    /// as evidence for inaccessible paths. This does not discard cached data
30    /// merely because a watched drive or subfolder cannot be read.
31    pub force_read: bool,
32    /// Number of reader threads. 0 = automatic detection.
33    pub threads: usize,
34    /// Follow symbolic links (with loop detection).
35    pub follow_symlinks: bool,
36    /// Skip files and folders starting with a dot.
37    pub skip_hidden: bool,
38    /// Folders never to walk into, canonical, whatever a root says.
39    ///
40    /// A music folder is rarely only music: `Audiobooks`, `Podcasts`,
41    /// `_incoming`, a `Samples` folder for a DAW. Without this the only way to
42    /// keep them out of the catalog is to reorganise the disk to suit the
43    /// program, which is the wrong way round.
44    ///
45    /// They live in the **catalog**, not in this run's options, because a
46    /// plain `aede scan` re-reads every watched root: an exclusion that had to
47    /// be retyped would be forgotten exactly when it mattered.
48    pub excluded: Vec<PathBuf>,
49    /// Spellings the owner of the disk has said are one artist.
50    ///
51    /// The half of artist identity no identifier can settle, and the reason it
52    /// travels here rather than in the catalog: it is **the user's**, it lives
53    /// in `user.json`, and this crate does not read that file — the caller
54    /// does, as it does for every other statement in it.
55    ///
56    /// It is applied when the graph is built, which is why a merge takes
57    /// effect on the next `aede scan` and not before: the spelling a track is
58    /// filed under is decided as the artist is interned, and there is no later
59    /// moment at which one row can become another without rebuilding
60    /// everything that points at it.
61    pub same_artist: Vec<crate::model::identity::Chosen>,
62}
63
64impl Default for ScanOptions {
65    fn default() -> Self {
66        ScanOptions {
67            force_read: false,
68            threads: 0,
69            follow_symlinks: false,
70            skip_hidden: true,
71            excluded: Vec::new(),
72            same_artist: Vec::new(),
73        }
74    }
75}

77/// Summary of a scan.
78#[derive(Debug, Default)]
79pub struct ScanReport {
80    /// Audio files spotted during traversal, after deduplication. Always the
81    /// sum of `read` and `reused`, failures included.
82    pub found: usize,
83    /// Files selected for a fresh read, including unsuccessful attempts.
84    pub read: usize,
85    /// Files taken unchanged from the previous catalog.
86    pub reused: usize,
87    /// Files present in the old catalog and gone since.
88    pub removed: usize,
89    /// Previous files retained because their file or directory was inaccessible.
90    /// These are separate from unchanged files counted in `reused`.
91    pub preserved: usize,
92    /// File paths newly entering the catalog, sorted by path.
93    pub added_paths: Vec<String>,
94    /// Existing file paths whose size or precise timestamp changed.
95    pub changed_paths: Vec<String>,
96    /// Previously catalogued file paths absent from accessible scanned folders.
97    pub removed_paths: Vec<String>,
98    /// Files or directories that could not be read, sorted by path.
99    /// Inaccessible paths retain their previous entries. Invalid audio content
100    /// excludes the file, with its parsing failure reported here. Non-UTF-8
101    /// entries are skipped and reported with escaped native path spelling.
102    pub failures: Vec<(String, String)>,
103    /// Analysis reports found inside the folders and taken in.
104    pub reports: usize,
105    /// Imported analyses that were waiting for a file and found it this time.
106    pub attached: usize,
107    /// Imported analyses the catalog holds once the scan is done, the ones
108    /// carried over from the previous catalog included.
109    pub analyses: usize,
110    /// Wall-clock time of the whole scan in milliseconds, traversal included.
111    pub elapsed_ms: u128,
112}
113
114/// Progress event, for display.
115#[derive(Debug, Clone, Copy)]
116pub enum Progress {
117    /// Directory traversal finished: number of audio files spotted.
118    Discovered(usize),
119    /// `n` files processed out of `total`.
120    Read {
121        /// Files pulled off the read queue so far, failures included.
122        done: usize,
123        /// Files needing a fresh read. On an incremental scan this stays well
124        /// below the count announced by [`Progress::Discovered`], since reused
125        /// files never enter the queue.
126        total: usize,
127    },
128}

130/// Scans the given folders and builds a catalog.
131///
132/// `previous` enables incremental reuse and retention of inaccessible paths.
133/// To reread accessible files while keeping that evidence, set
134/// [`ScanOptions::force_read`]. Passing `None` starts without any prior data.
135///
136/// Only ordinary files are opened as audio. Roots and directory entries are
137/// traversed in native path order so followed directory aliases choose the
138/// same spelling. A root that cannot be represented exactly as UTF-8 is an
139/// invalid input; unrepresentable entries are reported and skipped.
140///
141/// Fresh reads compare filesystem metadata before and after tag parsing. An
142/// observed rewrite is reported as temporary unavailability rather than
143/// cached or mistaken for permanent corruption. This is a filesystem snapshot
144/// check, not a content hash or a transaction with concurrent file writers.
145pub fn scan(
146    roots: &[PathBuf],
147    previous: Option<&Catalog>,
148    options: &ScanOptions,
149    mut on_progress: impl FnMut(Progress) + Send,
150) -> std::io::Result<(Catalog, ScanReport)> {
151    let started = Instant::now();
152    let mut report = ScanReport::default();
153    // The persistent path contract is UTF-8. Refuse an unrepresentable root
154    // before a lossy spelling could become watched or collide with another.
155    let mut roots_str: Vec<String> = roots
156        .iter()
157        .map(|path| {
158            path.to_str().map(str::to_owned).ok_or_else(|| {
159                std::io::Error::new(
160                    std::io::ErrorKind::InvalidInput,
161                    format!("{path:?}: path is not valid UTF-8"),
162                )
163            })
164        })
165        .collect::<std::io::Result<_>>()?;
166    roots_str.sort();
167    roots_str.dedup();
168    let mut traversal_roots: Vec<&PathBuf> = roots.iter().collect();
169    traversal_roots.sort();
170    traversal_roots.dedup();
171
172    // --- 1. Traversal ------------------------------------------------------
173    let mut walker = Walker::new(options);
174    for root in traversal_roots {
175        walker.walk(root)?;
176    }
177    let Walker {
178        mut audio_files,
179        reports: mut reports_found,
180        folder_covers,
181        sidecars,
182        failures: traversal_failures,
183        inaccessible,
184        ..
185    } = walker;
186    reports_found.sort();
187    audio_files.sort();
188    audio_files.dedup();
189    report.found = audio_files.len();
190    on_progress(Progress::Discovered(audio_files.len()));
191
192    // --- 2. What can be taken as is ----------------------------------------
193    let mut cache: HashMap<&str, &model::AudioFile> = HashMap::new();
194    if let Some(prev) = previous {
195        for file in &prev.files {
196            cache.insert(file.path.as_str(), file);
197        }
198    }
199
200    let mut to_read: Vec<PathBuf> = Vec::new();
201    let mut reused: Vec<ScannedFile> = Vec::new();
202    let mut fractions = BTreeMap::new();
203
204    for path in &audio_files {
205        let path_str = path.to_string_lossy().to_string();
206        let Ok(meta) = file_metadata(path, options.follow_symlinks) else {
207            to_read.push(path.clone());
208            continue;
209        };
210        let size = meta.len();
211        let mtime = mtime_seconds(&meta);
212        let fraction = mtime_subseconds(&meta);
213
214        match cache.get(path_str.as_str()) {
215            Some(old)
216                if !options.force_read
217                    && old.size == size
218                    && old.mtime == mtime
219                    && previous
220                        .and_then(|catalog| catalog.file_mtime_subseconds.get(&path_str))
221                        == Some(&fraction) =>
222            {
223                fractions.insert(path_str.clone(), fraction);
224                let mut tags = RawTags {
225                    fields: old.tags.clone(),
226                    properties: old.properties.clone(),
227                    has_embedded_art: old.has_embedded_art,
228                };
229                // Cover art may have appeared in the folder since.
230                tags.properties.container = old.properties.container.clone();
231                reused.push(ScannedFile {
232                    path: path_str,
233                    size,
234                    mtime,
235                    tags,
236                    folder_cover: cover_for(&folder_covers, path),
237                    // Read from the fresh walk rather than carried over: a
238                    // `.lrc` dropped beside a track nobody touched must attach
239                    // on the next scan, and the track's own bytes have not
240                    // changed to say so.
241                    sidecar: sidecar_for(&sidecars, path),
242                    // The file has not moved and has not changed, so what was
243                    // concluded about it still holds — both of the expensive
244                    // conclusions, the checksum verdict and the fingerprint.
245                    integrity: old.integrity.clone(),
246                    fingerprint: old.fingerprint.clone(),
247                });
248            }
249            _ => to_read.push(path.clone()),
250        }
251    }
252    report.reused = reused.len();
253    report.read = to_read.len();
254
255    // --- 3. Parallel reading ----------------------------------------------
256    let thread_count = resolve_threads(options.threads).min(to_read.len().max(1));
257    let queue = Mutex::new(to_read);
258    let results: Mutex<Vec<(ScannedFile, u32)>> = Mutex::new(Vec::new());
259    let failures: Mutex<Vec<(String, String)>> = Mutex::new(Vec::new());
260    let unavailable: Mutex<HashSet<String>> = Mutex::new(HashSet::new());
261    let done = AtomicUsize::new(0);
262    let total = report.read;
263
264    std::thread::scope(|scope| {
265        for _ in 0..thread_count {
266            scope.spawn(|| {
267                loop {
268                    let Some(path) = queue.lock().unwrap_or_else(|e| e.into_inner()).pop() else {
269                        break;
270                    };
271                    let path_str = path.to_string_lossy().to_string();
272                    let outcome = read_fresh(&path, options.follow_symlinks, tags::read);
273                    match outcome {
274                        Ok((metadata, tags)) => {
275                            let fraction = mtime_subseconds(&metadata);
276                            let file = ScannedFile {
277                                path: path_str,
278                                size: metadata.len(),
279                                mtime: mtime_seconds(&metadata),
280                                tags,
281                                folder_cover: cover_for(&folder_covers, &path),
282                                sidecar: sidecar_for(&sidecars, &path),
283                                // A file read again is a file that changed:
284                                // any earlier verdict is about other bytes,
285                                // and so is any earlier fingerprint.
286                                integrity: None,
287                                fingerprint: None,
288                            };
289                            results.lock().unwrap_or_else(|e| e.into_inner()).push((file, fraction));
290                        }
291                        Err(error) => {
292                            if matches!(&error, tags::TagError::Io(error) if error.kind() != std::io::ErrorKind::NotFound) {
293                                unavailable.lock().unwrap_or_else(|e| e.into_inner()).insert(path_str.clone());
294                            }
295                            failures
296                                .lock()
297                                .unwrap_or_else(|e| e.into_inner())
298                                .push((path_str, error.to_string()));
299                        }
300                    }
301                    done.fetch_add(1, Ordering::Relaxed);
302                }
303            });
304        }
305
306        // The main thread only reports progress.
307        let mut last = 0usize;
308        while done.load(Ordering::Relaxed) < total {
309            let current = done.load(Ordering::Relaxed);
310            if current != last {
311                last = current;
312                on_progress(Progress::Read {
313                    done: current,
314                    total,
315                });
316            }
317            std::thread::sleep(std::time::Duration::from_millis(60));
318        }
319    });
320
321    let mut scanned: Vec<ScannedFile> = results
322        .into_inner()
323        .unwrap_or_else(|e| e.into_inner())
324        .into_iter()
325        .map(|(file, fraction)| {
326            fractions.insert(file.path.clone(), fraction);
327            file
328        })
329        .collect();
330    report.failures = failures.into_inner().unwrap_or_else(|e| e.into_inner());
331    report.failures.extend(traversal_failures.iter().cloned());
332    report.failures.sort();
333    report.failures.dedup();
334    scanned.extend(reused);
335    // An unsuccessful walk cannot establish absence. Preserve only entries
336    // below the failed paths, never excluded folders or intentionally removed
337    // roots. An I/O failure while opening an audio file has the same meaning.
338    let unavailable = unavailable.into_inner().unwrap_or_else(|e| e.into_inner());
339    if let Some(previous) = previous {
340        let present: HashSet<&str> = scanned.iter().map(|file| file.path.as_str()).collect();
341        let retained: Vec<&model::AudioFile> = previous
342            .files
343            .iter()
344            .filter(|file| {
345                !present.contains(file.path.as_str())
346                    && !options
347                        .excluded
348                        .iter()
349                        .any(|folder| crate::text::is_under(&file.path, &folder.to_string_lossy()))
350                    && (Path::new(&file.path)
351                        .ancestors()
352                        .any(|ancestor| inaccessible.contains(ancestor))
353                        || unavailable.contains(&file.path))
354            })
355            .collect();
356        let covers: HashMap<model::Id, &str> = if retained.is_empty() {
357            HashMap::new()
358        } else {
359            previous
360                .tracks
361                .iter()
362                .filter_map(|track| {
363                    let release = previous.release(track.release_id?)?;
364                    Some((track.file_id, release.cover_path.as_deref()?))
365                })
366                .collect()
367        };
368        for file in retained {
369            if let Some(fraction) = previous.file_mtime_subseconds.get(&file.path) {
370                fractions.insert(file.path.clone(), *fraction);
371            }
372            scanned.push(cached_file(
373                file,
374                covers.get(&file.id).map(|path| (*path).to_owned()),
375            ));
376            report.preserved += 1;
377        }
378    }
379    on_progress(Progress::Read { done: total, total });
380
381    // --- 4. Building the graph ---------------------------------------------
382    let mut catalog = model::build(scanned, roots_str, now_seconds(), &options.same_artist);
383    catalog.file_mtime_subseconds = fractions;
384    let current_paths: HashSet<&str> = catalog
385        .files
386        .iter()
387        .map(|file| file.path.as_str())
388        .collect();
389    catalog
390        .file_mtime_subseconds
391        .retain(|path, _| current_paths.contains(path.as_str()));
392    record_changes(&catalog, previous, &mut report);
393
394    // Analyses are keyed by path, so they simply travel: nothing to remap, and
395    // nothing to lose. They are the one thing in a catalog that reading the
396    // files again cannot recompute.
397    //
398    // Exclusions travel for the same reason and it is the same rule — **a scan
399    // may not destroy what it cannot recompute**. They are typed by the user
400    // and derived from nothing, so a rebuild that dropped them would forget
401    // them on the very run they were meant to shape. That is not theory: the
402    // first version of this feature lost them exactly here, and the symptom
403    // was an exclusion that worked once and then vanished.
404    if let Some(previous) = previous {
405        catalog.analyses = previous.analyses.clone();
406        catalog.excluded = previous.excluded.clone();
407    }
408    // Reports lying in the library are taken in, so that analysing a folder and
409    // then scanning it works as well as the other way round.
410    report.reports = import_reports(&reports_found, &mut catalog);
411    // And whatever was waiting is given another chance now that the files are
412    // known — including records naming the same file by another route, which
413    // is what a report written against a symbolic link looks like.
414    report.attached = analysis::reconcile(&mut catalog);
415    report.analyses = catalog.analyses.len();
416
417    report.elapsed_ms = started.elapsed().as_millis();
418    Ok((catalog, report))
419}

421fn read_fresh(
422    path: &Path,
423    follow_symlinks: bool,
424    read_tags: impl FnOnce(&Path) -> Result<RawTags, tags::TagError>,
425) -> Result<(std::fs::Metadata, RawTags), tags::TagError> {
426    let before = file_metadata(path, follow_symlinks)?;
427    let tags = read_tags(path);
428    let after = file_metadata(path, follow_symlinks)?;
429    if !same_file_snapshot(&before, &after) {
430        return Err(std::io::Error::new(
431            std::io::ErrorKind::Interrupted,
432            "file changed while its tags were read; scan again once it is stable",
433        )
434        .into());
435    }
436    tags.map(|tags| (after, tags))
437}

439fn file_metadata(path: &Path, follow_symlinks: bool) -> std::io::Result<std::fs::Metadata> {
440    let metadata = if follow_symlinks {
441        std::fs::metadata(path)?
442    } else {
443        std::fs::symlink_metadata(path)?
444    };
445    if !metadata.is_file() {
446        return Err(std::io::Error::new(
447            std::io::ErrorKind::InvalidInput,
448            "audio path is no longer an ordinary file",
449        ));
450    }
451    Ok(metadata)
452}
453
454fn same_file_snapshot(before: &std::fs::Metadata, after: &std::fs::Metadata) -> bool {
455    if before.len() != after.len()
456        || before.modified().ok() != after.modified().ok()
457        || before.created().ok() != after.created().ok()
458    {
459        return false;
460    }
461    #[cfg(unix)]
462    {
463        use std::os::unix::fs::MetadataExt;
464        if before.dev() != after.dev() || before.ino() != after.ino() {
465            return false;
466        }
467    }
468    true
469}

489fn record_changes(catalog: &Catalog, previous: Option<&Catalog>, report: &mut ScanReport) {
490    let old: HashMap<&str, &model::AudioFile> = previous
491        .into_iter()
492        .flat_map(|catalog| &catalog.files)
493        .map(|file| (file.path.as_str(), file))
494        .collect();
495    let current: HashSet<&str> = catalog
496        .files
497        .iter()
498        .map(|file| file.path.as_str())
499        .collect();
500    for file in &catalog.files {
501        match old.get(file.path.as_str()) {
502            None => report.added_paths.push(file.path.clone()),
503            Some(old)
504                if old.size != file.size
505                    || old.mtime != file.mtime
506                    || previous
507                        .and_then(|catalog| catalog.file_mtime_subseconds.get(&file.path))
508                        != catalog.file_mtime_subseconds.get(&file.path) =>
509            {
510                report.changed_paths.push(file.path.clone());
511            }
512            _ => {}
513        }
514    }
515    report.removed_paths = old
516        .keys()
517        .filter(|path| !current.contains(**path))
518        .map(|path| (*path).to_string())
519        .collect();
520    report.removed_paths.sort();
521    report.removed = report.removed_paths.len();
522}

524/// Reads the reports found while walking, and merges what they hold.
525///
526/// Returns how many were actually reports. A file is only parsed once
527/// [`analysis::looks_like_a_report`] has recognised it, and one that turns out
528/// to be unreadable is passed over: a scan is not the place to fail over a
529/// sidecar file nobody asked about.
530///
531/// The records go through the same matching as `aede import` — the scan has no
532/// business being more trusting than the command whose whole job this is.
533fn import_reports(found: &[PathBuf], catalog: &mut Catalog) -> usize {
534    let mut count = 0;
535    let now = now_seconds();
536    for path in found {
537        let Ok(report) = analysis::read_report(path) else {
538            continue;
539        };
540        count += 1;
541        analysis::merge_into(catalog, report.files, now);
542    }
543    count
544}

546/// How many threads a run should use: what was asked for, or what the machine
547/// offers.
548///
549/// Shared with `spectrum`, which spawns one ffmpeg per track and has exactly
550/// the same question to answer. Two defaults that could disagree about what
551/// `--threads` with no value means is one too many.
552pub fn resolve_threads(requested: usize) -> usize {
553    if requested > 0 {
554        return requested;
555    }
556    std::thread::available_parallelism()
557        .map(|n| n.get())
558        .unwrap_or(4)
559}

561/// The `.lrc` beside this file, if the walk saw one.
562fn sidecar_for(sidecars: &HashSet<PathBuf>, file: &Path) -> Option<String> {
563    let expected = crate::lyrics::sidecar_of(file);
564    sidecars
565        .contains(&expected)
566        .then(|| expected.to_string_lossy().to_string())
567}

569fn cover_for(covers: &HashMap<PathBuf, PathBuf>, file: &Path) -> Option<String> {
570    let folder = file.parent()?;
571    covers.get(folder).map(|p| p.to_string_lossy().to_string())

578struct Walker<'a> {
579    options: &'a ScanOptions,
580    audio_files: Vec<PathBuf>,
581    /// Analysis reports met on the way, to be taken in at the end.
582    reports: Vec<PathBuf>,
583    /// Best cover art found per folder.
584    folder_covers: HashMap<PathBuf, PathBuf>,
585    sidecars: HashSet<PathBuf>,
586    /// Folders already visited, so as not to go round in circles on a
587    /// circular symbolic link.
588    visited: HashSet<PathBuf>,
589    /// Failed paths are evidence of an unavailable subtree, not its deletion.
590    failures: Vec<(String, String)>,
591    /// Native paths used for ancestor lookups. One failed subtree must not
592    /// trigger a scan of every recorded failure for every previous file.
593    inaccessible: HashSet<PathBuf>,
594}
595
596impl<'a> Walker<'a> {
597    fn new(options: &'a ScanOptions) -> Self {
598        Walker {
599            options,
600            audio_files: Vec::new(),
601            reports: Vec::new(),
602            folder_covers: HashMap::new(),
603            sidecars: HashSet::new(),
604            visited: HashSet::new(),
605            failures: Vec::new(),
606            inaccessible: HashSet::new(),
607        }
608    }
609
610    fn failure(&mut self, path: &Path, reason: impl Into<String>) {
611        let spelling = path
612            .to_str()
613            .map(str::to_owned)
614            .unwrap_or_else(|| format!("{path:?}"));
615        self.failures.push((spelling, reason.into()));
616        self.inaccessible.insert(path.to_path_buf());
617    }
618
619    /// `true` when a folder is one the user asked never to read, or sits
620    /// inside one.
621    fn is_excluded(&self, canonical: &Path) -> bool {
622        let path = canonical.to_string_lossy();
623        self.options
624            .excluded
625            .iter()
626            .any(|folder| crate::text::is_under(&path, &folder.to_string_lossy()))
627    }
628
629    fn walk(&mut self, dir: &Path) -> std::io::Result<()> {
630        let canonical = std::fs::canonicalize(dir).unwrap_or_else(|_| dir.to_path_buf());
631        // Tested on the canonical path, so a folder reached through a symbolic
632        // link is excluded too — the same reason every comparison against a
633        // stored path in this program is made on a resolved one.
634        if self.is_excluded(&canonical) {
635            return Ok(());
636        }
637        if !self.visited.insert(canonical) {
638            return Ok(()); // already seen: symlink loop
639        }
640
641        let entries = match std::fs::read_dir(dir) {
642            Ok(entries) => entries,
643            // An unreadable folder must not interrupt the whole scan.
644            Err(error) => {
645                self.failure(dir, error.to_string());
646                return Ok(());
647            }
648        };
649
650        let mut best_cover: Option<(usize, PathBuf)> = None;
651
652        // Resolve directory aliases in a stable order. ReadDir order is not
653        // promised by the filesystem and otherwise decides which native path
654        // survives when two links reach the same directory.
655        let mut paths = Vec::new();
656        for entry in entries {
657            let entry = match entry {
658                Ok(entry) => entry,
659                Err(error) => {
660                    self.failure(dir, error.to_string());
661                    continue;
662                }
663            };
664            paths.push((entry.path(), entry.file_type()));
665        }
666        paths.sort_by(|(left, _), (right, _)| left.cmp(right));
667        for (path, file_type) in paths {
668            let name = path
669                .file_name()
670                .map(|name| name.to_string_lossy())
671                .unwrap_or_default();
672            if self.options.skip_hidden && name.starts_with('.') {
673                continue;
674            }
675            if path.to_str().is_none() {
676                self.failure(
677                    &path,
678                    "path is not valid UTF-8 and cannot be stored without losing its identity",
679                );
680                continue;
681            }
682            let file_type = match file_type {
683                Ok(file_type) => file_type,
684                Err(error) => {
685                    self.failure(&path, error.to_string());
686                    continue;
687                }
688            };
689
690            if file_type.is_symlink() && !self.options.follow_symlinks {
691                continue;
692            }
693            let file_type = if file_type.is_symlink() {
694                match std::fs::metadata(&path) {
695                    Ok(metadata) => metadata.file_type(),
696                    Err(error) => {
697                        self.failure(&path, error.to_string());
698                        continue;
699                    }
700                }
701            } else {
702                file_type
703            };
704
705            if file_type.is_dir() {
706                self.walk(&path)?;
707            } else if !file_type.is_file() {
708                // Opening a FIFO may wait forever; devices and sockets are
709                // not library audio regardless of their filename extension.
710                continue;
711            } else if tags::is_audio_path(&path) {
712                self.audio_files.push(path);
713            } else if name.to_ascii_lowercase().ends_with(".json")
714                && analysis::looks_like_a_report(&path)
715            {
716                // Someone may analyse a folder before ever scanning it, and
717                // leave the report sitting in it. Picking it up here is what
718                // makes the order of the two operations irrelevant.
719                self.reports.push(path);
720            } else if crate::lyrics::is_sidecar(&name) {
721                // Noted while the folder is open rather than looked for later:
722                // the walk already has the names in hand, and asking the
723                // filesystem again once per track would be ten thousand
724                // questions with the answers already on the table.
725                self.sidecars.insert(path);
726            } else if let Some(rank) = cover_rank(&name)
727                && best_cover
728                    .as_ref()
729                    .is_none_or(|(seen_rank, seen_path)| (rank, &path) < (*seen_rank, seen_path))
730            {
731                best_cover = Some((rank, path));
732            }
733        }
734
735        if let Some((_, cover)) = best_cover {
736            self.folder_covers.insert(dir.to_path_buf(), cover);
737        }
738        Ok(())
739    }
740}

742/// The image already serving as cover art in a folder, if there is one.
743///
744/// Reads the folder rather than the catalog, and shares its list of cover names with
745/// the scanner so that "does this album have a cover" has one answer in this
746/// program rather than two that drift. Used before anything writes an image
747/// beside music: the catalog is a snapshot and the disk is not.
748pub fn cover_in(folder: &std::path::Path) -> Option<std::path::PathBuf> {
749    let mut best: Option<(usize, std::path::PathBuf)> = None;
750    for entry in std::fs::read_dir(folder).ok()?.flatten() {
751        let path = entry.path();
752        if !path.is_file() {
753            continue;
754        }
755        let Some(name) = path.file_name().and_then(|n| n.to_str()) else {
756            continue;
757        };
758        if let Some(rank) = cover_rank(name)
759            && best
760                .as_ref()
761                .is_none_or(|(seen_rank, seen_path)| (rank, &path) < (*seen_rank, seen_path))
762        {
763            best = Some((rank, path));
764        }
765    }
766    best.map(|(_, path)| path)
767}
768
769fn cover_rank(name: &str) -> Option<usize> {
770    let lower = name.to_ascii_lowercase();
771    let (stem, ext) = lower.rsplit_once('.')?;
772    if !COVER_EXTENSIONS.contains(&ext) {
773        return None;
774    }
775    for (rank, candidate) in COVER_NAMES.iter().enumerate() {
776        if stem == *candidate || stem.starts_with(candidate) {
777            return Some(rank);
778        }
779    }
780    // Any image is still a candidate, but of last rank.
781    Some(COVER_NAMES.len())
782}
