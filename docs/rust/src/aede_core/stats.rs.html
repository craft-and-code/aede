2//!
3//! Everything is computed on demand from the in-memory catalog: no aggregate
4//! value is stored, so none can drift out of sync.

10/// Named breakdown entry: `(label, count, size in bytes)`.
11#[derive(Debug, Clone, PartialEq)]
12pub struct Bucket {
13    /// Name of the slice as it is shown, already formatted (`FLAC`, `44.1 kHz`, `1970`).
14    pub label: String,
15    /// How many items fall into this slice.
16    pub count: usize,
17    /// Disk space these items occupy, so a breakdown can be read by weight and not only by
18    /// number.
19    pub bytes: u64,
20}
21
22/// Snapshot of what the catalog contains, recomputed on each call.
23#[derive(Debug, Clone, Default)]
24pub struct Stats {
25    /// Audio files seen on disk, whether or not they could be attached to anything.
26    pub files: usize,
27    /// Recordings held by the catalog, one per file.
28    pub tracks: usize,
29    /// Albums, EPs and singles reconstructed from the tags.
30    pub releases: usize,
31    /// Releases gathering several artists rather than one, and so excluded from a discography.
32    pub compilations: usize,
33    /// Distinct artists, all roles taken together.
34    pub artists: usize,
35    /// Artists carrying at least one album.
36    pub album_artists: usize,
37    /// Distinct record labels credited across the releases.
38    pub labels: usize,
39    /// Distinct genres in use, after the tags have been normalized.
40    pub genres: usize,
41    /// Cumulative playing time of the library, in milliseconds.
42    pub total_duration_ms: u64,
43    /// Space the audio files take up on disk, in bytes; cover art is not counted.
44    pub total_bytes: u64,
45    /// Tracks with no identified album.
46    pub orphan_tracks: usize,
47
48    /// Files grouped by container format, the untagged ones landing under `unknown`.
49    pub by_codec: Vec<Bucket>,
50    /// Files grouped by perceived quality, following [`QualityTier`].
51    pub by_quality: Vec<Bucket>,
52    /// Files grouped by sampling frequency; files that declare none are left out.
53    pub by_sample_rate: Vec<Bucket>,
54    /// Releases grouped by decade of publication, undated ones gathered under a separate label.
55    pub by_decade: Vec<Bucket>,
56
57    /// Share of releases having cover art (embedded or in the folder).
58    ///
59    /// This field and the three that follow are proportions between 0 and 1, not percentages;
60    /// the display layer decides how to present them.
61    pub cover_ratio: f64,
62    /// Share of tracks carrying a year.
63    pub year_ratio: f64,
64    /// Share of tracks carrying at least one genre.
65    pub genre_ratio: f64,
66    /// Share of tracks already identified by a MusicBrainz MBID.
67    pub mbid_ratio: f64,
68}

70/// Quality class, as a listener perceives it.
71#[derive(Debug, Clone, Copy, PartialEq, Eq)]
72pub enum QualityTier {
73    /// Lossless beyond CD: more than 48 kHz or more than 16 bits.
74    HiRes,
75    /// Lossless, CD quality or below.
76    Lossless,
77    /// Lossy, 256 kbps or more.
78    LossyHigh,
79    /// Lossy, below 256 kbps.
80    LossyLow,
81    /// Nothing reliable to go on: unreadable file, or a bitrate the decoder never reported.
82    Unknown,
83}
84
85impl QualityTier {
86    /// Wording of the tier for a listing or a breakdown, in a form the reader can act on.
87    pub fn label(self) -> &'static str {
88        match self {
89            QualityTier::HiRes => "Hi-res",
90            QualityTier::Lossless => "Lossless (CD)",
91            QualityTier::LossyHigh => "Lossy (>= 256 kbps)",
92            QualityTier::LossyLow => "Lossy (< 256 kbps)",
93            QualityTier::Unknown => "Unknown",
94        }
95    }
96}
97
98/// Ranks a file from what its audio properties actually establish.
99///
100/// A file that names no codec, or a lossy one that declares no bitrate, stays `Unknown`: better
101/// an admitted gap than a tier guessed from the extension alone.
102pub fn quality_tier(props: &crate::tags::AudioProperties) -> QualityTier {
103    if props.codec.is_empty() {
104        return QualityTier::Unknown;
105    }
106    if props.lossless {
107        if props.is_hi_res() {
108            QualityTier::HiRes
109        } else {
110            QualityTier::Lossless
111        }
112    } else {
113        match props.bitrate_kbps {
114            Some(kbps) if kbps >= 256 => QualityTier::LossyHigh,
115            Some(_) => QualityTier::LossyLow,
116            None => QualityTier::Unknown,
117        }
118    }
119}

121/// Runs through the whole catalog and produces the counts, breakdowns and completeness ratios.
122///
123/// The cost grows with the size of the library; call it when the catalog has settled rather than
124/// after each change.
125pub fn compute(catalog: &Catalog) -> Stats {
126    let mut stats = Stats {
127        files: catalog.files.len(),
128        tracks: catalog.tracks.len(),
129        releases: catalog.releases.len(),
130        compilations: catalog.releases.iter().filter(|r| r.is_compilation).count(),
131        artists: catalog.artists.len(),
132        labels: catalog.labels.len(),
133        genres: catalog.genres.len(),
134        total_duration_ms: catalog.total_duration_ms(),
135        total_bytes: catalog.total_size(),
136        orphan_tracks: catalog
137            .tracks
138            .iter()
139            .filter(|t| t.release_id.is_none())
140            .count(),
141        ..Default::default()
142    };
143
144    stats.album_artists = catalog
145        .releases
146        .iter()
147        .filter_map(|r| r.album_artist_id)
148        .collect::<std::collections::BTreeSet<_>>()
149        .len();
150
151    // --- Per-file breakdowns ----------------------------------------------
152    let mut codecs: BTreeMap<String, (usize, u64)> = BTreeMap::new();
153    let mut qualities: BTreeMap<&'static str, (usize, u64)> = BTreeMap::new();
154    let mut rates: BTreeMap<u32, (usize, u64)> = BTreeMap::new();
155
156    for file in &catalog.files {
157        let codec = if file.properties.codec.is_empty() {
158            "unknown".to_string()
159        } else {
160            file.properties.codec.to_uppercase()
161        };
162        let entry = codecs.entry(codec).or_default();
163        entry.0 += 1;
164        entry.1 += file.size;
165
166        let tier = quality_tier(&file.properties).label();
167        let entry = qualities.entry(tier).or_default();
168        entry.0 += 1;
169        entry.1 += file.size;
170
171        if let Some(rate) = file.properties.sample_rate {
172            let entry = rates.entry(rate).or_default();
173            entry.0 += 1;
174            entry.1 += file.size;
175        }
176    }
177
178    stats.by_codec = sorted_buckets(codecs.into_iter());
179    stats.by_quality = sorted_buckets(qualities.into_iter().map(|(k, v)| (k.to_string(), v)));
180    stats.by_sample_rate = rates
181        .into_iter()
182        .map(|(rate, (count, bytes))| Bucket {
183            label: format_rate(rate),
184            count,
185            bytes,
186        })
187        .collect();
188
189    // --- Breakdown by decade ----------------------------------------------
190    let mut decades: BTreeMap<String, (usize, u64)> = BTreeMap::new();
191    for release in &catalog.releases {
192        let label = match release.year {
193            Some(year) => format!("{}", (year / 10) * 10),
194            None => "No year".to_string(),
195        };
196        let bytes: u64 = release
197            .track_ids
198            .iter()
199            .filter_map(|&id| catalog.track(id))
200            .filter_map(|t| catalog.file(t.file_id))
201            .map(|f| f.size)
202            .sum();
203        let entry = decades.entry(label).or_default();
204        entry.0 += 1;
205        entry.1 += bytes;
206    }
207    stats.by_decade = decades
208        .into_iter()
209        .map(|(label, (count, bytes))| Bucket {
210            label,
211            count,
212            bytes,
213        })
214        .collect();
215
216    // --- Completeness rates -----------------------------------------------
217    let with_cover = catalog
218        .releases
219        .iter()
220        .filter(|r| {
221            r.cover_path.is_some()
222                || r.track_ids
223                    .iter()
224                    .filter_map(|&id| catalog.track(id))
225                    .filter_map(|t| catalog.file(t.file_id))
226                    .any(|f| f.has_embedded_art)
227        })
228        .count();
229    stats.cover_ratio = ratio(with_cover, catalog.releases.len());
230
231    let with_year = catalog
232        .tracks
233        .iter()
234        .filter(|t| {
235            t.release_id
236                .and_then(|id| catalog.release(id))
237                .map(|r| r.year.is_some())
238                .unwrap_or(false)
239        })
240        .count();
241    stats.year_ratio = ratio(with_year, catalog.tracks.len());
242
243    let with_genre = catalog
244        .genre_links
245        .iter()
246        .filter(|g| g.entity_kind == EntityKind::Track)
247        .map(|g| g.entity_id)
248        .collect::<std::collections::BTreeSet<Id>>()
249        .len();
250    stats.genre_ratio = ratio(with_genre, catalog.tracks.len());
251
252    let with_mbid = catalog.tracks.iter().filter(|t| t.mbid.is_some()).count();
253    stats.mbid_ratio = ratio(with_mbid, catalog.tracks.len());
254
255    stats
256}

258/// The most present artists, by number of distinct tracks they are audible on.
259///
260/// Writing and production credits are excluded on purpose: a songwriter who
261/// never performs would otherwise outrank the singers of their own catalog.
262/// Those credits remain visible on the artist page and in [`top_writers`].
263///
264/// We count tracks, not credits: an artist who is both performer and composer
265/// of a piece must not count it twice.
266pub fn top_artists(catalog: &Catalog, limit: usize) -> Vec<(Id, usize)> {
267    top_by_role(catalog, limit, true)
268}
269
270/// The most credited writers and producers, by number of distinct tracks.
271pub fn top_writers(catalog: &Catalog, limit: usize) -> Vec<(Id, usize)> {
272    top_by_role(catalog, limit, false)
273}

275fn top_by_role(catalog: &Catalog, limit: usize, performing: bool) -> Vec<(Id, usize)> {
276    let mut sets: BTreeMap<Id, std::collections::BTreeSet<Id>> = BTreeMap::new();
277    for credit in &catalog.credits {
278        if credit.entity_kind == EntityKind::Track
279            && model::is_performing_role(&credit.role) == performing
280        {
281            sets.entry(credit.artist_id)
282                .or_default()
283                .insert(credit.entity_id);
284        }
285    }
286    let counts = sets.into_iter().map(|(id, set)| (id, set.len())).collect();
287    top(counts, limit)
288}

290/// The most represented genres, by number of tracks.
291pub fn top_genres(catalog: &Catalog, limit: usize) -> Vec<(Id, usize)> {
292    let mut sets: BTreeMap<Id, std::collections::BTreeSet<Id>> = BTreeMap::new();
293    for link in &catalog.genre_links {
294        if link.entity_kind == EntityKind::Track {
295            sets.entry(link.genre_id)
296                .or_default()
297                .insert(link.entity_id);
298        }
299    }
300    top(
301        sets.into_iter().map(|(id, set)| (id, set.len())).collect(),
302        limit,
303    )
304}

306/// The most present labels, by number of releases.
307pub fn top_labels(catalog: &Catalog, limit: usize) -> Vec<(Id, usize)> {
308    let mut counts: BTreeMap<Id, usize> = BTreeMap::new();
309    for release in &catalog.releases {
310        for &label_id in &release.label_ids {
311            *counts.entry(label_id).or_insert(0) += 1;
312        }
313    }
314    top(counts, limit)
315}
316
317fn top(counts: BTreeMap<Id, usize>, limit: usize) -> Vec<(Id, usize)> {
318    let mut list: Vec<(Id, usize)> = counts.into_iter().collect();
319    // On a tie, the smallest identifier comes first: stable result.
320    list.sort_by(|a, b| b.1.cmp(&a.1).then_with(|| a.0.cmp(&b.0)));
321    list.truncate(limit);
322    list
323}
324
325fn sorted_buckets(items: impl Iterator<Item = (String, (usize, u64))>) -> Vec<Bucket> {
326    let mut list: Vec<Bucket> = items
327        .map(|(label, (count, bytes))| Bucket {
328            label,
329            count,
330            bytes,
331        })
332        .collect();
333    list.sort_by(|a, b| b.count.cmp(&a.count).then_with(|| a.label.cmp(&b.label)));
334    list
335}

337fn ratio(part: usize, total: usize) -> f64 {
338    if total == 0 {
339        0.0
340    } else {
341        part as f64 / total as f64
342    }
343}
344
345fn format_rate(rate: u32) -> String {
346    let khz = rate as f64 / 1000.0;
347    if khz.fract() == 0.0 {
348        format!("{} kHz", khz as u32)
349    } else {
350        format!("{khz:.1} kHz")
351    }
352}
