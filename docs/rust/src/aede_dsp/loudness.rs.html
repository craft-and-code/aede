1//! Gated programme loudness and true peak measurements over decoded PCM.
2//!
3//! This layer knows only sample format and samples. Decoding, file identity,
4//! metadata precedence, and persistence belong to the playback core.
5
6use std::fmt;
7
8use ebur128::{EbuR128, Mode};
9
10use crate::PcmFormat;
11use crate::peak_tail::PeakTail;
12
13/// Integrated programme loudness and peak of the original PCM.
14#[derive(Clone, Copy, Debug, PartialEq)]
15pub struct Measurement {
16    /// Gated programme loudness in LUFS.
17    pub integrated_lufs: f32,
18    /// Linear true peak, when every programme section has an oversampled
19    /// estimate; may exceed one. Unavailable at 192 kHz and above.
20    pub true_peak: Option<f32>,
21}
22
23/// Invalid sample layout or a measurement-library failure.
24#[derive(Debug)]
25pub enum LoudnessError {
26    /// Speaker positions are unknown for more than two channels.
27    UnsupportedLayout,
28    /// An input block is not a complete set of finite frames.
29    InvalidBuffer,
30    /// The underlying EBU R128 meter refused the operation.
31    Meter(ebur128::Error),
32}
33
34impl fmt::Display for LoudnessError {
35    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
36        match self {
37            Self::UnsupportedLayout => f.write_str("loudness requires mono or stereo PCM"),
38            Self::InvalidBuffer => f.write_str("loudness requires complete finite PCM frames"),
39            Self::Meter(error) => write!(f, "loudness meter: {error}"),
40        }
41    }
42}
43
44impl std::error::Error for LoudnessError {
45    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
46        match self {
47            Self::Meter(error) => Some(error),
48            _ => None,
49        }
50    }
51}
52
53impl From<ebur128::Error> for LoudnessError {
54    fn from(error: ebur128::Error) -> Self {
55        Self::Meter(error)
56    }
57}
58
59/// Streaming meter for an ordered programme, including format changes.
60///
61/// Consecutive blocks of the same format share one meter. At a format change,
62/// a new meter starts and the final gated result combines both histories.
63/// This avoids averaging per-track LUFS; a format boundary does not contribute
64/// windows that cross that boundary.
65#[derive(Default)]
66pub struct LoudnessProgramme {
67    sections: Vec<Section>,
68}
69
70struct Section {
71    meter: EbuR128,
72    peak_tail: PeakTail,
73}
74
75impl LoudnessProgramme {
76    /// Create an empty programme meter.
77    pub fn new() -> Self {
78        Self::default()
79    }
80
81    /// Add finite, interleaved PCM without changing the caller's samples.
82    pub fn push(&mut self, format: PcmFormat, samples: &[f32]) -> Result<(), LoudnessError> {
83        let channels = usize::from(format.channels());
84        if !matches!(format.channels(), 1 | 2) {
85            return Err(LoudnessError::UnsupportedLayout);
86        }
87        if samples.is_empty()
88            || !samples.len().is_multiple_of(channels)
89            || samples.iter().any(|sample| !sample.is_finite())
90        {
91            return Err(LoudnessError::InvalidBuffer);
92        }
93        let reuse = self.sections.last().is_some_and(|section| {
94            section.meter.rate() == format.sample_rate()
95                && section.meter.channels() == u32::from(format.channels())
96        });
97        if !reuse {
98            self.sections.push(Section {
99                meter: EbuR128::new(
100                    u32::from(format.channels()),
101                    format.sample_rate(),
102                    Mode::I | Mode::TRUE_PEAK,
103                )?,
104                peak_tail: PeakTail::new(format),
105            });
106        }
107        if let Some(section) = self.sections.last_mut() {
108            section.meter.add_frames_f32(samples)?;
109            section.peak_tail.push(samples);
110        }
111        Ok(())
112    }
113
114    /// Measure the finite programme; silence and streams under one gate have no LUFS.
115    ///
116    /// Resolve delayed true-peak outputs with a separate zero-extended tail.
117    /// This does not add silent loudness frames or modify subsequent capture.
118    pub fn measurement(&self) -> Result<Option<Measurement>, LoudnessError> {
119        if self.sections.is_empty() {
120            return Ok(None);
121        }
122        let loudness =
123            EbuR128::loudness_global_multiple(self.sections.iter().map(|section| &section.meter))?;
124        if !loudness.is_finite() || !(-100.0..=20.0).contains(&loudness) {
125            return Ok(None);
126        }
127        // ebur128 does not oversample at 192 kHz or above. A sample peak
128        // from even one section cannot establish the programme's true peak.
129        let true_peak = if self
130            .sections
131            .iter()
132            .all(|section| section.meter.rate() < 192_000)
133        {
134            let mut peak = 0.0f32;
135            for section in &self.sections {
136                if let Some(section_peak) = section.peak_tail.peak(&section.meter)? {
137                    peak = peak.max(section_peak);
138                }
139            }
140            if !peak.is_finite() {
141                return Ok(None);
142            }
143            Some(peak)
144        } else {
145            None
146        };
147        Ok(Some(Measurement {
148            integrated_lufs: loudness as f32,
149            true_peak,
150        }))
151    }
152}
