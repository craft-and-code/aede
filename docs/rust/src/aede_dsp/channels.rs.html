1//! Explicit speaker positions and conservative multichannel downmix.
2
3use crate::DspError;

5/// Speaker positions in the interleaved order defined by the channel mask.
6///
7/// The low bits follow WAVEFORMATEXTENSIBLE and Symphonia. An unknown layout
8/// retains its channel count but cannot be downmixed by guessing positions.
9#[derive(Clone, Copy, Debug, Eq, PartialEq)]
10pub enum ChannelLayout {
11    /// A channel count without trustworthy speaker positions.
12    Unknown(u16),
13    /// Speaker positions in ascending WAVEFORMATEXTENSIBLE mask-bit order.
14    /// Unrecognized nonzero masks can be retained without implying downmix support.
15    Mask(u32),
16}
17
18impl ChannelLayout {
19    /// Canonical one-channel layout used for decoded mono.
20    pub const MONO: Self = Self::Mask(0x1);
21    /// Front left followed by front right.
22    pub const STEREO: Self = Self::Mask(0x3);
23
24    /// Retain a nonempty speaker mask; downmix support is checked separately.
25    pub fn from_mask(mask: u32) -> Result<Self, DspError> {
26        if mask == 0 {
27            return Err(DspError::InvalidLayout);
28        }
29        Ok(Self::Mask(mask))
30    }
31
32    /// Number of interleaved samples in each complete PCM frame.
33    pub fn channels(self) -> u16 {
34        match self {
35            Self::Unknown(channels) => channels,
36            Self::Mask(mask) => mask.count_ones() as u16,
37        }
38    }
39
40    /// The retained positions, or `None` for a count-only layout.
41    pub fn mask(self) -> Option<u32> {
42        match self {
43            Self::Unknown(_) => None,
44            Self::Mask(mask) => Some(mask),
45        }
46    }
47
48    /// A diagnostic label for recognized positions, with an explicit unknown fallback.
49    pub fn name(self) -> &'static str {
50        match self {
51            Self::Mask(0x1) => "mono",
52            Self::Mask(0x3) => "stereo (FL FR)",
53            Self::Mask(0xb) => "2.1 (FL FR LFE)",
54            Self::Mask(0x7) => "3.0 (FL FR FC)",
55            Self::Mask(0xf) => "3.1 (FL FR FC LFE)",
56            Self::Mask(0x33) => "quad (FL FR BL BR)",
57            Self::Mask(0x603) => "quad side (FL FR SL SR)",
58            Self::Mask(0x107) => "4.0 (FL FR FC BC)",
59            Self::Mask(0x37) => "5.0 (FL FR FC BL BR)",
60            Self::Mask(0x3f) => "5.1 (FL FR FC LFE BL BR)",
61            Self::Mask(0x607) => "5.0 side (FL FR FC SL SR)",
62            Self::Mask(0x60f) => "5.1 side (FL FR FC LFE SL SR)",
63            Self::Mask(0x637) => "7.0 (FL FR FC BL BR SL SR)",
64            Self::Mask(0x63f) => "7.1 (FL FR FC LFE BL BR SL SR)",
65            _ => "unrecognized channel layout",
66        }
67    }
68
69    /// Explicit FFmpeg raw-PCM layout name; unknown masks must not be guessed.
70    pub fn ffmpeg_name(self) -> Option<&'static str> {
71        match self {
72            Self::Mask(0x1) => Some("mono"),
73            Self::Mask(0x3) => Some("stereo"),
74            Self::Mask(0xb) => Some("2.1"),
75            Self::Mask(0x7) => Some("3.0"),
76            Self::Mask(0xf) => Some("3.1"),
77            Self::Mask(0x33) => Some("quad"),
78            Self::Mask(0x603) => Some("quad(side)"),
79            Self::Mask(0x107) => Some("4.0"),
80            Self::Mask(0x37) => Some("5.0"),
81            Self::Mask(0x3f) => Some("5.1"),
82            Self::Mask(0x607) => Some("5.0(side)"),
83            Self::Mask(0x60f) => Some("5.1(side)"),
84            Self::Mask(0x637) => Some("7.0"),
85            Self::Mask(0x63f) => Some("7.1"),
86            _ => None,
87        }
88    }
89}

91/// A stateless Lo/Ro stereo fold-down for known speaker layouts.
92///
93/// The center and one surround pair are mixed at -3 dB into their sides,
94/// following ITU-R BS.775's conventional downmix. LFE is omitted. When both
95/// side and rear pairs are present, each gets -6 dB. The matrix is then
96/// divided by its largest absolute row sum to prevent coherent full-scale
97/// channels from exceeding full scale. This may make isolated channels quiet.
98#[derive(Clone, Copy, Debug)]
99pub struct StereoDownmixer {
100    channels: usize,
101    left: [f64; 8],
102    right: [f64; 8],
103}
104
105impl StereoDownmixer {
106    /// Prepare a supported conventional 2.1–7.1 fold-down. Mono/stereo use
107    /// their original frames; unknown or unsupported masks are refused.
108    pub fn new(layout: ChannelLayout) -> Result<Self, DspError> {
109        let mask = layout.mask().ok_or(DspError::InvalidLayout)?;
110        if !matches!(
111            mask,
112            0xb | 0x7 | 0xf | 0x33 | 0x603 | 0x107 | 0x37 | 0x3f | 0x607 | 0x60f | 0x637 | 0x63f
113        ) {
114            return Err(DspError::InvalidLayout);
115        }
116        let mut left = [0.0; 8];
117        let mut right = [0.0; 8];
118        let both_surround_pairs = mask & 0x630 == 0x630;
119        for (index, bit) in (0..32).filter(|bit| mask & (1 << bit) != 0).enumerate() {
120            let surround = if both_surround_pairs {
121                0.5
122            } else {
123                std::f64::consts::FRAC_1_SQRT_2
124            };
125            let (l, r) = match bit {
126                0 => (1.0, 0.0), // front left
127                1 => (0.0, 1.0), // front right
128                2 => (
129                    std::f64::consts::FRAC_1_SQRT_2,
130                    std::f64::consts::FRAC_1_SQRT_2,
131                ),
132                3 => (0.0, 0.0), // LFE: no bass management in a stereo downmix
133                4 | 9 => (surround, 0.0),
134                5 | 10 => (0.0, surround),
135                8 => (0.5, 0.5), // rear center
136                _ => return Err(DspError::InvalidLayout),
137            };
138            left[index] = l;
139            right[index] = r;
140        }
141        let row_sum = left.iter().sum::<f64>().max(right.iter().sum::<f64>());
142        for weight in &mut left {
143            *weight /= row_sum;
144        }
145        for weight in &mut right {
146            *weight /= row_sum;
147        }
148        Ok(Self {
149            channels: usize::from(layout.channels()),
150            left,
151            right,
152        })
153    }
154
155    /// Write one stereo frame for each source frame, without allocation.
156    /// Invalid or incomplete source/destination frames leave output unchanged.
157    pub fn process(&self, source: &[f32], stereo: &mut [f32]) -> Result<(), DspError> {
158        if !source.len().is_multiple_of(self.channels)
159            || stereo.len() != source.len() / self.channels * 2
160        {
161            return Err(DspError::IncompleteFrame);
162        }
163        if source.iter().any(|sample| !sample.is_finite()) {
164            return Err(DspError::NonFiniteSample);
165        }
166        for (frame, out) in source
167            .chunks_exact(self.channels)
168            .zip(stereo.as_chunks_mut::<2>().0)
169        {
170            let mut left = 0.0;
171            let mut right = 0.0;
172            for (index, &sample) in frame.iter().enumerate() {
173                left += f64::from(sample) * self.left[index];
174                right += f64::from(sample) * self.right[index];
175            }
176            out[0] = left as f32;
177            out[1] = right as f32;
178        }
179        Ok(())
180    }
181}
