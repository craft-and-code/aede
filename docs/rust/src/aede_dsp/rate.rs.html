1//! Stateful, bandlimited conversion between fixed PCM sample rates.
2
3use std::fmt;
4
5use rubato::audioadapter_buffers::direct::InterleavedSlice;
6use rubato::{
7    Async, Fft, FixedAsync, FixedSync, Indexing, Resampler, SincInterpolationParameters,
8    WindowFunction, calculate_cutoff,
9};
10
11const CHUNK_FRAMES: usize = 1024;
12const MIN_FILTER_FRAMES: usize = 256;
13
14#[derive(Debug)]
15pub enum RateError {
16    InvalidFormat,
17    IncompleteFrame,
18    NonFiniteSample,
19    Finished,
20    Resampler(String),
21}
22
23impl fmt::Display for RateError {
24    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
25        match self {
26            Self::InvalidFormat => f.write_str("unsupported PCM rate or channel count"),
27            Self::IncompleteFrame => f.write_str("PCM buffer does not contain complete frames"),
28            Self::NonFiniteSample => f.write_str("PCM buffer contains a non-finite sample"),
29            Self::Finished => f.write_str("sample-rate conversion has already finished"),
30            Self::Resampler(error) => write!(f, "sample-rate conversion failed: {error}"),
31        }
32    }
33}
34
35impl std::error::Error for RateError {}
36
37/// A fixed-ratio converter for one uninterrupted source stream.
38///
39/// `push` accepts any number of complete interleaved frames. Set `last` on
40/// the final call to emit the filter tail, trim its startup delay, and produce
41/// exactly ceil(input frames × output rate / input rate) frames. Calls do not
42/// allocate in the output callback; this type runs on the decoding thread.
43pub struct RateConverter {
44    resampler: Box<dyn Resampler<f32>>,
45    channels: usize,
46    input_rate: u32,
47    output_rate: u32,
48    pending: Vec<f32>,
49    scratch: Vec<f32>,
50    output: Vec<f32>,
51    skip_frames: usize,
52    input_frames: u64,
53    emitted_frames: u64,
54    finished: bool,
55}
56
57impl RateConverter {
58    pub fn new(input: PcmFormat, output_rate: u32) -> Result<Self, RateError> {
59        let input_rate = input.sample_rate();
60        let channels = usize::from(input.channels());
61        if !(8_000..=384_000).contains(&input_rate)
62            || !(8_000..=384_000).contains(&output_rate)
63            || channels == 0
64            || channels > 32
65            || input_rate == output_rate
66        {
67            return Err(RateError::InvalidFormat);
68        }
69        let common = gcd(input_rate, output_rate);
70        let filter_input_frames = (MIN_FILTER_FRAMES * input_rate as usize)
71            .div_ceil(input_rate.min(output_rate) as usize);
72        let resampler: Box<dyn Resampler<f32>> = if input_rate / common > CHUNK_FRAMES as u32
73            || output_rate / common > CHUNK_FRAMES as u32
74        {
75            // Sinc taps are in input frames too. Keep the same lower-rate
76            // bandwidth and guard band when downsampling; an automatic cutoff
77            // based on the enlarged input length would admit aliases.
78            let parameters = SincInterpolationParameters::new(
79                filter_input_frames,
80                WindowFunction::BlackmanHarris2,
81            )
82            .f_cutoff(calculate_cutoff(
83                MIN_FILTER_FRAMES,
84                WindowFunction::BlackmanHarris2,
85            ));
86            Box::new(
87                Async::<f32>::new_sinc(
88                    f64::from(output_rate) / f64::from(input_rate),
89                    1.0,
90                    &parameters,
91                    CHUNK_FRAMES,
92                    channels,
93                    FixedAsync::Input,
94                )
95                .map_err(|error| RateError::Resampler(error.to_string()))?,
96            )
97        } else {
98            // Rubato sizes its filter on the smaller side of the rate pair.
99            // Its default 256 input frames leave only 64 filter frames at
100            // 192 -> 48 kHz, attenuating 19.2 kHz by nearly 8 dB. Preserve
101            // at least 256 frames at the lower rate, even for large ratios.
102            let chunk_frames = CHUNK_FRAMES.max(filter_input_frames);
103            Box::new(
104                Fft::<f32>::new_custom(
105                    input_rate as usize,
106                    output_rate as usize,
107                    chunk_frames,
108                    chunk_frames / filter_input_frames,
109                    channels,
110                    WindowFunction::BlackmanHarris2,
111                    FixedSync::Input,
112                )
113                .map_err(|error| RateError::Resampler(error.to_string()))?,
114            )
115        };
116        let skip_frames = resampler.output_delay();
117        let scratch = vec![0.0; resampler.output_frames_max() * channels];
118        let pending = Vec::with_capacity(resampler.input_frames_max() * channels * 2);
119        Ok(Self {
120            resampler,
121            channels,
122            input_rate,
123            output_rate,
124            pending,
125            scratch,
126            output: Vec::new(),
127            skip_frames,
128            input_frames: 0,
129            emitted_frames: 0,
130            finished: false,
131        })
132    }
133
134    pub fn push(&mut self, samples: &[f32], last: bool) -> Result<&mut [f32], RateError> {
135        if self.finished {
136            return Err(RateError::Finished);
137        }
138        if !samples.len().is_multiple_of(self.channels) {
139            return Err(RateError::IncompleteFrame);
140        }
141        if samples.iter().any(|sample| !sample.is_finite()) {
142            return Err(RateError::NonFiniteSample);
143        }
144        self.output.clear();
145        self.input_frames += (samples.len() / self.channels) as u64;
146        self.pending.extend_from_slice(samples);
147        let target = last.then(|| {
148            self.input_frames
149                .saturating_mul(u64::from(self.output_rate))
150                .div_ceil(u64::from(self.input_rate))
151        });
152        while self.pending.len() / self.channels >= self.resampler.input_frames_next() {
153            self.process_chunk(None, target)?;
154        }
155        if let Some(target) = target {
156            if !self.pending.is_empty() {
157                let partial = self.pending.len() / self.channels;
158                self.process_chunk(Some(partial), Some(target))?;
159            }
160            while self.emitted_frames < target {
161                self.process_chunk(Some(0), Some(target))?;
162            }
163            self.finished = true;
164        }
165        Ok(&mut self.output)
166    }
167
168    fn process_chunk(
169        &mut self,
170        partial: Option<usize>,
171        target: Option<u64>,
172    ) -> Result<(), RateError> {
173        let needed = self.resampler.input_frames_next();
174        let valid = partial.unwrap_or(needed);
175        let input = InterleavedSlice::new(&self.pending, self.channels, valid)
176            .map_err(|error| RateError::Resampler(error.to_string()))?;
177        let mut output = InterleavedSlice::new_mut(
178            &mut self.scratch,
179            self.channels,
180            self.resampler.output_frames_max(),
181        )
182        .map_err(|error| RateError::Resampler(error.to_string()))?;
183        let indexing = Indexing {
184            partial_len: partial,
185            ..Indexing::default()
186        };
187        let (_, produced) = self
188            .resampler
189            .process_into_buffer(&input, &mut output, Some(&indexing))
190            .map_err(|error| RateError::Resampler(error.to_string()))?;
191        let remaining = self.pending.len() - valid * self.channels;
192        self.pending.copy_within(valid * self.channels.., 0);
193        self.pending.truncate(remaining);
194        let skipped = self.skip_frames.min(produced);
195        self.skip_frames -= skipped;
196        let available = produced - skipped;
197        let accepted = target
198            .map(|limit| available.min(limit.saturating_sub(self.emitted_frames) as usize))
199            .unwrap_or(available);
200        self.output.extend_from_slice(
201            &self.scratch[skipped * self.channels..(skipped + accepted) * self.channels],
202        );
203        self.emitted_frames += accepted as u64;
204        Ok(())
205    }
206}
207
208fn gcd(mut left: u32, mut right: u32) -> u32 {
209    while right != 0 {
210        (left, right) = (right, left % right);
211    }
212    left
213}
