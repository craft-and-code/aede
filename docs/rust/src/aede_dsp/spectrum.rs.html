1//! Frequency-band levels for a playback visualizer.
2
3use std::f32::consts::TAU;
4
5use crate::{DspError, PcmFormat};
6
7/// Number of logarithmic display bands between 45 Hz and min(Nyquist, 16 kHz).
8pub const SPECTRUM_BANDS: usize = 24;
9const WINDOW: usize = 2048;
10
11/// Twenty-four smoothed frequency-band levels from interleaved PCM.
12///
13/// The values are display levels in `0..=1`, not loudness measurements. Each
14/// completed window uses a Hann taper and a real-input FFT. The source PCM is
15/// only read; playback sends the same samples to the output device.
16pub struct Spectrum {
17    format: PcmFormat,
18    samples: [f32; WINDOW],
19    taper: [f32; WINDOW],
20    filled: usize,
21    levels: [f32; SPECTRUM_BANDS],
22}
23
24impl Spectrum {
25    /// Prepare a fresh fixed-format 2048-frame analysis window.
26    /// Build this outside the real-time audio callback.
27    pub fn new(format: PcmFormat) -> Self {
28        let mut taper = [0.0; WINDOW];
29        for (index, weight) in taper.iter_mut().enumerate() {
30            *weight = 0.5 * (1.0 - (TAU * index as f32 / (WINDOW - 1) as f32).cos());
31        }
32        Self {
33            format,
34            samples: [0.0; WINDOW],
35            taper,
36            filled: 0,
37            levels: [0.0; SPECTRUM_BANDS],
38        }
39    }
40
41    /// Consume complete PCM frames and return the latest completed window.
42    ///
43    /// Incomplete or non-finite input leaves both smoothing and partially
44    /// collected PCM unchanged. A block may complete multiple windows; the
45    /// result describes the last one, or `None` if no window completed.
46    pub fn push(&mut self, samples: &[f32]) -> Result<Option<[f32; SPECTRUM_BANDS]>, DspError> {
47        let channels = usize::from(self.format.channels());
48        if !samples.len().is_multiple_of(channels) {
49            return Err(DspError::IncompleteFrame);
50        }
51        if samples.iter().any(|sample| !sample.is_finite()) {
52            return Err(DspError::NonFiniteSample);
53        }
54        let mut latest = None;
55        for frame in samples.chunks_exact(channels) {
56            // Giving the first channel a larger weight keeps opposite-phase
57            // stereo from cancelling into silence in this display mix.
58            self.samples[self.filled] = if channels == 1 {
59                frame[0]
60            } else {
61                let others = frame[1..].iter().copied().sum::<f32>() / (channels - 1) as f32;
62                frame[0] * 0.75 + others * 0.25
63            };
64            self.filled += 1;
65            if self.filled == WINDOW {
66                self.analyze();
67                self.filled = 0;
68                latest = Some(self.levels);
69            }
70        }
71        Ok(latest)
72    }
73
74    fn analyze(&mut self) {
75        let mut real = [0.0; WINDOW];
76        let mut imaginary = [0.0; WINDOW];
77        for ((out, sample), taper) in real.iter_mut().zip(&self.samples).zip(&self.taper) {
78            *out = sample * taper;
79        }
80        fft(&mut real, &mut imaginary);
81
82        let high = (self.format.sample_rate() as f32 / 2.0).min(16_000.0);
83        let mut peaks = [0.0_f32; SPECTRUM_BANDS];
84        if high > 45.0 {
85            let logarithmic_width = (high / 45.0).ln() / SPECTRUM_BANDS as f32;
86            for index in 1..WINDOW / 2 {
87                let hz = index as f32 * self.format.sample_rate() as f32 / WINDOW as f32;
88                if !(45.0..=high).contains(&hz) {
89                    continue;
90                }
91                let band =
92                    (((hz / 45.0).ln() / logarithmic_width) as usize).min(SPECTRUM_BANDS - 1);
93                let amplitude = real[index].hypot(imaginary[index]) * 4.0 / WINDOW as f32;
94                peaks[band] = peaks[band].max(amplitude);
95            }
96        }
97        for (level, peak) in self.levels.iter_mut().zip(peaks) {
98            let target = if peak > 0.0 {
99                ((20.0 * peak.log10() + 60.0) / 50.0).clamp(0.0, 1.0)
100            } else {
101                0.0
102            };
103            let retained = if target > *level { 0.25 } else { 0.70 };
104            *level = *level * retained + target * (1.0 - retained);
105        }
106    }
107}
108
109fn fft(real: &mut [f32; WINDOW], imaginary: &mut [f32; WINDOW]) {
110    let mut swapped = 0;
111    for index in 1..WINDOW {
112        let mut bit = WINDOW / 2;
113        while swapped & bit != 0 {
114            swapped ^= bit;
115            bit /= 2;
116        }
117        swapped ^= bit;
118        if index < swapped {
119            real.swap(index, swapped);
120            imaginary.swap(index, swapped);
121        }
122    }
123
124    let mut span = 2;
125    while span <= WINDOW {
126        let phase = -TAU / span as f32;
127        let (step_imaginary, step_real) = phase.sin_cos();
128        for start in (0..WINDOW).step_by(span) {
129            let (mut rotation_real, mut rotation_imaginary) = (1.0, 0.0);
130            for offset in 0..span / 2 {
131                let left = start + offset;
132                let right = left + span / 2;
133                let right_real =
134                    real[right] * rotation_real - imaginary[right] * rotation_imaginary;
135                let right_imaginary =
136                    real[right] * rotation_imaginary + imaginary[right] * rotation_real;
137                real[right] = real[left] - right_real;
138                imaginary[right] = imaginary[left] - right_imaginary;
139                real[left] += right_real;
140                imaginary[left] += right_imaginary;
141                let next_real = rotation_real * step_real - rotation_imaginary * step_imaginary;
142                rotation_imaginary =
143                    rotation_real * step_imaginary + rotation_imaginary * step_real;
144                rotation_real = next_real;
145            }
146        }
147        span *= 2;
148    }
149}
