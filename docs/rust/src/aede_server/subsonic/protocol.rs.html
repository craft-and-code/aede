28pub(super) fn response(format: Format, result: Result<Value, ProtocolError>) -> Response {
29    let (status, payload) = match result {
30        Ok(payload) => ("ok", payload),
31        Err(failure) => (
32            "failed",
33            json!({"error": {"code": failure.code, "message": failure.message}}),
34        ),
35    };
36    let mut envelope = json!({"status": status, "version": "1.16.1", "type": "Aede",
37        "serverVersion": env!("CARGO_PKG_VERSION"), "openSubsonic": true});
38    if let (Some(envelope), Some(payload)) = (envelope.as_object_mut(), payload.as_object()) {
39        envelope.extend(
40            payload
41                .iter()
42                .map(|(name, value)| (name.clone(), value.clone())),
43        );
44    }
45    let mut response = match format {
46        Format::Json => Json(json!({"subsonic-response": envelope})).into_response(),
47        Format::Xml => {
48            let mut output = String::from("<?xml version=\"1.0\" encoding=\"UTF-8\"?>");
49            xml_element(&mut output, "subsonic-response", &envelope, true);
50            (
51                [(header::CONTENT_TYPE, "application/xml; charset=utf-8")],
52                output,
53            )
54                .into_response()
55        }
56    };
57    response.headers_mut().insert(
58        header::CACHE_CONTROL,
59        axum::http::HeaderValue::from_static("no-store"),
60    );
61    response.headers_mut().insert(
62        header::HeaderName::from_static("x-content-type-options"),
63        axum::http::HeaderValue::from_static("nosniff"),
64    );
65    response
66}

68fn scalar(value: &Value) -> Option<String> {
69    match value {
70        Value::String(text) => Some(text.clone()),
71        Value::Bool(value) => Some(value.to_string()),
72        Value::Number(value) => Some(value.to_string()),
73        _ => None,
74    }
75}

77// Element/attribute names come only from our payload builders, never request
78// or tag names. Strings are escaped and invalid XML 1.0 controls replaced.
79fn escaped(output: &mut String, text: &str) {
80    for ch in text.chars() {
81        match ch {
82            '&' => output.push_str("&amp;"),
83            '<' => output.push_str("&lt;"),
84            '>' => output.push_str("&gt;"),
85            '"' => output.push_str("&quot;"),
86            '\'' => output.push_str("&apos;"),
87            '\t' => output.push_str("&#9;"),
88            '\n' => output.push_str("&#10;"),
89            '\r' => output.push_str("&#13;"),
90            '\u{20}'..='\u{d7ff}' | '\u{e000}'..='\u{fffd}' | '\u{10000}'..='\u{10ffff}' => {
91                output.push(ch)
92            }
93            _ => output.push('\u{fffd}'),
94        }
95    }
96}

98fn xml_element(output: &mut String, name: &str, value: &Value, root: bool) {
99    if let Value::Array(items) = value {
100        for item in items {
101            xml_element(output, name, item, false);
102        }
103        return;
104    }
105    output.push('<');
106    output.push_str(name);
107    if root {
108        output.push_str(" xmlns=\"http://subsonic.org/restapi\"");
109    }
110    if let Some(object) = value.as_object() {
111        for (key, value) in object {
112            if key == "value" {
113                continue;
114            }
115            if let Some(text) = scalar(value) {
116                output.push(' ');
117                output.push_str(key);
118                output.push_str("=\"");
119                escaped(output, &text);
120                output.push('"');
121            }
122        }
123    }
124    output.push('>');
125    if let Some(object) = value.as_object() {
126        for (key, value) in object {
127            if key == "value" {
128                if let Some(text) = scalar(value) {
129                    escaped(output, &text);
130                }
131            } else if value.is_object() || value.is_array() {
132                xml_element(output, key, value, false);
133            }
134        }
135    } else if let Some(text) = scalar(value) {
136        escaped(output, &text);
137    }
138    output.push_str("</");
139    output.push_str(name);
140    output.push('>');
141}

143pub(super) fn sha256(text: &str) -> Result<String, ProtocolError> {
144    // Rustls already exposes the approved provider's SHA-256 implementation.
145    // Use the named suite so algorithm/provider defaults cannot change IDs.
146    let suite = rustls::crypto::ring::cipher_suite::TLS13_AES_128_GCM_SHA256
147        .tls13()
148        .ok_or_else(|| ProtocolError::new(0, "SHA-256 provider unavailable"))?;
149    let digest = suite.common.hash_provider.hash(text.as_bytes());
150    let mut output = String::with_capacity(64);
151    for byte in digest.as_ref() {
152        write!(&mut output, "{byte:02x}")
153            .map_err(|_| ProtocolError::new(0, "could not encode identifier"))?;
154    }
155    Ok(output)
156}

158pub(super) fn opaque_id(
159    catalog: &Catalog,
160    kind: EntityKind,
161    id: Id,
162) -> Result<String, ProtocolError> {
163    let reference = EntityRef::of(catalog, kind, id)
164        .ok_or_else(|| ProtocolError::new(70, "entity not found"))?;
165    Ok(format!(
166        "{}-{}",
167        kind.as_str(),
168        sha256(&reference.to_token())?
169    ))
170}

172/// UTC snapshot timestamp; the catalog has no independent album-added date.
173pub(super) fn utc_date(seconds: u64) -> Result<String, ProtocolError> {
174    // Gregorian civil-date conversion from Unix days. The accepted range
175    // limits intermediate arithmetic and yields XML Schema's four-digit year.
176    if seconds > 253_402_300_799 {
177        return Err(ProtocolError::new(0, "catalog timestamp is out of range"));
178    }
179    let days = seconds / 86_400;
180    let z = days as i64 + 719_468;
181    let era = z / 146_097;
182    let doe = z - era * 146_097;
183    let yoe = (doe - doe / 1460 + doe / 36_524 - doe / 146_096) / 365;
184    let mut year = yoe + era * 400;
185    let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
186    let mp = (5 * doy + 2) / 153;
187    let day = doy - (153 * mp + 2) / 5 + 1;
188    let month = mp + if mp < 10 { 3 } else { -9 };
189    year += i64::from(month <= 2);
190    let time = seconds % 86_400;
191    Ok(format!(
192        "{year:04}-{month:02}-{day:02}T{:02}:{:02}:{:02}Z",
193        time / 3600,
194        time / 60 % 60,
195        time % 60
196    ))
197}
