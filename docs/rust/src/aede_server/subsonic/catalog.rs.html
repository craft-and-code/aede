43/// One borrowed projection per catalog operation. Resolve opaque IDs and
44/// credits once so listing a page does not repeatedly scan the whole graph.
45pub(super) struct Index<'a> {
46    catalog: &'a Catalog,
47    artists: Vec<ArtistRow<'a>>,
48    albums: Vec<AlbumRow<'a>>,
49    songs: Vec<SongRow<'a>>,
50    ids: HashMap<String, (EntityKind, Id)>,
51    references: BTreeMap<EntityRef, Id>,
52    created: String,
53}
54
55impl<'a> Index<'a> {
56    pub(super) fn new(catalog: &'a Catalog) -> Result<Self, ProtocolError> {
57        if catalog
58            .artists
59            .iter()
60            .enumerate()
61            .any(|(at, row)| row.id as usize != at)
62            || catalog
63                .releases
64                .iter()
65                .enumerate()
66                .any(|(at, row)| row.id as usize != at)
67            || catalog
68                .tracks
69                .iter()
70                .enumerate()
71                .any(|(at, row)| row.id as usize != at)
72        {
73            return Err(ProtocolError::new(0, "catalog identities are invalid"));
74        }
75        let mut index = Self {
76            catalog,
77            artists: catalog
78                .artists
79                .iter()
80                .map(|artist| {
81                    Ok(ArtistRow {
82                        entity: artist,
83                        id: opaque_id(catalog, EntityKind::Artist, artist.id)?,
84                        sort: text::normalize(&artist.sort_name),
85                        albums: Vec::new(),
86                    })
87                })
88                .collect::<Result<_, ProtocolError>>()?,
89            albums: catalog
90                .releases
91                .iter()
92                .map(|release| {
93                    Ok(AlbumRow {
94                        entity: release,
95                        id: opaque_id(catalog, EntityKind::Release, release.id)?,
96                        sort: text::normalize(&release.title),
97                        genres: Vec::new(),
98                        cover: super::artwork::cover_id(catalog, release)?,
99                        duration_ms: release
100                            .track_ids
101                            .iter()
102                            .filter_map(|id| catalog.track(*id)?.duration_ms)
103                            .fold(0, u64::saturating_add),
104                    })
105                })
106                .collect::<Result<_, ProtocolError>>()?,
107            songs: catalog
108                .tracks
109                .iter()
110                .map(|track| {
111                    Ok(SongRow {
112                        entity: track,
113                        id: opaque_id(catalog, EntityKind::Track, track.id)?,
114                        sort: text::normalize(&track.title),
115                        artists: Vec::new(),
116                        genres: Vec::new(),
117                    })
118                })
119                .collect::<Result<_, ProtocolError>>()?,
120            ids: HashMap::new(),
121            references: BTreeMap::new(),
122            // The core stores the last scan time, not a per-album addition
123            // date. Use that explicit proxy for the required legacy field.
124            created: utc_date(catalog.scanned_at)?,
125        };
126        for (id, kind, entity) in index
127            .artists
128            .iter()
129            .map(|row| (&row.id, EntityKind::Artist, row.entity.id))
130            .chain(
131                index
132                    .albums
133                    .iter()
134                    .map(|row| (&row.id, EntityKind::Release, row.entity.id)),
135            )
136            .chain(
137                index
138                    .songs
139                    .iter()
140                    .map(|row| (&row.id, EntityKind::Track, row.entity.id)),
141            )
142        {
143            if index.ids.insert(id.clone(), (kind, entity)).is_some() {
144                return Err(ProtocolError::new(0, "catalog identities are ambiguous"));
145            }
146            let reference = EntityRef::of(catalog, kind, entity)
147                .ok_or_else(|| ProtocolError::new(0, "catalog identity is invalid"))?;
148            index.references.insert(reference, entity);
149        }
150        // Core personal references may name an artist by its unique MBID.
151        // Native keys win, and ambiguous MBIDs never invent an attachment.
152        let mut mbids = HashMap::new();
153        for artist in &catalog.artists {
154            if let Some(mbid) = artist.mbid.as_deref() {
155                mbids
156                    .entry(mbid)
157                    .and_modify(|id| *id = None)
158                    .or_insert(Some(artist.id));
159            }
160        }
161        for (mbid, id) in mbids {
162            if let Some(id) = id {
163                index
164                    .references
165                    .entry(EntityRef::new(EntityKind::Artist, format!("mbid:{mbid}")))
166                    .or_insert(id);
167            }
168        }
169        let mut album_credits = HashSet::new();
170        for album in &index.albums {
171            if let Some(artist) = album
172                .entity
173                .album_artist_id
174                .and_then(|id| index.artists.get_mut(id as usize))
175            {
176                album_credits.insert((artist.entity.id, album.entity.id));
177                artist.albums.push(album.entity.id);
178            }
179        }
180        for credit in &catalog.credits {
181            if credit.entity_kind == EntityKind::Release
182                && credit.role == "album"
183                && index.albums.get(credit.entity_id as usize).is_some()
184                && album_credits.insert((credit.artist_id, credit.entity_id))
185                && let Some(artist) = index.artists.get_mut(credit.artist_id as usize)
186            {
187                artist.albums.push(credit.entity_id);
188            }
189        }
190        let mut credits: Vec<_> = catalog
191            .credits
192            .iter()
193            .filter(|credit| {
194                credit.entity_kind == EntityKind::Track
195                    && matches!(credit.role.as_str(), "main" | "featured")
196            })
197            .collect();
198        credits.sort_by(|a, b| {
199            let artist_key = |id: Id| {
200                index
201                    .artists
202                    .get(id as usize)
203                    .map(|row| (row.sort.as_str(), row.id.as_str()))
204            };
205            (a.role != "main", a.order, artist_key(a.artist_id)).cmp(&(
206                b.role != "main",
207                b.order,
208                artist_key(b.artist_id),
209            ))
210        });
211        let mut assigned = HashSet::new();
212        for credit in credits {
213            if let Some(song) = index.songs.get_mut(credit.entity_id as usize)
214                && assigned.insert((credit.entity_id, credit.artist_id))
215            {
216                song.artists.push(credit.artist_id);
217            }
218        }
219        for song in &mut index.songs {
220            if song.artists.is_empty()
221                && let Some(artist) = song
222                    .entity
223                    .release_id
224                    .and_then(|id| catalog.release(id))
225                    .and_then(|album| album.album_artist_id)
226            {
227                song.artists.push(artist);
228            }
229        }
230        let mut attached = HashSet::new();
231        for link in &catalog.genre_links {
232            if !attached.insert((link.entity_kind, link.entity_id, link.genre_id)) {
233                continue;
234            }
235            match link.entity_kind {
236                EntityKind::Release => {
237                    if let Some(album) = index.albums.get_mut(link.entity_id as usize) {
238                        album.genres.push(link.genre_id);
239                    }
240                }
241                EntityKind::Track => {
242                    if let Some(song) = index.songs.get_mut(link.entity_id as usize) {
243                        song.genres.push(link.genre_id);
244                    }
245                }
246                _ => {}
247            }
248        }
249        for song in &mut index.songs {
250            if song.genres.is_empty()
251                && let Some(album) = song
252                    .entity
253                    .release_id
254                    .and_then(|id| index.albums.get(id as usize))
255            {
256                song.genres.clone_from(&album.genres);
257            }
258        }
259        Ok(index)
260    }

262    fn resolve(&self, id: &str, kind: EntityKind) -> Result<Id, ProtocolError> {
263        self.ids
264            .get(id)
265            .filter(|(found, _)| *found == kind)
266            .map(|(_, entity)| *entity)
267            .ok_or_else(|| ProtocolError::new(70, "requested media was not found"))
268    }

270    /// Resolve a public ID only for the entity kinds accepted by its operation.
271    /// Stable private references stay within the server and its personal store.
272    pub(super) fn reference(
273        &self,
274        wanted: &str,
275        allowed_kinds: &[EntityKind],
276    ) -> Result<EntityRef, ProtocolError> {
277        let &(kind, id) = self
278            .ids
279            .get(wanted)
280            .filter(|(kind, _)| allowed_kinds.contains(kind))
281            .ok_or_else(|| ProtocolError::new(70, "requested media was not found"))?;
282        EntityRef::of(self.catalog, kind, id)
283            .ok_or_else(|| ProtocolError::new(0, "catalog identity is invalid"))
284    }

286    /// Render a native artist, release or track as one compatibility list item.
287    pub(super) fn render(&self, kind: EntityKind, id: Id) -> Result<Value, ProtocolError> {
288        let value = match kind {
289            EntityKind::Artist => self.artists.get(id as usize).map(|row| self.artist(row)),
290            EntityKind::Release => self.albums.get(id as usize).map(|row| self.album(row)),
291            EntityKind::Track => self.songs.get(id as usize).map(|row| self.song(row)),
292            _ => None,
293        };
294        value.ok_or_else(|| ProtocolError::new(70, "requested media was not found"))
295    }

297    /// Render a saved stable reference without rescanning the catalog per row.
298    pub(super) fn render_reference(&self, reference: &EntityRef) -> Result<Value, ProtocolError> {
299        let legacy = (reference.kind == EntityKind::Release)
300            .then(|| {
301                let (prefix, folder) = reference
302                    .key
303                    .rsplit_once(aede_core::user::RELEASE_KEY_SEPARATOR)?;
304                text::disc_folder(text::file_name(folder))?;
305                Some(EntityRef::new(
306                    EntityKind::Release,
307                    format!("{prefix}|{}", text::folder(folder)),
308                ))
309            })
310            .flatten();
311        let &id = self
312            .references
313            .get(reference)
314            .or_else(|| {
315                legacy
316                    .as_ref()
317                    .and_then(|reference| self.references.get(reference))
318            })
319            .ok_or_else(|| ProtocolError::new(70, "requested media was not found"))?;
320        self.render(reference.kind, id)
321    }

323    /// Known song duration in whole seconds without constructing a JSON item.
324    /// A missing song is an error; an unmeasured duration stays unknown.
325    pub(super) fn track_duration(
326        &self,
327        reference: &EntityRef,
328    ) -> Result<Option<u64>, ProtocolError> {
329        let track = (reference.kind == EntityKind::Track)
330            .then(|| self.references.get(reference))
331            .flatten()
332            .and_then(|id| self.catalog.track(*id))
333            .ok_or_else(|| ProtocolError::new(70, "requested song was not found"))?;
334        Ok(track.duration_ms.map(|duration| duration / 1000))
335    }

354    fn album(&self, row: &AlbumRow<'_>) -> Value {
355        let mut value = json!({
356            "id": row.id,
357            "name": row.entity.title,
358            "songCount": row.entity.track_ids.len(),
359            "duration": row.duration_ms / 1000,
360            "created": self.created,
361        });
362        if let Some(artist) = row
363            .entity
364            .album_artist_id
365            .and_then(|id| self.artists.get(id as usize))
366        {
367            value["artist"] = json!(artist.entity.name);
368            value["artistId"] = json!(artist.id);
369        }
370        if let Some(year) = row.entity.year {
371            value["year"] = json!(year);
372        }
373        if let Some(genre) = self.genre_name(&row.genres) {
374            value["genre"] = json!(genre);
375        }
376        if let Some(cover) = &row.cover {
377            value["coverArt"] = json!(cover);
378        }
379        value
380    }

382    fn song(&self, row: &SongRow<'_>) -> Value {
383        let mut value = json!({
384            "id": row.id,
385            "title": row.entity.title,
386            "isDir": false,
387            "isVideo": false,
388            "type": "music",
389        });
390        if let Some(album) = row
391            .entity
392            .release_id
393            .and_then(|id| self.albums.get(id as usize))
394        {
395            value["album"] = json!(album.entity.title);
396            value["albumId"] = json!(album.id);
397            value["parent"] = json!(album.id);
398            if let Some(cover) = &album.cover {
399                value["coverArt"] = json!(cover);
400            }
401            if let Some(year) = album.entity.year {
402                value["year"] = json!(year);
403            }
404        }
405        if let Some(artist) = row
406            .artists
407            .first()
408            .and_then(|id| self.artists.get(*id as usize))
409        {
410            value["artistId"] = json!(artist.id);
411            value["artist"] = json!(
412                row.artists
413                    .iter()
414                    .filter_map(|id| {
415                        self.artists
416                            .get(*id as usize)
417                            .map(|artist| artist.entity.name.as_str())
418                    })
419                    .collect::<Vec<_>>()
420                    .join(" / ")
421            );
422        }
423        if let Some(position) = row.entity.track_no {
424            value["track"] = json!(position);
425        }
426        if let Some(disc) = row.entity.disc_no {
427            value["discNumber"] = json!(disc);
428        }
429        if let Some(duration) = row.entity.duration_ms {
430            value["duration"] = json!(duration / 1000);
431        }
432        if let Some(genre) = self.genre_name(&row.genres) {
433            value["genre"] = json!(genre);
434        }
435        if let Some(file) = self.catalog.file(row.entity.file_id) {
436            value["size"] = json!(file.size);
437            value["contentType"] = json!(super::media::content_type(&file.properties));
438            value["suffix"] = json!(super::media::suffix(&file.properties));
439            if let Some(bitrate) = file.properties.bitrate_kbps {
440                value["bitRate"] = json!(bitrate);
441            }
442        }
443        value
444    }

454    if params

456        .is_some_and(|id| id != MUSIC_FOLDER_ID)
457    {
458        return Err(ProtocolError::new(70, "music folder was not found"));
459    }
460    Ok(())

463/// Resolve a song's public ID to the native reference consumed by playback.
464/// The reference stays within the server; responses never serialize its key.
465pub(super) fn track_reference(catalog: &Catalog, wanted: &str) -> Result<EntityRef, ProtocolError> {
466    for track in &catalog.tracks {
467        if opaque_id(catalog, EntityKind::Track, track.id)? == wanted {
468            return EntityRef::of(catalog, EntityKind::Track, track.id)
469                .ok_or_else(|| ProtocolError::new(0, "catalog track identity is invalid"));
470        }
471    }
472    Err(ProtocolError::new(70, "requested media was not found"))
473}

475/// Translate one supported ID3 browsing method into its payload fields.
476///
477/// The caller runs this CPU work under the server's catalog worker limit and
478/// wraps the result in the common JSON/XML response envelope.
479#[cfg(test)]
480pub(super) fn dispatch(
481    method: &str,
482    params: &Parameters,
483    catalog: &Catalog,
484) -> Result<Value, ProtocolError> {
485    catalog_parameters(method, params)?;
486    if method == "getMusicFolders" {
487        return Ok(music_folders());
488    }
489    dispatch_validated(method, params, &Index::new(catalog)?)
490}

507pub(super) fn catalog_parameters(method: &str, params: &Parameters) -> Result<(), ProtocolError> {
508    match method {
509        "getMusicFolders" => params.allowed(&[])?,
510        "getGenres" => params.allowed(&[])?,
511        "getArtists" => params.allowed(&["musicFolderId"])?,
512        "getArtist" | "getAlbum" | "getSong" => params.allowed(&["id"])?,
513        "getAlbumList2" => params.allowed(&[
514            "type",
515            "size",
516            "limit",
517            "offset",
518            "fromYear",
519            "toYear",
520            "genre",
521            "musicFolderId",
522        ])?,
523        "getRandomSongs" => {
524            params.allowed(&["size", "genre", "fromYear", "toYear", "musicFolderId"])?
525        }
526        "getSongsByGenre" => params.allowed(&["genre", "count", "offset", "musicFolderId"])?,
527        "search3" => params.allowed(&[
528            "query",
529            "artistCount",
530            "artistOffset",
531            "albumCount",
532            "albumOffset",
533            "songCount",
534            "songOffset",
535            "musicFolderId",
536        ])?,
537        _ => return Err(ProtocolError::new(0, "catalog method is not supported")),
538    }
539    folder(params)
540}

542fn dispatch_validated(
543    method: &str,
544    params: &Parameters,
545    index: &Index<'_>,
546) -> Result<Value, ProtocolError> {
547    match method {
548        "getMusicFolders" => Ok(music_folders()),
549        "getArtists" => {
550            let mut rows: Vec<_> = index.artists.iter().collect();
551            rows.sort_by(|a, b| (a.sort.as_str(), &a.id).cmp(&(b.sort.as_str(), &b.id)));
552            let mut groups: BTreeMap<String, Vec<Value>> = BTreeMap::new();
553            for row in rows {
554                let letter = row
555                    .sort
556                    .chars()
557                    .next()
558                    .filter(|letter| letter.is_alphabetic())
559                    .map(|letter| letter.to_uppercase().collect::<String>())
560                    .unwrap_or_else(|| "#".into());
561                groups.entry(letter).or_default().push(index.artist(row));
562            }
563            Ok(
564                json!({"artists": {"ignoredArticles": "", "index": groups.into_iter()
565                .map(|(name, artist)| json!({"name": name, "artist": artist})).collect::<Vec<_>>()}}),
566            )
567        }
568        "getArtist" => {
569            let id = index.resolve(params.required("id")?, EntityKind::Artist)?;
570            let row = &index.artists[id as usize];
571            let mut value = index.artist(row);
572            let mut albums: Vec<_> = row
573                .albums
574                .iter()
575                .filter_map(|id| index.albums.get(*id as usize))
576                .collect();
577            albums.sort_by(|a, b| (a.sort.as_str(), &a.id).cmp(&(b.sort.as_str(), &b.id)));
578            value["album"] = json!(
579                albums
580                    .into_iter()
581                    .map(|album| index.album(album))
582                    .collect::<Vec<_>>()
583            );
584            Ok(json!({"artist": value}))
585        }
586        "getAlbum" => {
587            let id = index.resolve(params.required("id")?, EntityKind::Release)?;
588            let row = &index.albums[id as usize];
589            let mut value = index.album(row);
590            value["song"] = json!(
591                row.entity
592                    .track_ids
593                    .iter()
594                    .filter_map(|id| index.songs.get(*id as usize))
595                    .map(|song| index.song(song))
596                    .collect::<Vec<_>>()
597            );
598            Ok(json!({"album": value}))
599        }
600        "getSong" => {
601            let id = index.resolve(params.required("id")?, EntityKind::Track)?;
602            Ok(json!({"song": index.song(&index.songs[id as usize])}))
603        }
604        "getAlbumList2" => album_list(params, index),
605        "getRandomSongs" => random_songs(params, index),
606        "getSongsByGenre" => songs_by_genre(params, index),
607        "search3" => search(params, index),
608        "getGenres" => Ok(genres(index)),
609        _ => Err(ProtocolError::new(0, "catalog method is not supported")),
610    }
611}

613fn album_list(params: &Parameters, index: &Index<'_>) -> Result<Value, ProtocolError> {
614    if params.get("size").is_some() && params.get("limit").is_some() {
615        return Err(ProtocolError::new(10, "specify only one of size or limit"));
616    }
617    // Supersonic sends limit in its year/genre iterators. Honour that explicit
618    // page-size alias with the same bounds rather than ignoring the argument.
619    let size = params.number(
620        if params.get("limit").is_some() {
621            "limit"
622        } else {
623            "size"
624        },
625        10,
626        PAGE_LIMIT,
627    )?;
628    let offset = params.number("offset", 0, usize::MAX)?;
629    let kind = params.required("type")?;
630    if kind != "byYear" && (params.get("fromYear").is_some() || params.get("toYear").is_some()) {
631        return Err(ProtocolError::new(0, "year bounds require type=byYear"));
632    }
633    if kind != "byGenre" && params.get("genre").is_some() {
634        return Err(ProtocolError::new(0, "genre requires type=byGenre"));
635    }
636    let mut rows = index.sorted_albums();
637    match kind {
638        "alphabeticalByName" | "alphabeticalByArtist" => {
639            if kind == "alphabeticalByArtist" {
640                let artist_key = |row: &AlbumRow<'_>| {
641                    row.entity
642                        .album_artist_id
643                        .and_then(|id| index.artists.get(id as usize))
644                        .map(|artist| artist.sort.as_str())
645                        .unwrap_or_default()
646                };
647                rows.sort_by(|a, b| {
648                    (artist_key(a), a.sort.as_str(), &a.id).cmp(&(
649                        artist_key(b),
650                        b.sort.as_str(),
651                        &b.id,
652                    ))
653                });
654            }
655        }
656        "byYear" => {
657            params.required("fromYear")?;
658            params.required("toYear")?;
659            let from = params.number("fromYear", 0, u32::MAX as usize)? as u32;
660            let to = params.number("toYear", 0, u32::MAX as usize)? as u32;
661            rows.retain(|row| {
662                row.entity
663                    .year
664                    .is_some_and(|year| (from.min(to)..=from.max(to)).contains(&year))
665            });
666            rows.sort_by(|a, b| {
667                let years = a.entity.year.cmp(&b.entity.year);
668                let years = if from > to { years.reverse() } else { years };
669                years.then_with(|| (a.sort.as_str(), &a.id).cmp(&(b.sort.as_str(), &b.id)))
670            });
671        }
672        "byGenre" => {
673            let genres = genre_ids(index, params.required("genre")?);
674            rows.retain(|row| row.genres.iter().any(|id| genres.contains(id)));
675        }
676        "random" => {
677            let order = random_order(rows.iter().map(|row| row.entity.id).collect())?;
678            rows = order
679                .ordered_tracks()
680                .map(|id| &index.albums[id as usize])
681                .collect();
682        }
683        _ => return Err(ProtocolError::new(0, "album list type is not supported")),
684    }
685    Ok(
686        json!({"albumList2": {"album": rows.into_iter().skip(offset).take(size)
687        .map(|row| index.album(row)).collect::<Vec<_>>()}}),
688    )
689}

691/// The queue already owns the platform-independent, unbiased shuffle. Draw
692/// one secure seed per request rather than duplicating its permutation logic.
693fn random_order(ids: Vec<Id>) -> Result<aede_core::playback::Queue, ProtocolError> {
694    let mut seed = [0; 8];
695    rustls::crypto::ring::default_provider()
696        .secure_random
697        .fill(&mut seed)
698        .map_err(|_| ProtocolError::new(0, "random selection is unavailable"))?;
699    Ok(aede_core::playback::Queue::new(
700        ids,
701        Some(u64::from_le_bytes(seed)),
702    ))
703}

732fn random_songs(params: &Parameters, index: &Index<'_>) -> Result<Value, ProtocolError> {
733    let size = params.number("size", 10, PAGE_LIMIT)?;
734    let genres = params
735        .get("genre")
736        .map(|_| {
737            params
738                .required("genre")
739                .map(|genre| genre_ids(index, genre))
740        })
741        .transpose()?;
742    let year = |name| {
743        params
744            .get(name)
745            .map(|_| {
746                params.required(name)?;
747                params
748                    .number(name, 0, u32::MAX as usize)
749                    .map(|year| year as u32)
750            })
751            .transpose()
752    };
753    let from = year("fromYear")?;
754    let to = year("toYear")?;
755    if from.zip(to).is_some_and(|(from, to)| from > to) {
756        return Err(ProtocolError::new(0, "fromYear must not exceed toYear"));
757    }
758    let ids = index
759        .songs
760        .iter()
761        .filter(|row| {
762            let genre_matches = genres
763                .as_ref()
764                .is_none_or(|genres| row.genres.iter().any(|id| genres.contains(id)));
765            let year = row
766                .entity
767                .release_id
768                .and_then(|id| index.catalog.release(id))
769                .and_then(|album| album.year);
770            genre_matches
771                && (from.is_none() && to.is_none()
772                    || year.is_some_and(|year| {
773                        from.is_none_or(|from| year >= from) && to.is_none_or(|to| year <= to)
774                    }))
775        })
776        .map(|row| row.entity.id)
777        .collect();
778    let order = random_order(ids)?;
779    Ok(
780        json!({"randomSongs": {"song": order.ordered_tracks().take(size)
781        .map(|id| index.song(&index.songs[id as usize])).collect::<Vec<_>>()}}),
782    )
783}

791fn search(params: &Parameters, index: &Index<'_>) -> Result<Value, ProtocolError> {
792    let query = params
793        .get("query")
794        .ok_or_else(|| ProtocolError::new(10, "query is required"))?;
795    if query.len() > QUERY_LIMIT {
796        return Err(ProtocolError::new(0, "query exceeds 1024 bytes"));
797    }
798    let wanted = text::normalize(query);
799    let artist_count = params.number("artistCount", 20, SEARCH_LIMIT)?;
800    let artist_offset = params.number("artistOffset", 0, usize::MAX)?;
801    let album_count = params.number("albumCount", 20, SEARCH_LIMIT)?;
802    let album_offset = params.number("albumOffset", 0, usize::MAX)?;
803    let song_count = params.number("songCount", 20, SEARCH_LIMIT)?;
804    let song_offset = params.number("songOffset", 0, usize::MAX)?;
805    let matching_artists: Vec<_> = index
806        .artists
807        .iter()
808        .map(|row| {
809            wanted.is_empty()
810                || text::normalize(&row.entity.name).contains(&wanted)
811                || row
812                    .entity
813                    .aliases
814                    .iter()
815                    .any(|alias| text::normalize(alias).contains(&wanted))
816        })
817        .collect();
818    let artist_matches = |id: Id| matching_artists.get(id as usize).copied().unwrap_or(false);
819    let mut matching_albums = vec![false; index.albums.len()];
820    for artist in &index.artists {
821        if artist_matches(artist.entity.id) {
822            for album in &artist.albums {
823                if let Some(matched) = matching_albums.get_mut(*album as usize) {
824                    *matched = true;
825                }
826            }
827        }
828    }
829    let mut artists: Vec<_> = index
830        .artists
831        .iter()
832        .filter(|row| artist_matches(row.entity.id))
833        .collect();
834    artists.sort_by(|a, b| (a.sort.as_str(), &a.id).cmp(&(b.sort.as_str(), &b.id)));
835    let albums = index.sorted_albums().into_iter().filter(|row| {
836        row.sort.contains(&wanted)
837            || matching_albums
838                .get(row.entity.id as usize)
839                .copied()
840                .unwrap_or(false)
841    });
842    let mut songs: Vec<_> = index
843        .songs
844        .iter()
845        .filter(|row| {
846            row.sort.contains(&wanted)
847                || row.artists.iter().copied().any(artist_matches)
848                || row
849                    .entity
850                    .release_id
851                    .and_then(|id| index.albums.get(id as usize))
852                    .is_some_and(|album| album.sort.contains(&wanted))
853        })
854        .collect();
855    songs.sort_by(|a, b| (a.sort.as_str(), &a.id).cmp(&(b.sort.as_str(), &b.id)));
856    Ok(json!({"searchResult3": {
857        "artist": artists.into_iter().skip(artist_offset).take(artist_count)
858            .map(|row| index.artist(row)).collect::<Vec<_>>(),
859        "album": albums.skip(album_offset).take(album_count)
860            .map(|row| index.album(row)).collect::<Vec<_>>(),
861        "song": songs.into_iter().skip(song_offset).take(song_count)
862            .map(|row| index.song(row)).collect::<Vec<_>>(),
863    }}))
864}

866fn genres(index: &Index<'_>) -> Value {
867    let mut song_counts = vec![0usize; index.catalog.genres.len()];
868    let mut album_counts = vec![0usize; index.catalog.genres.len()];
869    for song in &index.songs {
870        for id in &song.genres {
871            if let Some(count) = song_counts.get_mut(*id as usize) {
872                *count += 1;
873            }
874        }
875    }
876    for album in &index.albums {
877        for id in &album.genres {
878            if let Some(count) = album_counts.get_mut(*id as usize) {
879                *count += 1;
880            }
881        }
882    }
883    let mut rows: Vec<_> = index.catalog.genres.iter().collect();
884    rows.sort_by_cached_key(|genre| (text::normalize(&genre.name), genre.key.as_str()));
885    json!({"genres": {"genre": rows.into_iter().map(|genre| json!({
886        "value": genre.name,
887        "songCount": song_counts.get(genre.id as usize).copied().unwrap_or_default(),
888        "albumCount": album_counts.get(genre.id as usize).copied().unwrap_or_default(),
889    })).collect::<Vec<_>>()}})
890}
