1//! Subsonic-shaped translation over the shared graph, never a second catalog.
2//!
3//! Revocable application keys authenticate ID3 browsing, original audio,
4//! sidecar artwork and owner-private annotations, playlists and declarations.
5//! Unsupported operations/options fail explicitly; native bearer sessions
6//! and the PCM/DSP WebSocket keep their separate contracts.

28async fn handle(
29    State(state): State<ApiState>,
30    RoutePath(method): RoutePath<String>,
31    request: Request,
32) -> Response {
33    // Axum's GET route also accepts HEAD. Subsonic GET operations can mutate
34    // personal data, so implicit HEAD execution must never reach the adapter.
35    if request.method() == axum::http::Method::HEAD {
36        return StatusCode::METHOD_NOT_ALLOWED.into_response();
37    }
38    let method = method.strip_suffix(".view").unwrap_or(&method).to_string();
39    let headers = request.headers().clone();
40    let parameters = match parameters::read(request, &method).await {
41        Ok(parameters) => parameters,
42        Err(failure) => return protocol::response(Format::Xml, Err(failure)),
43    };
44    let format = match parameters.format() {
45        Ok(format) => format,
46        Err(failure) => return protocol::response(Format::Xml, Err(failure)),
47    };
48    let result = execute(state, &method, parameters, headers).await;
49    match result {
50        Ok(Answer::Protocol(payload)) => protocol::response(format, Ok(payload)),
51        Ok(Answer::Binary(response)) => response,
52        Err(failure) => protocol::response(format, Err(failure)),
53    }
54}
55
56enum Answer {
57    Protocol(Value),
58    Binary(Response),
59}
60
61async fn execute(
62    state: ApiState,
63    method: &str,
64    parameters: Parameters,
65    headers: HeaderMap,
66) -> Result<Answer, ProtocolError> {
67    parameters.common(method == "getOpenSubsonicExtensions")?;
68    if method == "getOpenSubsonicExtensions" {
69        parameters.allowed(&[])?;
70        return Ok(Answer::Protocol(json!({"openSubsonicExtensions": [
71            {"name": "apiKeyAuthentication", "versions": [1]},
72            {"name": "formPost", "versions": [1]}
73        ]})));
74    }
75    let identity = authentication::authenticate(&state, &parameters).await?;
76    let payload = match method {
77        "ping" => {
78            parameters.allowed(&[])?;
79            json!({})
80        }
81        "getLicense" => {
82            parameters.allowed(&[])?;
83            json!({"license": {"valid": true}})
84        }
85        "tokenInfo" => {
86            parameters.allowed(&[])?;
87            json!({"tokenInfo": {"username": identity.username}})
88        }
89        "getUser" => {
90            parameters.allowed(&["username"])?;
91            if !parameters
92                .required("username")?
93                .eq_ignore_ascii_case(&identity.username)
94            {
95                return Err(ProtocolError::new(50, "only your own user is available"));
96            }
97            let audio = identity.role != aede_core::accounts::Role::Auditor;
98            // Installation administration is not exposed by this adapter.
99            json!({"user": {"username": identity.username, "scrobblingEnabled": audio,
100                "adminRole": false, "settingsRole": false, "downloadRole": audio,
101                "uploadRole": false, "playlistRole": audio, "coverArtRole": true,
102                "commentRole": false, "podcastRole": false, "streamRole": audio,
103                "jukeboxRole": false, "shareRole": false, "videoConversionRole": false,
104                "folder": [1]}})
105        }
106        "getCoverArt" => {
107            return artwork::response(state, identity, parameters)
108                .await
109                .map(Answer::Binary);
110        }
111        "getScanStatus" => {
112            parameters.allowed(&[])?;
113            let running = state.scan_activity.is_running();
114            inspect(state.clone(), move |catalog| {
115                catalog::scan_status(catalog, running)
116            })
117            .await?
118        }
119        "getNowPlaying" => now_playing::list(&state, &identity, &parameters).await?,
120        "scrobble" => match parameters.get("submission") {
121            Some("false") => now_playing::report(&state, &identity, &parameters).await?,
122            None | Some("true") => {
123                personal::execute(state.clone(), identity.clone(), method, parameters).await?
124            }
125            _ => return Err(ProtocolError::new(10, "submission must be true or false")),
126        },
127        "getStarred2" | "star" | "unstar" | "setRating" | "getPlaylists" | "getPlaylist"
128        | "createPlaylist" | "updatePlaylist" | "deletePlaylist" => {
129            personal::execute(state.clone(), identity.clone(), method, parameters).await?
130        }
131        "stream" | "download" => {
132            parameters.allowed(&[
133                "id",
134                "maxBitRate",
135                "format",
136                "timeOffset",
137                "estimateContentLength",
138                "converted",
139                "videoSize",
140            ])?;
141            original_only(&parameters)?;
142            authentication::require_audio(&identity)?;
143            let wanted = parameters.required("id")?.to_string();
144            let track = inspect(state.clone(), move |catalog| {
145                catalog::track_reference(catalog, &wanted)
146            })
147            .await?;
148            return media::response(state, identity, track, headers, method == "download")
149                .await
150                .map(Answer::Binary);
151        }
152        _ => personal::catalog(state.clone(), identity.clone(), method, parameters).await?,
153    };
154    authentication::recheck(&state, &identity).await?;
155    Ok(Answer::Protocol(payload))
156}
157
158fn original_only(parameters: &Parameters) -> Result<(), ProtocolError> {
159    for name in ["maxBitRate", "timeOffset"] {
160        if parameters.get(name).is_some_and(|value| value != "0") {
161            return Err(ProtocolError::new(
162                0,
163                "transcoding and time offsets are not implemented; request the original",
164            ));
165        }
166    }
167    if parameters.get("format").is_some_and(|value| value != "raw")
168        || parameters.get("videoSize").is_some()
169        || parameters
170            .get("converted")
171            .is_some_and(|value| value != "false")
172        || parameters
173            .get("estimateContentLength")
174            .is_some_and(|value| !matches!(value, "true" | "false"))
175    {
176        return Err(ProtocolError::new(0, "only original audio is supported"));
177    }
178    Ok(())
179}

181async fn inspect<T: Send + 'static>(
182    state: ApiState,
183    operation: impl FnOnce(&Catalog) -> Result<T, ProtocolError> + Send + 'static,
184) -> Result<T, ProtocolError> {
185    let permit = state
186        .inspection_slots
187        .clone()
188        .try_acquire_owned()
189        .map_err(|_| ProtocolError::new(0, "catalog work is busy; retry shortly"))?;
190    tokio::task::spawn_blocking(move || {
191        let _permit = permit;
192        let guard = state.catalog.blocking_read();
193        operation(
194            guard
195                .as_ref()
196                .ok_or_else(|| ProtocolError::new(0, "catalog unavailable"))?,
197        )
198    })
199    .await
200    .map_err(|_| ProtocolError::new(0, "catalog work failed"))?
201}
