27const CHUNK_BYTES: usize = 32 * 1024;
28const QUEUED_CHUNKS: usize = 2;
29const OPEN_TIMEOUT: Duration = Duration::from_secs(10);
30const AUTH_INTERVAL: Duration = Duration::from_secs(1);
31const AUTH_TIMEOUT: Duration = Duration::from_secs(5);
32const SOURCE_PROGRESS_TIMEOUT: Duration = Duration::from_secs(10);

34fn requested_range(headers: &HeaderMap, size: u64) -> Result<Option<ByteRange>, ProtocolError> {
35    aede_devices::original::requested_range(headers, size)
36        .map_err(|message| ProtocolError::new(10, message))

39fn unavailable() -> ProtocolError {
40    ProtocolError::new(
41        70,
42        "Media source is unavailable or changed; refresh the catalog",
43    )
44}

46fn valid_opened(source: &TrackSource, metadata: &Metadata) -> bool {
47    metadata.is_file()
48        && metadata.len() == source.file.size
49        && clock::mtime_seconds(metadata) == source.file.mtime
50        && clock::mtime_subseconds(metadata) == source.mtime_subseconds
51}

53pub(super) fn same_file(left: &Metadata, right: &Metadata) -> bool {
54    #[cfg(unix)]
55    {
56        use std::os::unix::fs::MetadataExt as _;
57        left.dev() == right.dev() && left.ino() == right.ino()
58    }
59    #[cfg(not(unix))]
60    {
61        left.len() == right.len()
62            && matches!((left.modified(), right.modified()), (Ok(left), Ok(right)) if left == right)
63    }
64}

66struct Opened {
67    file: File,
68    source: TrackSource,
69    _permit: Arc<OwnedSemaphorePermit>,
70}
71
72fn open_source(source: TrackSource, permit: OwnedSemaphorePermit) -> Result<Opened, ProtocolError> {
73    validate_source(&source).map_err(|_| unavailable())?;
74    let before = fs::symlink_metadata(&source.path).map_err(|_| unavailable())?;
75    let file = File::open(&source.path).map_err(|_| unavailable())?;
76    let metadata = file.metadata().map_err(|_| unavailable())?;
77    if !valid_opened(&source, &metadata) || !same_file(&before, &metadata) {
78        return Err(unavailable());
79    }
80    validate_source(&source).map_err(|_| unavailable())?;
81    Ok(Opened {
82        file,
83        source,
84        _permit: Arc::new(permit),
85    })
86}

88/// Retrieve the exact catalogued original, without DSP or listening mutations.
89pub(super) async fn response(
90    state: ApiState,
91    identity: Identity,
92    track: EntityRef,
93    headers: HeaderMap,
94    download: bool,
95) -> Result<Response, ProtocolError> {
96    authentication::require_audio(&identity)?;
97    let source = {
98        let catalog = state.catalog.read().await;
99        let catalog = catalog.as_ref().ok_or_else(unavailable)?;
100        source_from_catalog(catalog, &track).map_err(|_| unavailable())?
101    };
102    let range = requested_range(&headers, source.file.size)?;
103    let permit = state
104        .playback_slots
105        .clone()
106        .try_acquire_owned()
107        .map_err(|_| ProtocolError::new(0, "Too many active audio transfers; retry later"))?;
108    let mut shutdown = state.shutdown.subscribe();
109    let opening = tokio::time::timeout(
110        OPEN_TIMEOUT,
111        tokio::task::spawn_blocking(move || open_source(source, permit)),
112    );
113    let opened = tokio::select! {
114        _ = shutdown.recv() => return Err(ProtocolError::new(0, "server is shutting down")),
115        opened = opening => opened.map_err(|_| unavailable())?.map_err(|_| unavailable())??,
116    };
117    tokio::select! {
118        _ = shutdown.recv() => return Err(ProtocolError::new(0, "server is shutting down")),
119        confirmation = tokio::time::timeout(AUTH_TIMEOUT, authentication::recheck(&state, &identity)) => {
120            confirmation.map_err(|_| ProtocolError::new(40, "API key could not be confirmed"))??;
121        }
122    }
123    let mut response = Response::builder()
124        .header(header::CACHE_CONTROL, "no-store")
125        .header(header::ACCEPT_RANGES, "bytes")
126        .header(
127            header::CONTENT_TYPE,
128            content_type(&opened.source.file.properties),
129        )
130        .header(header::X_CONTENT_TYPE_OPTIONS, "nosniff");
131    let Some(range) = range else {
132        return response
133            .status(StatusCode::RANGE_NOT_SATISFIABLE)
134            .header(
135                header::CONTENT_RANGE,
136                format!("bytes */{}", opened.source.file.size),
137            )
138            .header(header::CONTENT_LENGTH, "0")
139            .body(Body::empty())
140            .map_err(|_| ProtocolError::new(0, "Media response could not be prepared"));
141    };
142    if range.partial {
143        response = response.status(StatusCode::PARTIAL_CONTENT).header(
144            header::CONTENT_RANGE,
145            format!(
146                "bytes {}-{}/{}",
147                range.start,
148                range.start + range.length - 1,
149                opened.source.file.size
150            ),
151        );
152    }
153    if download {
154        response = response.header(
155            header::CONTENT_DISPOSITION,
156            format!(
157                "attachment; filename=\"audio.{}\"",
158                suffix(&opened.source.file.properties)
159            ),
160        );
161    }
162    let body = original_body(opened, range, state, identity, shutdown);
163    response
164        .header(header::CONTENT_LENGTH, range.length.to_string())
165        .body(Body::new(body))
166        .map_err(|_| ProtocolError::new(0, "Media response could not be prepared"))
167}

169fn read_original(
170    opened: &mut Opened,
171    range: ByteRange,
172    sender: mpsc::Sender<io::Result<Bytes>>,
173    cancelled: Arc<AtomicBool>,
174) -> io::Result<()> {
175    // The permit belongs to the real filesystem worker, not the HTTP waiter.
176    // Dropping a timed-out response therefore cannot admit unbounded reads.
177    opened.file.seek(SeekFrom::Start(range.start))?;
178    let mut remaining = range.length;
179    while remaining > 0 {
180        if cancelled.load(Ordering::Acquire) {
181            return Ok(());
182        }
183        let count = usize::try_from(remaining.min(CHUNK_BYTES as u64))
184            .map_err(|_| io::Error::other("invalid media chunk size"))?;
185        let mut bytes = vec![0; count];
186        opened.file.read_exact(&mut bytes)?;
187        remaining -= count as u64;
188        if remaining == 0 {
189            let metadata = opened.file.metadata()?;
190            if !valid_opened(&opened.source, &metadata)
191                || validate_source(&opened.source).is_err()
192                || !same_file(&fs::symlink_metadata(&opened.source.path)?, &metadata)
193            {
194                return Err(io::Error::other("media source changed during transfer"));
195            }
196        }
197        if sender.blocking_send(Ok(Bytes::from(bytes))).is_err() {
198            return Ok(());
199        }
200    }
201    Ok(())
202}

204type MediaChunk = io::Result<Bytes>;

206fn original_body(
207    opened: Opened,
208    range: ByteRange,
209    state: ApiState,
210    identity: Identity,
211    shutdown: broadcast::Receiver<()>,
212) -> OriginalBody {
213    let cancelled = Arc::new(AtomicBool::new(false));
214    let permit = opened._permit.clone();
215    let (producer, chunks) = mpsc::channel(QUEUED_CHUNKS);
216    let worker_cancelled = cancelled.clone();
217    tokio::task::spawn_blocking(move || {
218        let mut opened = opened;
219        if let Err(failure) = read_original(&mut opened, range, producer.clone(), worker_cancelled)
220        {
221            let _ = producer.blocking_send(Err(failure));
222        }
223    });
224    guarded_body(
225        chunks,
226        permit,
227        range.length,
228        cancelled,
229        state,
230        identity,
231        shutdown,
232    )
233}

235/// Supervise an already validated in-memory original or derived artwork.
236/// Its transfer slot remains held through consumer completion or cancellation.
237pub(super) fn buffered_body(
238    mut bytes: Bytes,
239    permit: Arc<OwnedSemaphorePermit>,
240    state: ApiState,
241    identity: Identity,
242    shutdown: broadcast::Receiver<()>,
243) -> Body {
244    let length = bytes.len() as u64;
245    let cancelled = Arc::new(AtomicBool::new(false));
246    let (producer, chunks) = mpsc::channel(QUEUED_CHUNKS);
247    let worker_cancelled = cancelled.clone();
248    let worker_permit = permit.clone();
249    tokio::spawn(async move {
250        let _permit = worker_permit;
251        while !bytes.is_empty() && !worker_cancelled.load(Ordering::Acquire) {
252            let chunk = bytes.split_to(bytes.len().min(CHUNK_BYTES));
253            if producer.send(Ok(chunk)).await.is_err() {
254                break;
255            }
256        }
257    });
258    Body::new(guarded_body(
259        chunks, permit, length, cancelled, state, identity, shutdown,
260    ))
261}

263fn guarded_body(
264    chunks: mpsc::Receiver<MediaChunk>,
265    permit: Arc<OwnedSemaphorePermit>,
266    length: u64,
267    cancelled: Arc<AtomicBool>,
268    state: ApiState,
269    identity: Identity,
270    shutdown: broadcast::Receiver<()>,
271) -> OriginalBody {
272    let (output, receiver) = mpsc::channel(QUEUED_CHUNKS);
273    tokio::spawn(forward_original(
274        chunks,
275        output,
276        cancelled.clone(),
277        state,
278        identity,
279        shutdown,
280    ));
281    OriginalBody {
282        receiver,
283        cancelled,
284        remaining: length,
285        ended: false,
286        _permit: permit,
287    }
288}

290async fn forward_original(
291    mut chunks: mpsc::Receiver<MediaChunk>,
292    output: mpsc::Sender<MediaChunk>,
293    cancelled: Arc<AtomicBool>,
294    state: ApiState,
295    identity: Identity,
296    mut shutdown: broadcast::Receiver<()>,
297) {
298    let mut authorization =
299        tokio::time::interval_at(tokio::time::Instant::now() + AUTH_INTERVAL, AUTH_INTERVAL);
300    authorization.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
301    let mut pending = None;
302    let mut source_finished = false;
303    let mut source_progress = tokio::time::Instant::now();
304    loop {
305        let source_timeout = tokio::time::sleep_until(source_progress + SOURCE_PROGRESS_TIMEOUT);
306        tokio::pin!(source_timeout);
307        tokio::select! {
308            _ = shutdown.recv() => break,
309            _ = output.closed() => break,
310            _ = &mut source_timeout, if pending.is_none() && !source_finished => break,
311            _ = authorization.tick() => {
312                let confirmation = tokio::select! {
313                    _ = shutdown.recv() => break,
314                    _ = output.closed() => break,
315                    confirmation = tokio::time::timeout(AUTH_TIMEOUT, authentication::recheck(&state, &identity)) => confirmation,
316                };
317                if !matches!(confirmation, Ok(Ok(()))) {
318                    break;
319                }
320            }
321            chunk = chunks.recv(), if pending.is_none() && !source_finished => match chunk {
322                Some(Ok(chunk)) => {
323                    source_progress = tokio::time::Instant::now();
324                    pending = Some(chunk);
325                }
326                Some(Err(_)) => break,
327                // Authorization must outlive the reader: small originals may
328                // already be buffered while the HTTP consumer is still idle.
329                None => source_finished = true,
330            },
331            // Reserving is cancellation-safe. Keeping the bytes outside this
332            // future prevents an authorization tick from losing file data.
333            reservation = output.reserve(), if pending.is_some() => match reservation {
334                Ok(slot) => {
335                    if let Some(chunk) = pending.take() {
336                        slot.send(Ok(chunk));
337                        source_progress = tokio::time::Instant::now();
338                    }
339                }
340                Err(_) => break,
341            },
342        }
343    }
344    cancelled.store(true, Ordering::Release);
345}

347struct OriginalBody {
348    receiver: mpsc::Receiver<MediaChunk>,
349    cancelled: Arc<AtomicBool>,
350    remaining: u64,
351    ended: bool,
352    _permit: Arc<OwnedSemaphorePermit>,
353}
354
355impl Drop for OriginalBody {
356    fn drop(&mut self) {
357        self.cancelled.store(true, Ordering::Release);
358    }
359}
360
361impl HttpBody for OriginalBody {
362    type Data = Bytes;
363    type Error = io::Error;
364
365    fn poll_frame(
366        self: Pin<&mut Self>,
367        context: &mut Context<'_>,
368    ) -> Poll<Option<io::Result<Frame<Bytes>>>> {
369        let body = self.get_mut();
370        if body.ended || body.remaining == 0 {
371            body.ended = true;
372            return Poll::Ready(None);
373        }
374        if body.cancelled.load(Ordering::Acquire) {
375            body.ended = true;
376            return Poll::Ready(Some(Err(io::Error::other("media transfer stopped"))));
377        }
378        match body.receiver.poll_recv(context) {
379            Poll::Ready(Some(Ok(bytes))) => {
380                let count = bytes.len() as u64;
381                if count > body.remaining {
382                    body.ended = true;
383                    return Poll::Ready(Some(Err(io::Error::other(
384                        "media transfer exceeded its range",
385                    ))));
386                }
387                body.remaining -= count;
388                Poll::Ready(Some(Ok(Frame::data(bytes))))
389            }
390            Poll::Ready(Some(Err(failure))) => {
391                body.ended = true;
392                Poll::Ready(Some(Err(failure)))
393            }
394            Poll::Ready(None) => {
395                body.ended = true;
396                if body.remaining == 0 {
397                    Poll::Ready(None)
398                } else {
399                    Poll::Ready(Some(Err(io::Error::other("media transfer ended early"))))
400                }
401            }
402            Poll::Pending => Poll::Pending,
403        }
404    }
405
406    fn is_end_stream(&self) -> bool {
407        self.ended || self.remaining == 0
408    }
409
410    fn size_hint(&self) -> SizeHint {
411        SizeHint::with_exact(self.remaining)
412    }
413}
