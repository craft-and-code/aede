33/// The MIME type of original audio, independent of its private filesystem path.
34pub(super) fn content_type(properties: &AudioProperties) -> &'static str {
35    match suffix(properties) {
36        "flac" => "audio/flac",
37        "mp3" => "audio/mpeg",
38        "ogg" | "opus" => "audio/ogg",
39        "wav" => "audio/wav",
40        "aiff" => "audio/aiff",
41        "m4a" => "audio/mp4",
42        "aac" => "audio/aac",
43        "wv" => "audio/wavpack",
44        _ => "application/octet-stream",
45    }
46}

48/// A conservative original-container suffix for the public song projection.
49pub(super) fn suffix(properties: &AudioProperties) -> &'static str {
50    let container = if properties.container.is_empty() {
51        properties.codec.as_str()
52    } else {
53        properties.container.as_str()
54    };
55    match container {
56        "flac" => "flac",
57        "mp3" => "mp3",
58        "ogg" => "ogg",
59        "opus" => "opus",
60        "wav" | "wave" => "wav",
61        "aiff" | "aif" => "aiff",
62        "mp4" | "m4a" | "m4b" => "m4a",
63        "aac" | "adts" => "aac",
64        "wavpack" | "wv" => "wv",
65        "ape" => "ape",
66        "dsf" => "dsf",
67        "dff" => "dff",
68        _ => "bin",
69    }
70}

79fn requested_range(headers: &HeaderMap, size: u64) -> Result<Option<ByteRange>, ProtocolError> {
80    let mut ranges = headers.get_all(header::RANGE).iter();
81    let range = ranges.next();
82    if ranges.next().is_some() {
83        return Err(ProtocolError::new(10, "Only one Range header is supported"));
84    }
85    let mut conditions = headers.get_all(header::IF_RANGE).iter();
86    let conditional = conditions.next().is_some();
87    if conditions.next().is_some() {
88        return Err(ProtocolError::new(
89            10,
90            "Only one If-Range header is supported",
91        ));
92    }
93    // No strong entity validator is published. An unrecognized If-Range
94    // cannot authorize appending bytes to a previously downloaded entity.
95    if conditional || range.is_none() {
96        return Ok(Some(ByteRange {
97            start: 0,
98            length: size,
99            partial: false,
100        }));
101    }
102    let value = range.and_then(|range| range.to_str().ok()).unwrap_or("");
103    Ok(parse_range(value, size))
104}
105
106fn parse_range(value: &str, size: u64) -> Option<ByteRange> {
107    let value = value.strip_prefix("bytes=")?;
108    let (start, end) = value.split_once('-')?;
109    if size == 0 || value.contains(',') {
110        return None;
111    }
112    let decimal = |value: &str| {
113        (!value.is_empty() && value.bytes().all(|byte| byte.is_ascii_digit()))
114            .then(|| value.parse::<u64>().ok())
115            .flatten()
116    };
117    let (start, end) = if start.is_empty() {
118        let suffix = decimal(end)?.min(size);
119        if suffix == 0 {
120            return None;
121        }
122        (size - suffix, size - 1)
123    } else {
124        let start = decimal(start)?;
125        if start >= size {
126            return None;
127        }
128        let end = if end.is_empty() {
129            size - 1
130        } else {
131            decimal(end)?.min(size - 1)
132        };
133        if end < start {
134            return None;
135        }
136        (start, end)
137    };
138    Some(ByteRange {
139        start,
140        length: end.checked_sub(start)?.checked_add(1)?,
141        partial: true,
142    })
143}

152fn valid_opened(source: &TrackSource, metadata: &Metadata) -> bool {
153    metadata.is_file()
154        && metadata.len() == source.file.size
155        && clock::mtime_seconds(metadata) == source.file.mtime
156        && clock::mtime_subseconds(metadata) == source.mtime_subseconds
157}

159pub(super) fn same_file(left: &Metadata, right: &Metadata) -> bool {
160    #[cfg(unix)]
161    {
162        use std::os::unix::fs::MetadataExt as _;
163        left.dev() == right.dev() && left.ino() == right.ino()
164    }
165    #[cfg(not(unix))]
166    {
167        left.len() == right.len()
168            && matches!((left.modified(), right.modified()), (Ok(left), Ok(right)) if left == right)
169    }
170}

172struct Opened {
173    file: File,
174    source: TrackSource,
175    _permit: Arc<OwnedSemaphorePermit>,
176}
177
178fn open_source(source: TrackSource, permit: OwnedSemaphorePermit) -> Result<Opened, ProtocolError> {
179    validate_source(&source).map_err(|_| unavailable())?;
180    let before = fs::symlink_metadata(&source.path).map_err(|_| unavailable())?;
181    let file = File::open(&source.path).map_err(|_| unavailable())?;
182    let metadata = file.metadata().map_err(|_| unavailable())?;
183    if !valid_opened(&source, &metadata) || !same_file(&before, &metadata) {
184        return Err(unavailable());
185    }
186    validate_source(&source).map_err(|_| unavailable())?;
187    Ok(Opened {
188        file,
189        source,
190        _permit: Arc::new(permit),
191    })
192}

194/// Retrieve the exact catalogued original, without DSP or listening mutations.
195pub(super) async fn response(
196    state: ApiState,
197    identity: Identity,
198    track: EntityRef,
199    headers: HeaderMap,
200    download: bool,
201) -> Result<Response, ProtocolError> {
202    authentication::require_audio(&identity)?;
203    let source = {
204        let catalog = state.catalog.read().await;
205        let catalog = catalog.as_ref().ok_or_else(unavailable)?;
206        source_from_catalog(catalog, &track).map_err(|_| unavailable())?
207    };
208    let range = requested_range(&headers, source.file.size)?;
209    let permit = state
210        .playback_slots
211        .clone()
212        .try_acquire_owned()
213        .map_err(|_| ProtocolError::new(0, "Too many active audio transfers; retry later"))?;
214    let mut shutdown = state.shutdown.subscribe();
215    let opening = tokio::time::timeout(
216        OPEN_TIMEOUT,
217        tokio::task::spawn_blocking(move || open_source(source, permit)),
218    );
219    let opened = tokio::select! {
220        _ = shutdown.recv() => return Err(ProtocolError::new(0, "server is shutting down")),
221        opened = opening => opened.map_err(|_| unavailable())?.map_err(|_| unavailable())??,
222    };
223    tokio::select! {
224        _ = shutdown.recv() => return Err(ProtocolError::new(0, "server is shutting down")),
225        confirmation = tokio::time::timeout(AUTH_TIMEOUT, authentication::recheck(&state, &identity)) => {
226            confirmation.map_err(|_| ProtocolError::new(40, "API key could not be confirmed"))??;
227        }
228    }
229    let mut response = Response::builder()
230        .header(header::CACHE_CONTROL, "no-store")
231        .header(header::ACCEPT_RANGES, "bytes")
232        .header(
233            header::CONTENT_TYPE,
234            content_type(&opened.source.file.properties),
235        )
236        .header(header::X_CONTENT_TYPE_OPTIONS, "nosniff");
237    let Some(range) = range else {
238        return response
239            .status(StatusCode::RANGE_NOT_SATISFIABLE)
240            .header(
241                header::CONTENT_RANGE,
242                format!("bytes */{}", opened.source.file.size),
243            )
244            .header(header::CONTENT_LENGTH, "0")
245            .body(Body::empty())
246            .map_err(|_| ProtocolError::new(0, "Media response could not be prepared"));
247    };
248    if range.partial {
249        response = response.status(StatusCode::PARTIAL_CONTENT).header(
250            header::CONTENT_RANGE,
251            format!(
252                "bytes {}-{}/{}",
253                range.start,
254                range.start + range.length - 1,
255                opened.source.file.size
256            ),
257        );
258    }
259    if download {
260        response = response.header(
261            header::CONTENT_DISPOSITION,
262            format!(
263                "attachment; filename=\"audio.{}\"",
264                suffix(&opened.source.file.properties)
265            ),
266        );
267    }
268    let body = original_body(opened, range, state, identity, shutdown);
269    response
270        .header(header::CONTENT_LENGTH, range.length.to_string())
271        .body(Body::new(body))
272        .map_err(|_| ProtocolError::new(0, "Media response could not be prepared"))
273}

275fn read_original(
276    opened: &mut Opened,
277    range: ByteRange,
278    sender: mpsc::Sender<io::Result<Bytes>>,
279    cancelled: Arc<AtomicBool>,
280) -> io::Result<()> {
281    // The permit belongs to the real filesystem worker, not the HTTP waiter.
282    // Dropping a timed-out response therefore cannot admit unbounded reads.
283    opened.file.seek(SeekFrom::Start(range.start))?;
284    let mut remaining = range.length;
285    while remaining > 0 {
286        if cancelled.load(Ordering::Acquire) {
287            return Ok(());
288        }
289        let count = usize::try_from(remaining.min(CHUNK_BYTES as u64))
290            .map_err(|_| io::Error::other("invalid media chunk size"))?;
291        let mut bytes = vec![0; count];
292        opened.file.read_exact(&mut bytes)?;
293        remaining -= count as u64;
294        if remaining == 0 {
295            let metadata = opened.file.metadata()?;
296            if !valid_opened(&opened.source, &metadata)
297                || validate_source(&opened.source).is_err()
298                || !same_file(&fs::symlink_metadata(&opened.source.path)?, &metadata)
299            {
300                return Err(io::Error::other("media source changed during transfer"));
301            }
302        }
303        if sender.blocking_send(Ok(Bytes::from(bytes))).is_err() {
304            return Ok(());
305        }
306    }
307    Ok(())
308}

312fn original_body(
313    opened: Opened,
314    range: ByteRange,
315    state: ApiState,
316    identity: Identity,
317    shutdown: broadcast::Receiver<()>,
318) -> OriginalBody {
319    let cancelled = Arc::new(AtomicBool::new(false));
320    let permit = opened._permit.clone();
321    let (producer, chunks) = mpsc::channel(QUEUED_CHUNKS);
322    let worker_cancelled = cancelled.clone();
323    tokio::task::spawn_blocking(move || {
324        let mut opened = opened;
325        if let Err(failure) = read_original(&mut opened, range, producer.clone(), worker_cancelled)
326        {
327            let _ = producer.blocking_send(Err(failure));
328        }
329    });
330    guarded_body(
331        chunks,
332        permit,
333        range.length,
334        cancelled,
335        state,
336        identity,
337        shutdown,
338    )
339}

341/// Supervise an already validated in-memory original or derived artwork.
342/// Its transfer slot remains held through consumer completion or cancellation.
343pub(super) fn buffered_body(
344    mut bytes: Bytes,
345    permit: Arc<OwnedSemaphorePermit>,
346    state: ApiState,
347    identity: Identity,
348    shutdown: broadcast::Receiver<()>,
349) -> Body {
350    let length = bytes.len() as u64;
351    let cancelled = Arc::new(AtomicBool::new(false));
352    let (producer, chunks) = mpsc::channel(QUEUED_CHUNKS);
353    let worker_cancelled = cancelled.clone();
354    let worker_permit = permit.clone();
355    tokio::spawn(async move {
356        let _permit = worker_permit;
357        while !bytes.is_empty() && !worker_cancelled.load(Ordering::Acquire) {
358            let chunk = bytes.split_to(bytes.len().min(CHUNK_BYTES));
359            if producer.send(Ok(chunk)).await.is_err() {
360                break;
361            }
362        }
363    });
364    Body::new(guarded_body(
365        chunks, permit, length, cancelled, state, identity, shutdown,
366    ))
367}

369fn guarded_body(
370    chunks: mpsc::Receiver<MediaChunk>,
371    permit: Arc<OwnedSemaphorePermit>,
372    length: u64,
373    cancelled: Arc<AtomicBool>,
374    state: ApiState,
375    identity: Identity,
376    shutdown: broadcast::Receiver<()>,
377) -> OriginalBody {
378    let (output, receiver) = mpsc::channel(QUEUED_CHUNKS);
379    tokio::spawn(forward_original(
380        chunks,
381        output,
382        cancelled.clone(),
383        state,
384        identity,
385        shutdown,
386    ));
387    OriginalBody {
388        receiver,
389        cancelled,
390        remaining: length,
391        ended: false,
392        _permit: permit,
393    }
394}

396async fn forward_original(
397    mut chunks: mpsc::Receiver<MediaChunk>,
398    output: mpsc::Sender<MediaChunk>,
399    cancelled: Arc<AtomicBool>,
400    state: ApiState,
401    identity: Identity,
402    mut shutdown: broadcast::Receiver<()>,
403) {
404    let mut authorization =
405        tokio::time::interval_at(tokio::time::Instant::now() + AUTH_INTERVAL, AUTH_INTERVAL);
406    authorization.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
407    let mut pending = None;
408    let mut source_finished = false;
409    let mut source_progress = tokio::time::Instant::now();
410    loop {
411        let source_timeout = tokio::time::sleep_until(source_progress + SOURCE_PROGRESS_TIMEOUT);
412        tokio::pin!(source_timeout);
413        tokio::select! {
414            _ = shutdown.recv() => break,
415            _ = output.closed() => break,
416            _ = &mut source_timeout, if pending.is_none() && !source_finished => break,
417            _ = authorization.tick() => {
418                let confirmation = tokio::select! {
419                    _ = shutdown.recv() => break,
420                    _ = output.closed() => break,
421                    confirmation = tokio::time::timeout(AUTH_TIMEOUT, authentication::recheck(&state, &identity)) => confirmation,
422                };
423                if !matches!(confirmation, Ok(Ok(()))) {
424                    break;
425                }
426            }
427            chunk = chunks.recv(), if pending.is_none() && !source_finished => match chunk {
428                Some(Ok(chunk)) => {
429                    source_progress = tokio::time::Instant::now();
430                    pending = Some(chunk);
431                }
432                Some(Err(_)) => break,
433                // Authorization must outlive the reader: small originals may
434                // already be buffered while the HTTP consumer is still idle.
435                None => source_finished = true,
436            },
437            // Reserving is cancellation-safe. Keeping the bytes outside this
438            // future prevents an authorization tick from losing file data.
439            reservation = output.reserve(), if pending.is_some() => match reservation {
440                Ok(slot) => {
441                    if let Some(chunk) = pending.take() {
442                        slot.send(Ok(chunk));
443                        source_progress = tokio::time::Instant::now();
444                    }
445                }
446                Err(_) => break,
447            },
448        }
449    }
450    cancelled.store(true, Ordering::Release);
451}

453struct OriginalBody {
454    receiver: mpsc::Receiver<MediaChunk>,
455    cancelled: Arc<AtomicBool>,
456    remaining: u64,
457    ended: bool,
458    _permit: Arc<OwnedSemaphorePermit>,
459}
460
461impl Drop for OriginalBody {
462    fn drop(&mut self) {
463        self.cancelled.store(true, Ordering::Release);
464    }
465}
466
467impl HttpBody for OriginalBody {
468    type Data = Bytes;
469    type Error = io::Error;
470
471    fn poll_frame(
472        self: Pin<&mut Self>,
473        context: &mut Context<'_>,
474    ) -> Poll<Option<io::Result<Frame<Bytes>>>> {
475        let body = self.get_mut();
476        if body.ended || body.remaining == 0 {
477            body.ended = true;
478            return Poll::Ready(None);
479        }
480        if body.cancelled.load(Ordering::Acquire) {
481            body.ended = true;
482            return Poll::Ready(Some(Err(io::Error::other("media transfer stopped"))));
483        }
484        match body.receiver.poll_recv(context) {
485            Poll::Ready(Some(Ok(bytes))) => {
486                let count = bytes.len() as u64;
487                if count > body.remaining {
488                    body.ended = true;
489                    return Poll::Ready(Some(Err(io::Error::other(
490                        "media transfer exceeded its range",
491                    ))));
492                }
493                body.remaining -= count;
494                Poll::Ready(Some(Ok(Frame::data(bytes))))
495            }
496            Poll::Ready(Some(Err(failure))) => {
497                body.ended = true;
498                Poll::Ready(Some(Err(failure)))
499            }
500            Poll::Ready(None) => {
501                body.ended = true;
502                if body.remaining == 0 {
503                    Poll::Ready(None)
504                } else {
505                    Poll::Ready(Some(Err(io::Error::other("media transfer ended early"))))
506                }
507            }
508            Poll::Pending => Poll::Pending,
509        }
510    }
511
512    fn is_end_stream(&self) -> bool {
513        self.ended || self.remaining == 0
514    }
515
516    fn size_hint(&self) -> SizeHint {
517        SizeHint::with_exact(self.remaining)
518    }
519}
