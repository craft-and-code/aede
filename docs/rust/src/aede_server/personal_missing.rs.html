1//! Owner-private missing-album decisions reuse the CLI's derived core report.
2use super::*;
3use aede_core::missing;
4use aede_core::sources::Sources;
5use serde_json::{Value, json};
6#[path = "personal_missing_cover.rs"]
7mod cover;
8pub(super) fn routes() -> Router<ApiState> {
9    Router::new()
10        .route("/api/me/v1/missing", get(list).put(update))
11        .merge(cover::routes())
12}
13#[derive(Default, Deserialize)]
14#[serde(deny_unknown_fields)]
15struct MissingQuery {
16    offset: Option<String>,
17    limit: Option<String>,
18    include_set_aside: Option<bool>,
19}
20#[derive(Deserialize)]
21#[serde(deny_unknown_fields)]
22struct Decision {
23    release_group: String,
24    set_aside: bool,
25}
26fn rows(
27    catalog: &Catalog,
28    sources: &Sources,
29    data: &UserData,
30    owner: &str,
31    include_aside: bool,
32) -> Result<Vec<Value>, ApiError> {
33    let aside: Vec<_> = data
34        .set_aside
35        .iter()
36        .filter(|row| row.owner == owner)
37        .cloned()
38        .collect();
39    missing::absent(catalog, sources, &aside, true).into_iter()
40        .filter(|row| row.known.is_studio_album() && (include_aside || !row.set_aside))
41        .map(|row| {
42            let artist = EntityRef::of(catalog, EntityKind::Artist, row.artist_id).ok_or_else(unavailable)?;
43            let year = row.known.first_released.as_deref().and_then(|date| date.get(..4)).and_then(|year| year.parse::<u32>().ok());
44            Ok(json!({"artist":artist.to_token(),"artist_name":row.artist,"release_group":row.known.mbid,"title":row.known.title,"year":year,"set_aside":row.set_aside,"source":"musicbrainz","url":format!("https://musicbrainz.org/release-group/{}",row.known.mbid)}))
45        }).collect()
46}
47async fn list(
48    State(state): State<ApiState>,
49    query: Result<Query<MissingQuery>, QueryRejection>,
50    request: Request,
51) -> Result<Json<Page<Value>>, ApiError> {
52    let owner = PersonalOwner::from_request(&state, &request)?;
53    empty_body(&body(request).await?)?;
54    let Query(query) = query.map_err(|failure| invalid_query(failure.body_text()))?;
55    let (offset, limit) = page_bounds(query.offset.as_deref(), query.limit.as_deref())?;
56    let data_dir = state.data_dir.clone();
57    spawn_personal(state.inspection_slots.clone(), move || {
58        let (_lock, catalog, data) = locked_personal(&state)?;
59        owner.recheck(&state)?;
60        let sources = crate::inspection::held_sources(&data_dir)?;
61        let all = rows(
62            &catalog,
63            &sources,
64            &data,
65            &owner.id,
66            query.include_set_aside.unwrap_or(false),
67        )?;
68        let total = all.len();
69        owner.admit(
70            &state,
71            Json(Page::new(
72                all.into_iter().skip(offset).take(limit).collect(),
73                total,
74                offset,
75                limit,
76                catalog.scanned_at,
77            )),
78        )
79    })?
80    .await
81    .map_err(|_| unavailable())?
82}
83async fn update(State(state): State<ApiState>, request: Request) -> Result<Json<Value>, ApiError> {
84    no_query(&request)?;
85    let owner = PersonalOwner::from_request(&state, &request)?;
86    owner.require_mutation()?;
87    let input: Decision = json_body(request).await?;
88    if input.release_group.len() != 36
89        || !input.release_group.bytes().enumerate().all(|(i, b)| {
90            if [8, 13, 18, 23].contains(&i) {
91                b == b'-'
92            } else {
93                b.is_ascii_hexdigit()
94            }
95        })
96    {
97        return Err(invalid_query("release_group must be a MusicBrainz UUID"));
98    }
99    let data_dir = state.data_dir.clone();
100    spawn_personal(state.inspection_slots.clone(), move || {
101        locked_update(&data_dir, &state, &owner, |catalog, data| {
102            let sources = crate::inspection::held_sources(&data_dir)?;
103            let found = rows(catalog, &sources, data, &owner.id, true)?
104                .into_iter()
105                .find(|row| row["release_group"] == input.release_group);
106            if input.set_aside {
107                let row = found.ok_or_else(|| {
108                    error(
109                        StatusCode::NOT_FOUND,
110                        "missing_not_found",
111                        "no missing studio album matches this identifier",
112                    )
113                })?;
114                if !data
115                    .set_aside
116                    .iter()
117                    .any(|row| row.owner == owner.id && row.release_group == input.release_group)
118                {
119                    if data
120                        .set_aside
121                        .iter()
122                        .filter(|row| row.owner == owner.id)
123                        .count()
124                        >= 4096
125                    {
126                        return Err(invalid_query("too many set-aside decisions"));
127                    }
128                    data.set_aside.push(user::SetAside {
129                        owner: owner.id.clone(),
130                        release_group: input.release_group.clone(),
131                        title: row["title"].as_str().ok_or_else(unavailable)?.into(),
132                        created_at: clock::now_seconds(),
133                    });
134                }
135            } else {
136                data.set_aside.retain(|row| {
137                    row.owner != owner.id || row.release_group != input.release_group
138                });
139            }
140            Ok(Json(
141                json!({"release_group":input.release_group,"set_aside":input.set_aside}),
142            ))
143        })
144    })?
145    .await
146    .map_err(|_| unavailable())?
147}
