1//! Opt-in, bounded HTTP summaries. Never record URIs, queries, headers or bodies.
2use axum::{
3    extract::{MatchedPath, Request, State},
4    middleware::Next,
5    response::Response,
6};
7use serde::Serialize;
8use std::io::Write;
9use std::{
10    sync::mpsc::{self, SyncSender},
11    time::Instant,
12};
13
14#[derive(Clone)]
15pub(super) struct AccessLog(SyncSender<Record>);
16#[derive(Serialize)]
17#[serde(rename_all = "camelCase")]
18pub(super) struct Record {
19    method: &'static str,
20    route: String,
21    status: u16,
22    elapsed_ms: u64,
23}
24#[derive(Clone)]
25struct RouteLabel(String);
26impl AccessLog {
27    pub(super) fn from_environment() -> Option<Self> {
28        if std::env::var_os("AEDE_REQUEST_LOG").as_deref() != Some(std::ffi::OsStr::new("1")) {
29            return None;
30        }
31        let (sender, receiver) = mpsc::sync_channel::<Record>(128);
32        // A slow terminal must not block the async HTTP workers or grow an unbounded queue.
33        std::thread::spawn(move || {
34            for record in receiver {
35                if let Ok(line) = serde_json::to_string(&record)
36                    && writeln!(std::io::stderr().lock(), "AEDE_HTTP {line}").is_err()
37                {
38                    break;
39                }
40            }
41        });
42        Some(Self(sender))
43    }
44}
45pub(super) async fn label(request: Request, next: Next) -> Response {
46    let route = request
47        .extensions()
48        .get::<MatchedPath>()
49        .map(|path| path.as_str().to_owned());
50    let mut response = next.run(request).await;
51    if let Some(route) = route {
52        response.extensions_mut().insert(RouteLabel(route));
53    }
54    response
55}
56pub(super) async fn record(State(log): State<AccessLog>, request: Request, next: Next) -> Response {
57    let method = match request.method().as_str() {
58        "GET" => "GET",
59        "HEAD" => "HEAD",
60        "POST" => "POST",
61        "PUT" => "PUT",
62        "PATCH" => "PATCH",
63        "DELETE" => "DELETE",
64        "OPTIONS" => "OPTIONS",
65        _ => "OTHER",
66    };
67    let start = Instant::now();
68    let response = next.run(request).await;
69    let record = Record {
70        method,
71        route: response
72            .extensions()
73            .get::<RouteLabel>()
74            .map_or_else(|| "unmatched/rejected".into(), |label| label.0.clone()),
75        status: response.status().as_u16(),
76        elapsed_ms: start.elapsed().as_millis().min(u64::MAX as u128) as u64,
77    };
78    // Logs are diagnostic, not an audit ledger: saturated queues drop summaries.
79    let _ = log.0.try_send(record);
80    response
81}
