27#[derive(Serialize)]
28struct Published {
29    id: String,
30    name: String,
31    filename: String,
32    content: String,
33    #[serde(skip_serializing_if = "Vec::is_empty")]
34    library_files: Vec<String>,
35}
36fn target_path(data_dir: &Path, owner: &str, id: &str) -> Result<PathBuf, ApiError> {
37    if owner.is_empty()
38        || !owner
39            .bytes()
40            .all(|byte| byte.is_ascii_alphanumeric() || byte == b'-' || byte == b'_')
41        || !id.starts_with("playlist-")
42        || !id
43            .bytes()
44            .all(|byte| byte.is_ascii_alphanumeric() || byte == b'-')
45    {
46        return Err(invalid_query("invalid private playlist identity"));
47    }
48    Ok(data_dir
49        .join("playlists")
50        .join(owner)
51        .join(format!("{id}.m3u")))
52}
53fn published(catalog: &Catalog, playlist: &user::Playlist) -> Result<Published, ApiError> {
54    let tracks = playlist
55        .tracks
56        .iter()
57        .map(|reference| {
58            reference
59                .resolve(catalog)
60                .filter(|_| reference.kind == EntityKind::Track)
61                .ok_or_else(|| {
62                    error(
63                        StatusCode::CONFLICT,
64                        "playlist_changed",
65                        "a stored track is no longer available",
66                    )
67                })
68        })
69        .collect::<Result<Vec<_>, _>>()?;
70    let content = aede_core::playlist::try_render(
71        catalog,
72        &tracks,
73        None,
74        aede_core::playlist::Style::Extended,
75    )
76    .map_err(|message| error(StatusCode::BAD_REQUEST, "m3u_refused", message))?;
77    if content.len() > 8 * 1024 * 1024 {
78        return Err(error(
79            StatusCode::PAYLOAD_TOO_LARGE,
80            "m3u_too_large",
81            "playlist is too large",
82        ));
83    }
84    Ok(Published {
85        id: playlist.id.clone(),
86        name: playlist.name.clone(),
87        filename: format!("{}.m3u", playlist.id),
88        content,
89        library_files: Vec::new(),
90    })
91}
92async fn save(
93    State(state): State<ApiState>,
94    request: Request,
95) -> Result<Json<Published>, ApiError> {
96    no_query(&request)?;
97    let owner = PersonalOwner::from_request(&state, &request)?;
98    owner.require_mutation()?;
99    let publication_admin = require_admin(&state, &request).map(|_| ());
100    let input: Selection = json_body(request).await?;
101    if input.library {
102        publication_admin?;
103    }
104    validate_collection_name(&input.name)?;
105    if input.library && input.release.is_none() {
106        return Err(invalid_query("library publication requires an album"));
107    }
108    let data_dir = state.data_dir.clone();
109    spawn_personal(state.inspection_slots.clone(), move || {
110        let (_guard, catalog, mut data) = locked_personal(&state)?;
111        owner.recheck(&state)?;
112        let (album, tracks) =
113            selection(&catalog, input.release.as_deref(), input.tracks.as_deref())?;
114        let mut answer = save_selection(
115            &catalog,
116            &mut data,
117            &data_dir,
118            &owner.id,
119            &input.name,
120            tracks,
121            || owner.recheck(&state),
122        )?;
123        if input.library {
124            owner.recheck(&state)?;
125            answer.library_files = publish_library(
126                &catalog,
127                album.ok_or_else(|| invalid_query("album required"))?,
128            )?;
129        }
130        owner.admit(&state, Json(answer))
131    })?
132    .await
133    .map_err(|_| unavailable())?
134}
135fn selection(
136    catalog: &Catalog,
137    release: Option<&str>,
138    tracks: Option<&[String]>,
139) -> Result<(Option<Id>, Vec<EntityRef>), ApiError> {
140    match (release, tracks) {
141        (Some(token), None) if token.len() <= MAX_REFERENCE_BYTES => {
142            let id = crate::catalog::filter_reference(catalog, token, EntityKind::Release)?;
143            let tracks = catalog
144                .release(id)
145                .ok_or_else(unavailable)?
146                .track_ids
147                .iter()
148                .map(|id| EntityRef::of(catalog, EntityKind::Track, *id).ok_or_else(unavailable))
149                .collect::<Result<_, _>>()?;
150            Ok((Some(id), tracks))
151        }
152        (None, Some(tracks)) if !tracks.is_empty() && tracks.len() <= 4096 => {
153            let tracks = tracks
154                .iter()
155                .map(|token| {
156                    if token.len() > MAX_REFERENCE_BYTES {
157                        return Err(invalid_query("track reference too long"));
158                    }
159                    let id = crate::catalog::filter_reference(catalog, token, EntityKind::Track)?;
160                    EntityRef::of(catalog, EntityKind::Track, id).ok_or_else(unavailable)
161                })
162                .collect::<Result<_, _>>()?;
163            Ok((None, tracks))
164        }
165        _ => Err(invalid_query(
166            "select exactly one album or a nonempty ordered track list",
167        )),
168    }
169}
170// Repeated explicit saves restore the same retained selection instead of consuming playlist slots.
171fn save_selection(
172    catalog: &Catalog,
173    data: &mut UserData,
174    data_dir: &Path,
175    owner: &str,
176    name: &str,
177    tracks: Vec<EntityRef>,
178    confirm: impl Fn() -> Result<(), ApiError>,
179) -> Result<Published, ApiError> {
180    let retained = data
181        .playlists
182        .iter()
183        .rev()
184        .find(|p| p.owner == owner && p.name == name && p.tracks == tracks)
185        .map(|p| p.id.clone());
186    let existing = retained.is_some();
187    let id = match retained {
188        Some(id) => id,
189        None => data
190            .create_playlist(owner, name, tracks, clock::now_seconds())
191            .map_err(|message| error(StatusCode::BAD_REQUEST, "playlist_refused", message))?,
192    };
193    let answer = published(catalog, data.playlist(owner, &id).ok_or_else(unavailable)?)?;
194    let path = target_path(data_dir, owner, &id)?;
195    confirm()?;
196    let folder = path.parent().ok_or_else(unavailable)?;
197    std::fs::create_dir_all(folder).map_err(|_| unavailable())?;
198    if !std::fs::symlink_metadata(folder).is_ok_and(|m| m.is_dir() && !m.file_type().is_symlink()) {
199        return Err(unavailable());
200    }
201    #[cfg(unix)]
202    {
203        use std::os::unix::fs::PermissionsExt;
204        std::fs::set_permissions(folder, std::fs::Permissions::from_mode(0o700))
205            .map_err(|_| unavailable())?;
206    }
207    let created =
208        store::write_new_atomic(&path, answer.content.as_bytes()).map_err(|_| unavailable())?;
209    if !created {
210        let metadata = std::fs::symlink_metadata(&path).map_err(|_| unavailable())?;
211        if !metadata.is_file()
212            || metadata.file_type().is_symlink()
213            || metadata.len() != answer.content.len() as u64
214            || {
215                use std::io::Read;
216                let mut held = Vec::new();
217                std::fs::File::open(&path)
218                    .map_err(|_| unavailable())?
219                    .take(8 * 1024 * 1024 + 1)
220                    .read_to_end(&mut held)
221                    .map_err(|_| unavailable())?;
222                held != answer.content.as_bytes()
223            }
224        {
225            return Err(error(
226                StatusCode::CONFLICT,
227                "playlist_exists",
228                "existing playlist file differs; it has not been overwritten",
229            ));
230        }
231    }
232    if !existing {
233        let saved = confirm().and_then(|_| {
234            user::save(data, &user::user_path(data_dir)).map_err(|_| {
235                error(
236                    StatusCode::INTERNAL_SERVER_ERROR,
237                    "store_error",
238                    "private playlist could not be saved",
239                )
240            })
241        });
242        if let Err(failure) = saved {
243            if created && let Err(cleanup) = std::fs::remove_file(path) {
244                eprintln!("API unregistered private playlist cleanup failed: {cleanup}");
245            }
246            return Err(failure);
247        }
248    }
249    Ok(answer)
250}
251async fn list(
252    State(state): State<ApiState>,
253    input: Result<Query<files::Parameters>, QueryRejection>,
254    request: Request,
255) -> Result<Json<Page<Summary>>, ApiError> {
256    let owner = PersonalOwner::from_request(&state, &request)?;
257    empty_body(&body(request).await?)?;
258    let (offset, limit, q) = files::parameters(parsed_query(input)?)?;
259    spawn_personal(state.inspection_slots.clone(), move || {
260        let (_guard, catalog, data) = locked_personal(&state)?;
261        owner.recheck(&state)?;
262        let own = unique_summaries(&data, &owner.id)
263            .into_iter()
264            .filter(|row| {
265                !crate::parental::supervised(&state)
266                    || data.playlist(&owner.id, &row.id).is_some_and(|playlist| {
267                        playlist
268                            .tracks
269                            .iter()
270                            .all(|track| track.resolve(&catalog).is_some())
271                    })
272            })
273            .filter(|row| {
274                q.as_ref()
275                    .is_none_or(|q| text::normalize(&row.name).contains(q))
276            })
277            .collect::<Vec<_>>();
278        let items = own.iter().skip(offset).take(limit).cloned().collect();
279        owner.admit(
280            &state,
281            Json(Page::new(
282                items,
283                own.len(),
284                offset,
285                limit,
286                catalog.scanned_at,
287            )),
288        )
289    })?
290    .await
291    .map_err(|_| unavailable())?
292}
293#[derive(Clone, Serialize)]
294struct Summary {
295    id: String,
296    name: String,
297    track_count: usize,
298    duplicates: usize,
299}
300// Legacy repeated saves remain stored; only exact ordered selections with the same name share one row.
301fn unique_summaries(data: &UserData, owner: &str) -> Vec<Summary> {
302    let mut rows: Vec<Summary> = Vec::new();
303    let mut keys = BTreeMap::new();
304    for playlist in data.playlists.iter().rev().filter(|p| p.owner == owner) {
305        let key = (
306            playlist.name.clone(),
307            playlist
308                .tracks
309                .iter()
310                .map(EntityRef::to_token)
311                .collect::<Vec<_>>(),
312        );
313        if let Some(index) = keys.get(&key).copied() {
314            let row: &mut Summary = &mut rows[index];
315            row.duplicates += 1;
316        } else {
317            keys.insert(key, rows.len());
318            rows.push(Summary {
319                id: playlist.id.clone(),
320                name: playlist.name.clone(),
321                track_count: playlist.tracks.len(),
322                duplicates: 1,
323            });
324        }
325    }
326    rows
327}
328async fn download(
329    State(state): State<ApiState>,
330    axum::extract::Path(id): axum::extract::Path<String>,
331    request: Request,
332) -> Result<Json<Published>, ApiError> {
333    no_query(&request)?;
334    let owner = PersonalOwner::from_request(&state, &request)?;
335    empty_body(&body(request).await?)?;
336    let data_dir = state.data_dir.clone();
337    target_path(&data_dir, &owner.id, &id)?;
338    spawn_personal(state.inspection_slots.clone(), move || {
339        let (_guard, catalog, data) = locked_personal(&state)?;
340        owner.recheck(&state)?;
341        let playlist = data.playlist(&owner.id, &id).ok_or_else(|| {
342            error(
343                StatusCode::NOT_FOUND,
344                "playlist_not_found",
345                "no private playlist has this identity",
346            )
347        })?;
348        owner.admit(&state, Json(published(&catalog, playlist)?))
349    })?
350    .await
351    .map_err(|_| unavailable())?
352}
