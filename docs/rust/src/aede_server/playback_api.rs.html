1//! Authenticated, bounded PCM playback for remote clients.
2//!
3//! This module transports processed PCM only. It deliberately does not own a
4//! device or a second audio pipeline. Opt-in controls manage source positioning,
5//! queue revisions and private checkpoints; decoding, stereo
6//! downmix, normalization, tone controls, rate conversion, and the final
7//! output guard come from `aede-core` and `aede-dsp`, just as they do for local
8//! playback. The client acknowledges cumulative output frames, so listening
9//! history represents audio it consumed rather than audio the server merely
10//! wrote to a socket buffer.

60async fn playback(
61    ws: WebSocketUpgrade,
62    State(state): State<ApiState>,
63    request: Request,
64) -> Result<Response, ApiError> {
65    personal::no_query(&request)?;
66    let principal = auth::request_principal(&request)?;
67    // Playback creates private listening history, so the explicit read-only
68    // role must be refused before the protocol is upgraded.
69    auth::require_mutation(&principal)?;
70    let permit = state
71        .playback_slots
72        .clone()
73        .try_acquire_owned()
74        .map_err(|_| {
75            error(
76                StatusCode::SERVICE_UNAVAILABLE,
77                "playback_busy",
78                "too many active playback streams; retry shortly",
79            )
80        })?;
81    let shutdown = state.shutdown.subscribe();
82    Ok(ws
83        .max_message_size(MAX_CONTROL_MESSAGE_BYTES)
84        .max_frame_size(MAX_CONTROL_MESSAGE_BYTES)
85        .on_upgrade(move |socket| async move {
86            playback_stream(socket, state, principal, permit, shutdown).await;
87        }))
88}

146    references: Vec<EntityRef>,
147    queue: bool,

149    output_rate: Option<u32>,
150    tone: ToneControls,
151    interactive: bool,
152    profile: Option<String>,

156impl Start {
157    fn parse(input: &str) -> Result<Self, StreamFailure> {
158        let initial = serde_json::from_str(input).map_err(|_| StreamFailure::INVALID_START)?;
159        let InitialFrame::Start {
160            track,
161            tracks,
162            normalize,
163            sample_rate,
164            bass,
165            treble,
166            interactive,
167            profile,
168        } = initial
169        else {
170            let InitialFrame::Resume { profile } = initial else {
171                return Err(StreamFailure::INVALID_START);
172            };
173            return Ok(Self {
174                references: Vec::new(),
175                queue: true,
176                normalize: Mode::Track,
177                output_rate: None,
178                tone: ToneControls::FLAT,
179                interactive: true,
180                profile: Some(profile),
181                resume: true,
182            });
183        };
184        if profile.is_some() && !interactive {
185            return Err(StreamFailure::INVALID_START);
186        }
187        let (tracks, queue) = match (track, tracks) {
188            (Some(track), None) => (vec![track], false),
189            (None, Some(tracks)) if !tracks.is_empty() && tracks.len() <= MAX_QUEUE_TRACKS => {
190                (tracks, true)
191            }
192            _ => return Err(StreamFailure::INVALID_START),
193        };
194        let references = tracks
195            .iter()
196            .map(|track| {
197                EntityRef::parse_token(track)
198                    .filter(|reference| {
199                        reference.kind == EntityKind::Track && !reference.key.is_empty()
200                    })
201                    .ok_or(StreamFailure::INVALID_START)
202            })
203            .collect::<Result<Vec<_>, _>>()?;
204        if sample_rate.is_some_and(|rate| !(MIN_SAMPLE_RATE..=MAX_SAMPLE_RATE).contains(&rate)) {
205            return Err(StreamFailure::INVALID_START);
206        }
207        let bass = bass.unwrap_or(0.0);
208        let treble = treble.unwrap_or(0.0);
209        let tone = ToneControls::new(bass, treble).map_err(|_| StreamFailure::INVALID_START)?;
210        Ok(Self {
211            references,
212            queue,
213            normalize: normalize.mode(),
214            output_rate: sample_rate,
215            tone,
216            interactive,
217            profile,
218            resume: false,
219        })
220    }
221}

231pub(super) fn source_from_catalog(
232    catalog: &Catalog,
233    requested: &EntityRef,
234) -> Result<TrackSource, StreamFailure> {
235    if requested.kind != EntityKind::Track {
236        return Err(StreamFailure::TRACK_NOT_FOUND);
237    }
238    let track_id = requested
239        .resolve(catalog)
240        .ok_or(StreamFailure::TRACK_NOT_FOUND)?;
241    let track = catalog
242        .track(track_id)
243        .ok_or(StreamFailure::TRACK_NOT_FOUND)?;
244    let file = catalog
245        .file(track.file_id)
246        .ok_or(StreamFailure::TRACK_NOT_FOUND)?;
247    let mtime_subseconds = catalog
248        .file_mtime_subseconds
249        .get(&file.path)
250        .copied()
251        // A legacy whole-second identity is not precise enough to safely
252        // attribute a remote listen. A normal scan upgrades it.
253        .ok_or(StreamFailure::SOURCE_CHANGED)?;
254    let reference = EntityRef::of(catalog, EntityKind::Track, track_id)
255        .ok_or(StreamFailure::TRACK_NOT_FOUND)?;
256    Ok(TrackSource {
257        reference,
258        path: PathBuf::from(&file.path),
259        // Raw lyrics/tags can be large. Repeated queue references need only
260        // source identity and technical properties, never copies of those tags.
261        file: AudioFile {
262            id: file.id,
263            path: file.path.clone(),
264            size: file.size,
265            mtime: file.mtime,
266            properties: file.properties.clone(),
267            ..AudioFile::default()
268        },
269        mtime_subseconds,
270    })
271}

287async fn current_sources(
288    state: &ApiState,
289    requested: &[EntityRef],
290) -> Result<(Vec<TrackSource>, Catalog), StreamFailure> {
291    // A catalog publisher may still own the write lock. Wait asynchronously
292    // so shutdown/deadlines do not depend on that writer, and cancellation
293    // cannot strand blocking workers queued behind it.
294    let mut shutdown = state.shutdown.subscribe();
295    let catalog = tokio::select! {
296        biased;
297        _ = shutdown.recv() => return Err(StreamFailure::SERVER_SHUTDOWN),
298        catalog = tokio::time::timeout(SOURCE_READY_TIMEOUT, state.catalog.clone().read_owned()) => {
299            catalog.map_err(|_| StreamFailure::SOURCE_UNAVAILABLE)?
300        }
301    };
302    // Track references and selected analyses can each scan a large catalog.
303    // The socket's playback permit bounds this work independently of HTTP
304    // workers; keep it away from Tokio's network and acknowledgement threads.
305    let state = state.clone();
306    let requested = requested.to_vec();
307    // A queued task must not keep a read lock after its caller is cancelled:
308    // catalogue publication must not depend on blocking-pool availability.
309    let pending = PendingCatalog(Arc::new(std::sync::Mutex::new(Some(catalog))));
310    let for_worker = pending.0.clone();
311    let mut worker = tokio::task::spawn_blocking(move || {
312        let catalog = for_worker
313            .lock()
314            .map_err(|_| StreamFailure::STREAM_FAILED)?
315            .take()
316            .ok_or(StreamFailure::STREAM_FAILED)?;
317        let catalog = catalog.as_ref().ok_or(StreamFailure::CATALOG_UNAVAILABLE)?;
318        selected_sources(&state, catalog, &requested)
319    });
320    tokio::select! {
321        biased;
322        _ = shutdown.recv() => {
323            // Abort can prevent queued work from starting. PendingCatalog also
324            // releases its snapshot without relying on the blocking pool to run.
325            // An already running selection retains its own guard until it exits.
326            worker.abort();
327            Err(StreamFailure::SERVER_SHUTDOWN)
328        },
329        result = &mut worker => result.map_err(|_| StreamFailure::STREAM_FAILED)?,
330    }
331}

333fn selected_sources(
334    state: &ApiState,
335    catalog: &Catalog,
336    requested: &[EntityRef],
337) -> Result<(Vec<TrackSource>, Catalog), StreamFailure> {
338    let visible = crate::parental::visible(state, catalog)
339        .map_err(|_| StreamFailure::parental("parental_policy_unavailable"))?;
340    let held_visible = visible.as_ref().map(|view| view.blocking_read());
341    let catalog = match held_visible.as_ref() {
342        Some(held) => held.as_ref().ok_or(StreamFailure::CATALOG_UNAVAILABLE)?,
343        None => catalog,
344    };
345    if crate::parental::supervised(state)
346        && requested
347            .iter()
348            .any(|reference| reference.resolve(catalog).is_none())
349    {
350        return Err(StreamFailure::parental("parental_artist_blocked"));
351    }
352    let sources = requested
353        .iter()
354        .map(|reference| source_from_catalog(catalog, reference))
355        .collect::<Result<Vec<_>, _>>()?;
356    let paths: BTreeSet<_> = sources
357        .iter()
358        .map(|source| source.file.path.as_str())
359        .collect();
360    // ReadyNormalization needs catalog analyses only for the selected files.
361    // Retaining the full graph here would copy a potentially huge library per
362    // active client, so keep just analyses that can describe this exact file.
363    let normalization_catalog = Catalog {
364        analyses: catalog
365            .analyses
366            .iter()
367            .filter(|analysis| paths.contains(analysis.path.as_str()))
368            .cloned()
369            .collect(),
370        ..Catalog::default()
371    };
372    Ok((sources, normalization_catalog))
373}

375/// File identity is checked around opening and decoding. This cannot make a
376/// pathname-based decoder immune to every operating-system race, but it
377/// refuses relative/traversing paths, links and special files, never accepts a
378/// caller-supplied path, and detects a replacement before it can become an
379/// acknowledged listen.
380pub(super) fn validate_source(source: &TrackSource) -> Result<(), StreamFailure> {
381    if !source.path.is_absolute()
382        || source
383            .path
384            .components()
385            .any(|part| matches!(part, Component::ParentDir))
386    {
387        return Err(StreamFailure::SOURCE_CHANGED);
388    }
389    let metadata = fs::symlink_metadata(&source.path).map_err(|failure| {
390        if failure.kind() == ErrorKind::NotFound {
391            StreamFailure::SOURCE_UNAVAILABLE
392        } else {
393            StreamFailure::SOURCE_CHANGED
394        }
395    })?;
396    let identity = user::PlaybackSource {
397        size: source.file.size,
398        mtime: source.file.mtime,
399        mtime_ns: Some(source.mtime_subseconds),
400    };
401    if metadata.file_type().is_symlink() || !identity.matches_metadata(&metadata) {
402        return Err(StreamFailure::SOURCE_CHANGED);
403    }
404    Ok(())
405}

630#[derive(Default)]
631struct Acknowledgements {
632    sent: u64,
633    consumed: u64,
634    max_pending: u64,
635    waiting_since: Option<Instant>,
636}
637
638impl Acknowledgements {
639    fn with_window(max_pending: u64) -> Self {
640        Self {
641            max_pending,
642            ..Self::default()
643        }
644    }
645
646    fn available(&self) -> u64 {
647        self.max_pending
648            .saturating_sub(self.sent.saturating_sub(self.consumed))
649    }
650
651    fn sent(&mut self, frames: u64, now: Instant) -> Result<(), StreamFailure> {
652        self.sent = self
653            .sent
654            .checked_add(frames)
655            .ok_or(StreamFailure::PROCESSING_FAILED)?;
656        if self.sent > self.consumed && self.waiting_since.is_none() {
657            self.waiting_since = Some(now);
658        }
659        Ok(())
660    }
661
662    fn acknowledge(&mut self, frames: u64, now: Instant) -> Result<bool, StreamFailure> {
663        if frames < self.consumed || frames > self.sent {
664            return Err(StreamFailure::INVALID_ACK);
665        }
666        if frames > self.consumed {
667            self.consumed = frames;
668            self.waiting_since = (self.sent > self.consumed).then_some(now);
669            return Ok(true);
670        }
671        Ok(false)
672    }
673
674    fn deadline(&self) -> Option<Instant> {
675        self.waiting_since.map(|waiting| waiting + ACK_TIMEOUT)
676    }
677}

706async fn receive_start(
707    socket: &mut WebSocket,
708    state: &ApiState,
709    captured: &auth::Principal,
710    shutdown: &mut crate::shutdown::Receiver,
711) -> Result<Start, Option<StreamFailure>> {
712    let mut authorization = tokio::time::interval(AUTH_RECHECK_INTERVAL);
713    authorization.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
714    let deadline = tokio::time::sleep(START_TIMEOUT);
715    tokio::pin!(deadline);
716    loop {
717        tokio::select! {
718            _ = &mut deadline => return Err(Some(StreamFailure::INVALID_START)),
719            _ = shutdown.recv() => return Err(Some(StreamFailure::SERVER_SHUTDOWN)),
720            _ = authorization.tick() => {
721                if let Err(failure) = authorized(state, captured, false, shutdown).await {
722                    return Err(Some(failure));
723                }
724            }
725            incoming = socket.recv() => {
726                let Some(incoming) = incoming else {
727                    return Err(None);
728                };
729                let incoming = match incoming {
730                    Ok(incoming) => incoming,
731                    Err(_) => return Err(None),
732                };
733                match incoming {
734                    Message::Text(input) => return Start::parse(&input).map_err(Some),
735                    Message::Ping(payload) => {
736                        if events::send_message(socket, Message::Pong(payload)).await.is_err() {
737                            return Err(None);
738                        }
739                    }
740                    Message::Pong(_) => {}
741                    Message::Close(_) => return Err(None),
742                    Message::Binary(_) => return Err(Some(StreamFailure::INVALID_START)),
743                }
744            }
745        }
746    }
747}

749async fn wait_for_format(socket: &mut WebSocket) -> Result<bool, StreamFailure> {
750    let Some(incoming) = socket.recv().await else {
751        return Ok(false);
752    };

754    match incoming {

764    }

767async fn authorized(
768    state: &ApiState,
769    captured: &auth::Principal,
770    touch: bool,
771    shutdown: &mut crate::shutdown::Receiver,
772) -> Result<(), StreamFailure> {
773    // Credential/policy workers can outlive this wait. Keep shutdown responsive
774    // and bound how long a disconnected peer can retain a playback connection.
775    let confirmation = tokio::select! {
776        biased;
777        _ = shutdown.recv() => return Err(StreamFailure::SERVER_SHUTDOWN),
778        confirmation = tokio::time::timeout(
779            AUTH_CHECK_TIMEOUT, check_authorized(state, captured, touch),
780        ) => confirmation,
781    };
782    if matches!(confirmation, Ok(true)) {
783        Ok(())
784    } else {
785        Err(crate::parental::pcm::failure(state).unwrap_or(StreamFailure::AUTHENTICATION_EXPIRED))
786    }
787}

789async fn check_authorized(state: &ApiState, captured: &auth::Principal, touch: bool) -> bool {
790    let publication = state.parental.catalog_revision();
791    let Ok(Some(accounts)) = auth::current_accounts(state).await else {
792        return false;
793    };
794    let Ok(principal) = auth::principal(
795        state,
796        &accounts,
797        auth::session_token(captured),
798        touch,
799        Instant::now(),
800    ) else {
801        return false;
802    };
803    if auth::require_mutation(&principal).is_err() {
804        return false;
805    }
806    let policy_allowed = if let Some(session) = &state.parental_pcm {
807        session.check(state).await
808    } else if state.parental_scope.is_some() {
809        let for_check = state.clone();
810        let owner = principal.owner.clone();
811        crate::parental::blocking(state, move || {
812            crate::parental::confirm(&for_check)?;
813            if !crate::parental::supervised(&for_check) {
814                return Ok(());
815            }
816            crate::parental::load(&for_check)?
817                .check_schedule(
818                    &owner,
819                    aede_core::clock::now_seconds(),
820                    &aede_core::parental::Media::Music { track: None },
821                )
822                .map_err(|reason| crate::parental::refused(&reason))
823        })
824        .await
825        .is_ok()
826    } else {
827        return true;
828    };
829    // Policy/lease work can wait after the first credential read. Its success
830    // must not revive a login revoked while that work was pending.
831    policy_allowed
832        && auth::socket_live(
833            state,
834            &auth::SocketIdentity {
835                session: Some(captured.clone()),
836                administrative: false,
837            },
838        )
839        .await
840        && publication == state.parental.catalog_revision()
841}

848async fn send_failure(
849    socket: &mut WebSocket,
850    failure: StreamFailure,
851    shutdown: &mut crate::shutdown::Receiver,
852) {
853    let payload = match serde_json::to_string(&ErrorFrame {
854        kind: "error",
855        code: failure.code,
856        message: failure.message,
857    }) {
858        Ok(payload) => payload,
859        Err(_) => {
860            eprintln!("API playback failure encoding failed: {}", failure.code);
861            return;
862        }
863    };
864    for (stage, message) in [
865        ("error", Message::Text(payload)),
866        (
867            "close",
868            Message::Close(Some(CloseFrame {
869                code: close_code::POLICY,
870                reason: "playback stream stopped".into(),
871            })),
872        ),
873    ] {
874        let reason = match tokio::time::timeout(WEBSOCKET_SEND_TIMEOUT, socket.send(message)).await
875        {
876            Ok(Ok(())) => continue,
877            Ok(Err(_)) => "transport failed",
878            Err(_) => "send deadline",
879        };
880        eprintln!(
881            "API playback {stage} notification failed: {} ({reason})",
882            failure.code
883        );
884        return;
885    }
886    // Do not drop TCP with client ACKs still in flight. They are transport
887    // cleanup only: the consumed-frame snapshot and history are already sealed.
888    if let Err(reason) = finish_failure_close(socket, shutdown, CLOSE_TIMEOUT).await {
889        eprintln!(
890            "API playback close handshake stopped: {} ({reason})",
891            failure.code
892        );
893    }
894}

896async fn finish_failure_close(
897    socket: &mut WebSocket,
898    shutdown: &mut crate::shutdown::Receiver,
899    timeout: Duration,
900) -> Result<(), &'static str> {
901    let closing = tokio::time::timeout(timeout, async {
902        let mut bytes = 0_usize;
903        for _ in 0..MAX_CLOSE_MESSAGES {
904            let size = match socket.recv().await {
905                Some(Ok(Message::Close(_))) => return Ok(()),
906                Some(Ok(Message::Text(text))) => text.len(),
907                Some(Ok(Message::Binary(bytes) | Message::Ping(bytes) | Message::Pong(bytes))) => {
908                    bytes.len()
909                }
910                Some(Err(_)) => return Err("transport failed"),
911                None => return Err("peer disconnected before close reply"),
912            };
913            bytes = bytes.saturating_add(size);
914            if bytes > MAX_CLOSE_BYTES {
915                return Err("close input byte limit");
916            }
917        }
918        Err("close input message limit")
919    });
920    tokio::select! {
921        biased;
922        _ = shutdown.recv() => Err("server shutdown"),
923        result = closing => result.unwrap_or(Err("close reply deadline")),
924    }
925}

945async fn receive_ack(
946    socket: &mut WebSocket,
947    acknowledgements: &mut Acknowledgements,
948) -> Result<bool, StreamFailure> {
949    let Some(incoming) = socket.recv().await else {

967        Message::Close(_) => Ok(false),
968        Message::Binary(_) => Err(StreamFailure::INVALID_ACK),

977async fn drive(
978    socket: &mut WebSocket,
979    state: &ApiState,
980    captured: &auth::Principal,
981    receiver: &mut mpsc::Receiver<ProducerEvent>,
982    shutdown: &mut crate::shutdown::Receiver,
983    sources: &[TrackSource],
984    mode: DriveMode<'_>,
985    timeline: &mut ListeningTimeline,
986) -> DriveEnd {
987    let mut owned = state.clone();
988    owned.parental_pcm = match crate::parental::pcm::begin(state, captured).await {
989        Ok(session) => session,
990        Err(error) => {
991            return DriveEnd {
992                acknowledged_frames: 0,
993                sample_rate: None,
994                completed: false,
995                failure: Some(StreamFailure::parental(error.code)),
996                notify: true,
997                control: None,
998            };
999        }
1000    };
1001    let mut end = drive_inner(
1002        socket, &owned, captured, receiver, shutdown, sources, mode, timeline,
1003    )
1004    .await;
1005    if let Some(session) = &owned.parental_pcm {
1006        if let Some(rate) = end.sample_rate {
1007            session.observe(frames_to_milliseconds(end.acknowledged_frames, rate));
1008        }
1009        if let Err(error) = session.close(&owned).await {
1010            end.failure = Some(StreamFailure::parental(error.code));
1011            end.notify = true;
1012            end.completed = false;
1013        }
1014    }
1015    end
1016}

1018async fn drive_inner(
1019    socket: &mut WebSocket,
1020    state: &ApiState,
1021    captured: &auth::Principal,
1022    receiver: &mut mpsc::Receiver<ProducerEvent>,
1023    shutdown: &mut crate::shutdown::Receiver,
1024    sources: &[TrackSource],
1025    mode: DriveMode<'_>,
1026    timeline: &mut ListeningTimeline,
1027) -> DriveEnd {
1028    let DriveMode { queue, interactive } = mode;
1029    let mut authorization = tokio::time::interval(AUTH_RECHECK_INTERVAL);
1030    authorization.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
1031    let mut last_authentication = Instant::now();
1032    let source_deadline = tokio::time::sleep(SOURCE_READY_TIMEOUT);
1033    tokio::pin!(source_deadline);
1034    let initial = loop {
1035        tokio::select! {
1036            event = receiver.recv() => match event {
1037                Some(event) => break event,
1038                None => return DriveEnd {
1039                    acknowledged_frames: 0,
1040                    sample_rate: None,
1041                    completed: false,
1042                    failure: Some(StreamFailure::STREAM_FAILED),
1043                    notify: true,
1044                    control: None,
1045                },
1046            },
1047            _ = authorization.tick() => {
1048                if let Err(failure) = keep_authorized(state, captured, &mut last_authentication, true, shutdown).await {
1049                    return DriveEnd {
1050                        acknowledged_frames: 0,
1051                        sample_rate: None,
1052                        completed: false,
1053                        failure: Some(failure),
1054                        notify: true,
1055                        control: None,
1056                    };
1057                }
1058            }
1059            _ = shutdown.recv() => return DriveEnd {
1060                acknowledged_frames: 0,
1061                sample_rate: None,
1062                completed: false,
1063                failure: Some(StreamFailure::SERVER_SHUTDOWN),
1064                notify: true,
1065                control: None,
1066            },
1067            incoming = wait_for_format(socket) => match incoming {
1068                Ok(true) => {}
1069                Ok(false) => return DriveEnd {
1070                    acknowledged_frames: 0,
1071                    sample_rate: None,
1072                    completed: false,
1073                    failure: None,
1074                    notify: false,
1075                    control: None,
1076                },
1077                Err(failure) => return DriveEnd {
1078                    acknowledged_frames: 0,
1079                    sample_rate: None,
1080                    completed: false,
1081                    failure: Some(failure),
1082                    notify: true,
1083                    control: None,
1084                },
1085            },
1086            _ = &mut source_deadline => return DriveEnd {
1087                acknowledged_frames: 0,
1088                sample_rate: None,
1089                completed: false,
1090                failure: Some(StreamFailure::SOURCE_UNAVAILABLE),
1091                notify: true,
1092                control: None,
1093            },
1094        }
1095    };
1096    let mut format = match initial {
1097        ProducerEvent::Format(format) => format,
1098        ProducerEvent::Error(failure) => {
1099            return DriveEnd {
1100                acknowledged_frames: 0,
1101                sample_rate: None,
1102                completed: false,
1103                failure: Some(failure),
1104                notify: true,
1105                control: None,
1106            };
1107        }
1108        ProducerEvent::Audio { .. }
1109        | ProducerEvent::Eof { .. }
1110        | ProducerEvent::Track(_)
1111        | ProducerEvent::TrackEnd(_) => {
1112            return DriveEnd {
1113                acknowledged_frames: 0,
1114                sample_rate: None,
1115                completed: false,
1116                failure: Some(StreamFailure::STREAM_FAILED),
1117                notify: true,
1118                control: None,
1119            };
1120        }
1121    };
1122    let mut acknowledgements = Acknowledgements::with_window(format.max_unacknowledged_frames);
1123    if let Some(session) = &state.parental_pcm
1124        && let Err(error) = session.activate(state).await
1125    {
1126        return DriveEnd::stopped(
1127            &acknowledgements,
1128            Some(format.sample_rate),
1129            Some(StreamFailure::parental(error.code)),
1130            true,
1131        );
1132    }
1133    if let Err(failure) =
1134        keep_authorized(state, captured, &mut last_authentication, true, shutdown).await
1135    {
1136        return DriveEnd::stopped(
1137            &acknowledgements,
1138            Some(format.sample_rate),
1139            Some(failure),
1140            true,
1141        );
1142    }
1143    if interactive::send_stream_json(socket, &format, interactive)
1144        .await
1145        .is_err()
1146    {
1147        return DriveEnd::stopped(&acknowledgements, Some(format.sample_rate), None, false);
1148    }
1149    let mut pending = None;
1150    let mut eof = None;
1151    let mut producer_progress = Instant::now();
1152    let mut checkpoint_at = Instant::now();
1153    loop {
1154        if let Some(session) = &state.parental_pcm {
1155            session.observe(frames_to_milliseconds(
1156                acknowledgements.consumed,
1157                format.sample_rate,
1158            ));
1159            session.sent(frames_to_milliseconds(
1160                acknowledgements.sent,
1161                format.sample_rate,
1162            ));
1163        }
1164        if interactive.is_some_and(|epoch| epoch.checkpoint_failed()) {
1165            return DriveEnd::stopped(
1166                &acknowledgements,
1167                Some(format.sample_rate),
1168                Some(StreamFailure::STATE_FAILED),
1169                true,
1170            );
1171        }
1172        if let Some(event) = pending.take() {
1173            match event {
1174                ProducerEvent::Audio {
1175                    mut bytes,
1176                    mut frames,
1177                } => {
1178                    let mut accepted = frames.min(acknowledgements.available());
1179                    if let Some(session) = &state.parental_pcm {
1180                        accepted = accepted
1181                            .min(session.available(acknowledgements.sent, format.sample_rate));
1182                    }
1183                    if accepted == 0 {
1184                        // The window is full. Requeue this exact producer
1185                        // block and fall through to `select!` so an ACK,
1186                        // shutdown, revocation or timeout can make progress.
1187                        pending = Some(ProducerEvent::Audio { bytes, frames });
1188                    } else {
1189                        if accepted < frames {
1190                            let frame_count =
1191                                usize::try_from(frames).ok().filter(|count| *count > 0);
1192                            let frame_bytes = frame_count
1193                                .and_then(|count| {
1194                                    bytes.len().checked_div(count).map(|width| (count, width))
1195                                })
1196                                .filter(|(count, width)| {
1197                                    *width > 0
1198                                        && count
1199                                            .checked_mul(*width)
1200                                            .is_some_and(|length| length == bytes.len())
1201                                })
1202                                .map(|(_, width)| width)
1203                                .ok_or(StreamFailure::PROCESSING_FAILED);
1204                            let Ok(frame_bytes) = frame_bytes else {
1205                                return DriveEnd::stopped(
1206                                    &acknowledgements,
1207                                    Some(format.sample_rate),
1208                                    Some(StreamFailure::PROCESSING_FAILED),
1209                                    true,
1210                                );
1211                            };
1212                            let split = usize::try_from(accepted)
1213                                .ok()
1214                                .and_then(|count| count.checked_mul(frame_bytes));
1215                            let Some(split) = split.filter(|split| *split <= bytes.len()) else {
1216                                return DriveEnd::stopped(
1217                                    &acknowledgements,
1218                                    Some(format.sample_rate),
1219                                    Some(StreamFailure::PROCESSING_FAILED),
1220                                    true,
1221                                );
1222                            };
1223                            let remaining = bytes.split_off(split);
1224                            frames -= accepted;
1225                            pending = Some(ProducerEvent::Audio {
1226                                bytes: remaining,
1227                                frames,
1228                            });
1229                            frames = accepted;
1230                        }
1231                        if let Err(failure) = keep_authorized(
1232                            state,
1233                            captured,
1234                            &mut last_authentication,
1235                            false,
1236                            shutdown,
1237                        )
1238                        .await
1239                        {
1240                            return DriveEnd::stopped(
1241                                &acknowledgements,
1242                                Some(format.sample_rate),
1243                                Some(failure),
1244                                true,
1245                            );
1246                        }
1247                        if events::send_message(socket, Message::Binary(bytes))
1248                            .await
1249                            .is_err()
1250                        {
1251                            return DriveEnd::stopped(
1252                                &acknowledgements,
1253                                Some(format.sample_rate),
1254                                None,
1255                                false,
1256                            );
1257                        }
1258                        if acknowledgements.sent(frames, Instant::now()).is_err() {
1259                            return DriveEnd::stopped(
1260                                &acknowledgements,
1261                                Some(format.sample_rate),
1262                                Some(StreamFailure::PROCESSING_FAILED),
1263                                true,
1264                            );
1265                        }
1266                        // Releasing this buffered item lets the decoder continue;
1267                        // give it a fresh bounded interval to produce the next one.
1268                        producer_progress = Instant::now();
1269                        continue;
1270                    }
1271                }
1272                ProducerEvent::Eof { frames } => {
1273                    if frames != acknowledgements.sent {
1274                        return DriveEnd::stopped(
1275                            &acknowledgements,
1276                            Some(format.sample_rate),
1277                            Some(StreamFailure::STREAM_FAILED),
1278                            true,
1279                        );
1280                    }
1281                    if let Err(failure) =
1282                        keep_authorized(state, captured, &mut last_authentication, false, shutdown)
1283                            .await
1284                    {
1285                        return DriveEnd::stopped(
1286                            &acknowledgements,
1287                            Some(format.sample_rate),
1288                            Some(failure),
1289                            true,
1290                        );
1291                    }
1292                    if interactive::send_stream_json(
1293                        socket,
1294                        &EofFrame {
1295                            kind: "eof",
1296                            frames,
1297                        },
1298                        interactive,
1299                    )
1300                    .await
1301                    .is_err()
1302                    {
1303                        return DriveEnd::stopped(
1304                            &acknowledgements,
1305                            Some(format.sample_rate),
1306                            None,
1307                            false,
1308                        );
1309                    }
1310                    eof = Some(frames);
1311                    continue;
1312                }
1313                ProducerEvent::Format(next) => {
1314                    if !queue
1315                        || next.start_frame != Some(acknowledgements.sent)
1316                        || next.sample_rate != format.sample_rate
1317                    {
1318                        return DriveEnd::stopped(
1319                            &acknowledgements,
1320                            Some(format.sample_rate),
1321                            Some(StreamFailure::STREAM_FAILED),
1322                            true,
1323                        );
1324                    }
1325                    if acknowledgements.sent != acknowledgements.consumed {
1326                        // Channel changes require the client to drain its old
1327                        // device format before any differently laid out bytes.
1328                        pending = Some(ProducerEvent::Format(next));
1329                    } else {
1330                        if let Err(failure) = keep_authorized(
1331                            state,
1332                            captured,
1333                            &mut last_authentication,
1334                            false,
1335                            shutdown,
1336                        )
1337                        .await
1338                        {
1339                            return DriveEnd::stopped(
1340                                &acknowledgements,
1341                                Some(format.sample_rate),
1342                                Some(failure),
1343                                true,
1344                            );
1345                        }
1346                        if interactive::send_stream_json(socket, &next, interactive)
1347                            .await
1348                            .is_err()
1349                        {
1350                            return DriveEnd::stopped(
1351                                &acknowledgements,
1352                                Some(format.sample_rate),
1353                                None,
1354                                false,
1355                            );
1356                        }
1357                        acknowledgements.max_pending = next.max_unacknowledged_frames;
1358                        format = next;
1359                        producer_progress = Instant::now();
1360                        continue;
1361                    }
1362                }
1363                ProducerEvent::Track(track) => {
1364                    if let Err(failure) =
1365                        timeline.begin(&track, acknowledgements.sent, format.sample_rate, sources)
1366                    {
1367                        return DriveEnd::stopped(
1368                            &acknowledgements,
1369                            Some(format.sample_rate),
1370                            Some(failure),
1371                            true,
1372                        );
1373                    }
1374                    if queue
1375                        && interactive::send_stream_json(socket, &track, interactive)
1376                            .await
1377                            .is_err()
1378                    {
1379                        return DriveEnd::stopped(
1380                            &acknowledgements,
1381                            Some(format.sample_rate),
1382                            None,
1383                            false,
1384                        );
1385                    }
1386                    continue;
1387                }
1388                ProducerEvent::TrackEnd(track) => {
1389                    if let Err(failure) = timeline.end(&track, acknowledgements.sent) {
1390                        return DriveEnd::stopped(
1391                            &acknowledgements,
1392                            Some(format.sample_rate),
1393                            Some(failure),
1394                            true,
1395                        );
1396                    }
1397                    if queue
1398                        && interactive::send_stream_json(socket, &track, interactive)
1399                            .await
1400                            .is_err()
1401                    {
1402                        return DriveEnd::stopped(
1403                            &acknowledgements,
1404                            Some(format.sample_rate),
1405                            None,
1406                            false,
1407                        );
1408                    }
1409                    continue;
1410                }
1411                ProducerEvent::Error(failure) => {
1412                    return DriveEnd::stopped(
1413                        &acknowledgements,
1414                        Some(format.sample_rate),
1415                        Some(failure),
1416                        true,
1417                    );
1418                }
1419            }
1420        }
1421        if eof.is_some_and(|frames| frames == acknowledgements.consumed) {
1422            return DriveEnd {
1423                acknowledged_frames: acknowledgements.consumed,
1424                sample_rate: Some(format.sample_rate),
1425                completed: true,
1426                failure: None,
1427                notify: false,
1428                control: None,
1429            };
1430        }
1431        let deadline = acknowledgements
1432            .deadline()
1433            .unwrap_or_else(|| Instant::now() + Duration::from_secs(24 * 60 * 60));
1434        let acknowledgement_timeout =
1435            tokio::time::sleep_until(tokio::time::Instant::from_std(deadline));
1436        tokio::pin!(acknowledgement_timeout);
1437        let progress_timeout = tokio::time::sleep_until(tokio::time::Instant::from_std(
1438            producer_progress + PRODUCER_PROGRESS_TIMEOUT,
1439        ));
1440        tokio::pin!(progress_timeout);
1441        let consumed_before_receive = acknowledgements.consumed;
1442        tokio::select! {
1443            _ = tokio::time::sleep(Duration::from_millis(20)), if pending.is_some() && state.parental_pcm.is_some() => {},
1444            _ = &mut acknowledgement_timeout, if acknowledgements.deadline().is_some() => {
1445                return DriveEnd::stopped(
1446                    &acknowledgements,
1447                    Some(format.sample_rate),
1448                    Some(StreamFailure::ACK_TIMEOUT),
1449                    true,
1450                );
1451            }
1452            _ = &mut progress_timeout,
1453                if pending.is_none()
1454                    && eof.is_none()
1455                    && acknowledgements.sent == acknowledgements.consumed =>
1456            {
1457                return DriveEnd::stopped(
1458                    &acknowledgements,
1459                    Some(format.sample_rate),
1460                    Some(StreamFailure::SOURCE_UNAVAILABLE),
1461                    true,
1462                );
1463            }
1464            _ = authorization.tick() => {
1465                if let Err(failure) = keep_authorized(state, captured, &mut last_authentication, true, shutdown).await {
1466                    return DriveEnd::stopped(
1467                        &acknowledgements,
1468                        Some(format.sample_rate),
1469                        Some(failure),
1470                        true,
1471                    );
1472                }
1473            }
1474            _ = shutdown.recv() => {
1475                return DriveEnd::stopped(
1476                    &acknowledgements,
1477                    Some(format.sample_rate),
1478                    Some(StreamFailure::SERVER_SHUTDOWN),
1479                    true,
1480                );
1481            }
1482            incoming = interactive::receive(socket, &mut acknowledgements, interactive) => match incoming {
1483                Ok(interactive::Incoming::Control(control)) => {
1484                    if let Err(failure) = authorized(state, captured, true, shutdown).await {
1485                        return DriveEnd::stopped(&acknowledgements, Some(format.sample_rate),
1486                            Some(failure), true);
1487                    }
1488                    let mut end = DriveEnd::stopped(&acknowledgements, Some(format.sample_rate), None, false);
1489                    end.control = Some(control);
1490                    return end;
1491                }
1492                Ok(interactive::Incoming::Continue) => {
1493                    // A growing ACK can reopen a fresh producer wait after a
1494                    // long client pause. Pings and duplicate ACKs leave this
1495                    // deadline untouched.
1496                    if acknowledgements.consumed > consumed_before_receive {
1497                        // The ACK is committed only by the selected branch.
1498                        // Awaiting this read inside `receive_ack` would let
1499                        // another ready producer event cancel it after the
1500                        // socket message and frame count were already consumed.
1501                        if let Err(failure) = authorized(state, captured, true, shutdown).await {
1502                            return DriveEnd::stopped(
1503                                &acknowledgements,
1504                                Some(format.sample_rate),
1505                                Some(failure),
1506                                true,
1507                            );
1508                        }
1509                        if checkpoint_at.elapsed() >= Duration::from_secs(5) {
1510                            if let Some(epoch) = interactive {
1511                                epoch.checkpoint(timeline, acknowledgements.consumed);
1512                            }
1513                            checkpoint_at = Instant::now();
1514                        }
1515                        producer_progress = Instant::now();
1516                    }
1517                }
1518                Ok(interactive::Incoming::Closed) => return DriveEnd::stopped(
1519                    &acknowledgements,
1520                    Some(format.sample_rate),
1521                    None,
1522                    false,
1523                ),
1524                Err(failure) => return DriveEnd::stopped(
1525                    &acknowledgements,
1526                    Some(format.sample_rate),
1527                    Some(failure),
1528                    true,
1529                ),
1530            },
1531            event = receiver.recv(), if pending.is_none() && eof.is_none() => match event {
1532                Some(event) => {
1533                    producer_progress = Instant::now();
1534                    pending = Some(event);
1535                }
1536                None => return DriveEnd::stopped(
1537                    &acknowledgements,
1538                    Some(format.sample_rate),
1539                    Some(StreamFailure::STREAM_FAILED),
1540                    true,
1541                ),
1542            },
1543        }
1544    }
1545}

1548    let milliseconds = u128::from(frames).saturating_mul(1_000) / u128::from(sample_rate);

1556fn lock_for_history(data_dir: &Path) -> Result<StoreLock, ApiError> {
1557    // A catalog watcher may briefly own the shared store lock immediately
1558    // after the stream ends. Keep the once-per-socket history record pending
1559    // for a short, bounded interval instead of silently losing a valid listen.
1560    let deadline = Instant::now() + HISTORY_WRITE_TIMEOUT;
1561    loop {
1562        match StoreLock::try_acquire(data_dir) {
1563            Ok(lock) => return Ok(lock),
1564            Err(failure)
1565                if failure.kind() == ErrorKind::WouldBlock && Instant::now() < deadline =>
1566            {
1567                std::thread::sleep(Duration::from_millis(25));
1568            }
1569            Err(failure) => {
1570                return Err(history_error(if failure.kind() == ErrorKind::WouldBlock {
1571                    StreamFailure::HISTORY_BUSY.message
1572                } else {
1573                    StreamFailure::HISTORY_WRITE.message
1574                }));
1575            }
1576        }
1577    }
1578}

1580fn load_history(data_dir: &Path, catalog: &Catalog) -> Result<UserData, ApiError> {
1581    let mut data = user::load(&user::user_path(data_dir))
1582        .map_err(|failure| {
1583            eprintln!("API playback history read failed: {failure}");
1584            history_error(StreamFailure::HISTORY_READ.message)
1585        })?
1586        .unwrap_or_default();
1587    user::reconcile(&mut data, catalog);
1588    Ok(data)

1591#[derive(Default)]
1592struct HistoryOutcome {
1593    saved: Vec<usize>,
1594    failed: bool,
1595}
1596
1597#[derive(Clone)]
1598struct ListenRecord {
1599    state: ApiState,
1600    captured: auth::Principal,
1601    listens: Vec<history::Listened>,
1602}
1603
1604impl ListenRecord {
1605    fn persist(&self) -> Result<HistoryOutcome, ApiError> {
1606        let _guard = lock_for_history(&self.state.data_dir)?;
1607        // Revocation wins over every pending occurrence in the batch. Source
1608        // invalidation is local to its occurrence and cannot erase another
1609        // acknowledged listen.
1610        let accounts = auth::load_accounts(&self.state)?.ok_or_else(auth::unauthorized)?;
1611        let principal = auth::principal(
1612            &self.state,
1613            &accounts,
1614            auth::session_token(&self.captured),
1615            false,
1616            Instant::now(),
1617        )?;
1618        auth::require_mutation(&principal)?;
1619        let catalog = store::load(&store::catalog_path(&self.state.data_dir))
1620            .map_err(|failure| {
1621                eprintln!("API playback catalog read failed: {failure}");
1622                history_error(StreamFailure::HISTORY_READ.message)
1623            })?
1624            .ok_or_else(unavailable)?;
1625        let mut data = load_history(&self.state.data_dir, &catalog)?;
1626        let mut outcome = HistoryOutcome::default();
1627        for listen in &self.listens {
1628            let original = &listen.source;
1629            let valid = source_from_catalog(&catalog, &original.reference).is_ok_and(|current| {
1630                current.reference == original.reference
1631                    && current.path == original.path
1632                    && current.file.size == original.file.size
1633                    && current.file.mtime == original.file.mtime
1634                    && current.mtime_subseconds == original.mtime_subseconds
1635            }) && validate_source(original).is_ok();
1636            if !valid {
1637                outcome.failed = true;
1638                continue;
1639            }
1640            data.record_play(Play {
1641                owner: principal.owner.clone(),
1642                track: original.reference.clone(),
1643                at: listen.started_at,
1644                ms_played: listen.ms_played,
1645                completed: listen.completed,
1646            });
1647            outcome.saved.push(listen.index);
1648        }
1649        if !outcome.saved.is_empty() {
1650            if let Err(failure) = aede_core::achievements::refresh(
1651                &catalog,
1652                &mut data,
1653                &principal.owner,
1654                false,
1655                aede_core::clock::now_seconds(),
1656            ) {
1657                eprintln!("API achievement projection unavailable: {failure}");
1658            }
1659            auth::recheck(&self.state, &principal)?;
1660            user::save(&data, &user::user_path(&self.state.data_dir)).map_err(|failure| {
1661                eprintln!("API playback history write failed: {failure}");
1662                history_error(StreamFailure::HISTORY_WRITE.message)
1663            })?;
1664        }
1665        Ok(outcome)
1666    }
1667}
1668
1669fn history_failure(failure: &ApiError) -> StreamFailure {
1670    if matches!(failure.code, "unauthorized" | "forbidden") {
1671        return StreamFailure::AUTHENTICATION_EXPIRED;
1672    }
1673    for reason in [
1674        StreamFailure::HISTORY_BUSY,
1675        StreamFailure::HISTORY_READ,
1676        StreamFailure::HISTORY_WRITE,
1677    ] {
1678        if failure.message == reason.message {
1679            return reason;
1680        }
1681    }
1682    StreamFailure::HISTORY_FAILED
1683}

1685async fn record_listen(
1686    record: ListenRecord,
1687    permit: tokio::sync::OwnedSemaphorePermit,
1688) -> (
1689    Result<HistoryOutcome, StreamFailure>,
1690    Option<tokio::sync::OwnedSemaphorePermit>,
1691) {
1692    let mut task = tokio::task::spawn_blocking(move || record.persist());
1693    tokio::select! {
1694        outcome = &mut task => match outcome {
1695            Ok(Ok(outcome)) => (Ok(outcome), Some(permit)),
1696            Ok(Err(failure)) => {
1697                eprintln!("API playback history update failed: {}", failure.message);
1698                (Err(history_failure(&failure)), Some(permit))
1699            }
1700            Err(_) => (Err(StreamFailure::HISTORY_FAILED), Some(permit)),
1701        },
1702        _ = tokio::time::sleep(HISTORY_WRITE_TIMEOUT) => {
1703            tokio::spawn(async move {
1704                let _permit = permit;
1705                match task.await {
1706                    Ok(Err(failure)) => eprintln!("API playback deferred history update failed: {}", failure.code),
1707                    Err(_) => eprintln!("API playback deferred history task stopped"),
1708                    Ok(Ok(outcome)) if outcome.failed => eprintln!("API playback deferred history rejected changed sources"),
1709                    Ok(Ok(_)) => {}
1710                }
1711            });
1712            (Err(StreamFailure::HISTORY_TIMEOUT), None)
1713        }
1714    }
1715}

1717async fn worker_finished(worker: &mut tokio::task::JoinHandle<()>) -> bool {
1718    tokio::select! {
1719        _ = worker => true,
1720        _ = tokio::time::sleep(WORKER_DRAIN_TIMEOUT) => false,
1721    }
1722}

1724fn retain_worker(
1725    permit: tokio::sync::OwnedSemaphorePermit,
1726    worker: tokio::task::JoinHandle<()>,
1727    record: Option<ListenRecord>,
1728) {
1729    tokio::spawn(async move {
1730        let _permit = permit;
1731        let _ = worker.await;
1732        if let Some(record) = record {
1733            match tokio::task::spawn_blocking(move || record.persist()).await {
1734                Ok(Err(failure)) => {
1735                    eprintln!(
1736                        "API playback deferred history update failed: {}",
1737                        failure.code
1738                    );
1739                }
1740                Err(_) => eprintln!("API playback deferred history task stopped"),
1741                Ok(Ok(outcome)) if outcome.failed => {
1742                    eprintln!("API playback deferred history rejected changed sources")
1743                }
1744                Ok(Ok(_)) => {}
1745            }
1746        }
1747    });
1748}

1750async fn playback_stream(
1751    mut socket: WebSocket,
1752    state: ApiState,
1753    captured: auth::Principal,
1754    permit: tokio::sync::OwnedSemaphorePermit,
1755    mut shutdown: crate::shutdown::Receiver,
1756) {
1757    let start = match receive_start(&mut socket, &state, &captured, &mut shutdown).await {
1758        Ok(start) => start,
1759        Err(Some(failure)) => {
1760            send_failure(&mut socket, failure, &mut shutdown).await;
1761            return;
1762        }
1763        Err(None) => return,
1764    };
1765    if let Err(failure) = authorized(&state, &captured, false, &mut shutdown).await {
1766        send_failure(&mut socket, failure, &mut shutdown).await;
1767        return;
1768    }
1769    if start.interactive {
1770        interactive::run(socket, state, captured, permit, shutdown, start).await;
1771        return;
1772    }
1773    let (sources, catalog) = match current_sources(&state, &start.references).await {
1774        Ok(sources) => sources,
1775        Err(failure) => {
1776            send_failure(&mut socket, failure, &mut shutdown).await;
1777            return;
1778        }
1779    };
1780    let settings = settings(catalog, state.data_dir.clone(), &start);
1781    let cancelled = std::sync::Arc::new(AtomicBool::new(false));
1782    let (mut receiver, mut worker) = spawn_producer(sources.clone(), settings, cancelled.clone());
1783    let mut timeline = ListeningTimeline::default();
1784    let end = drive(
1785        &mut socket,
1786        &state,
1787        &captured,
1788        &mut receiver,
1789        &mut shutdown,
1790        &sources,
1791        DriveMode {
1792            queue: start.queue,
1793            interactive: None,
1794        },
1795        &mut timeline,
1796    )
1797    .await;
1798    cancelled.store(true, AtomicOrdering::Release);
1799    // Closing the bounded receiver wakes a worker blocked on output. A
1800    // spawned blocking task cannot be forcibly cancelled, so its permit stays
1801    // reserved until it actually exits.
1802    drop(receiver);
1803    let mut listens = if end.sample_rate.is_some() {
1804        timeline.listens(end.acknowledged_frames, &sources)
1805    } else {
1806        Vec::new()
1807    };
1808    if !start.queue && !end.completed {
1809        // The original singleton contract completes only after the socket's
1810        // EOF handshake; additive queue markers must not change that rule.
1811        for listen in &mut listens {
1812            listen.completed = false;
1813        }
1814    }
1815    let record = (!listens.is_empty()).then(|| ListenRecord {
1816        state: state.clone(),
1817        captured: captured.clone(),
1818        listens: listens.clone(),
1819    });
1820    let needs_record = record.is_some();
1821    let worker_done = worker_finished(&mut worker).await;
1822    let (recorded, permit) = if worker_done {
1823        match record {
1824            Some(record) => record_listen(record, permit).await,
1825            None => (Ok(HistoryOutcome::default()), Some(permit)),
1826        }
1827    } else {
1828        retain_worker(permit, worker, record);
1829        (
1830            if needs_record {
1831                Err(StreamFailure::HISTORY_FAILED)
1832            } else {
1833                Ok(HistoryOutcome::default())
1834            },
1835            None,
1836        )
1837    };
1838    if end.completed {
1839        match recorded {
1840            Ok(outcome) => {
1841                for listen in listens
1842                    .iter()
1843                    .filter(|listen| outcome.saved.contains(&listen.index))
1844                {
1845                    if send_json(
1846                        &mut socket,
1847                        RecordedFrame {
1848                            kind: "recorded",
1849                            ms_played: listen.ms_played,
1850                            completed: listen.completed,
1851                            index: start.queue.then_some(listen.index),
1852                            track: start.queue.then(|| listen.source.reference.to_token()),
1853                        },
1854                    )
1855                    .await
1856                    .is_err()
1857                    {
1858                        return;
1859                    }
1860                }
1861                if outcome.failed {
1862                    send_failure(&mut socket, StreamFailure::HISTORY_SOURCE, &mut shutdown).await;
1863                } else {
1864                    send_close(&mut socket).await;
1865                }
1866            }
1867            Err(failure) => send_failure(&mut socket, failure, &mut shutdown).await,
1868        }
1869    } else if end.notify {
1870        send_failure(
1871            &mut socket,
1872            end.failure.unwrap_or(StreamFailure::STREAM_FAILED),
1873            &mut shutdown,
1874        )
1875        .await;
1876    }
1877    // Keep the connection's playback permit until its final protocol message
1878    // has been attempted. A timed-out worker/history reaper owns it instead.
1879    drop(permit);
1880}
