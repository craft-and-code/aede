1//! Resolve only the CLI's canonical album-playlist destination, under confirmed shared roots.
2use super::*;
3
4pub(super) fn target(
5    state: &ApiState,
6    scope: &Scope,
7    reference: &str,
8) -> Result<(Vec<String>, Option<PathBuf>), ApiError> {
9    bounded_text(reference, "ref", MAX_REFERENCE_BYTES)?;
10    let folder = {
11        let held = state.catalog.blocking_read();
12        let catalog = held.as_ref().ok_or_else(unavailable)?;
13        if catalog.scanned_at != scope.scanned_at {
14            return Err(failure(library_files::Error::Changed));
15        }
16        let id = crate::catalog::filter_reference(catalog, reference, EntityKind::Release)?;
17        PathBuf::from(&catalog.release(id).ok_or_else(unavailable)?.folder)
18    };
19    // Release paths are catalog facts, never client-selected filesystem arguments.
20    // Resolve them after releasing the catalog guard; final Scope confirmation catches retargeted roots.
21    let canonical = folder
22        .canonicalize()
23        .map_err(|_| failure(library_files::Error::Unavailable))?;
24    let admitted = scope
25        .canonical_paths
26        .iter()
27        .take(scope.roots.len())
28        .flatten()
29        .any(|root| canonical.starts_with(root));
30    let excluded = scope
31        .canonical_paths
32        .iter()
33        .skip(scope.roots.len())
34        .flatten()
35        .any(|path| canonical.starts_with(path))
36        || scope
37            .excluded
38            .iter()
39            .any(|path| canonical.starts_with(path));
40    if !admitted || excluded {
41        return Ok((Vec::new(), None));
42    }
43    let target = canonical.join(aede_core::playlist::file_name(&folder));
44    Ok((
45        vec![
46            canonical
47                .to_str()
48                .ok_or_else(|| failure(library_files::Error::InvalidPath))?
49                .to_owned(),
50        ],
51        Some(target),
52    ))
53}
