13const MAX_ACTIONS: usize = 128;
14const MAX_LISTENS: usize = 256;

16const MAX_EXACT_INTEGER: u64 = (1u64 << 53) - 1;

61pub(super) struct Epoch {
62    number: u64,
63    entries: Vec<PlaybackEntry>,
64    base: PlaybackState,
65    sender: Option<mpsc::Sender<Snapshot>>,
66    failed: Arc<AtomicBool>,
67}
68
69impl Epoch {
70    pub(super) fn checkpoint_failed(&self) -> bool {
71        self.failed.load(AtomicOrdering::Acquire)
72    }
73
74    fn snapshot(&self, timeline: &ListeningTimeline, consumed: u64) -> PlaybackState {
75        let mut snapshot = self.base.clone();
76        if let Some((index, position, at_end)) = timeline.cursor(consumed) {
77            let next = index + usize::from(at_end);
78            snapshot.current_occurrence = self.entries.get(next).map(|entry| entry.occurrence);
79            snapshot.position_ms = if at_end { 0 } else { position };
80        }
81        snapshot.settings.sample_rate = timeline.output_rate().or(snapshot.settings.sample_rate);
82        snapshot.updated_at = clock::now_seconds();
83        snapshot
84    }
85
86    pub(super) fn checkpoint(&self, timeline: &ListeningTimeline, consumed: u64) {
87        if let Some(sender) = &self.sender {
88            // One pending periodic write is enough. A final/control write is
89            // awaited separately and always follows any queued periodic write.
90            let _ = sender.try_send(Snapshot {
91                state: self.snapshot(timeline, consumed),
92                expected_revision: self.base.revision,
93                reply: None,
94            });
95        }
96    }
97}

99pub(super) async fn send_stream_json(
100    socket: &mut WebSocket,
101    value: &impl Serialize,
102    epoch: Option<&Epoch>,
103) -> Result<(), ()> {
104    let Some(epoch) = epoch else {
105        return send_json(socket, value).await;
106    };
107    let mut value = serde_json::to_value(value).map_err(|_| ())?;
108    let object = value.as_object_mut().ok_or(())?;
109    object.insert("epoch".into(), epoch.number.into());
110    object.insert("revision".into(), epoch.base.revision.into());
111    if let Some(index) = object.get("index").and_then(serde_json::Value::as_u64) {
112        let entry = usize::try_from(index)
113            .ok()
114            .and_then(|index| epoch.entries.get(index))
115            .ok_or(())?;
116        object.insert("occurrence".into(), entry.occurrence.into());
117    }
118    send_json(socket, value).await
119}

121pub(super) async fn receive(
122    socket: &mut WebSocket,
123    acknowledgements: &mut Acknowledgements,
124    epoch: Option<&Epoch>,
125) -> Result<Incoming, StreamFailure> {
126    let Some(epoch) = epoch else {
127        return receive_ack(socket, acknowledgements).await.map(|open| {
128            if open {
129                Incoming::Continue
130            } else {
131                Incoming::Closed
132            }
133        });
134    };
135    let Some(message) = socket.recv().await else {
136        return Ok(Incoming::Closed);
137    };
138    match message.map_err(|_| StreamFailure::STREAM_FAILED)? {
139        Message::Text(input) => {
140            let frame: Frame =
141                serde_json::from_str(&input).map_err(|_| StreamFailure::INVALID_CONTROL)?;
142            let (number, frames, control) = match frame {
143                Frame::Ack { epoch, frames } => (epoch, frames, None),
144                Frame::Seek {
145                    epoch,
146                    frames,
147                    position_ms,
148                } if position_ms <= MAX_POSITION_MS => {
149                    (epoch, frames, Some(Control::Seek(position_ms)))
150                }
151                Frame::Seek { .. } => return Err(StreamFailure::INVALID_SEEK),
152                Frame::EditQueue {
153                    epoch,
154                    frames,
155                    revision,
156                    items,
157                } if items.len() <= MAX_QUEUE_TRACKS => {
158                    (epoch, frames, Some(Control::Edit { revision, items }))
159                }
160                Frame::EditQueue { .. } => return Err(StreamFailure::INVALID_CONTROL),
161                Frame::Stop { epoch, frames } => (epoch, frames, Some(Control::Stop)),
162            };
163            if number != epoch.number {
164                return Err(StreamFailure::INVALID_CONTROL);
165            }
166            acknowledgements.acknowledge(frames, Instant::now())?;
167            Ok(control.map_or(Incoming::Continue, Incoming::Control))
168        }
169        Message::Ping(payload) => {
170            events::send_message(socket, Message::Pong(payload))
171                .await
172                .map_err(|_| StreamFailure::STREAM_FAILED)?;
173            Ok(Incoming::Continue)
174        }
175        Message::Pong(_) => Ok(Incoming::Continue),
176        Message::Close(_) => Ok(Incoming::Closed),
177        Message::Binary(_) => Err(StreamFailure::INVALID_CONTROL),
178    }
179}

181fn entry(source: &TrackSource, occurrence: u64) -> PlaybackEntry {
182    PlaybackEntry {
183        occurrence,
184        track: source.reference.clone(),
185        source: PlaybackSource {
186            size: source.file.size,
187            mtime: source.file.mtime,
188            mtime_ns: Some(source.mtime_subseconds),
189        },
190    }
191}
192
193fn sources_match(entries: &[PlaybackEntry], sources: &[TrackSource]) -> bool {
194    entries.len() == sources.len()
195        && entries.iter().zip(sources).all(|(entry, source)| {
196            entry.track == source.reference
197                && entry.source.size == source.file.size
198                && entry.source.mtime == source.file.mtime
199                && entry.source.mtime_ns == Some(source.mtime_subseconds)
200        })
201}

203async fn initial_state(
204    state: &ApiState,
205    captured: &auth::Principal,
206    start: &Start,
207) -> Result<PlaybackState, StreamFailure> {
208    if start
209        .profile
210        .as_deref()
211        .is_some_and(|profile| !user::valid_playback_profile(profile))
212    {
213        return Err(StreamFailure::INVALID_START);
214    }
215    let previous = if let Some(profile) = &start.profile {
216        let state = state.clone();
217        let captured = captured.clone();
218        let profile = profile.clone();
219        tokio::task::spawn_blocking(move || read_profile(&state, &captured, &profile))
220            .await
221            .map_err(|_| StreamFailure::STATE_FAILED)??
222    } else {
223        None
224    };
225    let references = if start.resume {
226        previous
227            .as_ref()
228            .ok_or(StreamFailure::STATE_FAILED)?
229            .entries
230            .iter()
231            .map(|entry| entry.track.clone())
232            .collect()
233    } else {
234        start.references.clone()
235    };
236    let (sources, _) = current_sources(state, &references).await?;
237    let sources = tokio::task::spawn_blocking(move || {
238        sources.iter().try_for_each(validate_source)?;
239        Ok::<_, StreamFailure>(sources)
240    })
241    .await
242    .map_err(|_| StreamFailure::STATE_FAILED)??;
243    let mut snapshot = if start.resume {
244        let previous = previous.clone().ok_or(StreamFailure::STATE_FAILED)?;
245        if !sources_match(&previous.entries, &sources) {
246            return Err(StreamFailure::SOURCE_CHANGED);
247        }
248        previous
249    } else {
250        PlaybackState {
251            owner: captured.owner.clone(),
252            profile: start.profile.clone().unwrap_or_else(|| "volatile".into()),
253            session_id: String::new(),
254            revision: 1,
255            entries: sources
256                .iter()
257                .enumerate()
258                .map(|(index, source)| entry(source, index as u64 + 1))
259                .collect(),
260            current_occurrence: Some(1),
261            position_ms: 0,
262            settings: PlaybackSettings {
263                normalize: start.normalize,
264                sample_rate: start.output_rate,
265                bass_db: start.tone.bass_db(),
266                treble_db: start.tone.treble_db(),
267            },
268            updated_at: clock::now_seconds(),
269        }
270    };
271    if snapshot.position_ms > MAX_POSITION_MS {
272        return Err(StreamFailure::INVALID_SEEK);
273    }
274    snapshot.revision = previous
275        .as_ref()
276        .map_or(Some(1), |state| state.revision.checked_add(1))
277        .filter(|revision| *revision <= MAX_EXACT_INTEGER)
278        .ok_or(StreamFailure::STATE_FAILED)?;
279    snapshot.session_id =
280        aede_core::accounts::random_token().map_err(|_| StreamFailure::STATE_FAILED)?;
281    snapshot.updated_at = clock::now_seconds();
282    if start.profile.is_some() {
283        let state = state.clone();
284        let captured = captured.clone();
285        let initial = snapshot.clone();
286        tokio::task::spawn_blocking(move || {
287            persist_snapshot(
288                &state,
289                &captured,
290                initial,
291                previous
292                    .as_ref()
293                    .map(|state| (state.session_id.as_str(), state.revision)),
294            )
295        })
296        .await
297        .map_err(|_| StreamFailure::STATE_FAILED)??;
298    }
299    Ok(snapshot)
300}

302async fn reset_ack(
303    socket: &mut WebSocket,
304    state: &ApiState,
305    captured: &auth::Principal,
306    shutdown: &mut broadcast::Receiver<()>,
307    epoch: u64,
308) -> Result<(), StreamFailure> {
309    #[derive(Deserialize)]
310    #[serde(deny_unknown_fields)]
311    struct ResetAck {
312        #[serde(rename = "type")]
313        kind: String,
314        epoch: u64,
315    }
316    let deadline = tokio::time::sleep(START_TIMEOUT);
317    tokio::pin!(deadline);
318    let mut authorization = tokio::time::interval(AUTH_RECHECK_INTERVAL);
319    loop {
320        tokio::select! {
321            _ = &mut deadline => return Err(StreamFailure::ACK_TIMEOUT),
322            _ = shutdown.recv() => return Err(StreamFailure::SERVER_SHUTDOWN),
323            _ = authorization.tick() => {
324                if !authorized(state,captured,false).await { return Err(StreamFailure::AUTHENTICATION_EXPIRED) }
325            }
326            incoming = socket.recv() => match incoming {
327                Some(Ok(Message::Text(input))) => {
328                    let ack: ResetAck = serde_json::from_str(&input).map_err(|_| StreamFailure::INVALID_CONTROL)?;
329                    if ack.kind != "reset_ack" || ack.epoch != epoch { return Err(StreamFailure::INVALID_CONTROL) }
330                    if !authorized(state,captured,true).await { return Err(StreamFailure::AUTHENTICATION_EXPIRED) }
331                    return Ok(());
332                }
333                Some(Ok(Message::Ping(payload))) => { events::send_message(socket,Message::Pong(payload)).await.map_err(|_|StreamFailure::STREAM_FAILED)?; }
334                Some(Ok(Message::Pong(_))) => {}
335                _ => return Err(StreamFailure::STREAM_FAILED),
336            }
337        }
338    }
339}

341async fn announce(
342    socket: &mut WebSocket,
343    snapshot: &PlaybackState,
344    epoch: u64,
345    reset: bool,
346) -> Result<(), StreamFailure> {
347    let items: Vec<_> = snapshot.entries.iter().map(|entry| serde_json::json!({"occurrence":entry.occurrence,"track":entry.track.to_token()})).collect();
348    send_json(socket, serde_json::json!({
349        "type":if reset {"reset"} else {"queue"}, "epoch":epoch,"revision":snapshot.revision,
350        "items":items,"current_occurrence":snapshot.current_occurrence,"position_ms":snapshot.position_ms
351    })).await.map_err(|_| StreamFailure::STREAM_FAILED)
352}

354fn selected_index(snapshot: &PlaybackState) -> Option<usize> {
355    snapshot.current_occurrence.and_then(|id| {
356        snapshot
357            .entries
358            .iter()
359            .position(|entry| entry.occurrence == id)

363async fn edit(
364    state: &ApiState,
365    snapshot: &mut PlaybackState,
366    revision: u64,
367    items: Vec<Item>,
368    next_id: &mut u64,
369) -> Result<(), StreamFailure> {
370    if revision != snapshot.revision {
371        return Err(StreamFailure::INVALID_CONTROL);
372    }
373    let prefix = selected_index(snapshot).map_or(snapshot.entries.len(), |index| index + 1);
374    if prefix + items.len() > MAX_QUEUE_TRACKS {
375        return Err(StreamFailure::INVALID_CONTROL);
376    }
377    let references = items
378        .iter()
379        .filter_map(|item| item.track.as_deref())
380        .map(|track| {
381            EntityRef::parse_token(track)
382                .filter(|reference| reference.kind == EntityKind::Track)
383                .ok_or(StreamFailure::INVALID_CONTROL)
384        })
385        .collect::<Result<Vec<_>, _>>()?;
386    let (new_sources, _) = current_sources(state, &references).await?;
387    let mut new_sources = new_sources.into_iter();
388    let mut seen = BTreeSet::new();
389    let mut future = Vec::new();
390    let mut next = *next_id;
391    for item in items {
392        match (item.occurrence, item.track) {
393            (Some(id), None) if seen.insert(id) => {
394                let entry = snapshot.entries[prefix..]
395                    .iter()
396                    .find(|entry| entry.occurrence == id)
397                    .ok_or(StreamFailure::INVALID_CONTROL)?;
398                future.push(entry.clone());
399            }
400            (None, Some(_)) => {
401                let source = new_sources.next().ok_or(StreamFailure::TRACK_NOT_FOUND)?;
402                next = next
403                    .checked_add(1)
404                    .filter(|id| *id <= MAX_EXACT_INTEGER)
405                    .ok_or(StreamFailure::INVALID_CONTROL)?;
406                future.push(entry(&source, next));
407            }
408            _ => return Err(StreamFailure::INVALID_CONTROL),
409        }
410    }
411    if snapshot.current_occurrence.is_none() {
412        snapshot.current_occurrence = future.first().map(|entry| entry.occurrence);
413        snapshot.position_ms = 0;
414    }
415    snapshot.entries.truncate(prefix);
416    snapshot.entries.extend(future);
417    *next_id = next;
418    Ok(())
419}

421struct Visit {
422    order: usize,
423    frames: u64,
424    sample_rate: u32,
425    occurrence: u64,
426    listen: history::Listened,
427    moved: bool,
428}
429
430fn accumulate(
431    visits: &mut BTreeMap<u64, Visit>,
432    epoch: &Epoch,
433    timeline: &ListeningTimeline,
434    consumed: u64,
435    sources: &[TrackSource],
436    moved: &BTreeSet<u64>,
437) -> Result<(), StreamFailure> {
438    for (listen, frames, sample_rate) in timeline.acknowledged(consumed, sources) {
439        let id = epoch
440            .entries
441            .get(listen.index)
442            .ok_or(StreamFailure::STREAM_FAILED)?
443            .occurrence;
444        if let Some(visit) = visits.get_mut(&id) {
445            if visit.sample_rate != sample_rate {
446                return Err(StreamFailure::STREAM_FAILED);
447            }
448            visit.frames = visit
449                .frames
450                .checked_add(frames)
451                .ok_or(StreamFailure::STREAM_FAILED)?;
452            visit.listen.ms_played = frames_to_milliseconds(visit.frames, sample_rate);
453            visit.moved |= moved.contains(&id);
454            // An edit restart continues one visit from its consumed cursor.
455            visit.listen.completed = !visit.moved && listen.completed;
456        } else {
457            if visits.len() >= MAX_LISTENS {
458                return Err(StreamFailure::INVALID_CONTROL);
459            }
460            let mut listen = listen;
461            let was_moved = moved.contains(&id) || epoch.base.position_ms > 0 && listen.index == 0;
462            listen.completed &= !was_moved;
463            visits.insert(
464                id,
465                Visit {
466                    order: visits.len(),
467                    frames,
468                    sample_rate,
469                    occurrence: id,
470                    listen,
471                    moved: was_moved,
472                },
473            );
474        }
475    }
476    Ok(())
477}

479pub(super) async fn run(
480    mut socket: WebSocket,
481    state: ApiState,
482    captured: auth::Principal,
483    permit: tokio::sync::OwnedSemaphorePermit,
484    mut shutdown: broadcast::Receiver<()>,
485    mut start: Start,
486) {
487    start.queue = true;
488    let lease = match start.profile.as_deref() {
489        Some(profile) => match ProfileLease::acquire(&state, &captured.owner, profile) {
490            Ok(lease) => Some(lease),
491            Err(failure) => {
492                send_failure(&mut socket, failure).await;
493                return;
494            }
495        },
496        None => None,
497    };
498    let mut snapshot = match initial_state(&state, &captured, &start).await {
499        Ok(initial) => initial,
500        Err(failure) => {
501            send_failure(&mut socket, failure).await;
502            return;
503        }
504    };
505    let writer = start
506        .profile
507        .as_ref()
508        .map(|_| Writer::spawn(state.clone(), captured.clone()));
509    let mut number = 0;
510    let mut next_id = snapshot
511        .entries
512        .iter()
513        .map(|entry| entry.occurrence)
514        .max()
515        .unwrap_or(0);
516    let mut visits = BTreeMap::new();
517    let mut moved = BTreeSet::new();
518    let mut failure = announce(&mut socket, &snapshot, number, false).await.err();
519    let mut final_worker = None;
520    let mut completed = false;
521    let mut actions = 0;
522    let mut seek_backup: Option<PlaybackState> = None;
523    while failure.is_none() {
524        let Some(index) = selected_index(&snapshot) else {
525            completed = true;
526            break;
527        };
528        let entries = snapshot.entries[index..].to_vec();
529        let references: Vec<_> = entries.iter().map(|entry| entry.track.clone()).collect();
530        let (sources, catalog) = match current_sources(&state, &references).await {
531            Ok(value) => value,
532            Err(error) => {
533                failure = Some(error);
534                break;
535            }
536        };
537        if !sources_match(&entries, &sources) {
538            failure = Some(StreamFailure::SOURCE_CHANGED);
539            break;
540        }
541        start.normalize = snapshot.settings.normalize;
542        start.output_rate = snapshot.settings.sample_rate;
543        start.tone = match ToneControls::new(snapshot.settings.bass_db, snapshot.settings.treble_db)
544        {
545            Ok(tone) => tone,
546            Err(_) => {
547                failure = Some(StreamFailure::STATE_FAILED);
548                break;
549            }
550        };
551        let mut settings = settings(catalog, state.data_dir.clone(), &start);
552        settings.first_position_ms = snapshot.position_ms;
553        let epoch = Epoch {
554            number,
555            entries,
556            base: snapshot.clone(),
557            sender: writer.as_ref().map(|writer| writer.sender.clone()),
558            failed: writer.as_ref().map_or_else(
559                || Arc::new(AtomicBool::new(false)),
560                |writer| writer.failed.clone(),
561            ),
562        };
563        let cancelled = Arc::new(AtomicBool::new(false));
564        let (mut receiver, mut worker) =
565            spawn_producer(sources.clone(), settings, cancelled.clone());
566        let mut timeline = ListeningTimeline::default();
567        let end = drive(
568            &mut socket,
569            &state,
570            &captured,
571            &mut receiver,
572            &mut shutdown,
573            &sources,
574            DriveMode {
575                queue: true,
576                interactive: Some(&epoch),
577            },
578            &mut timeline,
579        )
580        .await;
581        cancelled.store(true, AtomicOrdering::Release);
582        drop(receiver);
583        let worker_done = worker_finished(&mut worker).await;
584        if let Err(error) = accumulate(
585            &mut visits,
586            &epoch,
587            &timeline,
588            end.acknowledged_frames,
589            &sources,
590            &moved,
591        ) {
592            failure = Some(error)
593        }
594        snapshot = if end.sample_rate.is_none() {
595            seek_backup
596                .take()
597                .unwrap_or_else(|| epoch.snapshot(&timeline, end.acknowledged_frames))
598        } else {
599            seek_backup = None;
600            epoch.snapshot(&timeline, end.acknowledged_frames)
601        };
602        snapshot.settings.sample_rate = end.sample_rate.or(snapshot.settings.sample_rate);
603        if end.completed {
604            snapshot.current_occurrence = None;
605            snapshot.position_ms = 0;
606            completed = true
607        }
608        if !worker_done {
609            final_worker = Some(worker);
610            failure = Some(StreamFailure::STREAM_FAILED);
611            break;
612        }
613        if failure.is_some() {
614            break;
615        }
616        if let Some(control) = end.control {
617            actions += 1;
618            if actions > MAX_ACTIONS {
619                failure = Some(StreamFailure::INVALID_CONTROL);
620                break;
621            }
622            let previous_revision = snapshot.revision;
623            match control {
624                Control::Stop => break,
625                Control::Seek(position) => {
626                    let Some(id) = snapshot.current_occurrence else {
627                        failure = Some(StreamFailure::INVALID_SEEK);
628                        break;
629                    };
630                    moved.insert(id);
631                    if let Some(visit) = visits.get_mut(&id) {
632                        visit.moved = true;
633                        visit.listen.completed = false
634                    }
635                    seek_backup = Some(snapshot.clone());
636                    snapshot.position_ms = position;
637                }
638                Control::Edit { revision, items } => {
639                    if let Err(error) =
640                        edit(&state, &mut snapshot, revision, items, &mut next_id).await
641                    {
642                        failure = Some(error);
643                        break;
644                    }
645                }
646            }
647            snapshot.revision = match snapshot
648                .revision
649                .checked_add(1)
650                .filter(|revision| *revision <= MAX_EXACT_INTEGER)
651            {
652                Some(revision) => revision,
653                None => {
654                    failure = Some(StreamFailure::INVALID_CONTROL);
655                    break;
656                }
657            };
658            if let Some(writer) = &writer {
659                let mut published = seek_backup.clone().unwrap_or_else(|| snapshot.clone());
660                published.revision = snapshot.revision;
661                if let Some(backup) = &mut seek_backup {
662                    backup.revision = snapshot.revision;
663                }
664                if let Err(error) = writer.save(published, previous_revision).await {
665                    failure = Some(error);
666                    break;
667                }
668            }
669            number += 1;
670            if let Err(error) = announce(&mut socket, &snapshot, number, true).await {
671                failure = Some(error);
672                break;
673            }
674            if let Err(error) =
675                reset_ack(&mut socket, &state, &captured, &mut shutdown, number).await
676            {
677                failure = Some(error);
678                break;
679            }
680        } else {
681            failure = end.failure;
682            break;
683        }
684    }
685    if let Some(confirmed) = seek_backup.take() {
686        snapshot = confirmed;
687    }
688    // Saving a resume point is independent of listening history and never
689    // fills in frames that the client did not acknowledge.
690    if let Some(writer) = &writer
691        && let Err(error) = writer.save(snapshot.clone(), snapshot.revision).await
692        && failure.is_none()
693    {
694        failure = Some(error);
695    }
696    let mut ordered_visits: Vec<_> = visits
697        .values()
698        .filter(|visit| visit.listen.ms_played > 0)
699        .collect();
700    ordered_visits.sort_by_key(|visit| visit.order);
701    let listens: Vec<_> = ordered_visits
702        .iter()
703        .enumerate()
704        .map(|(index, visit)| {
705            let mut listen = visit.listen.clone();
706            listen.index = index;
707            listen
708        })
709        .collect();
710    let record = (!listens.is_empty()).then(|| ListenRecord {
711        state: state.clone(),
712        captured: captured.clone(),
713        listens: listens.clone(),
714    });
715    let mut writer_worker = writer.map(|writer| {
716        drop(writer.sender);
717        writer.worker
718    });
719    let writer_done = if let Some(worker) = &mut writer_worker {
720        worker_finished(worker).await
721    } else {
722        true
723    };
724    let (recorded, permit) = if final_worker.is_some() || !writer_done {
725        let deferred_worker = final_worker;
726        let deferred_writer = writer_worker;
727        tokio::spawn(async move {
728            let _permit = permit;
729            let _lease = lease;
730            if let Some(worker) = deferred_worker {
731                let _ = worker.await;
732            }
733            if let Some(worker) = deferred_writer {
734                let _ = worker.await;
735            }
736            if let Some(record) = record {
737                let _ = tokio::task::spawn_blocking(move || record.persist()).await;
738            }
739        });
740        (Err(StreamFailure::HISTORY_FAILED), None)
741    } else {
742        drop(lease);
743        match record {
744            Some(record) => record_listen(record, permit).await,
745            None => (Ok(HistoryOutcome::default()), Some(permit)),
746        }
747    };
748    match recorded {
749        Ok(outcome) => {
750            for (_, visit) in ordered_visits
751                .iter()
752                .enumerate()
753                .filter(|(index, _)| outcome.saved.contains(index))
754            {
755                if send_json(&mut socket,serde_json::json!({"type":"recorded","occurrence":visit.occurrence,"track":visit.listen.source.reference.to_token(),"ms_played":visit.listen.ms_played,"completed":visit.listen.completed})).await.is_err(){break}
756            }
757            if outcome.failed && failure.is_none() {
758                failure = Some(StreamFailure::HISTORY_FAILED)
759            }
760        }
761        Err(error) => {
762            if failure.is_none() {
763                failure = Some(error)
764            }
765        }
766    }
767    if let Some(failure) = failure {
768        send_failure(&mut socket, failure).await;
769    } else {
770        let _=send_json(&mut socket,serde_json::json!({"type":"saved","profile":start.profile,"revision":snapshot.revision,"completed":completed})).await;
771        send_close(&mut socket).await;
772    }
773    drop(permit);
774}
