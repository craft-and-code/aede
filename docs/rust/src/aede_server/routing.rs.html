5pub(super) fn router(state: ApiState, address: SocketAddr) -> Router {
6    router_with_policy(state, OriginPolicy::Local(address))
7}

9/// HTTPS administration requires a named administrator session, never a legacy token.
10pub(super) fn remote_router(mut state: ApiState, authority: RemoteAuthority) -> Router {
11    if let Some(admin) = state.admin.as_mut() {
12        admin.token.clear();
13    }
14    router_with_policy(state, OriginPolicy::Remote(authority))
15}

17fn application_routes(admin_enabled: bool) -> Router<ApiState> {
18    let mut routes = Router::new()
19        .route("/", get(landing::page))
20        .route("/api/v1/status", get(status))
21        .route("/api/v1/library", get(library))
22        .route("/api/v1/artists", get(artists))
23        .route("/api/v1/releases", get(releases))
24        .route("/api/v1/tracks", get(tracks))
25        .route("/api/v1/recordings", get(recordings))
26        .route("/api/v1/entities", get(entity))
27        .route("/api/v1/events", get(events))
28        .route("/api/v1/activity", get(activity))
29        .merge(accounts_api::routes())
30        .merge(personal::account_routes())
31        .merge(playback_api::routes())
32        .merge(exports::routes())
33        .merge(crate::parental::http::routes())
34        .merge(subsonic::routes())
35        .merge(catalog_commands::routes())
36        .merge(inspection::routes())
37        .merge(lyrics::routes())
38        .merge(waveform::routes())
39        .merge(audio_features::routes())
40        .merge(presentation::routes())
41        .merge(artwork::routes())
42        .merge(analysis_report::routes());
43    if admin_enabled {
44        routes = routes
45            .route("/api/admin/v1/scan", post(jobs::scan_route))
46            .merge(jobs::routes())
47            .merge(personal::routes());
48    }
49    routes
50        .method_not_allowed_fallback(method_not_allowed)
51        .fallback(not_found)
52}

54fn router_with_policy(state: ApiState, policy: OriginPolicy) -> Router {
55    use tower::ServiceExt;
56    let remote = matches!(policy, OriginPolicy::Remote(_));
57    let access_log = access_log::AccessLog::from_environment();
58    let mut template = application_routes(state.admin.is_some());
59    if access_log.is_some() {
60        template = template.layer(middleware::from_fn(access_log::label));
61    }
62    let mut routes =
63        Router::new().fallback(move |State(state): State<ApiState>, request: Request| {
64            let template = template.clone();
65            async move {
66                let principal = request.extensions().get::<auth::Principal>().cloned();
67                let scoped = match principal.clone() {
68                    Some(owner)
69                        if !request.uri().path().starts_with("/api/auth/")
70                            && !request.uri().path().starts_with("/api/admin/")
71                            && !request.uri().path().starts_with("/api/me/v1/parental") =>
72                    {
73                        crate::parental::scoped(
74                            state,
75                            owner.owner,
76                            owner.role == aede_core::accounts::Role::Administrator,
77                            request.uri().path() != "/api/me/v1/playback",
78                        )
79                        .await
80                    }
81                    _ => Ok(state),
82                };
83                let scoped = match scoped {
84                    Ok(state) => state,
85                    Err(error) => return error.into_response(),
86                };
87                if let Err(error) = crate::parental::route_allowed(&scoped, request.uri().path()) {
88                    return error.into_response();
89                }
90                let response = match template.with_state(scoped.clone()).oneshot(request).await {
91                    Ok(response) => response,
92                    Err(never) => match never {},
93                };
94                if scoped.parental_scope.is_some() {
95                    let for_check = scoped.clone();
96                    if let Err(error) = crate::parental::blocking(&scoped, move || {
97                        if let Some(principal) = &principal {
98                            auth::recheck(&for_check, principal)?;
99                        }
100                        crate::parental::finish(&for_check)
101                    })
102                    .await
103                    {
104                        return error.into_response();
105                    }
106                }
107                response
108            }
109        });
110    if remote {
111        routes = routes.layer(middleware::from_fn_with_state(
112            state.clone(),
113            enforce_remote_request_budget,
114        ));
115    }
116    let routes = routes
117        .layer(middleware::from_fn_with_state(
118            state.clone(),
119            auth::enforce_accounts,
120        ))
121        .layer(middleware::from_fn_with_state(policy, enforce_origin))
122        .with_state(state);
123    match access_log {
124        Some(log) => routes.layer(middleware::from_fn_with_state(log, access_log::record)),
125        None => routes,
126    }
127}
