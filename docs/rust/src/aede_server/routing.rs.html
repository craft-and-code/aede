5pub(super) fn router(state: ApiState, address: SocketAddr) -> Router {
6    router_with_policy(state, OriginPolicy::Local(address))
7}
8
9/// Build the restricted HTTPS router. Administrative and legacy-token routes
10/// deliberately do not exist on a remotely reachable listener.
11pub(super) fn remote_router(mut state: ApiState, authority: RemoteAuthority) -> Router {
12    state.admin = None;
13    router_with_policy(state, OriginPolicy::Remote(authority))
14}
15
16fn router_with_policy(state: ApiState, policy: OriginPolicy) -> Router {
17    let remote = matches!(policy, OriginPolicy::Remote(_));
18    let admin_enabled = !remote && state.admin.is_some();
19    let mut routes = Router::new()
20        .route("/api/v1/status", get(status))
21        .route("/api/v1/library", get(library))
22        .route("/api/v1/artists", get(artists))
23        .route("/api/v1/releases", get(releases))
24        .route("/api/v1/tracks", get(tracks))
25        .route("/api/v1/recordings", get(recordings))
26        .route("/api/v1/entities", get(entity))
27        .route("/api/v1/events", get(events))
28        .route("/api/v1/activity", get(activity))
29        .merge(if remote {
30            accounts_api::remote_routes()
31        } else {
32            accounts_api::routes()
33        })
34        .merge(personal::account_routes())
35        .merge(playback_api::routes())
36        .merge(subsonic::routes())
37        .merge(catalog_commands::routes())
38        .merge(inspection::routes());
39    if admin_enabled {
40        routes = routes
41            .route("/api/admin/v1/scan", post(jobs::scan_route))
42            .merge(jobs::routes())
43            .merge(personal::routes());
44    }
45    if remote {
46        routes = routes.layer(middleware::from_fn_with_state(
47            state.clone(),
48            enforce_remote_request_budget,
49        ));
50    }
51    routes
52        .method_not_allowed_fallback(method_not_allowed)
53        .fallback(not_found)
54        .layer(middleware::from_fn_with_state(
55            state.clone(),
56            auth::enforce_accounts,
57        ))
58        .layer(middleware::from_fn_with_state(policy, enforce_origin))
59        .with_state(state)
60}
