96fn spawn_personal<T: Send + 'static>(
97    slots: Arc<Semaphore>,
98    work: impl FnOnce() -> Result<T, ApiError> + Send + 'static,
99) -> Result<tokio::task::JoinHandle<Result<T, ApiError>>, ApiError> {
100    let permit = slots.try_acquire_owned().map_err(|_| {
101        error(
102            StatusCode::SERVICE_UNAVAILABLE,
103            "personal_busy",
104            "personal data workers are busy; retry shortly",
105        )
106    })?;
107    Ok(tokio::task::spawn_blocking(move || {
108        // Keep the slot until the blocking work finishes even if HTTP disconnects.
109        let _permit = permit;
110        work()
111    }))
112}

114impl PersonalOwner {
115    fn from_request(state: &ApiState, request: &Request) -> Result<Self, ApiError> {
116        if request.uri().path().starts_with("/api/admin/v1/") {
117            require_admin(state, request)?;
118            let principal = request.extensions().get::<auth::Principal>().cloned();
119            if principal
120                .as_ref()
121                .is_some_and(|principal| principal.owner != LOCAL_USER)
122            {
123                return Err(error(
124                    StatusCode::FORBIDDEN,
125                    "forbidden",
126                    "use /api/me/v1 for this account's personal data",
127                ));
128            }
129            Ok(Self {
130                id: LOCAL_USER.into(),
131                principal,
132            })
133        } else {
134            let principal = auth::request_principal(request)?;
135            Ok(Self {
136                id: principal.owner.clone(),
137                principal: Some(principal),
138            })
139        }
140    }
141
142    fn recheck(&self, state: &ApiState) -> Result<(), ApiError> {
143        match &self.principal {
144            Some(principal) => auth::recheck(state, principal),
145            None => Ok(()),
146        }?;
147        crate::parental::confirm(state)
148    }
149
150    // Preparation may include blocking filesystem reads or catalog projection.
151    // Revalidate after that work, immediately before admitting its result.
152    fn admit<T>(&self, state: &ApiState, prepared: T) -> Result<T, ApiError> {
153        self.recheck(state)?;
154        Ok(prepared)
155    }
156
157    /// Administrative-token routes have no account role; session routes must
158    /// refuse the explicit read-only role before parsing or changing a body.
159    fn require_mutation(&self) -> Result<(), ApiError> {
160        match &self.principal {
161            Some(principal) => auth::require_mutation(principal),
162            None => Ok(()),
163        }
164    }
165}

187/// Distinguishes an omitted field from an explicit JSON null.
188#[derive(Default)]
189enum Field<T> {
190    #[default]
191    Missing,
192    Value(Option<T>),
193}
194
195impl<'de, T: Deserialize<'de>> Deserialize<'de> for Field<T> {
196    fn deserialize<D: Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
197        Option::<T>::deserialize(deserializer).map(Self::Value)
198    }
199}

201#[derive(Default, Deserialize)]
202#[serde(deny_unknown_fields)]
203struct AnnotationPatch {
204    #[serde(default)]
205    loved: Field<bool>,
206    #[serde(default)]
207    rating: Field<u8>,
208    #[serde(default)]
209    note: Field<String>,
210    #[serde(default)]
211    tags: Field<Vec<String>>,
212}
213
214impl AnnotationPatch {
215    fn validate(&self) -> Result<(), ApiError> {
216        if matches!(
217            self,
218            Self {
219                loved: Field::Missing,
220                rating: Field::Missing,
221                note: Field::Missing,
222                tags: Field::Missing,
223            }
224        ) {
225            return Err(error(
226                StatusCode::BAD_REQUEST,
227                "invalid_parameters",
228                "provide at least one annotation field",
229            ));
230        }
231        if matches!(self.loved, Field::Value(None)) {
232            return Err(error(
233                StatusCode::BAD_REQUEST,
234                "invalid_parameters",
235                "loved must be true or false",
236            ));
237        }
238        if matches!(self.rating, Field::Value(Some(0 | 6..))) {
239            return Err(error(
240                StatusCode::BAD_REQUEST,
241                "invalid_parameters",
242                "rating must be from 1 to 5, or null to remove it",
243            ));
244        }
245        if let Field::Value(Some(note)) = &self.note
246            && note.trim().is_empty()
247        {
248            return Err(error(
249                StatusCode::BAD_REQUEST,
250                "invalid_parameters",
251                "note must contain text, or be null to remove it",
252            ));
253        }
254        if let Field::Value(Some(tags)) = &self.tags {
255            validate_tags(tags)?;
256        }
257        Ok(())
258    }
259}

263struct PlayRequest {
264    track: String,
265    #[serde(default)]
266    at: Option<u64>,
267    ms_played: u64,
268    completed: bool,
269}
270
271impl PlayRequest {
272    fn validate(&self, now: u64) -> Result<EntityRef, ApiError> {
273        let track = EntityRef::parse_token(&self.track)
274            .filter(|reference| reference.kind == EntityKind::Track && !reference.key.is_empty())
275            .ok_or_else(|| invalid_reference("track must be a nonempty track reference"))?;
276        if self.ms_played > MAX_HISTORY_MS {
277            return Err(error(
278                StatusCode::BAD_REQUEST,
279                "invalid_parameters",
280                "ms_played must be at most 24 hours",
281            ));
282        }
283        if self
284            .at
285            .is_some_and(|at| at > now.saturating_add(24 * 60 * 60))
286        {
287            return Err(error(
288                StatusCode::BAD_REQUEST,
289                "invalid_parameters",
290                "at cannot be more than one day in the future",
291            ));
292        }
293        Ok(track)
294    }
295}

328            "request body did not finish",
329        )
330    })?
331    .map_err(|_| {
332        error(
333            StatusCode::BAD_REQUEST,
334            "invalid_body",
335            "request body is malformed or too large",
336        )
337    })

340pub(super) async fn json_body<T: DeserializeOwned>(request: Request) -> Result<T, ApiError> {
341    let bytes = body(request).await?;
342    if bytes
343        .iter()
344        .copied()
345        .find(|byte| !byte.is_ascii_whitespace())
346        != Some(b'{')
347    {
348        return Err(error(
349            StatusCode::BAD_REQUEST,
350            "invalid_body",
351            "expected a JSON object",
352        ));
353    }
354    serde_json::from_slice(&bytes).map_err(|_| {
355        error(
356            StatusCode::BAD_REQUEST,
357            "invalid_body",
358            "expected a JSON object with supported fields and types",
359        )
360    })
361}

381fn canonical_target(catalog: &Catalog, reference: &EntityRef) -> Result<EntityRef, ApiError> {
382    let id = reference.resolve(catalog).ok_or_else(|| {
383        error(
384            StatusCode::NOT_FOUND,
385            "entity_not_found",
386            "no entity matches this reference",
387        )
388    })?;
389    // Accepted legacy aliases must address the same personal row as current
390    // catalog responses, including after reconciliation or identification.
391    EntityRef::of(catalog, reference.kind, id).ok_or_else(unavailable)
392}

394fn load_personal(data_dir: &Path, catalog: &Catalog) -> Result<UserData, ApiError> {
395    let path = user::user_path(data_dir);
396    let mut data = user::load(&path)
397        .map_err(|failure| {
398            eprintln!("API user data read failed: {failure}");
399            error(
400                StatusCode::INTERNAL_SERVER_ERROR,
401                "user_unavailable",
402                "personal data could not be read",
403            )
404        })?
405        .unwrap_or_default();
406    user::reconcile(&mut data, catalog);
407    Ok(data)
408}

410// Catalog and user data must describe the same completed writer operation.
411// A cached catalog can predate user.json and reconcile a new track's note onto
412// a different track. Keep the lock through validation and any resulting save.
413fn locked_personal_full(data_dir: &Path) -> Result<(StoreLock, Catalog, UserData), ApiError> {
414    let guard = StoreLock::try_acquire(data_dir).map_err(|failure| {
415        if failure.kind() == std::io::ErrorKind::WouldBlock {
416            error(
417                StatusCode::CONFLICT,
418                "store_busy",
419                "Aède data is busy; retry after the current write",
420            )
421        } else {
422            error(
423                StatusCode::INTERNAL_SERVER_ERROR,
424                "store_error",
425                "Aède data could not be locked for the update",
426            )
427        }
428    })?;
429    let catalog = store::load(&store::catalog_path(data_dir))
430        .map_err(|failure| {
431            eprintln!("API personal catalog read failed: {failure}");
432            error(
433                StatusCode::INTERNAL_SERVER_ERROR,
434                "store_error",
435                "the current catalog could not be read",
436            )
437        })?
438        .ok_or_else(unavailable)?;
439    let data = load_personal(data_dir, &catalog)?;
440    Ok((guard, catalog, data))
441}
442
443fn locked_personal(state: &ApiState) -> Result<(StoreLock, Catalog, UserData), ApiError> {
444    let (guard, catalog, data) = locked_personal_full(&state.data_dir)?;
445    let visible = crate::parental::view_owned(state, catalog)?;
446    Ok((guard, visible, data))
447}

449fn locked_update<T>(
450    data_dir: &Path,
451    state: &ApiState,
452    owner: &PersonalOwner,
453    update: impl FnOnce(&Catalog, &mut UserData) -> Result<T, ApiError>,
454) -> Result<T, ApiError> {
455    let (_guard, catalog, mut data) = locked_personal(state)?;
456    owner.recheck(state)?;
457    let result = update(&catalog, &mut data)?;
458    if let Err(failure) = aede_core::achievements::refresh(
459        &catalog,
460        &mut data,
461        &owner.id,
462        false,
463        clock::now_seconds(),
464    ) {
465        eprintln!("Private achievement projection retained after failure: {failure}");
466    }
467    data.forget_empty();
468    owner.recheck(state)?;
469    user::save(&data, &user::user_path(data_dir)).map_err(|failure| {
470        eprintln!("API user data write failed: {failure}");
471        error(
472            StatusCode::INTERNAL_SERVER_ERROR,
473            "store_error",
474            "personal data could not be saved",
475        )
476    })?;
477    Ok(result)
478}

509async fn annotation(
510    State(state): State<ApiState>,
511    input: Result<Query<RefQuery>, QueryRejection>,
512    request: Request,
513) -> Result<Json<AnnotationResponse>, ApiError> {
514    let owner = PersonalOwner::from_request(&state, &request)?;
515    empty_body(&body(request).await?)?;
516    let reference = reference(parsed_query(input)?)?;
517    spawn_personal(state.inspection_slots.clone(), move || {

526            }),
527        )
528    })?
529    .await
530    .map_err(|_| {
531        error(
532            StatusCode::INTERNAL_SERVER_ERROR,
533            "personal_failed",
534            "personal data read could not finish",
535        )
536    })?

539async fn update_annotation(
540    State(state): State<ApiState>,
541    input: Result<Query<RefQuery>, QueryRejection>,
542    request: Request,
543) -> Result<Json<AnnotationResponse>, ApiError> {
544    let owner = PersonalOwner::from_request(&state, &request)?;
545    owner.require_mutation()?;
546    let patch: AnnotationPatch = json_body(request).await?;
547    patch.validate()?;
548    let reference = reference(parsed_query(input)?)?;
549    let data_dir = state.data_dir.clone();
550    spawn_personal(state.inspection_slots.clone(), move || {
551        let now = clock::now_seconds();
552        let saved = locked_update(&data_dir, &state, &owner, |catalog, data| {
553            let reference = canonical_target(catalog, &reference)?;
554            let entry = data.entry(&owner.id, &reference, now);
555            if let Field::Value(Some(loved)) = patch.loved {
556                entry.loved = loved;
557            }
558            if let Field::Value(rating) = patch.rating {
559                entry.rating = rating;
560            }
561            if let Field::Value(note) = patch.note {
562                entry.note = note;
563            }
564            if let Field::Value(tags) = patch.tags {
565                entry.tags = tags.unwrap_or_default().into_iter().collect();
566            }
567            entry.updated_at = now;
568            Ok(entry.clone())
569        })?;
570        Ok(Json(AnnotationResponse {
571            reference: saved.target.to_token(),
572            annotation: (!saved.is_empty()).then(|| annotation_view(&saved)),
573        }))
574    })?
575    .await
576    .map_err(|_| {
577        error(
578            StatusCode::INTERNAL_SERVER_ERROR,
579            "personal_failed",
580            "personal data update could not finish",
581        )
582    })?
583}

595fn play_view(data: &UserData, play: &Play) -> PlayView {
596    let count = data
597        .counts
598        .iter()
599        .find(|count| count.owner == play.owner && count.track == play.track);
600    play_view_with_count(play, count)
601}

603fn play_view_with_count(play: &Play, count: Option<&user::PlayCount>) -> PlayView {
604    PlayView {
605        track: play.track.to_token(),
606        at: play.at,
607        ms_played: play.ms_played,
608        completed: play.completed,
609        play_count: count.map_or(0, |count| count.count),
610        last_played: count.map_or(0, |count| count.last_played),
611    }
612}

614#[derive(Serialize)]
615struct HistoryResponse {
616    #[serde(flatten)]
617    page: Page<PlayView>,
618}

620fn history_page(
621    data: &UserData,
622    catalog: &Catalog,
623    owner: &str,
624    offset: usize,
625    limit: usize,
626) -> HistoryResponse {
627    let plays = data.plays.iter().filter(|play| play.owner == owner);
628    let total = plays.clone().count();
629    let mut counts = BTreeMap::new();
630    for count in data.counts.iter().filter(|count| count.owner == owner) {
631        // Keep the core's first-row behavior for legacy duplicates.
632        counts.entry(&count.track).or_insert(count);
633    }
634    let items = plays
635        .rev()
636        .skip(offset)
637        .take(limit)
638        .map(|play| play_view_with_count(play, counts.get(&play.track).copied()))
639        .collect();
640    HistoryResponse {
641        page: Page {
642            total,
643            items,
644            offset,
645            limit,
646            scanned_at: catalog.scanned_at,
647        },
648    }
649}
650
651fn page(query: PageQuery) -> Result<(usize, usize), ApiError> {
652    page_bounds(query.offset.as_deref(), query.limit.as_deref())
653}

655async fn history(
656    State(state): State<ApiState>,
657    input: Result<Query<PageQuery>, QueryRejection>,
658    request: Request,
659) -> Result<Json<HistoryResponse>, ApiError> {
660    let owner = PersonalOwner::from_request(&state, &request)?;
661    empty_body(&body(request).await?)?;
662    let (offset, limit) = page(parsed_query(input)?)?;
663    spawn_personal(state.inspection_slots.clone(), move || {
664        let (_guard, catalog, mut data) = locked_personal(&state)?;
665        owner.recheck(&state)?;
666        if crate::parental::supervised(&state) {
667            data.plays
668                .retain(|play| play.track.resolve(&catalog).is_some());
669        }
670        owner.admit(
671            &state,
672            Json(history_page(&data, &catalog, &owner.id, offset, limit)),
673        )
674    })?
675    .await
676    .map_err(|_| {
677        error(
678            StatusCode::INTERNAL_SERVER_ERROR,
679            "personal_failed",
680            "personal history could not finish",
681        )
682    })?
683}

685async fn record_history(
686    State(state): State<ApiState>,
687    request: Request,
688) -> Result<(StatusCode, Json<PlayView>), ApiError> {
689    let owner = PersonalOwner::from_request(&state, &request)?;
690    owner.require_mutation()?;
691    no_query(&request)?;
692    let input: PlayRequest = json_body(request).await?;
693    let now = clock::now_seconds();
694    let track = input.validate(now)?;
695    let data_dir = state.data_dir.clone();
696    spawn_personal(state.inspection_slots.clone(), move || {
697        let at = input.at.unwrap_or(now);
698        locked_update(&data_dir, &state, &owner, |catalog, data| {
699            let track = canonical_target(catalog, &track)?;
700            let play = Play {
701                owner: owner.id.clone(),
702                track: track.clone(),
703                at,
704                ms_played: input.ms_played,
705                completed: input.completed,
706            };
707            data.record_play(play.clone());
708            Ok(play_view(data, &play))
709        })
710        .map(|view| (StatusCode::CREATED, Json(view)))
711    })?
712    .await
713    .map_err(|_| {
714        error(
715            StatusCode::INTERNAL_SERVER_ERROR,
716            "personal_failed",
717            "personal history update could not finish",
718        )
719    })?
720}

809async fn save_collection(
810    State(state): State<ApiState>,
811    input: Result<Query<NameQuery>, QueryRejection>,
812    request: Request,
813) -> Result<Json<CollectionView>, ApiError> {
814    let owner = PersonalOwner::from_request(&state, &request)?;
815    owner.require_mutation()?;
816    let query = parsed_query(input)?;
817    validate_collection_name(&query.name)?;
818    let input: CollectionRequest = json_body(request).await?;
819    if input.expression.trim().is_empty() || input.expression.len() > 8192 {
820        return Err(error(
821            StatusCode::BAD_REQUEST,
822            "invalid_parameters",
823            "expression must contain 1 to 8192 bytes",
824        ));
825    }
826    aede_core::query::parse(&input.expression).map_err(|failure| {
827        error(
828            StatusCode::BAD_REQUEST,
829            "invalid_parameters",
830            failure.to_string(),
831        )
832    })?;
833    let data_dir = state.data_dir.clone();
834    spawn_personal(state.inspection_slots.clone(), move || {
835        let now = clock::now_seconds();
836        locked_update(&data_dir, &state, &owner, |_, data| {
837            if input.create_only && data.collection(&owner.id, &query.name).is_some() {
838                return Err(error(
839                    StatusCode::CONFLICT,
840                    "collection_exists",
841                    "a collection already has this name",
842                ));
843            }
844            data.save_collection(&owner.id, &query.name, &input.expression, now);
845            data.collection(&owner.id, &query.name)
846                .map(collection_view)
847                .ok_or_else(|| {
848                    error(
849                        StatusCode::INTERNAL_SERVER_ERROR,
850                        "personal_failed",
851                        "collection was not saved",
852                    )
853                })
854        })
855        .map(Json)
856    })?
857    .await
858    .map_err(|_| {
859        error(
860            StatusCode::INTERNAL_SERVER_ERROR,
861            "personal_failed",
862            "collection update could not finish",
863        )
864    })?
865}

901async fn collection_albums(
902    State(state): State<ApiState>,
903    input: Result<Query<NameQuery>, QueryRejection>,
904    request: Request,
905) -> Result<Json<serde_json::Value>, ApiError> {
906    let owner = PersonalOwner::from_request(&state, &request)?;
907    empty_body(&body(request).await?)?;
908    let query = parsed_query(input)?;
909    validate_collection_name(&query.name)?;
910    let data_dir = state.data_dir.clone();
911    spawn_personal(state.inspection_slots.clone(), move || {
912        let (_guard, catalog, data) = locked_personal(&state)?;
913        owner.recheck(&state)?;
914        let saved = data.collection(&owner.id, &query.name).ok_or_else(|| {
915            error(
916                StatusCode::NOT_FOUND,
917                "collection_not_found",
918                "collection is unavailable",
919            )
920        })?;
921        let parsed = aede_core::query::parse(&saved.expression)
922            .map_err(|failure| invalid_query(failure.to_string()))?;
923        let sources = inspection::held_sources(&data_dir)?;
924        let context =
925            aede_core::query::Context::new(&catalog, &data, &owner.id).with_sources(&sources);
926        let releases: BTreeSet<_> = aede_core::query::run(&parsed, &context)
927            .into_iter()
928            .filter_map(|id| catalog.track(id).and_then(|t| t.release_id))
929            .collect();
930        let items = releases
931            .into_iter()
932            .map(|id| {
933                EntityRef::of(&catalog, EntityKind::Release, id)
934                    .map(|r| r.to_token())
935                    .ok_or_else(unavailable)
936            })
937            .collect::<Result<Vec<_>, _>>()?;
938        owner.admit(
939            &state,
940            Json(serde_json::json!({"items":items,"scanned_at":catalog.scanned_at})),
941        )
942    })?
943    .await
944    .map_err(|_| unavailable())?

950    page: Page<CollectionView>,

953async fn collections(
954    State(state): State<ApiState>,
955    input: Result<Query<PageQuery>, QueryRejection>,
956    request: Request,
957) -> Result<Json<CollectionsResponse>, ApiError> {
958    let owner = PersonalOwner::from_request(&state, &request)?;
959    empty_body(&body(request).await?)?;
960    let (offset, limit) = page(parsed_query(input)?)?;
961    let data_dir = state.data_dir.clone();
962    spawn_personal(state.inspection_slots.clone(), move || {
963        let (_guard, catalog, data) = locked_personal(&state)?;
964        owner.recheck(&state)?;
965        let mut items: Vec<_> = data
966            .collections
967            .iter()
968            .filter(|collection| collection.owner == owner.id)
969            .map(collection_view)
970            .collect();
971        items.sort_by(|left, right| {
972            text::normalize(&left.name)
973                .cmp(&text::normalize(&right.name))
974                .then_with(|| left.name.cmp(&right.name))
975        });
976        let total = items.len();
977        let sources = inspection::held_sources(&data_dir)?;
978        let context =
979            aede_core::query::Context::new(&catalog, &data, &owner.id).with_sources(&sources);
980        let items = items
981            .into_iter()
982            .skip(offset)
983            .take(limit)
984            .map(|mut row| {
985                row.duration_ms = collection_duration::duration(&context, &row.expression);
986                row
987            })
988            .collect();
989        owner.admit(
990            &state,
991            Json(CollectionsResponse {
992                page: Page {
993                    total,
994                    items,
995                    offset,
996                    limit,
997                    scanned_at: catalog.scanned_at,
998                },
999            }),
1000        )
1001    })?
1002    .await
1003    .map_err(|_| {
1004        error(
1005            StatusCode::INTERNAL_SERVER_ERROR,
1006            "personal_failed",
1007            "collections read could not finish",
1008        )
1009    })?
1010}

1052fn validate_tags(tags: &[String]) -> Result<(), ApiError> {
1053    if tags.len() > MAX_TAGS
1054        || tags
1055            .iter()
1056            .any(|tag| tag.trim().is_empty() || tag.len() > MAX_TAG_BYTES || tag.contains('\0'))
1057    {
1058        return Err(error(
1059            StatusCode::BAD_REQUEST,
1060            "invalid_parameters",
1061            "tags must contain at most 100 nonempty values of at most 256 bytes without NUL",
1062        ));
1063    }
1064    let unique: BTreeSet<_> = tags.iter().map(|tag| tag.as_str()).collect();
1065    if unique.len() != tags.len() {
1066        return Err(error(
1067            StatusCode::BAD_REQUEST,
1068            "invalid_parameters",
1069            "tags must not contain duplicates",
1070        ));
1071    }
1072    Ok(())
1073}

1077        return Err(invalid_query("name must contain 1 to 256 bytes"));
1078    }
1079    Ok(())
1080}
