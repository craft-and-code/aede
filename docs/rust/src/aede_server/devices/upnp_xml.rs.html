1//! Bounded XML subset used for device descriptions and SOAP, without DTDs.
2
3use std::collections::BTreeMap;
4
5const MAX_DEPTH: usize = 32;
6const MAX_NODES: usize = 8192;
7const MAX_BYTES: usize = 512 * 1024;
8
9#[derive(Debug)]
10pub(super) struct Node {
11    pub name: String,
12    pub namespace: String,
13    pub text: String,
14    pub children: Vec<Node>,
15}
16
17impl Node {
18    pub fn child(&self, name: &str) -> Result<&Self, String> {
19        self.optional(name)?
20            .ok_or_else(|| format!("missing XML field {name}"))
21    }
22
23    pub fn optional(&self, name: &str) -> Result<Option<&Self>, String> {
24        let mut found = self.children.iter().filter(|node| node.name == name);
25        let result = found.next();
26        if found.next().is_some() {
27            return Err(format!("duplicate XML field {name}"));
28        }
29        Ok(result)
30    }
31
32    pub fn value(&self, name: &str) -> Result<&str, String> {
33        let node = self.child(name)?;
34        if !node.children.is_empty() {
35            return Err(format!("nested XML field {name}"));
36        }
37        Ok(node.text.trim())
38    }
39}
40
41pub(super) fn escape(value: &str) -> Result<String, String> {
42    let mut escaped = String::with_capacity(value.len());
43    for character in value.chars() {
44        if !xml_character(character) {
45            return Err("invalid character in XML value".into());
46        }
47        escaped.push_str(match character {
48            '&' => "&amp;",
49            '<' => "&lt;",
50            '>' => "&gt;",
51            '"' => "&quot;",
52            '\'' => "&apos;",
53            _ => {
54                escaped.push(character);
55                continue;
56            }
57        });
58    }
59    Ok(escaped)
60}
61
62fn xml_character(value: char) -> bool {
63    matches!(value, '\t' | '\n' | '\r' | '\u{20}'..='\u{d7ff}' | '\u{e000}'..='\u{fffd}' | '\u{10000}'..='\u{10ffff}')
64}
65
66fn decode(value: &str) -> Result<String, String> {
67    let mut result = String::with_capacity(value.len());
68    let mut remainder = value;
69    while let Some(offset) = remainder.find('&') {
70        result.push_str(&remainder[..offset]);
71        remainder = &remainder[offset + 1..];
72        let end = remainder.find(';').ok_or("unterminated XML entity")?;
73        let entity = &remainder[..end];
74        let decoded = match entity {
75            "amp" => '&',
76            "lt" => '<',
77            "gt" => '>',
78            "quot" => '"',
79            "apos" => '\'',
80            _ => {
81                let number = if let Some(number) = entity.strip_prefix("#x") {
82                    u32::from_str_radix(number, 16).ok()
83                } else if let Some(number) = entity.strip_prefix('#') {
84                    number.parse::<u32>().ok()
85                } else {
86                    None
87                };
88                number
89                    .and_then(char::from_u32)
90                    .ok_or("unsupported XML entity")?
91            }
92        };
93        if !xml_character(decoded) {
94            return Err("invalid XML character entity".into());
95        }
96        result.push(decoded);
97        remainder = &remainder[end + 1..];
98    }
99    result.push_str(remainder);
100    Ok(result)
101}

103pub(super) fn parse(value: &str) -> Result<Node, String> {
104    if value.len() > MAX_BYTES || value.chars().any(|value| !xml_character(value)) {
105        return Err("device XML exceeds its limit or contains invalid characters".into());
106    }
107    let mut parser = Parser {
108        source: value.strip_prefix('\u{feff}').unwrap_or(value),
109        position: 0,
110        nodes: 0,
111    };
112    parser.whitespace();
113    if parser.rest().starts_with("<?xml ") {
114        let end = parser
115            .rest()
116            .find("?>")
117            .ok_or("unterminated XML declaration")?;
118        if parser.rest()[..end].contains("<!") {
119            return Err("invalid XML declaration".into());
120        }
121        parser.position += end + 2;
122    }
123    parser.misc()?;
124    let mut namespaces = BTreeMap::new();
125    namespaces.insert(
126        "xml".to_string(),
127        "http://www.w3.org/XML/1998/namespace".to_string(),
128    );
129    let root = parser.element(0, &namespaces)?;
130    parser.misc()?;
131    if !parser.rest().is_empty() {
132        return Err("trailing device XML".into());
133    }
134    Ok(root)
135}
136
137struct Parser<'a> {
138    source: &'a str,
139    position: usize,
140    nodes: usize,
141}
142
143impl Parser<'_> {
144    fn rest(&self) -> &str {
145        &self.source[self.position..]
146    }
147
148    fn whitespace(&mut self) {
149        self.position += self
150            .rest()
151            .bytes()
152            .take_while(|b| matches!(b, b' ' | b'\n' | b'\r' | b'\t'))
153            .count();
154    }
155
156    fn misc(&mut self) -> Result<(), String> {
157        loop {
158            self.whitespace();
159            if !self.rest().starts_with("<!--") {
160                return Ok(());
161            }
162            self.comment()?;
163        }
164    }
165
166    fn comment(&mut self) -> Result<(), String> {
167        let end = self.rest().find("-->").ok_or("unterminated XML comment")?;
168        if end < 4 || self.rest()[4..end].contains("--") {
169            return Err("invalid XML comment".into());
170        }
171        self.position += end + 3;
172        Ok(())
173    }
174
175    fn name(&mut self) -> Result<String, String> {
176        let count = self
177            .rest()
178            .bytes()
179            .take_while(|b| b.is_ascii_alphanumeric() || matches!(b, b'_' | b':' | b'-' | b'.'))
180            .count();
181        if count > 128 {
182            return Err("XML name exceeds its limit".into());
183        }
184        let value = self.rest()[..count].to_owned();
185        if value.is_empty()
186            || !value.as_bytes()[0].is_ascii_alphabetic() && value.as_bytes()[0] != b'_'
187            || value.matches(':').count() > 1
188        {
189            return Err("unsupported XML name".into());
190        }
191        self.position += count;
192        Ok(value)
193    }
194
195    fn element(
196        &mut self,
197        depth: usize,
198        inherited: &BTreeMap<String, String>,
199    ) -> Result<Node, String> {
200        if depth >= MAX_DEPTH || self.nodes >= MAX_NODES {
201            return Err("device XML nesting/node limit exceeded".into());
202        }
203        self.nodes += 1;
204        if !self.rest().starts_with('<') {
205            return Err("expected XML element".into());
206        }
207        self.position += 1;
208        let qualified = self.name()?;
209        let mut namespaces = inherited.clone();
210        let mut attributes = BTreeMap::new();
211        let self_closed;
212        loop {
213            let before = self.position;
214            self.whitespace();
215            if self.rest().starts_with("/>") {
216                self.position += 2;
217                self_closed = true;
218                break;
219            }
220            if self.rest().starts_with('>') {
221                self.position += 1;
222                self_closed = false;
223                break;
224            }
225            if before == self.position {
226                return Err("missing XML attribute whitespace".into());
227            }
228            let key = self.name()?;
229            if attributes.len() == 64 {
230                return Err("too many XML attributes".into());
231            }
232            self.whitespace();
233            if !self.rest().starts_with('=') {
234                return Err("invalid XML attribute".into());
235            }
236            self.position += 1;
237            self.whitespace();
238            let quote = self
239                .rest()
240                .chars()
241                .next()
242                .ok_or("truncated XML attribute")?;
243            if quote != '\'' && quote != '"' {
244                return Err("unquoted XML attribute".into());
245            }
246            self.position += 1;
247            let end = self.rest().find(quote).ok_or("truncated XML attribute")?;
248            let raw = &self.rest()[..end];
249            if raw.len() > 4096 || raw.contains('<') {
250                return Err("invalid XML attribute value".into());
251            }
252            let value = decode(raw)?;
253            self.position += end + 1;
254            if attributes.insert(key.clone(), value.clone()).is_some() {
255                return Err("duplicate XML attribute".into());
256            }
257            if key == "xmlns" {
258                if value.len() > 256 {
259                    return Err("XML namespace exceeds its limit".into());
260                }
261                namespaces.insert(String::new(), value);
262            } else if let Some(prefix) = key.strip_prefix("xmlns:") {
263                if value.len() > 256 {
264                    return Err("XML namespace exceeds its limit".into());
265                }
266                if prefix == "xmlns"
267                    || prefix == "xml" && value != "http://www.w3.org/XML/1998/namespace"
268                {
269                    return Err("reserved XML namespace declaration".into());
270                }
271                namespaces.insert(prefix.to_owned(), value);
272            }
273            if namespaces.len() > 32 {
274                return Err("too many XML namespaces".into());
275            }
276        }
277        let (prefix, name) = qualified.split_once(':').unwrap_or(("", &qualified));
278        if name.is_empty()
279            || !name.as_bytes()[0].is_ascii_alphabetic() && name.as_bytes()[0] != b'_'
280        {
281            return Err("invalid XML qualified name".into());
282        }
283        let namespace = namespaces.get(prefix).cloned().unwrap_or_default();
284        if !prefix.is_empty() && namespace.is_empty() {
285            return Err("undeclared XML namespace".into());
286        }
287        for key in attributes.keys() {
288            if let Some((prefix, _)) = key.split_once(':')
289                && prefix != "xmlns"
290                && !namespaces.contains_key(prefix)
291            {
292                return Err("undeclared XML attribute namespace".into());
293            }
294        }
295        let mut node = Node {
296            name: name.into(),
297            namespace,
298            text: String::new(),
299            children: Vec::new(),
300        };
301        if self_closed {
302            return Ok(node);
303        }
304        loop {
305            if self.rest().starts_with("</") {
306                self.position += 2;
307                let close = self.name()?;
308                self.whitespace();
309                if close != qualified || !self.rest().starts_with('>') {
310                    return Err("mismatched XML closing element".into());
311                }
312                self.position += 1;
313                return Ok(node);
314            }
315            if self.rest().starts_with("<!--") {
316                self.comment()?;
317            } else if self.rest().starts_with("<![CDATA[") {
318                self.position += 9;
319                let end = self.rest().find("]]>").ok_or("unterminated XML CDATA")?;
320                node.text.push_str(&self.rest()[..end]);
321                self.position += end + 3;
322            } else if self.rest().starts_with("<!") || self.rest().starts_with("<?") {
323                return Err("DTDs, entities and processing instructions are forbidden".into());
324            } else if self.rest().starts_with('<') {
325                node.children.push(self.element(depth + 1, &namespaces)?);
326            } else {
327                let end = self.rest().find('<').ok_or("truncated XML content")?;
328                let raw = &self.rest()[..end];
329                if raw.contains("]]>") {
330                    return Err("invalid XML text".into());
331                }
332                node.text.push_str(&decode(raw)?);
333                self.position += end;
334            }
335        }
336    }
337}
