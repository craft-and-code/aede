1//! Explicit, finite LAN playback on SlimProto, UPnP AV and OpenHome devices.
2//!
3//! This controller is separate from the authenticated catalog API. A local
4//! caller selects up to 64 original files and exactly one private IPv4 device.
5//! The temporary media listener exposes only that selection, with a random
6//! session capability and a matching peer address. No transfer creates a listen
7//! or applies server DSP. Physical device compatibility needs real acceptance.
8

21/// The device's control language, independent of the audio file format.
22#[derive(Clone, Copy, Debug, Eq, PartialEq)]
23pub enum DeviceProtocol {
24    /// A Squeezebox-compatible player explicitly connects to Aède's TCP port.
25    Slimproto,
26    /// Aède controls an advertised UPnP AVTransport renderer.
27    Upnp,
28    /// Aède supplies a finite queue to an OpenHome Playlist renderer.
29    Openhome,
30}

41/// Explicit LAN addresses and queue replacement policy for [`cast`].
42#[derive(Clone, Debug)]
43pub struct CastOptions {
44    /// A specific local private or loopback IPv4 address, never `0.0.0.0`.
45    pub bind: Ipv4Addr,
46    /// The chosen transport.
47    pub protocol: DeviceProtocol,
48    /// SlimProto peer IP, or an HTTP device-description URL for UPnP/OpenHome.
49    pub device: String,
50    /// SlimProto listener port; use 3483 normally. Other protocols require None.
51    pub port: Option<u16>,
52    /// Explicit permission to replace an existing OpenHome device playlist.
53    pub replace: bool,
54    /// Optional explicit SlimProto linear digital gain percentage, 0 to 100.
55    /// None preserves device gain; fresh software players may be muted.
56    pub volume: Option<u8>,
57}

59pub(super) fn validate_ip(ip: Ipv4Addr) -> Result<(), String> {
60    if ip.is_unspecified()
61        || ip.is_broadcast()
62        || !(ip.is_private() || ip.is_loopback() || ip.is_link_local())
63    {
64        return Err(
65            "device playback requires a specific private, link-local or loopback IPv4 address"
66                .into(),
67        );
68    }
69    Ok(())
70}
71
72impl CastOptions {
73    /// Validate every option before opening files or network listeners.
74    pub fn validate(&self) -> Result<Ipv4Addr, String> {
75        validate_ip(self.bind)?;
76        if self.volume.is_some_and(|volume| volume > 100) {
77            return Err("--device-volume must be between 0 and 100".into());
78        }
79        match self.protocol {
80            DeviceProtocol::Slimproto => {
81                if self.replace {
82                    return Err("--replace is only supported by OpenHome".into());
83                }
84                if self.port == Some(0) {
85                    return Err("the SlimProto listener needs a non-zero port".into());
86                }
87                let peer = self
88                    .device
89                    .parse::<Ipv4Addr>()
90                    .map_err(|_| "SlimProto --device must be a literal IPv4 address")?;
91                validate_ip(peer)?;
92                Ok(peer)
93            }
94            DeviceProtocol::Upnp | DeviceProtocol::Openhome => {
95                if self.volume.is_some() {
96                    return Err("--device-volume is only supported by SlimProto".into());
97                }
98                if self.port.is_some() {
99                    return Err("--port is only supported by SlimProto".into());
100                }
101                if self.replace && self.protocol != DeviceProtocol::Openhome {
102                    return Err("--replace is only supported by OpenHome".into());
103                }
104                let peer = upnp::device_ip(&self.device)?;
105                validate_ip(peer)?;
106                Ok(peer)
107            }
108        }
109    }
110}

112fn runtime() -> Result<tokio::runtime::Runtime, String> {
113    tokio::runtime::Builder::new_multi_thread()
114        .worker_threads(2)
115        .enable_all()
116        .build()
117        .map_err(|error| format!("cannot start the device controller: {error}"))
118}
119
120async fn finish_controller(
121    result: Result<(), String>,
122    signal_task: tokio::task::JoinHandle<Result<(), String>>,
123) -> Result<(), String> {
124    signal_task.abort();
125    let signal_result = match signal_task.await {
126        Ok(result) => result,
127        Err(error) if error.is_cancelled() => Ok(()),
128        Err(error) => Err(format!("Ctrl-C monitor failed: {error}")),
129    };
130    match (result, signal_result) {
131        (Err(transport), Err(signal)) => Err(format!("{transport}; {signal}")),
132        (Err(error), _) | (_, Err(error)) => Err(error),
133        (Ok(()), Ok(())) => Ok(()),
134    }
135}

137/// Perform one bounded SSDP discovery on an explicitly selected interface.
138///
139/// Network activity occurs only during this call. Devices outside the LAN IP
140/// policy, cross-host descriptions and unsupported services are not followed.
141pub fn discover(bind: Ipv4Addr) -> Result<Vec<DeviceDescription>, String> {
142    validate_ip(bind)?;
143    runtime()?.block_on(upnp::discover(bind))
144}

146/// Play a finite selection on exactly one LAN device until completion or Ctrl-C.
147///
148/// Files are opened read-only and sent unchanged, without normalization or
149/// transcoding. Each adapter preflights advertised format support. This method
150/// does not create listening history: transport/device status is not a native
151/// PCM consumption acknowledgement. A request to stop is sent on Ctrl-C;
152/// a powered-off or disconnected device cannot acknowledge that request.
153pub fn cast(options: CastOptions, paths: Vec<PathBuf>) -> Result<(), String> {
154    let peer = options.validate()?;
155    let sources = media::prepare(paths)?;
156    let runtime = runtime()?;
157    let result = runtime.block_on(async move {
158        let (shutdown, signal) = watch::channel(false);
159        let signal_task = tokio::spawn(async move {
160            let result = tokio::signal::ctrl_c()
161                .await
162                .map_err(|error| format!("cannot monitor Ctrl-C: {error}"));
163            let _ = shutdown.send(true);
164            result
165        });
166        let tracks = sources
167            .iter()
168            .map(|source| source.track.clone())
169            .collect::<Vec<_>>();
170        let media = media::MediaServer::start(options.bind, peer, sources).await?;
171        println!(
172            "Device media: {} (selected originals only; Ctrl-C stops playback)",
173            media.address()
174        );
175        let result = match options.protocol {
176            DeviceProtocol::Slimproto => {
177                match tokio::net::TcpListener::bind(SocketAddrV4::new(
178                    options.bind,
179                    options.port.unwrap_or(3483),
180                ))
181                .await
182                {
183                    Ok(listener) => {
184                        println!(
185                            "SlimProto: {} — connect the selected player to this address",
186                            listener.local_addr().map_err(|error| error.to_string())?
187                        );
188                        slimproto::run(
189                            listener,
190                            peer,
191                            &tracks,
192                            media.urls(),
193                            options.volume,
194                            signal,
195                        )
196                        .await
197                    }
198                    Err(error) => Err(format!("cannot open the SlimProto listener: {error}")),
199                }
200            }
201            protocol => {
202                upnp::run(
203                    protocol,
204                    &options.device,
205                    &tracks,
206                    media.urls(),
207                    options.replace,
208                    signal,
209                )
210                .await
211            }
212        };
213        let result = finish_controller(result, signal_task).await;
214        media.stop().await;
215        result
216    });
217    // A stuck OS/filesystem operation must not prevent explicit shutdown.
218    runtime.shutdown_timeout(std::time::Duration::from_secs(2));
219    result
220}
