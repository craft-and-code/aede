1//! Explicit, single-player SlimProto control for original local audio.
2//!
3//! Packet layouts and event meanings follow the Squeezelite and Lyrion
4//! implementations. Download EOF and decoder EOF are not audible completion:
5//! this first profile waits for decoded EOF followed by a drained output.
6

17const MAX_PAYLOAD: usize = 4096;
18const MAX_QUEUE: usize = 64;
19const MAX_REFUSED_PEERS: usize = 16;
20const MAX_PACKET_BURST: usize = 256;
21const STRM_BYTES: usize = 24;
22const STAT_BYTES: usize = 53;

24#[derive(Clone, Copy)]
25struct Limits {
26    connection: Duration,
27    hello: Duration,
28    packet: Duration,
29    write: Duration,
30    heartbeat: Duration,
31    completion_grace: Duration,
32}

34impl Default for Limits {
35    fn default() -> Self {
36        Self {
37            connection: Duration::from_secs(60),
38            hello: Duration::from_secs(5),
39            packet: Duration::from_secs(30),
40            write: Duration::from_secs(2),
41            heartbeat: Duration::from_secs(5),
42            completion_grace: Duration::from_secs(120),
43        }
44    }
45}

52/// Runs one finite original-file queue for the explicitly selected IPv4 player.
53/// Cancellation is cooperative so the player receives a bounded stop request.
54/// No listening history is inferred from these unauthenticated device reports.
55pub(super) async fn run(
56    listener: TcpListener,
57    expected_peer: Ipv4Addr,
58    tracks: &[DeviceTrack],
59    urls: &[String],
60    volume: Option<u8>,
61    shutdown: watch::Receiver<bool>,
62) -> Result<(), String> {
63    run_with_limits(
64        listener,
65        expected_peer,
66        tracks,
67        urls,
68        shutdown,
69        SessionOptions {
70            volume,
71            limits: Limits::default(),
72        },
73    )
74    .await
75}

77async fn run_with_limits(
78    listener: TcpListener,
79    expected_peer: Ipv4Addr,
80    tracks: &[DeviceTrack],
81    urls: &[String],
82    mut shutdown: watch::Receiver<bool>,
83    options: SessionOptions,
84) -> Result<(), String> {
85    let SessionOptions { volume, limits } = options;
86    if tracks.is_empty() || tracks.len() > MAX_QUEUE || tracks.len() != urls.len() {
87        return Err("SlimProto requires 1 to 64 tracks and one media URL per occurrence".into());
88    }
89    if volume.is_some_and(|volume| volume > 100) {
90        return Err("SlimProto device volume must be between 0 and 100 percent".into());
91    }
92    let Some(mut stream) = accept_player(&listener, expected_peer, &mut shutdown, limits).await?
93    else {
94        return Ok(());
95    };
96    stream
97        .set_nodelay(true)
98        .map_err(|error| format!("SlimProto TCP: {error}"))?;
99
100    let hello = tokio::select! {
101        result = timeout(limits.hello, read_packet(&mut stream)) => {
102            result.map_err(|_| "SlimProto HELO timed out".to_string()).and_then(|packet| packet)
103        }
104        () = cancelled(&mut shutdown) => {
105            stop(&mut stream, limits).await?;
106            return Ok(());
107        },
108    };
109    let (mut reader, mut writer) = stream.into_split();
110    let prepared = hello.and_then(|packet| {
111        let capabilities = Capabilities::parse(&packet)?;
112        tracks
113            .iter()
114            .zip(urls)
115            .map(|(track, url)| start_payload(track, url, &capabilities))
116            .collect::<Result<Vec<_>, _>>()
117    });
118    let prepared = match prepared {
119        Ok(prepared) => prepared,
120        Err(error) => {
121            if let Err(stop_error) = stop(&mut writer, limits).await {
122                return Err(format!("{error}; stop request also failed: {stop_error}"));
123            }
124            return Err(error);
125        }
126    };
127
128    // A single reader preserves fragmented frames while the controller sends
129    // heartbeats or handles cancellation. Cancelling read_exact would discard
130    // its partial prefix and desynchronise the next frame.
131    let (sender, mut packets) = mpsc::channel(8);
132    let reader_task = tokio::spawn(async move {
133        loop {
134            let packet = timeout(limits.packet, read_packet(&mut reader))
135                .await
136                .map_err(|_| "SlimProto player stopped responding".to_string())
137                .and_then(|packet| packet);
138            let failed = packet.is_err();
139            if sender.send(packet).await.is_err() || failed {
140                break;
141            }
142        }
143    });
144    let result = play_queue(
145        &mut writer,
146        &mut packets,
147        tracks,
148        &prepared,
149        volume,
150        &mut shutdown,
151        limits,
152    )
153    .await;
154    let stopped = stop(&mut writer, limits).await;
155    reader_task.abort();
156    // Preserve the original transport/decoder diagnosis if stopping also fails.
157    match (result, stopped) {
158        (Err(error), Err(stop_error)) => {
159            Err(format!("{error}; stop request also failed: {stop_error}"))
160        }
161        (Err(error), Ok(())) => Err(error),
162        (Ok(()), stopped) => stopped,
163    }
164}

166async fn accept_player(
167    listener: &TcpListener,
168    expected: Ipv4Addr,
169    shutdown: &mut watch::Receiver<bool>,
170    limits: Limits,
171) -> Result<Option<TcpStream>, String> {
172    let deadline = Instant::now() + limits.connection;
173    for _ in 0..MAX_REFUSED_PEERS {
174        let accepted = tokio::select! {
175            result = timeout_at(deadline, listener.accept()) => result
176                .map_err(|_| "SlimProto player connection timed out".to_string())?
177                .map_err(|error| format!("SlimProto accept: {error}"))?,
178            () = cancelled(shutdown) => return Ok(None),
179        };
180        if accepted.1.ip() == IpAddr::V4(expected) {
181            return Ok(Some(accepted.0));
182        }
183        // Other hosts cannot claim this explicitly selected receiver session.
184        drop(accepted.0);
185    }
186    Err("SlimProto refused too many connections from another address".into())
187}

189async fn cancelled(shutdown: &mut watch::Receiver<bool>) {
190    loop {
191        if *shutdown.borrow_and_update() || shutdown.changed().await.is_err() {
192            return;
193        }
194    }
195}

197#[derive(Debug)]
198struct Packet {
199    opcode: [u8; 4],
200    payload: Vec<u8>,
201}
202
203async fn read_packet(reader: &mut (impl AsyncRead + Unpin)) -> Result<Packet, String> {
204    let mut header = [0; 8];
205    reader
206        .read_exact(&mut header)
207        .await
208        .map_err(|error| format!("SlimProto disconnected: {error}"))?;
209    let length = u32::from_be_bytes([header[4], header[5], header[6], header[7]]) as usize;
210    if length > MAX_PAYLOAD {
211        return Err("SlimProto player packet exceeds 4096 bytes".into());
212    }
213    let mut payload = vec![0; length];
214    reader
215        .read_exact(&mut payload)
216        .await
217        .map_err(|error| format!("SlimProto truncated packet: {error}"))?;
218    Ok(Packet {
219        opcode: [header[0], header[1], header[2], header[3]],
220        payload,
221    })
222}

224struct Capabilities {
225    flac: bool,
226    mp3: bool,
227    wav: bool,
228    maximum_rate: u32,
229}
230
231impl Capabilities {
232    fn parse(packet: &Packet) -> Result<Self, String> {
233        if packet.opcode != *b"HELO" || packet.payload.len() < 36 {
234            return Err("SlimProto requires a modern HELO with advertised codecs".into());
235        }
236        let text = std::str::from_utf8(&packet.payload[36..])
237            .map_err(|_| "SlimProto capabilities are not UTF-8".to_string())?;
238        if text.bytes().any(|byte| !(b' '..=b'~').contains(&byte)) {
239            return Err("SlimProto capabilities contain control or non-ASCII bytes".into());
240        }
241        let mut capabilities = Self {
242            flac: false,
243            mp3: false,
244            wav: false,
245            maximum_rate: 48_000,
246        };
247        let mut has_rate = false;
248        for item in text.split(',') {
249            match item {
250                "flc" => capabilities.flac = true,
251                "mp3" => capabilities.mp3 = true,
252                "wav" => capabilities.wav = true,
253                _ => {
254                    if let Some(rate) = item.strip_prefix("MaxSampleRate=") {
255                        if has_rate {
256                            return Err(
257                                "SlimProto advertises duplicate maximum sample rates".into()
258                            );
259                        }
260                        capabilities.maximum_rate = rate
261                            .parse::<u32>()
262                            .ok()
263                            .filter(|rate| (8_000..=1_536_000).contains(rate))
264                            .ok_or_else(|| {
265                                "SlimProto maximum sample rate is invalid".to_string()
266                            })?;
267                        has_rate = true;
268                    }
269                }
270            }
271        }
272        Ok(capabilities)
273    }
274}

276fn start_payload(
277    track: &DeviceTrack,
278    url: &str,
279    capabilities: &Capabilities,
280) -> Result<Vec<u8>, String> {
281    let properties = &track.properties;
282    let rate = properties
283        .sample_rate
284        .filter(|rate| *rate > 0)
285        .ok_or_else(|| "SlimProto needs a known source sample rate".to_string())?;
286    if rate > capabilities.maximum_rate {
287        return Err("SlimProto source sample rate exceeds the player's advertised limit".into());
288    }
289    if !matches!(properties.channels, Some(1 | 2)) {
290        return Err("SlimProto's initial profile supports mono or stereo sources".into());
291    }
292    if properties.duration_ms.is_none_or(|duration| duration == 0) {
293        return Err("SlimProto needs a positive known track duration".into());
294    }
295    let mut payload = control_payload(b's');
296    payload[1] = b'1';
297    payload[7] = 1; // Small sources must not wait for a large network threshold.
298    let supported = match (
299        properties.codec.as_str(),
300        properties.container.as_str(),
301        track.mime,
302    ) {
303        ("flac", "flac", "audio/flac") if capabilities.flac => {
304            if !matches!(properties.bit_depth, Some(4..=24)) {
305                return Err("SlimProto's initial FLAC profile requires a known precision of at most 24 bits".into());
306            }
307            payload[2] = b'f';
308            true
309        }
310        ("mp3", "mp3", "audio/mpeg") if capabilities.mp3 => {
311            payload[2] = b'm';
312            true
313        }
314        ("pcm", "wav", "audio/wav") if capabilities.wav => {
315            payload[2] = b'p';
316            payload[3] = match properties.bit_depth {
317                Some(8) => b'0',
318                Some(16) => b'1',
319                Some(24) => b'2',
320                Some(32) => b'3',
321                _ => return Err("SlimProto WAV requires 8, 16, 24 or 32-bit integer PCM".into()),
322            };
323            let rates = [
324                11_025, 22_050, 32_000, 44_100, 48_000, 8_000, 12_000, 16_000, 24_000, 96_000,
325                88_200, 176_400, 192_000,
326            ];
327            let index = rates
328                .iter()
329                .position(|candidate| *candidate == rate)
330                .ok_or_else(|| {
331                    "SlimProto WAV sample rate has no supported protocol code".to_string()
332                })?;
333            payload[4] = b'0' + index as u8;
334            payload[5] = if properties.channels == Some(1) {
335                b'1'
336            } else {
337                b'2'
338            };
339            payload[6] = b'1';
340            true
341        }
342        _ => false,
343    };
344    if !supported {
345        return Err(
346            "SlimProto source codec/container is unsupported or not advertised by this player"
347                .into(),
348        );
349    }
350    let (address, path) = media_address(url)?;
351    payload[18..20].copy_from_slice(&address.port().to_be_bytes());
352    payload[20..24].copy_from_slice(&address.ip().octets());
353    let request = format!("GET {path} HTTP/1.0\r\nHost: {address}\r\nConnection: close\r\n\r\n");
354    payload.extend_from_slice(request.as_bytes());
355    Ok(payload)
356}

358fn media_address(url: &str) -> Result<(SocketAddrV4, &str), String> {
359    let remainder = url
360        .strip_prefix("http://")
361        .ok_or_else(|| "SlimProto needs an explicit HTTP IPv4 media URL".to_string())?;
362    let (authority, suffix) = remainder
363        .split_once('/')
364        .ok_or_else(|| "SlimProto media URL has no path".to_string())?;
365    let address = authority
366        .parse::<SocketAddrV4>()
367        .map_err(|_| "SlimProto media URL requires an IPv4 address and port".to_string())?;
368    let path = &remainder[authority.len()..];
369    if address.port() == 0
370        || suffix.is_empty()
371        || path.len() > 2048
372        || path
373            .bytes()
374            .any(|byte| !(b'!'..=b'~').contains(&byte) || byte == b'#')
375    {
376        return Err("SlimProto media URL path is invalid or too long".into());
377    }
378    Ok((address, path))
379}

381fn control_payload(command: u8) -> Vec<u8> {
382    let mut payload = vec![0; STRM_BYTES];
383    payload[0] = command;
384    payload[2] = b'm';
385    payload[3..7].fill(b'?');
386    payload[10] = b'0'; // Disable fades; volume and replay gain remain untouched.
387    payload
388}

390fn volume_payload(volume: u8) -> [u8; 18] {
391    let gain = (u32::from(volume) * 65_536 + 50) / 100;
392    let legacy_gain = (u32::from(volume) * 128 + 50) / 100;
393    let mut payload = [0; 18];
394    payload[..4].copy_from_slice(&legacy_gain.to_be_bytes());
395    payload[4..8].copy_from_slice(&legacy_gain.to_be_bytes());
396    payload[8] = 1;
397    payload[10..14].copy_from_slice(&gain.to_be_bytes());
398    payload[14..18].copy_from_slice(&gain.to_be_bytes());
399    payload
400}

402async fn send_frame(
403    writer: &mut (impl AsyncWrite + Unpin),
404    opcode: &[u8; 4],
405    payload: &[u8],
406    limits: Limits,
407) -> Result<(), String> {
408    let length = u16::try_from(payload.len() + 4)
409        .map_err(|_| "SlimProto server packet is too long".to_string())?;
410    let mut frame = Vec::with_capacity(usize::from(length) + 2);
411    frame.extend_from_slice(&length.to_be_bytes());
412    frame.extend_from_slice(opcode);
413    frame.extend_from_slice(payload);
414    timeout(limits.write, writer.write_all(&frame))
415        .await
416        .map_err(|_| "SlimProto write timed out".to_string())?
417        .map_err(|error| format!("SlimProto write: {error}"))
418}

420async fn stop(writer: &mut (impl AsyncWrite + Unpin), limits: Limits) -> Result<(), String> {
421    send_frame(writer, b"strm", &control_payload(b'q'), limits).await
422}

424async fn play_queue(
425    writer: &mut (impl AsyncWrite + Unpin),
426    packets: &mut mpsc::Receiver<Result<Packet, String>>,
427    tracks: &[DeviceTrack],
428    prepared: &[Vec<u8>],
429    volume: Option<u8>,
430    shutdown: &mut watch::Receiver<bool>,
431    limits: Limits,
432) -> Result<(), String> {
433    if *shutdown.borrow_and_update() {
434        return Ok(());
435    }
436    // strm s preserves old output for LMS gapless joins. An explicit new cast
437    // must first discard any samples left by a previous controller session.
438    stop(writer, limits).await?;
439    if let Some(volume) = volume {
440        send_frame(writer, b"audg", &volume_payload(volume), limits).await?;
441    }
442    send_frame(writer, b"aude", &[1, 1], limits).await?;
443    let mut heartbeat = interval(limits.heartbeat);
444    // The first interval tick is immediate; strm s itself starts the heartbeat.
445    heartbeat.tick().await;
446    let mut burst_start = Instant::now();
447    let mut burst = 0;
448    for (track, request) in tracks.iter().zip(prepared) {
449        if *shutdown.borrow_and_update() {
450            return Ok(());
451        }
452        let budget = Duration::from_millis(track.properties.duration_ms.unwrap_or(0))
453            .checked_add(limits.completion_grace)
454            .ok_or_else(|| "SlimProto track duration is too large".to_string())?;
455        let deadline = Instant::now()
456            .checked_add(budget)
457            .ok_or_else(|| "SlimProto track duration is too large".to_string())?;
458        send_frame(writer, b"strm", request, limits).await?;
459        let mut started = false;
460        let mut decoded = false;
461        let mut released = false;
462        loop {
463            let packet = tokio::select! {
464                () = cancelled(shutdown) => return Ok(()),
465                () = tokio::time::sleep_until(deadline) => return Err("SlimProto track did not finish within its playback budget".into()),
466                _ = heartbeat.tick() => {
467                    send_frame(writer, b"strm", &control_payload(b't'), limits).await?;
468                    continue;
469                },
470                packet = packets.recv() => packet.ok_or_else(|| "SlimProto packet reader stopped".to_string())??,
471            };
472            if burst_start.elapsed() >= Duration::from_secs(5) {
473                burst_start = Instant::now();
474                burst = 0;
475            }
476            burst += 1;
477            if burst > MAX_PACKET_BURST {
478                return Err("SlimProto player sent too many control packets".into());
479            }
480            match packet.opcode {
481                opcode if opcode == *b"STAT" => {
482                    if packet.payload.len() != STAT_BYTES {
483                        return Err("SlimProto STAT must contain 53 payload bytes".into());
484                    }
485                    match &packet.payload[..4] {
486                        b"STMs" => started = true,
487                        b"STMd" => {
488                            decoded = true;
489                            // A short source can finish decoding before the
490                            // player's normal output threshold is reached.
491                            if !started && !released {
492                                send_frame(writer, b"strm", &control_payload(b'u'), limits).await?;
493                                released = true;
494                            }
495                        }
496                        b"STMl" if !started && !released => {
497                            send_frame(writer, b"strm", &control_payload(b'u'), limits).await?;
498                            released = true;
499                        }
500                        b"STMu" if started && decoded && packet.payload[33..37] == [0; 4] => break,
501                        b"STMn" => return Err("SlimProto player could not decode the source".into()),
502                        b"STMo" => return Err("SlimProto player ran out of audio while still downloading".into()),
503                        b"STMp" => return Err("SlimProto device-side pause is not supported by this finite-queue profile".into()),
504                        _ => {},
505                    }
506                }
507                opcode if opcode == *b"DSCO" => {
508                    if packet.payload.len() != 1 || packet.payload[0] > 1 {
509                        return Err("SlimProto player could not retrieve the source".into());
510                    }
511                    // Normal download EOF or decoder-triggered disconnect
512                    // leaves decoded samples in the player's output buffer.
513                }
514                opcode if opcode == *b"RESP" => {
515                    let successful = packet.payload.starts_with(b"HTTP/1.0 200 ")
516                        || packet.payload.starts_with(b"HTTP/1.1 200 ");
517                    if !successful {
518                        return Err(
519                            "SlimProto player received an unsuccessful media HTTP response".into(),
520                        );
521                    }
522                }
523                opcode
524                    if opcode == *b"SETD"
525                        || opcode == *b"IR  "
526                        || opcode == *b"BUTN"
527                        || opcode == *b"KNOB"
528                        || opcode == *b"ANIC" => {}
529                _ => return Err("SlimProto player sent an unsupported control packet".into()),
530            }
531        }
532    }
533    Ok(())
534}
