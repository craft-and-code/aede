31fn validate_path(path: &Path) -> Result<Metadata, String> {
32    if !path.is_absolute() || path.components().any(|part| part == Component::ParentDir) {
33        return Err("device media needs a canonical absolute source path".into());
34    }
35    let mut prefix = PathBuf::new();
36    for part in path.components() {
37        prefix.push(part);
38        let metadata = fs::symlink_metadata(&prefix).map_err(|error| error.to_string())?;
39        if metadata.file_type().is_symlink() {
40            return Err("device media source changed to a symbolic link".into());
41        }
42    }
43    let metadata = fs::metadata(path).map_err(|error| error.to_string())?;
44    if !metadata.is_file() || metadata.modified().is_err() {
45        return Err(
46            "device media source must be a regular file with a known modification time".into(),
47        );
48    }
49    Ok(metadata)
50}

52fn same_source(expected: &Metadata, actual: &Metadata) -> bool {
53    let stable = actual.is_file()
54        && expected.len() == actual.len()
55        && matches!((expected.modified(), actual.modified()), (Ok(left), Ok(right)) if left == right);
56    #[cfg(unix)]
57    {
58        use std::os::unix::fs::MetadataExt as _;
59        stable && expected.dev() == actual.dev() && expected.ino() == actual.ino()
60    }
61    #[cfg(not(unix))]
62    {
63        stable
64    }
65}

67fn open_regular(path: &Path) -> Result<File, String> {
68    let mut options = OpenOptions::new();
69    options.read(true);
70    #[cfg(unix)]
71    {
72        use std::os::unix::fs::OpenOptionsExt;
73        options.custom_flags(unix_open_flags()?);
74    }
75    #[cfg(windows)]
76    {
77        use std::os::windows::fs::OpenOptionsExt;
78        // FILE_FLAG_OPEN_REPARSE_POINT: inspect the opened link itself, rather
79        // than following a replacement introduced after path validation.
80        options.custom_flags(0x0020_0000);
81    }
82    #[cfg(not(any(unix, windows)))]
83    return Err("guarded device media opening is unsupported on this platform".into());
84    let file = options.open(path).map_err(|error| error.to_string())?;
85    if !file
86        .metadata()
87        .map_err(|error| error.to_string())?
88        .is_file()
89    {
90        return Err("device media source must be a regular file".into());
91    }
92    Ok(file)
93}

95#[cfg(unix)]
96fn unix_open_flags() -> Result<i32, String> {
97    // O_NONBLOCK prevents a swapped FIFO from retaining a worker forever;
98    // O_NOFOLLOW refuses a swapped final symbolic link. These ABI constants
99    // are shared by Darwin/BSD but differ among Linux architectures. Unknown
100    // Unix targets are refused rather than receiving guessed open flags.
101    #[cfg(any(
102        target_os = "macos",
103        target_os = "freebsd",
104        target_os = "netbsd",
105        target_os = "openbsd",
106        target_os = "dragonfly"
107    ))]
108    return Ok(0x0004 | 0x0100);
109    #[cfg(all(
110        target_os = "linux",
111        any(
112            target_arch = "x86",
113            target_arch = "x86_64",
114            target_arch = "riscv32",
115            target_arch = "riscv64",
116            target_arch = "loongarch64"
117        )
118    ))]
119    return Ok(0x0800 | 0x0002_0000);
120    #[cfg(all(
121        target_os = "linux",
122        any(
123            target_arch = "arm",
124            target_arch = "aarch64",
125            target_arch = "powerpc",
126            target_arch = "powerpc64"
127        )
128    ))]
129    return Ok(0x0800 | 0x8000);
130    #[cfg(not(any(
131        any(
132            target_os = "macos",
133            target_os = "freebsd",
134            target_os = "netbsd",
135            target_os = "openbsd",
136            target_os = "dragonfly"
137        ),
138        all(
139            target_os = "linux",
140            any(
141                target_arch = "x86",
142                target_arch = "x86_64",
143                target_arch = "riscv32",
144                target_arch = "riscv64",
145                target_arch = "loongarch64",
146                target_arch = "arm",
147                target_arch = "aarch64",
148                target_arch = "powerpc",
149                target_arch = "powerpc64"
150            )
151        )
152    )))]
153    Err("guarded device media opening is unsupported on this Unix target".into())
154}

156impl Source {
157    fn open(&self) -> Result<File, String> {
158        if !same_source(&self.metadata, &validate_path(&self.track.path)?) {
159            return Err("device media source changed; start a new session".into());
160        }
161        let file = open_regular(&self.track.path)?;
162        self.check(&file)?;
163        Ok(file)
164    }
165
166    fn check(&self, file: &File) -> Result<(), String> {
167        self.check_descriptor(file)?;
168        if !same_source(&self.metadata, &validate_path(&self.track.path)?) {
169            return Err("device media source changed during transfer".into());
170        }
171        Ok(())
172    }
173
174    fn check_descriptor(&self, file: &File) -> Result<(), String> {
175        if !same_source(
176            &self.metadata,
177            &file.metadata().map_err(|error| error.to_string())?,
178        ) {
179            return Err("opened device media source changed during transfer".into());
180        }
181        Ok(())
182    }
183}

185pub(super) fn prepare(paths: Vec<PathBuf>) -> Result<Vec<Source>, String> {
186    if paths.is_empty() || paths.len() > MAX_TRACKS {
187        return Err("device playback needs between 1 and 64 selection occurrences".into());
188    }
189    paths
190        .into_iter()
191        .map(|path| {
192            let path = fs::canonicalize(path).map_err(|error| error.to_string())?;
193            if path.to_str().is_none() {
194                return Err("device media requires UTF-8 source paths".into());
195            }
196            let metadata = validate_path(&path)?;
197            let mut file = open_regular(&path)?;
198            if !same_source(
199                &metadata,
200                &file.metadata().map_err(|error| error.to_string())?,
201            ) {
202                return Err("device media source changed before inspecting its format".into());
203            }
204            let tags =
205                aede_core::tags::read_from_file(&mut file).map_err(|error| error.to_string())?;
206            let mime = content_type(&tags.properties);
207            if mime == "application/octet-stream" {
208                return Err("this original container has no supported device media type".into());
209            }
210            if !same_source(
211                &metadata,
212                &file.metadata().map_err(|error| error.to_string())?,
213            ) || !same_source(&metadata, &validate_path(&path)?)
214            {
215                return Err("device media source changed while inspecting its format".into());
216            }
217            let title = path
218                .file_stem()
219                .and_then(|value| value.to_str())
220                .unwrap_or("Audio")
221                .chars()
222                .filter(|character| !character.is_control())
223                .take(512)
224                .collect();
225            Ok(Source {
226                track: DeviceTrack {
227                    path,
228                    title,
229                    mime,
230                    properties: tags.properties,
231                },
232                metadata,
233            })
234        })
235        .collect()
236}

238pub(super) struct MediaServer {
239    address: SocketAddr,
240    urls: Vec<String>,
241    shutdown: watch::Sender<bool>,
242    task: JoinHandle<()>,
243}
244
245impl MediaServer {
246    pub async fn start(
247        bind: Ipv4Addr,
248        peer: Ipv4Addr,
249        sources: Vec<Source>,
250    ) -> Result<Self, String> {
251        let listener = TcpListener::bind((bind, 0))
252            .await
253            .map_err(|error| error.to_string())?;
254        let address = listener.local_addr().map_err(|error| error.to_string())?;
255        let token = aede_core::accounts::random_token()?;
256        let urls = (0..sources.len())
257            .map(|index| format!("http://{address}/{token}/{index}"))
258            .collect();
259        let (shutdown, signal) = watch::channel(false);
260        let task = tokio::spawn(run(listener, peer, Arc::new(sources), token, signal));
261        Ok(Self {
262            address,
263            urls,
264            shutdown,
265            task,
266        })
267    }
268
269    pub fn address(&self) -> SocketAddr {
270        self.address
271    }
272    pub fn urls(&self) -> &[String] {
273        &self.urls
274    }
275
276    pub async fn stop(self) {
277        let _ = self.shutdown.send(true);
278        let _ = self.task.await;
279    }
280}
281
282async fn run(
283    listener: TcpListener,
284    peer: Ipv4Addr,
285    sources: Arc<Vec<Source>>,
286    token: String,
287    mut shutdown: watch::Receiver<bool>,
288) {
289    let slots = Arc::new(Semaphore::new(4));
290    let mut tasks = JoinSet::new();
291    loop {
292        tokio::select! {
293            _ = shutdown.changed() => break,
294            _ = tasks.join_next(), if !tasks.is_empty() => {},
295            accepted = listener.accept() => {
296                let Ok((stream, address)) = accepted else { break };
297                if address.ip() != peer { continue; }
298                let Ok(permit) = slots.clone().try_acquire_owned() else { continue };
299                let sources = sources.clone();
300                let token = token.clone();
301                tasks.spawn(async move { let _ = response(stream, sources, &token, permit).await; });
302            }
303        }
304    }
305    tasks.abort_all();
306    while tasks.join_next().await.is_some() {}
307}

309struct MediaRequest {
310    index: usize,
311    head: bool,
312    headers: HeaderMap,
313}

315fn parse_request(raw: &[u8], token: &str, count: usize) -> Result<MediaRequest, u16> {
316    let text = std::str::from_utf8(raw).map_err(|_| 400_u16)?;
317    let mut lines = text.strip_suffix("\r\n\r\n").ok_or(400_u16)?.split("\r\n");
318    let words = lines.next().ok_or(400_u16)?.split(' ').collect::<Vec<_>>();
319    if words.len() != 3 || !matches!(words[2], "HTTP/1.0" | "HTTP/1.1") {
320        return Err(400);
321    }
322    if !matches!(words[0], "GET" | "HEAD") {
323        return Err(405);
324    }
325    let parts = words[1]
326        .strip_prefix('/')
327        .ok_or(404_u16)?
328        .split('/')
329        .collect::<Vec<_>>();
330    if parts.len() != 2 || parts[0] != token {
331        return Err(404);
332    }
333    let index = parts[1].parse::<usize>().map_err(|_| 404_u16)?;
334    if index >= count || index.to_string() != parts[1] {
335        return Err(404);
336    }
337    let mut headers = HeaderMap::new();
338    for (index, line) in lines.enumerate() {
339        if index >= 32 {
340            return Err(400);
341        }
342        let (name, value) = line.split_once(':').ok_or(400_u16)?;
343        let name = HeaderName::from_bytes(name.as_bytes()).map_err(|_| 400_u16)?;
344        let value = HeaderValue::from_str(value.trim()).map_err(|_| 400_u16)?;
345        if headers.contains_key(&name) {
346            return Err(400);
347        }
348        headers.insert(name, value);
349    }
350    if headers.contains_key("transfer-encoding")
351        || headers
352            .get("content-length")
353            .is_some_and(|length| length != "0")
354        || (words[2] == "HTTP/1.1" && !headers.contains_key("host"))
355    {
356        return Err(400);
357    }
358    Ok(MediaRequest {
359        index,
360        head: words[0] == "HEAD",
361        headers,
362    })
363}

365async fn read_request(stream: &mut TcpStream) -> Result<Vec<u8>, String> {
366    let mut raw = Vec::with_capacity(1024);
367    loop {
368        if raw.len() >= MAX_HEADER {
369            return Err("device HTTP header is too large".into());
370        }
371        let mut bytes = [0; 1024];
372        let size = (MAX_HEADER - raw.len()).min(bytes.len());
373        let count = stream
374            .read(&mut bytes[..size])
375            .await
376            .map_err(|error| error.to_string())?;
377        if count == 0 {
378            return Err("device HTTP header ended early".into());
379        }
380        raw.extend_from_slice(&bytes[..count]);
381        if let Some(end) = raw.windows(4).position(|bytes| bytes == b"\r\n\r\n") {
382            if end + 4 != raw.len() {
383                return Err("device HTTP requests must have no body or pipeline".into());
384            }
385            return Ok(raw);
386        }
387    }
388}

390async fn write(stream: &mut TcpStream, bytes: &[u8]) -> Result<(), String> {
391    tokio::time::timeout(PROGRESS_TIMEOUT, stream.write_all(bytes))
392        .await
393        .map_err(|_| "device HTTP output stopped making progress".to_string())?
394        .map_err(|error| error.to_string())
395}

397async fn failure(stream: &mut TcpStream, status: u16, extra: &str) -> Result<(), String> {
398    write(stream, format!("HTTP/1.1 {status} Refused\r\nContent-Length: 0\r\nConnection: close\r\nCache-Control: no-store\r\n{extra}\r\n").as_bytes()).await
399}

401struct Cancelled(Arc<AtomicBool>);
402impl Drop for Cancelled {
403    fn drop(&mut self) {
404        self.0.store(true, Ordering::Release);
405    }

408async fn response(
409    mut stream: TcpStream,
410    sources: Arc<Vec<Source>>,
411    token: &str,
412    permit: OwnedSemaphorePermit,
413) -> Result<(), String> {
414    let raw = match tokio::time::timeout(PROGRESS_TIMEOUT, read_request(&mut stream)).await {
415        Ok(Ok(raw)) => raw,
416        _ => return failure(&mut stream, 400, "").await,
417    };
418    let request = match parse_request(&raw, token, sources.len()) {
419        Ok(request) => request,
420        Err(status) => return failure(&mut stream, status, "").await,
421    };
422    let permit = Arc::new(permit);
423    let opening_sources = sources.clone();
424    let worker_permit = permit.clone();
425    let opened = tokio::task::spawn_blocking(move || {
426        let _permit = worker_permit;
427        opening_sources[request.index].open()
428    });
429    let file = match tokio::time::timeout(PROGRESS_TIMEOUT, opened).await {
430        Ok(Ok(Ok(file))) => file,
431        _ => return failure(&mut stream, 409, "").await,
432    };
433    let source = &sources[request.index];
434    let range_headers = if request.head {
435        HeaderMap::new()
436    } else {
437        request.headers
438    };
439    let range = match requested_range(&range_headers, source.metadata.len()) {
440        Ok(Some(range)) => range,
441        Ok(None) => {
442            return failure(
443                &mut stream,
444                416,
445                &format!("Content-Range: bytes */{}\r\n", source.metadata.len()),
446            )
447            .await;
448        }
449        Err(_) => return failure(&mut stream, 400, "").await,
450    };
451    let mut header = format!(
452        "HTTP/1.1 {} OK\r\nContent-Type: {}\r\nContent-Length: {}\r\nAccept-Ranges: bytes\r\nConnection: close\r\nCache-Control: no-store\r\nX-Content-Type-Options: nosniff\r\n",
453        if range.partial { 206 } else { 200 },
454        source.track.mime,
455        range.length
456    );
457    if range.partial {
458        header.push_str(&format!(
459            "Content-Range: bytes {}-{}/{}\r\n",
460            range.start,
461            range.start + range.length - 1,
462            source.metadata.len()
463        ));
464    }
465    header.push_str("\r\n");
466    write(&mut stream, header.as_bytes()).await?;
467    if request.head {
468        return Ok(());
469    }
470    let cancelled = Cancelled(Arc::new(AtomicBool::new(false)));
471    let worker_cancelled = cancelled.0.clone();
472    let (sender, mut chunks) = mpsc::channel(2);
473    tokio::task::spawn_blocking(move || {
474        let _permit = permit;
475        let source = &sources[request.index];
476        let mut file = file;
477        let result = (|| -> Result<(), String> {
478            file.seek(SeekFrom::Start(range.start))
479                .map_err(|error| error.to_string())?;
480            let mut remaining = range.length;
481            let mut path_checked = Instant::now();
482            while remaining > 0 && !worker_cancelled.load(Ordering::Acquire) {
483                let mut bytes = vec![0; remaining.min(CHUNK_BYTES as u64) as usize];
484                file.read_exact(&mut bytes)
485                    .map_err(|error| error.to_string())?;
486                remaining -= bytes.len() as u64;
487                source.check_descriptor(&file)?;
488                if remaining == 0 || path_checked.elapsed() >= PATH_CHECK_INTERVAL {
489                    source.check(&file)?;
490                    path_checked = Instant::now();
491                }
492                if sender.blocking_send(Ok(bytes)).is_err() {
493                    return Ok(());
494                }
495            }
496            Ok(())
497        })();
498        if let Err(error) = result {
499            let _ = sender.blocking_send(Err(error));
500        }
501    });
502    while let Some(chunk) = tokio::time::timeout(PROGRESS_TIMEOUT, chunks.recv())
503        .await
504        .map_err(|_| "device media read stopped making progress")?
505    {
506        write(&mut stream, &chunk?).await?;
507    }
508    Ok(())
509}
