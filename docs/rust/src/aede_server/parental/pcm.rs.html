1//! Bounded PCM reservations retain transmitted prebuffers independently of history.
2use super::*;
3use aede_core::parental::LeaseGrant;
4use std::sync::atomic::AtomicU64;
5
6struct Held {
7    grant: LeaseGrant,
8    base_ms: u64,
9    sent_base_ms: u64,
10}
11pub(crate) struct Session {
12    owner: String,
13    captured: auth::Principal,
14    held: BlockingMutex<Option<Held>>,
15    observed_ms: AtomicU64,
16    sent_ms: AtomicU64,
17    started: std::sync::OnceLock<Instant>,
18    failure: BlockingMutex<Option<&'static str>>,
19    last_checked: BlockingMutex<Option<Instant>>,
20}
21impl Session {
22    pub(crate) fn observe(&self, ms: u64) {
23        self.observed_ms.fetch_max(ms, Ordering::Relaxed);
24    }
25    pub(crate) fn sent(&self, ms: u64) {
26        self.sent_ms.fetch_max(ms, Ordering::Relaxed);
27    }
28    /// Reserve at most one second ahead of real time and never exceed this grant.
29    /// A client's rapid acknowledgements cannot turn the controlled transport into a download.
30    pub(crate) fn available(&self, sent: u64, rate: u32) -> u64 {
31        let Some(started) = self.started.get() else {
32            return 0;
33        };
34        let clock_ms = u64::try_from(started.elapsed().as_millis())
35            .unwrap_or(u64::MAX)
36            .saturating_add(1000);
37        let allowed = self
38            .held
39            .lock()
40            .ok()
41            .and_then(|held| {
42                held.as_ref().map(|held| {
43                    held.sent_base_ms
44                        .saturating_add(u64::from(held.grant.granted_seconds) * 1000)
45                })
46            })
47            .unwrap_or(0);
48        let grant_frames = allowed.saturating_mul(u64::from(rate)) / 1000;
49        let available =
50            (clock_ms.min(allowed).saturating_mul(u64::from(rate)) / 1000).saturating_sub(sent);
51        // A fast-ACK client must not cause thousands of one-frame socket writes.
52        // Accumulate about twenty milliseconds, except the exact end of a grant.
53        let quantum = (u64::from(rate) / 50)
54            .max(1)
55            .min(grant_frames.saturating_sub(sent));
56        if available < quantum { 0 } else { available }
57    }
58    pub(crate) fn failure(&self) -> Option<&'static str> {
59        self.failure.lock().ok().and_then(|f| *f)
60    }
61    pub(crate) async fn check(self: &Arc<Self>, state: &ApiState) -> bool {
62        let fresh = self
63            .last_checked
64            .lock()
65            .is_ok_and(|checked| checked.is_some_and(|at| at.elapsed() < Duration::from_secs(1)));
66        let publication_matches = state.parental_scope.as_ref().is_some_and(|scope| {
67            scope.publication == state.parental.publication.load(Ordering::Acquire)
68        });
69        let unexpired = self.held.lock().is_ok_and(|held| {
70            held.as_ref()
71                .is_none_or(|held| aede_core::clock::now_seconds() < held.grant.expires_at)
72        });
73        // Growing device ACKs can arrive much more often than the authorization tick.
74        // Keep policy reads off that cadence; admin changes and expiry bypass this cache.
75        if fresh && publication_matches && unexpired {
76            return true;
77        }
78        let held = self.clone();
79        let for_work = state.clone();
80        let result = tokio::task::spawn_blocking(move || held.refresh(&for_work, false))
81            .await
82            .unwrap_or_else(|_| Err(unavailable()));
83        if let Err(error) = result {
84            if let Ok(mut reason) = self.failure.lock() {
85                *reason = Some(error.code);
86            }
87            false
88        } else {
89            if let Ok(mut checked) = self.last_checked.lock() {
90                *checked = Some(Instant::now());
91            }
92            true
93        }
94    }
95    fn refresh(&self, state: &ApiState, close: bool) -> Result<(), ApiError> {
96        self.refresh_with_clock(state, close, aede_core::clock::now_seconds)
97    }
98    fn refresh_with_clock(
99        &self,
100        state: &ApiState,
101        close: bool,
102        mut clock: impl FnMut() -> u64,
103    ) -> Result<(), ApiError> {
104        if !close {
105            finish(state)?;
106        }
107        let now = clock();
108        if !close {
109            let data = load(state)?;
110            data.check_schedule(&self.owner, now, &Media::Music { track: None })
111                .map_err(|reason| refused(&reason))?;
112            let held = self.held.lock().map_err(|_| unavailable())?;
113            if held.as_ref().is_none_or(|held| now < held.grant.expires_at) {
114                return Ok(());
115            }
116        }
117        if self.held.lock().map_err(|_| unavailable())?.is_none() {
118            return Ok(());
119        }
120        let _lock = match lock(state) {
121            Ok(lock) => lock,
122            Err(_)
123                if !close
124                    && self.held.lock().is_ok_and(|held| {
125                        held.as_ref()
126                            .is_some_and(|held| now < held.grant.expires_at)
127                    }) =>
128            {
129                return Ok(());
130            }
131            Err(error) => return Err(error),
132        };
133        if !close {
134            auth::recheck(state, &self.captured)?;
135        }
136        let mut data = load(state)?;
137        let now = clock();
138        let mut held = self.held.lock().map_err(|_| unavailable())?;
139        let Some(held) = held.as_mut() else {
140            return Ok(());
141        };
142        let acknowledged = self
143            .observed_ms
144            .load(Ordering::Relaxed)
145            .saturating_sub(held.base_ms);
146        // A client already possesses transmitted PCM even without acknowledging it.
147        // Preserve that short prebuffer on close instead of refunding repeated
148        // receive/disconnect cycles inside the whole-second reservation clock.
149        let received = if close {
150            self.sent_ms
151                .load(Ordering::Relaxed)
152                .saturating_sub(held.sent_base_ms)
153        } else {
154            0
155        };
156        let consumed = acknowledged
157            .max(received)
158            .min(u64::from(held.grant.granted_seconds) * 1000);
159        if close {
160            data.settle(&self.owner, &held.grant.lease_id, consumed, now, true)
161                .map_err(|_| unavailable())?;
162        } else {
163            data.check_schedule(&self.owner, now, &Media::Music { track: None })
164                .map_err(|reason| refused(&reason))?;
165            held.grant = data
166                .renew(&self.owner, &held.grant.lease_id, consumed, 10, now)
167                .map_err(|reason| refused(&reason))?;
168            // All previously sent PCM belongs to the expired, fully charged
169            // grant. A delayed ACK must not spend part of this new one again.
170            let sent_ms = self.sent_ms.load(Ordering::Relaxed);
171            held.base_ms = sent_ms;
172            held.sent_base_ms = sent_ms;
173        }
174        save(state, &data)
175    }
176    pub(crate) async fn close(self: &Arc<Self>, state: &ApiState) -> Result<(), ApiError> {
177        let held = self.clone();
178        let for_work = state.clone();
179        tokio::task::spawn_blocking(move || held.refresh(&for_work, true))
180            .await
181            .unwrap_or_else(|_| Err(unavailable()))
182    }
183    pub(crate) async fn activate(self: &Arc<Self>, state: &ApiState) -> Result<(), ApiError> {
184        let session = self.clone();
185        let state = state.clone();
186        tokio::task::spawn_blocking(move || {
187            let _lock = lock(&state)?;
188            finish(&state)?;
189            auth::recheck(&state, &session.captured)?;
190            let mut held = session.held.lock().map_err(|_| unavailable())?;
191            if held.is_some() {
192                return Ok(());
193            }
194            let mut data = load(&state)?;
195            let id = aede_core::accounts::random_token().map_err(|_| unavailable())?;
196            let grant = data
197                .reserve(
198                    &session.owner,
199                    &id,
200                    Media::Music { track: None },
201                    10,
202                    aede_core::clock::now_seconds(),
203                )
204                .map_err(|reason| refused(&reason))?;
205            save(&state, &data)?;
206            *held = Some(Held {
207                grant,
208                base_ms: 0,
209                sent_base_ms: 0,
210            });
211            let _ = session.started.set(Instant::now());
212            if let Ok(mut checked) = session.last_checked.lock() {
213                *checked = Some(Instant::now());
214            }
215            Ok(())
216        })
217        .await
218        .map_err(|_| unavailable())?
219    }
220}
221pub(crate) async fn begin(
222    state: &ApiState,
223    captured: &auth::Principal,
224) -> Result<Option<Arc<Session>>, ApiError> {
225    if !supervised(state) {
226        return Ok(None);
227    }
228    let owner = captured.owner.clone();
229    Ok(Some(Arc::new(Session {
230        owner,
231        captured: captured.clone(),
232        held: BlockingMutex::new(None),
233        observed_ms: AtomicU64::new(0),
234        sent_ms: AtomicU64::new(0),
235        started: std::sync::OnceLock::new(),
236        failure: BlockingMutex::new(None),
237        last_checked: BlockingMutex::new(None),
238    })))
239}
240
241pub(crate) fn failure(state: &ApiState) -> Option<crate::playback_api::StreamFailure> {
242    state
243        .parental_pcm
244        .as_ref()
245        .and_then(|session| session.failure())
246        .map(crate::playback_api::StreamFailure::parental)
247}
