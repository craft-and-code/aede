1//! Explicit original-audio exports reuse the copy plan and revocable media transport.
2use crate::*;
3use aede_core::copy::{self, Extras, Recipe};
4use axum::http::HeaderMap;

7pub(super) fn routes() -> Router<ApiState> {
8    Router::new()
9        .route("/api/me/v1/exports/plan", post(plan))
10        .route("/api/me/v1/exports/original", get(original))
11}

13/// Raw originals cannot enforce supervised playback schedules after export.
14pub(super) fn admit(state: &ApiState, principal: &auth::Principal) -> Result<(), ApiError> {
15    auth::recheck(state, principal)?;
16    auth::require_mutation(principal)?;
17    if parental::snapshot(state, &principal.owner)?.policy.enabled {
18        return Err(parental::refused("artist_blocked"));
19    }
20    Ok(())
21}

23#[derive(Deserialize)]
24#[serde(deny_unknown_fields)]
25struct Selection {
26    references: Vec<String>,
27    format: Option<String>,
28}

30fn track_reference(token: &str) -> Result<EntityRef, ApiError> {
31    EntityRef::parse_token(token)
32        .filter(|value| value.kind == EntityKind::Track && token.len() <= MAX_REFERENCE_BYTES)
33        .ok_or_else(|| invalid_query("a bounded track reference is required"))
34}

36async fn plan(State(state): State<ApiState>, request: Request) -> Result<Json<Value>, ApiError> {
37    personal::no_query(&request)?;
38    let principal = auth::request_principal(&request)?;
39    let selection: Selection = personal::json_body(request).await?;
40    if selection.references.is_empty() || selection.references.len() > 200 {
41        return Err(invalid_query(
42            "select between 1 and 200 tracks per export plan",
43        ));
44    }
45    let target = match selection.format.as_deref() {
46        None | Some("original") => None,
47        Some(value) => Some(
48            copy::transcode::Target::parse(value)
49                .ok_or_else(|| invalid_query("unsupported export conversion format"))?,
50        ),
51    };
52    let to_read = state.clone();
53    let result = parental::blocking(&state, move || {
54        admit(&to_read, &principal)?;
55        let held = to_read.catalog.blocking_read();
56        let catalog = held.as_ref().ok_or_else(unavailable)?;
57        let mut tracks = Vec::new();
58        let mut seen = std::collections::BTreeSet::new();
59        for token in &selection.references {
60            let reference = track_reference(token)?;
61            let id = reference.resolve(catalog).ok_or_else(unavailable)?;
62            if !seen.insert(id) { return Err(invalid_query("duplicate track reference")); }
63            tracks.push(id);
64        }
65        let recipe = Recipe { extras: Extras::None, restrict_names: true, convert: None, quality: None };
66        let plan = copy::plan(catalog, &tracks, &recipe);
67        if !plan.rejected.is_empty() || !plan.rootless.is_empty() {
68            return Err(unavailable());
69        }
70        let mut rows = Vec::new();
71        for token in selection.references {
72            let reference = track_reference(&token)?;
73            let source = playback_api::source_from_catalog(catalog, &reference).map_err(|_| unavailable())?;
74            let item = plan.items.iter().find(|item| item.source == source.path).ok_or_else(unavailable)?;
75            copy::validate_source(&source.path).map_err(|_| unavailable())?;
76            let components: Vec<_> = item.relative.components().map(|part| part.as_os_str().to_str().map(str::to_owned).ok_or_else(unavailable)).collect::<Result<_, _>>()?;
77            let key: Vec<_> = components.iter().map(|part| copy::names::portable_key(part)).collect();
78            let output = copy::staged_output_path(&source.file, &components.join("/"), target);
79            let output_components: Vec<_> = output.components().map(|part| part.as_os_str().to_str().map(str::to_owned).ok_or_else(unavailable)).collect::<Result<_, _>>()?;
80            let output_key: Vec<_> = output_components.iter().map(|part| copy::names::portable_key(part)).collect();
81            rows.push(json!({"reference":token,"components":components,"key":key,"output_components":output_components,"output_key":output_key,"bytes":source.file.size,"stamp":crate::subsonic::media::source_stamp(&source)}));
82        }
83        admit(&to_read, &principal)?;
84        Ok(json!({"version":1,"scanned_at":catalog.scanned_at,"items":rows,"roots":catalog.roots}))
85    }).await?;
86    Ok(Json(result))
87}

89#[derive(Deserialize)]
90#[serde(deny_unknown_fields)]
91struct Original {
92    stamp: String,
93    #[serde(rename = "ref")]
94    reference: String,
95}
96async fn original(State(state): State<ApiState>, request: Request) -> Result<Response, ApiError> {
97    let principal = auth::request_principal(&request)?;
98    let Query(query): Query<Original> = Query::try_from_uri(request.uri())
99        .map_err(|_| invalid_query("invalid original reference"))?;
100    let reference = track_reference(&query.reference)?;
101    let headers: HeaderMap = request.headers().clone();
102    let to_check = state.clone();
103    let captured = principal.clone();
104    parental::blocking(&state, move || admit(&to_check, &captured)).await?;
105    subsonic::media::response_for(
106        state,
107        subsonic::media::MediaIdentity::Session(principal),
108        reference,
109        headers,
110        true,
111        Some(query.stamp),
112    )
113    .await
114    .map_err(|_| {
115        error(
116            StatusCode::CONFLICT,
117            "export_unavailable",
118            "original audio is unavailable or authorization changed; refresh and retry",
119        )
120    })
121}
