1//! Account preferences can be saved before the first catalog scan.
2use super::*;
3#[derive(Default, Deserialize)]
4#[serde(deny_unknown_fields)]
5struct EmptyQuery {}
6#[derive(Deserialize)]
7#[serde(deny_unknown_fields)]
8struct Patch {
9    #[serde(deserialize_with = "unique_values")]
10    values: BTreeMap<String, serde_json::Value>,
11}
12fn unique_values<'de, D: serde::Deserializer<'de>>(
13    deserializer: D,
14) -> Result<BTreeMap<String, serde_json::Value>, D::Error> {
15    struct Values;
16    impl<'de> serde::de::Visitor<'de> for Values {
17        type Value = BTreeMap<String, serde_json::Value>;
18        fn expecting(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
19            formatter.write_str("application preferences with unique keys")
20        }
21        fn visit_map<A: serde::de::MapAccess<'de>>(
22            self,
23            mut input: A,
24        ) -> Result<Self::Value, A::Error> {
25            let mut values = BTreeMap::new();
26            while let Some((key, value)) = input.next_entry::<String, serde_json::Value>()? {
27                if values.insert(key, value).is_some() {
28                    return Err(serde::de::Error::custom("duplicate application preference"));
29                }
30            }
31            Ok(values)
32        }
33    }
34    deserializer.deserialize_map(Values)
35}
36#[derive(Serialize)]
37struct View {
38    values: BTreeMap<String, serde_json::Value>,
39}
40pub(super) fn routes() -> Router<ApiState> {
41    Router::new().route("/api/me/v1/preferences", get(read).patch(write))
42}
43fn snapshot(data: &UserData, owner: &str) -> Result<View, ApiError> {
44    let values = data
45        .client_preferences
46        .iter()
47        .filter(|row| row.owner == owner && row.key.starts_with("phemios."))
48        .map(|row| serde_json::from_str(&row.value).map(|value| (row.key.clone(), value)))
49        .collect::<Result<_, _>>()
50        .map_err(|_| {
51            error(
52                StatusCode::INTERNAL_SERVER_ERROR,
53                "user_unavailable",
54                "invalid preferences",
55            )
56        })?;
57    Ok(View { values })
58}
59fn locked(state: &ApiState, owner: &PersonalOwner, patch: Option<Patch>) -> Result<View, ApiError> {
60    let _guard = StoreLock::try_acquire(&state.data_dir)
61        .map_err(|_| error(StatusCode::CONFLICT, "store_busy", "personal store is busy"))?;
62    owner.recheck(state)?;
63    let path = user::user_path(&state.data_dir);
64    let mut data = user::load(&path)
65        .map_err(|_| {
66            error(
67                StatusCode::INTERNAL_SERVER_ERROR,
68                "user_unavailable",
69                "personal store could not be read",
70            )
71        })?
72        .unwrap_or_default();
73    if let Some(patch) = patch {
74        for (key, value) in patch.values {
75            let value = value.to_string();
76            if let Some(row) = data
77                .client_preferences
78                .iter_mut()
79                .find(|row| row.owner == owner.id && row.key == key)
80            {
81                row.value = value;
82            } else {
83                data.client_preferences.push(user::ClientPreference {
84                    owner: owner.id.clone(),
85                    key,
86                    value,
87                });
88            }
89        }
90        if data.client_preferences.len() > 8192
91            || data
92                .client_preferences
93                .iter()
94                .filter(|row| row.owner == owner.id)
95                .count()
96                > 64
97        {
98            return Err(invalid_query("too many preferences"));
99        }
100        owner.recheck(state)?;
101        user::save(&data, &path).map_err(|_| {
102            error(
103                StatusCode::INTERNAL_SERVER_ERROR,
104                "store_error",
105                "preferences could not be saved",
106            )
107        })?;
108    }
109    owner.admit(state, snapshot(&data, &owner.id)?)
110}
111async fn read(
112    State(state): State<ApiState>,
113    input: Result<Query<EmptyQuery>, QueryRejection>,
114    request: Request,
115) -> Result<Json<View>, ApiError> {
116    parsed_query(input)?;
117    let owner = PersonalOwner::from_request(&state, &request)?;
118    spawn_personal(state.inspection_slots.clone(), move || {
119        locked(&state, &owner, None)
120    })?
121    .await
122    .map_err(|_| {
123        error(
124            StatusCode::INTERNAL_SERVER_ERROR,
125            "personal_failed",
126            "preferences worker failed",
127        )
128    })?
129    .map(Json)
130}
131async fn write(
132    State(state): State<ApiState>,
133    input: Result<Query<EmptyQuery>, QueryRejection>,
134    request: Request,
135) -> Result<Json<View>, ApiError> {
136    parsed_query(input)?;
137    let owner = PersonalOwner::from_request(&state, &request)?;
138    owner.require_mutation()?;
139    let patch: Patch = json_body(request).await?;
140    if patch.values.is_empty()
141        || patch.values.len() > 32
142        || patch.values.iter().any(|(key, value)| {
143            !key.starts_with("phemios.")
144                || !user::valid_preference(key, &value.to_string())
145                || key == "phemios.moods"
146                    && value.as_array().is_some_and(|rows| {
147                        rows.iter().any(|row| {
148                            row.get("id")
149                                .and_then(serde_json::Value::as_str)
150                                .is_some_and(aede_moods::is_audio_recipe_id)
151                        })
152                    })
153        })
154    {
155        return Err(invalid_query("invalid application preferences"));
156    }
157    spawn_personal(state.inspection_slots.clone(), move || {
158        locked(&state, &owner, Some(patch))
159    })?
160    .await
161    .map_err(|_| {
162        error(
163            StatusCode::INTERNAL_SERVER_ERROR,
164            "personal_failed",
165            "preferences worker failed",
166        )
167    })?
168    .map(Json)
169}
