1//! Bounded, owner-private album completion; public catalog caches never carry this state.
2use super::*;
3pub(super) fn routes() -> Router<ApiState> {
4    Router::new().route("/api/me/v1/album-listening", get(read))
5}
6async fn read(
7    State(state): State<ApiState>,
8    input: Result<Query<PageQuery>, QueryRejection>,
9    request: Request,
10) -> Result<Json<serde_json::Value>, ApiError> {
11    let owner = PersonalOwner::from_request(&state, &request)?;
12    empty_body(&body(request).await?)?;
13    let (offset, limit) = page(parsed_query(input)?)?;
14    spawn_personal(state.inspection_slots.clone(), move || {
15        let (_guard, catalog, data) = locked_personal(&state)?;
16        owner.recheck(&state)?;
17        let completed = aede_core::listening::completed_albums(&catalog, &data, &owner.id);
18        let items: Vec<_> = catalog.releases.iter().skip(offset).take(limit).filter_map(|release| {
19            EntityRef::of(&catalog, aede_core::model::EntityKind::Release, release.id).map(|reference| serde_json::json!({"reference":reference.to_token(),"completed":completed.contains(&release.id)}))
20        }).collect();
21        owner.admit(&state, Json(serde_json::json!({"total":catalog.releases.len(),"offset":offset,"items":items})))
22    })?.await.map_err(|_| unavailable())?
23}
