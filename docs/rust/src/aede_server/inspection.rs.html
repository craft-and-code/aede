34#[derive(Clone, Copy)]
35struct Window {
36    offset: usize,
37    limit: usize,
38}
39
40impl Window {
41    fn page<T>(self, rows: Vec<T>, scanned_at: u64) -> Page<T> {
42        Page {
43            total: rows.len(),
44            items: rows
45                .into_iter()
46                .skip(self.offset)
47                .take(self.limit)
48                .collect(),
49            offset: self.offset,
50            limit: self.limit,
51            scanned_at,
52        }
53    }
54}
55
56fn parameters(
57    input: Result<Query<Parameters>, QueryRejection>,
58    allowed: &[&str],
59) -> Result<(Parameters, Window), ApiError> {
60    let Query(params) = input.map_err(|rejection| invalid_query(rejection.body_text()))?;
61    for (name, present) in [
62        ("q", params.q.is_some()),
63        ("comments", params.comments.is_some()),
64        ("severity", params.severity.is_some()),
65        ("sort", params.sort.is_some()),
66    ] {
67        if present && !allowed.contains(&name) {
68            return Err(invalid_query(format!(
69                "{name} is not supported for this route"
70            )));
71        }
72    }
73    let (offset, limit) = page_bounds(params.offset.as_deref(), params.limit.as_deref())?;
74    Ok((params, Window { offset, limit }))
75}
76
77fn required_text(params: &Parameters) -> Result<&str, ApiError> {
78    let value = params
79        .q
80        .as_deref()
81        .ok_or_else(|| invalid_query("q is required"))?;
82    if value.len() > MAX_TEXT_BYTES || text::normalize(value).is_empty() {
83        return Err(invalid_query(format!(
84            "q must contain text and be at most {MAX_TEXT_BYTES} bytes"
85        )));
86    }
87    Ok(value)
88}

103/// CPU-bound graph inspection and local store reads never occupy a runtime
104/// worker. Keep the permit until the blocking work finishes, even if its HTTP
105/// client disconnects; dropping a handler must not bypass the admission limit.
106pub(super) async fn inspect<T, F>(state: ApiState, operation: F) -> Result<Json<T>, ApiError>
107where
108    T: Send + 'static,
109    F: FnOnce(&ApiState) -> Result<T, ApiError> + Send + 'static,
110{
111    let permit = state
112        .inspection_slots
113        .clone()
114        .try_acquire_owned()
115        .map_err(|_| {
116            error(
117                StatusCode::TOO_MANY_REQUESTS,
118                "inspection_busy",
119                "too many catalog inspections; retry later",
120            )
121        })?;
122    tokio::task::spawn_blocking(move || {
123        let _permit = permit;
124        operation(&state)
125    })
126    .await
127    .map_err(|failure| {
128        eprintln!("API catalog inspection failed: {failure}");
129        error(
130            StatusCode::INTERNAL_SERVER_ERROR,
131            "inspection_failed",
132            "catalog inspection could not finish",
133        )
134    })?
135    .map(Json)
136}

188fn diagnosis(
189    catalog: &Catalog,
190    held: &sources::Sources,
191    severity: Option<&str>,
192    window: Window,
193) -> Diagnosis {
194    let mut issues = doctor::diagnose(catalog, held);
195    if let Some(severity) = severity {
196        issues.retain(|issue| issue.severity().label() == severity);
197    }
198    let mut summary = BTreeMap::from([("error", 0), ("warning", 0), ("info", 0)]);
199    for (severity, count) in doctor::summary(&issues) {
200        summary.insert(severity.label(), count);
201    }
202    let items = issues
203        .into_iter()
204        .map(|issue| {
205            let file_count = issue.files.len();
206            IssueItem {
207                kind: issue_code(issue.kind),
208                label: issue.kind.label(),
209                severity: issue.severity().label(),
210                detail: issue.detail,
211                files: issue.files.into_iter().take(ISSUE_FILE_PREVIEW).collect(),
212                file_count,
213                files_truncated: file_count > ISSUE_FILE_PREVIEW,
214            }
215        })
216        .collect();
217    Diagnosis {
218        page: window.page(items, catalog.scanned_at),
219        summary,
220        unverified_files: catalog
221            .files
222            .iter()
223            .filter(|file| file.integrity.is_none())
224            .count(),
225        pending_analyses: catalog.pending_analyses(),
226    }
227}

229async fn diagnose(
230    State(state): State<ApiState>,
231    input: Result<Query<Parameters>, QueryRejection>,
232) -> Result<Json<Diagnosis>, ApiError> {
233    let (params, window) = parameters(input, &["severity"])?;
234    if params
235        .severity
236        .as_deref()
237        .is_some_and(|value| !["error", "warning", "info"].contains(&value))
238    {
239        return Err(invalid_query("severity must be error, warning or info"));
240    }
241    inspect(state, move |state| {
242        // A check/import can update conclusions without replacing the catalog.
243        // Read all diagnostic inputs under the writer lock instead of reporting
244        // stale integrity findings from the cached catalog snapshot.
245        let _guard = StoreLock::try_acquire(&state.data_dir).map_err(|failure| {
246            if failure.kind() == std::io::ErrorKind::WouldBlock {
247                error(
248                    StatusCode::CONFLICT,
249                    "store_busy",
250                    "Aède data is busy; retry after the current write",
251                )
252            } else {
253                error(
254                    StatusCode::INTERNAL_SERVER_ERROR,
255                    "store_error",
256                    "Aède data could not be locked for diagnosis",
257                )
258            }
259        })?;
260        let catalog = store::load(&store::catalog_path(&state.data_dir))
261            .map_err(|_| {
262                error(
263                    StatusCode::INTERNAL_SERVER_ERROR,
264                    "catalog_read_failed",
265                    "catalog or conclusions could not be read",
266                )
267            })?
268            .ok_or_else(unavailable)?;
269        let held = held_sources(&state.data_dir)?;
270        Ok(diagnosis(
271            &catalog,
272            &held,
273            params.severity.as_deref(),
274            window,
275        ))
276    })
277    .await
278}

457struct RootItem {
458    status: &'static str,
459    path: Option<String>,
460    tracks: usize,
461    duration_ms: u64,
462    bytes: u64,
463}
464
465fn root_item(
466    catalog: &Catalog,
467    status: &'static str,
468    path: Option<String>,
469    tracks: impl Iterator<Item = Id>,
470) -> RootItem {
471    let mut item = RootItem {
472        status,
473        path,
474        tracks: 0,
475        duration_ms: 0,
476        bytes: 0,
477    };
478    for track in tracks.filter_map(|id| catalog.track(id)) {
479        item.tracks += 1;
480        item.duration_ms = item
481            .duration_ms
482            .saturating_add(track.duration_ms.unwrap_or(0));
483        item.bytes = item.bytes.saturating_add(
484            catalog
485                .file(track.file_id)
486                .map(|file| file.size)
487                .unwrap_or(0),
488        );
489    }
490    item
491}
492
493fn root_items(catalog: &Catalog) -> Vec<RootItem> {
494    let tracks_under = |root: &str| {
495        catalog
496            .tracks
497            .iter()
498            .filter_map(|track| {
499                catalog
500                    .file(track.file_id)
501                    .filter(|file| text::is_under(&file.path, root))
502                    .map(|_| track.id)
503            })
504            .collect::<Vec<_>>()
505    };
506    let mut watched = std::collections::BTreeSet::new();
507    let mut items = Vec::new();
508    for root in &catalog.roots {
509        let tracks = tracks_under(root);
510        watched.extend(tracks.iter().copied());
511        items.push(root_item(
512            catalog,
513            "watched",
514            Some(root.clone()),
515            tracks.into_iter(),
516        ));
517    }
518    for root in &catalog.excluded {
519        items.push(root_item(
520            catalog,
521            "excluded",
522            Some(root.clone()),
523            tracks_under(root).into_iter(),
524        ));
525    }
526    let orphaned: Vec<_> = catalog
527        .tracks
528        .iter()
529        .map(|track| track.id)
530        .filter(|id| !watched.contains(id))
531        .collect();
532    if !orphaned.is_empty() {
533        items.push(root_item(catalog, "unwatched", None, orphaned.into_iter()));
534    }
535    items
536}

587fn search_items(catalog: &Catalog, wanted: &str, comments: bool) -> Vec<SearchItem> {

604                .filter_map(|id| {
605                    catalog.track(id).map(|track| SearchItem {
606                        kind: "track",
607                        reference: reference(catalog, EntityKind::Track, id),
608                        name: track.title.clone(),
609                        context: catalog.comment_of_track(id).unwrap_or_default().to_string(),
610                        found_in: "comment",
611                    })
612                }),
613        );
614    }
615    items

619    State(state): State<ApiState>,
620    input: Result<Query<Parameters>, QueryRejection>,
621) -> Result<Json<Page<SearchItem>>, ApiError> {

625        "true" => true,

632        Ok(window.page(search_items(catalog, &wanted, comments), catalog.scanned_at))
633    })
634    .await

637fn public_query(expression: &str) -> Result<query::Query, ApiError> {
638    // Bound tokens/parentheses/unary negations before the recursive core parser.
639    // Quoted values remain intact, using the parser's single/double quote rules.
640    let mut quote = None;
641    let mut in_word = false;
642    let mut parts = 0;
643    for ch in expression.chars() {
644        match quote {
645            Some(current) if current == ch => quote = None,
646            Some(_) => {}
647            None if ch == '\'' || ch == '"' => {
648                quote = Some(ch);
649                if !in_word {
650                    parts += 1;
651                }
652                in_word = true;
653            }
654            None if ch == '(' || ch == ')' || ch == '-' => {
655                parts += 1;
656                in_word = false;
657            }
658            None if ch.is_whitespace() => in_word = false,
659            None => {
660                if !in_word {
661                    parts += 1;
662                }
663                in_word = true;
664            }
665        }
666        if parts > MAX_QUERY_PARTS {
667            return Err(invalid_query(
668                "query has too many terms, parentheses or negations",
669            ));
670        }
671    }
672    let parsed = query::parse(expression).map_err(|failure| invalid_query(failure.to_string()))?;
673    let mut pending = vec![&parsed];
674    while let Some(part) = pending.pop() {
675        match part {
676            query::Query::And(parts) | query::Query::Or(parts) => pending.extend(parts),
677            query::Query::Not(part) => pending.push(part),
678            query::Query::Term(term) => match term.field {
679                query::Field::Rating(_)
680                | query::Field::Loved(_)
681                | query::Field::LovedAnywhere
682                | query::Field::Tag(_)
683                | query::Field::Note(_)
684                | query::Field::Played => {
685                    return Err(invalid_query(
686                        "personal query fields require authenticated accounts and are not exposed",
687                    ));
688                }
689                query::Field::Lyrics => {
690                    return Err(invalid_query(
691                        "lyrics queries are not exposed by the catalog API",
692                    ));
693                }
694                query::Field::Title
695                | query::Field::Artist
696                | query::Field::Album
697                | query::Field::Recording
698                | query::Field::Work
699                | query::Field::Movement
700                | query::Field::ReleaseGroup
701                | query::Field::AlbumArtist
702                | query::Field::Genre
703                | query::Field::Label
704                | query::Field::Comment
705                | query::Field::Path
706                | query::Field::Codec
707                | query::Field::Year
708                | query::Field::Duration
709                | query::Field::Size
710                | query::Field::Bitrate
711                | query::Field::SampleRate
712                | query::Field::Lossless
713                | query::Field::Compilation
714                | query::Field::Performing
715                | query::Field::Credit(_)
716                | query::Field::Instrument
717                | query::Field::Guest
718                | query::Field::CompilationArtist
719                | query::Field::Contributor
720                | query::Field::Collaborator
721                | query::Field::Anything => {}
722            },
723            query::Query::All => {}
724        }
725    }
726    Ok(parsed)
727}

729fn query_items(
730    catalog: &Catalog,
731    held: &sources::Sources,
732    parsed: &query::Query,
733    sort: query::Sort,
734    window: Window,
735) -> Result<Page<TrackItem>, ApiError> {
736    let empty_private_data = aede_core::user::UserData::default();
737    let context = query::Context::new(catalog, &empty_private_data, "").with_sources(held);
738    if let Some((field, value)) = query::unknown_values(parsed, &context).first() {
739        return Err(invalid_query(format!("no {field} matches {value:?}")));
740    }
741    let mut tracks = query::run(parsed, &context);
742    query::sort(&mut tracks, sort, &context);
743    let total = tracks.len();
744    let items = tracks
745        .into_iter()
746        .skip(window.offset)
747        .take(window.limit)
748        .filter_map(|id| {
749            catalog.track(id).map(|track| TrackItem {
750                reference: reference(catalog, EntityKind::Track, id).unwrap_or_default(),
751                title: track.title.clone(),
752                release: track
753                    .release_id
754                    .and_then(|id| reference(catalog, EntityKind::Release, id)),
755                recording: reference(catalog, EntityKind::Recording, track.recording_id),
756                duration_ms: track.duration_ms,
757            })
758        })
759        .collect();
760    Ok(Page::new(
761        items,
762        total,
763        window.offset,
764        window.limit,
765        catalog.scanned_at,
766    ))
767}
768
769async fn run_query(
770    State(state): State<ApiState>,
771    input: Result<Query<Parameters>, QueryRejection>,
772) -> Result<Json<Page<TrackItem>>, ApiError> {
773    let (params, window) = parameters(input, &["q", "sort"])?;
774    let parsed = public_query(required_text(&params)?)?;
775    let sort = query::Sort::parse(params.sort.as_deref().unwrap_or("catalog"))
776        .map_err(|failure| invalid_query(failure.to_string()))?;
777    if matches!(sort.key, query::SortKey::Rating | query::SortKey::Played) {
778        return Err(invalid_query("personal sort fields are not exposed"));
779    }
780    inspect(state, move |state| {
781        let guard = state.catalog.blocking_read();
782        let catalog = guard.as_ref().ok_or_else(unavailable)?;
783        query_items(
784            catalog,
785            &held_sources(&state.data_dir)?,
786            &parsed,
787            sort,
788            window,
789        )
790    })
791    .await
792}

794#[derive(Serialize)]
795struct CountryItem {
796    name: String,
797    iso_code: Option<String>,
798    derived_initials: Option<String>,
799    artists: usize,
800    tracks: usize,
801    duration_ms: u64,
802    bytes: u64,
803}
804
805#[derive(Serialize)]
806struct CountryCoverage {
807    artists_total: usize,
808    artists_asked: usize,
809    artists_with_area: usize,
810    artists_without_area: usize,
811    places_without_iso_code: usize,
812}
813
814#[derive(Serialize)]
815struct Countries {
816    #[serde(flatten)]
817    page: Page<CountryItem>,
818    source: &'static str,
819    matched_by: Option<&'static str>,
820    coverage: CountryCoverage,
821}
822
823fn country_rows(
824    catalog: &Catalog,
825    held: &sources::Sources,
826    wanted: Option<&str>,
827    window: Window,
828) -> Result<Countries, ApiError> {
829    let mut places = aede_core::places::countries(catalog, held);
830    let known = places.iter().map(|place| place.artists.len()).sum();
831    let coverage = CountryCoverage {
832        artists_total: catalog.artists.len(),
833        artists_asked: aede_core::places::asked_about(catalog, held),
834        artists_with_area: known,
835        artists_without_area: catalog.artists.len().saturating_sub(known),
836        places_without_iso_code: aede_core::places::without_code(&places),
837    };
838    let matched_by = if let Some(wanted) = wanted {
839        let (found, matched) = aede_core::places::find(&places, wanted);
840        if found.is_empty() {
841            return Err(invalid_query("no known country or area matches q"));
842        }
843        places = found;
844        Some(match matched {
845            aede_core::model::TitleMatch::Exact => "exact",
846            aede_core::model::TitleMatch::Partial => "partial",
847        })
848    } else {
849        None
850    };
851    let items = places
852        .into_iter()
853        .map(|place| {
854            let tracks: std::collections::BTreeSet<_> = place
855                .artists
856                .iter()
857                .flat_map(|&id| catalog.tracks_of_artist(id))
858                .collect();
859            let totals = root_item(catalog, "country", None, tracks.into_iter());
860            CountryItem {
861                name: place.name,
862                iso_code: place.code,
863                derived_initials: place.initials,
864                artists: place.artists.len(),
865                tracks: totals.tracks,
866                duration_ms: totals.duration_ms,
867                bytes: totals.bytes,
868            }
869        })
870        .collect();
871    Ok(Countries {
872        page: window.page(items, catalog.scanned_at),
873        source: sources::MUSICBRAINZ,
874        matched_by,
875        coverage,
876    })
877}

902#[derive(Serialize)]
903struct YearItem {
904    year: u32,
905    albums: usize,
906    tracks: usize,
907    duration_ms: u64,
908    bytes: u64,
909}
910
911#[derive(Serialize)]
912struct Years {
913    #[serde(flatten)]
914    page: Page<YearItem>,
915    albums_total: usize,
916    albums_without_year: usize,
917    tracks_without_dated_album: usize,
918}
919
920fn year_rows(catalog: &Catalog, window: Window) -> Years {
921    let mut groups: BTreeMap<u32, (usize, Vec<Id>)> = BTreeMap::new();
922    let mut dated_tracks = std::collections::BTreeSet::new();
923    let mut albums_without_year = 0;
924    for release in &catalog.releases {
925        let Some(year) = release.year else {
926            albums_without_year += 1;
927            continue;
928        };
929        let group = groups.entry(year).or_default();
930        group.0 += 1;
931        group.1.extend(release.track_ids.iter().copied());
932        dated_tracks.extend(release.track_ids.iter().copied());
933    }
934    let items = groups
935        .into_iter()
936        .map(|(year, (albums, tracks))| {
937            let totals = root_item(catalog, "year", None, tracks.into_iter());
938            YearItem {
939                year,
940                albums,
941                tracks: totals.tracks,
942                duration_ms: totals.duration_ms,
943                bytes: totals.bytes,
944            }
945        })
946        .collect();
947    Years {
948        page: window.page(items, catalog.scanned_at),
949        albums_total: catalog.releases.len(),
950        albums_without_year,
951        tracks_without_dated_album: catalog.tracks.len().saturating_sub(dated_tracks.len()),
952    }
953}
954
955async fn years(
956    State(state): State<ApiState>,
957    input: Result<Query<Parameters>, QueryRejection>,
958) -> Result<Json<Years>, ApiError> {
959    let (_, window) = parameters(input, &[])?;
960    inspect(state, move |state| {
961        let guard = state.catalog.blocking_read();
962        let catalog = guard.as_ref().ok_or_else(unavailable)?;
963        Ok(year_rows(catalog, window))
964    })
965    .await
966}
