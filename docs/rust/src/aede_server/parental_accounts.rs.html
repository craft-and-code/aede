1//! Administrative parental policy editing under the account writer lock.
2use super::*;
3use aede_core::parental::{Policy, PolicySnapshot};

5pub(super) fn routes() -> Router<ApiState> {
6    Router::new().route(
7        "/api/admin/v1/accounts/:username/parental",
8        get(read).put(update),
9    )
10}
11#[derive(Deserialize)]
12#[serde(deny_unknown_fields)]
13struct Change {
14    expected_owner: String,
15    expected_revision: u64,
16    policy: Policy,
17}
18
19fn target<'a>(accounts: &'a Accounts, name: &str) -> Result<&'a Account, ApiError> {
20    let account = accounts.find(name).ok_or_else(missing)?;
21    if account.role == Role::Administrator {
22        return Err(refused(
23            "parental policies cannot restrict an administrator",
24        ));
25    }
26    Ok(account)
27}
28async fn read(
29    State(state): State<ApiState>,
30    input: Result<PathParameter<String>, PathRejection>,
31    request: Request,
32) -> Result<Json<PolicySnapshot>, ApiError> {
33    let captured = admin_capture(&state, &request)?;
34    personal::empty_body(&personal::body(request).await?)?;
35    let name = path_name(input)?;
36    let for_work = state.clone();
37    crate::parental::blocking(&state, move || {
38        let _guard = lock_accounts(&for_work)?;
39        check_capture(&for_work, captured.as_ref())?;
40        let accounts = enabled_accounts(&for_work)?;
41        let account = target(&accounts, &name)?;
42        Ok(Json(crate::parental::snapshot(&for_work, &account.id)?))
43    })
44    .await
45}
46async fn update(
47    State(state): State<ApiState>,
48    input: Result<PathParameter<String>, PathRejection>,
49    request: Request,
50) -> Result<Json<PolicySnapshot>, ApiError> {
51    let captured = admin_capture(&state, &request)?;
52    let name = path_name(input)?;
53    let mut change: Change = personal::json_body(request).await?;
54    change.policy.blocked_podcasts = change
55        .policy
56        .blocked_podcasts
57        .iter()
58        .map(|feed| crate::parental::http::canonical_feed(feed))
59        .collect::<Result<_, _>>()?;
60    let for_work = state.clone();
61    crate::parental::blocking(&state, move || {
62        let _guard = lock_accounts(&for_work)?;
63        check_capture(&for_work, captured.as_ref())?;
64        let accounts = enabled_accounts(&for_work)?;
65        let account = target(&accounts, &name)?;
66        if account.id != change.expected_owner {
67            return Err(error(
68                StatusCode::CONFLICT,
69                "parental_conflict",
70                "the target account changed; reload before saving",
71            ));
72        }
73        let mut data = crate::parental::load(&for_work)?;
74        let snapshot = data.policy(&account.id);
75        let administrator = captured.as_ref().map(|principal| principal.owner.as_str());
76        if snapshot
77            .locked_by
78            .as_deref()
79            .is_some_and(|id| Some(id) != administrator)
80        {
81            return Err(locked());
82        }
83        if change.policy.enabled && administrator.is_none() {
84            return Err(error(
85                StatusCode::FORBIDDEN,
86                "parental_administrator_required",
87                "sign in as an administrator to own the supervision lock",
88            ));
89        }
90        if data.policy(&account.id).revision != change.expected_revision {
91            return Err(error(
92                StatusCode::CONFLICT,
93                "parental_conflict",
94                "the parental policy changed; reload before saving",
95            ));
96        }
97        let saved = data
98            .set_policy_by(
99                &account.id,
100                administrator,
101                change.policy,
102                change.expected_revision,
103            )
104            .map_err(refused)?;
105        check_capture(&for_work, captured.as_ref())?;
106        crate::parental::save(&for_work, &data)?;
107        for_work.parental.invalidate();
108        Ok(Json(saved))
109    })
110    .await
111}

113pub(super) fn locked() -> ApiError {
114    error(
115        StatusCode::FORBIDDEN,
116        "parental_locked",
117        "this account is locked by its supervising administrator",
118    )

121pub(super) fn check_account_edit(
122    state: &ApiState,
123    actor: Option<&auth::Principal>,
124    account: &Account,
125) -> Result<(), ApiError> {
126    let data = crate::parental::load(state)?;
127    // Another administrator must not reset the supervisor's password and impersonate its lock.
128    if data.supervises(&account.id)
129        && actor.map(|value| value.owner.as_str()) != Some(account.id.as_str())
130    {
131        return Err(locked());
132    }
133    if data
134        .policy(&account.id)
135        .locked_by
136        .as_deref()
137        .is_some_and(|id| Some(id) != actor.map(|value| value.owner.as_str()))
138    {
139        return Err(locked());
140    }
141    Ok(())
142}

144pub(super) fn check_administrator_retained(
145    state: &ApiState,
146    account: &Account,
147    role: Option<&str>,
148    enabled: Option<bool>,
149) -> Result<(), ApiError> {
150    if role == Some("admin")
151        && crate::parental::load(state)?
152            .policy(&account.id)
153            .policy
154            .enabled
155    {
156        return Err(error(
157            StatusCode::FORBIDDEN,
158            "parental_locked",
159            "disable supervision before promoting this account",
160        ));
161    }
162    if (enabled == Some(false) || role.is_some_and(|role| role != "admin"))
163        && crate::parental::load(state)?.supervises(&account.id)
164    {
165        return Err(error(
166            StatusCode::FORBIDDEN,
167            "parental_locked",
168            "release the supervised account locks before disabling or demoting this administrator",
169        ));
170    }
171    Ok(())
