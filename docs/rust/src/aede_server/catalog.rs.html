5pub(super) fn filter_reference(
6    catalog: &Catalog,
7    token: &str,
8    kind: EntityKind,
9) -> Result<Id, ApiError> {
10    let requested = EntityRef::parse_token(token)
11        .filter(|reference| reference.kind == kind && !reference.key.trim().is_empty())
12        .ok_or_else(|| invalid_query(format!("filter must be a {} reference", kind.as_str())))?;
13    requested.resolve(catalog).ok_or_else(|| {
14        error(
15            StatusCode::NOT_FOUND,
16            "entity_not_found",
17            "no entity matches this reference in the current catalog",
18        )
19    })
20}

26pub(super) fn release_year_key(
27    release: &aede_core::model::Release,
28    descending: bool,
29) -> (bool, u32, String) {
30    let year = release.year.unwrap_or_default();
31    (
32        release.year.is_none(),
33        if descending { u32::MAX - year } else { year },
34        text::normalize(&release.title),
35    )
36}

75pub(super) async fn artists(
76    State(state): State<ApiState>,
77    query: Result<Query<ListQuery>, QueryRejection>,
78) -> Result<Json<Page<ArtistItem>>, ApiError> {
79    let options = list_query(query, ListKind::Artist)?;
80    let guard = state.catalog.read().await;
81    let catalog = guard.as_ref().ok_or_else(unavailable)?;
82    let mut rows: Vec<_> = catalog
83        .artists
84        .iter()
85        .filter(|artist| {
86            options
87                .mbid
88                .as_deref()
89                .is_none_or(|mbid| artist.mbid.as_deref() == Some(mbid))
90                && options.q.as_deref().is_none_or(|q| {
91                    text::normalize(&artist.name).contains(q)
92                        || artist
93                            .aliases
94                            .iter()
95                            .any(|alias| text::normalize(alias).contains(q))
96                })
97        })
98        .collect();
99    match options.sort {
100        SortMode::Catalog if options.descending => rows.reverse(),
101        SortMode::Name if options.descending => {
102            rows.sort_by_cached_key(|artist| std::cmp::Reverse(text::normalize(&artist.sort_name)));
103        }
104        SortMode::Name => rows.sort_by_cached_key(|artist| text::normalize(&artist.sort_name)),
105        _ => {}
106    }
107    let total = rows.len();
108    let items = rows
109        .into_iter()
110        .skip(options.offset)
111        .take(options.limit)
112        .filter_map(|artist| {
113            Some(ArtistItem {
114                reference: reference(catalog, EntityKind::Artist, artist.id)?,
115                name: artist.name.clone(),
116                sort_name: artist.sort_name.clone(),
117                mbid: artist.mbid.clone(),
118                aliases: artist.aliases.clone(),
119            })
120        })
121        .collect();
122    Ok(Json(Page::new(
123        items,
124        total,
125        options.offset,
126        options.limit,
127        catalog.scanned_at,
128    )))
129}

131pub(super) async fn releases(
132    State(state): State<ApiState>,
133    query: Result<Query<ListQuery>, QueryRejection>,
134) -> Result<Json<Page<ReleaseItem>>, ApiError> {
135    let options = list_query(query, ListKind::Release)?;
136    let guard = state.catalog.read().await;
137    let catalog = guard.as_ref().ok_or_else(unavailable)?;
138    let artist_id = options
139        .artist
140        .as_deref()
141        .map(|token| filter_reference(catalog, token, EntityKind::Artist))
142        .transpose()?;
143    let mut rows: Vec<_> = catalog
144        .releases
145        .iter()
146        .filter(|release| {
147            options.year.is_none_or(|year| release.year == Some(year))
148                && artist_id.is_none_or(|id| release.album_artist_id == Some(id))
149                && options
150                    .q
151                    .as_deref()
152                    .is_none_or(|q| text::normalize(&release.title).contains(q))
153        })
154        .collect();
155    match options.sort {
156        SortMode::Catalog if options.descending => rows.reverse(),
157        SortMode::Title if options.descending => {
158            rows.sort_by_cached_key(|release| std::cmp::Reverse(text::normalize(&release.title)));
159        }
160        SortMode::Title => rows.sort_by_cached_key(|release| text::normalize(&release.title)),
161        SortMode::Year => {
162            rows.sort_by_cached_key(|release| release_year_key(release, options.descending))
163        }
164        _ => {}
165    }
166    let total = rows.len();
167    let items = rows
168        .into_iter()
169        .skip(options.offset)
170        .take(options.limit)
171        .filter_map(|release| release_item(catalog, release))
172        .collect();
173    Ok(Json(Page::new(
174        items,
175        total,
176        options.offset,
177        options.limit,
178        catalog.scanned_at,
179    )))
180}

293pub(super) async fn entity(
294    State(state): State<ApiState>,
295    query: Result<Query<EntityQuery>, QueryRejection>,
296) -> Result<Json<EntityDetail>, ApiError> {
297    let Query(query) = query.map_err(|rejection| {
298        error(
299            StatusCode::BAD_REQUEST,
300            "invalid_reference",
301            rejection.body_text(),
302        )
303    })?;
304    let requested = EntityRef::parse_token(&query.reference)
305        .filter(|reference| !reference.key.trim().is_empty())
306        .ok_or_else(|| {
307            error(
308                StatusCode::BAD_REQUEST,
309                "invalid_reference",
310                "ref must be a stable entity reference such as artist:miles davis",
311            )
312        })?;
313    let guard = state.catalog.read().await;
314    let catalog = guard.as_ref().ok_or_else(unavailable)?;
315    entity_detail(catalog, &requested).map(Json)
316}

318pub(super) fn entity_detail(
319    catalog: &Catalog,
320    requested: &EntityRef,
321) -> Result<EntityDetail, ApiError> {
322    let id = requested.resolve(catalog).ok_or_else(|| {
323        error(
324            StatusCode::NOT_FOUND,
325            "entity_not_found",
326            "no entity matches this reference in the current catalog",
327        )
328    })?;
329    let detail = match requested.kind {
330        EntityKind::Artist => {
331            let artist = catalog.artist(id).ok_or_else(unavailable)?;
332            EntityDetail::Artist {
333                reference: requested.to_token(),
334                name: artist.name.clone(),
335                sort_name: artist.sort_name.clone(),
336                mbid: artist.mbid.clone(),
337                aliases: artist.aliases.clone(),
338                releases: catalog
339                    .releases_as_album_artist(id)
340                    .into_iter()
341                    .filter_map(|id| reference(catalog, EntityKind::Release, id))
342                    .collect(),
343            }
344        }
345        EntityKind::Release => {
346            let release = catalog.release(id).ok_or_else(unavailable)?;
347            EntityDetail::Release {
348                reference: requested.to_token(),
349                title: release.title.clone(),
350                year: release.year,
351                album_artist: release
352                    .album_artist_id
353                    .and_then(|id| reference(catalog, EntityKind::Artist, id)),
354                tracks: release
355                    .track_ids
356                    .iter()
357                    .filter_map(|&id| reference(catalog, EntityKind::Track, id))
358                    .collect(),
359                release_group: release
360                    .release_group_id
361                    .and_then(|id| reference(catalog, EntityKind::ReleaseGroup, id)),
362                labels: release
363                    .label_ids
364                    .iter()
365                    .filter_map(|&id| reference(catalog, EntityKind::Label, id))
366                    .collect(),
367                cover_path: release.cover_path.clone(),
368            }
369        }
370        EntityKind::Track => {
371            let track = catalog.track(id).ok_or_else(unavailable)?;
372            let file = catalog.file(track.file_id).ok_or_else(unavailable)?;
373            let analyses = catalog
374                .analyses_of(file)
375                .map(|record| {
376                    let mut analysis = aede_core::store::analysis_to_json(record);
377                    analysis.set(
378                        "stale",
379                        (!record.still_applies(file.size, file.mtime)).into(),
380                    );
381                    serde_json::from_str(&analysis.to_string_compact()).map_err(|_| {
382                        error(
383                            StatusCode::INTERNAL_SERVER_ERROR,
384                            "analysis_unavailable",
385                            "stored analysis could not be serialized",
386                        )
387                    })
388                })
389                .collect::<Result<Vec<_>, ApiError>>()?;
390            EntityDetail::Track {
391                reference: requested.to_token(),
392                title: track.title.clone(),
393                release: track
394                    .release_id
395                    .and_then(|id| reference(catalog, EntityKind::Release, id)),
396                recording: reference(catalog, EntityKind::Recording, track.recording_id),
397                duration_ms: track.duration_ms,
398                path: file.path.clone(),
399                size: file.size,
400                analyses,
401            }
402        }
403        EntityKind::Recording => {
404            let recording = catalog.recording(id).ok_or_else(unavailable)?;
405            EntityDetail::Recording {
406                reference: requested.to_token(),
407                title: recording.title.clone(),
408                isrc: recording.isrc.clone(),
409                mbid: recording.mbid.clone(),
410                tracks: recording
411                    .track_ids
412                    .iter()
413                    .filter_map(|&id| reference(catalog, EntityKind::Track, id))
414                    .collect(),
415                works: recording
416                    .work_ids
417                    .iter()
418                    .filter_map(|&id| reference(catalog, EntityKind::Work, id))
419                    .collect(),
420            }
421        }
422        EntityKind::Work => {
423            let work = catalog.work(id).ok_or_else(unavailable)?;
424            EntityDetail::Work {
425                reference: requested.to_token(),
426                title: work.title.clone(),
427                mbid: work.mbid.clone(),
428                recordings: work
429                    .recording_ids
430                    .iter()
431                    .filter_map(|&id| reference(catalog, EntityKind::Recording, id))
432                    .collect(),
433            }
434        }
435        EntityKind::ReleaseGroup => {
436            let group = catalog.release_group(id).ok_or_else(unavailable)?;
437            EntityDetail::ReleaseGroup {
438                reference: requested.to_token(),
439                title: group.title.clone(),
440                mbid: group.mbid.clone(),
441                releases: group
442                    .release_ids
443                    .iter()
444                    .filter_map(|&id| reference(catalog, EntityKind::Release, id))
445                    .collect(),
446            }
447        }
448        EntityKind::Label => {
449            let label = catalog.label(id).ok_or_else(unavailable)?;
450            EntityDetail::Label {
451                reference: requested.to_token(),
452                name: label.name.clone(),
453                mbid: label.mbid.clone(),
454                releases: catalog
455                    .releases_of_label(id)
456                    .into_iter()
457                    .filter_map(|id| reference(catalog, EntityKind::Release, id))
458                    .collect(),
459            }
460        }
461        EntityKind::Genre => {
462            let genre = catalog.genre(id).ok_or_else(unavailable)?;
463            EntityDetail::Genre {
464                reference: requested.to_token(),
465                name: genre.name.clone(),
466                releases: catalog
467                    .releases_of_genre(id)
468                    .into_iter()
469                    .filter_map(|id| reference(catalog, EntityKind::Release, id))
470                    .collect(),
471                tracks: catalog
472                    .tracks_of_genre(id)
473                    .into_iter()
474                    .filter_map(|id| reference(catalog, EntityKind::Track, id))
475                    .collect(),
476            }
477        }
478    };
479    Ok(detail)
