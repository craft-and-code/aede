5pub(super) fn filter_reference(
6    catalog: &Catalog,
7    token: &str,
8    kind: EntityKind,
9) -> Result<Id, ApiError> {
10    let requested = EntityRef::parse_token(token)
11        .filter(|reference| reference.kind == kind && !reference.key.trim().is_empty())
12        .ok_or_else(|| invalid_query(format!("filter must be a {} reference", kind.as_str())))?;
13    requested.resolve(catalog).ok_or_else(|| {
14        error(
15            StatusCode::NOT_FOUND,
16            "entity_not_found",
17            "no entity matches this reference in the current catalog",
18        )
19    })
20}

22pub(super) fn reference(catalog: &Catalog, kind: EntityKind, id: u32) -> Option<String> {
23    EntityRef::of(catalog, kind, id).map(|reference| reference.to_token())

26pub(super) fn release_year_key(
27    release: &aede_core::model::Release,
28    descending: bool,
29) -> (bool, u32, String) {
30    let year = release.year.unwrap_or_default();
31    (
32        release.year.is_none(),
33        if descending { u32::MAX - year } else { year },
34        text::normalize(&release.title),
35    )
36}

132pub(super) async fn artists(
133    State(state): State<ApiState>,
134    query: Result<Query<ListQuery>, QueryRejection>,
135) -> Result<Json<Page<ArtistItem>>, ApiError> {
136    let options = list_query(query, ListKind::Artist)?;
137    inspection::inspect(state, move |state| {
138        let guard = state.catalog.blocking_read();
139        let catalog = guard.as_ref().ok_or_else(unavailable)?;
140        let mut rows: Vec<_> = catalog
141            .artists
142            .iter()
143            .filter(|artist| {
144                options
145                    .mbid
146                    .as_deref()
147                    .is_none_or(|mbid| artist.mbid.as_deref() == Some(mbid))
148                    && options.q.as_deref().is_none_or(|q| {
149                        text::normalize(&artist.name).contains(q)
150                            || artist
151                                .aliases
152                                .iter()
153                                .any(|alias| text::normalize(alias).contains(q))
154                    })
155            })
156            .collect();
157        match options.sort {
158            SortMode::Catalog if options.descending => rows.reverse(),
159            SortMode::Name if options.descending => {
160                rows.sort_by_cached_key(|artist| {
161                    std::cmp::Reverse(text::normalize(&artist.sort_name))
162                });
163            }
164            SortMode::Name => rows.sort_by_cached_key(|artist| text::normalize(&artist.sort_name)),
165            _ => {}
166        }
167        let total = rows.len();
168        let items = rows
169            .into_iter()
170            .skip(options.offset)
171            .take(options.limit)
172            .filter_map(|artist| {
173                Some(ArtistItem {
174                    reference: reference(catalog, EntityKind::Artist, artist.id)?,
175                    name: artist.name.clone(),
176                    sort_name: artist.sort_name.clone(),
177                    mbid: artist.mbid.clone(),
178                    aliases: artist.aliases.clone(),
179                })
180            })
181            .collect();
182        Ok(Page::new(
183            items,
184            total,
185            options.offset,
186            options.limit,
187            catalog.scanned_at,
188        ))
189    })
190    .await
191}

193pub(super) async fn releases(
194    State(state): State<ApiState>,
195    query: Result<Query<ListQuery>, QueryRejection>,
196) -> Result<Json<Page<ReleaseItem>>, ApiError> {
197    let options = list_query(query, ListKind::Release)?;
198    inspection::inspect(state, move |state| {
199        let guard = state.catalog.blocking_read();
200        let catalog = guard.as_ref().ok_or_else(unavailable)?;
201        let artist_id = options
202            .artist
203            .as_deref()
204            .map(|token| filter_reference(catalog, token, EntityKind::Artist))
205            .transpose()?;
206        let mut rows: Vec<_> = catalog
207            .releases
208            .iter()
209            .filter(|release| {
210                options.year.is_none_or(|year| release.year == Some(year))
211                    && artist_id.is_none_or(|id| release.album_artist_id == Some(id))
212                    && options
213                        .q
214                        .as_deref()
215                        .is_none_or(|q| text::normalize(&release.title).contains(q))
216            })
217            .collect();
218        match options.sort {
219            SortMode::Catalog if options.descending => rows.reverse(),
220            SortMode::Title if options.descending => {
221                rows.sort_by_cached_key(|release| {
222                    std::cmp::Reverse(text::normalize(&release.title))
223                });
224            }
225            SortMode::Title => rows.sort_by_cached_key(|release| text::normalize(&release.title)),
226            SortMode::Year => {
227                rows.sort_by_cached_key(|release| release_year_key(release, options.descending))
228            }
229            _ => {}
230        }
231        let total = rows.len();
232        let items = rows
233            .into_iter()
234            .skip(options.offset)
235            .take(options.limit)
236            .filter_map(|release| release_item(catalog, release))
237            .collect();
238        Ok(Page::new(
239            items,
240            total,
241            options.offset,
242            options.limit,
243            catalog.scanned_at,
244        ))
245    })
246    .await
247}

369pub(super) async fn entity(
370    State(state): State<ApiState>,
371    query: Result<Query<EntityQuery>, QueryRejection>,
372) -> Result<Json<EntityDetail>, ApiError> {
373    let Query(query) = query.map_err(|rejection| {
374        error(
375            StatusCode::BAD_REQUEST,
376            "invalid_reference",
377            rejection.body_text(),
378        )
379    })?;
380    bounded_text(&query.reference, "ref", MAX_REFERENCE_BYTES)?;
381    let requested = EntityRef::parse_token(&query.reference)
382        .filter(|reference| !reference.key.trim().is_empty())
383        .ok_or_else(|| {
384            error(
385                StatusCode::BAD_REQUEST,
386                "invalid_reference",
387                "ref must be a stable entity reference such as artist:miles davis",
388            )
389        })?;
390    inspection::inspect(state, move |state| {
391        let guard = state.catalog.blocking_read();
392        let catalog = guard.as_ref().ok_or_else(unavailable)?;
393        entity_detail(catalog, &requested)
394    })
395    .await
396}

398pub(super) fn entity_detail(
399    catalog: &Catalog,
400    requested: &EntityRef,
401) -> Result<EntityDetail, ApiError> {
402    let id = requested.resolve(catalog).ok_or_else(|| {
403        error(
404            StatusCode::NOT_FOUND,
405            "entity_not_found",
406            "no entity matches this reference in the current catalog",
407        )
408    })?;
409    let detail = match requested.kind {
410        EntityKind::Artist => {
411            let artist = catalog.artist(id).ok_or_else(unavailable)?;
412            EntityDetail::Artist {
413                reference: requested.to_token(),
414                name: artist.name.clone(),
415                sort_name: artist.sort_name.clone(),
416                mbid: artist.mbid.clone(),
417                aliases: artist.aliases.clone(),
418                releases: catalog
419                    .releases_as_album_artist(id)
420                    .into_iter()
421                    .filter_map(|id| reference(catalog, EntityKind::Release, id))
422                    .collect(),
423            }
424        }
425        EntityKind::Release => {
426            let release = catalog.release(id).ok_or_else(unavailable)?;
427            EntityDetail::Release {
428                reference: requested.to_token(),
429                title: release.title.clone(),
430                year: release.year,
431                album_artist: release
432                    .album_artist_id
433                    .and_then(|id| reference(catalog, EntityKind::Artist, id)),
434                tracks: release
435                    .track_ids
436                    .iter()
437                    .filter_map(|&id| reference(catalog, EntityKind::Track, id))
438                    .collect(),
439                release_group: release
440                    .release_group_id
441                    .and_then(|id| reference(catalog, EntityKind::ReleaseGroup, id)),
442                labels: release
443                    .label_ids
444                    .iter()
445                    .filter_map(|&id| reference(catalog, EntityKind::Label, id))
446                    .collect(),
447                cover_path: release.cover_path.clone(),
448            }
449        }
450        EntityKind::Track => {
451            let track = catalog.track(id).ok_or_else(unavailable)?;
452            let file = catalog.file(track.file_id).ok_or_else(unavailable)?;
453            let analyses = catalog
454                .analyses_of(file)
455                .map(|record| {
456                    let mut analysis = aede_core::store::analysis_to_json(record);
457                    analysis.set(
458                        "stale",
459                        (!record.still_applies(file.size, file.mtime)).into(),
460                    );
461                    serde_json::from_str(&analysis.to_string_compact()).map_err(|_| {
462                        error(
463                            StatusCode::INTERNAL_SERVER_ERROR,
464                            "analysis_unavailable",
465                            "stored analysis could not be serialized",
466                        )
467                    })
468                })
469                .collect::<Result<Vec<_>, ApiError>>()?;
470            EntityDetail::Track {
471                reference: requested.to_token(),
472                title: track.title.clone(),
473                release: track
474                    .release_id
475                    .and_then(|id| reference(catalog, EntityKind::Release, id)),
476                recording: reference(catalog, EntityKind::Recording, track.recording_id),
477                duration_ms: track.duration_ms,
478                disc_no: track.disc_no,
479                track_no: track.track_no,
480                properties: serde_json::json!({"codec":file.properties.codec,"container":file.properties.container,"sample_rate":file.properties.sample_rate,"bit_depth":file.properties.bit_depth,"channels":file.properties.channels,"bitrate_kbps":file.properties.bitrate_kbps,"lossless":file.properties.lossless}),
481                tags: file.tags.clone(),
482                credits: catalog.credits.iter().filter(|credit| credit.entity_kind == EntityKind::Track && credit.entity_id == id).map(|credit| {
483                    let artist = catalog.artist(credit.artist_id).ok_or_else(unavailable)?;
484                    Ok(serde_json::json!({"reference":reference(catalog,EntityKind::Artist,artist.id),"name":artist.name,"role":credit.role,"source":credit.source}))
485                }).collect::<Result<Vec<_>, ApiError>>()?,
486                path: file.path.clone(),
487                size: file.size,
488                analyses,
489                lyrics_available: aede_core::lyrics::has_source(
490                    Path::new(&file.path),
491                    file.first_tag("lyrics"),
492                    file.lyrics_path.as_deref().map(Path::new),
493                ),
494                spectrum_available: aede_core::spectrum::picture_for(Path::new(&file.path))
495                    .is_file(),
496            }
497        }
498        EntityKind::Recording => {
499            let recording = catalog.recording(id).ok_or_else(unavailable)?;
500            EntityDetail::Recording {
501                reference: requested.to_token(),
502                title: recording.title.clone(),
503                isrc: recording.isrc.clone(),
504                mbid: recording.mbid.clone(),
505                tracks: recording
506                    .track_ids
507                    .iter()
508                    .filter_map(|&id| reference(catalog, EntityKind::Track, id))
509                    .collect(),
510                works: recording
511                    .work_ids
512                    .iter()
513                    .filter_map(|&id| reference(catalog, EntityKind::Work, id))
514                    .collect(),
515            }
516        }
517        EntityKind::Work => {
518            let work = catalog.work(id).ok_or_else(unavailable)?;
519            EntityDetail::Work {
520                reference: requested.to_token(),
521                title: work.title.clone(),
522                mbid: work.mbid.clone(),
523                recordings: work
524                    .recording_ids
525                    .iter()
526                    .filter_map(|&id| reference(catalog, EntityKind::Recording, id))
527                    .collect(),
528            }
529        }
530        EntityKind::ReleaseGroup => {
531            let group = catalog.release_group(id).ok_or_else(unavailable)?;
532            EntityDetail::ReleaseGroup {
533                reference: requested.to_token(),
534                title: group.title.clone(),
535                mbid: group.mbid.clone(),
536                releases: group
537                    .release_ids
538                    .iter()
539                    .filter_map(|&id| reference(catalog, EntityKind::Release, id))
540                    .collect(),
541            }
542        }
543        EntityKind::Label => {
544            let label = catalog.label(id).ok_or_else(unavailable)?;
545            EntityDetail::Label {
546                reference: requested.to_token(),
547                name: label.name.clone(),
548                mbid: label.mbid.clone(),
549                releases: catalog
550                    .releases_of_label(id)
551                    .into_iter()
552                    .filter_map(|id| reference(catalog, EntityKind::Release, id))
553                    .collect(),
554            }
555        }
556        EntityKind::Genre => {
557            let genre = catalog.genre(id).ok_or_else(unavailable)?;
558            EntityDetail::Genre {
559                reference: requested.to_token(),
560                name: genre.name.clone(),
561                releases: catalog
562                    .releases_of_genre(id)
563                    .into_iter()
564                    .filter_map(|id| reference(catalog, EntityKind::Release, id))
565                    .collect(),
566                tracks: catalog
567                    .tracks_of_genre(id)
568                    .into_iter()
569                    .filter_map(|id| reference(catalog, EntityKind::Track, id))
570                    .collect(),
571            }
572        }
573    };
574    Ok(detail)
575}
