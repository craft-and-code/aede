27    limit: Option<usize>,

29fn bounds(query: Result<Query<Paging>, QueryRejection>) -> Result<(usize, usize), ApiError> {
30    let Query(query) = query.map_err(|failure| invalid_query(failure.body_text()))?;
31    let limit = query.limit.unwrap_or(DEFAULT_LIMIT);
32    if limit == 0 || limit > MAX_LIMIT {
33        return Err(invalid_query("limit must be between 1 and 200"));
34    }
35    Ok((query.offset.unwrap_or(0), limit))
36}
37fn reference_of(catalog: &Catalog, kind: EntityKind, id: Id) -> Result<String, ApiError> {
38    reference(catalog, kind, id).ok_or_else(unavailable)
39}
40fn related_releases(catalog: &Catalog, id: Id) -> Vec<Id> {
41    let mut releases: BTreeSet<_> = catalog.releases_of_artist(id).into_iter().collect();
42    releases.extend(
43        catalog
44            .releases
45            .iter()
46            .filter(|release| release.album_artist_id == Some(id))
47            .map(|release| release.id),
48    );
49    releases.into_iter().collect()
50}
51fn artist_row(
52    catalog: &Catalog,
53    sources: &Sources,
54    data_dir: &Path,
55    artist: &aede_core::model::Artist,
56) -> Result<Value, ApiError> {
57    let entity = EntityRef::of(catalog, EntityKind::Artist, artist.id).ok_or_else(unavailable)?;
58    let origin = crate::catalog_commands::artist_origin(catalog, &entity, sources);
59    let releases = related_releases(catalog, artist.id);
60    let genres: BTreeSet<_> = releases
61        .iter()
62        .flat_map(|id| catalog.genres_of(EntityKind::Release, *id))
63        .map(|genre| genre.name.clone())
64        .collect();
65    let biography = sources
66        .about(&entity)
67        .filter(|record| sources.is_trusted(catalog, record))
68        .find_map(|record| {
69            if let Facts::Artist(facts) = &record.facts {
70                facts.summary.as_ref()
71            } else {
72                None
73            }
74        });
75    let folders = crate::artwork::artist_folders(catalog, sources, artist.id, data_dir);
76    let media = |kind| {
77        folders
78            .iter()
79            .any(|folder| crate::artwork::beside(folder.clone(), kind).is_some())
80    };
81    let relations:Vec<_>=catalog.neighbours_of_artist(artist.id).into_iter().map(|(other,weight,kind)|json!({"reference":reference(catalog,EntityKind::Artist,other.id),"name":other.name,"weight":weight,"kind":kind,"source":"tags"})).collect();
82    Ok(
83        json!({"reference":entity.to_token(),"name":artist.name,"genre":genres.iter().cloned().collect::<Vec<_>>().join(" · "),"genres":genres,"country":origin.country_code.as_ref().or(origin.area.as_ref()),"country_code":origin.country_code,"origin":origin,"biography":biography.map(|prose|prose.text.as_str()).unwrap_or(""),"biography_credit":biography.map(|prose|json!({"url":prose.url,"lang":prose.lang,"licence":prose.licence})),"artwork":false,"media":{"banner":media(Kind::Banner),"background":media(Kind::Background),"portrait":media(Kind::Artist),"logo":media(Kind::Logo)},"release_refs":releases.into_iter().map(|id|reference_of(catalog,EntityKind::Release,id)).collect::<Result<Vec<_>,_>>()?,"search_fields":[artist.sort_name,artist.mbid.as_deref().unwrap_or(""),artist.aliases.join(" ")],"relations":relations}),
84    )
85}
86fn album_row(
87    catalog: &Catalog,
88    release: &aede_core::model::Release,
89    composers: &std::collections::BTreeMap<Id, BTreeSet<Id>>,
90) -> Result<Value, ApiError> {
91    let reference = reference_of(catalog, EntityKind::Release, release.id)?;
92    let mut contributors: BTreeSet<_> = catalog
93        .credits_on(EntityKind::Release, release.id)
94        .into_iter()
95        .map(|(artist, _)| artist.id)
96        .collect();
97    let mut search: Vec<_> = composers
98        .get(&release.id)
99        .into_iter()
100        .flatten()
101        .filter_map(|id| catalog.artist(*id))
102        .flat_map(|artist| {
103            std::iter::once(artist.name.clone()).chain(artist.aliases.iter().cloned())
104        })
105        .collect();
106    for id in &release.track_ids {
107        if let Some(track) = catalog.track(*id) {
108            contributors.extend(
109                catalog
110                    .credits_on(EntityKind::Track, *id)
111                    .into_iter()
112                    .map(|(artist, _)| artist.id),
113            );
114            if let Some(file) = catalog.file(track.file_id) {
115                for (name, values) in &file.tags {
116                    search.push(name.clone());
117                    search.extend(values.iter().cloned());
118                }
119            }
120        }
121    }
122    if let Some(artist) = release.album_artist_id {
123        contributors.insert(artist);
124    }
125    let genres: Vec<_> = catalog
126        .genres_of(EntityKind::Release, release.id)
127        .iter()
128        .map(|genre| genre.name.clone())
129        .collect();
130    Ok(
131        json!({"reference":reference,"artist":release.album_artist_id.and_then(|id|crate::catalog::reference(catalog,EntityKind::Artist,id)),"title":release.title,"year":release.year,"compilation":release.is_compilation,"tags":genres,"composers":composers.get(&release.id).into_iter().flatten().map(|id|reference_of(catalog,EntityKind::Artist,*id).map(|reference|json!({"reference":reference}))).collect::<Result<Vec<_>,_>>()?,"contributors":contributors.into_iter().map(|id|reference_of(catalog,EntityKind::Artist,id)).collect::<Result<Vec<_>,_>>()?,"cover":crate::subsonic::artwork::candidate(release).is_some_and(|source|crate::subsonic::artwork::available(&source)),"track_count":release.track_ids.len(),"added_at":catalog.release_added_at(release.id),"search_fields":search}),
132    )
133}
134async fn artists(
135    State(state): State<ApiState>,
136    query: Result<Query<Paging>, QueryRejection>,
137) -> Result<Json<Page<Value>>, ApiError> {
138    let (offset, limit) = bounds(query)?;
139    inspection::inspect(state, move |state| {
140        let guard = state.catalog.blocking_read();
141        let catalog = guard.as_ref().ok_or_else(unavailable)?;
142        let sources = inspection::held_sources(&state.data_dir)?;
143        let items = catalog
144            .artists
145            .iter()
146            .skip(offset)
147            .take(limit)
148            .map(|artist| artist_row(catalog, &sources, &state.data_dir, artist))
149            .collect::<Result<_, _>>()?;
150        Ok(Page::new(
151            items,
152            catalog.artists.len(),
153            offset,
154            limit,
155            catalog.scanned_at,
156        ))
157    })
158    .await
159}
160async fn albums(
161    State(state): State<ApiState>,
162    query: Result<Query<Paging>, QueryRejection>,
163) -> Result<Json<Page<Value>>, ApiError> {
164    let (offset, limit) = bounds(query)?;
165    inspection::inspect(state, move |state| {
166        let guard = state.catalog.blocking_read();
167        let catalog = guard.as_ref().ok_or_else(unavailable)?;
168        let composers = catalog.composers_by_release();
169        let items = catalog
170            .releases
171            .iter()
172            .skip(offset)
173            .take(limit)
174            .map(|release| album_row(catalog, release, &composers))
175            .collect::<Result<_, _>>()?;
176        Ok(Page::new(
177            items,
178            catalog.releases.len(),
179            offset,
180            limit,
181            catalog.scanned_at,
182        ))
183    })
184    .await
185}

190    #[serde(rename = "ref")]
191    reference: String,
192}
193pub(super) fn playback_rows(catalog: &Catalog, token: &str) -> Result<Vec<Value>, ApiError> {
194    if token.len() > MAX_REFERENCE_BYTES {
195        return Err(invalid_query("reference is too long"));
196    }
197    let entity = EntityRef::parse_token(token)
198        .ok_or_else(|| invalid_query("invalid selection reference"))?;
199    let id = entity.resolve(catalog).ok_or_else(|| {
200        error(
201            StatusCode::NOT_FOUND,
202            "entity_not_found",
203            "selection is no longer available",
204        )
205    })?;
206    if entity.kind == EntityKind::Track {
207        return Ok(vec![playback_track_row(catalog, id)?]);
208    }
209    let ids = match entity.kind {
210        EntityKind::Release => vec![id],
211        EntityKind::Artist => catalog.playback_release_order(catalog.releases_as_album_artist(id)),
212        EntityKind::Label => catalog.playback_release_order(catalog.releases_of_label(id)),
213        _ => {
214            return Err(invalid_query(
215                "selection must be an artist, album, track or label",
216            ));
217        }
218    };
219    let mut rows = Vec::new();
220    for id in ids {
221        let release = catalog.release(id).ok_or_else(unavailable)?;
222        let reference = reference_of(catalog, EntityKind::Release, id)?;
223        for (index, id) in release.track_ids.iter().enumerate() {
224            if rows.len() >= 4096 {
225                return Err(invalid_query("selection exceeds 4096 tracks"));
226            }
227            let track = catalog.track(*id).ok_or_else(unavailable)?;
228            rows.push(json!({"reference": reference_of(catalog, EntityKind::Track, *id)?, "title": track.title, "release": reference, "number": index + 1, "album_total": release.track_ids.len()}));
229        }
230    }
231    Ok(rows)
232}
233/// Render an already resolved catalogue track without repeating reference resolution.
234pub(super) fn playback_track_row(catalog: &Catalog, id: Id) -> Result<Value, ApiError> {
235    let track = catalog.track(id).ok_or_else(unavailable)?;
236    let release = track.release_id.and_then(|id| catalog.release(id));
237    let position = release.and_then(|r| r.track_ids.iter().position(|p| *p == id));
238    Ok(
239        json!({"reference":reference_of(catalog,EntityKind::Track,id)?,"title":track.title,
240        "release":release.map(|r| reference_of(catalog,EntityKind::Release,r.id)).transpose()?,
241        "number":position.map_or(1, |p| p+1), "album_total":release.map_or(1, |r| r.track_ids.len())}),
242    )
243}
244async fn playback_selection(
245    State(state): State<ApiState>,
246    query: Result<Query<PlaybackQuery>, QueryRejection>,
247) -> Result<Json<Value>, ApiError> {
248    let Query(query) = query.map_err(|failure| invalid_query(failure.body_text()))?;
249    inspection::inspect(state, move |state| {
250        let guard = state.catalog.blocking_read();
251        let catalog = guard.as_ref().ok_or_else(unavailable)?;
252        Ok(json!({"items": playback_rows(catalog, &query.reference)?, "scanned_at": catalog.scanned_at}))
253    }).await
254}
