1//! Lightweight ordered track pages; retained analyses are read on demand.
2use super::*;

4#[derive(Deserialize)]
5#[serde(deny_unknown_fields)]
6pub(super) struct Parameters {
7    #[serde(rename = "ref")]
8    reference: Option<String>,
9    q: Option<String>,
10    offset: Option<usize>,
11    limit: Option<usize>,
12}
13pub(super) async fn list(
14    State(state): State<ApiState>,
15    query: Result<Query<Parameters>, QueryRejection>,
16) -> Result<Json<Value>, ApiError> {
17    let Query(query) = query.map_err(|e| invalid_query(e.body_text()))?;
18    let (offset, limit) = bounds(Ok(Query(Paging {
19        offset: query.offset,
20        limit: query.limit,
21    })))?;
22    if query.q.as_ref().is_some_and(|q| q.len() > 1024) {
23        return Err(invalid_query("track search is too long"));
24    }
25    inspection::inspect(state, move |state| {
26        let guard = state.catalog.blocking_read();
27        listing(
28            guard.as_ref().ok_or_else(unavailable)?,
29            query.reference.as_deref(),
30            offset,
31            limit,
32            query.q.as_deref(),
33        )
34    })
35    .await
36}
37#[cfg(test)]
38pub(super) fn page(
39    catalog: &Catalog,
40    token: &str,
41    offset: usize,
42    limit: usize,
43) -> Result<Value, ApiError> {
44    listing(catalog, Some(token), offset, limit, None)
45}
46fn searchable(
47    c: &Catalog,
48    track: &aede_core::model::Track,
49    query: &str,
50    credited: &std::collections::HashSet<Id>,
51) -> bool {
52    let has = |value: &str| aede_core::text::normalize(value).contains(query);
53    has(&track.title)
54        || c.file(track.file_id).is_some_and(|f| {
55            f.tags
56                .iter()
57                .any(|(key, values)| has(key) || values.iter().any(|v| has(v)))
58        })
59        || track
60            .release_id
61            .and_then(|id| c.release(id))
62            .is_some_and(|r| {
63                has(&r.title)
64                    || r.album_artist_id
65                        .and_then(|id| c.artist(id))
66                        .is_some_and(|a| has(&a.name))
67            })
68        || credited.contains(&track.id)
69}
70pub(super) fn listing(
71    c: &Catalog,
72    token: Option<&str>,
73    offset: usize,
74    limit: usize,
75    query: Option<&str>,
76) -> Result<Value, ApiError> {
77    let ids: Vec<_> = if let Some(token) = token {
78        if token.len() > MAX_REFERENCE_BYTES {
79            return Err(invalid_query("reference is too long"));
80        }
81        let id = catalog::filter_reference(c, token, EntityKind::Release)?;
82        c.release(id).ok_or_else(unavailable)?.track_ids.clone()
83    } else {
84        c.tracks.iter().map(|t| t.id).collect()
85    };
86    let query = query
87        .map(aede_core::text::normalize)
88        .filter(|q| !q.is_empty());
89    // A search visits each credit once rather than rescanning the credit table per track.
90    let mut credited = std::collections::HashSet::new();
91    if let Some(query) = &query {
92        let artists: std::collections::HashSet<_> = c
93            .artists
94            .iter()
95            .filter(|a| aede_core::text::normalize(&a.name).contains(query))
96            .map(|a| a.id)
97            .collect();
98        for credit in &c.credits {
99            if credit.entity_kind == EntityKind::Track
100                && (artists.contains(&credit.artist_id)
101                    || aede_core::text::normalize(&credit.role).contains(query))
102            {
103                credited.insert(credit.entity_id);
104            }
105        }
106    }
107    let rows: Vec<_> = ids
108        .into_iter()
109        .filter(|id| {
110            c.track(*id).is_some_and(|t| {
111                query
112                    .as_ref()
113                    .is_none_or(|q| searchable(c, t, q, &credited))
114            })
115        })
116        .collect();
117    render_rows(c, &rows, offset, limit)
118}

120pub(crate) fn render_rows(
121    c: &Catalog,
122    rows: &[Id],
123    offset: usize,
124    limit: usize,
125) -> Result<Value, ApiError> {
126    let items = rows.iter().skip(offset).take(limit).map(|id| {
127        let track = c.track(*id).ok_or_else(unavailable)?;
128        let file = c.file(track.file_id).ok_or_else(unavailable)?;
129        let release = track.release_id.and_then(|id| c.release(id));
130        Ok(json!({
131            "reference":reference_of(c,EntityKind::Track,*id)?, "title":track.title,
132            "release":release.map(|r| reference_of(c,EntityKind::Release,r.id)).transpose()?,
133            "album_title":release.map(|r| r.title.as_str()),
134            "disc_no":track.disc_no,"track_no":track.track_no,
135            "album_position":release.and_then(|r| r.track_ids.iter().position(|id| *id == track.id)).map(|p| p+1),
136            "duration_ms":track.duration_ms,
137            "analysis_available":c.analyses_of(file).next().is_some(),
138            "lyrics_available":aede_core::lyrics::has_source(Path::new(&file.path), file.first_tag("lyrics"), file.lyrics_path.as_deref().map(Path::new)),
139            "spectrum_available":aede_core::spectrum::picture_for(Path::new(&file.path)).is_file()
140        }))
141    }).collect::<Result<Vec<_>,ApiError>>()?;
142    Ok(
143        json!({"items":items,"total":rows.len(),"offset":offset,"limit":limit,"scanned_at":c.scanned_at}),
144    )
145}
