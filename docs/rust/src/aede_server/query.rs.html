5pub(super) const MAX_SEARCH_TEXT_BYTES: usize = 2048;
6pub(super) const MAX_REFERENCE_BYTES: usize = 16 * 1024;
7
8pub(super) fn bounded_text(value: &str, field: &str, maximum: usize) -> Result<(), ApiError> {
9    if value.len() > maximum {
10        return Err(invalid_query(format!(
11            "{field} must be at most {maximum} bytes"
12        )));
13    }
14    Ok(())
15}
16
17// Stable references include native paths and release keys, so their limit is
18// larger than a name search. Check the prefix without allocating a reference.
19pub(super) fn bounded_selector(value: &str, field: &str) -> Result<(), ApiError> {
20    let maximum = if value
21        .split_once(':')
22        .is_some_and(|(kind, _)| EntityKind::parse_kind(kind).is_some())
23    {
24        MAX_REFERENCE_BYTES
25    } else {
26        MAX_SEARCH_TEXT_BYTES
27    };
28    bounded_text(value, field, maximum)
29}

31pub(super) fn decimal(value: &str, field: &str, pagination: bool) -> Result<usize, ApiError> {
32    if value.is_empty() || !value.bytes().all(|byte| byte.is_ascii_digit()) {
33        let message = format!("{field} must be a non-negative decimal integer");
34        return Err(if pagination {
35            error(StatusCode::BAD_REQUEST, "invalid_pagination", message)
36        } else {
37            invalid_query(message)
38        });
39    }
40    value.parse::<usize>().map_err(|_| {
41        let message = format!("{field} is too large");
42        if pagination {
43            error(StatusCode::BAD_REQUEST, "invalid_pagination", message)
44        } else {
45            invalid_query(message)
46        }
47    })
48}

50pub(super) fn page_bounds(
51    offset: Option<&str>,
52    limit: Option<&str>,
53) -> Result<(usize, usize), ApiError> {
54    let offset = offset
55        .map(|value| decimal(value, "offset", true))
56        .transpose()?
57        .unwrap_or(0);
58    let limit = limit
59        .map(|value| decimal(value, "limit", true))
60        .transpose()?
61        .unwrap_or(DEFAULT_LIMIT);
62    if !(1..=MAX_LIMIT).contains(&limit) {
63        return Err(error(
64            StatusCode::BAD_REQUEST,
65            "invalid_pagination",
66            format!("limit must be between 1 and {MAX_LIMIT}"),
67        ));
68    }
69    Ok((offset, limit))
70}

72pub(super) fn list_query(
73    query: Result<Query<ListQuery>, QueryRejection>,
74    kind: ListKind,
75) -> Result<ListOptions, ApiError> {
76    let Query(query) = query.map_err(|rejection| invalid_query(rejection.body_text()))?;
77    for (field, value) in [("q", &query.q), ("mbid", &query.mbid)] {
78        if let Some(value) = value {
79            bounded_text(value, field, MAX_SEARCH_TEXT_BYTES)?;
80        }
81    }
82    for (field, value) in [
83        ("artist", &query.artist),
84        ("release", &query.release),
85        ("work", &query.work),
86    ] {
87        if let Some(value) = value {
88            bounded_text(value, field, MAX_REFERENCE_BYTES)?;
89        }
90    }
91    let (offset, limit) = page_bounds(query.offset.as_deref(), query.limit.as_deref())?;
92    let q = query
93        .q
94        .map(|value| text::normalize(&value))
95        .map(|normalized| {
96            if normalized.is_empty() {
97                Err(invalid_query("q must contain searchable text"))
98            } else {
99                Ok(normalized)
100            }
101        })
102        .transpose()?;
103    let sort = match query.sort.as_deref().unwrap_or("catalog") {
104        "catalog" => SortMode::Catalog,
105        "name" if kind == ListKind::Artist => SortMode::Name,
106        "title" if kind != ListKind::Artist => SortMode::Title,
107        "year" if kind == ListKind::Release => SortMode::Year,
108        _ => return Err(invalid_query("sort is not supported for this list")),
109    };
110    let descending = match query.order.as_deref().unwrap_or("asc") {
111        "asc" => false,
112        "desc" => true,
113        _ => return Err(invalid_query("order must be asc or desc")),
114    };
115    if query.mbid.is_some() && kind != ListKind::Artist
116        || query.year.is_some() && kind != ListKind::Release
117        || query.artist.is_some() && kind != ListKind::Release
118        || query.release.is_some() && kind != ListKind::Track
119        || query.work.is_some() && kind != ListKind::Recording
120    {
121        return Err(invalid_query("filter is not supported for this list"));
122    }
123    if query.mbid.as_deref() == Some("") {
124        return Err(invalid_query("mbid cannot be empty"));
125    }
126    let year = query
127        .year
128        .as_deref()
129        .map(|value| decimal(value, "year", false))
130        .transpose()?
131        .map(|value| u32::try_from(value).map_err(|_| invalid_query("year is too large")))
132        .transpose()?;
133    Ok(ListOptions {
134        offset,
135        limit,
136        q,
137        sort,
138        descending,
139        mbid: query.mbid,
140        year,
141        artist: query.artist,
142        release: query.release,
143        work: query.work,
144    })
145}
