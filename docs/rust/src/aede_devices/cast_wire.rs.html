1//! Bounded Cast V2 envelope framing; protocol fields follow Chromium CastMessage.
2
3use serde_json::Value;
4use tokio::io::{AsyncRead, AsyncReadExt, AsyncWrite, AsyncWriteExt};
5
6pub(super) const MAX_FRAME: usize = 64 * 1024;
7
8#[derive(Debug)]
9pub(super) struct Message {
10    pub source: String,
11    pub destination: String,
12    pub namespace: String,
13    pub payload: Value,
14}
15
16fn varint(input: &[u8], offset: &mut usize) -> Result<u64, String> {
17    let mut value = 0u64;
18    for shift in (0..70).step_by(7) {
19        let byte = *input.get(*offset).ok_or("truncated Cast varint")?;
20        *offset += 1;
21        if shift == 63 && byte > 1 {
22            return Err("overflowing Cast varint".into());
23        }
24        value |= u64::from(byte & 127) << shift;
25        if byte & 128 == 0 {
26            return Ok(value);
27        }
28    }
29    Err("overflowing Cast varint".into())
30}
31
32fn bytes<'a>(input: &'a [u8], offset: &mut usize) -> Result<&'a [u8], String> {
33    let length = usize::try_from(varint(input, offset)?).map_err(|_| "Cast field too large")?;
34    let end = offset.checked_add(length).ok_or("Cast field too large")?;
35    let value = input.get(*offset..end).ok_or("truncated Cast field")?;
36    *offset = end;
37    Ok(value)
38}
39
40fn text(input: &[u8], maximum: usize) -> Result<String, String> {
41    if input.is_empty() || input.len() > maximum {
42        return Err("invalid Cast text length".into());
43    }
44    let value = std::str::from_utf8(input).map_err(|_| "invalid Cast UTF-8")?;
45    if value.chars().any(char::is_control) {
46        return Err("invalid Cast routing text".into());
47    }
48    Ok(value.into())
49}

51pub(super) fn decode(input: &[u8]) -> Result<Message, String> {
52    if input.is_empty() || input.len() > MAX_FRAME {
53        return Err("Cast frame exceeds its limit".into());
54    }
55    let mut offset = 0;
56    let mut seen = 0u8;
57    let mut source = None;
58    let mut destination = None;
59    let mut namespace = None;
60    let mut payload = None;
61    while offset < input.len() {
62        let tag = varint(input, &mut offset)?;
63        let field = tag >> 3;
64        let wire = tag & 7;
65        if field == 0 || field > 0x1fff_ffff {
66            return Err("invalid Cast field number".into());
67        }
68        if field <= 7 {
69            let bit = 1 << (field - 1);
70            if seen & bit != 0 {
71                return Err("duplicate Cast field".into());
72            }
73            seen |= bit;
74        }
75        match (field, wire) {
76            (1 | 5, 0) => {
77                if varint(input, &mut offset)? != 0 {
78                    return Err("unsupported Cast version or binary payload".into());
79                }
80            }
81            (2, 2) => source = Some(text(bytes(input, &mut offset)?, 128)?),
82            (3, 2) => destination = Some(text(bytes(input, &mut offset)?, 128)?),
83            (4, 2) => namespace = Some(text(bytes(input, &mut offset)?, 256)?),
84            (6, 2) => {
85                let value = serde_json::from_slice::<Value>(bytes(input, &mut offset)?)
86                    .map_err(|_| "invalid Cast JSON payload")?;
87                if !value.is_object() {
88                    return Err("Cast payload must be an object".into());
89                }
90                payload = Some(value);
91            }
92            (1..=7, _) => return Err("invalid Cast field type".into()),
93            (_, 0) => {
94                varint(input, &mut offset)?;
95            }
96            (_, 2) => {
97                bytes(input, &mut offset)?;
98            }
99            (_, 1 | 5) => {
100                let length = if wire == 1 { 8 } else { 4 };
101                let end = offset.checked_add(length).ok_or("Cast field too large")?;
102                input.get(offset..end).ok_or("truncated Cast field")?;
103                offset = end;
104            }
105            _ => return Err("unsupported Cast wire type".into()),
106        }
107    }
108    if seen & 63 != 63 {
109        return Err("missing required Cast field".into());
110    }
111    Ok(Message {
112        source: source.ok_or("missing Cast source")?,
113        destination: destination.ok_or("missing Cast destination")?,
114        namespace: namespace.ok_or("missing Cast namespace")?,
115        payload: payload.ok_or("missing Cast payload")?,
116    })
117}

119fn put_varint(output: &mut Vec<u8>, mut value: u64) {
120    while value >= 128 {
121        output.push((value as u8 & 127) | 128);
122        value >>= 7;
123    }
124    output.push(value as u8);
125}
126
127fn put_text(output: &mut Vec<u8>, field: u8, value: &str) {
128    output.push(field << 3 | 2);
129    put_varint(output, value.len() as u64);
130    output.extend_from_slice(value.as_bytes());
131}

133pub(super) fn encode(message: &Message) -> Result<Vec<u8>, String> {
134    text(message.source.as_bytes(), 128)?;
135    text(message.destination.as_bytes(), 128)?;
136    text(message.namespace.as_bytes(), 256)?;
137    if !message.payload.is_object() {
138        return Err("Cast payload must be an object".into());
139    }
140    let payload = serde_json::to_string(&message.payload).map_err(|error| error.to_string())?;
141    if payload.len() > MAX_FRAME {
142        return Err("Cast payload exceeds its limit".into());
143    }
144    let mut output = vec![8, 0];
145    put_text(&mut output, 2, &message.source);
146    put_text(&mut output, 3, &message.destination);
147    put_text(&mut output, 4, &message.namespace);
148    output.extend_from_slice(&[40, 0]);
149    put_text(&mut output, 6, &payload);
150    if output.len() > MAX_FRAME {
151        return Err("Cast frame exceeds its limit".into());
152    }
153    Ok(output)
154}

156pub(super) async fn read<R: AsyncRead + Unpin>(reader: &mut R) -> Result<Message, String> {
157    let length = reader
158        .read_u32()
159        .await
160        .map_err(|error| format!("Cast read: {error}"))?;
161    if length == 0 || length as u64 > MAX_FRAME as u64 {
162        return Err("Cast frame exceeds its limit".into());
163    }
164    let mut input = vec![0; length as usize];
165    reader
166        .read_exact(&mut input)
167        .await
168        .map_err(|error| format!("Cast read: {error}"))?;
169    decode(&input)
170}
171
172pub(super) async fn write<W: AsyncWrite + Unpin>(
173    writer: &mut W,
174    message: &Message,
175) -> Result<(), String> {
176    let output = encode(message)?;
177    writer
178        .write_u32(output.len() as u32)
179        .await
180        .map_err(|error| format!("Cast write: {error}"))?;
181    writer
182        .write_all(&output)
183        .await
184        .map_err(|error| format!("Cast write: {error}"))?;
185    writer
186        .flush()
187        .await
188        .map_err(|error| format!("Cast flush: {error}"))
189}
