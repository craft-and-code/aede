8/// The short, command-specific part of every help page.

10/// The detailed prose lives in the command where it earns it; this shared

12/// discoverable in the same way without duplicating the option-scope table.

15    pub(crate) summary: &'static str,

256pub fn is_command(command: &str) -> bool {
257    COMMANDS
258        .iter()
259        .any(|(name, _, _)| *name == canonical(command))
260}
261
262/// `help` as a function, so that it sits in the table like every other command
263/// rather than being a special case the table could forget.
264pub fn run(args: &Args) -> commands::Res {
265    let Some(topic) = args.positionals.first() else {
266        print_index();
267        return Ok(());
268    };
269    let command = canonical(topic);
270    if !is_command(command) {
271        return Err(
272            format!("no command named \"{topic}\"; run aede help for the command list").into(),
273        );
274    }
275    print_command(command);
276    Ok(())
277}

1012/// Prints one command's options, operating context and examples.
1013pub fn print_command(command: &str) {
1014    if command == "fetch" {
1015        print_fetch_help();
1016        return;
1017    }
1018    let page = command_page(command);
1019    let alias = COMMANDS
1020        .iter()
1021        .find(|(name, _, _)| *name == command)
1022        .and_then(|(_, alias, _)| *alias);
1023
1024    println!("{}", ui::bold(&format!("aede {command} — command help")));
1025    println!("\n{}\n  {}", ui::cyan("USAGE"), page.usage);
1026    println!("\n{}\n  {}", ui::cyan("DESCRIPTION"), page.summary);
1027    if let Some(alias) = alias {
1028        println!("\n  Also available as: aede {alias}");
1029    }
1030    match command {
1031        "analyze" => {
1032            print_analyze_help();
1033            print_delegation_help(command);
1034        }
1035        "play" => print_play_help(),
1036        "serve" => print_server_help(),
1037        "cancel" => print_cancel_help(),
1038        _ if crate::mutates_store(command) => print_delegation_help(command),
1039        _ => {}
1040    }
1041
1042    let options = OPTION_SCOPE
1043        .iter()
1044        .filter(|(_, commands, _)| commands.contains(&command))
1045        .collect::<Vec<_>>();
1046    if !options.is_empty() {
1047        println!("\n{}", ui::cyan("OPTIONS"));
1048        for (option, _, what) in options {
1049            println!("  --{option:<19} {what}");
1050        }
1051    }
1052    let examples = command_examples(command);
1053    if !examples.is_empty() {
1054        println!("\n{}", ui::cyan("EXAMPLES"));
1055        for example in examples {
1056            println!("  {example}");
1057        }
1058    }
1059    print_global_options(command != "play");
1060}
