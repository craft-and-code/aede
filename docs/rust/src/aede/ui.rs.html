1//! Terminal presentation: colours, aligned tables, bars.
2//!
3//! Two constraints guide this module. First, display width: an accented
4//! character takes several bytes but a single column, so aligning on `len()`
5//! would produce crooked tables. Then restraint: colours disappear as soon as
6//! the output is not a terminal, so that `aede stats > file.txt` stays
7//! readable.
8
9use std::io::IsTerminal;
10use std::sync::atomic::{AtomicBool, Ordering};
11
12static COLOR_ENABLED: AtomicBool = AtomicBool::new(true);

14pub fn init_color(force_off: bool) {
15    let enabled = !force_off && is_interactive() && std::env::var_os("NO_COLOR").is_none();
16    COLOR_ENABLED.store(enabled, Ordering::Relaxed);
17}

19/// `true` when the output goes to a terminal.
20///
21/// A progress line redrawn with a carriage return is meant for a human
22/// watching; piped into a file it turns into thousands of useless lines.
23pub fn is_interactive() -> bool {
24    std::io::stdout().is_terminal()
25        || (std::env::var_os("AEDE_DELEGATED_CHILD").is_some()
26            && std::env::var_os("AEDE_DELEGATED_STDOUT_TTY").is_some())
27}

29/// Shows untrusted text literally, without terminal control instructions.
30///
31/// Paragraph newlines and tabs remain readable. Other C0/C1 controls, including
32/// ESC, BEL and carriage return, are escaped; storage and machine exports keep
33/// the original text. Apply this before adding Aède's own colour sequences.
34pub fn literal(text: &str) -> String {
35    let mut out = String::with_capacity(text.len());
36    for ch in text.chars() {
37        if ch.is_control() && !matches!(ch, '\n' | '\t') {
38            out.extend(ch.escape_default());
39        } else {
40            out.push(ch);
41        }
42    }
43    out
44}

46/// Animate a temporary line while a blocking operation runs. The operation
47/// must not print until it returns; redirected output gets no animation.
48pub fn with_loading<T>(message: &str, operation: impl FnOnce() -> T) -> T {
49    with_loading_output(std::io::stdout(), is_interactive(), message, operation)
50}
51
52fn with_loading_output<T>(
53    mut output: impl std::io::Write + Send,
54    interactive: bool,
55    message: &str,
56    operation: impl FnOnce() -> T,
57) -> T {
58    if !interactive {
59        return operation();
60    }
61    // Flush before starting the request, so even a short wait is visible.
62    if write!(output, "\r  | {message}")
63        .and_then(|_| output.flush())
64        .is_err()
65    {
66        return operation();
67    }
68    std::thread::scope(|scope| {
69        let (stop, stopped) = std::sync::mpsc::channel::<()>();
70        scope.spawn(move || {
71            let frames = ['/', '-', '\\', '|'];
72            let mut frame = 0;
73            while matches!(
74                stopped.recv_timeout(std::time::Duration::from_millis(100)),
75                Err(std::sync::mpsc::RecvTimeoutError::Timeout)
76            ) {
77                if write!(output, "\r  {} {message}", frames[frame])
78                    .and_then(|_| output.flush())
79                    .is_err()
80                {
81                    break;
82                }
83                frame = (frame + 1) % frames.len();
84            }
85            // Presentation is best effort if the terminal has gone away.
86            let _ = write!(output, "\r{}\r", " ".repeat(display_width(message) + 4))
87                .and_then(|_| output.flush());
88        });
89        let result = operation();
90        // Disconnecting also stops and clears the line when operation unwinds.
91        drop(stop);
92        result
93    })
94}

96fn colorize(code: &str, text: &str) -> String {
97    let enabled = COLOR_ENABLED.load(Ordering::Relaxed);
98    let text = literal_styled_with_color(text, enabled);
99    if enabled {
100        format!("\x1b[{code}m{text}\x1b[0m")
101    } else {
102        text
103    }
104}

106// Keep only the handful of SGR colour/style sequences produced by this module,
107// so nested styling still works. OSC, cursor movement, erase and C1 controls
108// remain literal even in a pre-coloured table cell.
109fn literal_styled(text: &str) -> String {
110    literal_styled_with_color(text, COLOR_ENABLED.load(Ordering::Relaxed))
111}
112
113fn literal_styled_with_color(text: &str, enabled: bool) -> String {
114    let mut parts = text.split('\x1b');
115    let mut out = literal(parts.next().unwrap_or_default());
116    for part in parts {
117        if let Some(prefix) = ["[0m", "[1m", "[2m", "[31m", "[32m", "[33m", "[36m"]
118            .into_iter()
119            .find(|prefix| part.starts_with(prefix))
120        {
121            if enabled {
122                out.push('\x1b');
123                out.push_str(prefix);
124            }
125            out.push_str(&literal(&part[prefix.len()..]));
126        } else {
127            out.push_str("\\u{1b}");
128            out.push_str(&literal(part));
129        }
130    }
131    out
132}

153/// Approximate display width of a string, in terminal columns.
154///
155/// We count Unicode characters, not bytes, and give two columns to ideographs
156/// and emoji. That is enough to align artist names.
157pub fn display_width(text: &str) -> usize {
158    text.chars()
159        .filter(|c| !is_combining(*c))
160        .map(|c| if is_wide(c) { 2 } else { 1 })
161        .sum()
162}
163
164fn is_combining(c: char) -> bool {
165    matches!(c as u32, 0x0300..=0x036F | 0x1AB0..=0x1AFF | 0x20D0..=0x20FF)
166}
167
168fn is_wide(c: char) -> bool {
169    matches!(
170        c as u32,
171        0x1100..=0x115F
172            | 0x2E80..=0xA4CF
173            | 0xAC00..=0xD7A3
174            | 0xF900..=0xFAFF
175            | 0xFF00..=0xFF60
176            | 0xFFE0..=0xFFE6
177            | 0x1F300..=0x1FAFF
178    )
179}

181/// Truncates to `max` columns, adding an ellipsis when needed.
182pub fn truncate(text: &str, max: usize) -> String {
183    if display_width(text) <= max || max == 0 {
184        return text.to_string();
185    }
186    let mut out = String::new();
187    let mut width = 0;
188    for c in text.chars() {
189        let w = if is_wide(c) { 2 } else { 1 };
190        if width + w > max.saturating_sub(1) {
191            break;
192        }
193        out.push(c);
194        width += w;
195    }
196    out.push('…');
197    out
198}

200/// Breaks a paragraph into lines of at most `max` columns.
201///
202/// Everything else in this file shortens text to fit a column, because a table
203/// cell that grows breaks the table. A paragraph is the one thing here that is
204/// meant to be *read* rather than scanned, and shortening it to 34 characters
205/// would not be a summary but a tease — so it is wrapped instead.
206///
207/// A word longer than the width is left alone rather than cut: a URL split
208/// across two lines is a URL nobody can click, and one long line is the lesser
209/// harm.
210pub fn wrap(text: &str, max: usize) -> Vec<String> {
211    if max == 0 {
212        return vec![text.to_string()];
213    }
214    let mut lines: Vec<String> = Vec::new();
215    let mut line = String::new();
216    let mut width = 0;
217    for word in text.split_whitespace() {
218        let w = display_width(word);
219        // `+ 1` for the space this word would need after a non-empty line.
220        if width > 0 && width + 1 + w > max {
221            lines.push(std::mem::take(&mut line));
222            width = 0;
223        }
224        if width > 0 {
225            line.push(' ');
226            width += 1;
227        }
228        line.push_str(word);
229        width += w;
230    }
231    if !line.is_empty() {
232        lines.push(line);
233    }
234    lines
235}

237/// Truncates to `max` columns by dropping the **start** of the text.
238///
239/// For a file path, the end is what identifies the file. Cutting the tail off
240/// `/temporary/cache/with/a/long/generated/directory/name/bad.flac` leaves a
241/// row that names no file at all.
242pub fn truncate_start(text: &str, max: usize) -> String {
243    if display_width(text) <= max || max == 0 {
244        return text.to_string();
245    }
246    let mut tail: Vec<char> = Vec::new();
247    let mut width = 0;
248    for c in text.chars().rev() {
249        let w = if is_wide(c) { 2 } else { 1 };
250        if width + w > max.saturating_sub(1) {
251            break;
252        }
253        tail.push(c);
254        width += w;
255    }
256    let mut out = String::from("…");
257    out.extend(tail.into_iter().rev());
258    out
259}

261/// Pads on the right up to `width` columns.
262pub fn pad(text: &str, width: usize) -> String {
263    let current = display_width(text);
264    if current >= width {
265        text.to_string()
266    } else {
267        format!("{text}{}", " ".repeat(width - current))
268    }
269}
270
271/// Pads on the left up to `width` columns.
272pub fn pad_left(text: &str, width: usize) -> String {
273    let current = display_width(text);
274    if current >= width {
275        text.to_string()
276    } else {
277        format!("{}{text}", " ".repeat(width - current))
278    }
279}

281/// Column alignment.
282#[derive(Debug, Clone, Copy, PartialEq)]
283pub enum Align {
284    Left,
285    Right,
286}

288/// Aligned table, with an underlined header.
289pub struct Table {
290    headers: Vec<String>,
291    aligns: Vec<Align>,
292    /// Maximum width per column; 0 = unlimited.
293    limits: Vec<usize>,
294    /// Columns whose overflow is cut from the start rather than from the end.
295    keep_end: Vec<bool>,
296    rows: Vec<Vec<String>>,
297    show_header: bool,
298}
299
300impl Table {
301    pub fn new(headers: &[&str]) -> Self {
302        Table {
303            headers: headers.iter().map(|h| h.to_string()).collect(),
304            aligns: vec![Align::Left; headers.len()],
305            limits: vec![0; headers.len()],
306            keep_end: vec![false; headers.len()],
307            rows: Vec::new(),
308            show_header: true,
309        }
310    }
311
312    /// Table of values without a header row: for the "label → value" cards,
313    /// where an empty header would be noise.
314    pub fn plain(columns: usize) -> Self {
315        let mut table = Table::new(&vec![""; columns]);
316        table.show_header = false;
317        table
318    }
319
320    pub fn align(mut self, index: usize, align: Align) -> Self {
321        if let Some(slot) = self.aligns.get_mut(index) {
322            *slot = align;
323        }
324        self
325    }
326
327    pub fn limit(mut self, index: usize, max: usize) -> Self {
328        if let Some(slot) = self.limits.get_mut(index) {
329            *slot = max;
330        }
331        self
332    }
333
334    /// Bounds a column like [`Table::limit`], but keeps its **end** when it
335    /// overflows. For paths, where the file name is the informative part.
336    pub fn path_limit(mut self, index: usize, max: usize) -> Self {
337        if let Some(slot) = self.limits.get_mut(index) {
338            *slot = max;
339        }
340        if let Some(slot) = self.keep_end.get_mut(index) {
341            *slot = true;
342        }
343        self
344    }
345
346    pub fn push(&mut self, row: Vec<String>) {
347        self.rows.push(row);
348    }
349
350    pub fn is_empty(&self) -> bool {
351        self.rows.is_empty()
352    }
353
354    pub fn render(&self) -> String {
355        if self.rows.is_empty() {
356            return dim("  (no results)\n");
357        }
358        let columns = self.headers.len();
359        let mut widths: Vec<usize> = self.headers.iter().map(|h| display_width(h)).collect();
360
361        let cells: Vec<Vec<String>> = self
362            .rows
363            .iter()
364            .map(|row| {
365                row.iter()
366                    .enumerate()
367                    .map(|(i, cell)| {
368                        let cell = literal_styled(cell).replace(['\n', '\t'], " ");
369                        let limit = self.limits.get(i).copied().unwrap_or(0);
370                        match (limit > 0, self.keep_end.get(i).copied().unwrap_or(false)) {
371                            (true, true) => truncate_start(&cell, limit),
372                            (true, false) => truncate(&cell, limit),
373                            (false, _) => cell,
374                        }
375                    })
376                    .collect()
377            })
378            .collect();
379
380        for row in &cells {
381            for (i, cell) in row.iter().enumerate().take(columns) {
382                widths[i] = widths[i].max(display_width(cell));
383            }
384        }
385
386        let mut out = String::new();
387        if !self.show_header {
388            for row in &cells {
389                self.render_row(&mut out, row, &widths);
390            }
391            return out;
392        }
393        // Header
394        out.push_str("  ");
395        for (i, header) in self.headers.iter().enumerate() {
396            let text = match self.aligns[i] {
397                Align::Left => pad(header, widths[i]),
398                Align::Right => pad_left(header, widths[i]),
399            };
400            out.push_str(&bold(&text));
401            if i + 1 < columns {
402                out.push_str("  ");
403            }
404        }
405        out.push('\n');
406        // Rule
407        out.push_str("  ");
408        let rule: Vec<String> = widths.iter().map(|w| "─".repeat(*w)).collect();
409        out.push_str(&dim(&rule.join("  ")));
410        out.push('\n');
411        // Rows
412        for row in &cells {
413            self.render_row(&mut out, row, &widths);
414        }
415        out
416    }
417
418    /// Renders a data row, column by column, without trailing spaces (they
419    /// pollute copy-paste and test comparisons).
420    fn render_row(&self, out: &mut String, row: &[String], widths: &[usize]) {
421        out.push_str("  ");
422        let last = widths.len().saturating_sub(1);
423        for (i, width) in widths.iter().enumerate() {
424            let cell = row.get(i).map(String::as_str).unwrap_or("");
425            let text = match self.aligns[i] {
426                Align::Left => pad(cell, *width),
427                Align::Right => pad_left(cell, *width),
428            };
429            out.push_str(&text);
430            if i < last {
431                out.push_str("  ");
432            }
433        }
434        while out.ends_with(' ') {
435            out.pop();
436        }
437        out.push('\n');
438    }
439}

441/// Proportional horizontal bar, for breakdowns.
442pub fn bar(value: usize, max: usize, width: usize) -> String {
443    if max == 0 || width == 0 {
444        return String::new();
445    }
446    let filled = ((value * width + max / 2) / max).min(width);
447    let rest = width - filled;
448    // No escape sequence for an empty portion: it would pollute comparisons
449    // and redirected output.
450    if rest == 0 {
451        "█".repeat(filled)
452    } else {
453        format!("{}{}", "█".repeat(filled), dim(&"·".repeat(rest)))
454    }
455}

457/// Pluralises a noun: `plural(1, "album")` → "1 album".
458///
459/// Only the two endings the program actually uses are special-cased — `-is`
460/// becoming `-es` (analysis → analyses) and the sibilants that take `-es`. A
461/// full English pluraliser would be a library; "1 analysiss" is a typo the
462/// user reads every time.
463/// How long ago a moment was, in one short phrase.
464///
465/// **Takes the moment, not the age**, and that is the whole reason it exists
466/// beside [`ago`]. Ten call sites wrote `ago(now_seconds() - at)` and the
467/// eleventh wrote `ago(at)`, which reads perfectly and says "56 years ago" for
468/// something that happened a second before — a Unix timestamp is a plausible
469/// number of seconds. A signature two callers can read two ways is a trap
470/// however well documented, so the arithmetic lives here, once.
471///
472/// The clock is read here rather than inside [`ago`] so that the part with
473/// rules in it stays a pure function of its argument, testable without a clock.
474pub fn since(at: u64) -> String {
475    ago(aede_core::clock::now_seconds().saturating_sub(at))
476}

478/// How long an elapsed span is, in one short phrase.
479///
480/// Coarse on purpose: a history is read for its order, not for its arithmetic,
481/// and "3 days ago" is easier to place than a date nobody remembers.
482///
483/// Takes a **duration in seconds**. Callers holding a moment want [`since`].
484pub fn ago(seconds: u64) -> String {
485    const MINUTE: u64 = 60;
486    const HOUR: u64 = 60 * MINUTE;
487    const DAY: u64 = 24 * HOUR;
488    match seconds {
489        s if s < MINUTE => "just now".into(),
490        s if s < HOUR => format!("{} min ago", s / MINUTE),
491        s if s < DAY => plural((s / HOUR) as usize, "hour") + " ago",
492        s if s < 30 * DAY => plural((s / DAY) as usize, "day") + " ago",
493        s if s < 365 * DAY => plural((s / (30 * DAY)) as usize, "month") + " ago",
494        s => plural((s / (365 * DAY)) as usize, "year") + " ago",
495    }
496}

518/// A measured elapsed time, in the largest unit that stays readable.
519///
520/// Milliseconds below a second, seconds below a minute, minutes beyond: a scan
521/// that reports `260604 ms` makes the reader do the division.
522pub fn elapsed(ms: u128) -> String {
523    if ms < 1_000 {
524        return format!("{ms} ms");
525    }
526    if ms < 60_000 {
527        return format!("{:.1} s", ms as f64 / 1000.0);
528    }
529    let seconds = ms / 1000 + u128::from(ms % 1000 >= 500);
530    let (minutes, rest) = (seconds / 60, seconds % 60);
531    if minutes < 60 {
532        format!("{minutes} min {rest} s")
533    } else {
534        format!("{} h {} min", minutes / 60, minutes % 60)
535    }
536}

538/// Long duration, in days/hours/minutes — and in seconds below a minute, so a
539/// small library does not report "0 min".
540pub fn long_duration(ms: u64) -> String {
541    // Rounded like `text::format_duration`, so the two never disagree by a
542    // second on the same value.
543    let total_sec = ms / 1000 + u64::from(ms % 1000 >= 500);
544    let total_min = total_sec / 60;
545    let days = total_min / 1440;
546    let hours = (total_min % 1440) / 60;
547    let minutes = total_min % 60;
548    if days > 0 {
549        format!("{days} d {hours} h {minutes} min")
550    } else if hours > 0 {
551        format!("{hours} h {minutes} min")
552    } else if total_min > 0 {
553        format!("{minutes} min {} s", total_sec % 60)
554    } else {
555        format!("{total_sec} s")
556    }
557}
