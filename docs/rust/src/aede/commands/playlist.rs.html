1//! The `playlist` command: an `.m3u` in every album folder.
2//!
3//! Same shape as `spectrum`, and for the same reason: the catalog already
4//! knows what an album is, which tracks it holds and in what order, and a
5//! second answer read off the filesystem would drift from the first. The
6//! folders on the command line narrow it; they do not replace it.
7
8use std::collections::BTreeMap;
9use std::path::PathBuf;
10
11use aede_core::model::{Catalog, Id};
12use aede_core::playlist::{self, Style};
13
14use super::{Res, load};
15use crate::args::Args;
16use crate::ui::{self, Align, Table};
17
18pub fn playlist(args: &Args) -> Res {
19    let catalog = load(args)?;
20    let scope = super::scope_of(args, &catalog)?;
21    let style = match args.has("simple") {
22        true => Style::Simple,
23        false => Style::Extended,
24    };
25
26    let mut wanted: Vec<(PathBuf, String)> = albums(&catalog, &scope, style);
27    if args.has("artists") {
28        wanted.extend(discographies(&catalog, &scope, style));
29    }
30
31    println!("{}", ui::section("Playlists"));
32    if wanted.is_empty() {
33        println!(
34            "  {}",
35            ui::yellow("no album of the catalog is in that folder")
36        );
37        return Ok(());
38    }
39
40    // Split before writing anything, so the count on screen is the work and
41    // not the library — and so a --dry-run says the same thing the real run
42    // will do rather than something close to it.
43    let (to_write, current): (Vec<_>, Vec<_>) = wanted
44        .into_iter()
45        .partition(|(path, text)| !playlist::already_says(path, text));
46
47    if args.has("dry-run") {
48        let mut t = Table::new(&["Playlist", "Tracks"]).align(1, Align::Right);
49        for (path, text) in to_write.iter().take(20) {
50            t.push(vec![path.display().to_string(), lines(text).to_string()]);
51        }
52        print!("{}", t.render());
53        if to_write.len() > 20 {
54            println!(
55                "{}",
56                ui::dim(&format!("  … and {} more", to_write.len() - 20))
57            );
58        }
59        summarise(to_write.len(), 0, current.len(), &[]);
60        return Ok(());
61    }
62
63    let mut written = 0usize;
64    let mut failures: Vec<(String, String)> = Vec::new();
65    for (path, text) in &to_write {
66        match std::fs::write(path, text) {
67            Ok(()) => written += 1,
68            Err(e) => failures.push((path.display().to_string(), e.to_string())),
69        }
70    }
71    summarise(0, written, current.len(), &failures);
72    Ok(())
73}
74
75fn summarise(planned: usize, written: usize, current: usize, failures: &[(String, String)]) {
76    let mut t = Table::plain(2).align(1, Align::Right);
77    if planned > 0 {
78        t.push(vec!["To write".into(), planned.to_string()]);
79    }
80    if written > 0 || planned == 0 {
81        t.push(vec!["Written".into(), written.to_string()]);
82    }
83    // Said even when it is the whole answer: a second run over an unchanged
84    // library writes nothing, and an empty screen there reads as a failure
85    // rather than as "there was nothing to do".
86    t.push(vec!["Already up to date".into(), current.to_string()]);
87    if !failures.is_empty() {
88        t.push(vec!["Failed".into(), failures.len().to_string()]);
89    }
90    print!("{}", t.render());
91    if failures.is_empty() {
92        return;
93    }
94    super::print_failure_table("What could not be written", "Playlist", failures);
95}

97/// Tracks in a rendered playlist, for the dry run's column.
98fn lines(text: &str) -> usize {
99    text.lines()
100        .filter(|l| !l.is_empty() && !l.starts_with('#'))
101        .count()
102}

104/// One playlist per album folder, holding that album in its own order.
105fn albums(catalog: &Catalog, scope: &[String], style: Style) -> Vec<(PathBuf, String)> {
106    let mut out = Vec::new();
107    for release in &catalog.releases {
108        if !super::in_scope(&release.folder, scope) {
109            continue;
110        }
111        // A box set laid out as `Album/Disc 1`, `Album/Disc 2` is *one*
112        // release living in the parent, so its playlist goes there and spans
113        // the discs — which is exactly the file somebody wants when the tracks
114        // are numbered 1..17 twice over.
115        let folder = PathBuf::from(&release.folder);
116        let text = playlist::render(catalog, &release.track_ids, Some(&folder), style);
117        if lines(&text) == 0 {
118            continue;
119        }
120        out.push((folder.join(playlist::file_name(&folder)), text));
121    }
122    out
123}

125/// One playlist per artist folder, holding every album of that artist in order.
126///
127/// The artist folder is not a thing the catalog holds — it is inferred as the
128/// folder every one of that artist's albums sits in. Where they do not share
129/// one, or share only a watched root, nothing is written: a library laid out
130/// flat would otherwise get one playlist per artist dumped in its root, which
131/// is not tidying but littering.
132fn discographies(catalog: &Catalog, scope: &[String], style: Style) -> Vec<(PathBuf, String)> {
133    let mut by_artist: BTreeMap<Id, Vec<&aede_core::model::Release>> = BTreeMap::new();
134    for release in &catalog.releases {
135        if let Some(artist) = release.album_artist_id
136            && super::in_scope(&release.folder, scope)
137        {
138            by_artist.entry(artist).or_default().push(release);
139        }
140    }
141
142    let roots: Vec<&str> = catalog.roots.iter().map(String::as_str).collect();
143    let mut out = Vec::new();
144    for (_, mut releases) in by_artist {
145        let Some(folder) = super::shared_folder(&releases) else {
146            continue;
147        };
148        let name = folder.to_string_lossy().to_string();
149        // The root of a library is not an artist folder, and neither is a
150        // folder that *is* one of the album folders — that one already has its
151        // own playlist under the very same name.
152        if roots.contains(&name.as_str()) || releases.iter().any(|r| r.folder == name) {
153            continue;
154        }
155        // Chronological, which is the order a discography is read in; the
156        // title breaks a tie so that two records of the same year come out the
157        // same way on every machine.
158        releases.sort_by(|a, b| {
159            a.year
160                .unwrap_or(0)
161                .cmp(&b.year.unwrap_or(0))
162                .then_with(|| a.title.cmp(&b.title))
163        });
164        let tracks: Vec<Id> = releases.iter().flat_map(|r| r.track_ids.clone()).collect();
165        let text = playlist::render(catalog, &tracks, Some(&folder), style);
166        if lines(&text) == 0 {
167            continue;
168        }
169        out.push((folder.join(playlist::file_name(&folder)), text));
170    }
171    out
172}
