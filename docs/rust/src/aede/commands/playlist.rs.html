1//! The `playlist` command: an `.m3u` in every album folder.
2//!
3//! Same shape as `spectrum`, and for the same reason: the catalog already
4//! knows what an album is, which tracks it holds and in what order, and a
5//! second answer read off the filesystem would drift from the first. The
6//! folders on the command line narrow it; they do not replace it.
7
8use std::collections::BTreeMap;
9use std::path::PathBuf;
10
11use aede_core::model::{Catalog, Id};
12use aede_core::playlist::{self, Style};

14use super::{Res, load};
15use crate::args::Args;
16use crate::ui::{self, Align, Table};

18pub fn playlist(args: &Args) -> Res {
19    let catalog = load(args)?;
20    let scope = super::scope_of(args, &catalog)?;
21    let style = match args.has("simple") {
22        true => Style::Simple,
23        false => Style::Extended,
24    };
25
26    let mut folders = albums(&catalog, &scope);
27    if args.has("artists") {
28        for (folder, tracks) in discographies(&catalog, &scope) {
29            folders.entry(folder).or_default().extend(tracks);
30        }
31    }
32    let wanted = render_folders(&catalog, folders, style)?;
33
34    println!("{}", ui::section("Playlists"));
35    if wanted.is_empty() {
36        println!(
37            "  {}",
38            ui::yellow("no album of the catalog is in that folder")
39        );
40        return Ok(());
41    }
42
43    // Split before writing anything, so the count on screen is the work and
44    // not the library — and so a --dry-run says the same thing the real run
45    // will do rather than something close to it.
46    let (to_write, current): (Vec<_>, Vec<_>) = wanted
47        .into_iter()
48        .partition(|(path, text)| !playlist::already_says(path, text));
49
50    if args.has("dry-run") {
51        let mut t = Table::new(&["Playlist", "Tracks"]).align(1, Align::Right);
52        for (path, text) in to_write.iter().take(20) {
53            t.push(vec![path.display().to_string(), lines(text).to_string()]);
54        }
55        print!("{}", t.render());
56        if to_write.len() > 20 {
57            println!(
58                "{}",
59                ui::dim(&format!("  … and {} more", to_write.len() - 20))
60            );
61        }
62        summarise(to_write.len(), 0, current.len(), &[]);
63        return Ok(());
64    }
65
66    let mut written = 0usize;
67    let mut failures: Vec<(String, String)> = Vec::new();
68    for (path, text) in &to_write {
69        match playlist::write_atomic(path, text) {
70            Ok(()) => written += 1,
71            Err(e) => failures.push((path.display().to_string(), e.to_string())),
72        }
73    }
74    summarise(0, written, current.len(), &failures);
75    if failures.is_empty() {
76        Ok(())
77    } else {
78        Err(format!("{} playlist(s) could not be written", failures.len()).into())
79    }
80}

86fn summarise(planned: usize, written: usize, current: usize, failures: &[(String, String)]) {
87    let mut t = Table::plain(2).align(1, Align::Right);
88    if planned > 0 {
89        t.push(vec!["To write".into(), planned.to_string()]);
90    }
91    if written > 0 || planned == 0 {
92        t.push(vec!["Written".into(), written.to_string()]);
93    }
94    // Said even when it is the whole answer: a second run over an unchanged
95    // library writes nothing, and an empty screen there reads as a failure
96    // rather than as "there was nothing to do".
97    t.push(vec!["Already up to date".into(), current.to_string()]);
98    if !failures.is_empty() {
99        t.push(vec!["Failed".into(), failures.len().to_string()]);
100    }
101    print!("{}", t.render());
102    if failures.is_empty() {
103        return;
104    }
105    super::print_failure_table("What could not be written", "Playlist", failures);
106}

108/// Tracks in a rendered playlist, for the dry run's column.
109fn lines(text: &str) -> usize {
110    text.lines()
111        .filter(|l| !l.is_empty() && !l.starts_with('#'))
112        .count()
113}

115/// One playlist per album folder, holding that album in its own order.
116fn albums(catalog: &Catalog, scope: &[String]) -> BTreeMap<PathBuf, Vec<Id>> {
117    let mut by_folder: BTreeMap<PathBuf, Vec<Id>> = BTreeMap::new();
118    for release in &catalog.releases {
119        if !super::in_scope(&release.folder, scope) {
120            continue;
121        }
122        // A box set laid out as `Album/Disc 1`, `Album/Disc 2` is *one*
123        // release living in the parent, so its playlist goes there and spans
124        // the discs — which is exactly the file somebody wants when the tracks
125        // are numbered 1..17 twice over.
126        let folder = PathBuf::from(&release.folder);
127        by_folder
128            .entry(folder)
129            .or_default()
130            .extend(&release.track_ids);
131    }
132    by_folder
133}

135fn render_folders(
136    catalog: &Catalog,
137    folders: BTreeMap<PathBuf, Vec<Id>>,
138    style: Style,
139) -> Result<Vec<(PathBuf, String)>, String> {
140    let mut out = Vec::new();
141    for (folder, mut tracks) in folders {
142        let mut seen = std::collections::BTreeSet::new();
143        tracks.retain(|id| seen.insert(*id));
144        let text = playlist::try_render(catalog, &tracks, Some(&folder), style)?;
145        if lines(&text) == 0 {
146            continue;
147        }
148        out.push((folder.join(playlist::file_name(&folder)), text));
149    }
150    Ok(out)
151}

153/// One playlist per artist folder, holding every album of that artist in order.
154///
155/// The artist folder is not a thing the catalog holds — it is inferred as the
156/// folder every one of that artist's albums sits in. Where they do not share
157/// one, or share only a watched root, nothing is written: a library laid out
158/// flat would otherwise get one playlist per artist dumped in its root, which
159/// is not tidying but littering.
160fn discographies(catalog: &Catalog, scope: &[String]) -> BTreeMap<PathBuf, Vec<Id>> {
161    let mut by_artist: BTreeMap<Id, Vec<&aede_core::model::Release>> = BTreeMap::new();
162    for release in &catalog.releases {
163        if let Some(artist) = release.album_artist_id
164            && super::in_scope(&release.folder, scope)
165        {
166            by_artist.entry(artist).or_default().push(release);
167        }
168    }
169
170    let roots: Vec<&str> = catalog.roots.iter().map(String::as_str).collect();
171    let mut by_folder: BTreeMap<PathBuf, Vec<&aede_core::model::Release>> = BTreeMap::new();
172    for (_, releases) in by_artist {
173        let Some(folder) = super::shared_folder(&releases) else {
174            continue;
175        };
176        let name = folder.to_string_lossy().to_string();
177        // The root of a library is not an artist folder, and neither is a
178        // folder that *is* one of the album folders — that one already has its
179        // own playlist under the very same name.
180        if roots.contains(&name.as_str()) || releases.iter().any(|r| r.folder == name) {
181            continue;
182        }
183        by_folder.entry(folder).or_default().extend(releases);
184    }
185    by_folder
186        .into_iter()
187        .map(|(folder, mut releases)| {
188            // Chronological, which is the order a discography is read in; the
189            // title breaks a tie so that two records of the same year come out the
190            // same way on every machine.
191            releases.sort_by(|a, b| {
192                a.year
193                    .unwrap_or(0)
194                    .cmp(&b.year.unwrap_or(0))
195                    .then_with(|| a.title.cmp(&b.title))
196            });
197            let tracks: Vec<Id> = releases.iter().flat_map(|r| r.track_ids.clone()).collect();
198            (folder, tracks)
199        })
200        .collect()
201}
