1//! The `playlist` command: an `.m3u` in every album folder.
2//!
3//! Same shape as `spectrum`, and for the same reason: the catalog already
4//! knows what an album is, which tracks it holds and in what order, and a
5//! second answer read off the filesystem would drift from the first. The
6//! folders on the command line narrow it; they do not replace it.
7

17pub fn playlist(args: &Args) -> Res {
18    let catalog = load(args)?;
19    let scope = super::scope_of(args, &catalog)?;
20    let style = match args.has("simple") {
21        true => Style::Simple,
22        false => Style::Extended,
23    };
24
25    let mut folders = albums(&catalog, &scope);
26    if args.has("artists") {
27        for (folder, tracks) in discographies(&catalog, &scope) {
28            folders.entry(folder).or_default().extend(tracks);
29        }
30    }
31    let wanted = render_folders(&catalog, folders, style)?;
32
33    println!("{}", ui::section("Playlists"));
34    if wanted.is_empty() {
35        println!(
36            "  {}",
37            ui::yellow("no album of the catalog is in that folder")
38        );
39        return Ok(());
40    }
41
42    // Split before writing anything, so the count on screen is the work and
43    // not the library — and so a --dry-run says the same thing the real run
44    // will do rather than something close to it.
45    let (to_write, current): (Vec<_>, Vec<_>) = wanted
46        .into_iter()
47        .partition(|(path, text)| !playlist::already_says(path, text));
48
49    if args.has("dry-run") {
50        let mut t = Table::new(&["Playlist", "Tracks"]).align(1, Align::Right);
51        for (path, text) in to_write.iter().take(20) {
52            t.push(vec![path.display().to_string(), lines(text).to_string()]);
53        }
54        print!("{}", t.render());
55        if to_write.len() > 20 {
56            println!(
57                "{}",
58                ui::dim(&format!("  … and {} more", to_write.len() - 20))
59            );
60        }
61        summarise(to_write.len(), 0, current.len(), &[]);
62        return Ok(());
63    }
64
65    let mut written = 0usize;
66    let mut failures: Vec<(String, String)> = Vec::new();
67    for (path, text) in &to_write {
68        // Catalogued folders may be on an unavailable volume. The generic
69        // export writer creates parents, but a library sidecar must not
70        // recreate a missing album/artist directory from retained metadata.
71        let result = if path.parent().is_some_and(std::path::Path::is_dir) {
72            playlist::write_atomic(path, text)
73        } else {
74            Err("playlist folder is unavailable; scan and retry".into())
75        };
76        match result {
77            Ok(()) => written += 1,
78            Err(e) => failures.push((path.display().to_string(), e.to_string())),
79        }
80    }
81    summarise(0, written, current.len(), &failures);
82    if failures.is_empty() {
83        Ok(())
84    } else {
85        Err(format!("{} playlist(s) could not be written", failures.len()).into())
86    }
87}

93fn summarise(planned: usize, written: usize, current: usize, failures: &[(String, String)]) {
94    let mut t = Table::plain(2).align(1, Align::Right);
95    if planned > 0 {
96        t.push(vec!["To write".into(), planned.to_string()]);
97    }
98    if written > 0 || planned == 0 {
99        t.push(vec!["Written".into(), written.to_string()]);
100    }
101    // Said even when it is the whole answer: a second run over an unchanged
102    // library writes nothing, and an empty screen there reads as a failure
103    // rather than as "there was nothing to do".
104    t.push(vec!["Already up to date".into(), current.to_string()]);
105    if !failures.is_empty() {
106        t.push(vec!["Failed".into(), failures.len().to_string()]);
107    }
108    print!("{}", t.render());
109    if failures.is_empty() {
110        return;
111    }
112    super::print_failure_table("What could not be written", "Playlist", failures);
113}

115/// Tracks in a rendered playlist, for the dry run's column.
116fn lines(text: &str) -> usize {
117    text.lines()
118        .filter(|l| !l.is_empty() && !l.starts_with('#'))
119        .count()
120}
