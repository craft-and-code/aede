19/// The one artist a name reaches, or an error naming the several it reaches.
20///
21/// **Several equally good answers are refused rather than arbitrated**, which
22/// is the rule everywhere else in this program and was missing here: the page
23/// matched exactly, then fell back on the fuzzy search and took its *first
24/// hit*. On a shelf holding both `Ozzy Osbourne` and `O. Osbourne`,
25/// `aede artist osbourne` and `aede artist ozzy` answered about two different
26/// people — one with a biography and thirteen albums, the other with seven —
27/// and nothing on screen suggested a choice had been made.
28///
29/// Each line of the refusal carries what tells the two apart, because repeating
30/// the names would repeat the ambiguity: the track count, and the MusicBrainz
31/// identifier where there is one, since two spellings with **different**
32/// identifiers are genuinely two people and two spellings with the same one
33/// should have been merged by the scan.
34fn one_artist<'a>(catalog: &'a Catalog, name: &str) -> Result<&'a Artist, Box<dyn Error>> {
35    if let Some(artist) = catalog
36        .artists
37        .iter()
38        .find(|artist| artist.mbid.as_deref() == Some(name))
39    {
40        return Ok(artist);
41    }
42    let (found, _) = catalog.find_artists(name);
43    match found.as_slice() {
44        [one] => return Ok(one),
45        [] => {
46            // Nothing matched even loosely: the fuzzy search is the last
47            // resort, and it answers about spelling rather than about
48            // substrings — `ozy` reaching Ozzy is what it is for.
49            return catalog
50                .search(name, 1)
51                .first()
52                .filter(|hit| hit.kind == EntityKind::Artist)
53                .and_then(|hit| catalog.artist(hit.id))
54                .ok_or_else(|| format!("no artist matches \"{name}\"").into());
55        }
56        _ => {}
57    }
58    let mut lines = String::new();
59    for artist in &found {
60        // Tracks *and* albums, because either alone can be zero for a real
61        // artist: a name credited only as an album artist performs on nothing,
62        // and a guest performs on tracks and owns no album. "0 track" on its
63        // own reads as an empty row rather than as a distinguishing fact.
64        let mbid = match &artist.mbid {
65            Some(id) => format!(" · musicbrainz {id}"),
66            None => String::new(),
67        };
68        lines.push_str(&format!(
69            "\n\t{} ({}, {}){mbid}",
70            artist.name,
71            crate::ui::plural(catalog.tracks_of_artist(artist.id).len(), "track"),
72            crate::ui::plural(catalog.releases_of_artist(artist.id).len(), "album")
73        ));
74    }
75    Err(format!(
76        "\"{name}\" matches {}, and a page is about exactly one.\n\
77         Name the one you mean:{lines}",
78        crate::ui::plural(found.len(), "artist")
79    )
80    .into())
81}

83enum ArtistPage<'a> {
84    Local(&'a Artist),
85    Sourced(&'a SourcedContributor),
86}
87
88fn resolve_artist<'a>(
89    catalog: &'a Catalog,
90    sourced: &'a [SourcedContributor],
91    query: &str,
92) -> Result<ArtistPage<'a>, Box<dyn Error>> {
93    if let Some(artist) = catalog
94        .artists
95        .iter()
96        .find(|artist| artist.mbid.as_deref() == Some(query))
97    {
98        return Ok(ArtistPage::Local(artist));
99    }
100    if let Some(artist) = sourced.iter().find(|artist| artist.mbid == query) {
101        return Ok(ArtistPage::Sourced(artist));
102    }
103    let (local, match_kind) = catalog.find_artists(query);
104    // A name that already opened a local artist must keep doing so. A source
105    // credit with the same spelling is a separate possible identity until
106    // tags establish its MBID; its exact ID remains directly openable.
107    if match_kind == aede_core::model::TitleMatch::Exact && !local.is_empty() {
108        return one_artist(catalog, query).map(ArtistPage::Local);
109    }
110    let exact: Vec<_> = sourced
111        .iter()
112        .filter(|artist| contributors::matches_name(artist, query, true))
113        .collect();
114    let candidates = if !exact.is_empty() {
115        exact
116            .into_iter()
117            .map(ArtistPage::Sourced)
118            .collect::<Vec<_>>()
119    } else {
120        let partial: Vec<_> = sourced
121            .iter()
122            .filter(|artist| contributors::matches_name(artist, query, false))
123            .collect();
124        if partial.is_empty() {
125            return one_artist(catalog, query).map(ArtistPage::Local);
126        }
127        local
128            .iter()
129            .copied()
130            .map(ArtistPage::Local)
131            .chain(partial.into_iter().map(ArtistPage::Sourced))
132            .collect::<Vec<_>>()
133    };
134    match candidates.as_slice() {
135        [one] => Ok(match one {
136            ArtistPage::Local(artist) => ArtistPage::Local(artist),
137            ArtistPage::Sourced(artist) => ArtistPage::Sourced(artist),
138        }),
139        [] => one_artist(catalog, query).map(ArtistPage::Local),
140        _ => {
141            let mut names = Vec::new();
142            for candidate in candidates {
143                match candidate {
144                    ArtistPage::Local(artist) => names.push(format!(
145                        "{} · {} · local tags",
146                        artist.name,
147                        artist.mbid.as_deref().unwrap_or("no MusicBrainz ID")
148                    )),
149                    ArtistPage::Sourced(artist) => names.push(format!(
150                        "{} · {} · sourced credits",
151                        artist.name, artist.mbid
152                    )),
153                }
154            }
155            Err(format!(
156                "\"{query}\" matches several artists; use a MusicBrainz ID:\n  {}",
157                names.join("\n  ")
158            )
159            .into())
160        }
161    }
162}

164pub fn show_artist(args: &Args) -> Res {
165    let catalog = load(args)?;
166    let name = args.positionals.join(" ");
167    if name.trim().is_empty() {
168        return Err("give a name: aede artist \"Miles Davis\"".into());
169    }
170    let window = args.window(50)?;
171    let questions: Vec<_> = ["with", "members", "role"]
172        .into_iter()
173        .filter(|option| args.has(option))
174        .collect();
175    if questions.len() > 1 {
176        return Err(format!(
177            "choose one artist question: {} cannot be combined",
178            questions
179                .iter()
180                .map(|option| format!("--{option}"))
181                .collect::<Vec<_>>()
182                .join(", ")
183        )
184        .into());
185    }
186    if args.has("members") && !["csv", "json", "m3u"].iter().any(|option| args.has(option)) {
187        super::refuse_output_without_a_format(args)?;
188    }
189    if args.has("members")
190        && ["csv", "json", "m3u", "output", "limit", "offset", "all"]
191            .iter()
192            .any(|option| args.has(option))
193    {
194        return Err(
195            "--members is a dated membership page; export and pagination options are not supported"
196                .into(),
197        );
198    }
199    let held = super::sources_held(args)?;
200    let contributors = contributors::sourced(&catalog, &held);
201    let artist = match resolve_artist(&catalog, &contributors, &name)? {
202        ArtistPage::Local(artist) => artist,
203        ArtistPage::Sourced(artist) => {
204            return super::sourced_artist::show(args, &catalog, &held, artist);
205        }
206    };
207
208    // `--with` turns one line of the collaboration table into the tracks it
209    // counts: the graph is only useful if one can walk down it.
210    if let Some(wanted) = args.value("with") {
211        return print_tracks_in_common(args, &catalog, artist.id, wanted);
212    }
213
214    // `--members` is the line-up, and it is its own answer rather than a
215    // panel on the page: a band's history is a table with dates in it, and a
216    // person's is the same table read from the other end.
217    if args.has("members") {
218        return print_members(args, &catalog, artist);
219    }
220
221    // `--role` narrows the page to what this person did *in that role*. The
222    // page below already separates performing from writing; this goes one step
223    // finer, and is the only way to ask "what did Ozzy sing on" as opposed to
224    // "what is Ozzy on at all".
225    if let Some(role) = args.value("role") {
226        return print_tracks_in_role(&catalog, artist.id, role, args);
227    }
228
229    // The tracks the artist is audible on, which is what one wants to hear or
230    // to tabulate.
231    let selected: Vec<Id> = catalog
232        .performed_tracks_of_artist(artist.id)
233        .into_iter()
234        .skip(window.offset)
235        .take(window.limit)
236        .collect();
237    if let Some(result) = selection_output(&catalog, &selected, args) {
238        return result;
239    }
240    // Reached only when none of `--csv`, `--json` or `--m3u` was given, so
241    // the ordinary biography page below is what is about to print — and
242    // `main.rs` lets `--output` through for every `artist` invocation
243    // without knowing which of its several shapes was actually asked for.
244    super::refuse_output_without_a_format(args)?;
245
246    println!("{}", ui::section(&artist.name));
247    if artist.sort_name != artist.name {
248        println!("  {}", ui::dim(&format!("sort: {}", artist.sort_name)));
249    }
250    if let Some(mbid) = &artist.mbid {
251        println!("  {}", ui::dim(&format!("MusicBrainz: {mbid}")));
252    }
253    // **A merge the reader cannot see is a merge they cannot check.** Two rows
254    // became one because the files carrying them carried the same MusicBrainz
255    // identifier, and a shelf that simply stops listing `O. Osbourne` leaves
256    // somebody unable to tell that from a folder that was never scanned. The
257    // spellings are normalised because that is the form the merge was decided
258    // on, and showing them as they were written would suggest the tags had been
259    // changed — which nothing here ever does.
260    if !artist.aliases.is_empty() {
261        println!(
262            "  {}",
263            ui::dim(&format!(
264                "also spelled {} in your tags, merged by that identifier",
265                artist.aliases.join(", ")
266            ))
267        );
268    }
269
270    let artist_id = artist.id;
271    let quoted_name = super::navigation::shell_arg(&artist.name);
272    let mut navigation = super::navigation::Navigation::default();
273    navigation.add(
274        "Filtered albums",
275        format!("aede albums --artist={quoted_name}"),
276    );
277    for sourced in contributors
278        .iter()
279        .filter(|sourced| contributors::matches_name(sourced, &artist.name, true))
280    {
281        println!(
282            "  {}",
283            ui::dim(&format!(
284                "separate source credit identity with this name: {} (not linked to local tags)",
285                sourced.mbid
286            ))
287        );
288        navigation.source_artist(&sourced.mbid);
289    }
290    let own = catalog.releases_as_album_artist(artist_id);
291    let guest = catalog.guest_appearances(artist_id);
292    let tracks = catalog.performed_tracks_of_artist(artist_id);
293    // Two lines, always labelled. An unlabelled count reads as "everything
294    // about this artist", and a band's lyricist — credited on forty albums,
295    // audible on none — was announced as "0 album · 0 track" right above the
296    // forty rows that contradict it.
297    //
298    // And each line counts **everything** in its class. The writing line used
299    // to report the size of the table further down, which leaves out whatever
300    // the artist also plays on: Ozzy Osbourne, sixty-nine composer credits and
301    // sixty-eight as lyricist, was announced as writing one track. A figure
302    // that answers a narrower question than its label is worse than no figure.
303    let writing_releases = catalog.releases_with_writing_credit(artist_id);
304    let writing_tracks = catalog.writing_tracks_of_artist(artist_id);
305    print_measures(&catalog, "performing", own.len(), &tracks);
306    print_measures(&catalog, "writing", writing_releases.len(), &writing_tracks);
307
308    // The table below shows only what the two tables above do not, so it needs
309    // its own set — and, since the two numbers differ, its own heading.
310    let written_elsewhere = catalog.releases_written_without_performing(artist_id);
311
312    let genres = collect_genres_for_artist(&catalog, artist.id);
313    if !genres.is_empty() {
314        println!(
315            "  {} {}",
316            ui::dim("genres:"),
317            ui::literal(&genres.join(", "))
318        );
319    }
320
321    print_release_table(
322        &catalog,
323        "Discography",
324        &own,
325        TrackColumn::WholeRelease,
326        window,
327    );
328    // Under *this* table and no other. It was under the last one printed,
329    // which is "Appears on" — so a line about the records that are theirs sat
330    // beneath a table of records that are not, and read as being about it.
331    // **A line belongs under the table it is about, not at the end of the
332    // page**: where a note is put is part of what it says.
333    //
334    // Guarded on the table having been printed at all. An artist with no
335    // album of their own is never browsed — `discography::has_shelf` asks the
336    // same question — so the count would be zero anyway; relying on that
337    // coincidence in another module is how a heading ends up with nothing
338    // under it.
339    if !own.is_empty() {
340        say_what_is_missing(args, &catalog, artist.id);
341    }
342    print_release_table(
343        &catalog,
344        "Appears on",
345        &guest,
346        TrackColumn::OnlyArtist {
347            header: "Tracks here",
348            artist_id,
349        },
350        window,
351    );
352    print_release_table(
353        &catalog,
354        "Written or produced, without performing on it",
355        &written_elsewhere,
356        TrackColumn::WholeRelease,
357        window,
358    );
359
360    let neighbours = catalog.neighbours_of_artist(artist.id);
361    if !neighbours.is_empty() {
362        println!("{}", ui::section("Played with"));
363        let mut t = Table::new(&["Artist", "Tracks in common"])
364            .align(1, Align::Right)
365            .limit(0, 40);
366        let neighbours_window = args.window(20)?;
367        for (other, weight, _) in neighbours
368            .iter()
369            .skip(neighbours_window.offset)
370            .take(neighbours_window.limit)
371        {
372            t.push(vec![other.name.clone(), weight.to_string()]);
373        }
374        print!("{}", t.render());
375        announce_window(neighbours_window, neighbours.len(), "collaborator");
376        println!(
377            "{}",
378            ui::dim("  (inferred from the credits found in tags; MusicBrainz will enrich these)")
379        );
380        println!(
381            "{}",
382            ui::dim("  aede artist \"<name>\" --with=\"<other>\" lists the tracks in common")
383        );
384    }
385
386    // Shown as soon as the artist does something beyond being the main credit,
387    // even if that is the only thing they do: an artist who is *only* a
388    // producer used to get no panel at all, which read as "no role recorded".
389    let roles = collect_roles(&catalog, artist.id);
390    if roles
391        .iter()
392        .any(|(role, _)| role != "main" && role != "album")
393    {
394        println!("{}", ui::section("Roles"));
395        let mut t = Table::new(&["Role", "Occurrences"]).align(1, Align::Right);
396        for (role, count) in roles {
397            t.push(vec![role_label(&role), count.to_string()]);
398        }
399        print!("{}", t.render());
400    }
401    let source_credits: Vec<_> = held
402        .credit_links(&catalog)
403        .into_iter()
404        .filter(|link| artist.mbid.as_deref() == Some(link.credit.artist_mbid.as_str()))
405        .collect();
406    for relation in catalog
407        .relations_from(EntityKind::Artist, artist.id)
408        .into_iter()
409        .filter(|relation| {
410            relation.target_kind == EntityKind::Recording && relation.kind.starts_with("credit:")
411        })
412    {
413        navigation.entity(
414            &catalog,
415            "Credited recording",
416            EntityKind::Recording,
417            relation.target_id,
418        );
419    }
420    for link in &source_credits {
421        if !link.trusted {
422            continue;
423        }
424        navigation.entity(
425            &catalog,
426            "Sourced recording",
427            EntityKind::Recording,
428            link.recording_id,
429        );
430        if let Some(work) = &link.work {
431            if let Some(local) = catalog.works.iter().find(|local| local.mbid == work.mbid) {
432                navigation.entity(&catalog, "Credited work", EntityKind::Work, local.id);
433            } else {
434                navigation.add(
435                    "Sourced work",
436                    format!("aede work {}", super::navigation::shell_arg(&work.mbid)),
437                );
438            }
439        }
440    }
441    super::print_sourced_credits(&catalog, source_credits);
442    let edition_credits: Vec<_> = held
443        .edition_credit_links(&catalog)
444        .into_iter()
445        .filter(|link| artist.mbid.as_deref() == Some(link.credit.artist_mbid.as_str()))
446        .collect();
447    for link in edition_credits.iter().filter(|link| link.trusted) {
448        navigation.entity(
449            &catalog,
450            "Credited edition",
451            EntityKind::Release,
452            link.release_id,
453        );
454    }
455    super::print_sourced_edition_credits(&catalog, edition_credits);
456    super::sources_panel_for(args, &catalog, EntityKind::Artist, artist.id);
457    say_who_played(args, &catalog, artist.id, &artist.name);
458    // A rating given and never shown again is a rating nobody trusts.
459    super::panel_for(args, &catalog, EntityKind::Artist, artist.id);
460    navigation.print();
461    Ok(())
462}

464/// Names `aede missing` under the artist's own discography.
465///
466/// Called from directly beneath that table rather than at the end of the page:
467/// `aede missing` was in the help, in the README and in the manual, and still
468/// could not be found, because it was named nowhere near the question it
469/// answers. **A command named only where nobody is looking is a command
470/// nobody has** — and "near" means under the right table, not merely on the
471/// right page.
472///
473/// Silent in three cases, each of which would otherwise be noise on a page
474/// that is about something else: no discography has been browsed for this
475/// artist, the shelf holds everything MusicBrainz credits to them, or the
476/// layer cannot be read at all. A page does not fail because a panel could
477/// not be filled.
478fn say_what_is_missing(args: &Args, catalog: &aede_core::model::Catalog, artist: Id) {
479    let Ok(held) = super::sources_held(args) else {
480        return;
481    };
482    let aside = match super::user_data(args, catalog) {
483        Ok(user) => user.set_aside,
484        Err(_) => Vec::new(),
485    };
486    let count = super::discography::absent_for(catalog, &held, &aside, artist);
487    if count == 0 {
488        return;
489    }
490    let name = catalog
491        .artist(artist)
492        .map(|a| a.name.clone())
493        .unwrap_or_default();
494    println!(
495        "  {}",
496        ui::dim(&format!(
497            "{} MusicBrainz credits to them and this shelf does not hold: \
498             aede missing \"{name}\"",
499            ui::plural(count, "studio album")
500        ))
501    );
502}

504/// Names the line-up on the page that already knows about it.
505///
506/// The same rule as the missing-albums line just below: **a command named only
507/// where nobody is looking is a command nobody has**. The table itself does not
508/// belong here — it is long, dated, and about a different question than the
509/// discography this page is built around — but its existence does, under the
510/// artist it is about.
511fn say_who_played(args: &Args, catalog: &Catalog, artist: Id, name: &str) {
512    let Some(facts) = super::artist_facts_for(args, catalog, artist) else {
513        return;
514    };
515    let (players, bands) = (facts.line_up().count(), facts.bands().count());
516    let said = match (players, bands) {
517        (0, 0) => return,
518        (0, bands) => format!("played in {}", ui::plural(bands, "band")),
519        // "musician", not "member": most of what a solo artist's record holds
520        // are supporting musicians, and calling Randy Rhoads a member of Ozzy
521        // Osbourne would be the page inventing a relationship the source never
522        // stated.
523        (players, 0) => format!("{} played with them", ui::plural(players, "musician")),
524        (players, bands) => format!(
525            "{} played with them · played in {}",
526            ui::plural(players, "musician"),
527            ui::plural(bands, "band")
528        ),
529    };
530    println!(
531        "  {}",
532        ui::dim(&format!(
533            "{said}, with dates: aede artist \"{name}\" --members"
534        ))
535    );
536}

538/// `aede artist <name> --members`: the line-up, dated.
539///
540/// **One relation, read from either end.** MusicBrainz states a membership
541/// once, between two artists, and returns it on both of them; for a band the
542/// rows are the musicians who played in it, for a person they are the bands
543/// they played in. This prints whichever the artist has, and both when an
544/// artist is both — a soloist whose backing band carries their own name is
545/// exactly that, and picking one at write time would lose the other.
546fn print_members(args: &Args, catalog: &Catalog, artist: &Artist) -> Res {
547    // `--members` never had a `--csv`/`--json`/`--m3u` form to begin with.
548    super::refuse_output_without_a_format(args)?;
549    let facts = super::artist_facts_for(args, catalog, artist.id);
550    let Some(facts) = facts else {
551        return Err(format!(
552            "nothing has been fetched about {name} yet: aede fetch \"{name}\"",
553            name = artist.name
554        )
555        .into());
556    };
557
558    let mut printed = false;
559    // "Played with", not "Line-up": Ozzy Osbourne's record holds no members at
560    // all — a solo artist is not a band — and every musician on it is there as
561    // a supporting musician. A heading that called them the line-up would be
562    // the page asserting what the Relation column then contradicts.
563    for (title, rows) in [
564        ("Played with them", facts.line_up().collect::<Vec<_>>()),
565        ("Played in", facts.bands().collect::<Vec<_>>()),
566    ] {
567        if rows.is_empty() {
568            continue;
569        }
570        printed = true;
571        println!("{}", ui::section(title));
572        // "As" rather than "Instruments", because the source's attributes are
573        // not all instruments: a real answer carries `guitar family, original`,
574        // where `original` marks an original member. And "Relation" carries
575        // MusicBrainz's own phrase — a founding member and a guitarist hired
576        // for one tour are both on the record, and only one was in the band.
577        let mut table = Table::new(&["Name", "Relation", "As", "Years", "Identity"])
578            .limit(1, 32)
579            .limit(4, 40);
580        for row in rows_in_order(rows) {
581            let local = catalog
582                .artists
583                .iter()
584                .any(|artist| artist.mbid.as_deref() == Some(&row.mbid));
585            table.push(vec![
586                row.name.clone(),
587                row.kind.clone(),
588                row.attributes.join(", "),
589                row.years(),
590                format!("{}{}", row.mbid, if local { " · local" } else { "" }),
591            ]);
592        }
593        println!("{}", table.render());
594    }
595
596    if !printed {
597        println!("{}", ui::section(&artist.name));
598        // Said rather than left blank, because the two cases look identical on
599        // an empty screen and only one of them is worth acting on: MusicBrainz
600        // holds no membership for a solo artist who never joined anything, and
601        // holds plenty for a band nobody has fetched again.
602        println!(
603            "  {}",
604            ui::dim(&format!(
605                "MusicBrainz lists nobody playing with them. Somebody who never \
606                 joined a group and never hired a band has nobody; if you \
607                 expected somebody, aede fetch \"{}\" --full asks again",
608                artist.name
609            ))
610        );
611    }
612    Ok(())
613}

615/// Memberships in the order a reader expects: earliest first, then by name.
616///
617/// Earliest first because a line-up is a history, and a history read out of
618/// order is a list. The name breaks the tie so that two runs agree, and a
619/// missing date sorts last rather than first — an undated row is not the
620/// beginning of the story, it is the one nobody has written down.
621fn rows_in_order(
622    mut rows: Vec<&aede_core::sources::Membership>,
623) -> Vec<&aede_core::sources::Membership> {
624    rows.sort_by(|a, b| {
625        let key = |m: &aede_core::sources::Membership| {
626            m.began.clone().unwrap_or_else(|| "9999".to_string())
627        };
628        key(a).cmp(&key(b)).then_with(|| a.name.cmp(&b.name))
629    });
630    rows
631}

633/// Which release tables the artist page can print.
634///
635/// A guest appearance concerns only part of the release, so the track column
636/// must count the artist's own tracks rather than the whole album.
637enum TrackColumn {
638    WholeRelease,
639    OnlyArtist { header: &'static str, artist_id: Id },
640}
641
642/// Renders one of the artist page's release tables.
643fn print_release_table(
644    catalog: &Catalog,
645    title: &str,
646    ids: &[Id],
647    column: TrackColumn,
648    window: Window,
649) {
650    if ids.is_empty() {
651        return;
652    }
653    println!("{}", ui::section(title));
654    let track_header = match column {
655        TrackColumn::WholeRelease => "Tracks",
656        TrackColumn::OnlyArtist { header, .. } => header,
657    };
658    let mut t = super::album_summary_table(track_header, 24);
659
660    let mut list: Vec<&aede_core::model::Release> =
661        ids.iter().filter_map(|&id| catalog.release(id)).collect();
662    list.sort_by_key(|r| (r.year.unwrap_or(u32::MAX), r.title.clone()));
663
664    let total = list.len();
665    for release in list.into_iter().skip(window.offset).take(window.limit) {
666        // The whole row describes the same set of tracks. Counting one track
667        // and timing the entire album made a guest appearance of one song look
668        // like forty minutes of music.
669        let counted: Vec<Id> = match column {
670            TrackColumn::WholeRelease => release.track_ids.clone(),
671            TrackColumn::OnlyArtist { artist_id, .. } => release
672                .track_ids
673                .iter()
674                .copied()
675                .filter(|&track_id| {
676                    catalog
677                        .credits_on(EntityKind::Track, track_id)
678                        .iter()
679                        .any(|(a, role)| {
680                            a.id == artist_id && aede_core::model::is_performing_role(role)
681                        })
682                })
683                .collect(),
684        };
685        let (duration, size) = totals(catalog, &counted);
686        let album_artist = super::album_artist_name(catalog, release);
687        t.push(vec![
688            super::year_label(release.year),
689            format!("{}{}", release.title, copy_marker(catalog, release.id)),
690            album_artist,
691            counted.len().to_string(),
692            text::format_duration(duration),
693            text::format_size(size),
694            super::track_formats(catalog, &counted),
695        ]);
696    }
697    print!("{}", t.render());
698    announce_window(window, total, "album");
699}

712fn collect_roles(catalog: &Catalog, artist_id: Id) -> Vec<(String, usize)> {
713    let mut counts: std::collections::BTreeMap<String, usize> = Default::default();
714    for credit in catalog.credits.iter().filter(|c| c.artist_id == artist_id) {
715        *counts.entry(credit.role.clone()).or_insert(0) += 1;
716    }
717    let mut list: Vec<(String, usize)> = counts.into_iter().collect();
718    list.sort_by(|a, b| b.1.cmp(&a.1).then_with(|| a.0.cmp(&b.0)));
719    list
720}

722/// Lists the tracks two artists both perform on.
723///
724/// The pair is what the collaboration weight counts, so the number of rows
725/// here always matches the figure shown in the "Played with" table.
726/// The tracks one artist holds one role on.
727///
728/// An empty answer names the roles this person *does* hold: told only that
729/// nobody is credited that way, the user cannot tell a wrong spelling from a
730/// library whose tags never carried the field.
731fn print_tracks_in_role(catalog: &Catalog, artist_id: Id, typed: &str, args: &Args) -> Res {
732    let name = catalog
733        .artist(artist_id)
734        .map(|a| a.name.clone())
735        .unwrap_or_default();
736    let held = catalog.roles_of_artist(artist_id);
737    let listed: Vec<String> = held
738        .iter()
739        .map(|(role, count)| format!("{} ({count})", role_label(role)))
740        .collect();
741
742    // What the user typed is matched against both spellings of every role,
743    // because "album artist" is the only one they have ever been shown.
744    let Some(role) = role_key(catalog, typed) else {
745        return Err(match listed.is_empty() {
746            true => format!("{name} carries no credit at all"),
747            false => format!(
748                "no role is called \"{typed}\".\nRoles in use: {}",
749                roles_offered(catalog)
750            ),
751        }
752        .into());
753    };
754    let tracks = tracks_in_page_order(catalog, &catalog.tracks_of_artist_in_role(artist_id, &role));
755    if tracks.is_empty() {
756        return Err(match listed.is_empty() {
757            true => format!("{name} carries no credit at all"),
758            false => format!(
759                "{name} is not credited as {}.\nCredited as: {}",
760                role_label(&role),
761                listed.join(", ")
762            ),
763        }
764        .into());
765    }
766
767    let window = args.window(50)?;
768    let selected: Vec<Id> = tracks
769        .iter()
770        .copied()
771        .skip(window.offset)
772        .take(window.limit)
773        .collect();
774    if let Some(result) = selection_output(catalog, &selected, args) {
775        return result;
776    }
777    // Reached only when none of `--csv`, `--json` or `--m3u` was given, so
778    // `--output` — asked for on its own — has nothing here to write either.
779    super::refuse_output_without_a_format(args)?;
780    print_track_table(
781        catalog,
782        &format!("{name} as {}", role_label(&role)),
783        &tracks,
784        args,
785    )
786}

825/// The chronological order shared by narrowed human and exported track pages.
826/// Catalog IDs follow file paths; slicing those IDs before the human sort
827/// would put different tracks on the same numbered page of an export.
828fn tracks_in_page_order(catalog: &Catalog, tracks: &[Id]) -> Vec<Id> {
829    let mut rows: Vec<_> = tracks.iter().filter_map(|&id| catalog.track(id)).collect();
830    rows.sort_by(|left, right| {
831        let left_release = left.release_id.and_then(|id| catalog.release(id));
832        let right_release = right.release_id.and_then(|id| catalog.release(id));
833        left_release
834            .and_then(|release| release.year)
835            .unwrap_or(u32::MAX)
836            .cmp(
837                &right_release
838                    .and_then(|release| release.year)
839                    .unwrap_or(u32::MAX),
840            )
841            .then_with(|| {
842                left_release
843                    .map(|release| release.title.as_str())
844                    .unwrap_or_default()
845                    .cmp(
846                        right_release
847                            .map(|release| release.title.as_str())
848                            .unwrap_or_default(),
849                    )
850            })
851            .then_with(|| left.title.cmp(&right.title))
852            .then_with(|| left.id.cmp(&right.id))
853    });
854    rows.into_iter().map(|track| track.id).collect()
855}

857/// One ordered track table, with totals for the complete matching selection.
858fn print_track_table(catalog: &Catalog, heading: &str, tracks: &[Id], args: &Args) -> Res {
859    let window = args.window(50)?;
860    println!("{}", ui::section(heading));
861    let mut t = Table::new(&["Year", "Album", "Track", "Duration", "Size", "Format"])
862        .align(3, Align::Right)
863        .align(4, Align::Right)
864        .limit(1, 32)
865        .limit(2, 36);
866    let rows: Vec<(Option<u32>, String, String, u64, u64, String)> = tracks
867        .iter()
868        .filter_map(|&id| catalog.track(id))
869        .map(|track| {
870            let release = track.release_id.and_then(|id| catalog.release(id));
871            let file = catalog.file(track.file_id);
872            (
873                release.and_then(|r| r.year),
874                release.map(|r| r.title.clone()).unwrap_or_default(),
875                track.title.clone(),
876                track.duration_ms.unwrap_or(0),
877                file.map(|f| f.size).unwrap_or(0),
878                file.map(|f| f.properties.quality_label())
879                    .unwrap_or_default(),
880            )
881        })
882        .collect();
883
884    let total = rows.len();
885    let (duration, size) = totals(catalog, tracks);
886    for (year, album, title, track_ms, bytes, format) in
887        rows.into_iter().skip(window.offset).take(window.limit)
888    {
889        t.push(vec![
890            year.map(|y| y.to_string()).unwrap_or_else(|| "—".into()),
891            album,
892            title,
893            text::format_duration(track_ms),
894            text::format_size(bytes),
895            format,
896        ]);
897    }
898    print!("{}", t.render());
899    announce_window(window, total, "track");
900    println!(
901        "  {}",
902        ui::dim(&format!(
903            "{} · {} · {}",
904            ui::plural(tracks.len(), "track"),
905            ui::long_duration(duration),
906            text::format_size(size)
907        ))
908    );
909    Ok(())
910}

912/// One summary line: how many releases, how many tracks, and what they weigh.
913///
914/// `label` says which credits the figures cover, because performing on a
915/// record and writing for it are two different presences and the same artist
916/// can have both.
917fn print_measures(catalog: &Catalog, label: &str, releases: usize, tracks: &[Id]) {
918    if releases == 0 && tracks.is_empty() {
919        return;
920    }
921    let (duration, size) = totals(catalog, tracks);
922    println!(
923        "  {:<11} {} · {} · {} · {}",
924        format!("{label}:"),
925        ui::plural(releases, "album"),
926        ui::plural(tracks.len(), "track"),
927        ui::long_duration(duration),
928        text::format_size(size)
929    );
930}
