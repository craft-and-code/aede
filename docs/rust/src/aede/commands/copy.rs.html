1//! The `copy` command: put a selection somewhere that is not the library.
2//!
3//! A player, a card, an external drive. The destination is not a catalog and
4//! will never be scanned, which is what separates this from everything else the
5//! program does: it writes derived audio and companion files **outside** the
6//! watched library. Nothing it does touches the catalog or what the
7//! user wrote.
8//!
9//! The selection is the grammar's, not a set of filters of its own — the same
10//! rule the listings follow. `aede copy /Volumes/USB --query "loved
11//! rating:>=4"` reuses, exactly, what `aede query` would have shown.
12

26/// Files copied between two redraws of the progress line.
27const REDRAW_EVERY: usize = 4;
28
29pub fn copy(args: &Args) -> Res {
30    let destination = destination(args)?;
31    let catalog = load(args)?;
32
33    // Copying a library into itself is not a mistake to notice afterwards: the
34    // next scan reads the copies as new files, the catalog doubles, and
35    // `doctor` reports every album as a duplicate of itself.
36    if let Some(root) = copy::inside_a_watched_root(&catalog, &destination) {
37        return Err(format!(
38            "\"{}\" is inside the watched folder \"{root}\".\n\
39             A copy is not a library: the next scan would read it back in and \
40             every album would become its own duplicate.\n\
41             Choose a destination outside your watched folders.",
42            destination.display()
43        )
44        .into());
45    }
46
47    let extras = match args.value("extras") {
48        None => Extras::default(),
49        Some(word) => Extras::parse(word)
50            .ok_or_else(|| format!("--extras takes {}: got \"{word}\"", Extras::NAMES))?,
51    };
52    let tracks = selection(args, &catalog)?;
53    if tracks.is_empty() {
54        return Err("nothing to copy: that selection matches no track".into());
55    }
56
57    // Asked of the volume rather than inferred from a filesystem name, and
58    // overridable in both directions because a probe can only answer for the
59    // folder it was run in.
60    let target = convert(args)?;
61    let chosen_quality = quality(args, target)?;
62    // Invalid options remain errors in a no-write preview, before probing.
63    args.number_or("threads", 0)?;
64    let restrict = match (args.has("safe-names"), args.has("raw-names")) {
65        (true, true) => return Err("--safe-names and --raw-names ask for opposite things".into()),
66        (true, false) => true,
67        (false, true) => false,
68        (false, false) => args.has("dry-run") || copy::names::restricts_names(&destination),
69    };
70
71    let recipe = Recipe {
72        extras,
73        restrict_names: restrict,
74        convert: target,
75        quality: chosen_quality,
76    };
77
78    let mut plan = copy::plan(&catalog, &tracks, &recipe);
79    if let Some(failure) = plan.rejected.first() {
80        return Err(format!("{}: {}", failure.source.display(), failure.reason).into());
81    }
82    for item in &plan.items {
83        copy::validate_source(&item.source)?;
84    }
85    let omitted = if args.has("playlists") {
86        plan.prepare_playlists(&catalog, &tracks)?
87    } else {
88        0
89    };
90    for item in &plan.items {
91        copy::validate_destination(&destination, &item.relative)?;
92    }
93    announce(&plan, &destination, restrict, &recipe);
94    if omitted > 0 {
95        println!(
96            "  {}",
97            ui::dim(&format!(
98                "{omitted} playlist entries outside the selected audio were omitted"
99            ))
100        );
101    }
102    if args.has("dry-run") && !args.has("safe-names") && !args.has("raw-names") {
103        println!(
104            "  {}",
105            ui::dim(
106                "names are planned conservatively without probing; --raw-names previews original characters"
107            )
108        );
109    }
110    if args.has("dry-run") {
111        println!(
112            "  {}",
113            ui::dim("--dry-run: nothing was written. Drop it to copy.")
114        );
115        return Ok(());
116    }
117    if plan.items.is_empty() {
118        return Err("nothing to copy".into());
119    }
120    // Looked for once, before the first file rather than on each of nine
121    // hundred — and before the first byte is written, so a missing encoder is
122    // an error at the start rather than half a copy.
123    let ffmpeg = match plan.converted() > 0 {
124        false => None,
125        true => Some(transcode::find_ffmpeg().ok_or_else(transcode::missing_ffmpeg)?),
126    };
127    let threads = workers(args, &recipe, plan.items.len())?;
128    room_for(&plan, &destination, args, threads)?;
129    copy::validate_filesystem_names(&destination, &plan)?;
130    if !args.has("replace")
131        && plan
132            .items
133            .iter()
134            .any(|item| !destination.join(&item.relative).exists())
135    {
136        copy::validate_new_publication(&destination)?;
137    }
138    run(
139        &plan,
140        &destination,
141        args,
142        &recipe,
143        ffmpeg.as_deref(),
144        threads,
145    )
146}

148/// The format to encode into, when one was asked for.
149fn convert(args: &Args) -> Result<Option<Target>, Box<dyn std::error::Error>> {
150    let Some(word) = args.value("compress") else {
151        return Ok(None);
152    };
153    Target::parse(word)
154        .map(Some)
155        .ok_or_else(|| format!("--compress takes {}: got \"{word}\"", Target::names()).into())
156}

158/// How hard the encoder should try, when it was said.
159///
160/// Refused rather than ignored in the two cases where it cannot be honoured,
161/// and the second is the interesting one. `--compress wav --quality 128k` reads
162/// as a request for small files; WAV has no quality setting at all, so the
163/// option went into the void and the run produced files roughly eleven times
164/// larger than the number that had just been typed. On the card this command
165/// exists to fill, that is the difference between fitting and not.
166///
167/// Stopping costs nothing here — the check happens before a single file is
168/// read — which is what settles it against a warning: a warning scrolls past a
169/// plan and a progress line, and the run that follows is the wrong one.
170fn quality(
171    args: &Args,
172    target: Option<Target>,
173) -> Result<Option<Quality>, Box<dyn std::error::Error>> {
174    let Some(word) = args.value("quality") else {
175        return Ok(None);
176    };
177    let Some(target) = target else {
178        return Err(
179            "--quality says how to encode, and nothing is being encoded: add --compress".into(),
180        );
181    };
182    if target.lossless() {
183        return Err(format!(
184            "--quality means nothing for {}: a lossless format keeps every sample, \
185             so there is no quality to choose.\n\
186             It applies to {}.\n\
187             Drop --quality, or compress to one of those.",
188            args.value("compress").unwrap_or("that format"),
189            Target::lossy_names()
190        )
191        .into());
192    }
193    Quality::parse_for(target, word).map(Some).ok_or_else(|| {
194        format!(
195            "--quality for .{} takes {}: got \"{word}\"",
196            target.extension(),
197            target.quality_forms()
198        )
199        .into()
200    })
201}

203/// Where the copy is going, checked before anything else is read.
204///
205/// **The folder must already exist.** Creating it would be friendlier exactly
206/// once and catastrophic the rest of the time: `aede copy /Volumes/Player` with
207/// the player unplugged would create `/Volumes/Player` on the internal disk and
208/// quietly fill it with sixty gigabytes, leaving the user with a full disk, no
209/// copy, and nothing on screen that said so.
210///
211/// **And it comes back canonical**, because the guard downstream compares it
212/// against the watched roots — which `scan` stores canonical — as strings. A
213/// destination reached through a symbolic link names the same folder by a
214/// string that never compares equal, and "is this inside my library" answered
215/// `no` for a folder that plainly was. On macOS that is the ordinary case
216/// rather than a corner one: `/var` is a link to `/private/var`, so every path
217/// under it exists in two spellings and only one of them is the catalog's.
218/// `scan` and `check` both canonicalize what they are given; this is the same
219/// step, in the one command that writes.
220fn destination(args: &Args) -> Result<PathBuf, Box<dyn std::error::Error>> {
221    let Some(raw) = args.positionals.first() else {
222        return Err("where to? aede copy /Volumes/Player --query \"loved\"".into());
223    };
224    if args.positionals.len() > 1 {
225        return Err(format!(
226            "one destination at a time: \"{}\" was not understood.\n\
227             The selection goes in --query, not in the arguments.",
228            args.positionals[1..].join(" ")
229        )
230        .into());
231    }
232    let path = PathBuf::from(raw);
233    if !path.exists() {
234        return Err(format!(
235            "\"{raw}\" does not exist.\n\
236             The folder is not created for you: a destination that is missing is \
237             usually a drive that is not plugged in, and filling the internal disk \
238             instead is the one outcome nobody wants."
239        )
240        .into());
241    }
242    if !path.is_dir() {
243        return Err(format!("\"{raw}\" is not a folder").into());
244    }
245    Ok(canonical(&path))
246}

248/// The tracks to copy: an expression, a saved collection, or the whole library.
249fn selection(
250    args: &Args,
251    catalog: &Catalog,
252) -> Result<Vec<aede_core::model::Id>, Box<dyn std::error::Error>> {
253    let data = user_data(args, catalog)?;
254    let held = super::sources_held(args)?;
255    let context = aede_core::query::Context::new(catalog, &data, aede_core::user::LOCAL_USER)
256        .with_sources(&held);
257    let expression = match (args.value("query"), args.value("collection")) {
258        (Some(_), Some(_)) => {
259            return Err("--query and --collection both name a selection: give one".into());
260        }
261        (Some(expression), None) => expression.to_string(),
262        (None, Some(name)) => data
263            .collection(aede_core::user::LOCAL_USER, name)
264            .map(|c| c.expression.clone())
265            .ok_or_else(|| format!("no collection is called \"{name}\""))?,
266        // No selection is the whole library, which is what somebody filling an
267        // empty drive means. It is not a mistake, so it is not an error.
268        (None, None) => String::new(),
269    };
270    let parsed = aede_core::query::parse(&expression)?;
271    if let Some((what, value)) = aede_core::query::unknown_values(&parsed, &context).first() {
272        return Err(
273            format!("no {what} matches \"{value}\".\nRun \"aede {what}s\" for the list.").into(),
274        );
275    }
276    Ok(aede_core::query::run(&parsed, &context))
277}

279/// What the copy is about to do, said before it does it.
280fn announce(plan: &Plan, destination: &Path, restrict: bool, recipe: &Recipe) {
281    println!("{}", ui::section("Copy"));
282    let counts = plan.counts();
283    let mut table = Table::plain(2).align(1, Align::Right);
284    for (kind, label) in [
285        (ItemKind::Audio, "Tracks"),
286        (ItemKind::Cover, "Covers"),
287        (ItemKind::Other, "Other files"),
288    ] {
289        if let Some(count) = counts.get(&kind) {
290            table.push(vec![label.into(), count.to_string()]);
291        }
292    }
293    if plan.converted() > 0 {
294        table.push(vec!["To encode".into(), plan.converted().to_string()]);
295        let untouched = counts.get(&ItemKind::Audio).copied().unwrap_or(0) - plan.converted();
296        if untouched > 0 {
297            table.push(vec!["Copied as they are".into(), untouched.to_string()]);
298        }
299    }
300    table.push(vec![
301        match plan.size_is_estimated() {
302            true => "Size (estimated)".into(),
303            false => "Size".to_string(),
304        },
305        text::format_size(plan.total_bytes()),
306    ]);
307    print!("{}", table.render());
308    println!(
309        "  {}",
310        ui::dim(&format!(
311            "to {}",
312            ui::literal(&destination.to_string_lossy())
313        ))
314    );
315    if plan.size_is_estimated() {
316        println!(
317            "  {}",
318            ui::dim("what an encoder produces is not known until it has: the size is a guess")
319        );
320    }
321    // Said plainly, because it is the one thing about a conversion that
322    // surprises people: a library that is half MP3 already comes out half
323    // untouched, and a silent skip would look like files had been lost.
324    let audio = counts.get(&ItemKind::Audio).copied().unwrap_or(0);
325    if plan.converted() > 0 && plan.converted() < audio {
326        println!(
327            "  {}",
328            ui::dim("what is already compressed is copied rather than encoded a second time")
329        );
330    }
331    // And the case where *nothing* is encoded, which is the same silence seen
332    // from the other side: `--compress mp3` over a selection that is already
333    // MP3 did its job perfectly and said nothing at all about it, leaving the
334    // option looking ignored. It was honoured; it simply had nothing to do.
335    if recipe.convert.is_some() && plan.converted() == 0 && audio > 0 {
336        println!(
337            "  {}",
338            ui::yellow(
339                "nothing here needs encoding: every track is already compressed, \
340                       or already in that format"
341            )
342        );
343    }
344    // Said plainly and once, rather than left for a player to notice: an
345    // embedded cover or an extended tag an encode is about to drop is data
346    // lost silently unless something here says so.
347    if plan.covers_the_target_cannot_hold > 0 {
348        let target = recipe.convert.map(Target::extension).unwrap_or_default();
349        println!(
350            "  {}",
351            ui::yellow(&format!(
352                "{} will lose its embedded cover: .{target} cannot carry a picture stream",
353                ui::plural(plan.covers_the_target_cannot_hold, "file")
354            ))
355        );
356    }
357    if plan.tags_the_target_cannot_hold > 0 {
358        println!(
359            "  {}",
360            ui::yellow(&format!(
361                "{} will lose a tag wav's fixed vocabulary has no room for \
362                       (composer, album artist, disc or publisher)",
363                ui::plural(plan.tags_the_target_cannot_hold, "file")
364            ))
365        );
366    }
367    if restrict {
368        println!(
369            "  {}",
370            ui::dim("this destination refuses some characters: names are adapted")
371        );
372    }
373
374    // Renames are listed, not counted. A copy whose names differ from the
375    // library is a copy nobody can compare against the original, and "37 files
376    // renamed" is not something anyone can act on.
377    if !plan.renamed.is_empty() {
378        println!("{}", ui::section("Renamed"));
379        let mut t = Table::new(&["In the library", "On the destination"]);
380        for renamed in plan.renamed.iter().take(20) {
381            t.push(vec![
382                renamed.from.to_string_lossy().to_string(),
383                renamed.to.to_string_lossy().to_string(),
384            ]);
385        }
386        print!("{}", t.render());
387        if plan.renamed.len() > 20 {
388            println!(
389                "  {}",
390                ui::dim(&format!("… and {} more", plan.renamed.len() - 20))
391            );
392        }
393    }
394
395    if !plan.rootless.is_empty() {
396        println!("{}", ui::section("No tree to keep"));
397        println!(
398            "  {}",
399            ui::yellow(&format!(
400                "{} under no watched folder, and so with no place in the copy",
401                ui::plural(plan.rootless.len(), "file")
402            ))
403        );
404        for path in plan.rootless.iter().take(5) {
405            println!("      {}", ui::dim(&ui::literal(&path.to_string_lossy())));
406        }
407    }
408}

410/// Refuses a copy that cannot fit, before writing the first byte.
411///
412/// Filling a card to the last byte and failing on the final album is a waste of
413/// twenty minutes that one `statvfs`-shaped question could have avoided. The
414/// answer is approximate — other things may be writing to the same volume — so
415/// it refuses only when the shortfall is plain.
416fn room_for(plan: &Plan, destination: &Path, args: &Args, threads: usize) -> Res {
417    room_for_available(plan, destination, args, threads, free_space(destination))
418}
419
420fn room_for_available(
421    plan: &Plan,
422    destination: &Path,
423    args: &Args,
424    threads: usize,
425    free: Option<u64>,
426) -> Res {
427    let Some(free) = free else {
428        return Ok(());
429    };
430    let needed = pending_bytes(plan, destination, args, threads)?;
431    if needed <= free {
432        return Ok(());
433    }
434    Err(format!(
435        "not enough room: {} to copy, {} free on {}.\n\
436         Narrow the selection with --query, or free some space.",
437        text::format_size(needed),
438        text::format_size(free),
439        destination.display()
440    )
441    .into())
442}

444/// Space for outputs still to write, plus concurrent comparison encodes.
445/// Existing plain files need no temporary output when they are skipped or
446/// refused; verifying converted files does need temporary space, released
447/// after each comparison rather than accumulated for the whole selection.
448fn pending_bytes(
449    plan: &Plan,
450    destination: &Path,
451    args: &Args,
452    threads: usize,
453) -> Result<u64, String> {
454    let mut needed = 0u64;
455    let mut comparisons = Vec::new();
456    for item in &plan.items {
457        match std::fs::symlink_metadata(destination.join(&item.relative)) {
458            Ok(metadata) if !metadata.is_file() || metadata.file_type().is_symlink() => {
459                return Err(format!(
460                    "{}: output is not an ordinary file",
461                    item.relative.display()
462                ));
463            }
464            Ok(_) if !args.has("replace") => {
465                if item.convert.is_some() && args.has("verify-existing") {
466                    comparisons.push(item.size);
467                }
468            }
469            Ok(_) => needed = needed.saturating_add(item.size),
470            Err(error) if error.kind() == std::io::ErrorKind::NotFound => {
471                needed = needed.saturating_add(item.size);
472            }
473            Err(error) => return Err(format!("{}: {error}", item.relative.display())),
474        }
475    }
476    comparisons.sort_unstable_by(|left, right| right.cmp(left));
477    Ok(comparisons
478        .into_iter()
479        .take(threads)
480        .fold(needed, u64::saturating_add))
481}

483/// Bytes free on the volume holding a folder, when the platform will say.
484///
485/// `None` rather than a guess when it will not: refusing a copy on a number
486/// nobody produced would be worse than not checking.
487fn free_space(destination: &Path) -> Option<u64> {
488    // No `libc` in this project, and one number does not justify one. `df` is
489    // specified by POSIX, present everywhere this runs, and its `-k` output is
490    // the one shape that does not vary between implementations.
491    let output = std::process::Command::new("df")
492        .arg("-k")
493        .arg(destination)
494        .output()
495        .ok()?;
496    let text = String::from_utf8_lossy(&output.stdout);
497    let line = text.lines().nth(1)?;
498    let available: u64 = line.split_whitespace().nth(3)?.parse().ok()?;
499    Some(available * 1024)
500}

502/// How many files to work on at once, which is not the same question when the
503/// run is encoding as when it is only copying.
504///
505/// **Encoding is arithmetic; copying is a queue at a device.** With
506/// `--compress`, each file is an ffmpeg encode — seconds of CPU that no other
507/// file waits on, and running one at a time leaves most of the machine idle.
508/// A plain copy is the opposite: the destination is usually *one* card, one
509/// stick, one slow drive, and several writers on it do not go faster — they
510/// seek against each other, and on cheap flash they go markedly slower.
511/// `--verify` makes that worse still by reading back what was just written.
512///
513/// So the default follows the work: parallel when there is something to encode,
514/// one at a time when there is not. `--threads` overrides it in both
515/// directions, for the person copying to an NVMe or encoding on a laptop they
516/// still want to use.
517fn workers(
518    args: &Args,
519    recipe: &Recipe,
520    total: usize,
521) -> Result<usize, Box<dyn std::error::Error>> {
522    let asked = args.number_or("threads", 0)?;
523    if asked > 0 {
524        return Ok(asked.min(total.max(1)));
525    }
526    match recipe.convert.is_some() {
527        true => Ok(aede_core::scan::resolve_threads(0).min(total.max(1))),
528        false => Ok(1),
529    }
530}

532/// Writes the plan, saying where it is as it goes.
533fn run(
534    plan: &Plan,
535    destination: &Path,
536    args: &Args,
537    recipe: &Recipe,
538    ffmpeg: Option<&str>,
539    threads: usize,
540) -> Res {
541    let verify = args.has("verify");
542    let replace = args.has("replace");
543    let verify_existing = args.has("verify-existing");
544    let interactive = ui::is_interactive();
545    let total = plan.items.len();
546
547    // Taken in reverse so that popping from the back walks the plan forwards,
548    // which is what makes the progress line agree with the folder tree being
549    // filled — a run that jumps about looks like it is doing something else.
550    let queue: Mutex<Vec<&Item>> = Mutex::new(plan.items.iter().rev().collect());
551    let counts = Mutex::new((0usize, 0usize)); // copied, skipped
552    let failures: Mutex<Vec<(PathBuf, String)>> = Mutex::new(Vec::new());
553    let done = AtomicUsize::new(0);
554
555    std::thread::scope(|scope| {
556        for _ in 0..threads {
557            scope.spawn(|| {
558                // A poisoned lock must not strand the rest of the plan:
559                // one file that panicked is not a reason to stop copying.
560                while let Some(item) = super::take_queued(&queue) {
561                    match write_one(
562                        item,
563                        destination,
564                        verify,
565                        replace,
566                        verify_existing,
567                        recipe,
568                        ffmpeg,
569                    ) {
570                        Ok(copy::Wrote::Copied) => {
571                            counts.lock().unwrap_or_else(|e| e.into_inner()).0 += 1;
572                        }
573                        Ok(copy::Wrote::Skipped) => {
574                            counts.lock().unwrap_or_else(|e| e.into_inner()).1 += 1;
575                        }
576                        // One unreadable file does not end the run: the other
577                        // nine hundred are still worth copying, and the report
578                        // names every failure.
579                        Err(reason) => failures
580                            .lock()
581                            .unwrap_or_else(|e| e.into_inner())
582                            .push((item.source.clone(), reason)),
583                    }
584                    done.fetch_add(1, Ordering::Relaxed);
585                }
586            });
587        }
588        if !interactive {
589            return;
590        }
591        let mut last = usize::MAX;
592        while done.load(Ordering::Relaxed) < total {
593            let current = done.load(Ordering::Relaxed);
594            if current != last && current.is_multiple_of(REDRAW_EVERY) {
595                last = current;
596                print!("\r  {current}/{total}   ");
597                use std::io::Write;
598                let _ = std::io::stdout().flush();
599            }
600            std::thread::sleep(std::time::Duration::from_millis(60));
601        }
602    });
603    if interactive {
604        println!("\r  {total}/{total}   ");
605    }
606
607    let (copied, skipped) = counts.into_inner().unwrap_or_else(|e| e.into_inner());
608    let mut failures = failures.into_inner().unwrap_or_else(|e| e.into_inner());
609    // Sorted, because the order several threads finish in is not an order:
610    // the same run twice must report the same list in the same sequence.
611    failures.sort();
612
613    println!("{}", ui::section("Copied"));
614    let mut table = Table::plain(2).align(1, Align::Right);
615    table.push(vec!["Written".into(), copied.to_string()]);
616    if skipped > 0 {
617        table.push(vec!["Already there".into(), skipped.to_string()]);
618    }
619    if !failures.is_empty() {
620        table.push(vec!["Failed".into(), failures.len().to_string()]);
621    }
622    print!("{}", table.render());
623    if verify {
624        println!(
625            "  {}",
626            ui::dim(
627                "new plain files were compared byte for byte; converted audio was checked for readable duration"
628            )
629        );
630    }
631    if verify_existing {
632        println!(
633            "  {}",
634            ui::dim(
635                "existing plain bytes and freshly encoded comparison outputs were checked before skipping"
636            )
637        );
638    }
639    if skipped > 0 && !verify_existing {
640        println!(
641            "  {}",
642            ui::dim(
643                "what was already there at the right size was left alone — \
644                    --replace writes it again"
645            )
646        );
647    }
648
649    if !failures.is_empty() {
650        println!("{}", ui::section("Failed"));
651        let mut t = Table::new(&["File", "Reason"]).path_limit(0, 50);
652        for (path, reason) in failures.iter().take(20) {
653            t.push(vec![path.to_string_lossy().to_string(), reason.clone()]);
654        }
655        print!("{}", t.render());
656        // A run that lost files must not report success: a script that copies
657        // to a card and then wipes the source has to be able to tell.
658        return Err(format!("{} could not be copied", ui::plural(failures.len(), "file")).into());
659    }
660    Ok(())
661}
662
663fn write_one(
664    item: &Item,
665    destination: &Path,
666    verify: bool,
667    replace: bool,
668    verify_existing: bool,
669    recipe: &Recipe,
670    ffmpeg: Option<&str>,
671) -> Result<copy::Wrote, String> {
672    copy::validate_destination(destination, &item.relative)?;
673    let target = destination.join(&item.relative);
674    if let Some(contents) = &item.contents {
675        if target.exists() && !replace {
676            if std::fs::read(&target).map_err(|error| error.to_string())? == contents.as_bytes() {
677                return Ok(copy::Wrote::Skipped);
678            }
679            return Err(format!(
680                "{}: existing playlist differs; use --replace to refresh it",
681                target.display()
682            ));
683        }
684        let temporary = copy::TemporaryOutput::new(&target)?;
685        std::fs::write(temporary.path(), contents).map_err(|error| error.to_string())?;
686        if verify {
687            std::fs::File::open(temporary.path())
688                .and_then(|file| file.sync_all())
689                .map_err(|error| error.to_string())?;
690            if std::fs::read(temporary.path()).map_err(|error| error.to_string())?
691                != contents.as_bytes()
692            {
693                return Err(format!(
694                    "{}: written playlist does not match its planned text",
695                    target.display()
696                ));
697            }
698        }
699        copy::validate_destination(destination, &item.relative)?;
700        publish(temporary, replace)?;
701        return Ok(copy::Wrote::Copied);
702    }
703    let Some(format) = item.convert else {
704        return copy::copy_one_checked(
705            &item.source,
706            &target,
707            item.size,
708            verify,
709            replace,
710            verify_existing,
711        );
712    };
713    let Some(ffmpeg) = ffmpeg else {
714        return Err("no encoder".into());
715    };
716
717    // A file already there is left alone, as in a plain copy — but the test
718    // cannot be the size, which for an encoder's output is a guess. Existence
719    // and a non-empty length is what can honestly be checked, so an
720    // complete-looking outputs can resume without encoding, and --replace
721    // explicitly refreshes an empty output or a changed recipe.
722    if !replace
723        && !verify_existing
724        && let Ok(existing) = std::fs::metadata(&target)
725        && existing.is_file()
726        && existing.len() > 0
727    {
728        return Ok(copy::Wrote::Skipped);
729    }
730    if !replace && !verify_existing && target.exists() {
731        return Err(format!(
732            "{}: existing encoded output is empty; use --replace to refresh it",
733            target.display()
734        ));
735    }
736    let temporary = copy::TemporaryOutput::new(&target)?;
737    transcode::convert(
738        ffmpeg,
739        &item.source,
740        temporary.path(),
741        format,
742        recipe.quality,
743    )?;
744    if verify || verify_existing {
745        transcode::verify(temporary.path(), source_duration(&item.source))?;
746    }
747    copy::validate_destination(destination, &item.relative)?;
748    if verify_existing && !replace && target.exists() {
749        return match copy::files_match(temporary.path(), &target)? {
750            true => Ok(copy::Wrote::Skipped),
751            false => Err(format!(
752                "{}: existing content differs from this source and encoding recipe; use --replace to refresh it",
753                target.display()
754            )),
755        };
756    }
757    publish(temporary, replace)?;
758    Ok(copy::Wrote::Copied)
759}
760
761fn publish(temporary: copy::TemporaryOutput, replace: bool) -> Result<(), String> {
762    if replace {
763        temporary.publish()
764    } else {
765        temporary.publish_new()
766    }
767}

769/// How long the source plays, read from the file rather than from the catalog.
770///
771/// The catalog would be quicker, and wrong the day the file changed without a
772/// scan: a verification that trusts a stale figure verifies nothing.
773fn source_duration(source: &Path) -> Option<u64> {
774    aede_core::tags::read(source)
775        .ok()
776        .and_then(|tags| tags.properties.duration_ms)
777}
