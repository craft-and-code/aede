1//! The `search` command: one query across every entity.
2//!
3//! Names only, by default. `--comments` widens it to the **comment** tag —
4//! where a rip came from, which pressing this is — `--notes` to what the user
5//! wrote themselves, and `--lyrics` to the words of the songs. All three are
6//! free prose, and a common word in any of them would bury the entity that
7//! actually bears the name, which is why none joins an ordinary search and why
8//! each keeps a section of its own: a hit found in a note was found by another
9//! route, and the reader has to be able to tell which.
10//!
11//! The three are not the same field and must not be folded together. A comment
12//! lives **inside the audio file**, put there by whoever tagged it; a note
13//! lives in `user.json`, put there by the person using Aède; the words are the
14//! song itself, and belong to nobody here. Searching one is searching the
15//! library, searching another is searching yourself.
16

27struct ParentWorkHit {
28    mbid: String,
29    title: String,
30    part_ids: BTreeSet<String>,
31}
32
33fn sourced_parent_works(
34    catalog: &Catalog,
35    held: &sources::Sources,
36    query: &str,
37) -> Vec<ParentWorkHit> {
38    let wanted = text::normalize(query);
39    let mut grouped: BTreeMap<String, ParentWorkHit> = BTreeMap::new();
40    for link in held
41        .work_parent_links(catalog)
42        .into_iter()
43        .filter(|link| link.trusted)
44    {
45        if link.parent.mbid != query
46            && (wanted.is_empty() || !text::normalize(&link.parent.title).contains(&wanted))
47        {
48            continue;
49        }
50        if catalog
51            .works
52            .iter()
53            .any(|work| work.mbid == link.parent.mbid)
54        {
55            continue;
56        }
57        let hit = grouped
58            .entry(link.parent.mbid.clone())
59            .or_insert_with(|| ParentWorkHit {
60                mbid: link.parent.mbid.clone(),
61                title: link.parent.title.clone(),
62                part_ids: BTreeSet::new(),
63            });
64        if hit.title.is_empty() && !link.parent.title.is_empty() {
65            hit.title = link.parent.title;
66        }
67        hit.part_ids.insert(link.child_mbid);
68    }
69    let mut hits: Vec<_> = grouped.into_values().collect();
70    hits.sort_by(|left, right| {
71        (left.mbid != query, text::normalize(&left.title), &left.mbid).cmp(&(
72            right.mbid != query,
73            text::normalize(&right.title),
74            &right.mbid,
75        ))
76    });
77    hits
78}

80pub fn search(args: &Args) -> Res {
81    let catalog = load(args)?;
82    let query = args.positionals.join(" ");
83    if query.trim().is_empty() {
84        return Err("give some text to search for".into());
85    }
86    let window = args.window(30)?;
87    // Keep the complete ranking until the requested shape is known. A track
88    // export has its own rows: paging mixed graph hits first could leave a
89    // playlist empty merely because albums occupied the chosen page.
90    let hits = catalog.search(&query, usize::MAX);
91    let sources = sources_held(args)?;
92    let sourced_parents = sourced_parent_works(&catalog, &sources, &query);
93    let mut sourced_contributors: Vec<SourcedContributor> =
94        contributors::sourced(&catalog, &sources)
95            .into_iter()
96            .filter(|person| {
97                person.mbid == query || contributors::matches_name(person, &query, false)
98            })
99            .collect();
100    sourced_contributors.sort_by(|left, right| {
101        (
102            left.mbid != query,
103            aede_core::text::normalize(&left.name),
104            &left.mbid,
105        )
106            .cmp(&(
107                right.mbid != query,
108                aede_core::text::normalize(&right.name),
109                &right.mbid,
110            ))
111    });
112
113    // Kept apart from the hits above rather than merged into them: a hit found
114    // in a comment was found by another route, and the reader has to be able to
115    // tell which. The same reason an imported analysis sits in its own panel.
116    let in_comments: Vec<Id> = if args.has("comments") {
117        catalog.tracks_with_comment(&query)
118    } else {
119        Vec::new()
120    };
121
122    // The words, with the line that carried the text rather than the song:
123    // "that one that goes something about a train" is answered by showing the
124    // line, in the shape it was half-remembered.
125    let in_lyrics: Vec<(Id, String)> = if args.has("lyrics") {
126        catalog.tracks_with_lyric(&query)
127    } else {
128        Vec::new()
129    };
130
131    // A note can be about anything — a label, a genre, an artist — so what
132    // comes back is not a list of tracks the way a comment hit is. It is shown
133    // as what it is.
134    let in_notes: Vec<(aede_core::user::EntityRef, String)> = if args.has("notes") {
135        notes_matching(args, &catalog, &query)?
136    } else {
137        Vec::new()
138    };
139
140    // Only the tracks: an artist or an album is not something to play, nor a
141    // row in a table of tracks. Comment hits are tracks, so they join in.
142    let mut ids: Vec<Id> = hits
143        .iter()
144        .filter(|h| h.kind == EntityKind::Track)
145        .map(|h| h.id)
146        .collect();
147    let mut included: BTreeSet<Id> = ids.iter().copied().collect();
148    for &id in in_comments.iter().chain(in_lyrics.iter().map(|(id, _)| id)) {
149        if included.insert(id) {
150            ids.push(id);
151        }
152    }
153    // Same reason as `track`: the JSON branch below returns on its own,
154    // before it would ever reach the shared gate inside `selection_output`,
155    // so the conflict has to be caught here too or `--json --csv` would
156    // print the JSON and drop the CSV without a word.
157    if let Some(message) = args.output_conflict(&["m3u", "csv", "json"]) {
158        return Err(message.into());
159    }
160    // A command with a JSON shape of its own answers first: `search --json`
161    // reports the hits — artists and albums included — which is a better answer
162    // than the flat track table the shared selection path would give.
163    if args.has("json") {
164        return print_json(
165            &catalog,
166            &hits,
167            &sourced_contributors,
168            &sourced_parents,
169            &in_comments,
170            &in_lyrics,
171            &in_notes,
172            args,
173        );
174    }
175    let ids: Vec<Id> = ids
176        .into_iter()
177        .skip(window.offset)
178        .take(window.limit)
179        .collect();
180    if let Some(result) = selection_output(&catalog, &ids, args) {
181        return result;
182    }
183    super::refuse_output_without_a_format(args)?;
184
185    println!("{}", ui::section(&format!("Results for \"{query}\"")));
186    if hits.is_empty() && sourced_contributors.is_empty() && sourced_parents.is_empty() {
187        // Symmetric with the three sections below it, each of which says what
188        // it did not find rather than falling back to the generic "(no
189        // results)" a reader could otherwise mistake for nothing at all
190        // having matched anywhere.
191        println!("  {}", ui::dim("nothing by name"));
192    } else if !hits.is_empty() {
193        let mut t = Table::new(&["Type", "Name", "Context", "Open"])
194            .limit(1, 32)
195            .limit(2, 22)
196            .limit(3, 64);
197        for hit in hits.iter().skip(window.offset).take(window.limit) {
198            // "release" is the model's word; "album" is the user's. On screen
199            // the user's wins — the JSON keeps the model's, for a client that
200            // has to map it back onto a table.
201            let kind = match hit.kind {
202                EntityKind::Artist => "artist",
203                EntityKind::Release => "album",
204                EntityKind::Track => "track",
205                EntityKind::Recording => "recording",
206                EntityKind::Work => "work",
207                EntityKind::ReleaseGroup => "release group",
208                EntityKind::Label => "label",
209                EntityKind::Genre => "genre",
210            };
211            t.push(vec![
212                kind.to_string(),
213                hit.name.clone(),
214                hit.detail.clone(),
215                super::navigation::open_command(&catalog, hit.kind, hit.id).unwrap_or_default(),
216            ]);
217        }
218        print!("{}", t.render());
219        announce_window(window, hits.len(), "name match");
220    }
221
222    if !sourced_contributors.is_empty() {
223        println!(
224            "{}",
225            ui::section("Sourced contributors (not in local artist tags)")
226        );
227        let mut table = Table::new(&["Name", "MusicBrainz ID", "Recordings", "Open"])
228            .align(2, crate::ui::Align::Right)
229            .limit(0, 30)
230            .limit(3, 65);
231        for person in sourced_contributors
232            .iter()
233            .skip(window.offset)
234            .take(window.limit)
235        {
236            table.push(vec![
237                person.name.clone(),
238                person.mbid.clone(),
239                person.recording_ids.len().to_string(),
240                format!("aede artist {}", super::navigation::shell_arg(&person.mbid)),
241            ]);
242        }
243        print!("{}", table.render());
244        announce_window(window, sourced_contributors.len(), "sourced contributor");
245    }
246
247    if !sourced_parents.is_empty() {
248        println!(
249            "{}",
250            ui::section("Sourced parent works (locally held parts)")
251        );
252        let mut table = Table::new(&["Work", "MusicBrainz ID", "Parts", "Open"])
253            .align(2, crate::ui::Align::Right)
254            .limit(0, 34)
255            .limit(3, 65);
256        for work in sourced_parents
257            .iter()
258            .skip(window.offset)
259            .take(window.limit)
260        {
261            table.push(vec![
262                work.title.clone(),
263                work.mbid.clone(),
264                work.part_ids.len().to_string(),
265                format!("aede work {}", super::navigation::shell_arg(&work.mbid)),
266            ]);
267        }
268        print!("{}", table.render());
269        announce_window(window, sourced_parents.len(), "sourced parent work");
270    }
271
272    if args.has("comments") {
273        print_comment_hits(&catalog, &in_comments, window);
274    }
275    if args.has("lyrics") {
276        print_lyric_hits(&catalog, &in_lyrics, window);
277    }
278    if args.has("notes") {
279        print_note_hits(&catalog, &in_notes, window);
280    }
281    Ok(())
282}

284/// What the user wrote, wherever the text appears in it.
285///
286/// Accent- and case-insensitive through `text::normalize`, like every other
287/// search in the program: somebody who wrote "pressage vinyle" must find it by
288/// typing "Vinyle".
289fn notes_matching(
290    args: &Args,
291    catalog: &Catalog,
292    query: &str,
293) -> Result<Vec<(aede_core::user::EntityRef, String)>, Box<dyn std::error::Error>> {
294    let data = super::user_data(args, catalog)?;
295    let wanted = aede_core::text::normalize(query);
296    let mut found: Vec<(aede_core::user::EntityRef, String)> = data
297        .annotations
298        .iter()
299        .filter(|a| a.owner == aede_core::user::LOCAL_USER)
300        .filter_map(|a| a.note.as_ref().map(|note| (a.target.clone(), note.clone())))
301        .filter(|(_, note)| aede_core::text::normalize(note).contains(&wanted))
302        .collect();
303    // Sorted so two runs agree, and so the kinds group together on screen.
304    found.sort_by(|a, b| (a.0.kind, &a.0.key).cmp(&(b.0.kind, &b.0.key)));
305    Ok(found)
306}

308/// The notes carrying the text, in their own section.
309fn print_note_hits(
310    catalog: &Catalog,
311    notes: &[(aede_core::user::EntityRef, String)],
312    window: Window,
313) {
314    if notes.is_empty() {
315        println!("  {}", ui::dim("nothing in your notes"));
316        return;
317    }
318    println!("{}", ui::section("In your notes"));
319    let mut t = Table::new(&["Kind", "Name", "Note"])
320        .limit(1, 30)
321        .limit(2, 50);
322    for (reference, note) in notes.iter().skip(window.offset).take(window.limit) {
323        t.push(vec![
324            match reference.kind {
325                EntityKind::Release => "album".to_string(),
326                other => other.as_str().to_string(),
327            },
328            reference.display_name(catalog),
329            note.replace('\n', " "),
330        ]);
331    }
332    print!("{}", t.render());
333    announce_window(window, notes.len(), "note");
334}

336/// Machine-readable form. Every row says **where** it was found, so a client
337/// can tell a name match from a comment match without guessing.
338fn print_json(
339    catalog: &Catalog,
340    hits: &[aede_core::model::SearchHit],
341    sourced_contributors: &[SourcedContributor],
342    sourced_parents: &[ParentWorkHit],
343    in_comments: &[Id],
344    in_lyrics: &[(Id, String)],
345    in_notes: &[(aede_core::user::EntityRef, String)],
346    args: &Args,
347) -> Res {
348    let window = args.window(30)?;
349    let mut rows: Vec<Json> = hits
350        .iter()
351        .skip(window.offset)
352        .take(window.limit)
353        .map(|h| {
354            let mut o = Json::obj();
355            o.set("type", h.kind.as_str().into());
356            o.set("id", h.id.into());
357            o.set("name", h.name.clone().into());
358            o.set("context", h.detail.clone().into());
359            o.set("found_in", "name".to_string().into());
360            o
361        })
362        .collect();
363    for person in sourced_contributors
364        .iter()
365        .skip(window.offset)
366        .take(window.limit)
367    {
368        let mut row = Json::obj();
369        row.set("type", "artist".into());
370        row.set("id", Json::Null);
371        row.set("musicbrainz_id", person.mbid.clone().into());
372        row.set("name", person.name.clone().into());
373        row.set(
374            "context",
375            format!(
376                "{} credited local recordings · sourced",
377                person.recording_ids.len()
378            )
379            .into(),
380        );
381        row.set("found_in", "source_credit".into());
382        row.set(
383            "open",
384            format!("aede artist {}", super::navigation::shell_arg(&person.mbid)).into(),
385        );
386        rows.push(row);
387    }
388    for work in sourced_parents
389        .iter()
390        .skip(window.offset)
391        .take(window.limit)
392    {
393        let mut row = Json::obj();
394        row.set("type", "work".into());
395        row.set("id", Json::Null);
396        row.set("musicbrainz_id", work.mbid.clone().into());
397        row.set("name", work.title.clone().into());
398        row.set(
399            "context",
400            format!("{} locally held parts · sourced", work.part_ids.len()).into(),
401        );
402        row.set("found_in", "source_work_part".into());
403        row.set(
404            "open",
405            format!("aede work {}", super::navigation::shell_arg(&work.mbid)).into(),
406        );
407        rows.push(row);
408    }
409    for &id in in_comments.iter().skip(window.offset).take(window.limit) {
410        let Some(track) = catalog.track(id) else {
411            continue;
412        };
413        let mut o = Json::obj();
414        o.set("type", EntityKind::Track.as_str().into());
415        o.set("id", id.into());
416        o.set("name", track.title.clone().into());
417        o.set(
418            "context",
419            catalog
420                .comment_of_track(id)
421                .unwrap_or_default()
422                .to_string()
423                .into(),
424        );
425        o.set("found_in", "comment".to_string().into());
426        rows.push(o);
427    }
428    for (id, line) in in_lyrics.iter().skip(window.offset).take(window.limit) {
429        let Some(track) = catalog.track(*id) else {
430            continue;
431        };
432        let mut o = Json::obj();
433        o.set("type", EntityKind::Track.as_str().into());
434        o.set("id", (*id).into());
435        o.set("name", track.title.clone().into());
436        // The line that matched, as on screen: a client that wanted the whole
437        // song would ask for the track, not for a search result.
438        o.set("context", line.clone().into());
439        o.set("found_in", "lyrics".to_string().into());
440        rows.push(o);
441    }
442    // A note can land on anything the catalog holds, not only a track — the
443    // JSON says what kind it landed on, same as every other row here.
444    for (reference, note) in in_notes.iter().skip(window.offset).take(window.limit) {
445        let Some(id) = reference.resolve(catalog) else {
446            continue;
447        };
448        let mut o = Json::obj();
449        o.set("type", reference.kind.as_str().into());
450        o.set("id", id.into());
451        o.set("name", reference.display_name(catalog).into());
452        o.set("context", note.clone().into());
453        o.set("found_in", "note".to_string().into());
454        rows.push(o);
455    }
456    super::export::emit(args, &Json::Arr(rows).to_string_pretty())
457}

459/// The tracks whose comment carries the text, in their own section.
460/// The tracks whose words carry the text, in their own section.
461///
462/// One line per hit, not one song: a cell holding four hundred lines is a table
463/// nobody can read, and the line is what was being looked for.
464fn print_lyric_hits(catalog: &Catalog, found: &[(Id, String)], window: Window) {
465    if found.is_empty() {
466        println!("  {}", ui::dim("nothing in the lyrics"));
467        return;
468    }
469    println!("{}", ui::section("In lyrics"));
470    let mut t = Table::new(&["Track", "Album", "Line"])
471        .limit(0, 30)
472        .limit(1, 25)
473        .limit(2, 45);
474    for (id, line) in found.iter().skip(window.offset).take(window.limit) {
475        let Some(track) = catalog.track(*id) else {
476            continue;
477        };
478        let album = track
479            .release_id
480            .and_then(|r| catalog.release(r))
481            .map(|r| r.title.clone())
482            .unwrap_or_default();
483        t.push(vec![track.title.clone(), album, line.clone()]);
484    }
485    print!("{}", t.render());
486    announce_window(window, found.len(), "line");
487}

489fn print_comment_hits(catalog: &Catalog, tracks: &[Id], window: Window) {
490    if tracks.is_empty() {
491        println!("  {}", ui::dim("nothing in the comments"));
492        return;
493    }
494    println!("{}", ui::section("In comments"));
495    let mut t = Table::new(&["Track", "Album", "Comment"])
496        .limit(0, 30)
497        .limit(1, 25)
498        .limit(2, 45);
499    for &id in tracks.iter().skip(window.offset).take(window.limit) {
500        let Some(track) = catalog.track(id) else {
501            continue;
502        };
503        let album = track
504            .release_id
505            .and_then(|r| catalog.release(r))
506            .map(|r| r.title.clone())
507            .unwrap_or_default();
508        t.push(vec![
509            track.title.clone(),
510            album,
511            catalog.comment_of_track(id).unwrap_or_default().to_string(),
512        ]);
513    }
514    print!("{}", t.render());
515    announce_window(window, tracks.len(), "comment");
516}
