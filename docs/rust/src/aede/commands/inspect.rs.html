1//! Reading a single file, and exporting the catalog.

3use std::path::Path;
4
5use aede_core::tags;
6use aede_core::text;
7
8use super::Res;
9use crate::args::Args;
10use crate::ui::{self, Table};
11
12/// Inspects a single file, without going through the catalog. Handy to
13/// understand why a file ends up misfiled.
14pub fn inspect(args: &Args) -> Res {
15    let Some(raw) = args.positionals.first() else {
16        return Err("give a file: aede file track.flac".into());
17    };
18    let path = Path::new(raw);
19    let tags = tags::read(path)?;
20    let size = std::fs::metadata(path).map(|m| m.len()).unwrap_or(0);
21
22    println!("{}", ui::section(raw));
23    let p = &tags.properties;
24    let mut t = Table::plain(2);
25    t.push(vec!["Container".into(), p.container.clone()]);
26    t.push(vec!["Codec".into(), p.codec.clone()]);
27    t.push(vec!["Quality".into(), p.quality_label()]);
28    t.push(vec![
29        "Sample rate".into(),
30        p.sample_rate
31            .map(|r| format!("{r} Hz"))
32            .unwrap_or_else(|| "—".into()),
33    ]);
34    t.push(vec![
35        "Bit depth".into(),
36        p.bit_depth
37            .map(|b| format!("{b} bits"))
38            .unwrap_or_else(|| "—".into()),
39    ]);
40    t.push(vec![
41        "Channels".into(),
42        p.channels
43            .map(|c| c.to_string())
44            .unwrap_or_else(|| "—".into()),
45    ]);
46    t.push(vec![
47        "Duration".into(),
48        p.duration_ms
49            .map(text::format_duration)
50            .unwrap_or_else(|| "—".into()),
51    ]);
52    t.push(vec![
53        "Bitrate".into(),
54        p.bitrate_kbps
55            .map(|b| format!("{b} kbps"))
56            .unwrap_or_else(|| "—".into()),
57    ]);
58    t.push(vec![
59        "Lossless".into(),
60        if p.lossless { "yes" } else { "no" }.into(),
61    ]);
62    t.push(vec![
63        "Embedded cover art".into(),
64        if tags.has_embedded_art { "yes" } else { "no" }.into(),
65    ]);
66    t.push(vec!["Size".into(), text::format_size(size)]);
67    print!("{}", t.render());
68
69    println!("{}", ui::section("Tags"));
70    if tags.is_empty() {
71        println!("  {}", ui::yellow("no tag in this file"));
72    } else {
73        let mut t = Table::new(&["Field", "Value"]).limit(1, 70);
74        for (key, values) in &tags.fields {
75            t.push(vec![key.clone(), values.join(" / ")]);
76        }
77        print!("{}", t.render());
78    }
79    Ok(())
80}
