114fn ensure_query_values(parsed: &query::Query, context: &query::Context<'_>) -> Res {
115    if let Some((what, value)) = query::unknown_values(parsed, context).first() {
116        return Err(
117            format!("no {what} matches \"{value}\".\nRun \"aede {what}s\" for the list.").into(),
118        );
119    }
120    Ok(())
121}

123fn quoted_query_value(value: &str) -> String {
124    if value.contains(char::is_whitespace) {
125        format!("\"{}\"", value.replace('"', ""))
126    } else {
127        value.to_string()
128    }

131/// Asks before something irreversible, unless `--yes` was given.
132///
133/// With no terminal to ask on — a script, a pipe — it refuses rather than
134/// assuming an answer. Assuming "no" would make a scripted run fail silently;
135/// assuming "yes" would destroy something nobody agreed to lose.
136///
137/// `what` completes both the question and the refusal, so a message names the
138/// act rather than talking about "the operation". Shared rather than copied:
139/// it lived in `reset` alone until `history --remove` needed the same
140/// question, and a second confirmation prompt worded slightly differently is
141/// how a user learns to stop reading them.
142pub fn confirmed(args: &Args, what: &str) -> Result<bool, Box<dyn Error>> {
143    use std::io::{IsTerminal, Write};
144    if args.has("yes") {
145        return Ok(true);
146    }
147    let delegated_terminal = std::env::var_os("AEDE_DELEGATED_CHILD").is_some()
148        && std::env::var_os("AEDE_DELEGATED_STDIN_TTY").is_some();
149    if !std::io::stdin().is_terminal() && !delegated_terminal {
150        return Err(format!("no terminal to confirm on: add --yes to {what}").into());
151    }
152    print!("  Type \"yes\" to confirm: ");
153    std::io::stdout().flush()?;
154    let mut answer = String::new();
155    std::io::stdin().read_line(&mut answer)?;
156    Ok(answer.trim().eq_ignore_ascii_case("yes"))
157}

159/// A folder the user named, resolved the way the catalog stores folders.
160///
161/// **Every path that arrives from the command line and will be compared against
162/// a stored one goes through here.** The catalog keeps its watched roots
163/// canonical, and the comparisons that matter — is this file under that root,
164/// is this destination inside my library — are string comparisons on a
165/// separator boundary ([`aede_core::text::is_under`]). A path reached through a
166/// symbolic link names the same folder by a string that never compares equal,
167/// so the answer comes back "no" for a folder that plainly is.
168///
169/// On macOS this is not a corner case but the ordinary one: `/var` is a link to
170/// `/private/var` and `/tmp` to `/private/tmp`, so most paths exist in two
171/// spellings and only one of them is ever the catalog's.
172///
173/// The step was written out four times, slightly differently, in `scan`,
174/// `roots`, `check` and `copy` — and `copy`, the one command that *writes*, was
175/// the one that had left it out. Its "this destination is inside your library"
176/// refusal therefore waved through precisely the case it exists to catch. One
177/// helper, so that the fifth command cannot forget it.
178///
179/// A path the filesystem will not resolve comes back as it was: acting on the
180/// name given is better than refusing outright.
181pub fn canonical(path: &Path) -> PathBuf {
182    std::fs::canonicalize(path).unwrap_or_else(|_| path.to_path_buf())
183}

185/// Folders or files a run is restricted to, canonicalized. Empty means the
186/// whole catalog.
187///
188/// Shared by every command that takes `[folder…]` and acts on the files under
189/// it — `check`, `spectrum`, `playlist`, `extract`, `fingerprint`. They must
190/// agree on what "under this folder" means, and on macOS that agreement is not
191/// free: `/var` and `/private/var` name the same place by two strings that
192/// never compare equal, which is what [`canonical`] exists for.
193///
194/// **A folder the catalog knows nothing about is refused here**, which is why
195/// this takes the catalog. Every one of these commands works from the catalog
196/// rather than from the disk, so a folder that was never scanned produces a
197/// run with nothing to do — and each of them then says something cheerful and
198/// wrong: `aede extract <a folder added yesterday>` answered "nothing to
199/// extract", which reads as *your files already have their covers* and is in
200/// fact *I have never heard of that folder*. One reader lost an evening to it,
201/// and worked it out themselves.
202///
203/// It is an error rather than a warning, and refuses the whole run rather than
204/// dropping the unknown folder: a command answers the question it was asked or
205/// refuses, and quietly acting on two of the three folders named is the
206/// swallowed argument in another costume. Checked here rather than in five
207/// commands, so the sixth cannot forget it — the same reason [`canonical`]
208/// exists.
209pub fn scope_of(args: &Args, catalog: &Catalog) -> Result<Vec<String>, Box<dyn Error>> {
210    scope_from(&args.positionals, catalog)
211}
212
213/// [`scope_of`], for a command whose positionals are not all folders.
214///
215/// `fetch` takes names and folders in the same place — `aede fetch --lyrics
216/// alastis ~/Music/Alastis` is both — so it sorts them out itself and hands
217/// the folders here. Splitting the reading from the *rules* is the point: what
218/// "under this folder" means, and what happens to a folder the catalog has
219/// never heard of, are decided once for every command that takes one.
220pub fn scope_from(folders: &[String], catalog: &Catalog) -> Result<Vec<String>, Box<dyn Error>> {
221    let mut scope = Vec::new();
222    for raw in folders {
223        let path = Path::new(raw);
224        if !path.exists() {
225            return Err(format!("\"{raw}\" does not exist").into());
226        }
227        let resolved = canonical(path).to_string_lossy().to_string();
228        if !catalog
229            .files
230            .iter()
231            .any(|f| text::is_under(&f.path, &resolved))
232        {
233            return Err(unknown_folder(raw, catalog).into());
234        }
235        scope.push(resolved);
236    }
237    Ok(scope)
238}

240/// What to say about a folder that exists on disk and not in the catalog.
241///
242/// The age of the catalog is the fact that explains it, and it is given rather
243/// than the advice: "run a scan" is something a reader weighs at nothing, while
244/// "this catalog was built three days ago" is something they can weigh against
245/// what they did three days ago. The command that fixes it is spelled out with
246/// the folder already in it, because the next thing they will do is type it.
247fn unknown_folder(raw: &str, catalog: &Catalog) -> String {
248    let age = match catalog.scanned_at {
249        0 => "this catalog has never been scanned".to_string(),
250        at => format!("this catalog was scanned {}", ui::since(at)),
251    };
252    format!(
253        "no file in the catalog is under \"{raw}\".\n\
254         It is on disk, so {age} and has not seen it — a folder added since is \
255         not in it yet.\n\
256         Add it: aede scan \"{raw}\""
257    )
258}

260/// `true` when the path is inside one of the folders given, or is one of them.
261pub fn in_scope(path: &str, scope: &[String]) -> bool {
262    scope.is_empty() || scope.iter().any(|root| text::is_under(path, root))
263}

265/// The deepest folder holding every one of these releases, or `None`.
266///
267/// Not a thing the catalog holds — an artist folder is inferred, not scanned
268/// — so this is the one place that infers it, shared by `playlist`'s
269/// `--discography` mode and the artist artwork passes. Each writes a file into
270/// it. When albums do not share one folder, there is no destination to infer;
271/// callers also reject a watched root, so a flat library does not get a stray
272/// file in its root for every artist.
273pub(super) fn shared_folder(releases: &[&Release]) -> Option<PathBuf> {
274    let mut shared: Option<PathBuf> = None;
275    for release in releases {
276        let parent = Path::new(&release.folder).parent()?.to_path_buf();
277        shared = Some(match shared {
278            None => parent,
279            Some(so_far) if so_far == parent => so_far,
280            Some(_) => return None,
281        });
282    }
283    shared
284}

296fn artist_artwork_destination(
297    catalog: &Catalog,
298    releases: &[&Release],
299    data_dir: &Path,
300    mbid: &str,
301) -> PathBuf {
302    shared_folder(releases)

312/// Where the catalog lives: what `--data` names, or the default location.
313pub fn data_dir(args: &Args) -> PathBuf {
314    if std::env::var_os("AEDE_DELEGATED_CHILD").is_some()
315        && let Some(forced) = std::env::var_os("AEDE_DELEGATED_DATA_DIR")
316    {
317        return PathBuf::from(forced);
318    }
319    args.value("data")
320        .map(PathBuf::from)
321        .unwrap_or_else(store::default_data_dir)
322}

324fn load(args: &Args) -> Result<Catalog, Box<dyn Error>> {
325    let path = store::catalog_path(&data_dir(args));
326    match store::load(&path)? {
327        Some(catalog) => Ok(catalog),
328        None => Err(format!(
329            "no catalog in {}.\nRun this first: aede scan <folder>",
330            data_dir(args).display()
331        )
332        .into()),
333    }
334}

347/// What other sources said, as it sits on disk.
348///
349/// Loaded and not reconciled, unlike the annotations: the layer is keyed on
350/// what an entity calls itself and resolves on read, so a rebuilt catalog
351/// needs nothing rewritten here.
352/// What MusicBrainz said about this artist, when anything has been fetched.
353///
354/// Returns the facts rather than the record: every caller here wants the
355/// contents, and handing back the wrapper would have each of them repeat the
356/// same two `match` arms to get inside it.
357pub(super) fn artist_facts_for(
358    args: &Args,
359    catalog: &Catalog,
360    artist: Id,
361) -> Option<aede_core::sources::ArtistFacts> {
362    let held = sources_held(args).ok()?;
363    let entity =
364        aede_core::user::EntityRef::of(catalog, aede_core::model::EntityKind::Artist, artist)?;
365    let record = held.get(&entity, aede_core::sources::MUSICBRAINZ)?;
366    if !held.is_trusted(catalog, record) {
367        return None;
368    }
369    match &record.facts {
370        aede_core::sources::Facts::Artist(facts) => Some(facts.clone()),
371        _ => None,
372    }
373}

380/// The role vocabulary, as `(key stored, name shown)`.
381///
382/// One table read in both directions. It used to be a one-way `match`, and the
383/// consequence was a message contradicting itself in a single breath: asked for
384/// `--role "album artist"` — the only spelling ever shown on screen — the
385/// program answered that the artist was *not* credited as album artist and then
386/// listed "album artist (14)" among their credits. The user can only type what
387/// they are shown; whatever is shown must therefore be accepted.
388const ROLE_NAMES: &[(&str, &str)] = &[
389    ("main", "main artist"),
390    ("album", "album artist"),
391    ("composer", "composer"),
392    ("conductor", "conductor"),
393    ("remixer", "remixer"),
394    ("lyricist", "lyricist"),
395    ("performer", "performer"),
396    ("producer", "producer"),
397    ("engineer", "engineer"),
398    ("featured", "featured"),
399];
400
401/// Display name for an internal role key; unknown keys are shown as they are.
402fn role_label(role: &str) -> String {
403    ROLE_NAMES
404        .iter()
405        .find(|(key, _)| *key == role)
406        .map(|(_, label)| (*label).to_string())
407        .unwrap_or_else(|| role.to_string())
408}

410/// The key behind whatever the user typed, or `None` if it names no role.
411///
412/// Accepts the stored key and the displayed name alike, up to case and
413/// accents. It looks first at the roles the catalog **holds** — so a role
414/// arriving from MusicBrainz at M1, absent from the table above, is reachable
415/// by its own name without a line of code — then at the known vocabulary, so
416/// that a real role nobody holds here can be told apart from a word that is no
417/// role at all. The two deserve different answers: one is an empty result, the
418/// other a misunderstanding.
419fn role_key(catalog: &Catalog, typed: &str) -> Option<String> {
420    let wanted = text::normalize(typed);
421    let in_use = catalog
422        .roles_in_use()
423        .into_iter()
424        .find(|key| text::normalize(key) == wanted || text::normalize(&role_label(key)) == wanted)
425        .map(|key| key.to_string());
426    in_use.or_else(|| {
427        ROLE_NAMES
428            .iter()
429            .find(|(key, label)| text::normalize(key) == wanted || text::normalize(label) == wanted)
430            .map(|(key, _)| (*key).to_string())
431    })
432}

434/// The roles a catalog holds, spelled the way they are shown.
435///
436/// What an error message offers has to be typeable back in: listing the stored

438fn roles_offered(catalog: &Catalog) -> String {

447/// Technical description of a stream, shown identically by `file` and by
448/// `track` — one is read from the disk, the other from the catalog, and the
449/// reader has no reason to see a difference.
450fn properties_table(properties: &AudioProperties, has_embedded_art: bool, size: u64) -> Table {
451    let mut t = Table::plain(2);
452    let dash = || "—".to_string();
453    t.push(vec!["Container".into(), properties.container.clone()]);
454    t.push(vec!["Codec".into(), properties.codec.clone()]);
455    t.push(vec!["Quality".into(), properties.quality_label()]);
456    t.push(vec![
457        "Sample rate".into(),
458        properties
459            .sample_rate
460            .map(|r| format!("{r} Hz"))
461            .unwrap_or_else(dash),
462    ]);
463    t.push(vec![
464        "Bit depth".into(),
465        properties
466            .bit_depth
467            .map(|b| format!("{b} bits"))
468            .unwrap_or_else(dash),
469    ]);
470    t.push(vec![
471        "Channels".into(),
472        properties
473            .channels
474            .map(|c| c.to_string())
475            .unwrap_or_else(dash),
476    ]);
477    t.push(vec![
478        "Duration".into(),
479        properties
480            .duration_ms
481            .map(text::format_duration)
482            .unwrap_or_else(dash),
483    ]);
484    t.push(vec![
485        "Bitrate".into(),
486        properties
487            .bitrate_kbps
488            .map(|b| format!("{b} kbps"))
489            .unwrap_or_else(dash),
490    ]);
491    t.push(vec![
492        "Lossless".into(),
493        if properties.lossless { "yes" } else { "no" }.into(),
494    ]);
495    t.push(vec![
496        "Embedded cover art".into(),
497        if has_embedded_art { "yes" } else { "no" }.into(),
498    ]);
499    if size > 0 {
500        t.push(vec!["Size".into(), text::format_size(size)]);
501    }
502    t
503}

514/// Prints attributed MusicBrainz credits in one consistent shape.
515fn print_sourced_credits(catalog: &Catalog, mut credits: Vec<core_sources::SourcedCreditLink>) {
516    if credits.is_empty() {
517        return;
518    }
519    credits.sort_by(|a, b| {
520        a.work
521            .as_ref()
522            .map(|work| work.title.as_str())
523            .cmp(&b.work.as_ref().map(|work| work.title.as_str()))
524            .then_with(|| a.credit.order.cmp(&b.credit.order))
525            .then_with(|| a.credit.role.cmp(&b.credit.role))
526            .then_with(|| a.credit.artist_name.cmp(&b.credit.artist_name))
527    });
528    println!("{}", ui::section("Sourced credits"));
529    let mut table = Table::new(&[
530        "Artist",
531        "Role",
532        "Credited as",
533        "Details",
534        "Scope",
535        "Dates",
536        "Evidence",
537    ])
538    .limit(0, 28)
539    .limit(2, 24)
540    .limit(3, 36)
541    .limit(4, 28)
542    .limit(6, 42);
543    for link in credits {
544        let details = link
545            .credit
546            .attributes
547            .iter()
548            .map(|attribute| {
549                let shown = attribute
550                    .credited_as
551                    .as_deref()
552                    .or(attribute.value.as_deref())
553                    .unwrap_or(&attribute.name);
554                match shown == attribute.name {
555                    true => shown.to_string(),
556                    false => format!("{} ({shown})", attribute.name),
557                }
558            })
559            .collect::<Vec<_>>()
560            .join(", ");
561        let recording = catalog
562            .recording(link.recording_id)
563            .map(|recording| recording.title.as_str())
564            .unwrap_or("unknown recording");
565        let scope = link.work.as_ref().map_or_else(
566            || format!("recording: {recording}"),
567            |work| format!("work: {} · recording: {recording}", work.title),
568        );
569        let mut dates = match (&link.credit.began, &link.credit.ended) {
570            (Some(begin), Some(end)) => format!("{begin}–{end}"),
571            (Some(begin), None) => format!("from {begin}"),
572            (None, Some(end)) => format!("until {end}"),
573            (None, None) => String::new(),
574        };
575        if link.credit.over == Some(true) && link.credit.ended.is_none() {
576            dates.push_str(match dates.is_empty() {
577                true => "ended",
578                false => " · ended",
579            });
580        }
581        let confidence = if link.excluded {
582            "excluded by you".to_string()
583        } else {
584            source_status(link.confidence, link.review, link.trusted)
585        };
586        let relation = link
587            .credit
588            .relation_id
589            .as_deref()
590            .or(link.credit.role_id.as_deref())
591            .map(|id| format!(" · {id}"))
592            .unwrap_or_default();
593        table.push(vec![
594            link.credit.artist_name,
595            role_label(&link.credit.role),
596            link.credit.credited_as.unwrap_or_default(),
597            details,
598            scope,
599            dates,
600            format!(
601                "{} · {confidence} · {}{relation}",
602                link.source,
603                ui::since(link.fetched_at)
604            ),
605        ]);
606    }
607    print!("{}", table.render());
608}

610fn print_sourced_edition_credits(
611    catalog: &Catalog,
612    mut credits: Vec<core_sources::SourcedEditionCreditLink>,
613) {
614    if credits.is_empty() {
615        return;
616    }
617    credits.sort_by(|a, b| {
618        a.credit
619            .order
620            .cmp(&b.credit.order)
621            .then_with(|| a.credit.role.cmp(&b.credit.role))
622            .then_with(|| a.credit.artist_name.cmp(&b.credit.artist_name))
623    });
624    println!("{}", ui::section("Edition credits from sources"));
625    let mut table = Table::new(&[
626        "Artist",
627        "Role",
628        "Credited as",
629        "Details",
630        "Edition",
631        "Evidence",
632    ])
633    .limit(0, 28)
634    .limit(2, 24)
635    .limit(3, 36)
636    .limit(4, 36)
637    .limit(5, 42);
638    for link in credits {
639        let details = link
640            .credit
641            .attributes
642            .iter()
643            .map(|attribute| attribute.name.as_str())
644            .collect::<Vec<_>>()
645            .join(", ");
646        let edition = catalog
647            .release(link.release_id)
648            .map(|release| release.title.as_str())
649            .unwrap_or("unknown edition");
650        table.push(vec![
651            link.credit.artist_name,
652            role_label(&link.credit.role),
653            link.credit.credited_as.unwrap_or_default(),
654            details,
655            edition.to_string(),
656            format!(
657                "{} · {} · {}",
658                link.source,
659                if link.excluded {
660                    "excluded by you".to_string()
661                } else {
662                    source_status(link.confidence, link.review, link.trusted)
663                },
664                ui::since(link.fetched_at)
665            ),
666        ]);
667    }
668    print!("{}", table.render());
669}

671/// Human reading of confidence after an explicit review has had its say.
672fn source_status(
673    confidence: core_sources::Confidence,
674    review: Option<core_sources::ReviewDecision>,
675    trusted: bool,
676) -> String {
677    let evidence = match confidence {
678        core_sources::Confidence::Identified => "identified".to_string(),
679        core_sources::Confidence::Matched(score) => format!("matched {score}%"),
680    };
681    match review {
682        Some(core_sources::ReviewDecision::Accepted) => format!("{evidence} · accepted"),
683        Some(core_sources::ReviewDecision::Rejected) => format!("{evidence} · rejected"),
684        None if !trusted => format!("{evidence} · pending review"),
685        None => evidence,
686    }
687}

689/// Playing time and size on disk of a set of tracks.
690///
691/// Count, duration and size are the three measures every listing and every
692/// entity page carries: what a part of the library weighs should not depend on
693/// the command used to look at it.
694fn totals(catalog: &Catalog, tracks: &[Id]) -> (u64, u64) {
695    tracks
696        .iter()
697        .filter_map(|&id| catalog.track(id))
698        .fold((0, 0), |(duration, size), track| {
699            (
700                duration + track.duration_ms.unwrap_or(0),
701                size + catalog.file(track.file_id).map(|f| f.size).unwrap_or(0),
702            )
703        })
704}

706fn print_track_totals(catalog: &Catalog, tracks: &[Id]) {
707    let (duration, size) = totals(catalog, tracks);
708    println!(
709        "  {}",
710        ui::dim(&format!(
711            "{} · {} · {}",
712            ui::plural(tracks.len(), "track"),
713            text::format_duration(duration),
714            text::format_size(size)
715        ))
716    );
717}

796/// Says that a listing was cut, and what to do about it.
797///
798/// Every listing stops at `--limit` rows so as not to fill the terminal, and
799/// all five of them used to stop **in silence**. A cut list reads as "that is
800/// all there is", which is the one thing it is not: an album sitting past the
801/// fiftieth row simply did not exist as far as the user could tell. The header
802/// counting the matches is not enough — nobody compares it against the rows
803/// they were given.
804///
805/// Nothing is printed when everything was shown, so the notice keeps meaning
806/// something.
807fn announce_window(window: Window, total: usize, what: &str) {
808    if let Some(said) = window_note(window, total, what) {
809        println!("  {}", ui::yellow(&said));
810    }
811}
812
813/// The sentence [`announce_window`] prints, or `None` when there is nothing to
814/// say.
815///
816/// Split out of the printing because what it decides is not obvious and was
817/// wrong: the *advice* at the end has to follow from the window, and a line
818/// that offers a page which is not there is worse than no line. `aede doctor
819/// --offset=25 --all` answered `26–52 of 52 issues — --offset=52 for the next
820/// page`, and `--offset=52` leads to "starts past the end". The command had
821/// done exactly what it was asked; the sentence under it had not.
822fn window_note(window: Window, total: usize, what: &str) -> Option<String> {
823    let Some((first, last)) = window.shown(total) else {
824        // Two different emptinesses, and naming the wrong one sends the reader
825        // looking for a page that was never there. `--offset` explains an empty
826        // screen only when there was something to page through; a listing that
827        // matched nothing at all is not a paging accident. The confusion became
828        // easy to meet the day the listings learned `--query`: `aede artists
829        // --query "year:2050"` answered "0 artist in all, and --offset=0 starts
830        // past the end", which blames a page number nobody typed.
831        return Some(match total {
832            0 => format!("nothing here: no {what} to show"),
833            _ => format!(
834                "nothing here: {} in all, and --offset={} starts past the end",
835                ui::plural(total, what),
836                window.offset
837            ),
838        });
839    };
840    if first == 1 && last == total {
841        return None;
842    }
843
844    // Two questions, and only the answers that are true of *this* screen.
845    //
846    // **A page is offered only when it exists.** The last row shown being the
847    // last row there is means there is no next page, and `--offset={last}` then
848    // names the one number guaranteed to answer "starts past the end". A reader
849    // on the last screen wants to be told that they are, not sent onward.
850    //
851    // **`--all` is advice only while it would change something.** Given
852    // already, the rows are unbounded and what shaped this screen was
853    // `--offset` alone; repeating it names an option the reader has typed and
854    // tells them to type it, which is how a reader learns to stop reading these
855    // lines.
856    let unbounded = window.limit == usize::MAX;
857    let onward = match (last < total, unbounded) {
858        (true, true) => format!("--offset={last} for the next page"),
859        (true, false) => format!("--offset={last} for the next page, --all for every row"),
860        // On the last screen `--all` is not advice either, whatever the limit:
861        // what cut this one is the offset, and lifting the limit would show the
862        // same rows again. The only move left is back to the start.
863        (false, _) => "these are the last; drop --offset to start from the first".to_string(),
864    };
865    Some(format!(
866        "{first}–{last} of {} — {onward}",
867        ui::plural(total, what)
868    ))
869}

871/// Marker put after an album title when the same album sits elsewhere too.
872///
873/// Two words rather than a symbol: a legend nobody reads is worse than four
874/// characters of prose, and the two cases call for opposite reactions.
875fn copy_marker(catalog: &Catalog, release_id: Id) -> String {
876    use aede_core::model::{DUPLICATE, OTHER_EDITION};
877    if !catalog.related_releases(release_id, DUPLICATE).is_empty() {
878        return " (duplicate)".to_string();
879    }
880    if !catalog
881        .related_releases(release_id, OTHER_EDITION)
882        .is_empty()
883    {
884        return " (other edition)".to_string();
885    }
886    String::new()
887}

889/// Hands the tracks on screen to another program, when asked for.
890///
891/// `None` means nothing was asked and the command should print its usual page.
892/// Every command that shows a track list offers the same two exits, so the
893/// option means the same thing wherever it is typed.
894fn selection_output(catalog: &Catalog, tracks: &[Id], args: &Args) -> Option<Res> {
895    // The one gate every one of these formats passes through, whichever
896    // command got them here: `--m3u --csv` used to hand back the playlist and
897    // drop the CSV without a word, because this function looked at `--m3u`
898    // first and never noticed the other flag was there at all.
899    if let Some(message) = args.output_conflict(&["m3u", "csv", "json"]) {
900        return Some(Err(message.into()));
901    }
902    if args.has("m3u") {
903        return Some(play_list(catalog, tracks, args));
904    }
905    if args.has("csv") || args.has("json") {
906        if tracks.is_empty() {
907            return Some(Err("nothing to put in a table".into()));
908        }
909        return Some(export::tracks_table(catalog, tracks, args));
910    }
911    None
912}

914/// Refuses `--output` on a page that has no file to write it to.
915///
916/// `-o`/`--output` only ever writes something through [`selection_output`] or
917/// a listing's `--csv`/`--json` table — everywhere else is a page meant for
918/// the screen, and until now `-o` vanished in front of one without a word.
919/// That is exactly the "wrong answer standing in for a missing one" this
920/// program refuses everywhere else an option cannot be honoured, and the rule
921/// stated at the top of `docs/commands.md`: an option a command cannot
922/// honour is refused, never ignored.
923pub fn refuse_output_without_a_format(args: &Args) -> Res {
924    if args.has("output") {
925        return Err(
926            "--output writes what --csv, --json or --m3u produce; this page has none of those to give it.\n\
927             Drop --output to see it on screen, or add one of the three to write it out."
928                .into(),
929        );
930    }
931    Ok(())
932}

934/// Prints the tracks shown as an M3U playlist instead of the usual page.
935///
936/// Every command that puts a track list on screen can hand it to a player;
937/// that the selection was reached through an album, an artist or a search is
938/// the caller's business, not the playlist's.
939fn play_list(catalog: &Catalog, tracks: &[Id], args: &Args) -> Res {
940    if tracks.is_empty() {
941        return Err("nothing to put in a playlist".into());
942    }
943    export::emit(args, &export::m3u(catalog, tracks))
944}
