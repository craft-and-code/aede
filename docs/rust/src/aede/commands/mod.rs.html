116fn ensure_query_values(parsed: &query::Query, context: &query::Context<'_>) -> Res {
117    if let Some((what, value)) = query::unknown_values(parsed, context).first() {
118        return Err(
119            format!("no {what} matches \"{value}\".\nRun \"aede {what}s\" for the list.").into(),
120        );
121    }
122    Ok(())
123}

125fn quoted_query_value(value: &str) -> String {
126    if value.contains(char::is_whitespace) {
127        format!("\"{}\"", value.replace('"', ""))
128    } else {
129        value.to_string()
130    }
131}

133/// Asks before something irreversible, unless `--yes` was given.
134///
135/// With no terminal to ask on — a script, a pipe — it refuses rather than
136/// assuming an answer. Assuming "no" would make a scripted run fail silently;
137/// assuming "yes" would destroy something nobody agreed to lose.
138///
139/// `what` completes both the question and the refusal, so a message names the
140/// act rather than talking about "the operation". Shared rather than copied:
141/// it lived in `reset` alone until `history --remove` needed the same
142/// question, and a second confirmation prompt worded slightly differently is
143/// how a user learns to stop reading them.
144pub fn confirmed(args: &Args, what: &str) -> Result<bool, Box<dyn Error>> {
145    use std::io::{IsTerminal, Write};
146    if args.has("yes") {
147        return Ok(true);
148    }
149    let delegated_terminal = std::env::var_os("AEDE_DELEGATED_CHILD").is_some()
150        && std::env::var_os("AEDE_DELEGATED_STDIN_TTY").is_some();
151    if !std::io::stdin().is_terminal() && !delegated_terminal {
152        return Err(format!("no terminal to confirm on: add --yes to {what}").into());
153    }
154    print!("  Type \"yes\" to confirm: ");
155    std::io::stdout().flush()?;
156    let mut answer = String::new();
157    std::io::stdin().read_line(&mut answer)?;
158    Ok(answer.trim().eq_ignore_ascii_case("yes"))
159}

161/// A folder the user named, resolved the way the catalog stores folders.
162///
163/// **Every path that arrives from the command line and will be compared against
164/// a stored one goes through here.** The catalog keeps its watched roots
165/// canonical, and the comparisons that matter — is this file under that root,
166/// is this destination inside my library — are string comparisons on a
167/// separator boundary ([`aede_core::text::is_under`]). A path reached through a
168/// symbolic link names the same folder by a string that never compares equal,
169/// so the answer comes back "no" for a folder that plainly is.
170///
171/// On macOS this is not a corner case but the ordinary one: `/var` is a link to
172/// `/private/var` and `/tmp` to `/private/tmp`, so most paths exist in two
173/// spellings and only one of them is ever the catalog's.
174///
175/// The step was written out four times, slightly differently, in `scan`,
176/// `roots`, `check` and `copy` — and `copy`, the one command that *writes*, was
177/// the one that had left it out. Its "this destination is inside your library"
178/// refusal therefore waved through precisely the case it exists to catch. One
179/// helper, so that the fifth command cannot forget it.
180///
181/// A path the filesystem will not resolve comes back as it was: acting on the
182/// name given is better than refusing outright.
183pub fn canonical(path: &Path) -> PathBuf {
184    std::fs::canonicalize(path).unwrap_or_else(|_| path.to_path_buf())
185}
186
187/// Converts a resolved path to a catalog key without changing native spelling.
188///
189/// A symlink's target need not be UTF-8 even when its command-line name is.
190/// Refusing it avoids comparing or saving the key of a different path.
191fn path_key(path: &Path) -> Result<String, Box<dyn Error>> {
192    path.to_str()
193        .map(str::to_owned)
194        .ok_or_else(|| format!("{path:?}: path cannot be represented as UTF-8").into())
195}

197/// Folders or files a run is restricted to, canonicalized. Empty means the
198/// whole catalog.
199///
200/// Shared by every command that takes `[folder…]` and acts on the files under
201/// it — `check`, `spectrum`, `playlist`, `extract`, `fingerprint`. They must
202/// agree on what "under this folder" means, and on macOS that agreement is not
203/// free: `/var` and `/private/var` name the same place by two strings that
204/// never compare equal, which is what [`canonical`] exists for.
205///
206/// **A folder the catalog knows nothing about is refused here**, which is why
207/// this takes the catalog. Every one of these commands works from the catalog
208/// rather than from the disk, so a folder that was never scanned produces a
209/// run with nothing to do — and each of them then says something cheerful and
210/// wrong: `aede extract <a folder added yesterday>` answered "nothing to
211/// extract", which reads as *your files already have their covers* and is in
212/// fact *I have never heard of that folder*. One reader lost an evening to it,
213/// and worked it out themselves.
214///
215/// It is an error rather than a warning, and refuses the whole run rather than
216/// dropping the unknown folder: a command answers the question it was asked or
217/// refuses, and quietly acting on two of the three folders named is the
218/// swallowed argument in another costume. Checked here rather than in five
219/// commands, so the sixth cannot forget it — the same reason [`canonical`]
220/// exists.
221pub fn scope_of(args: &Args, catalog: &Catalog) -> Result<Vec<String>, Box<dyn Error>> {
222    scope_from(&args.positionals, catalog)
223}
224
225/// [`scope_of`], for a command whose positionals are not all folders.
226///
227/// `fetch` takes names and folders in the same place — `aede fetch --lyrics
228/// alastis ~/Music/Alastis` is both — so it sorts them out itself and hands
229/// the folders here. Splitting the reading from the *rules* is the point: what
230/// "under this folder" means, and what happens to a folder the catalog has
231/// never heard of, are decided once for every command that takes one.
232pub fn scope_from(folders: &[String], catalog: &Catalog) -> Result<Vec<String>, Box<dyn Error>> {
233    let mut scope = Vec::new();
234    for raw in folders {
235        let path = Path::new(raw);
236        if !path.exists() {
237            return Err(format!("\"{raw}\" does not exist").into());
238        }
239        let resolved = path_key(&canonical(path))?;
240        if !catalog
241            .files
242            .iter()
243            .any(|f| text::is_under(&f.path, &resolved))
244        {
245            return Err(unknown_folder(raw, catalog).into());
246        }
247        scope.push(resolved);
248    }
249    Ok(scope)
250}

252/// What to say about a folder that exists on disk and not in the catalog.
253///
254/// The age of the catalog is the fact that explains it, and it is given rather
255/// than the advice: "run a scan" is something a reader weighs at nothing, while
256/// "this catalog was built three days ago" is something they can weigh against
257/// what they did three days ago. The command that fixes it is spelled out with
258/// the folder already in it, because the next thing they will do is type it.
259fn unknown_folder(raw: &str, catalog: &Catalog) -> String {
260    let age = match catalog.scanned_at {
261        0 => "this catalog has never been scanned".to_string(),
262        at => format!("this catalog was scanned {}", ui::since(at)),
263    };
264    format!(
265        "no file in the catalog is under \"{raw}\".\n\
266         It is on disk, so {age} and has not seen it — a folder added since is \
267         not in it yet.\n\
268         Add it: aede scan \"{raw}\""
269    )
270}

272/// `true` when the path is inside one of the folders given, or is one of them.
273pub fn in_scope(path: &str, scope: &[String]) -> bool {
274    scope.is_empty() || scope.iter().any(|root| text::is_under(path, root))
275}

277/// The deepest folder holding every one of these releases, or `None`.
278///
279/// Not a thing the catalog holds — an artist folder is inferred, not scanned
280/// — so this is the one place that infers it, shared by `playlist`'s
281/// `--discography` mode and the artist artwork passes. Each writes a file into
282/// it. When albums do not share one folder, there is no destination to infer;
283/// callers also reject a watched root, so a flat library does not get a stray
284/// file in its root for every artist.
285pub(super) fn shared_folder(releases: &[&Release]) -> Option<PathBuf> {
286    let mut shared: Option<PathBuf> = None;
287    for release in releases {
288        let parent = Path::new(&release.folder).parent()?.to_path_buf();
289        shared = Some(match shared {
290            None => parent,
291            Some(so_far) if so_far == parent => so_far,
292            Some(_) => return None,
293        });
294    }
295    shared
296}

308fn artist_artwork_destination(
309    catalog: &Catalog,
310    releases: &[&Release],
311    data_dir: &Path,
312    mbid: &str,
313) -> PathBuf {
314    shared_folder(releases)

316            // A watched root is not an artist folder, even when every album
317            // happens to sit directly beneath it.

321        .unwrap_or_else(|| store::assets_dir(data_dir).join("artists").join(mbid))

324/// Where the catalog lives: what `--data` names, or the default location.
325pub fn data_dir(args: &Args) -> PathBuf {
326    if std::env::var_os("AEDE_DELEGATED_CHILD").is_some()
327        && let Some(forced) = std::env::var_os("AEDE_DELEGATED_DATA_DIR")
328    {
329        return PathBuf::from(forced);
330    }
331    args.value("data")
332        .map(PathBuf::from)
333        .unwrap_or_else(store::default_data_dir)
334}

336fn load(args: &Args) -> Result<Catalog, Box<dyn Error>> {
337    let path = store::catalog_path(&data_dir(args));
338    match store::load(&path)? {
339        Some(catalog) => Ok(catalog),
340        None => Err(format!(
341            "no catalog in {}.\nRun this first: aede scan <folder>",
342            data_dir(args).display()
343        )
344        .into()),
345    }
346}

359/// What other sources said, as it sits on disk.
360///
361/// Loaded and not reconciled, unlike the annotations: the layer is keyed on
362/// what an entity calls itself and resolves on read, so a rebuilt catalog
363/// needs nothing rewritten here.
364/// What MusicBrainz said about this artist, when anything has been fetched.
365///
366/// Returns the facts rather than the record: every caller here wants the
367/// contents, and handing back the wrapper would have each of them repeat the
368/// same two `match` arms to get inside it.
369pub(super) fn artist_facts_for(
370    args: &Args,
371    catalog: &Catalog,
372    artist: Id,
373) -> Option<aede_core::sources::ArtistFacts> {
374    let held = sources_held(args).ok()?;
375    let entity =
376        aede_core::user::EntityRef::of(catalog, aede_core::model::EntityKind::Artist, artist)?;
377    let record = held.get(&entity, aede_core::sources::MUSICBRAINZ)?;
378    if !held.is_trusted(catalog, record) {
379        return None;
380    }
381    match &record.facts {
382        aede_core::sources::Facts::Artist(facts) => Some(facts.clone()),
383        _ => None,
384    }
385}

392/// The role vocabulary, as `(key stored, name shown)`.
393///
394/// One table read in both directions. It used to be a one-way `match`, and the
395/// consequence was a message contradicting itself in a single breath: asked for
396/// `--role "album artist"` — the only spelling ever shown on screen — the
397/// program answered that the artist was *not* credited as album artist and then
398/// listed "album artist (14)" among their credits. The user can only type what
399/// they are shown; whatever is shown must therefore be accepted.
400const ROLE_NAMES: &[(&str, &str)] = &[
401    ("main", "main artist"),
402    ("album", "album artist"),
403    ("composer", "composer"),
404    ("conductor", "conductor"),
405    ("remixer", "remixer"),
406    ("lyricist", "lyricist"),
407    ("performer", "performer"),
408    ("producer", "producer"),
409    ("engineer", "engineer"),
410    ("featured", "featured"),
411];
412
413/// Display name for an internal role key; unknown keys are shown as they are.
414fn role_label(role: &str) -> String {
415    ROLE_NAMES
416        .iter()
417        .find(|(key, _)| *key == role)
418        .map(|(_, label)| (*label).to_string())
419        .unwrap_or_else(|| role.to_string())
420}

422/// The key behind whatever the user typed, or `None` if it names no role.
423///
424/// Accepts the stored key and the displayed name alike, up to case and
425/// accents. It looks first at the roles the catalog **holds** — so a role
426/// arriving from MusicBrainz at M1, absent from the table above, is reachable
427/// by its own name without a line of code — then at the known vocabulary, so
428/// that a real role nobody holds here can be told apart from a word that is no
429/// role at all. The two deserve different answers: one is an empty result, the
430/// other a misunderstanding.
431fn role_key(catalog: &Catalog, typed: &str) -> Option<String> {
432    let wanted = text::normalize(typed);
433    let in_use = catalog
434        .roles_in_use()
435        .into_iter()
436        .find(|key| text::normalize(key) == wanted || text::normalize(&role_label(key)) == wanted)
437        .map(|key| key.to_string());
438    in_use.or_else(|| {
439        ROLE_NAMES
440            .iter()
441            .find(|(key, label)| text::normalize(key) == wanted || text::normalize(label) == wanted)
442            .map(|(key, _)| (*key).to_string())
443    })
444}

459/// Technical description of a stream, shown identically by `file` and by
460/// `track` — one is read from the disk, the other from the catalog, and the
461/// reader has no reason to see a difference.
462fn properties_table(properties: &AudioProperties, has_embedded_art: bool, size: u64) -> Table {
463    let mut t = Table::plain(2);
464    let dash = || "—".to_string();
465    t.push(vec!["Container".into(), properties.container.clone()]);
466    t.push(vec!["Codec".into(), properties.codec.clone()]);
467    t.push(vec!["Quality".into(), properties.quality_label()]);
468    t.push(vec![
469        "Sample rate".into(),
470        properties
471            .sample_rate
472            .map(|r| format!("{r} Hz"))
473            .unwrap_or_else(dash),
474    ]);
475    t.push(vec![
476        "Bit depth".into(),
477        properties
478            .bit_depth
479            .map(|b| format!("{b} bits"))
480            .unwrap_or_else(dash),
481    ]);
482    t.push(vec![
483        "Channels".into(),
484        properties
485            .channels
486            .map(|c| c.to_string())
487            .unwrap_or_else(dash),
488    ]);
489    t.push(vec![
490        "Duration".into(),
491        properties
492            .duration_ms
493            .map(text::format_duration)
494            .unwrap_or_else(dash),
495    ]);
496    t.push(vec![
497        "Bitrate".into(),
498        properties
499            .bitrate_kbps
500            .map(|b| format!("{b} kbps"))
501            .unwrap_or_else(dash),
502    ]);
503    t.push(vec![
504        "Lossless".into(),
505        if properties.lossless { "yes" } else { "no" }.into(),
506    ]);
507    t.push(vec![
508        "Embedded cover art".into(),
509        if has_embedded_art { "yes" } else { "no" }.into(),
510    ]);
511    if size > 0 {
512        t.push(vec!["Size".into(), text::format_size(size)]);
513    }
514    t
515}

526/// Prints attributed MusicBrainz credits in one consistent shape.
527fn print_sourced_credits(catalog: &Catalog, mut credits: Vec<core_sources::SourcedCreditLink>) {
528    if credits.is_empty() {
529        return;
530    }
531    credits.sort_by(|a, b| {
532        a.work
533            .as_ref()
534            .map(|work| work.title.as_str())
535            .cmp(&b.work.as_ref().map(|work| work.title.as_str()))
536            .then_with(|| a.credit.order.cmp(&b.credit.order))
537            .then_with(|| a.credit.role.cmp(&b.credit.role))
538            .then_with(|| a.credit.artist_name.cmp(&b.credit.artist_name))
539    });
540    println!("{}", ui::section("Sourced credits"));
541    let mut table = Table::new(&[
542        "Artist",
543        "Role",
544        "Credited as",
545        "Details",
546        "Scope",
547        "Dates",
548        "Evidence",
549    ])
550    .limit(0, 28)
551    .limit(2, 24)
552    .limit(3, 36)
553    .limit(4, 28)
554    .limit(6, 42);
555    for link in credits {
556        let details = link
557            .credit
558            .attributes
559            .iter()
560            .map(|attribute| {
561                let shown = attribute
562                    .credited_as
563                    .as_deref()
564                    .or(attribute.value.as_deref())
565                    .unwrap_or(&attribute.name);
566                match shown == attribute.name {
567                    true => shown.to_string(),
568                    false => format!("{} ({shown})", attribute.name),
569                }
570            })
571            .collect::<Vec<_>>()
572            .join(", ");
573        let recording = catalog
574            .recording(link.recording_id)
575            .map(|recording| recording.title.as_str())
576            .unwrap_or("unknown recording");
577        let scope = link.work.as_ref().map_or_else(
578            || format!("recording: {recording}"),
579            |work| format!("work: {} · recording: {recording}", work.title),
580        );
581        let mut dates = match (&link.credit.began, &link.credit.ended) {
582            (Some(begin), Some(end)) => format!("{begin}–{end}"),
583            (Some(begin), None) => format!("from {begin}"),
584            (None, Some(end)) => format!("until {end}"),
585            (None, None) => String::new(),
586        };
587        if link.credit.over == Some(true) && link.credit.ended.is_none() {
588            dates.push_str(match dates.is_empty() {
589                true => "ended",
590                false => " · ended",
591            });
592        }
593        let confidence = if link.excluded {
594            "excluded by you".to_string()
595        } else {
596            source_status(link.confidence, link.review, link.trusted)
597        };
598        let relation = link
599            .credit
600            .relation_id
601            .as_deref()
602            .or(link.credit.role_id.as_deref())
603            .map(|id| format!(" · {id}"))
604            .unwrap_or_default();
605        table.push(vec![
606            link.credit.artist_name,
607            role_label(&link.credit.role),
608            link.credit.credited_as.unwrap_or_default(),
609            details,
610            scope,
611            dates,
612            format!(
613                "{} · {confidence} · {}{relation}",
614                link.source,
615                ui::since(link.fetched_at)
616            ),
617        ]);
618    }
619    print!("{}", table.render());
620}

644    ])
645    .limit(0, 28)
646    .limit(2, 24)
647    .limit(3, 36)
648    .limit(4, 36)
649    .limit(5, 42);
650    for link in credits {
651        let details = link
652            .credit
653            .attributes
654            .iter()
655            .map(|attribute| attribute.name.as_str())
656            .collect::<Vec<_>>()
657            .join(", ");
658        let edition = catalog
659            .release(link.release_id)
660            .map(|release| release.title.as_str())
661            .unwrap_or("unknown edition");
662        table.push(vec![
663            link.credit.artist_name,
664            role_label(&link.credit.role),
665            link.credit.credited_as.unwrap_or_default(),
666            details,
667            edition.to_string(),
668            format!(
669                "{} · {} · {}",
670                link.source,
671                if link.excluded {
672                    "excluded by you".to_string()
673                } else {
674                    source_status(link.confidence, link.review, link.trusted)
675                },
676                ui::since(link.fetched_at)
677            ),
678        ]);
679    }
680    print!("{}", table.render());
681}

683/// Human reading of confidence after an explicit review has had its say.
684fn source_status(
685    confidence: core_sources::Confidence,
686    review: Option<core_sources::ReviewDecision>,
687    trusted: bool,
688) -> String {
689    let evidence = match confidence {
690        core_sources::Confidence::Identified => "identified".to_string(),
691        core_sources::Confidence::Matched(score) => format!("matched {score}%"),
692    };
693    match review {
694        Some(core_sources::ReviewDecision::Accepted) => format!("{evidence} · accepted"),
695        Some(core_sources::ReviewDecision::Rejected) => format!("{evidence} · rejected"),
696        None if !trusted => format!("{evidence} · pending review"),
697        None => evidence,
698    }
699}

701/// Playing time and size on disk of a set of tracks.
702///
703/// Count, duration and size are the three measures every listing and every
704/// entity page carries: what a part of the library weighs should not depend on
705/// the command used to look at it.
706fn totals(catalog: &Catalog, tracks: &[Id]) -> (u64, u64) {
707    tracks
708        .iter()
709        .filter_map(|&id| catalog.track(id))
710        .fold((0, 0), |(duration, size), track| {
711            (
712                duration + track.duration_ms.unwrap_or(0),
713                size + catalog.file(track.file_id).map(|f| f.size).unwrap_or(0),
714            )
715        })
716}

718fn print_track_totals(catalog: &Catalog, tracks: &[Id]) {

720    println!(
721        "  {}",
722        ui::dim(&format!(
723            "{} · {} · {}",
724            ui::plural(tracks.len(), "track"),
725            text::format_duration(duration),
726            text::format_size(size)
727        ))
728    );
729}

808/// Says that a listing was cut, and what to do about it.
809///
810/// Every listing stops at `--limit` rows so as not to fill the terminal, and
811/// all five of them used to stop **in silence**. A cut list reads as "that is
812/// all there is", which is the one thing it is not: an album sitting past the
813/// fiftieth row simply did not exist as far as the user could tell. The header
814/// counting the matches is not enough — nobody compares it against the rows
815/// they were given.
816///
817/// Nothing is printed when everything was shown, so the notice keeps meaning
818/// something.
819fn announce_window(window: Window, total: usize, what: &str) {
820    if let Some(said) = window_note(window, total, what) {
821        println!("  {}", ui::yellow(&said));
822    }
823}
824
825/// The sentence [`announce_window`] prints, or `None` when there is nothing to
826/// say.
827///
828/// Split out of the printing because what it decides is not obvious and was
829/// wrong: the *advice* at the end has to follow from the window, and a line
830/// that offers a page which is not there is worse than no line. `aede doctor
831/// --offset=25 --all` answered `26–52 of 52 issues — --offset=52 for the next
832/// page`, and `--offset=52` leads to "starts past the end". The command had
833/// done exactly what it was asked; the sentence under it had not.
834fn window_note(window: Window, total: usize, what: &str) -> Option<String> {
835    let Some((first, last)) = window.shown(total) else {
836        // Two different emptinesses, and naming the wrong one sends the reader
837        // looking for a page that was never there. `--offset` explains an empty
838        // screen only when there was something to page through; a listing that
839        // matched nothing at all is not a paging accident. The confusion became
840        // easy to meet the day the listings learned `--query`: `aede artists
841        // --query "year:2050"` answered "0 artist in all, and --offset=0 starts
842        // past the end", which blames a page number nobody typed.
843        return Some(match total {
844            0 => format!("nothing here: no {what} to show"),
845            _ => format!(
846                "nothing here: {} in all, and --offset={} starts past the end",
847                ui::plural(total, what),
848                window.offset
849            ),
850        });
851    };
852    if first == 1 && last == total {
853        return None;
854    }
855
856    // Two questions, and only the answers that are true of *this* screen.
857    //
858    // **A page is offered only when it exists.** The last row shown being the
859    // last row there is means there is no next page, and `--offset={last}` then
860    // names the one number guaranteed to answer "starts past the end". A reader
861    // on the last screen wants to be told that they are, not sent onward.
862    //
863    // **`--all` is advice only while it would change something.** Given
864    // already, the rows are unbounded and what shaped this screen was
865    // `--offset` alone; repeating it names an option the reader has typed and
866    // tells them to type it, which is how a reader learns to stop reading these
867    // lines.
868    let unbounded = window.limit == usize::MAX;
869    let onward = match (last < total, unbounded) {
870        (true, true) => format!("--offset={last} for the next page"),
871        (true, false) => format!("--offset={last} for the next page, --all for every row"),
872        // On the last screen `--all` is not advice either, whatever the limit:
873        // what cut this one is the offset, and lifting the limit would show the
874        // same rows again. The only move left is back to the start.
875        (false, _) => "these are the last; drop --offset to start from the first".to_string(),
876    };
877    Some(format!(
878        "{first}–{last} of {} — {onward}",
879        ui::plural(total, what)
880    ))
881}

883/// Marker put after an album title when the same album sits elsewhere too.
884///
885/// Two words rather than a symbol: a legend nobody reads is worse than four
886/// characters of prose, and the two cases call for opposite reactions.
887fn copy_marker(catalog: &Catalog, release_id: Id) -> String {
888    use aede_core::model::{DUPLICATE, OTHER_EDITION};
889    if !catalog.related_releases(release_id, DUPLICATE).is_empty() {
890        return " (duplicate)".to_string();
891    }
892    if !catalog
893        .related_releases(release_id, OTHER_EDITION)
894        .is_empty()
895    {
896        return " (other edition)".to_string();
897    }
898    String::new()
899}

901/// Hands the tracks on screen to another program, when asked for.
902///
903/// `None` means nothing was asked and the command should print its usual page.
904/// Every command that shows a track list offers the same two exits, so the
905/// option means the same thing wherever it is typed.
906fn selection_output(catalog: &Catalog, tracks: &[Id], args: &Args) -> Option<Res> {
907    // The one gate every one of these formats passes through, whichever
908    // command got them here: `--m3u --csv` used to hand back the playlist and
909    // drop the CSV without a word, because this function looked at `--m3u`
910    // first and never noticed the other flag was there at all.
911    if let Some(message) = args.output_conflict(&["m3u", "csv", "json"]) {
912        return Some(Err(message.into()));
913    }
914    if args.has("m3u") {
915        return Some(play_list(catalog, tracks, args));
916    }
917    if args.has("csv") || args.has("json") {
918        return Some(export::tracks_table(catalog, tracks, args));
919    }
920    None
921}

923/// Refuses `--output` on a page that has no file to write it to.
924///
925/// `-o`/`--output` only ever writes something through [`selection_output`] or
926/// a listing's `--csv`/`--json` table — everywhere else is a page meant for
927/// the screen, and until now `-o` vanished in front of one without a word.
928/// That is exactly the "wrong answer standing in for a missing one" this
929/// program refuses everywhere else an option cannot be honoured, and the rule
930/// stated at the top of `docs/commands.md`: an option a command cannot
931/// honour is refused, never ignored.
932pub fn refuse_output_without_a_format(args: &Args) -> Res {
933    if args.has("output") {
934        return Err(
935            "--output writes what --csv, --json or --m3u produce; this page has none of those to give it.\n\
936             Drop --output to see it on screen, or add one of the three to write it out."
937                .into(),
938        );
939    }
940    Ok(())
941}

943/// Prints the tracks shown as an M3U playlist instead of the usual page.
944///
945/// Every command that puts a track list on screen can hand it to a player;
946/// that the selection was reached through an album, an artist or a search is
947/// the caller's business, not the playlist's.
948fn play_list(catalog: &Catalog, tracks: &[Id], args: &Args) -> Res {
949    if tracks.is_empty() {
950        return Err("nothing to put in a playlist".into());
951    }
952    export::emit(args, &export::m3u(catalog, tracks)?)
953}
