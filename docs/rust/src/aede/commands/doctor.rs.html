1//! The `doctor` command: what is wrong with the library.
2
3use aede_core::doctor::{self, Issue, Severity};
4use aede_core::json::Json;
5
6use super::{Res, announce_window, data_dir, load};
7use crate::args::Args;
8use crate::ui::{self, Align, Table};

10pub fn show_doctor(args: &Args) -> Res {
11    let window = args.window(25)?;
12    let catalog = load(args)?;
13    let sources = super::sources_held(args)?;
14    let mut issues = doctor::diagnose(&catalog, &sources);
15    let user =
16        aede_core::user::load(&aede_core::user::user_path(&data_dir(args)))?.unwrap_or_default();
17    issues.extend(doctor::orphaned_relation_annotations(
18        &catalog, &sources, &user,
19    ));
20    issues.sort_by(|left, right| {
21        left.severity()
22            .cmp(&right.severity())
23            .then_with(|| left.kind.cmp(&right.kind))
24            .then_with(|| left.detail.cmp(&right.detail))
25    });
26
27    // A level nobody recognises used to leave the filter off entirely: the
28    // whole diagnosis came back under a name that had asked for a third of it,
29    // and nothing said so. Same fault as `--year=abc`, same answer.
30    if let Some(filter) = args.value("severity") {
31        let wanted = match filter {
32            "error" | "errors" => Severity::Error,
33            "warning" | "warnings" => Severity::Warning,
34            "info" | "infos" => Severity::Info,
35            other => {
36                return Err(
37                    format!("--severity takes error, warning or info: got \"{other}\"").into(),
38                );
39            }
40        };
41        issues.retain(|i| i.severity() == wanted);
42    }
43
44    if args.has("json") {
45        let start = window.offset.min(issues.len());
46        let end = start.saturating_add(window.limit).min(issues.len());
47        println!("{}", issues_to_json(&issues[start..end]).to_string_pretty());
48        return Ok(());
49    }
50
51    let summary = doctor::summary(&issues);
52    println!("{}", ui::section("Diagnosis"));
53    if issues.is_empty() {
54        println!("  {}", ui::green("No issue found."));
55        print_unverified(&catalog);
56        print_waiting_analyses(&catalog);
57        return Ok(());
58    }
59    for (severity, count) in &summary {
60        let line = ui::plural(*count, severity.label());
61        let coloured = match severity {
62            Severity::Error => ui::red(&line),
63            Severity::Warning => ui::yellow(&line),
64            Severity::Info => ui::dim(&line),
65        };
66        println!("  {coloured}");
67    }
68
69    // Grouping by kind: a damaged library produces thousands of lines, so the
70    // summary comes first.
71    println!("{}", ui::section("By issue type"));
72    let mut by_kind: std::collections::BTreeMap<&str, usize> = Default::default();
73    for issue in &issues {
74        *by_kind.entry(issue.kind.label()).or_insert(0) += 1;
75    }
76    let mut t = Table::new(&["Issue", "Count"]).align(1, Align::Right);
77    let mut rows: Vec<(&str, usize)> = by_kind.into_iter().collect();
78    rows.sort_by(|a, b| b.1.cmp(&a.1).then_with(|| a.0.cmp(b.0)));
79    for (label, count) in rows {
80        t.push(vec![label.to_string(), count.to_string()]);
81    }
82    print!("{}", t.render());
83
84    println!("{}", ui::section("Details"));
85    for issue in issues.iter().skip(window.offset).take(window.limit) {
86        let mark = match issue.severity() {
87            Severity::Error => ui::red("✗"),
88            Severity::Warning => ui::yellow("!"),
89            Severity::Info => ui::dim("·"),
90        };
91        println!(
92            "  {mark} {} — {}",
93            ui::bold(issue.kind.label()),
94            ui::literal(&issue.detail)
95        );
96        for file in issue.files.iter().take(4) {
97            println!("      {}", ui::dim(&ui::literal(file)));
98        }
99        if issue.files.len() > 4 {
100            println!(
101                "      {}",
102                ui::dim(&format!("… and {} more", issue.files.len() - 4))
103            );
104        }
105    }
106    println!();
107    announce_window(window, issues.len(), "issue");
108    print_unverified(&catalog);
109    print_waiting_analyses(&catalog);
110    Ok(())
111}

113fn issues_to_json(issues: &[Issue]) -> Json {
114    Json::Arr(
115        issues
116            .iter()
117            .map(|i| {
118                let mut o = Json::obj();
119                o.set("type", i.kind.label().into());
120                o.set("severity", i.severity().label().into());
121                o.set("detail", i.detail.clone().into());
122                o.set(
123                    "files",
124                    Json::Arr(i.files.iter().map(|f| Json::Str(f.clone())).collect()),
125                );
126                o
127            })
128            .collect(),
129    )
130}

132/// Says how many files carry no integrity verdict.
133///
134/// `doctor` reads no file, so it can only report what `aede check` established.
135/// Staying silent would let a library look healthy when nothing was verified.
136fn print_unverified(catalog: &aede_core::model::Catalog) {
137    let unverified = catalog
138        .files
139        .iter()
140        .filter(|f| f.integrity.is_none())
141        .count();
142    if unverified == 0 {
143        return;
144    }
145    println!(
146        "  {}",
147        ui::dim(&format!(
148            "{} not verified — run aede check",
149            ui::plural(unverified, "file")
150        ))
151    );
152}

154/// Says how many imported analyses describe files the catalog does not hold.
155///
156/// They are not lost and they are not a defect: they are waiting for the folder
157/// they name to be scanned. Saying so is the difference between "waiting" and
158/// "swallowed without a word".
159fn print_waiting_analyses(catalog: &aede_core::model::Catalog) {
160    let waiting = catalog.pending_analyses();
161    if waiting == 0 {
162        return;
163    }
164    println!(
165        "  {}",
166        ui::dim(&format!(
167            "{} waiting for the folders they name to be scanned",
168            ui::plural(waiting, "imported analysis")
169        ))
170    );
