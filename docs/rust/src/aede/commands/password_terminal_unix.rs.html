1//! UTF-8 terminal keys using the existing scoped stty mode, without FFI.
2
3use std::io::{self, Read, StdinLock};
4
5use crate::commands::terminal_mode::TerminalMode;
6
7use super::{Key, character_key};
8
9pub(super) struct Input {
10    mode: TerminalMode,
11    reader: StdinLock<'static>,
12    restored: bool,
13}
14
15impl Input {
16    pub(super) fn start() -> io::Result<Self> {
17        let mode = TerminalMode::start(&[
18            // Read characters ourselves: neither flow control, extended
19            // editing nor input translation may swallow or rewrite a secret.
20            "-echo", "-echonl", "-icanon", "-isig", "-iexten", "-ixon", "-ixoff", "-istrip",
21            "-inlcr", "-igncr", "-icrnl", "-parmrk", "min", "1", "time", "0",
22        ])?;
23        Ok(Self {
24            mode,
25            reader: io::stdin().lock(),
26            restored: false,
27        })
28    }
29
30    pub(super) fn read_key(&mut self) -> io::Result<Key> {
31        let mut first = [0];
32        self.reader.read_exact(&mut first)?;
33        if first[0] == 0x1b {
34            // A lone Escape must cancel rather than wait for another key.
35            // The nested mode changes only the read timeout, retaining the
36            // outer guard's continuous echo/signal suppression.
37            let mut timeout = TerminalMode::start(&["min", "0", "time", "1"])?;
38            let result = escape(|| {
39                let mut byte = [0];
40                match self.reader.read(&mut byte)? {
41                    0 => Ok(None),
42                    _ => Ok(Some(byte[0])),
43                }
44            });
45            timeout.restore()?;
46            return result;
47        }
48        character(first[0], &mut self.reader).map(character_key)
49    }
50
51    pub(super) fn restore(&mut self) -> io::Result<()> {
52        if self.restored {
53            return Ok(());
54        }
55        // On cancellation or invalid input, the rest of a paste may still be
56        // buffered by StdinLock or queued in the terminal driver. Discard both
57        // while echo remains disabled, including extra lines after success.
58        // VTIME waits for a quiet 100 ms without waiting indefinitely for a key.
59        let mut timeout = TerminalMode::start(&["min", "0", "time", "1"])?;
60        discard_pending(&mut self.reader)?;
61        timeout.restore()?;
62        self.mode.restore()?;
63        self.restored = true;
64        Ok(())
65    }
66}
67
68impl Drop for Input {
69    fn drop(&mut self) {
70        // Cover prompt write/flush failures and unwinding, as well as retrying
71        // a failed explicit cleanup. The terminal mode guard is the last resort.
72        if let Err(error) = self.restore() {
73            // A generic mode-guard drop must not re-enable echo over input
74            // that could not be drained. Prefer a masked terminal and a clear
75            // error to exposing queued secret bytes; playback keeps its guard.
76            self.mode.retain_current();
77            eprintln!(
78                "cannot finish masked terminal input; terminal echo remains disabled: {error}"
79            );
80        }
81    }
82}
83
84fn discard_pending(mut reader: impl Read) -> io::Result<()> {
85    let mut discarded = [0; 1024];
86    loop {
87        match reader.read(&mut discarded) {
88            Ok(0) => return Ok(()),
89            Ok(_) => {}
90            Err(error) if error.kind() == io::ErrorKind::Interrupted => {}
91            Err(error) => return Err(error),
92        }
93    }
94}
95
96fn character(first: u8, mut remaining: impl Read) -> io::Result<char> {
97    let length = match first {
98        0..=0x7f => return Ok(char::from(first)),
99        0xc2..=0xdf => 2,
100        0xe0..=0xef => 3,
101        0xf0..=0xf4 => 4,
102        _ => return Err(invalid_utf8()),
103    };
104    let mut bytes = [0; 4];
105    bytes[0] = first;
106    remaining.read_exact(&mut bytes[1..length])?;
107    std::str::from_utf8(&bytes[..length])
108        .map_err(|_| invalid_utf8())?
109        .chars()
110        .next()
111        .ok_or_else(invalid_utf8)
112}
113
114fn invalid_utf8() -> io::Error {
115    io::Error::new(
116        io::ErrorKind::InvalidData,
117        "terminal input is not valid UTF-8",
118    )
119}
120
121fn escape(mut next: impl FnMut() -> io::Result<Option<u8>>) -> io::Result<Key> {
122    match next()? {
123        None => Ok(Key::Escape),
124        Some(0x03) => Ok(Key::CtrlC),
125        Some(0x04) => Ok(Key::Char('\u{4}')),
126        Some(0x1b) => Ok(Key::Escape),
127        Some(b'[' | b'O') => {
128            // CSI/SS3 keys and bracketed-paste delimiters are terminal events,
129            // never password text. Bound both their size and inter-byte wait.
130            for _ in 0..16 {
131                match next()? {
132                    Some(0x03) => return Ok(Key::CtrlC),
133                    Some(0x04) => return Ok(Key::Char('\u{4}')),
134                    Some(0x1b) => return Ok(Key::Escape),
135                    Some(0x40..=0x7e) => return Ok(Key::Ignored),
136                    Some(0x20..=0x3f) => {}
137                    _ => break,
138                }
139            }
140            Err(io::Error::new(
141                io::ErrorKind::InvalidData,
142                "incomplete or oversized terminal key sequence",
143            ))
144        }
145        Some(first) => {
146            // Alt plus a printable character is also a key event. Consume
147            // its complete scalar so UTF-8 continuation bytes cannot leak
148            // into the next password character.
149            let mut bytes = [0; 3];
150            let length = match first {
151                0..=0x7f => 0,
152                0xc2..=0xdf => 1,
153                0xe0..=0xef => 2,
154                0xf0..=0xf4 => 3,
155                _ => return Err(invalid_utf8()),
156            };
157            for byte in &mut bytes[..length] {
158                *byte = next()?.ok_or_else(invalid_utf8)?;
159            }
160            let scalar = character(first, &bytes[..length])?;
161            if scalar.is_control() {
162                return Ok(character_key(scalar));
163            }
164            Ok(Key::Ignored)
165        }
166    }
167}
