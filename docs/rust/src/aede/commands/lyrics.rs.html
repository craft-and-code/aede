1//! `fetch --lyrics`: the words, for tracks that have none.
2//!
3//! # Never on by default, and that is not a technical decision
4//!
5//! Lyrics are the **composition's** copyright. Owning a FLAC grants no rights
6//! in it — the recording and the song are legally two different objects — and
7//! every comparable project keeps online fetching out of its core for exactly
8//! that reason: Navidrome and Jellyfin read files and leave the network to
9//! plugins, beets ships a `lyrics` plugin, foobar2000 and MusicBee do it
10//! through add-ons.
11//!
12//! So this pass runs only when somebody types `--lyrics`, and it says what it
13//! is about to do **before it asks anything** — every run, not once. That is a
14//! decision about somebody else's rights, and this program does not get to
15//! make it silently on a user's behalf. `docs/design/lyrics.md` has the whole
16//! of the reasoning.
17//!
18//! The typed option is the consent, and it is deliberately not doubled by a
19//! prompt on every run: a confirmation asked every time is a confirmation
20//! nobody reads, which would leave the caveat *less* read than printing it
21//! plainly does. A long run is still agreed to, on the same threshold and for
22//! the same reason as the ordinary fetch — that is about the ten minutes, not
23//! about the rights.
24//!
25//! # It never touches a track that already has words
26//!
27//! The catalog answers that offline and exactly: a file's tags say whether the
28//! words are inside it, and [`aede_core::model::AudioFile::lyrics_path`] says
29//! whether a `.lrc` sits beside it. Either of them and the track is not asked
30//! about at all — no request, no file, nothing.
31//!
32//! There is deliberately **no `--replace`**. Overwriting a lyrics file somebody
33//! wrote or corrected is not a thing this command should be able to do by
34//! accident, and it is the same refusal `--covers` makes about artwork.
35//!
36//! # What is written
37//!
38//! A `.lrc` beside the track, which is the file every player already looks for
39//! and the one the scanner already picks up from the walk. It is registered
40//! nowhere: the next scan simply discovers it, exactly as it would one put
41//! there by hand.
42//!
43//! **Nothing is written into an audio file, here or anywhere in this program.**
44//!
45//! # A track the service has nothing for is asked about again
46//!
47//! Said plainly rather than hidden, because it is the one cost of this pass
48//! that a reader might not expect. `--covers` records that the archive had
49//! nothing, so a second run does not ask twice; here a *successful* answer
50//! records itself — the `.lrc` is on the disk and the track is skipped from
51//! then on — and only the misses come back. Recording those would mean a store
52//! of negative claims about a service that gains entries every day, which is a
53//! worse trade than one wasted request on a run somebody typed on purpose.
54
55// Compiled in every build, for the reason `fetch` is.
56#![cfg_attr(not(feature = "fetch"), allow(dead_code))]

68/// A track to ask about, and where its words would go.
69struct Target {
70    /// What to ask.
71    url: String,
72    /// The track, for a line a reader can recognise.
73    title: String,
74    artist: String,
75    /// Where the `.lrc` goes: beside the audio, same name.
76    sidecar: std::path::PathBuf,
77}
78
79/// Why a track was left out, so the header can say so rather than silently
80/// shortening the list.
81#[derive(Default)]
82struct Skipped {
83    /// The words are in the file's own tags.
84    tagged: usize,
85    /// A `.lrc` is already beside it.
86    sidecar: usize,
87    /// No length, no artist or no title.
88    ///
89    /// Only the first two are the service's requirement — a live answer to the
90    /// artist and title alone came back `200`. The **length** is this program's
91    /// insistence: a song has a studio take, a live one and three covers, they
92    /// share a title, and without the length the answer would be about
93    /// whichever of them the service reached first.
94    unaskable: usize,
95}

97/// The pass.
98pub fn run(
99    args: &Args,
100    catalog: &Catalog,
101    transport: &mut dyn Ask,
102    backoff: &[std::time::Duration],
103    asked: &super::fetch::Asked,
104) -> Res {
105    let (targets, skipped) = survey(catalog, asked.names, asked.scope);
106
107    println!("{}", ui::section("Lyrics"));
108    // Before the counts, and before anything is asked. A reader deciding
109    // whether to run this needs to know what it is they are deciding.
110    println!(
111        "  {}",
112        ui::dim(
113            "lyrics are the song's copyright, not the recording's: owning the \
114             file grants no rights in the words. This asks LRCLIB, which \
115             permits it, and runs only when you type --lyrics"
116        )
117    );
118    report(&skipped);
119
120    if targets.is_empty() {
121        match asked.names.is_empty() && asked.scope.is_empty() {
122            true => println!("  {}", ui::dim("no track is missing its words")),
123            false => println!(
124                "  {}",
125                ui::dim(&format!(
126                    "no track missing its words answers to {}",
127                    super::fetch::narrowing(asked.names, asked.scope)
128                ))
129            ),
130        }
131        return Ok(());
132    }
133
134    let wait = targets.len() as u64 * musicbrainz::REQUEST_INTERVAL.as_millis() as u64;
135    println!(
136        "  {} without words, one request each, about {}",
137        ui::plural(targets.len(), "track"),
138        ui::long_duration(wait)
139    );
140    println!(
141        "  {}",
142        ui::dim(&format!(
143            "each answer is written as a .{} beside its track; a track that \
144             already has words is never touched, and nothing is ever written \
145             into an audio file",
146            aede_core::lyrics::EXTENSION
147        ))
148    );
149    // The polite alternative, named where somebody about to spend six hours
150    // will read it. LRCLIB publishes its whole database as SQLite dumps for
151    // exactly this case, and a program that knew and did not say would be
152    // spending a free service's bandwidth on the user's behalf.
153    if targets.len() > BULK {
154        println!(
155            "  {}",
156            ui::yellow(&format!(
157                "that is {} requests to a free service — for a library this \
158                 size the polite way is LRCLIB's published database dump",
159                targets.len()
160            ))
161        );
162    }
163    if super::fetch::asked_nothing(
164        asked,
165        &targets.iter().map(|t| t.title.clone()).collect::<Vec<_>>(),
166    ) {
167        return Ok(());
168    }
169    // A run of ten minutes is something to agree to, not something to
170    // discover — the same threshold the ordinary fetch uses, and about the
171    // same thing: the time, not the rights. The caveat above is printed
172    // whatever the length.
173    if targets.len() > super::fetch::CONFIRM_ABOVE
174        && !super::confirmed(args, "ask about all of them")?
175    {
176        println!("  {}", ui::dim("nothing was asked"));
177        return Ok(());
178    }
179
180    let (mut written, mut none, mut instrumental, mut failed) = (0usize, 0usize, 0usize, 0usize);
181    let mut timed = 0usize;
182    let mut pending = queue(&targets);
183    let mut done = 0usize;
184    let total = targets.len();
185    while let Some((target, retried)) = pending.pop_front() {
186        print!("\r  asking: {}/{}", done + 1, total);
187        let _ = std::io::Write::flush(&mut std::io::stdout());
188
189        match ask_with_backoff(transport, &target.url, backoff) {
190            Ok(answer) => match lrclib::read(&answer) {
191                Found::Words(words) => match write_beside(&target.sidecar, &words.text) {
192                    Ok(()) => {
193                        written += 1;
194                        if words.synced {
195                            timed += 1;
196                        }
197                    }
198                    Err(why) => {
199                        failed += 1;
200                        eprintln!("\r  {} {}: {why}", ui::red("×"), target.title);
201                    }
202                },
203                Found::Instrumental => instrumental += 1,
204                Found::Nothing => none += 1,
205            },
206            // A track the service does not know is the ordinary case, not a
207            // failure: it answers 404, and a library of any size holds plenty
208            // it has never seen.
209            Err(Refusal::Missing) => none += 1,
210            Err(Refusal::RateLimited) => {
211                println!();
212                return Err(format!(
213                    "the service refused {} times in a row, waiting longer each \
214                     time; that is a rate limit rather than a hiccup, so nothing \
215                     more was asked.\n  it stopped on \"{}\"",
216                    backoff.len() + 1,
217                    target.title
218                )
219                .into());
220            }
221            Err(other) if defer(&mut pending, target, retried, &other) => {
222                continue;
223            }
224            Err(other) => {
225                failed += 1;
226                eprintln!(
227                    "\r  {} {} — {}: {other}",
228                    ui::red("×"),
229                    target.artist,
230                    target.title
231                );
232            }
233        }
234        done += 1;
235    }
236    println!();
237
238    println!("  {} written", ui::plural(written, "lyrics file"));
239    if timed > 0 {
240        println!(
241            "  {}",
242            ui::dim(&format!(
243                "{} of them timed, which a player can follow",
244                timed
245            ))
246        );
247    }
248    if instrumental > 0 {
249        println!(
250            "  {}",
251            ui::dim(&format!(
252                "{} — the service says so, and says it once",
253                ui::plural(instrumental, "instrumental")
254            ))
255        );
256    }
257    if none > 0 {
258        // Named, with what it costs, because it is the one thing here a second
259        // run repeats. See the note at the top of this file for why nothing is
260        // stored about it.
261        println!(
262            "  {}",
263            ui::dim(&format!(
264                "{} the service has nothing for; a later run asks again",
265                ui::plural(none, "track")
266            ))
267        );
268    }
269    if failed > 0 {
270        println!("  {}", ui::yellow(&format!("{failed} could not be asked")));
271    }
272    if written > 0 {
273        println!(
274            "  {}",
275            ui::dim("aede scan attaches them; aede track \"<title>\" --lyrics shows one")
276        );
277    }
278    Ok(())
279}

281/// Above this many requests, the run names the published database instead.
282///
283/// A number rather than a judgement call, and a low one: five hundred requests
284/// at a request a second is nine minutes, which is the point where somebody
285/// would rather have been told there was another way.
286const BULK: usize = 500;
287
288/// Writes a lyrics file beside its track, and never over one.
289///
290/// The check is here rather than only in [`survey`] because the two are
291/// separated by a network: a `.lrc` that appeared while the run was working —
292/// a second copy of the program, a file dropped in by hand — must not be
293/// overwritten by an answer that was asked for before it existed. The same
294/// reasoning as the cover art pass, and the same refusal.
295fn write_beside(path: &std::path::Path, text: &str) -> Result<(), String> {
296    if path.exists() {
297        return Err(format!(
298            "{} appeared while this was running and was left alone",
299            path.display()
300        ));
301    }
302    // A trailing newline, because this is a text file somebody will open in an
303    // editor and every other text file on their disk has one.
304    let text = match text.ends_with('\n') {
305        true => text.to_string(),
306        false => format!("{text}\n"),
307    };
308    match aede_core::store::write_new_atomic(path, text.as_bytes()) {
309        Ok(true) => Ok(()),
310        Ok(false) => Err(format!(
311            "{} appeared while this was running and was left alone",
312            path.display()
313        )),
314        Err(why) => Err(format!("could not write {}: {why}", path.display())),
315    }
316}

318/// The tracks worth asking about, and what was left out.
319///
320/// A track is worth asking about when it has **no words already** and can be
321/// asked about at all: the service matches on artist, title, album and length,
322/// so a track missing any of them cannot be looked up — and saying that is
323/// better than sending a request that cannot match.
324fn survey(
325    catalog: &Catalog,
326    wanted: &[String],
327    scope: &super::fetch::Scope,
328) -> (Vec<Target>, Skipped) {
329    let mut targets = Vec::new();
330    let mut skipped = Skipped::default();
331    for track in &catalog.tracks {
332        let Some(file) = catalog.file(track.file_id) else {
333            continue;
334        };
335        // Out of the folders asked about: not skipped, not counted, not here
336        // at all. The counts below explain the tracks this run *could* have
337        // asked about, and a track on another shelf was never one of them.
338        if !scope.has_track(track.id) {
339            continue;
340        }
341        // The main credit, which is what the service matches on. `credits_on`
342        // is the one place a track's artists are worked out, and going round
343        // it would be a second answer to the same question.
344        let artist = super::main_track_artist(catalog, track.id).unwrap_or_default();
345        if !super::fetch::reaches(wanted, &[track.title.as_str(), artist.as_str()]) {
346            continue;
347        }
348        // In the file's own tags: the words are there, and this pass has
349        // nothing to add.
350        if file.tags.contains_key("lyrics") {
351            skipped.tagged += 1;
352            continue;
353        }
354        if file.lyrics_path.is_some() {
355            skipped.sidecar += 1;
356            continue;
357        }
358        let album = track
359            .release_id
360            .and_then(|id| catalog.release(id))
361            .map(|r| r.title.clone())
362            .unwrap_or_default();
363        // The album may be missing and the question still be a good one — it
364        // is simply left out of the address. The length may not: see
365        // [`Skipped::unaskable`].
366        let (Some(duration), false, false) = (
367            track.duration_ms,
368            artist.trim().is_empty(),
369            track.title.trim().is_empty(),
370        ) else {
371            skipped.unaskable += 1;
372            continue;
373        };
374        targets.push(Target {
375            // Seconds, which is what the service matches on.
376            url: lrclib::get_url(&artist, &track.title, &album, duration / 1000),
377            title: track.title.clone(),
378            artist,
379            sidecar: aede_core::lyrics::sidecar_of(std::path::Path::new(&file.path)),
380        });
381    }
382    (targets, skipped)
383}

385/// Says what was left out and why.
386///
387/// A filter the reader cannot see is a trap: a run that asked about nine tracks
388/// on a library of four hundred owes an explanation, and "the other three
389/// hundred and ninety-one already have their words" is a good one.
390fn report(skipped: &Skipped) {
391    // **Participles, not verbs.** `ui::plural` agrees the noun and cannot agree
392    // what follows it, so "1 track already have a lyrics file beside them" is
393    // what a hard-coded verb produces — and it shipped, in the first run of
394    // this pass. A phrase that reads correctly at one and at many is the fix,
395    // not a second branch per line.
396    for (count, why) in [
397        (skipped.tagged, "already carrying the words in the tags"),
398        (skipped.sidecar, "already with a lyrics file alongside"),
399        (
400            skipped.unaskable,
401            "with no artist, title or measured length to ask about — a length \
402             is what tells a studio take from a live one",
403        ),
404    ] {
405        if count > 0 {
406            println!(
407                "  {}",
408                ui::dim(&format!("{} {why}", ui::plural(count, "track")))
409            );
410        }
411    }
412}
