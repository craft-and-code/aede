1//! The `fetch` command: ask MusicBrainz about what is in the catalog.
2//!
3//! Everything it decides — which answer is about us, how firmly, what to store
4//! — lives in [`aede_core::musicbrainz`] and is tested without a network. What
5//! is here is the walk, the progress, and the saving as it goes.
6//!
7//! **The rate is the shape of this command.** One request per second is not a
8//! detail to tune later: six hundred artists is ten minutes, and a run that
9//! long has to say so before it starts, show where it is, and lose nothing
10//! when it is interrupted. So it saves after every answer rather than at the
11//! end — the same rule `check` follows for a scan that may take an hour.
12
13// Compiled in every build, because the tests below prove it in every build;
14// only `fetch` itself needs the feature, and a build without it would
15// otherwise report this machinery as dead.
16#![cfg_attr(not(feature = "fetch"), allow(dead_code))]

31/// Whatever can answer a URL with JSON.
32///
33/// A trait for one implementation, which usually earns nothing — here it earns
34/// the only thing that matters: the walk below, the refusals, the counting and
35/// the saving all compile and run **without a network stack**, against a fake
36/// that answers from a fixture. What is left unproven is then the twenty lines
37/// that hand a URL to the client library, instead of this whole command.
38pub trait Ask {
39    /// Fetches a URL and parses the answer, or says why it could not.
40    fn get_json(&mut self, url: &str) -> Result<Json, Refusal>;
41
42    /// Fetches a URL and hands back what came, unread.
43    ///
44    /// For the one thing this program downloads that is not an answer: an
45    /// image. It sits on the same trait as [`Ask::get_json`] rather than on one
46    /// of its own so that a caller cannot fetch a picture without waiting its
47    /// turn — the throttle belongs to the client, and a second client would be
48    /// a second rate limiter to forget about.
49    fn get_bytes(&mut self, url: &str) -> Result<Vec<u8>, Refusal>;
50}

52/// Why an answer did not arrive.
53#[derive(Debug, Clone, PartialEq, Eq)]
54pub enum Refusal {
55    /// The service asked us to slow down: stop, do not retry.
56    RateLimited,
57    /// The service answered, and the answer was that it has nothing.
58    ///
59    /// **Not a failure**, and kept apart from one for that reason. A library of
60    /// four hundred tracks holds plenty LRCLIB has never seen, and a run that
61    /// reported four hundred failures would be describing a working service as
62    /// broken. It was read out of the message text before this existed — a
63    /// `detail.contains("404")` — which is a comparison that breaks the day the
64    /// wording changes and says nothing when it does.
65    Missing,
66    /// No route, no name, a timeout: the service was never reached at all.
67    ///
68    /// Kept apart from [`Refusal::Failed`] because it is the one refusal a
69    /// later attempt might not repeat — see [`worth_deferring`]. A run gives
70    /// it exactly one more try, at the end of the list, before treating it
71    /// the same as any other failure.
72    Unreachable(String),
73    /// Anything else, already worded for a reader.
74    Failed(String),
75}
76
77impl std::fmt::Display for Refusal {
78    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
79        match self {
80            Refusal::RateLimited => write!(
81                f,
82                "the service is refusing requests because too many were sent \
83                 (one per second is the limit); nothing was lost, try later"
84            ),
85            Refusal::Missing => write!(f, "the service has nothing for this"),
86            Refusal::Unreachable(detail) => write!(f, "could not reach the service: {detail}"),
87            Refusal::Failed(detail) => write!(f, "{detail}"),
88        }
89    }
90}

108#[cfg(feature = "fetch")]
109fn http_refusal(error: aede_core::http::Error) -> Refusal {
110    match error {
111        aede_core::http::Error::RateLimited => Refusal::RateLimited,
112        aede_core::http::Error::Status(404) => Refusal::Missing,
113        aede_core::http::Error::Network(detail) => Refusal::Unreachable(detail),
114        other => Refusal::Failed(other.to_string()),
115    }
116}

139/// How long to wait before asking again, after a `503`.
140///
141/// Backing off is the polite reaction to a service saying "not now", and it is
142/// also what tells the two meanings of `503` apart. MusicBrainz answers it both
143/// when the rate has been exceeded — a ban that lasts — and when its search
144/// server is momentarily overloaded, which passes. Nothing in the response
145/// distinguishes them, but their *behaviour* does: a transient one lets the
146/// next attempt through, a ban does not. So the client tries three times,
147/// waiting longer each time, and only then gives up.
148///
149/// The first version stopped the whole run on the first `503`, which turned a
150/// hiccup on request 5 of 402 into "come back later" — and looked exactly like
151/// a rate limit that had not been exceeded.
152const RETRY_AFTER: [std::time::Duration; 3] = [
153    std::time::Duration::from_secs(2),
154    std::time::Duration::from_secs(5),
155    std::time::Duration::from_secs(15),
156];

158/// What was typed after the command, sorted into folders and names.
159///
160/// **A path is not a name**, and every positional used to be read as one. So
161/// `aede fetch --lyrics ~/Music/Alastis` normalised a whole path into words,
162/// matched none of them against a title or an artist, and asked LRCLIB about
163/// the **entire library** — the swallowed argument this program refuses
164/// everywhere else, wearing a slash. Worse than a mistyped name, because it
165/// looks like it worked.
166///
167/// A positional naming something that exists on the disk is a folder;
168/// everything else is a name. That is the test [`super::scope_of`] already
169/// makes for the commands that take `[folder…]`, so `aede fetch --lyrics
170/// ~/Music/Alastis` and `aede check ~/Music/Alastis` mean the same thing by
171/// the same word. The reading is never silent: [`run_with`] prints the folders
172/// back before anything is asked, so a name that happens to also be a folder
173/// on the disk is visibly read as the folder.
174pub(super) fn folders_and_names(args: &Args) -> (Vec<String>, Vec<String>) {
175    let mut folders = Vec::new();
176    let mut names = Vec::new();
177    for raw in &args.positionals {
178        if std::path::Path::new(raw).exists() {
179            folders.push(raw.clone());
180            continue;
181        }
182        let name = text::normalize(raw);
183        if !name.is_empty() {
184            names.push(name);
185        }
186    }
187    (folders, names)
188}

204/// The folders typed after the command, and what the catalog holds under them.
205///
206/// A name and a folder narrow a run in two different ways — one asks *who*,
207/// the other asks *where* — and every pass has to honour both. Which artists,
208/// albums and tracks a folder holds is a walk over the catalog, so it is
209/// worked out **once**, here, exactly as the names are: six passes working it
210/// out separately is six chances for two of them to disagree about what
211/// `~/Music/Alastis` means.
212///
213/// [`Default`] is the whole library, which is what the `waiting` counters ask
214/// for: they report what a pass would find over everything, not over what
215/// happened to be typed this time.
216#[derive(Default)]
217pub(super) struct Scope {
218    /// The folders, canonical, as the catalog spells them; empty for the whole
219    /// library.
220    folders: Vec<String>,
221    /// What they hold. `None` when no folder was given, which is not the same
222    /// as a folder that holds nothing — the first reaches everything, the
223    /// second reaches nothing at all.
224    holds: Option<Held>,
225}
226
227/// No folders: the whole library, which is what most callers mean.
228///
229/// A `static` rather than a [`Default::default()`](Default) at each call site
230/// because several callers want a `&'static Scope` — the `waiting` counters,
231/// which report over everything, and the tests that build an [`Asked`] by
232/// hand. It costs nothing: an empty `Vec` allocates nothing and a `static` is
233/// never dropped.
234pub(super) static EVERYTHING: Scope = Scope {
235    folders: Vec::new(),
236    holds: None,
237};

239/// The catalog under the folders.
240#[derive(Default)]
241struct Held {
242    tracks: BTreeSet<Id>,
243    releases: BTreeSet<Id>,
244    /// Artists by their catalog key rather than their identifier: two of the
245    /// passes read the attributed layer instead of the catalog, and a stored
246    /// record carries a key and no identifier.
247    artists: BTreeSet<String>,
248    /// Labels credited on a release in scope, by their catalog key — the
249    /// same reason `artists` is keyed rather than indexed.
250    labels: BTreeSet<String>,
251}
252
253impl Scope {
254    /// The scope the folders name, refusing one the catalog has never seen.
255    ///
256    /// That refusal is [`super::scope_from`]'s, the same one `check` and
257    /// `playlist` make and for the same reason: a folder added since the last
258    /// scan would otherwise produce a run with nothing to do and a cheerful
259    /// line saying so, which reads as *there is nothing to fetch here* and is
260    /// in fact *I have never heard of that folder*.
261    pub(super) fn of(
262        catalog: &Catalog,
263        typed: &[String],
264    ) -> Result<Scope, Box<dyn std::error::Error>> {
265        if typed.is_empty() {
266            return Ok(Scope::default());
267        }
268        let folders = super::scope_from(typed, catalog)?;
269        let mut holds = Held::default();
270        for track in &catalog.tracks {
271            let Some(file) = catalog.file(track.file_id) else {
272                continue;
273            };
274            if !super::in_scope(&file.path, &folders) {
275                continue;
276            }
277            holds.tracks.insert(track.id);
278            if let Some(release) = track.release_id {
279                holds.releases.insert(release);
280            }
281        }
282        // The album artist is credited on the record rather than on each of
283        // its tracks, so a folder holding a record whose tracks name only the
284        // guests would otherwise not reach the person whose record it is.
285        for id in &holds.releases {
286            let Some(artist) = catalog
287                .release(*id)
288                .and_then(|release| release.album_artist_id)
289                .and_then(|id| catalog.artist(id))
290            else {
291                continue;
292            };
293            holds.artists.insert(artist.key.clone());
294        }
295        // Every label credited on a release in scope, the same derivation as
296        // the artist above and for the same reason: a label is never itself
297        // under a folder, only the records that name it are.
298        for id in &holds.releases {
299            let Some(release) = catalog.release(*id) else {
300                continue;
301            };
302            for &label_id in &release.label_ids {
303                if let Some(label) = catalog.label(label_id) {
304                    holds.labels.insert(label.key.clone());
305                }
306            }
307        }
308        Ok(Scope {
309            folders,
310            holds: Some(holds),
311        })
312    }
313
314    /// `true` when no folder was given, so the whole library is in reach.
315    pub(super) fn is_empty(&self) -> bool {
316        self.holds.is_none()
317    }
318
319    /// The folders, for a message that names them.
320    pub(super) fn folders(&self) -> &[String] {
321        &self.folders
322    }
323
324    pub(super) fn has_track(&self, id: Id) -> bool {
325        match &self.holds {
326            None => true,
327            Some(held) => held.tracks.contains(&id),
328        }
329    }
330
331    pub(super) fn has_release(&self, id: Id) -> bool {
332        match &self.holds {
333            None => true,
334            Some(held) => held.releases.contains(&id),
335        }
336    }
337
338    /// `true` when the artist filed under this key has something in reach.
339    pub(super) fn has_artist(&self, key: &str) -> bool {
340        match &self.holds {
341            None => true,
342            Some(held) => held.artists.contains(key),
343        }
344    }
345
346    /// `true` when the label filed under this key is credited on something in
347    /// reach.
348    pub(super) fn has_label(&self, key: &str) -> bool {
349        match &self.holds {
350            None => true,
351            Some(held) => held.labels.contains(key),
352        }
353    }
354}

365/// How a run was narrowed, worded for a message; empty when it was not.
366///
367/// A name and a folder are two different questions, and a run narrowed by both
368/// says both rather than running them together into a list the reader has to
369/// sort out again.
370pub(super) fn narrowing(names: &[String], scope: &Scope) -> String {
371    match (names.is_empty(), scope.is_empty()) {
372        (true, true) => String::new(),
373        (false, true) => names.join(", "),
374        (true, false) => scope.folders().join(", "),
375        (false, false) => format!("{} under {}", names.join(", "), scope.folders().join(", ")),
376    }
377}

383    } else {
384        format!("nothing to ask about for {narrowed}")

389/// `true` when one of the names typed reaches this thing.
390///
391/// Empty means everything, which is what makes a bare `aede fetch --covers`
392/// the whole library. A name matches on **any** of the strings offered — for
393/// an album that is its title and its artist, so `--covers manson` finds the
394/// records as well as the person, exactly as the ordinary fetch does.
395///
396/// Matching is `contains` on the normalised form, the same rule as the
397/// ordinary fetch: a reader who types `pink` should not have to remember
398/// whether the band is filed as "Pink Floyd" or "The Pink Floyd Sound".
399pub(super) fn reaches(wanted: &[String], candidates: &[&str]) -> bool {
400    wanted.is_empty()
401        || candidates.iter().any(|candidate| {
402            let key = text::normalize(candidate);
403            wanted.iter().any(|w| key.contains(w.as_str()))
404        })
405}

407/// MusicBrainz artist identities that can have local artwork beside an album.
408pub(super) fn artist_image_candidates<'a>(
409    catalog: &'a Catalog,
410    held: &'a sources::Sources,
411    wanted: &[String],
412    scope: &Scope,
413) -> Vec<(&'a SourceRecord, &'a Artist, &'a ArtistFacts, &'a str)> {
414    held.records
415        .iter()
416        .filter_map(|record| {
417            if record.source != sources::MUSICBRAINZ
418                || !reaches(wanted, &[record.key.as_str()])
419                || !scope.has_artist(&record.key)
420            {
421                return None;
422            }
423            let Facts::Artist(facts) = &record.facts else {
424                return None;
425            };
426            let mbid = record.source_id.as_deref()?;
427            let artist = catalog
428                .artists
429                .iter()
430                .find(|artist| artist.key == record.key)?;
431            has_album(catalog, artist.id).then_some((record, artist, facts, mbid))
432        })
433        .collect()
434}

436/// What to say when names were given and nothing came of them.
437///
438/// Three states again, and only the first two used to be told apart. "No such
439/// artist here" and "that artist is already done" are different problems with
440/// different next steps, and printing the general "run fetch first" for both
441/// sends somebody to re-run a pass that has nothing to do.
442pub(super) fn nothing_named(wanted: &[String], scope: &Scope, but_for_full: usize) -> String {
443    let named = narrowing(wanted, scope);
444    match but_for_full {
445        0 => format!("nothing here matches {named}"),
446        _ => format!(
447            "{} matching {named}, already done: --full asks again",
448            ui::plural(but_for_full, "artist")
449        ),
450    }
451}

453/// What the reader asked for, gathered once and handed to every pass.
454///
455/// The passes grew one parameter at a time — a name, a size, `--images`,
456/// `--dry-run` — until the cover pass took nine, which is the point at which a
457/// signature stops being read and starts being counted. Bundling them also
458/// makes the three passes **the same shape**, so adding a fourth is a call
459/// that looks like the others rather than a new argument list to invent.
460///
461/// Fields nobody but one pass reads — `size`, `images`, `key` — sit here all
462/// the same: they are things the reader arranged, which is what this is,
463/// whether they typed them or exported them.
464pub(super) struct Asked<'a> {
465    /// The names typed after the command; empty means the whole shelf.
466    pub names: &'a [String],
467    /// The folders typed after the command, and what they hold; empty means
468    /// the whole library.
469    ///
470    /// Beside the names rather than folded into them: a name asks *who* and a
471    /// folder asks *where*, they narrow a run independently, and a pass given
472    /// both honours both.
473    pub scope: &'a Scope,
474    /// `--full`: ask again about what is already held.
475    pub again: bool,
476    /// `--dry-run`: say what would happen and do none of it.
477    pub dry_run: bool,
478    /// `--size`: how large an image to keep.
479    pub size: aede_core::coverart::Size,
480    /// `--images`: keep the pictures that are not the cover.
481    pub images: bool,
482    /// Which parts of a Fanart.tv answer this run should keep.
483    pub fanart: FanartOptions,
484    /// Which language the prose is wanted in, most wanted first.
485    ///
486    /// `--lang` when it was given, the shell's own locale otherwise, and
487    /// English last in either case — for a great many artists it is the only
488    /// article there is, and being *last* means it never displaces a language
489    /// the reader actually asked for. Gathered here rather than read inside the
490    /// pass, like the key below and for the same reason.
491    pub langs: Vec<String>,
492    /// The AcoustID key, from `AEDE_ACOUSTID_KEY`, when this machine has one.
493    ///
494    /// Read here rather than inside the pass that needs it, for the reason
495    /// [`aede_core::acoustid::key`] gives: the process environment is shared by
496    /// every thread, so a pass that reaches for it cannot be handed a different
497    /// answer by a test. Gathered at the edge, like everything else in here.
498    pub key: Option<String>,
499    /// The Fanart.tv key, from `AEDE_FANARTTV_KEY`, when this machine has one.
500    ///
501    /// Gathered here for the same reason [`Asked::key`] is — see
502    /// [`aede_core::fanarttv::key`]. `--portraits` and `--logos` are the only
503    /// passes that read it — a portrait tries Wikidata first and needs no key
504    /// at all, a logo has no such alternative and needs one for every artist.
505    pub portrait_key: Option<String>,
506}

538/// Prints the list a pass would have asked about, and says nothing was.
539///
540/// **`--dry-run` is a promise, and two of the four passes were not keeping it.**
541/// `--covers` and `--identify` stopped; `--summaries` and `--discography` read
542/// the flag, said nothing, and went to the network — which for summaries is two
543/// requests per artist against Wikimedia by somebody who had just said *ask
544/// nothing*. The rule was stated in prose on the first pass that honoured it,
545/// which is exactly how the third and fourth came not to: **a rule stated in a
546/// comment is a rule the next caller does not have.**
547///
548/// So the sentinel is a function. A pass prints its own plan — the titles, the
549/// artists, whatever it is about — and then calls this, which prints the one
550/// line and answers whether to stop. Four call sites, one wording, and an
551/// end-to-end test walks all of them.
552pub(super) fn asked_nothing(asked: &Asked, what: &[String]) -> bool {
553    if !asked.dry_run {
554        return false;
555    }
556    for one in what {
557        println!("  {}", ui::dim(one));
558    }
559    println!("  {}", ui::dim("nothing was asked: --dry-run"));
560    true
561}

563/// A pass `fetch` can be asked for instead of its ordinary run.
564///
565/// Each answers a different question, and any combination of them is allowed.
566/// What is **not** allowed is the typed order deciding anything: the passes go
567/// out from the artist — who they are, what they recorded, what the records
568/// look like — and running them the other way round would ask about albums
569/// before the fetch that names them. So the order is fixed here and the
570/// command says so when there is more than one.
571#[derive(Debug, Clone, Copy, PartialEq, Eq)]
572enum Pass {
573    /// Wikipedia, through artist or label Wikidata links already in the layer.
574    Summaries,
575    /// Everything MusicBrainz credits to each artist.
576    Discography,
577    /// The front image of every album that has none.
578    Covers,
579    /// The words, for tracks that have none — see [`super::lyrics`].
580    Lyrics,
581    /// What AcoustID hears in the files that have been fingerprinted.
582    Identify,
583    /// Rich recording and work relationships already identified in local tags.
584    Credits,
585    /// A picture of the artist, from Wikidata or Fanart.tv — see
586    /// [`super::portraits`].
587    Portraits,
588    /// The artist's logo, from Fanart.tv — see [`super::logos`].
589    Logos,
590    /// A label's own MusicBrainz identifier, closing the gap passive capture
591    /// leaves — see [`super::labels`].
592    Labels,
593}
594
595impl Pass {
596    /// The passes asked for, in the order they will run.
597    fn asked_for(args: &Args) -> Vec<Pass> {
598        [
599            ("labels", Pass::Labels),
600            ("summaries", Pass::Summaries),
601            ("discography", Pass::Discography),
602            ("covers", Pass::Covers),
603            ("lyrics", Pass::Lyrics),
604            ("identify", Pass::Identify),
605            ("credits", Pass::Credits),
606            ("recordings", Pass::Credits),
607            ("portraits", Pass::Portraits),
608            ("logos", Pass::Logos),
609            ("fanart", Pass::Logos),
610        ]
611        .into_iter()
612        .filter(|(flag, _)| args.has(flag))
613        .map(|(_, pass)| pass)
614        .fold(Vec::new(), |mut passes, pass| {
615            if !passes.contains(&pass) {
616                passes.push(pass);
617            }
618            passes
619        })
620    }
621
622    /// The option that asks for it, for a message to name.
623    fn option(self) -> &'static str {
624        match self {
625            Pass::Summaries => "--summaries",
626            Pass::Discography => "--discography",
627            Pass::Covers => "--covers",
628            Pass::Lyrics => "--lyrics",
629            Pass::Identify => "--identify",
630            Pass::Credits => "--credits",
631            Pass::Portraits => "--portraits",
632            Pass::Logos => "--logos",
633            Pass::Labels => "--labels",
634        }
635    }
636
637    /// Whether it needs the catalog, which decides whether one is loaded.
638    ///
639    /// `--summaries` does not: its input is the wikidata link already stored,
640    /// and failing on a missing catalog would be a refusal with no reason
641    /// behind it. Loading one anyway "for symmetry" would break that.
642    ///
643    /// A **folder** overrides this, in [`run_with`] rather than here: the
644    /// catalog is the only thing that can say which artists a folder holds, so
645    /// `aede fetch --summaries ~/Music/Alastis` needs one even though the pass
646    /// alone does not. The pass still never reads it — it is handed the answer.
647    fn needs_the_catalog(self) -> bool {
648        self != Pass::Summaries
649    }
650}

652/// Runs the passes asked for, in order, stopping at the first that cannot go on.
653///
654/// A pass returning `Err` means it could not do its work at all — the layer
655/// could not be written, an option was unusable. Individual requests that fail
656/// are counted inside each pass and do not stop the next one, which is the
657/// distinction that lets a run over a large library survive a bad afternoon.
658///
659/// One argument per thing a pass might need to do its work — network, retry,
660/// storage, and where and whose library this is — because folding them into
661/// a context struct would still need one field per parameter and one
662/// accessor per call site, for no reader's benefit.
663#[allow(clippy::too_many_arguments)]
664fn second_passes(
665    passes: &[Pass],
666    args: &Args,
667    transport: &mut dyn Ask,
668    backoff: &[std::time::Duration],
669    held: &mut sources::Sources,
670    path: &std::path::Path,
671    data_dir: &std::path::Path,
672    asked: &Asked,
673    catalog: Option<&Catalog>,
674) -> Res {
675    // The order is not the typed one, so it is stated rather than left to be
676    // inferred from the order the sections happen to come out in.
677    if passes.len() > 1 {
678        println!(
679            "  {}",
680            ui::dim(&format!(
681                "{} in this order: {}",
682                ui::plural(passes.len(), "pass"),
683                passes
684                    .iter()
685                    .map(|p| p.option())
686                    .collect::<Vec<_>>()
687                    .join(", ")
688            ))
689        );
690    }
691
692    for pass in passes {
693        match pass {
694            Pass::Summaries => {
695                super::summaries::run(transport, backoff, &asked.langs, held, path, asked)?;
696            }
697            Pass::Discography => {
698                let catalog = catalog.expect("a catalog was loaded for it");
699                super::discography::run(catalog, transport, backoff, held, path, asked)?;
700            }
701            Pass::Covers => {
702                let catalog = catalog.expect("a catalog was loaded for it");
703                super::covers::run(catalog, transport, backoff, held, path, asked)?;
704            }
705            Pass::Lyrics => {
706                let catalog = catalog.expect("a catalog was loaded for it");
707                // The only pass that writes nothing into the attributed layer:
708                // its answer is a file beside the music, which the next scan
709                // discovers, exactly as it would one put there by hand.
710                super::lyrics::run(args, catalog, transport, backoff, asked)?;
711            }
712            Pass::Identify => {
713                let catalog = catalog.expect("a catalog was loaded for it");
714                super::identify::run(catalog, transport, backoff, held, path, asked)?;
715            }
716            Pass::Portraits => {
717                let catalog = catalog.expect("a catalog was loaded for it");
718                super::portraits::run(
719                    catalog,
720                    transport,
721                    backoff,
722                    held,
723                    path,
724                    data_dir,
725                    asked.portrait_key.as_deref(),
726                    asked,
727                )?;
728            }
729            Pass::Logos => {
730                let catalog = catalog.expect("a catalog was loaded for it");
731                super::logos::run(
732                    catalog,
733                    transport,
734                    backoff,
735                    held,
736                    path,
737                    data_dir,
738                    asked.portrait_key.as_deref(),
739                    asked,
740                )?;
741            }
742            Pass::Labels => {
743                let catalog = catalog.expect("a catalog was loaded for it");
744                super::labels::run(catalog, transport, backoff, held, path, asked)?;
745            }
746            Pass::Credits => {
747                let catalog = catalog.expect("a catalog was loaded for it");
748                if rich_credits(catalog, transport, backoff, held, path, asked, args)? {
749                    edition_credits(catalog, transport, backoff, held, path, asked, args)?;
750                }
751            }
752        }
753    }
754    Ok(())
755}

757/// Retrieves MusicBrainz recording credits, work relationships and work
758/// credits for recordings local tags already identify. No title search is
759/// offered: without a recording MBID there is no safe question to ask.
760fn rich_credits(
761    catalog: &Catalog,
762    transport: &mut dyn Ask,
763    backoff: &[std::time::Duration],
764    held: &mut sources::Sources,
765    path: &std::path::Path,
766    asked: &Asked,
767    args: &Args,
768) -> Result<bool, Box<dyn std::error::Error>> {
769    let mut targets = Vec::new();
770    for recording in &catalog.recordings {
771        let Some(mbid) = recording.mbid.as_deref() else {
772            continue;
773        };
774        let Some(&track_id) = recording
775            .track_ids
776            .iter()
777            .find(|&&track_id| asked.scope.has_track(track_id))
778        else {
779            continue;
780        };
781        let Some(entity) = EntityRef::of(catalog, EntityKind::Track, track_id) else {
782            continue;
783        };
784        if !asked.again
785            && held
786                .get(&entity, sources::MUSICBRAINZ)
787                .is_some_and(|record| {
788                    matches!(&record.facts, Facts::Track(facts) if facts.relationships_complete)
789                })
790        {
791            continue;
792        }
793        if !reaches(asked.names, &[recording.title.as_str()]) {
794            continue;
795        }
796        targets.push((entity, recording.title.as_str(), mbid));
797    }
798    println!("{}", ui::section("Recording and work credits"));
799    if targets.is_empty() {
800        println!("  {}", ui::dim("no identified recording is waiting"));
801        return Ok(true);
802    }
803    if asked.dry_run {
804        for (_, title, _) in &targets {
805            println!("  {}", ui::dim(title));
806        }
807        println!("  {}", ui::dim("nothing was asked: --dry-run"));
808        return Ok(true);
809    }
810    let total = targets.len();
811    let estimate_ms = total as u64 * musicbrainz::REQUEST_INTERVAL.as_millis() as u64;
812    println!(
813        "  {} at one request per second, about {}",
814        ui::plural(total, "recording"),
815        ui::long_duration(estimate_ms)
816    );
817    if total > CONFIRM_ABOVE && !super::confirmed(args, "ask about all of them")? {
818        println!("  {}", ui::dim("nothing was asked"));
819        return Ok(false);
820    }
821    let (mut stored, mut refused, mut failed) = (0, 0, 0);
822    for (index, (entity, title, mbid)) in targets.into_iter().enumerate() {
823        print!("\r  asking: {}/{}", index + 1, total);
824        let _ = std::io::Write::flush(&mut std::io::stdout());
825        let url = format!(
826            "{}/recording/{mbid}?fmt=json&inc={}",
827            musicbrainz::WEB_SERVICE,
828            musicbrainz::RECORDING_INCLUDES
829        );
830        match ask_with_backoff(transport, &url, backoff) {
831            Ok(answer) => match musicbrainz::recording(&answer) {
832                Some(candidate) => {
833                    held.set(SourceRecord {
834                        key: entity.key,
835                        source: sources::MUSICBRAINZ.to_string(),
836                        source_id: Some(candidate.mbid),
837                        fetched_at: clock::now_seconds(),
838                        confidence: sources::Confidence::Identified,
839                        facts: Facts::Track(candidate.facts),
840                    });
841                    sources::save(held, path)?;
842                    stored += 1;
843                }
844                None => {
845                    refused += 1;
846                    eprintln!(
847                        "  {} {title}: no readable MusicBrainz recording",
848                        ui::yellow("?")
849                    );
850                }
851            },
852            Err(Refusal::RateLimited) => {
853                println!();
854                sources::save(held, path)?;
855                return Err(
856                    "MusicBrainz is rate limiting requests; saved answers are safe, try later"
857                        .into(),
858                );
859            }
860            Err(error) => {
861                failed += 1;
862                eprintln!("\n  {} {title}: {error}", ui::red("×"));
863            }
864        }
865    }
866    println!();
867    println!(
868        "{} {stored} stored, {refused} left alone, {failed} failed",
869        ui::green("→")
870    );
871    Ok(true)
872}

874/// Fetches relationships attached to precise editions, separately from
875/// recording/work credits. A release group cannot safely stand in for one.
876fn edition_credits(
877    catalog: &Catalog,
878    transport: &mut dyn Ask,
879    backoff: &[std::time::Duration],
880    held: &mut sources::Sources,
881    path: &std::path::Path,
882    asked: &Asked,
883    args: &Args,
884) -> Res {
885    let targets: Vec<_> = catalog
886        .releases
887        .iter()
888        .filter_map(|release| {
889            let mbid = release.mbid.as_deref()?;
890            if !asked.scope.has_release(release.id) {
891                return None;
892            }
893            let artist = release
894                .album_artist_id
895                .and_then(|id| catalog.artist(id))
896                .map(|artist| artist.name.as_str())
897                .unwrap_or("");
898            if !reaches(asked.names, &[&release.title, artist]) {
899                return None;
900            }
901            let entity = EntityRef::of(catalog, EntityKind::Release, release.id)?;
902            if !asked.again
903                && held
904                    .get(&entity, sources::MUSICBRAINZ)
905                    .is_some_and(|record| {
906                        matches!(&record.facts, Facts::Release(facts)
907                            if facts.relationships_complete
908                            && facts.edition_mbid.as_deref() == Some(mbid))
909                    })
910            {
911                return None;
912            }
913            Some((entity, release.title.as_str(), mbid))
914        })
915        .collect();
916    println!("{}", ui::section("Edition credits"));
917    if targets.is_empty() {
918        println!("  {}", ui::dim("no identified edition is waiting"));
919        return Ok(());
920    }
921    if asked.dry_run {
922        for (_, title, _) in &targets {
923            println!("  {}", ui::dim(title));
924        }
925        println!("  {}", ui::dim("nothing was asked: --dry-run"));
926        return Ok(());
927    }
928    let total = targets.len();
929    let estimate_ms = total as u64 * musicbrainz::REQUEST_INTERVAL.as_millis() as u64;
930    println!(
931        "  {} at one request per second, about {}",
932        ui::plural(total, "edition"),
933        ui::long_duration(estimate_ms)
934    );
935    if total > CONFIRM_ABOVE && !super::confirmed(args, "ask about all of them")? {
936        println!("  {}", ui::dim("nothing was asked"));
937        return Ok(());
938    }
939    let (mut stored, mut refused, mut failed) = (0, 0, 0);
940    for (index, (entity, title, mbid)) in targets.into_iter().enumerate() {
941        print!("\r  asking: {}/{}", index + 1, total);
942        let _ = std::io::Write::flush(&mut std::io::stdout());
943        let url = format!(
944            "{}/release/{mbid}?fmt=json&inc={}",
945            musicbrainz::WEB_SERVICE,
946            musicbrainz::RELEASE_INCLUDES
947        );
948        match ask_with_backoff(transport, &url, backoff) {
949            Ok(answer) => match musicbrainz::release(&answer) {
950                Some(candidate) if candidate.facts.edition_mbid.as_deref() == Some(mbid) => {
951                    let mut facts = candidate.facts;
952                    if let Some(previous) = held.get(&entity, sources::MUSICBRAINZ)
953                        && let Facts::Release(previous) = &previous.facts
954                    {
955                        facts.cover_art = facts.cover_art.or_else(|| previous.cover_art.clone());
956                    }
957                    held.set(SourceRecord {
958                        key: entity.key,
959                        source: sources::MUSICBRAINZ.to_string(),
960                        source_id: Some(candidate.mbid),
961                        fetched_at: clock::now_seconds(),
962                        confidence: sources::Confidence::Identified,
963                        facts: Facts::Release(facts),
964                    });
965                    sources::save(held, path)?;
966                    stored += 1;
967                }
968                _ => {
969                    refused += 1;
970                    eprintln!(
971                        "\n  {} {title}: no matching MusicBrainz edition",
972                        ui::yellow("?")
973                    );
974                }
975            },
976            Err(Refusal::RateLimited) => {
977                println!();
978                sources::save(held, path)?;
979                return Err(
980                    "MusicBrainz is rate limiting requests; saved answers are safe, try later"
981                        .into(),
982                );
983            }
984            Err(error) => {
985                failed += 1;
986                eprintln!("\n  {} {title}: {error}", ui::red("×"));
987            }
988        }
989    }
990    println!();
991    println!(
992        "{} {stored} stored, {refused} left alone, {failed} failed",
993        ui::green("→")
994    );
995    Ok(())
996}

1003/// [`run`], with the waits made explicit so a test does not have to sit
1004/// through them.
1005pub fn run_with(args: &Args, transport: &mut dyn Ask, backoff: &[std::time::Duration]) -> Res {
1006    let data_dir = super::data_dir(args);
1007    let path = sources::sources_path(&data_dir);
1008    let mut held = sources::load_all(&path)?.unwrap_or_default();
1009
1010    // An option that only means something to another pass, given on its own,
1011    // is refused rather than ignored: a reader who typed `--images` and got an
1012    // ordinary fetch would conclude the feature does not work.
1013    if !args.has("covers") {
1014        for option in ["images", "size"] {
1015            if args.has(option) {
1016                return Err(format!(
1017                    "--{option} belongs to the cover art pass: aede fetch --covers --{option}"
1018                )
1019                .into());
1020            }
1021        }
1022    }
1023    if !args.has("logos") && !args.has("fanart") && args.has("banners") {
1024        return Err("--banners belongs to a Fanart.tv pass: aede fetch --logos --banners".into());
1025    }
1026    let exclusions = [
1027        "no-logo",
1028        "no-label-logo",
1029        "no-portrait",
1030        "no-background",
1031        "no-banner",
1032        "no-album-cover",
1033        "no-cdart",
1034    ];
1035    if !args.has("fanart")
1036        && let Some(option) = exclusions.iter().find(|option| args.has(option))
1037    {
1038        return Err(format!(
1039            "--{option} belongs to the complete pass: aede fetch --fanart --{option}"
1040        )
1041        .into());
1042    }
1043    for (positive, negative) in [("logos", "no-logo"), ("banners", "no-banner")] {
1044        if args.has("fanart") && args.has(positive) && args.has(negative) {
1045            return Err(format!("--{positive} and --{negative} ask for opposite things").into());
1046        }
1047    }
1048
1049    // A second pass is a different question, often of a different service, so
1050    // each is its own run rather than a stage of the ordinary fetch:
1051    // `--summaries` alone does not re-ask MusicBrainz about a library it has
1052    // already answered on.
1053    //
1054    // Several of them together run one after another. They used to be three
1055    // `return`s in a row, so `--covers --discography` ran the covers and
1056    // **dropped the discography without a word** — the fault this program
1057    // refuses everywhere else: an option that cannot be honoured is refused,
1058    // never swallowed. Here it could be honoured, so it is.
1059    // What to ask about: the names given, the folders given, or every artist
1060    // in the library. Read here rather than inside the ordinary run, because
1061    // every pass honours them now — `aede fetch --discography mika` used to
1062    // swallow the word, and `aede fetch --lyrics ~/Music/Alastis` the path.
1063    let (folders, wanted) = folders_and_names(args);
1064    let passes = Pass::asked_for(args);
1065
1066    // Loaded once for the whole run, and only when something needs it. An
1067    // ordinary fetch always does; a pass may not; and a **folder** always
1068    // does, whatever the pass, because the catalog is the only thing that can
1069    // say what a folder holds.
1070    let catalog = match passes.is_empty()
1071        || !folders.is_empty()
1072        || passes.iter().any(|pass| pass.needs_the_catalog())
1073    {
1074        true => Some(super::load(args)?),
1075        false => None,
1076    };
1077    let scope = match &catalog {
1078        Some(catalog) => Scope::of(catalog, &folders)?,
1079        // A folder would have loaded one, so there is no folder to place.
1080        None => Scope::default(),
1081    };
1082    // Said before anything is asked, and on every run that gives one: reading
1083    // a positional as a folder is a *reading*, and a reading the person cannot
1084    // see is one they cannot correct.
1085    if !scope.is_empty() {
1086        println!(
1087            "  {}",
1088            ui::dim(&format!(
1089                "only what is under {}",
1090                scope.folders().join(", ")
1091            ))
1092        );
1093    }
1094
1095    // Read once for the whole run, and before anything is asked: a width the
1096    // archive does not generate must be refused before a summaries pass has
1097    // spent ten minutes on the network for it.
1098    let shell_locale = std::env::var("LC_ALL")
1099        .or_else(|_| std::env::var("LANG"))
1100        .ok();
1101    let asked = Asked {
1102        names: &wanted,
1103        scope: &scope,
1104        again: args.has("full"),
1105        dry_run: args.has("dry-run"),
1106        key: aede_core::acoustid::key(),
1107        portrait_key: aede_core::fanarttv::key(),
1108        // `--lang` is a statement of intent and the locale is a guess about
1109        // one, so the option wins. Neither is read anywhere but here.
1110        langs: super::summaries::preferred_langs(match args.value("lang") {
1111            Some(asked) => Some(asked),
1112            None => shell_locale.as_deref(),
1113        }),
1114        size: match args.value("size") {
1115            Some(text) => aede_core::coverart::Size::parse(text).ok_or_else(|| {
1116                format!(
1117                    "--size takes 250, 500, 1200 or original; \"{text}\" is not \
1118                     one the archive generates"
1119                )
1120            })?,
1121            None => super::covers::DEFAULT_SIZE,
1122        },
1123        images: args.has("images"),
1124        fanart: FanartOptions::from_args(args),
1125    };
1126
1127    if !passes.is_empty() {
1128        return second_passes(
1129            &passes,
1130            args,
1131            transport,
1132            backoff,
1133            &mut held,
1134            &path,
1135            &data_dir,
1136            &asked,
1137            catalog.as_ref(),
1138        );
1139    }
1140
1141    let catalog = catalog.expect("an ordinary fetch always loads one");
1142
1143    // No `--limit` here on purpose: everywhere else in this program it means
1144    // "show a window of the result", and bounding how much work is done is a
1145    // different thing wearing the same word. Naming artists narrows the run.
1146    let mut targets: Vec<(EntityRef, String, Option<String>)> = Vec::new();
1147    for artist in &catalog.artists {
1148        if !has_album(&catalog, artist.id) {
1149            continue;
1150        }
1151        // A blank name would go out as an empty query, which the search server
1152        // does not answer politely: it fails, and the failure looks like a
1153        // rate limit three steps from its cause.
1154        if artist.name.trim().is_empty() {
1155            continue;
1156        }
1157        let key = text::normalize(&artist.name);
1158        if !wanted.is_empty() && !wanted.iter().any(|w| key.contains(w.as_str())) {
1159            continue;
1160        }
1161        // A folder narrows by where the music is rather than by what it is
1162        // called, and the two are asked together: `aede fetch ozzy ~/Music/80s`
1163        // is the artist, on that shelf.
1164        if !scope.has_artist(&artist.key) {
1165            continue;
1166        }
1167        if let Some(entity) = EntityRef::of(&catalog, EntityKind::Artist, artist.id) {
1168            // Already answered, unless asked to do it again. A second run over
1169            // a library should cost what changed, not ten minutes again.
1170            if !args.has("full") && held.get(&entity, sources::MUSICBRAINZ).is_some() {
1171                continue;
1172            }
1173            targets.push((entity, artist.name.clone(), artist.mbid.clone()));
1174        }
1175    }
1176    // The albums, decided before anything is asked, so that one estimate and
1177    // one confirmation cover the whole run. Two prompts for one question is
1178    // how a confirmation becomes something a reader clicks through.
1179    let albums = super::releases::targets(&catalog, &held, &wanted, &scope, args.has("full"));
1180
1181    if targets.is_empty() && albums.is_empty() {
1182        println!("{}", ui::section("Fetch"));
1183        println!(
1184            "  {}",
1185            ui::dim(match wanted.is_empty() && scope.is_empty() {
1186                true => "every artist and album has already been asked about (--full asks again)",
1187                false => "nothing in this catalog matches, or it was already asked about",
1188            })
1189        );
1190        // Offered here too, and this is the exit that matters most: a reader
1191        // who fetched their library before this existed reaches *this* branch
1192        // every time, never the one below, and would never be told the second
1193        // pass is available. An announcement made only on the path that has
1194        // just done work is an announcement nobody who finished first ever
1195        // sees.
1196        offer_next_steps(&catalog, &held, &data_dir);
1197        return Ok(());
1198    }
1199
1200    // How long this will take, before it starts. A predicted duration is
1201    // usually a bad idea here — how long a read takes depends on the disk —
1202    // but this one is not a guess: the rate is fixed by the service, and every
1203    // album costs exactly one request whatever route it takes in.
1204    let asks = targets.len() + albums.len();
1205    let total_ms = asks as u64 * musicbrainz::REQUEST_INTERVAL.as_millis() as u64;
1206    println!("{}", ui::section("Fetch"));
1207    println!(
1208        "  {} and {} at one request per second, about {}",
1209        ui::plural(targets.len(), "artist"),
1210        ui::plural(albums.len(), "album"),
1211        ui::long_duration(total_ms)
1212    );
1213    if args.has("dry-run") {
1214        for (_, name, _) in &targets {
1215            println!("  {}", ui::dim(name));
1216        }
1217        for title in super::releases::names(&albums) {
1218            println!("  {}", ui::dim(title));
1219        }
1220        println!("  {}", ui::dim("nothing was asked: --dry-run"));
1221        return Ok(());
1222    }
1223
1224    // A run of ten minutes is something to agree to, not something to
1225    // discover. Short ones are not worth a question — a confirmation asked
1226    // every time is a confirmation nobody reads.
1227    if asks > CONFIRM_ABOVE && !super::confirmed(args, "ask about all of them")? {
1228        println!("  {}", ui::dim("nothing was asked"));
1229        return Ok(());
1230    }
1231
1232    let (mut stored, mut refused, mut failed) = (0usize, 0usize, 0usize);
1233    let mut pending = queue(&targets);
1234    let mut done = 0usize;
1235    while let Some((item @ (entity, name, mbid), retried)) = pending.pop_front() {
1236        print!("\r  asking: {}/{asks}", done + 1);
1237        let _ = std::io::Write::flush(&mut std::io::stdout());
1238
1239        // Ask by identifier when the tags carry one — Picard writes it, and a
1240        // library it has been through knows exactly which artist it means.
1241        // Searching by name there would replace an answer with a guess, and
1242        // it also asks a poorer question: a search result is abbreviated,
1243        // while a lookup returns the entity.
1244        let url = match mbid {
1245            Some(mbid) => format!(
1246                "{}/artist/{mbid}?fmt=json&inc={}",
1247                musicbrainz::WEB_SERVICE,
1248                musicbrainz::ARTIST_INCLUDES
1249            ),
1250            None => format!(
1251                "{}/artist/?query={}&fmt=json&limit=5",
1252                musicbrainz::WEB_SERVICE,
1253                encode(&musicbrainz::escape_query(name))
1254            ),
1255        };
1256        let answer = match ask_with_backoff(transport, &url, backoff) {
1257            Ok(answer) => answer,
1258            Err(Refusal::RateLimited) => {
1259                // Nothing is lost: what was stored stays stored, and the run
1260                // stops rather than hammering a service that has just said no.
1261                //
1262                // The name and the URL go into the message because the service
1263                // answers `503` to two different things — the rate being
1264                // exceeded, and its search backend refusing a query it could
1265                // not parse — and nothing in the response tells them apart.
1266                // Without them, a query this program built badly reads as
1267                // "you are going too fast", which is where an hour goes.
1268                println!();
1269                sources::save(&held, &path)?;
1270                return Err(format!(
1271                    "the service refused {} times in a row, waiting longer each \
1272                     time; that is a rate limit rather than a hiccup, so nothing \
1273                     more was asked.\n  it stopped on \"{name}\", asking:\n  {url}",
1274                    backoff.len() + 1
1275                )
1276                .into());
1277            }
1278            Err(other) if defer(&mut pending, item, retried, &other) => {
1279                // Not shown, not counted: the same name goes back to the end
1280                // of the queue instead, on the theory that whatever kept the
1281                // service from answering will often have passed by the time
1282                // everything else here has had its turn.
1283                continue;
1284            }
1285            Err(other) => {
1286                failed += 1;
1287                done += 1;
1288                eprintln!("\r  {} {name}: {other}", ui::red("×"));
1289                continue;
1290            }
1291        };
1292
1293        // A lookup answered about the identifier it was given: that is a
1294        // certainty, and the only thing in this program that produces one.
1295        let found = match mbid {
1296            Some(_) => musicbrainz::artist(&answer)
1297                .map(|c| (c, aede_core::sources::Confidence::Identified))
1298                .ok_or(musicbrainz::NoMatch::Nothing),
1299            None => musicbrainz::best_match(&musicbrainz::artists(&answer), name),
1300        };
1301        match found {
1302            Ok((candidate, confidence)) => {
1303                // The discography pass enriches this same MusicBrainz artist
1304                // row. A normal `--full` refresh answers the artist lookup
1305                // again, and that endpoint does not carry the separately
1306                // browsed release groups: replacing the row verbatim would
1307                // therefore erase a successful `fetch --discography` and make
1308                // the missing-albums report silently disappear.
1309                let mut facts = candidate.facts;
1310                if let Some(SourceRecord {
1311                    facts: Facts::Artist(existing),
1312                    ..
1313                }) = held.get(entity, sources::MUSICBRAINZ)
1314                {
1315                    facts.discography = existing.discography.clone();
1316                }
1317                held.set(SourceRecord {
1318                    key: entity.key.clone(),
1319                    source: sources::MUSICBRAINZ.to_string(),
1320                    source_id: Some(candidate.mbid),
1321                    fetched_at: clock::now_seconds(),
1322                    confidence,
1323                    facts: Facts::Artist(facts),
1324                });
1325                stored += 1;
1326                // Saved after each answer, not at the end: ten minutes of
1327                // waiting must not be undone by one interruption.
1328                sources::save(&held, &path)?;
1329            }
1330            Err(why) => {
1331                refused += 1;
1332                eprintln!("\r  {} {name}: {}", ui::yellow("?"), refusal(&why));
1333            }
1334        }
1335        done += 1;
1336    }
1337
1338    // The albums, in the same run and counted into the same report: they are
1339    // the same question asked of the same service, and splitting the totals
1340    // would leave the reader adding two lines up by eye.
1341    let (albums_stored, albums_refused, albums_failed) = super::releases::run(
1342        transport,
1343        backoff,
1344        &albums,
1345        &mut held,
1346        &path,
1347        targets.len(),
1348        asks,
1349    )?;
1350    stored += albums_stored;
1351    refused += albums_refused;
1352    failed += albums_failed;
1353    println!();
1354
1355    println!(
1356        "{} {stored} stored, {refused} left alone, {failed} failed",
1357        ui::green("→")
1358    );
1359    if refused > 0 {
1360        // A refusal is the design working, not a fault, and a reader who is
1361        // not told that will read it as one.
1362        println!(
1363            "  {}",
1364            ui::dim(
1365                "left alone means no answer was clearly about that artist — nothing was guessed"
1366            )
1367        );
1368    }
1369    offer_next_steps(&catalog, &held, &data_dir);
1370    println!("  {}", ui::dim(&path.display().to_string()));
1371    Ok(())
1372}

1548/// The `User-Agent`, refused rather than sent empty.
1549///

1563                .into(),
1564        );
1565    }
1566    Ok(format!("aede/{version} ( {contact} )"))
1567}

1569/// Asks, and asks again after waiting when the service says "not now".
1570///
1571/// Only `503` is retried. A failure to reach the service at all, or an answer
1572/// that is not JSON, will not be cured by waiting — retrying those would only
1573/// take three times as long to report the same thing.
1574pub(super) fn ask_with_backoff(
1575    transport: &mut dyn Ask,
1576    url: &str,
1577    backoff: &[std::time::Duration],
1578) -> Result<Json, Refusal> {
1579    request_with_backoff(|| transport.get_json(url), backoff)
1580}

1582/// [`ask_with_backoff`], for bytes.
1583///
1584/// Promoted here once a second caller needed it — `fetch --covers` first,
1585/// `fetch --portraits` after — rather than kept as two copies that would
1586/// eventually differ.
1587pub(super) fn ask_bytes(
1588    transport: &mut dyn Ask,
1589    url: &str,
1590    backoff: &[std::time::Duration],
1591) -> Result<Vec<u8>, Refusal> {
1592    request_with_backoff(|| transport.get_bytes(url), backoff)
1593}
1594
1595fn request_with_backoff<T>(
1596    mut request: impl FnMut() -> Result<T, Refusal>,
1597    backoff: &[std::time::Duration],
1598) -> Result<T, Refusal> {
1599    let mut attempt = 0;
1600    loop {
1601        match request() {
1602            Err(Refusal::RateLimited) if attempt < backoff.len() => {
1603                std::thread::sleep(backoff[attempt]);
1604                attempt += 1;
1605            }
1606            other => return other,
1607        }
1608    }
1609}

1611/// Turns a plain list into a queue where nothing has been retried yet.
1612///
1613/// A queue rather than a plain list because of what a caller does with a
1614/// [`Refusal`] that [`worth_deferring`] returns `true` for: push the same
1615/// target back onto the end, with the flag now `true`, instead of reporting
1616/// it. Everything already asked before it gets its turn first, which is
1617/// usually enough time for a service that merely stumbled to have recovered —
1618/// so asking it again is the exception rather than the rule. A target only
1619/// gets that once: `true` on the way back in means the next refusal is final.
1620pub(super) fn queue<T>(targets: &[T]) -> std::collections::VecDeque<(&T, bool)> {
1621    targets.iter().map(|target| (target, false)).collect()
1622}
1623
1624/// Gives an unreachable target one more turn after the rest of the queue.
1625pub(super) fn defer<'a, T>(
1626    pending: &mut std::collections::VecDeque<(&'a T, bool)>,
1627    target: &'a T,
1628    retried: bool,
1629    why: &Refusal,
1630) -> bool {
1631    if !retried && worth_deferring(why) {
1632        pending.push_back((target, true));
1633        true
1634    } else {
1635        false
1636    }
1637}
1638
1639/// Whether a refusal is worth asking again later rather than reporting now.
1640///
1641/// Only [`Refusal::Unreachable`] qualifies: no route, no name, a timeout are
1642/// all a property of one attempt, and often gone by the time everything else
1643/// in the queue has had its turn. A `404`, a rate limit that has used up its
1644/// own retries, or an answer that was not JSON will not read differently for
1645/// having waited — deferring those would only delay the same report.
1646pub(super) fn worth_deferring(refusal: &Refusal) -> bool {
1647    matches!(refusal, Refusal::Unreachable(_))
1648}

1650/// Says why an answer was not taken, in words rather than a variant name.
1651pub(super) fn refusal(why: &aede_core::musicbrainz::NoMatch) -> String {
1652    use aede_core::musicbrainz::NoMatch;
1653    match why {
1654        NoMatch::Nothing => "MusicBrainz knows nobody by that name".to_string(),
1655        NoMatch::Ambiguous(names) => {
1656            format!("several answers are equally good: {}", names.join(", "))
1657        }
1658        NoMatch::TooWeak { best, score } => {
1659            format!("the closest was \"{best}\" at {score}%, not close enough")
1660        }
1661    }
1662}

1664/// Percent-encodes a query value.
1665///
1666/// Written by hand for the same reason the JSON reader is: one dependency, and
1667/// this is thirty lines. The unreserved set is RFC 3986's.
1668pub(super) fn encode(text: &str) -> String {
1669    let mut out = String::with_capacity(text.len());
1670    for byte in text.as_bytes() {
1671        match byte {
1672            b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'_' | b'.' | b'~' => {
1673                out.push(*byte as char)
1674            }
1675            _ => out.push_str(&format!("%{byte:02X}")),
1676        }
1677    }
1678    out
1679}
