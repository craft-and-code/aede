1//! Portable personal corrections and decisions.
2//!
3//! A backup preserves everything. A rules bundle is deliberately narrower:
4//! the small set of human decisions that can be replayed on another catalog
5//! without copying listening history or re-fetched biographies and artwork.
6
7use aede_core::json::Json;
8use aede_core::{sources, user};
9
10use super::{Res, data_dir};
11use crate::args::Args;
12use crate::ui::{self, Table};

14const RULES_FORMAT_VERSION: u32 = 1;
15
16/// Lists, exports, or imports reproducible personal rules.
17pub fn rules(args: &Args) -> Res {
18    if args.has("export") && args.has("import") {
19        return Err("--export and --import ask for opposite operations".into());
20    }
21    if args.value("output").is_some() && !args.has("export") {
22        return Err("--output means nothing without --export on rules".into());
23    }
24    let directory = data_dir(args);
25    let user_path = user::user_path(&directory);
26    let sources_path = sources::sources_path(&directory);
27    let mut personal = user::load(&user_path)?.unwrap_or_default();
28    let mut evidence = sources::load_all(&sources_path)?.unwrap_or_default();
29
30    if args.has("export") {
31        return export(args, &personal, &evidence);
32    }
33    if let Some(path) = args.value("import") {
34        let text = std::fs::read_to_string(path)
35            .map_err(|error| format!("cannot read \"{path}\": {error}"))?;
36        let root = aede_core::json::parse(&text)
37            .map_err(|error| format!("\"{path}\" is not JSON: {error}"))?;
38        if root.field_str("format").as_deref() != Some("aede-rules") {
39            return Err("this is not an Aède rules bundle".into());
40        }
41        let version = root.field_u32("format_version").unwrap_or(0);
42        if version != RULES_FORMAT_VERSION {
43            return Err(format!(
44                "rules format version {version} is not supported; expected {RULES_FORMAT_VERSION}"
45            )
46            .into());
47        }
48        let incoming_user = user::from_json(
49            root.get("user")
50                .ok_or_else(|| "rules bundle has no user document".to_string())?,
51        )?;
52        let incoming_sources = sources::from_json(
53            root.get("sources")
54                .ok_or_else(|| "rules bundle has no sources document".to_string())?,
55        )?;
56        if !incoming_user.annotations.is_empty()
57            || !incoming_user.plays.is_empty()
58            || !incoming_user.counts.is_empty()
59            || !incoming_user.track_identities.is_empty()
60            || !incoming_user.relinks.is_empty()
61        {
62            return Err("a rules bundle may contain relation notes, collections, set-aside releases, and artist filing rules; use notes or backup for other personal data".into());
63        }
64        if incoming_sources
65            .records
66            .iter()
67            .any(|record| record.source != "manual")
68        {
69            return Err("a rules bundle may contain only manual source records; use sources --import for external evidence".into());
70        }
71        // Check both stores before the first write. This avoids applying half
72        // a bundle when an existing destination is a link or special file.
73        std::fs::create_dir_all(&directory)?;
74        let base = directory.canonicalize()?;
75        for name in [user::USER_FILE, sources::SOURCES_FILE] {
76            aede_core::copy::validate_destination(&base, std::path::Path::new(name))?;
77        }
78
79        let report = user::merge(&mut personal, incoming_user);
80        let mut source_changes = 0usize;
81        for record in incoming_sources.records {
82            evidence.set(record);
83            source_changes += 1;
84        }
85        for review in incoming_sources.reviews {
86            evidence.set_review(review);
87            source_changes += 1;
88        }
89        for decision in incoming_sources.credit_exclusions {
90            evidence.exclude_credit(decision.relation, decision.excluded_at);
91            source_changes += 1;
92        }
93        user::save(&personal, &user_path)?;
94        sources::save(&evidence, &sources_path)?;
95        println!(
96            "{} {} added, {} updated, {} kept, {} source decisions imported",
97            ui::green("→"),
98            report.added,
99            report.updated,
100            report.kept,
101            source_changes
102        );
103        return Ok(());
104    }
105
106    print_summary(&personal, &evidence);
107    Ok(())
108}

110fn export(args: &Args, personal: &user::UserData, evidence: &sources::Sources) -> Res {
111    let portable_user = user::UserData {
112        relation_annotations: personal.relation_annotations.clone(),
113        collections: personal.collections.clone(),
114        set_aside: personal.set_aside.clone(),
115        same_artist: personal.same_artist.clone(),
116        ..Default::default()
117    };
118    let portable_sources = sources::Sources {
119        records: evidence
120            .records
121            .iter()
122            .filter(|record| record.source == "manual")
123            .cloned()
124            .collect(),
125        reviews: evidence.reviews.clone(),
126        credit_exclusions: evidence.credit_exclusions.clone(),
127    };
128    let mut root = Json::obj();
129    root.set("format", "aede-rules".into());
130    root.set("format_version", RULES_FORMAT_VERSION.into());
131    root.set("exported_at", aede_core::clock::now_seconds().into());
132    root.set("user", user::to_json(&portable_user));
133    root.set("sources", sources::to_json(&portable_sources));
134    super::export::emit(args, &root.to_string_pretty())
135}

137fn print_summary(personal: &user::UserData, evidence: &sources::Sources) {
138    let accepted = evidence
139        .reviews
140        .iter()
141        .filter(|review| review.decision == sources::ReviewDecision::Accepted)
142        .count();
143    let rejected = evidence.reviews.len().saturating_sub(accepted);
144    let manual = evidence
145        .records
146        .iter()
147        .filter(|record| record.source == "manual")
148        .count();
149    println!("{}", ui::section("Personal rules"));
150    let mut table = Table::new(&["Rule", "Count"]);
151    for (name, count) in [
152        ("accepted source identities", accepted),
153        ("rejected source identities", rejected),
154        ("manual source records", manual),
155        ("artist filing rules", personal.same_artist.len()),
156        ("missing releases set aside", personal.set_aside.len()),
157        ("relation annotations", personal.relation_annotations.len()),
158        ("excluded credits", evidence.credit_exclusions.len()),
159        ("saved queries", personal.collections.len()),
160    ] {
161        table.push(vec![name.into(), count.to_string()]);
162    }
163    print!("{}", table.render());
164    println!(
165        "  {}",
166        ui::dim("aede rules --export --output=rules.json makes these decisions portable")
167    );
168}
