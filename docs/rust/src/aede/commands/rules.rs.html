1//! Portable personal corrections and decisions.
2//!
3//! A backup preserves everything. A rules bundle is deliberately narrower:
4//! the small set of human decisions that can be replayed on another catalog
5//! without copying listening history or re-fetched biographies and artwork.
6
7use aede_core::json::Json;
8use aede_core::{sources, user};
9
10use super::{Res, data_dir};
11use crate::args::Args;
12use crate::ui::{self, Table};
13
14const RULES_FORMAT_VERSION: u32 = 1;
15
16/// Lists, exports, or imports reproducible personal rules.
17pub fn rules(args: &Args) -> Res {
18    if args.has("export") && args.has("import") {
19        return Err("--export and --import ask for opposite operations".into());
20    }
21    if args.value("output").is_some() && !args.has("export") {
22        return Err("--output means nothing without --export on rules".into());
23    }
24    let directory = data_dir(args);
25    let user_path = user::user_path(&directory);
26    let sources_path = sources::sources_path(&directory);
27    let mut personal = user::load(&user_path)?.unwrap_or_default();
28    let mut evidence = sources::load_all(&sources_path)?.unwrap_or_default();
29
30    if args.has("export") {
31        return export(args, &personal, &evidence);
32    }
33    if let Some(path) = args.value("import") {
34        let text = std::fs::read_to_string(path)
35            .map_err(|error| format!("cannot read \"{path}\": {error}"))?;
36        let root = aede_core::json::parse(&text)
37            .map_err(|error| format!("\"{path}\" is not JSON: {error}"))?;
38        if root.field_str("format").as_deref() != Some("aede-rules") {
39            return Err("this is not an Aède rules bundle".into());
40        }
41        let version = root.field_u32("format_version").unwrap_or(0);
42        if version != RULES_FORMAT_VERSION {
43            return Err(format!(
44                "rules format version {version} is not supported; expected {RULES_FORMAT_VERSION}"
45            )
46            .into());
47        }
48        let incoming_user = user::from_json(
49            root.get("user")
50                .ok_or_else(|| "rules bundle has no user document".to_string())?,
51        )?;
52        let incoming_sources = sources::from_json(
53            root.get("sources")
54                .ok_or_else(|| "rules bundle has no sources document".to_string())?,
55        )?;
56
57        let report = user::merge(&mut personal, incoming_user);
58        let mut source_changes = 0usize;
59        for record in incoming_sources.records {
60            evidence.set(record);
61            source_changes += 1;
62        }
63        for review in incoming_sources.reviews {
64            evidence.set_review(review);
65            source_changes += 1;
66        }
67        for decision in incoming_sources.credit_exclusions {
68            evidence.exclude_credit(decision.relation, decision.excluded_at);
69            source_changes += 1;
70        }
71        user::save(&personal, &user_path)?;
72        sources::save(&evidence, &sources_path)?;
73        println!(
74            "{} {} added, {} updated, {} kept, {} source decisions imported",
75            ui::green("→"),
76            report.added,
77            report.updated,
78            report.kept,
79            source_changes
80        );
81        return Ok(());
82    }
83
84    print_summary(&personal, &evidence);
85    Ok(())
86}

88fn export(args: &Args, personal: &user::UserData, evidence: &sources::Sources) -> Res {
89    let portable_user = user::UserData {
90        relation_annotations: personal.relation_annotations.clone(),
91        collections: personal.collections.clone(),
92        set_aside: personal.set_aside.clone(),
93        same_artist: personal.same_artist.clone(),
94        ..Default::default()
95    };
96    let portable_sources = sources::Sources {
97        records: evidence
98            .records
99            .iter()
100            .filter(|record| record.source == "manual")
101            .cloned()
102            .collect(),
103        reviews: evidence.reviews.clone(),
104        credit_exclusions: evidence.credit_exclusions.clone(),
105    };
106    let mut root = Json::obj();
107    root.set("format", "aede-rules".into());
108    root.set("format_version", RULES_FORMAT_VERSION.into());
109    root.set("exported_at", aede_core::clock::now_seconds().into());
110    root.set("user", user::to_json(&portable_user));
111    root.set("sources", sources::to_json(&portable_sources));
112    super::export::emit(args, &root.to_string_pretty())
113}

115fn print_summary(personal: &user::UserData, evidence: &sources::Sources) {
116    let accepted = evidence
117        .reviews
118        .iter()
119        .filter(|review| review.decision == sources::ReviewDecision::Accepted)
120        .count();
121    let rejected = evidence.reviews.len().saturating_sub(accepted);
122    let manual = evidence
123        .records
124        .iter()
125        .filter(|record| record.source == "manual")
126        .count();
127    println!("{}", ui::section("Personal rules"));
128    let mut table = Table::new(&["Rule", "Count"]);
129    for (name, count) in [
130        ("accepted source identities", accepted),
131        ("rejected source identities", rejected),
132        ("manual source records", manual),
133        ("artist filing rules", personal.same_artist.len()),
134        ("missing releases set aside", personal.set_aside.len()),
135        ("relation annotations", personal.relation_annotations.len()),
136        ("excluded credits", evidence.credit_exclusions.len()),
137        ("saved queries", personal.collections.len()),
138    ] {
139        table.push(vec![name.into(), count.to_string()]);
140    }
141    print!("{}", table.render());
142    println!(
143        "  {}",
144        ui::dim("aede rules --export --output=rules.json makes these decisions portable")
145    );
146}
