1//! The `release-group` command: the album identity shared by local editions.
2
3use std::collections::BTreeSet;
4
5use aede_core::model::{EDITION_OF, EntityKind, HAS_EDITION};
6use aede_core::text;
7
8use super::{Res, load, navigation::Navigation, totals};
9use crate::args::Args;
10use crate::ui::{self, Align, Table};
11
12pub fn show_release_group(args: &Args) -> Res {
13    let catalog = load(args)?;
14    let query = args.positionals.join(" ");
15    if query.trim().is_empty() {
16        return Err("give a release-group title or MusicBrainz ID".into());
17    }
18    let found = catalog.find_release_groups(&query);
19    if found.is_empty() {
20        return Err(format!("no release group matches \"{query}\"").into());
21    }
22    if found.len() > 1 {
23        let choices = found
24            .iter()
25            .map(|group| format!("\n\t{} · {}", group.title, group.mbid))
26            .collect::<String>();
27        return Err(format!(
28            "{} release groups match \"{query}\"; use the MusicBrainz ID:{choices}",
29            found.len()
30        )
31        .into());
32    }
33    let group = found[0];
34    println!("{}", ui::section(&group.title));
35    println!(
36        "  {}",
37        ui::dim(&format!("MusicBrainz release group: {}", group.mbid))
38    );
39
40    let mut edition_ids = catalog.related_entities(
41        EntityKind::ReleaseGroup,
42        group.id,
43        HAS_EDITION,
44        EntityKind::Release,
45    );
46    edition_ids.sort_unstable();
47    edition_ids.dedup();
48
49    println!("{}", ui::section("Local editions"));
50    let mut table = Table::new(&[
51        "Year", "Edition", "Identity", "Artist", "Tracks", "Duration", "Size", "Format",
52    ])
53    .align(4, Align::Right)
54    .align(5, Align::Right)
55    .align(6, Align::Right)
56    .limit(1, 38)
57    .limit(2, 38)
58    .limit(3, 28)
59    .limit(7, 28);
60    let mut navigation = Navigation::default();
61    let mut artists = BTreeSet::new();
62    for edition_id in edition_ids {
63        let Some(release) = catalog.release(edition_id) else {
64            continue;
65        };
66        // The inverse edge must agree with the edge used to reach this row.
67        if !catalog
68            .related_entities(
69                EntityKind::Release,
70                release.id,
71                EDITION_OF,
72                EntityKind::ReleaseGroup,
73            )
74            .contains(&group.id)
75        {
76            continue;
77        }
78        let (duration, size) = totals(&catalog, &release.track_ids);
79        let formats: BTreeSet<String> = release
80            .track_ids
81            .iter()
82            .filter_map(|&id| catalog.track(id))
83            .filter_map(|track| catalog.file(track.file_id))
84            .map(|file| file.properties.quality_label())
85            .collect();
86        let artist = release
87            .album_artist_id
88            .and_then(|id| catalog.artist(id))
89            .map(|artist| artist.name.clone())
90            .unwrap_or_else(|| "Various Artists".into());
91        if let Some(artist_id) = release.album_artist_id
92            && artists.insert(artist_id)
93        {
94            navigation.entity(&catalog, "Album artist", EntityKind::Artist, artist_id);
95        }
96        navigation.entity(&catalog, "Edition", EntityKind::Release, release.id);
97        table.push(vec![
98            release
99                .year
100                .map(|year| year.to_string())
101                .unwrap_or_else(|| "—".into()),
102            release.title.clone(),
103            release.mbid.clone().unwrap_or_else(|| "local".into()),
104            artist,
105            release.track_ids.len().to_string(),
106            text::format_duration(duration),
107            text::format_size(size),
108            formats.into_iter().collect::<Vec<_>>().join(", "),
109        ]);
110    }
111    print!("{}", table.render());
112    super::panel_for(args, &catalog, EntityKind::ReleaseGroup, group.id);
113    navigation.print();
114    Ok(())
115}
