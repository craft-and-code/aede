1//! `fetch --portraits`: a picture of the artist or the band.
2//!
3//! For the web interface this library exists to feed one day: a shelf of
4//! albums with no face beside any of them is not wrong, but it is not what
5//! anybody actually wants to look at. This pass fills that gap the same way
6//! `--covers` fills the one beside it — by asking a source that already has
7//! the picture, and never by generating or guessing one.
8//!
9//! # Two sources, tried in that order, and never both
10//!
11//! [`wikipedia::portrait_file`] first: the `P18` image claim on the Wikidata
12//! entity MusicBrainz already links every artist to, the same entity
13//! `fetch --summaries` reads for prose. A Commons image carries a stated
14//! licence, which is the property that makes it worth trying first.
15//!
16//! [`fanarttv`] second, and only where Wikidata has nothing: user-submitted,
17//! with no licence Aède can state, so it is the fallback rather than the
18//! first question. It needs a free key — see [`fanarttv::no_key`] — and
19//! where none is set this pass simply does not ask it, and says so once.
20//!
21//! Whichever answers is what gets stored. `Sources::set` replaces a record
22//! whole, and the two services are kept as two source names —
23//! [`wikipedia::PORTRAIT_SOURCE`] and [`fanarttv::SOURCE`] — for exactly the
24//! reason stated there: so that one being asked again never silently erases
25//! what the other already found.
26//!
27//! # What concert photos and line-ups are not part of this
28//!
29//! Deliberately out of scope. Those are not a fact about the artist that a
30//! source states once and Aède reads back — they are a curated choice, and
31//! the interface this feeds will let somebody add their own. Fetching them
32//! automatically would be answering a question nobody asked this pass.
33//!
34//! # Where the picture is written
35//!
36//! Beside the music, when the artist's answer is unambiguous: every album
37//! [`super::shared_folder`] can find for them shares one parent, which is
38//! the same folder `aede playlist --artists` already writes a discography
39//! playlist into. It rides along with the user's own backup exactly as
40//! `cover.jpg` does today — nothing about `aede backup` has to know it
41//! exists.
42//!
43//! Everywhere else — a discography spread across more than one watched root,
44//! an artist known only through compilations, credited on tracks that share
45//! no album folder at all — guessing a destination is not something this
46//! program does anywhere else, and it does not start here. The picture goes
47//! into a per-artist folder under `assets/` in Aède's own data directory
48//! instead, named by the MusicBrainz identifier so two artists that happen
49//! to share a name never collide.
50//!
51//! The record in `sources.json` carries only the address it was downloaded
52//! from — see [`aede_core::sources::Picture`] — because the file on disk is
53//! what a viewer actually looks at, and re-fetching costs one request, not a
54//! reason to duplicate the bytes into the attributed layer too.
55

72/// The width asked of Commons for a downloaded portrait.
73///
74/// Large enough to look right as the face of an artist page, small enough
75/// that a library of a few hundred names is a few hundred megabytes rather
76/// than a few gigabytes — the same reasoning [`super::covers::DEFAULT_SIZE`]
77/// is picked with, at a size suited to a portrait rather than a full sleeve.
78const PORTRAIT_WIDTH: u32 = 800;
79
80/// An artist to ask about, and where the picture would be written.
81struct Target {
82    entity: EntityRef,
83    name: String,
84    /// The Wikidata entity id, when it is still worth asking about: linked by
85    /// MusicBrainz, and — unless `--full` — not already answered.
86    ///
87    /// Decided once, here, rather than re-read inside [`attempt`]: whether a
88    /// source is worth trying is a fact about what `held` already says, and
89    /// asking it twice — once to decide whether this artist is a target at
90    /// all, once to decide whether to actually ask — is how the two would
91    /// eventually disagree.
92    wikidata_id: Option<String>,
93    /// The MusicBrainz artist identifier, only when Fanart.tv is still worth
94    /// asking: a key is set, and — unless `--full` — it has not already
95    /// answered. `None` skips it exactly as an absent [`Target::wikidata_id`]
96    /// skips Wikidata.
97    fanarttv_mbid: Option<String>,
98    /// The MusicBrainz artist identifier, kept on its own beside
99    /// [`Target::fanarttv_mbid`] because it is also what names the artist's
100    /// folder under `assets/`, whether or not Fanart.tv itself is asked.
101    mbid: String,
102    /// The folder the picture is written into — see the module doc for how
103    /// it is chosen.
104    destination: PathBuf,
105}
106
107/// What became of asking about one artist.
108enum Outcome {
109    /// A picture was found and is now on disk.
110    Written {
111        source: &'static str,
112        url: String,
113        /// `false` when a file was already there — written by a previous,
114        /// interrupted run, or dropped in by hand — in which case nothing
115        /// was downloaded twice, only recorded.
116        new: bool,
117    },
118    /// Asked, and nothing was found — everywhere it was still worth asking.
119    Nothing {
120        wikidata_asked: bool,
121        fanarttv_asked: bool,
122    },
123}

125/// The pass.
126pub fn run(
127    catalog: &Catalog,
128    transport: &mut dyn Ask,
129    backoff: &[std::time::Duration],
130    held: &mut sources::Sources,
131    path: &Path,
132    data_dir: &Path,
133    fanarttv_key: Option<&str>,
134    asked: &super::fetch::Asked,
135) -> Res {
136    let targets = targets(
137        catalog,
138        held,
139        asked.names,
140        asked.scope,
141        data_dir,
142        fanarttv_key,
143        asked.again,
144    );
145    println!("{}", ui::section("Portraits"));
146    if targets.is_empty() {
147        super::fetch::print_nothing_to_ask(asked);
148        return Ok(());
149    }
150
151    let least = targets.len() as u64 * wikipedia::REQUEST_INTERVAL.as_millis() as u64;
152    println!(
153        "  {}, one to four requests each, {} to {}",
154        ui::plural(targets.len(), "artist"),
155        ui::long_duration(least),
156        ui::long_duration(least * 4)
157    );
158    println!(
159        "  {}",
160        ui::dim(
161            "wikidata is asked first; fanart.tv only where it has nothing, \
162             and only for an artist already asked",
163        )
164    );
165    if fanarttv_key.is_none() {
166        println!("  {}", ui::dim(&fanarttv::no_key()));
167    }
168    println!(
169        "  {}",
170        ui::dim(
171            "written as artist.jpg or artist.png beside the music when every \
172             album shares a folder, in assets/ otherwise; an artist that \
173             already has one is never touched",
174        )
175    );
176
177    if super::fetch::asked_nothing(
178        asked,
179        &targets.iter().map(|t| t.name.clone()).collect::<Vec<_>>(),
180    ) {
181        return Ok(());
182    }
183
184    let (mut written, mut already, mut none, mut failed) = (0usize, 0usize, 0usize, 0usize);
185    let mut pending = queue(&targets);
186    let mut done = 0usize;
187    let total = targets.len();
188    while let Some((target, retried)) = pending.pop_front() {
189        print!("\r  asking: {}/{}", done + 1, total);
190        let _ = std::io::Write::flush(&mut std::io::stdout());
191
192        match attempt(transport, backoff, target, fanarttv_key) {
193            Ok(outcome) => {
194                store(held, target, &outcome);
195                sources::save(held, path)?;
196                match outcome {
197                    Outcome::Written { new: true, .. } => written += 1,
198                    Outcome::Written { new: false, .. } => already += 1,
199                    Outcome::Nothing { .. } => none += 1,
200                }
201                done += 1;
202            }
203            Err(why) if defer(&mut pending, target, retried, &why) => {
204                continue;
205            }
206            Err(why) => {
207                failed += 1;
208                done += 1;
209                eprintln!("\r  {} {}: {why}", ui::red("×"), target.name);
210            }
211        }
212    }
213    println!();
214
215    println!(
216        "{} {written} written, {none} with no portrait found, {failed} failed",
217        ui::green("→")
218    );
219    if already > 0 {
220        println!(
221            "  {}",
222            ui::dim(&format!(
223                "{} already had a picture on disk, placed by hand or by an \
224                 interrupted run — nothing was downloaded twice",
225                ui::plural(already, "artist")
226            ))
227        );
228    }
229    if written > 0 {
230        println!(
231            "  {}",
232            ui::dim(
233                "aede scan picks up the ones written beside the music — nothing \
234                     registers a file, the scan finds it"
235            )
236        );
237    }
238    Ok(())
239}

241/// Asks about one artist, in order, and writes the picture if one turns up.
242///
243/// A single function rather than a state machine so that a retry — the
244/// `Refusal` bubbling out of any request below, when it is worth deferring —
245/// simply redoes the whole thing on the next pass through the queue, exactly
246/// as `fetch --covers` retries its own first request. One extra ask on the
247/// rare retried artist costs far less than the bookkeeping needed to resume
248/// midway through.
249fn attempt(
250    transport: &mut dyn Ask,
251    backoff: &[std::time::Duration],
252    target: &Target,
253    fanarttv_key: Option<&str>,
254) -> Result<Outcome, Refusal> {
255    let wikidata_asked = target.wikidata_id.is_some();
256    if let Some(id) = &target.wikidata_id {
257        let doc = match ask_with_backoff(transport, &wikipedia::entity_data_url(id), backoff) {
258            Ok(doc) => Some(doc),
259            // A entity id MusicBrainz linked to should exist; if Wikidata
260            // says otherwise, that is the same "asked, and there is
261            // nothing" answer as a claim with no image in it.
262            Err(Refusal::Missing) => None,
263            Err(why) => return Err(why),
264        };
265        if let Some(file) = doc
266            .as_ref()
267            .and_then(|doc| wikipedia::portrait_file(doc, id))
268        {
269            let url = wikipedia::commons_file_url(&file, PORTRAIT_WIDTH);
270            let bytes = ask_bytes(transport, &url, backoff)?;
271            let new = write(target, &bytes)?;
272            return Ok(Outcome::Written {
273                source: wikipedia::PORTRAIT_SOURCE,
274                url,
275                new,
276            });
277        }
278    }
279
280    let fanarttv_asked = target.fanarttv_mbid.is_some();
281    if let Some(mbid) = &target.fanarttv_mbid {
282        // `fanarttv_mbid` is only ever set when [`targets`] already found a
283        // key, so this is never reached with none — but the URL still needs
284        // the key itself, not just the fact that one exists.
285        let key = fanarttv_key.expect("a key, since fanarttv_mbid was set from one");
286        let doc = match ask_with_backoff(transport, &fanarttv::lookup_url(mbid, key), backoff) {
287            // The service answers `404` for an artist it holds nothing of,
288            // which is a real answer, not a failure — the same distinction
289            // `fetch --covers` reads out of the same status for the archive.
290            Ok(doc) => Some(doc),
291            Err(Refusal::Missing) => None,
292            Err(why) => return Err(why),
293        };
294        if let Some(url) = doc.as_ref().and_then(fanarttv::portrait_url) {
295            let bytes = ask_bytes(transport, &url, backoff)?;
296            let new = write(target, &bytes)?;
297            return Ok(Outcome::Written {
298                source: fanarttv::SOURCE,
299                url,
300                new,
301            });
302        }
303    }
304
305    Ok(Outcome::Nothing {
306        wikidata_asked,
307        fanarttv_asked,
308    })
309}

311/// Writes the picture into the target's folder, unless one is already there.
312///
313/// The general-purpose writer this program keeps for every image it puts
314/// into somebody's library — see [`coverart::write_image`] — carrying both of
315/// its guards: the bytes are sniffed, and nothing already there is ever
316/// overwritten. `true` means this call is the one that wrote it.
317fn write(target: &Target, bytes: &[u8]) -> Result<bool, Refusal> {
318    match coverart::write_image(&target.destination, Kind::Artist, (0, 1), bytes) {
319        Ok(coverart::Written::New(_)) => Ok(true),
320        Ok(coverart::Written::Already(_)) => Ok(false),
321        Err(why) => Err(Refusal::Failed(why)),
322    }
323}

325/// Files what was found, including when that was nothing.
326///
327/// A record with no picture is not wasted, for the same reason `--summaries`
328/// keeps one with an empty prose: "asked, and there is nothing" is what
329/// keeps the next run from asking again. Only the sources actually asked get
330/// a record — Fanart.tv is never marked "asked" for an artist Wikidata
331/// already answered, because it never was.
332fn store(held: &mut sources::Sources, target: &Target, outcome: &Outcome) {
333    let mut file = |source: &str, portrait: Option<Picture>| {
334        held.set(SourceRecord {
335            key: target.entity.key.clone(),
336            source: source.to_string(),
337            source_id: Some(target.mbid.clone()),
338            fetched_at: clock::now_seconds(),
339            // Both sources are reached by identifier — the Wikidata entity
340            // MusicBrainz links to, the artist's own MusicBrainz id — never
341            // by matching a name, so nothing stored here is ever a guess.
342            confidence: sources::Confidence::Identified,
343            facts: Facts::Artist(sources::ArtistFacts {
344                portrait,
345                ..Default::default()
346            }),
347        });
348    };
349    match outcome {
350        Outcome::Written { source, url, .. } => file(source, Some(Picture { url: url.clone() })),
351        Outcome::Nothing {
352            wikidata_asked,
353            fanarttv_asked,
354        } => {
355            if *wikidata_asked {
356                file(wikipedia::PORTRAIT_SOURCE, None);
357            }
358            if *fanarttv_asked {
359                file(fanarttv::SOURCE, None);
360            }
361        }
362    }
363}

365/// How many artists this pass would ask about, if it ran now.
366///
367/// The same function as the walk, counted rather than re-derived — see
368/// [`super::summaries::waiting`] for why that discipline matters here too.
369/// Takes the Fanart.tv key for the same reason `run` does: an artist only
370/// Fanart.tv could still answer for is not "waiting" while no key is set.
371pub fn waiting(
372    catalog: &Catalog,
373    held: &sources::Sources,
374    data_dir: &Path,
375    fanarttv_key: Option<&str>,
376) -> usize {
377    targets(
378        catalog,
379        held,
380        &[],
381        &super::fetch::EVERYTHING,
382        data_dir,
383        fanarttv_key,
384        false,
385    )
386    .len()
387}

389/// Who to ask about: artists MusicBrainz answered, that have not already got
390/// a picture and have not already been asked everywhere they could be.
391fn targets(
392    catalog: &Catalog,
393    held: &sources::Sources,
394    wanted: &[String],
395    scope: &super::fetch::Scope,
396    data_dir: &Path,
397    fanarttv_key: Option<&str>,
398    again: bool,
399) -> Vec<Target> {
400    let releases_by_artist = super::releases_by_album_artist(catalog);
401
402    let mut targets = Vec::new();
403    for (record, artist_row, artist, mbid) in
404        super::fetch::artist_image_candidates(catalog, held, wanted, scope)
405    {
406        let entity = record.entity();
407        let linked = artist.wikidata.as_deref().and_then(wikipedia::entity_id);
408
409        if !again && has_portrait(held, &entity) {
410            continue;
411        }
412        // `again` reruns a source that has already answered; without it, a
413        // source is worth trying only when it has not been asked yet — and
414        // not at all when there is nothing to ask it with, a missing
415        // Fanart.tv key above all, which is never a reason to keep retrying.
416        let wikidata_id =
417            linked.filter(|_| again || held.get(&entity, wikipedia::PORTRAIT_SOURCE).is_none());
418        let fanarttv_mbid = fanarttv_key
419            .filter(|_| again || held.get(&entity, fanarttv::SOURCE).is_none())
420            .map(|_| mbid.to_string());
421        if wikidata_id.is_none() && fanarttv_mbid.is_none() {
422            continue;
423        }
424
425        let destination = super::artist_artwork_destination(
426            catalog,
427            releases_by_artist
428                .get(&artist_row.id)
429                .map(Vec::as_slice)
430                .unwrap_or_default(),
431            data_dir,
432            mbid,
433        );
434
435        targets.push(Target {
436            entity,
437            name: artist_row.name.clone(),
438            wikidata_id,
439            fanarttv_mbid,
440            mbid: mbid.to_string(),
441            destination,
442        });
443    }
444    targets
445}

447/// `true` when a picture — not just an answer — is already on record.
448fn has_portrait(held: &sources::Sources, entity: &EntityRef) -> bool {
449    [wikipedia::PORTRAIT_SOURCE, fanarttv::SOURCE]
450        .iter()
451        .any(|source| match held.get(entity, source) {
452            Some(SourceRecord {
453                facts: Facts::Artist(artist),
454                ..
455            }) => artist.portrait.is_some(),
456            _ => false,
457        })
458}
