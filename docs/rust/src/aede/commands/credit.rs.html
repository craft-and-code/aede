15const MANUAL: &str = "manual";
16
17enum Scope {
18    Recording(Id),
19    Work {
20        recording_id: Id,
21        mbid: String,
22        title: String,
23    },
24    Edition(Id),
25}
26
27pub fn credit(args: &Args) -> Res {
28    let adding = args.has("add");
29    let excluding = args.value("exclude");
30    let undoing = args.value("undo");
31    if usize::from(adding) + usize::from(excluding.is_some()) + usize::from(undoing.is_some()) != 1
32    {
33        return Err(
34            "choose one: --add <scope>, --exclude=<credit ID>, or --undo=<credit ID>".into(),
35        );
36    }
37    let catalog = load(args)?;
38    let mut held = sources::load_all(&sources::sources_path(&data_dir(args)))?.unwrap_or_default();
39    let path = sources::sources_path(&data_dir(args));
40    if let Some(selector) = excluding.or(undoing) {
41        if !args.positionals.is_empty()
42            || args.has("artist")
43            || args.has("artist-id")
44            || args.has("role")
45            || args.has("instrument")
46        {
47            return Err("credit decisions take an ID, not a scope or new credit fields".into());
48        }
49        let edge = graph::select(&graph::edges(&catalog, &held), selector)?;
50        if !edge.reference.kind.starts_with("credit:") || edge.reference.provenance == "tags" {
51            return Err(
52                "only a sourced credit can be excluded; tags remain the local record".into(),
53            );
54        }
55        if excluding.is_some() {
56            held.exclude_credit_in(&catalog, edge.reference, clock::now_seconds());
57            sources::save(&held, &path)?;
58            println!(
59                "{} credit excluded from navigation and search; source evidence kept",
60                ui::green("→")
61            );
62        } else if held.restore_credit_in(&catalog, &edge.reference) {
63            sources::save(&held, &path)?;
64            println!("{} credit restored", ui::green("→"));
65        } else {
66            return Err("this credit has no exclusion to undo".into());
67        }
68        return Ok(());
69    }
70
71    let Some(token) = args.positionals.first() else {
72        return Err("give a precise scope: recording:<ID>, work:<ID>, or release:<ID>".into());
73    };
74    if args.positionals.len() != 1 {
75        return Err("give exactly one credit scope".into());
76    }
77    let artist_name = args
78        .value("artist")
79        .map(str::trim)
80        .filter(|name| !name.is_empty())
81        .ok_or_else(|| "--artist names the person to credit".to_string())?;
82    let role = args
83        .value("role")
84        .map(str::trim)
85        .filter(|role| !role.is_empty())
86        .ok_or_else(|| "--role names what they did".to_string())?
87        .to_ascii_lowercase()
88        .replace([' ', '-'], "_");
89    let artist_mbid = args.value("artist-id").unwrap_or("").trim();
90    let instrument = args
91        .value("instrument")
92        .map(str::trim)
93        .filter(|value| !value.is_empty());
94    let scope = resolve_scope(&catalog, &held, token)?;
95    let (target, entity) = match &scope {
96        Scope::Recording(id) => (
97            EntityRef::of(&catalog, EntityKind::Recording, *id)
98                .ok_or("recording has no stable reference")?,
99            representative_track(&catalog, *id)?,
100        ),
101        Scope::Work {
102            recording_id, mbid, ..
103        } => (
104            EntityRef::new(EntityKind::Work, mbid.clone()),
105            representative_track(&catalog, *recording_id)?,
106        ),
107        Scope::Edition(id) => (
108            EntityRef::of(&catalog, EntityKind::Release, *id)
109                .ok_or("edition has no stable reference")?,
110            EntityRef::of(&catalog, EntityKind::Release, *id)
111                .ok_or("edition has no stable reference")?,
112        ),
113    };
114    let seed = RelationRef {
115        source: EntityRef::new(
116            EntityKind::Artist,
117            if artist_mbid.is_empty() {
118                format!("manual:{}", text::normalize(artist_name))
119            } else {
120                format!("mbid:{artist_mbid}")
121            },
122        ),
123        kind: format!("credit:{role}"),
124        target,
125        provenance: MANUAL.into(),
126        source_id: instrument.map(text::normalize),
127    };
128    let relation_id = format!("manual:{}", seed.id());
129    let credit = CreditLink {
130        relation_id: Some(relation_id.clone()),
131        role_id: None,
132        role,
133        direction: None,
134        artist_mbid: artist_mbid.to_string(),
135        artist_name: artist_name.to_string(),
136        credited_as: None,
137        attributes: instrument
138            .map(|name| aede_core::model::CreditAttribute {
139                id: None,
140                name: name.to_string(),
141                value: None,
142                credited_as: None,
143            })
144            .into_iter()
145            .collect(),
146        began: None,
147        ended: None,
148        over: None,
149        order: None,
150    };
151    let existing = held.get(&entity, MANUAL).cloned();
152    let mut facts = existing
153        .as_ref()
154        .map(|record| record.facts.clone())
155        .unwrap_or_else(|| match scope {
156            Scope::Edition(_) => Facts::Release(ReleaseFacts::default()),
157            _ => Facts::Track(TrackFacts::default()),
158        });
159    let credits = match (&scope, &mut facts) {
160        (Scope::Recording(_), Facts::Track(facts)) => &mut facts.credits,
161        (Scope::Work { mbid, title, .. }, Facts::Track(facts)) => {
162            let index = match facts.works.iter().position(|work| work.mbid == *mbid) {
163                Some(index) => index,
164                None => {
165                    facts.works.push(WorkLink {
166                        mbid: mbid.clone(),
167                        title: title.clone(),
168                        ..Default::default()
169                    });
170                    facts.works.len() - 1
171                }
172            };
173            &mut facts.works[index].credits
174        }
175        (Scope::Edition(id), Facts::Release(facts)) => {
176            facts.edition_mbid = catalog
177                .release(*id)
178                .and_then(|release| release.mbid.clone());
179            &mut facts.credits
180        }
181        _ => {
182            return Err(
183                "manual source record has a different scope; refusing to overwrite it".into(),
184            );
185        }
186    };
187    if credits
188        .iter()
189        .any(|held| held.relation_id.as_deref() == Some(&relation_id))
190    {
191        return Err("this exact manual credit is already present".into());
192    }
193    credits.push(credit);
194    held.set(SourceRecord {
195        key: entity.key,
196        source: MANUAL.into(),
197        source_id: existing.and_then(|record| record.source_id),
198        fetched_at: clock::now_seconds(),
199        confidence: Confidence::Identified,
200        facts,
201    });
202    sources::save(&held, &path)?;
203    let id = graph::edges(&catalog, &held)
204        .into_iter()
205        .find(|edge| {
206            edge.reference.provenance == MANUAL
207                && edge.reference.source_id.as_deref() == Some(&relation_id)
208        })
209        .map(|edge| edge.reference.id())
210        .unwrap_or(relation_id);
211    println!("{} manual credit saved: {id}", ui::green("→"));
212    Ok(())
213}

215fn representative_track(catalog: &Catalog, recording_id: Id) -> Result<EntityRef, String> {
216    catalog
217        .recording(recording_id)
218        .ok_or_else(|| "recording is not in the local catalog".to_string())?
219        .track_ids
220        .iter()
221        .filter_map(|&id| EntityRef::of(catalog, EntityKind::Track, id))
222        .min_by(|left, right| left.key.cmp(&right.key))
223        .ok_or_else(|| "recording has no local file".to_string())
224}

226fn resolve_scope(
227    catalog: &Catalog,
228    sources: &sources::Sources,
229    token: &str,
230) -> Result<Scope, String> {
231    let (kind, key) = token
232        .split_once(':')
233        .ok_or_else(|| "scope must begin recording:, work:, or release:".to_string())?;
234    match kind {
235        "recording" => EntityRef::new(EntityKind::Recording, key)
236            .resolve(catalog)
237            .map(Scope::Recording)
238            .ok_or_else(|| format!("no local recording has identity {key}")),
239        "release" => {
240            let matching: Vec<_> = catalog
241                .releases
242                .iter()
243                .filter(|release| release.mbid.as_deref() == Some(key))
244                .collect();
245            if matching.len() > 1 {
246                return Err(format!(
247                    "{key} identifies several local editions; use release:<local release key>"
248                ));
249            }
250            matching
251                .first()
252                .map(|release| release.id)
253                .or_else(|| EntityRef::new(EntityKind::Release, key).resolve(catalog))
254                .map(Scope::Edition)
255                .ok_or_else(|| format!("no local edition has identity {key}"))
256        }
257        "work" => {
258            let local = catalog
259                .works
260                .iter()
261                .find(|work| work.mbid == key)
262                .and_then(|work| {
263                    work.recording_ids
264                        .first()
265                        .copied()
266                        .map(|id| (id, work.title.clone()))
267                });
268            let sourced = sources
269                .find_sourced_works(catalog, key)
270                .into_iter()
271                .find(|work| work.mbid == key)
272                .and_then(|work| {
273                    work.links
274                        .first()
275                        .map(|link| (link.recording_id, work.title))
276                });
277            local
278                .or(sourced)
279                .map(|(recording_id, title)| Scope::Work {
280                    recording_id,
281                    mbid: key.into(),
282                    title,
283                })
284                .ok_or_else(|| format!("no local recording is linked to work {key}"))
285        }
286        _ => Err("scope must begin recording:, work:, or release:".into()),
287    }
