15const MANUAL: &str = "manual";
16
17enum Scope {
18    Recording(Id),
19    Work {
20        recording_id: Id,
21        mbid: String,
22        title: String,
23    },
24    Edition(Id),
25}

27pub fn credit(args: &Args) -> Res {
28    let adding = args.has("add");
29    let excluding = args.value("exclude");
30    let undoing = args.value("undo");
31    if usize::from(adding) + usize::from(excluding.is_some()) + usize::from(undoing.is_some()) != 1
32    {
33        return Err(
34            "choose one: --add <scope>, --exclude=<credit ID>, or --undo=<credit ID>".into(),
35        );
36    }
37    let catalog = load(args)?;
38    let mut held = sources::load_all(&sources::sources_path(&data_dir(args)))?.unwrap_or_default();
39    let path = sources::sources_path(&data_dir(args));
40    if let Some(selector) = excluding.or(undoing) {
41        if !args.positionals.is_empty()
42            || args.has("artist")
43            || args.has("artist-id")
44            || args.has("role")
45            || args.has("instrument")
46        {
47            return Err("credit decisions take an ID, not a scope or new credit fields".into());
48        }
49        let edge = graph::select(&graph::edges(&catalog, &held), selector)?;
50        if !edge.reference.kind.starts_with("credit:") || edge.reference.provenance == "tags" {
51            return Err(
52                "only a sourced credit can be excluded; tags remain the local record".into(),
53            );
54        }
55        if excluding.is_some() {
56            held.exclude_credit_in(&catalog, edge.reference, clock::now_seconds());
57            sources::save(&held, &path)?;
58            println!(
59                "{} credit excluded from navigation and search; source evidence kept",
60                ui::green("→")
61            );
62        } else if held.restore_credit_in(&catalog, &edge.reference) {
63            sources::save(&held, &path)?;
64            println!("{} credit restored", ui::green("→"));
65        } else {
66            return Err("this credit has no exclusion to undo".into());
67        }
68        return Ok(());
69    }
70
71    let Some(token) = args.positionals.first() else {
72        return Err("give a precise scope: recording:<ID>, work:<ID>, or release:<ID>".into());
73    };
74    if args.positionals.len() != 1 {
75        return Err("give exactly one credit scope".into());
76    }
77    let artist_name = args
78        .value("artist")
79        .map(str::trim)
80        .filter(|name| !name.is_empty())
81        .ok_or_else(|| "--artist names the person to credit".to_string())?;
82    let role = args
83        .value("role")
84        .map(str::trim)
85        .filter(|role| !role.is_empty())
86        .ok_or_else(|| "--role names what they did".to_string())?
87        .to_ascii_lowercase()
88        .replace([' ', '-'], "_");
89    let artist_mbid = args.value("artist-id").unwrap_or("").trim();
90    let instrument = args
91        .value("instrument")
92        .map(str::trim)
93        .filter(|value| !value.is_empty());
94    let scope = resolve_scope(&catalog, &held, token)?;
95    let (target, entity) = match &scope {
96        Scope::Recording(id) => (
97            EntityRef::of(&catalog, EntityKind::Recording, *id)
98                .ok_or("recording has no stable reference")?,
99            representative_track(&catalog, *id)?,
100        ),
101        Scope::Work {
102            recording_id, mbid, ..
103        } => (
104            EntityRef::new(EntityKind::Work, mbid.clone()),
105            representative_track(&catalog, *recording_id)?,
106        ),
107        Scope::Edition(id) => (
108            EntityRef::of(&catalog, EntityKind::Release, *id)
109                .ok_or("edition has no stable reference")?,
110            EntityRef::of(&catalog, EntityKind::Release, *id)
111                .ok_or("edition has no stable reference")?,
112        ),
113    };
114    let seed = RelationRef {
115        source: EntityRef::new(
116            EntityKind::Artist,
117            if artist_mbid.is_empty() {
118                format!("manual:{}", text::normalize(artist_name))
119            } else {
120                format!("mbid:{artist_mbid}")
121            },
122        ),
123        kind: format!("credit:{role}"),
124        target,
125        provenance: MANUAL.into(),
126        source_id: instrument.map(text::normalize),
127    };
128    let relation_id = format!("manual:{}", seed.id());
129    let credit = CreditLink {
130        relation_id: Some(relation_id.clone()),
131        role_id: None,
132        role,
133        direction: None,
134        artist_mbid: artist_mbid.to_string(),
135        artist_name: artist_name.to_string(),
136        credited_as: None,
137        attributes: instrument
138            .map(|name| aede_core::model::CreditAttribute {
139                id: None,
140                name: name.to_string(),
141                value: None,
142                credited_as: None,
143            })
144            .into_iter()
145            .collect(),
146        began: None,
147        ended: None,
148        over: None,
149        order: None,
150    };
151    let existing = held.get(&entity, MANUAL).cloned();
152    let mut facts = existing
153        .as_ref()
154        .map(|record| record.facts.clone())
155        .unwrap_or_else(|| match scope {
156            Scope::Edition(_) => Facts::Release(ReleaseFacts::default()),
157            _ => Facts::Track(TrackFacts::default()),
158        });
159    let credits = match (&scope, &mut facts) {
160        (Scope::Recording(id), Facts::Track(facts)) => {
161            let recording_mbid = catalog
162                .recording(*id)
163                .and_then(|recording| recording.mbid.clone());
164            // One observation can also carry work credits, whose scope is
165            // independent. Only its direct credits capture this performance.
166            if facts
167                .recording
168                .as_ref()
169                .is_some_and(|held| recording_mbid.as_ref() != Some(held))
170            {
171                return Err(
172                    "manual credits belong to a different recording; refusing to retarget them"
173                        .into(),
174                );
175            }
176            facts.recording = recording_mbid;
177            &mut facts.credits
178        }
179        (Scope::Work { mbid, title, .. }, Facts::Track(facts)) => {
180            let index = match facts.works.iter().position(|work| work.mbid == *mbid) {
181                Some(index) => index,
182                None => {
183                    facts.works.push(WorkLink {
184                        mbid: mbid.clone(),
185                        title: title.clone(),
186                        ..Default::default()
187                    });
188                    facts.works.len() - 1
189                }
190            };
191            &mut facts.works[index].credits
192        }
193        (Scope::Edition(id), Facts::Release(facts)) => {
194            let edition_mbid = catalog
195                .release(*id)
196                .and_then(|release| release.mbid.clone());
197            // This snapshot also carries earlier credits. A new assertion
198            // must not move those retained claims onto a different edition.
199            if facts
200                .edition_mbid
201                .as_ref()
202                .is_some_and(|held| edition_mbid.as_ref() != Some(held))
203            {
204                return Err(
205                    "manual credits belong to a different edition; refusing to retarget them"
206                        .into(),
207                );
208            }
209            facts.edition_mbid = edition_mbid;
210            &mut facts.credits
211        }
212        _ => {
213            return Err(
214                "manual source record has a different scope; refusing to overwrite it".into(),
215            );
216        }
217    };
218    if credits
219        .iter()
220        .any(|held| held.relation_id.as_deref() == Some(&relation_id))
221        || already_attached(&catalog, &held, &scope, &seed, &relation_id)
222    {
223        return Err("this exact manual credit is already present".into());
224    }
225    credits.push(credit);
226    held.set(SourceRecord {
227        key: entity.key,
228        source: MANUAL.into(),
229        source_id: existing.and_then(|record| record.source_id),
230        fetched_at: clock::now_seconds(),
231        confidence: Confidence::Identified,
232        facts,
233    });
234    sources::save(&held, &path)?;
235    let id = graph::edges(&catalog, &held)
236        .into_iter()
237        .find(|edge| {
238            edge.reference.provenance == MANUAL
239                && edge.reference.source_id.as_deref() == Some(&relation_id)
240        })
241        .map(|edge| edge.reference.id())
242        .unwrap_or(relation_id);
243    println!("{} manual credit saved: {id}", ui::green("→"));
244    Ok(())
245}

247/// The observation host can change without changing the user's assertion.
248/// Existing projections keep scope/attachment separate from trust/exclusion.
249fn already_attached(
250    catalog: &Catalog,
251    held: &sources::Sources,
252    scope: &Scope,
253    seed: &RelationRef,
254    id: &str,
255) -> bool {
256    // Identifying a previously local recording changes its seed. Retain only
257    // the deterministic IDs of its still-current file placements as aliases.
258    let mut local_ids = if let Scope::Recording(recording_id) = scope
259        && let Some(recording) = catalog
260            .recording(*recording_id)
261            .filter(|recording| recording.mbid.is_some())
262    {
263        let mut local_seed = seed.clone();
264        recording
265            .track_ids
266            .iter()
267            .filter_map(|id| {
268                catalog
269                    .track(*id)
270                    .and_then(|track| catalog.file(track.file_id))
271            })
272            .map(|file| {
273                local_seed.target.key = format!("local:{}", file.path);
274                format!("manual:{}", local_seed.id())
275            })
276            .collect::<Vec<_>>()
277    } else {
278        Vec::new()
279    };
280    local_ids.sort_unstable();
281    local_ids.dedup();
282    let same_id = |credit: &CreditLink| {
283        credit.relation_id.as_deref().is_some_and(|held| {
284            held == id
285                || local_ids
286                    .binary_search_by(|local| local.as_str().cmp(held))
287                    .is_ok()
288        })
289    };
290    // Most additions have a new ID. Avoid allocating projections unless a
291    // retained manual assertion could already supply this exact credit.
292    if !held.records.iter().any(|record| {
293        record.source == MANUAL
294            && match (scope, &record.facts) {
295                (Scope::Recording(_), Facts::Track(facts)) => facts.credits.iter().any(same_id),
296                (Scope::Work { mbid, .. }, Facts::Track(facts)) => facts
297                    .works
298                    .iter()
299                    .filter(|work| work.mbid == *mbid)
300                    .any(|work| work.credits.iter().any(same_id)),
301                _ => false,
302            }
303    }) {
304        return false;
305    }
306    held.credit_links(catalog).iter().any(|link| {
307        link.source == MANUAL
308            && same_id(&link.credit)
309            && match scope {
310                Scope::Recording(recording_id) => {
311                    link.work.is_none() && link.recording_id == *recording_id
312                }
313                Scope::Work { mbid, .. } => {
314                    link.work.as_ref().is_some_and(|work| work.mbid == *mbid)
315                }
316                // Editions retain their existing exact local-key guard.
317                Scope::Edition(_) => false,
318            }
319    })
320}

322fn representative_track(catalog: &Catalog, recording_id: Id) -> Result<EntityRef, String> {
323    catalog
324        .recording(recording_id)
325        .ok_or_else(|| "recording is not in the local catalog".to_string())?
326        .track_ids
327        .iter()
328        .filter_map(|&id| EntityRef::of(catalog, EntityKind::Track, id))
329        .min_by(|left, right| left.key.cmp(&right.key))
330        .ok_or_else(|| "recording has no local file".to_string())
331}

333fn resolve_scope(
334    catalog: &Catalog,
335    sources: &sources::Sources,
336    token: &str,
337) -> Result<Scope, String> {
338    let (kind, key) = token
339        .split_once(':')
340        .ok_or_else(|| "scope must begin recording:, work:, or release:".to_string())?;
341    match kind {
342        "recording" => EntityRef::new(EntityKind::Recording, key)
343            .resolve(catalog)
344            .map(Scope::Recording)
345            .ok_or_else(|| format!("no local recording has identity {key}")),
346        "release" => {
347            let matching: Vec<_> = catalog
348                .releases
349                .iter()
350                .filter(|release| release.mbid.as_deref() == Some(key))
351                .collect();
352            if matching.len() > 1 {
353                return Err(format!(
354                    "{key} identifies several local editions; use release:<local release key>"
355                ));
356            }
357            matching
358                .first()
359                .map(|release| release.id)
360                .or_else(|| EntityRef::new(EntityKind::Release, key).resolve(catalog))
361                .map(Scope::Edition)
362                .ok_or_else(|| format!("no local edition has identity {key}"))
363        }
364        "work" => {
365            let local = catalog
366                .works
367                .iter()
368                .find(|work| work.mbid == key)
369                .and_then(|work| {
370                    work.recording_ids
371                        .first()
372                        .copied()
373                        .map(|id| (id, work.title.clone()))
374                });
375            let sourced = sources
376                .find_sourced_works(catalog, key)
377                .into_iter()
378                .find(|work| work.mbid == key)
379                .and_then(|work| {
380                    work.links
381                        .first()
382                        .map(|link| (link.recording_id, work.title))
383                });
384            local
385                .or(sourced)
386                .map(|(recording_id, title)| Scope::Work {
387                    recording_id,
388                    mbid: key.into(),
389                    title,
390                })
391                .ok_or_else(|| format!("no local recording is linked to work {key}"))
392        }
393        _ => Err("scope must begin recording:, work:, or release:".into()),
394    }
