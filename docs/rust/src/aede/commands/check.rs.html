1//! The `check` command: verify that the files are still what they were.
2//!
3//! Deliberately opt-in. A verdict needs all supported checksums across the
4//! selected streams, within the verifier's read budget. Storage throughput and
5//! file sizes determine the cost; a metadata `scan` does not perform this work
6//! on its own.
7//!
8//! The verdict is stored per file and survives across scans, so the cost is
9//! paid once: only files that are new, or that changed, come back without one.
10//! It is also saved as the work goes, so an interrupted run keeps what it
11//! established and the next one carries on where it stopped.
12

28/// Above this many files, the run is worth announcing before it starts.
29const LONG_RUN: usize = 500;
30
31/// Files verified between two saves.
32///
33/// Small enough that an interruption costs little, large enough that writing
34/// the catalog stays a rounding error next to reading the audio.
35const SAVE_EVERY: usize = 250;
36
37pub fn check(args: &Args) -> Res {
38    let dir = data_dir(args);
39    let catalog_file = store::catalog_path(&dir);
40    let mut catalog = load(args)?;
41
42    // A folder given on the command line restricts the work to it. Verifying a
43    // whole library at once is the kind of thing one wants to try on a corner
44    // first.
45    let scope = super::scope_of(args, &catalog)?;
46    let queue = to_verify(&catalog, &scope, args.has("full"));
47
48    // Nothing to read is not nothing to say. The command answers "are my files
49    // intact?", and it used to withhold that answer precisely when the work was
50    // already done — leaving the user with "every file already has a verdict"
51    // and no way to learn which verdict.
52    if queue.is_empty() {
53        report(&catalog, &scope, 0, None, &[]);
54        return Ok(());
55    }
56
57    announce(&catalog, &queue);
58
59    let started = Instant::now();
60    let total = queue.len();
61    let done = AtomicUsize::new(0);
62    let mut failures: Vec<(String, String)> = Vec::new();
63    let now = now_seconds();
64    let threads = resolve_threads(args.number_or("threads", 0)?);
65    let interactive = ui::is_interactive();
66
67    // One batch at a time, each saved before the next starts: a run stopped
68    // half-way keeps everything it verified.
69    for batch in queue.chunks(SAVE_EVERY) {
70        let pending = Mutex::new(batch.to_vec());
71        let results: Mutex<Vec<(Id, IntegrityRecord)>> = Mutex::new(Vec::new());
72        let batch_failures: Mutex<Vec<(Id, String, String)>> = Mutex::new(Vec::new());
73
74        std::thread::scope(|scope| {
75            for _ in 0..threads.min(batch.len()) {
76                scope.spawn(|| {
77                    while let Some((id, path)) = super::take_queued(&pending) {
78                        match integrity::check(&path) {
79                            Ok(report) => {
80                                let record = IntegrityRecord {
81                                    verdict: report.verdict,
82                                    method: report.method.to_string(),
83                                    checked_at: now,
84                                };
85                                results
86                                    .lock()
87                                    .unwrap_or_else(|e| e.into_inner())
88                                    .push((id, record));
89                            }
90                            // Being unreadable is not a verdict on the audio:
91                            // the file keeps no verdict and is reported apart.
92                            Err(error) => batch_failures
93                                .lock()
94                                .unwrap_or_else(|e| e.into_inner())
95                                .push((id, path.to_string_lossy().to_string(), error.to_string())),
96                        }
97                        // Redrawn every few files: often enough to look alive,
98                        // rarely enough not to interleave between threads.
99                        let count = done.fetch_add(1, Ordering::Relaxed) + 1;
100                        if interactive && (count.is_multiple_of(4) || count == total) {
101                            print!("\r  {count}/{total}   ");
102                            use std::io::Write;
103                            let _ = std::io::stdout().flush();
104                        }
105                    }
106                });
107            }
108        });
109
110        for (id, record) in results.into_inner().unwrap_or_else(|e| e.into_inner()) {
111            if let Some(file) = catalog.files.get_mut(id as usize) {
112                file.integrity = Some(record);
113            }
114        }
115        for (id, path, error) in batch_failures
116            .into_inner()
117            .unwrap_or_else(|e| e.into_inner())
118        {
119            // A failed explicit recheck cannot leave a reusable old verdict:
120            // the report would still count it as intact and the next ordinary
121            // run would skip it. Keep other conclusions and unattempted files.
122            if let Some(file) = catalog.files.get_mut(id as usize) {
123                file.integrity = None;
124            }
125            failures.push((path, error));
126        }
127        store::save(&catalog, &catalog_file)?;
128    }
129    if interactive {
130        println!("\r  {total}/{total}   ");
131    }
132
133    report(
134        &catalog,
135        &scope,
136        total,
137        Some(started.elapsed().as_millis()),
138        &failures,
139    );
140    Ok(())
141}

143fn to_verify(catalog: &Catalog, scope: &[String], full: bool) -> Vec<(Id, PathBuf)> {
144    catalog
145        .files
146        .iter()
147        .filter(|file| super::in_scope(&file.path, scope))
148        // By default only what has no verdict yet, which is what makes a second
149        // run cheap. `--full` re-reads everything, for a disk under suspicion.
150        .filter(|file| full || file.integrity.is_none())
151        .map(|file| (file.id, PathBuf::from(&file.path)))
152        .collect()
153}

155/// Says what is about to happen, in volume rather than in a predicted time: how
156/// long it takes depends on the disk, and a wrong estimate is worse than none.
157fn announce(catalog: &Catalog, queue: &[(Id, PathBuf)]) {
158    let bytes: u64 = queue
159        .iter()
160        .filter_map(|(id, _)| catalog.file(*id))
161        .map(|f| f.size)
162        .sum();
163    println!(
164        "{} {}",
165        ui::bold("Verifying"),
166        ui::dim(&format!(
167            "{} to read, {}",
168            ui::plural(queue.len(), "file"),
169            text::format_size(bytes)
170        ))
171    );
172    if queue.len() >= LONG_RUN {
173        println!(
174            "  {}",
175            ui::yellow("this reads every byte: minutes on an SSD, longer on a mechanical disk")
176        );
177        println!("  {}", ui::dim(&interruption_hint()));
178    }
179}

181/// What an interrupted run actually keeps.
182///
183/// A batch is only committed to the catalog once every file in it is done
184/// (see the loop in [`check`]), so a `Ctrl-C` mid-batch loses that whole
185/// batch, not just the file being read at that instant. Saying "saved as
186/// the run goes" without naming the batch size overstates that guarantee —
187/// which is exactly what decides whether someone lets the interrupt through.
188fn interruption_hint() -> String {
189    format!(
190        "stopping it is safe — verdicts are saved every {SAVE_EVERY} files, so at most the batch in progress is lost"
191    )
192}

194/// The state of the library, and what this run did to reach it.
195///
196/// The two are separate on purpose. The table describes **every file in
197/// scope**, whatever run established each verdict; the line under it describes
198/// **this run**. Reading "137 files to read" and then "1304 intact" as one
199/// figure is the confusion that follows from mixing them, and the shape does
200/// not change when there was nothing to read: a command that answers in a
201/// different form depending on the result is a command you cannot learn.
202fn report(
203    catalog: &Catalog,
204    scope: &[String],
205    read: usize,
206    elapsed_ms: Option<u128>,
207    failures: &[(String, String)],
208) {
209    let (mut intact, mut damaged, mut nothing, mut unchecked) = (0usize, 0usize, 0usize, 0usize);
210    for file in catalog
211        .files
212        .iter()
213        .filter(|f| super::in_scope(&f.path, scope))
214    {
215        match file.integrity.as_ref().map(|r| &r.verdict) {
216            Some(Verdict::Intact) => intact += 1,
217            Some(Verdict::Damaged { .. }) => damaged += 1,
218            Some(Verdict::NothingToCheck) => nothing += 1,
219            None => unchecked += 1,
220        }
221    }
222
223    println!("{}", ui::section("Integrity"));
224    let mut table = Table::plain(2).align(1, Align::Right);
225    table.push(vec!["Intact".into(), intact.to_string()]);
226    table.push(vec!["Damaged".into(), damaged.to_string()]);
227    table.push(vec!["No checksum in the file".into(), nothing.to_string()]);
228    if unchecked > 0 {
229        table.push(vec!["Not verified".into(), unchecked.to_string()]);
230    }
231    print!("{}", table.render());
232    if !scope.is_empty() {
233        println!("  {}", ui::dim(&format!("in {}", scope.join(", "))));
234    }
235
236    // What this run did, said apart from what the library holds.
237    match elapsed_ms {
238        Some(ms) => println!(
239            "  {}",
240            ui::dim(&format!(
241                "{} read in {}",
242                ui::plural(read, "file"),
243                ui::elapsed(ms)
244            ))
245        ),
246        // A scope holding none of the catalog no longer reaches this point:
247        // `scope_of` refuses it, for all five commands that take a folder
248        // rather than for this one alone. What is left is the empty catalog.
249        None if intact + damaged + nothing + unchecked == 0 => {
250            println!("  {}", ui::yellow("the catalog holds no file"))
251        }
252        None => {
253            println!("  {}", ui::dim("nothing to read: it all has a verdict"));
254            println!("  {}", ui::dim("aede check --full verifies them again"));
255        }
256    }
257
258    if damaged > 0 {
259        println!("{}", ui::section("Damaged files"));
260        let mut t = Table::new(&["File", "Problem"]).path_limit(0, 60);
261        for file in catalog
262            .files
263            .iter()
264            .filter(|f| super::in_scope(&f.path, scope))
265        {
266            if let Some(record) = &file.integrity
267                && let Verdict::Damaged { detail } = &record.verdict
268            {
269                t.push(vec![file.path.clone(), detail.clone()]);
270            }
271        }
272        print!("{}", t.render());
273        println!(
274            "{}",
275            ui::dim(
276                "  a damaged file cannot be repaired: restore it from a backup or rip it again"
277            )
278        );
279    }
280
281    if !failures.is_empty() {
282        super::print_failure_table("Unreadable files", "File", failures);
283    }
284}
285
286fn resolve_threads(requested: usize) -> usize {
287    if requested > 0 {
288        return requested;
289    }
290    std::thread::available_parallelism()
291        .map(|n| n.get())
292        .unwrap_or(4)
293}
