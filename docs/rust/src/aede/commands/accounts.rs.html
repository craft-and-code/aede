13const USAGE: &str = "aede accounts [list | init <name> | create <name> <admin|user|auditor> | password <name> | role <name> <admin|user|auditor> | rename <name> <new-name> | enable <name> | disable <name> | revoke <name> | keys <name> [create <label> | revoke <id>]] [--password-stdin] [--json]";

15fn password(args: &Args) -> Result<String, Box<dyn std::error::Error>> {
16    if !args.has("password-stdin") || std::io::stdin().is_terminal() {
17        return Err("supply a password using --password-stdin with redirected input; passwords are never command-line arguments".into());
18    }
19    let mut password = String::new();
20    std::io::stdin()
21        .take(accounts::MAX_PASSWORD_BYTES as u64 + 3)
22        .read_to_string(&mut password)?;
23    if password.ends_with('\n') {
24        password.pop();
25        if password.ends_with('\r') {
26            password.pop();
27        }
28    }
29    if password.len() > accounts::MAX_PASSWORD_BYTES || password.contains(['\r', '\n']) {
30        return Err("provide one password line of at most 1024 UTF-8 bytes".into());
31    }
32    Ok(password)
33}

35/// Manage credentials using the OS owner's existing data-directory lock.
36/// Read-only listing requires neither a catalog nor a writer lock.
37pub fn accounts(args: &Args) -> Res {
38    let words: Vec<&str> = args.positionals.iter().map(String::as_str).collect();
39    let operation = words.first().copied().unwrap_or("list");
40    let needs_password = matches!(operation, "init" | "create" | "password");
41    if args.value("password-stdin").is_some() || args.has("password-stdin") && !needs_password {
42        return Err("--password-stdin is a flag for init, create and password only".into());
43    }
44    if operation == "keys" {
45        return keys::run(args, &words);
46    }
47    let count = match operation {
48        "list" => {
49            if words.is_empty() {
50                0
51            } else {
52                1
53            }
54        }
55        "init" | "password" | "enable" | "disable" | "revoke" => 2,
56        "create" | "role" | "rename" => 3,
57        _ => return Err(USAGE.into()),
58    };
59    if words.len() != count {
60        return Err(USAGE.into());
61    }
62    let path = accounts::accounts_path(&super::data_dir(args));
63    let mut data = accounts::load(&path)?;
64    let now = aede_core::clock::now_seconds();
65    if operation == "init" {
66        if data.is_some() {
67            return Err("accounts are already initialized; use create or password".into());
68        }
69        data = Some(Accounts::bootstrap(words[1], &password(args)?, now)?);
70    } else if operation != "list" {
71        let held = data.as_mut().ok_or(
72            "initialize the first administrator with aede accounts init <name> --password-stdin",
73        )?;
74        match operation {
75            "create" => held.create(
76                words[1],
77                &password(args)?,
78                Role::parse(words[2]).ok_or("role must be admin, user or auditor")?,
79                now,
80            )?,
81            "password" => held.set_password(words[1], &password(args)?, now)?,
82            "role" => held.set_role(
83                words[1],
84                Role::parse(words[2]).ok_or("role must be admin, user or auditor")?,
85                now,
86            )?,
87            "rename" => held.rename(words[1], words[2], now)?,
88            "enable" | "disable" => held.set_enabled(words[1], operation == "enable", now)?,
89            "revoke" => held.revoke(words[1], now)?,
90            _ => return Err(USAGE.into()),
91        }
92    }
93    if operation != "list" {
94        accounts::save(data.as_ref().ok_or("accounts were not initialized")?, &path)?;
95    }
96    if args.has("json") {
97        let rows = data.as_ref().map_or(&[][..], Accounts::all);
98        println!(
99            "{}",
100            Json::Arr(
101                rows.iter()
102                    .map(|account| {
103                        let mut row = Json::obj();
104                        row.set("id", account.id.clone().into());
105                        row.set("username", account.username.clone().into());
106                        row.set("role", account.role.as_str().into());
107                        row.set("enabled", account.enabled.into());
108                        row.set("created_at", account.created_at.into());
109                        row.set("updated_at", account.updated_at.into());
110                        row
111                    })
112                    .collect()
113            )
114            .to_string_pretty()
115        );
116    } else if let Some(data) = data {
117        let mut table = ui::Table::new(&["Login", "Role", "Access", "Personal owner"]);
118        for account in data.all() {
119            table.push(vec![
120                account.username.clone(),
121                account.role.as_str().into(),
122                if account.enabled {
123                    "enabled"
124                } else {
125                    "disabled"
126                }
127                .into(),
128                account.id.clone(),
129            ]);
130        }
131        print!("{}", table.render());
132        if operation != "list" {
133            println!("Account update saved; affected sessions and API keys are revoked.");
134        }
135    } else {
136        println!(
137            "No accounts configured. Initialize one with aede accounts init <name> --password-stdin."
138        );
139    }
140    Ok(())
141}
