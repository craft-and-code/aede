1//! Complete accepted artist-search identities without upgrading their confidence or changing tags.
2use super::*;
3use aede_core::{model::Catalog, musicbrainz};
4
5pub(super) fn complete(
6    transport: &mut dyn Ask,
7    backoff: &[std::time::Duration],
8    held: &mut sources::Sources,
9    path: &std::path::Path,
10    asked: &super::super::fetch::Asked,
11    catalog: Option<&Catalog>,
12) -> Res {
13    let empty = Catalog::default();
14    let catalog = catalog.unwrap_or(&empty);
15    let targets: Vec<_> = held
16        .records
17        .iter()
18        .enumerate()
19        .filter_map(|(index, record)| {
20            let Facts::Artist(facts) = &record.facts else {
21                return None;
22            };
23            let id = super::super::fetch::lookup_selector(catalog, record)?;
24            (record.source == sources::MUSICBRAINZ
25                && musicbrainz::valid_id(id)
26                && held.is_trusted(catalog, record)
27                && asked.scope.has_artist(&record.key)
28                && super::super::fetch::reaches(asked.names, &[&record.key])
29                && facts
30                    .wikidata
31                    .as_deref()
32                    .and_then(wikipedia::entity_id)
33                    .is_none()
34                && (!facts.relationships_complete || asked.again))
35                .then(|| (index, id.to_string(), record.key.clone()))
36        })
37        .collect();
38    if targets.is_empty() {
39        return Ok(());
40    }
41    if super::super::fetch::asked_nothing(
42        asked,
43        &targets
44            .iter()
45            .map(|(_, _, name)| name.clone())
46            .collect::<Vec<_>>(),
47    ) {
48        return Ok(());
49    }
50    for (index, id, name) in targets {
51        println!("  completing MusicBrainz links for {}", ui::literal(&name));
52        let url = format!(
53            "{}/artist/{id}?fmt=json&inc={}",
54            musicbrainz::WEB_SERVICE,
55            musicbrainz::ARTIST_INCLUDES
56        );
57        let answer = match ask_with_backoff(transport, &url, backoff) {
58            Ok(answer) => answer,
59            Err(failure) if failure.must_stop() => return Err(failure.into()),
60            Err(failure) => {
61                super::super::fetch::report_failed(&name, &failure);
62                continue;
63            }
64        };
65        let Some(found) =
66            musicbrainz::artist(&answer).filter(|candidate| musicbrainz::valid_id(&candidate.mbid))
67        else {
68            super::super::fetch::report_failed(
69                &name,
70                &"MusicBrainz artist details are unreadable; existing claims retained",
71            );
72            continue;
73        };
74        let previous = &held.records[index];
75        let review = super::super::fetch::continued_lookup_review(
76            held,
77            &previous.entity(),
78            &found.mbid,
79            Some(&id),
80        );
81        let previous = &mut held.records[index];
82        let Facts::Artist(existing) = &mut previous.facts else {
83            continue;
84        };
85        // A redirect changes its canonical address, not its score or decision.
86        let mut facts = found.facts;
87        facts.lookup_mbid = (id != found.mbid)
88            .then_some(id)
89            .or_else(|| existing.lookup_mbid.take());
90        facts.discography = std::mem::take(&mut existing.discography);
91        facts.discography_fetched_at = existing.discography_fetched_at;
92        let updated = SourceRecord {
93            key: previous.key.clone(),
94            source: previous.source.clone(),
95            source_id: Some(found.mbid),
96            confidence: previous.confidence,
97            fetched_at: clock::now_seconds(),
98            facts: Facts::Artist(facts),
99        };
100        held.set(updated);
101        if let Some(review) = review {
102            held.set_review(review);
103        }
104        sources::save(held, path)?;
105    }
106    Ok(())
107}
