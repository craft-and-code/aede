15#[derive(Clone, Copy, Debug, PartialEq, Eq)]
16enum Key {
17    Char(char),
18    Enter,
19    Backspace,
20    CtrlC,
21    Escape,
22    Tab,
23    #[cfg(any(unix, test))]
24    Ignored,
25}
26
27fn character_key(character: char) -> Key {
28    match character {
29        '\r' | '\n' => Key::Enter,
30        '\u{8}' | '\u{7f}' => Key::Backspace,
31        '\u{3}' => Key::CtrlC,
32        '\u{1b}' => Key::Escape,
33        '\t' => Key::Tab,
34        character => Key::Char(character),
35    }
36}

38pub(super) fn read(args: &Args) -> Result<String, Box<dyn std::error::Error>> {
39    if args.has("password-stdin") {
40        if io::stdin().is_terminal() {
41            return Err(
42                "--password-stdin requires redirected input; omit it for masked terminal entry"
43                    .into(),
44            );
45        }
46        return redirected(io::stdin().lock()).map_err(Into::into);
47    }
48    if !io::stdin().is_terminal() || !io::stderr().is_terminal() {
49        return Err("use a terminal for masked password entry, or --password-stdin with redirected input for scripts".into());
50    }
51    #[cfg(unix)]
52    let mut input = unix::Input::start()?;
53    #[cfg(windows)]
54    let mut input = windows::Input::start()?;
55    #[cfg(not(any(unix, windows)))]
56    return Err(
57        "masked terminal input is unsupported on this platform; use --password-stdin".into(),
58    );
59    #[cfg(any(unix, windows))]
60    {
61        let mut terminal = io::stderr().lock();
62        let result = confirmed(|prompt| {
63            terminal.write_all(prompt.as_bytes())?;
64            terminal.flush()?;
65            #[cfg(unix)]
66            let result = read_keys(|| input.read_key());
67            #[cfg(windows)]
68            let result = read_keys(|| input.read_character().map(character_key));
69            terminal.write_all(b"\n")?;
70            result
71        });
72        input.restore()?;
73        result.map_err(Into::into)
74    }
75}

77fn redirected(reader: impl Read) -> io::Result<String> {
78    let mut password = String::new();
79    reader
80        .take(MAX_PASSWORD_BYTES as u64 + 3)
81        .read_to_string(&mut password)?;
82    if password.ends_with('\n') {
83        password.pop();
84        if password.ends_with('\r') {
85            password.pop();
86        }
87    }
88    if password.len() > MAX_PASSWORD_BYTES || password.contains(['\r', '\n']) {
89        return Err(io::Error::new(
90            io::ErrorKind::InvalidInput,
91            "provide one password line of at most 1024 UTF-8 bytes",
92        ));
93    }
94    Ok(password)
95}
96
97fn confirmed(mut read_line: impl FnMut(&'static str) -> io::Result<String>) -> io::Result<String> {
98    let password = read_line("Password: ")?;
99    let confirmation = read_line("Confirm password: ")?;
100    if password != confirmation {
101        return Err(io::Error::new(
102            io::ErrorKind::InvalidInput,
103            "passwords do not match; no account changes were saved",
104        ));
105    }
106    Ok(password)
107}

109fn read_keys(mut next: impl FnMut() -> io::Result<Key>) -> io::Result<String> {
110    let mut password = String::new();
111    // Consume an oversized line without allocating it or silently truncating
112    // it. Backspace can remove its excess characters, and Ctrl-U clears it.
113    let mut excess = 0usize;
114    loop {
115        match next()? {
116            Key::Enter => {
117                if excess != 0 {
118                    return Err(io::Error::new(
119                        io::ErrorKind::InvalidInput,
120                        "password must contain at most 1024 UTF-8 bytes",
121                    ));
122                }
123                return Ok(password);
124            }
125            Key::CtrlC | Key::Escape | Key::Char('\u{4}') => {
126                return Err(io::Error::new(
127                    io::ErrorKind::Interrupted,
128                    "password entry cancelled; no account changes were saved",
129                ));
130            }
131            Key::Backspace => {
132                if excess != 0 {
133                    excess -= 1;
134                } else {
135                    password.pop();
136                }
137            }
138            Key::Char('\u{15}') => {
139                password.clear();
140                excess = 0;
141            }
142            Key::Char(character) if !character.is_control() => {
143                if excess != 0 || password.len() + character.len_utf8() > MAX_PASSWORD_BYTES {
144                    excess = excess.saturating_add(1);
145                } else {
146                    password.push(character);
147                }
148            }
149            Key::Char(_) | Key::Tab => {
150                return Err(io::Error::new(
151                    io::ErrorKind::InvalidInput,
152                    "terminal passwords cannot contain control characters",
153                ));
154            }
155            #[cfg(any(unix, test))]
156            Key::Ignored => {}
157        }
158    }
159}
