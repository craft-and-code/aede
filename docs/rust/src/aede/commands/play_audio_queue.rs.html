1//! Bounded single-producer PCM handoff to an audio callback.
2//!
3//! The ring stores interleaved samples and publishes only complete frames.
4//! Allocation and validation happen outside the callback. Rendering copies
5//! directly from preallocated ring storage and never converts inserted silence.
6
7use std::io;
8use std::sync::Arc;
9use std::sync::atomic::{AtomicBool, AtomicU64, Ordering};
10
11use rtrb::{Consumer, Producer, RingBuffer};
12
13/// A concurrent observation of the bounded queue, expressed in PCM frames.
14#[derive(Clone, Copy, Debug, Eq, PartialEq)]
15pub(super) struct QueueSnapshot {
16    pub(super) queued_frames: usize,
17    pub(super) capacity_frames: usize,
18    pub(super) consumed_frames: u64,
19    pub(super) underrun_frames: u64,
20    pub(super) underrun_callbacks: u64,
21}
22
23struct Counters {
24    submitted_samples: AtomicU64,
25    consumed_samples: AtomicU64,
26    underrun_frames: AtomicU64,
27    underrun_callbacks: AtomicU64,
28    closed: AtomicBool,
29}
30
31/// The decoding thread's sole writer to a fixed-capacity PCM queue.
32pub(super) struct PcmProducer {
33    ring: Producer<f32>,
34    channels: usize,
35    capacity_frames: usize,
36    counters: Arc<Counters>,
37}
38
39/// The audio callback's sole reader of complete interleaved PCM frames.
40pub(super) struct PcmConsumer {
41    ring: Consumer<f32>,
42    channels: usize,
43    counters: Arc<Counters>,
44}
45
46/// Allocate the queue before starting the audio stream. Zero channels, zero
47/// capacity and capacities that overflow an allocation are rejected.
48pub(super) fn pcm_queue(
49    channels: usize,
50    capacity_frames: usize,
51) -> io::Result<(PcmProducer, PcmConsumer)> {
52    let capacity_samples = capacity_frames
53        .checked_mul(channels)
54        .filter(|&samples| {
55            samples > 0 && samples <= isize::MAX as usize / std::mem::size_of::<f32>()
56        })
57        .ok_or_else(|| io::Error::new(io::ErrorKind::InvalidInput, "invalid PCM queue capacity"))?;
58    let (producer, consumer) = RingBuffer::new(capacity_samples);
59    let counters = Arc::new(Counters {
60        submitted_samples: AtomicU64::new(0),
61        consumed_samples: AtomicU64::new(0),
62        underrun_frames: AtomicU64::new(0),
63        underrun_callbacks: AtomicU64::new(0),
64        closed: AtomicBool::new(false),
65    });
66    Ok((
67        PcmProducer {
68            ring: producer,
69            channels,
70            capacity_frames,
71            counters: Arc::clone(&counters),
72        },
73        PcmConsumer {
74            ring: consumer,
75            channels,
76            counters,
77        },
78    ))
79}

81impl PcmProducer {
82    /// Validate the entire input, then publish the largest complete-frame
83    /// prefix that fits. Return samples written, or `WouldBlock` if no frame
84    /// fits. Invalid PCM never publishes even a preceding valid prefix.
85    pub(super) fn write_samples(&mut self, samples: &[f32]) -> io::Result<usize> {
86        if !samples.len().is_multiple_of(self.channels)
87            || samples.iter().any(|&sample| !valid_sample(sample))
88        {
89            return Err(io::Error::new(
90                io::ErrorKind::InvalidInput,
91                "PCM requires complete finite guarded frames",
92            ));
93        }
94        self.write_validated(samples.len(), samples.iter().copied())
95    }
96
97    /// Validate complete little-endian floating-point PCM before publication.
98    /// Return bytes written; the writer initializes ring slots directly from
99    /// the encoded samples and does not allocate an intermediate PCM buffer.
100    pub(super) fn write_f32le(&mut self, bytes: &[u8]) -> io::Result<usize> {
101        let (samples, remainder) = bytes.as_chunks::<4>();
102        if !remainder.is_empty()
103            || !samples.len().is_multiple_of(self.channels)
104            || samples
105                .iter()
106                .any(|&sample| !valid_sample(f32::from_le_bytes(sample)))
107        {
108            return Err(io::Error::new(
109                io::ErrorKind::InvalidInput,
110                "f32le requires complete finite guarded frames",
111            ));
112        }
113        self.write_validated(
114            samples.len(),
115            samples.iter().map(|&sample| f32::from_le_bytes(sample)),
116        )
117        .map(|written| written * 4)
118    }
119
120    fn write_validated(
121        &mut self,
122        samples: usize,
123        values: impl Iterator<Item = f32>,
124    ) -> io::Result<usize> {
125        if self.counters.closed.load(Ordering::Acquire) || self.ring.is_abandoned() {
126            return Err(io::Error::new(
127                io::ErrorKind::BrokenPipe,
128                "PCM queue is closed",
129            ));
130        }
131        if samples == 0 {
132            return Ok(0);
133        }
134        let available = self.ring.slots();
135        let accepted = samples.min(available / self.channels * self.channels);
136        if accepted == 0 {
137            return Err(io::ErrorKind::WouldBlock.into());
138        }
139        // Only the consumer can change available slots, and it can only add
140        // free slots. This reservation cannot shrink between these calls.
141        let chunk = self
142            .ring
143            .write_chunk_uninit(accepted)
144            .map_err(|_| io::Error::from(io::ErrorKind::WouldBlock))?;
145        let written = chunk.fill_from_iter(values.take(accepted));
146        self.counters
147            .submitted_samples
148            .fetch_add(written as u64, Ordering::Release);
149        Ok(written)
150    }
151
152    /// Seal the producer while allowing queued frames to reach the callback.
153    pub(super) fn close_input(&self) {
154        self.counters.closed.store(true, Ordering::Release);
155    }
156
157    /// Whether the callback has consumed every submitted sample.
158    pub(super) fn drained(&self) -> bool {
159        self.counters.submitted_samples.load(Ordering::Acquire)
160            == self.counters.consumed_samples.load(Ordering::Acquire)
161    }
162
163    /// Observe bounded queue occupancy and monotonically increasing counters.
164    /// Occupancy is approximate while the producer and callback are advancing.
165    pub(super) fn snapshot(&self) -> QueueSnapshot {
166        let submitted = self.counters.submitted_samples.load(Ordering::Acquire);
167        let consumed = self.counters.consumed_samples.load(Ordering::Acquire);
168        let queued = submitted.saturating_sub(consumed) / self.channels as u64;
169        QueueSnapshot {
170            queued_frames: usize::try_from(queued)
171                .unwrap_or(usize::MAX)
172                .min(self.capacity_frames),
173            capacity_frames: self.capacity_frames,
174            consumed_frames: consumed / self.channels as u64,
175            underrun_frames: self.counters.underrun_frames.load(Ordering::Acquire),
176            underrun_callbacks: self.counters.underrun_callbacks.load(Ordering::Acquire),
177        }
178    }
179}
180
181impl PcmConsumer {
182    /// Render available whole frames, then exact digital silence. The callback
183    /// performs bounded copying and atomic counter updates; it does not allocate,
184    /// deallocate, lock, log or wait. An incomplete final output frame receives
185    /// silence without consuming any source channel.
186    pub(super) fn render_mapped<T: Copy>(
187        &mut self,
188        output: &mut [T],
189        silence: T,
190        mut convert: impl FnMut(f32) -> T,
191    ) {
192        let closed_before = self.counters.closed.load(Ordering::Acquire);
193        let started_before = self.counters.submitted_samples.load(Ordering::Acquire) > 0;
194        output.fill(silence);
195        let requested_frames = output.len() / self.channels;
196        if requested_frames == 0 {
197            return;
198        }
199        let available_frames = self.ring.slots() / self.channels;
200        let samples = requested_frames.min(available_frames) * self.channels;
201        let mut consumed = 0;
202        if samples > 0
203            && let Ok(chunk) = self.ring.read_chunk(samples)
204        {
205            let (first, second) = chunk.as_slices();
206            for (source, target) in first.iter().chain(second).zip(&mut output[..samples]) {
207                *target = convert(*source);
208            }
209            chunk.commit_all();
210            consumed = samples;
211            self.counters
212                .consumed_samples
213                .fetch_add(consumed as u64, Ordering::Release);
214        }
215        let missing_frames = requested_frames - consumed / self.channels;
216        if missing_frames > 0 && started_before && !closed_before {
217            self.counters
218                .underrun_frames
219                .fetch_add(missing_frames as u64, Ordering::Release);
220            self.counters
221                .underrun_callbacks
222                .fetch_add(1, Ordering::Release);
223        }
224    }
225}
226
227fn valid_sample(sample: f32) -> bool {
228    sample.is_finite() && (-1.0..=1.0).contains(&sample)
229}
