1//! Bounded single-producer PCM handoff to an audio callback.
2//!
3//! The ring stores interleaved samples and publishes only complete frames.
4//! Allocation and validation happen outside the callback. Rendering copies
5//! directly from preallocated ring storage and never converts inserted silence.
6
7use std::io;
8use std::sync::Arc;
9use std::sync::atomic::{AtomicBool, AtomicU64, Ordering};
10
11use aede_core::playback::format::IntegerPcmFormat;
12use rtrb::{Consumer, Producer, RingBuffer};
13
14/// A concurrent observation of the bounded queue, expressed in PCM frames.
15#[derive(Clone, Copy, Debug, Eq, PartialEq)]
16pub(super) struct QueueSnapshot {
17    pub(super) queued_frames: usize,
18    pub(super) capacity_frames: usize,
19    pub(super) consumed_frames: u64,
20    pub(super) underrun_frames: u64,
21    pub(super) underrun_callbacks: u64,
22    pub(super) failed: bool,
23}
24
25struct Counters {
26    submitted_samples: AtomicU64,
27    consumed_samples: AtomicU64,
28    underrun_frames: AtomicU64,
29    underrun_callbacks: AtomicU64,
30    closed: AtomicBool,
31    failed: AtomicBool,
32}
33
34/// Shared callback-safe stop signal for host errors, without queue ownership.
35#[derive(Clone)]
36pub(super) struct QueueFailure(Arc<Counters>);
37
38impl QueueFailure {
39    pub(super) fn fail(&self) {
40        self.0.failed.store(true, Ordering::Release);
41    }
42}
43
44/// Queue admission, separate from any device representation conversion.
45pub(super) trait QueueSample: Copy {
46    fn valid(self) -> bool;
47}
48
49impl QueueSample for f32 {
50    fn valid(self) -> bool {
51        self.is_finite() && (-1.0..=1.0).contains(&self)
52    }
53}
54
55impl QueueSample for i32 {
56    fn valid(self) -> bool {
57        // ExactPcmSession validates the source-depth range before submission.
58        true
59    }
60}

62/// The decoding thread's sole writer to a fixed-capacity PCM queue.
63pub(super) struct PcmProducer<T: QueueSample = f32> {
64    ring: Producer<T>,
65    channels: usize,
66    capacity_frames: usize,
67    counters: Arc<Counters>,
68}
69
70/// The audio callback's sole reader of complete interleaved PCM frames.
71pub(super) struct PcmConsumer<T: QueueSample = f32> {
72    ring: Consumer<T>,
73    channels: usize,
74    counters: Arc<Counters>,
75    strict: bool,
76}
77
78/// Allocate the queue before starting the audio stream. Zero channels, zero
79/// capacity and capacities that overflow an allocation are rejected.
80pub(super) fn pcm_queue<T: QueueSample>(
81    channels: usize,
82    capacity_frames: usize,
83    strict: bool,
84) -> io::Result<(PcmProducer<T>, PcmConsumer<T>)> {
85    let capacity_samples = capacity_frames
86        .checked_mul(channels)
87        .filter(|&samples| {
88            samples > 0
89                && std::mem::size_of::<T>() > 0
90                && samples <= isize::MAX as usize / std::mem::size_of::<T>()
91        })
92        .ok_or_else(|| io::Error::new(io::ErrorKind::InvalidInput, "invalid PCM queue capacity"))?;
93    let (producer, consumer) = RingBuffer::new(capacity_samples);
94    let counters = Arc::new(Counters {
95        submitted_samples: AtomicU64::new(0),
96        consumed_samples: AtomicU64::new(0),
97        underrun_frames: AtomicU64::new(0),
98        underrun_callbacks: AtomicU64::new(0),
99        closed: AtomicBool::new(false),
100        failed: AtomicBool::new(false),
101    });
102    Ok((
103        PcmProducer {
104            ring: producer,
105            channels,
106            capacity_frames,
107            counters: Arc::clone(&counters),
108        },
109        PcmConsumer {
110            ring: consumer,
111            channels,
112            counters,
113            strict,
114        },
115    ))
116}

118impl<T: QueueSample> PcmProducer<T> {
119    pub(super) fn failure_handle(&self) -> QueueFailure {
120        QueueFailure(Arc::clone(&self.counters))
121    }
122    /// Validate the entire input, then publish the largest complete-frame
123    /// prefix that fits. Return samples written, or `WouldBlock` if no frame
124    /// fits. Invalid PCM never publishes even a preceding valid prefix.
125    pub(super) fn write_samples(&mut self, samples: &[T]) -> io::Result<usize> {
126        if !samples.len().is_multiple_of(self.channels)
127            || samples.iter().any(|&sample| !sample.valid())
128        {
129            return Err(io::Error::new(
130                io::ErrorKind::InvalidInput,
131                "PCM requires complete valid frames",
132            ));
133        }
134        self.write_validated(samples.len(), samples.iter().copied())
135    }
136
137    fn write_validated(
138        &mut self,
139        samples: usize,
140        values: impl Iterator<Item = T>,
141    ) -> io::Result<usize> {
142        if self.counters.failed.load(Ordering::Acquire) {
143            return Err(io::Error::new(
144                io::ErrorKind::BrokenPipe,
145                "strict PCM queue has failed",
146            ));
147        }
148        if self.counters.closed.load(Ordering::Acquire) || self.ring.is_abandoned() {
149            return Err(io::Error::new(
150                io::ErrorKind::BrokenPipe,
151                "PCM queue is closed",
152            ));
153        }
154        if samples == 0 {
155            return Ok(0);
156        }
157        let available = self.ring.slots();
158        let accepted = samples.min(available / self.channels * self.channels);
159        if accepted == 0 {
160            return Err(io::ErrorKind::WouldBlock.into());
161        }
162        // Only the consumer can change available slots, and it can only add
163        // free slots. This reservation cannot shrink between these calls.
164        let chunk = self
165            .ring
166            .write_chunk_uninit(accepted)
167            .map_err(|_| io::Error::from(io::ErrorKind::WouldBlock))?;
168        let written = chunk.fill_from_iter(values.take(accepted));
169        self.counters
170            .submitted_samples
171            .fetch_add(written as u64, Ordering::Release);
172        Ok(written)
173    }
174
175    /// Seal the producer while allowing queued frames to reach the callback.
176    pub(super) fn close_input(&self) {
177        self.counters.closed.store(true, Ordering::Release);
178    }
179
180    /// Whether the callback has consumed every submitted sample.
181    pub(super) fn drained(&self) -> bool {
182        self.counters.submitted_samples.load(Ordering::Acquire)
183            == self.counters.consumed_samples.load(Ordering::Acquire)
184    }
185
186    /// Observe bounded queue occupancy and monotonically increasing counters.
187    /// Occupancy is approximate while the producer and callback are advancing.
188    pub(super) fn snapshot(&self) -> QueueSnapshot {
189        let submitted = self.counters.submitted_samples.load(Ordering::Acquire);
190        let consumed = self.counters.consumed_samples.load(Ordering::Acquire);
191        let queued = submitted.saturating_sub(consumed) / self.channels as u64;
192        QueueSnapshot {
193            queued_frames: usize::try_from(queued)
194                .unwrap_or(usize::MAX)
195                .min(self.capacity_frames),
196            capacity_frames: self.capacity_frames,
197            consumed_frames: consumed / self.channels as u64,
198            underrun_frames: self.counters.underrun_frames.load(Ordering::Acquire),
199            underrun_callbacks: self.counters.underrun_callbacks.load(Ordering::Acquire),
200            failed: self.counters.failed.load(Ordering::Acquire),
201        }
202    }
203}

205impl PcmProducer<i32> {
206    /// Refuse invalid source-depth samples before any prefix enters the ring.
207    pub(super) fn write_integer(
208        &mut self,
209        samples: &[i32],
210        format: IntegerPcmFormat,
211    ) -> io::Result<usize> {
212        let limit = 1_i32 << (format.bits_per_sample() - 1);
213        if usize::from(format.channels()) != self.channels
214            || samples
215                .iter()
216                .any(|sample| !(-limit..limit).contains(sample))
217        {
218            return Err(io::Error::new(
219                io::ErrorKind::InvalidInput,
220                "integer PCM disagrees with source format",
221            ));
222        }
223        self.write_samples(samples)
224    }
225}

227impl PcmProducer<f32> {
228    /// Validate complete little-endian floating-point PCM before publication.
229    /// Return bytes written; the writer initializes ring slots directly from
230    /// the encoded samples and does not allocate an intermediate PCM buffer.
231    pub(super) fn write_f32le(&mut self, bytes: &[u8]) -> io::Result<usize> {
232        let (samples, remainder) = bytes.as_chunks::<4>();
233        if !remainder.is_empty()
234            || !samples.len().is_multiple_of(self.channels)
235            || samples
236                .iter()
237                .any(|&sample| !f32::from_le_bytes(sample).valid())
238        {
239            return Err(io::Error::new(
240                io::ErrorKind::InvalidInput,
241                "f32le requires complete finite guarded frames",
242            ));
243        }
244        self.write_validated(
245            samples.len(),
246            samples.iter().map(|&sample| f32::from_le_bytes(sample)),
247        )
248        .map(|written| written * 4)
249    }
250}

252impl<S: QueueSample> PcmConsumer<S> {
253    #[cfg(any(test, all(target_os = "linux", target_env = "gnu")))]
254    pub(super) fn available_frames(&self) -> usize {
255        self.ring.slots() / self.channels
256    }
257
258    #[cfg(any(test, all(target_os = "linux", target_env = "gnu")))]
259    pub(super) fn input_closed(&self) -> bool {
260        self.counters.closed.load(Ordering::Acquire)
261    }
262
263    #[cfg(any(test, all(target_os = "linux", target_env = "gnu")))]
264    pub(super) fn has_failed(&self) -> bool {
265        self.counters.failed.load(Ordering::Acquire)
266    }
267
268    /// Render available whole frames, then exact digital silence. The callback
269    /// performs bounded copying and atomic counter updates; it does not allocate,
270    /// deallocate, lock, log or wait. Ordinary playback silences an incomplete
271    /// final output frame without consuming any source channel. Strict playback
272    /// fails before consuming source samples if the callback is not frame-aligned.
273    pub(super) fn render_mapped<T: Copy>(
274        &mut self,
275        output: &mut [T],
276        silence: T,
277        mut convert: impl FnMut(S) -> T,
278    ) {
279        let closed_before = self.counters.closed.load(Ordering::Acquire);
280        let started_before = self.counters.submitted_samples.load(Ordering::Acquire) > 0;
281        output.fill(silence);
282        if self.counters.failed.load(Ordering::Acquire) {
283            return;
284        }
285        if self.strict && !output.len().is_multiple_of(self.channels) {
286            // A partial frame would insert a channel sample outside the source
287            // timeline. There is no exact continuation through that callback.
288            self.counters.failed.store(true, Ordering::Release);
289            return;
290        }
291        let requested_frames = output.len() / self.channels;
292        if requested_frames == 0 {
293            return;
294        }
295        let available_frames = self.ring.slots() / self.channels;
296        let samples = requested_frames.min(available_frames) * self.channels;
297        let mut consumed = 0;
298        if samples > 0
299            && let Ok(chunk) = self.ring.read_chunk(samples)
300        {
301            let (first, second) = chunk.as_slices();
302            for (source, target) in first.iter().chain(second).zip(&mut output[..samples]) {
303                *target = convert(*source);
304            }
305            chunk.commit_all();
306            consumed = samples;
307            self.counters
308                .consumed_samples
309                .fetch_add(consumed as u64, Ordering::Release);
310        }
311        let missing_frames = requested_frames - consumed / self.channels;
312        // A producer may finish while this callback is copying the final
313        // source frames. In strict mode, fresh closure plus an empty ring
314        // proves that the unfilled callback tail is outside the programme.
315        let finished = if self.strict {
316            self.counters.closed.load(Ordering::Acquire) && self.ring.slots() == 0
317        } else {
318            closed_before
319        };
320        if missing_frames > 0 && (started_before || (self.strict && consumed > 0)) && !finished {
321            self.counters
322                .underrun_frames
323                .fetch_add(missing_frames as u64, Ordering::Release);
324            self.counters
325                .underrun_callbacks
326                .fetch_add(1, Ordering::Release);
327            if self.strict {
328                self.counters.failed.store(true, Ordering::Release);
329            }
330        }
331    }
332}
