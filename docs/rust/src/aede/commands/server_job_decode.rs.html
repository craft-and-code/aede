1//! Observe complete JSON row lines while preserving the ordinary final JSON capture.
2use super::{JobOutput, MAX_OUTPUT, redact};
3use aede_core::json::{self, Json};
4pub(super) struct Snapshots {
5    check: Option<aede_server::CheckRequest>,
6    secrets: Vec<String>,
7    offset: usize,
8    started: bool,
9    files: Vec<Json>,
10}
11impl Snapshots {
12    pub(super) fn new(check: Option<aede_server::CheckRequest>, secrets: Vec<String>) -> Self {
13        Self {
14            check,
15            secrets,
16            offset: 0,
17            started: false,
18            files: Vec::new(),
19        }
20    }
21    pub(super) fn observe(&mut self, bytes: &[u8], truncated: bool) {
22        if truncated {
23            self.check = None;
24        }
25        let Some(check) = &self.check else {
26            return;
27        };
28        while let Some(end) = bytes[self.offset..].iter().position(|byte| *byte == b'\n') {
29            let line = &bytes[self.offset..self.offset + end];
30            self.offset += end + 1;
31            if !self.started {
32                self.started = line == b"{\"version\":1,\"files\":[";
33                continue;
34            }
35            let Ok(line) = std::str::from_utf8(line) else {
36                continue;
37            };
38            if line == "]}" {
39                continue;
40            }
41            let Ok(file) = json::parse(line.trim_start_matches(',')) else {
42                continue;
43            };
44            if file.get("path").and_then(Json::as_str).is_none() {
45                continue;
46            }
47            if self.files.len() >= check.files.len() {
48                return;
49            }
50            self.files.push(file);
51            let output = self.output();
52            if output.stdout.len() > MAX_OUTPUT {
53                return;
54            }
55            check.report(output);
56        }
57    }
58    fn output(&self) -> JobOutput {
59        let mut value = Json::obj();
60        value.set("version", Json::Num(1.0));
61        value.set("files", Json::Arr(self.files.clone()));
62        let failed = self
63            .files
64            .iter()
65            .any(|file| file.get("status").and_then(Json::as_str) == Some("ko"));
66        JobOutput {
67            stdout: redact(value.to_string_compact(), &self.secrets),
68            exit_code: i32::from(failed),
69            ..Default::default()
70        }
71    }
72}
