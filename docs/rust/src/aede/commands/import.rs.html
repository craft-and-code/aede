1//! The `import` command: take in what another tool measured.
2//!
3//! Entirely optional. Aède reads the structure of a file; it does not decode.
4//! A user who has already run FlacCompagnon over an album has answers Aède
5//! cannot compute before its own decoder arrives — whether the audio still
6//! matches its MD5, whether the FLAC was made from an MP3, where the spectrum
7//! stops — and those answers are worth keeping.
8//!
9//! Imported measurements never overwrite Aède's own. They are attributed to
10//! their source and stored beside them, which is what lets `doctor` say that
11//! two methods disagree.
12//!
13//! An analysis is filed under the **path** it describes, not under a catalog
14//! entry. A report may therefore be imported before the library is scanned:
15//! the records wait, and the scan that brings the files in picks them up. The
16//! two operations can be done in either order.
17

30pub fn import(args: &Args) -> Res {
31    let dir = data_dir(args);
32    let catalog_file = store::catalog_path(&dir);
33    // The one command that does not demand a catalog. Analysing a library and
34    // then building it is a legitimate order, and refusing the import until a
35    // scan has run would force the other one for no reason.
36    let mut catalog = match store::load(&catalog_file)? {
37        Some(catalog) => catalog,
38        None => {
39            let mut empty = Catalog::default();
40            if let Some(gathered) =
41                aede_core::conclusions::load(&aede_core::conclusions::conclusions_path(&dir))?
42            {
43                gathered.attach(&mut empty);
44            }
45            empty
46        }
47    };
48
49    if args.has("forget") {
50        return forget(&mut catalog, &catalog_file, args);
51    }
52
53    if args.has("pending") {
54        return list_pending(&catalog, args);
55    }
56
57    if args.has("list") {
58        return list_all(&catalog, args);
59    }
60
61    if args.positionals.is_empty() {
62        return Err("give a report, or a folder holding some: aede import report.json".into());
63    }
64
65    let mut reports = Vec::new();
66    for raw in &args.positionals {
67        collect_reports(Path::new(raw), &mut reports)?;
68    }
69    if reports.is_empty() {
70        return Err("no .json report found in what was given".into());
71    }
72
73    // The matching itself lives in the core, because the scan does exactly the
74    // same thing with the reports it walks over. Two copies of "which file is
75    // this record about" would drift.
76    let mut records: Vec<FileAnalysis> = Vec::new();
77    for report_path in &reports {
78        let report = match analysis::read_report(report_path) {
79            Ok(report) => report,
80            Err(error) => {
81                return Err(format!("{}: {error}", report_path.display()).into());
82            }
83        };
84        records.extend(report.files);
85    }
86
87    let outcome = analysis::merge_into(&mut catalog, records, now_seconds());
88    store::save(&catalog, &catalog_file)?;
89
90    println!("{}", ui::section("Import"));
91    let mut table = Table::plain(2).align(1, Align::Right);
92    table.push(vec!["Reports read".into(), reports.len().to_string()]);
93    table.push(vec!["Files matched".into(), outcome.matched.to_string()]);
94    if outcome.older > 0 {
95        table.push(vec![
96            "Older results skipped".into(),
97            outcome.older.to_string(),
98        ]);
99    }
100    if outcome.moved > 0 {
101        table.push(vec![
102            "Matched by name and size".into(),
103            outcome.moved.to_string(),
104        ]);
105    }
106    if outcome.stale > 0 {
107        table.push(vec![
108            "Changed since the report".into(),
109            outcome.stale.to_string(),
110        ]);
111    }
112    if outcome.waiting > 0 {
113        table.push(vec![
114            "Waiting for a scan".into(),
115            outcome.waiting.to_string(),
116        ]);
117    }
118    table.push(vec![
119        "Analyses stored".into(),
120        catalog.analyses.len().to_string(),
121    ]);
122    print!("{}", table.render());
123
124    if !outcome.stale_folders.is_empty() {
125        // Named for the same reason `waiting_folders` is, and the same
126        // reason the count alone used to be the whole message: "some files
127        // changed" says how many, never which, and which is the one thing a
128        // reader needs to point FlacCompagnon at the right album again.
129        // These records are never stored — this printout is the only place
130        // they are ever named, so there is no `--stale` to list the rest of
131        // them later the way `--pending` can for waiting ones.
132        println!("{}", ui::section("Changed since the report"));
133        let shown: usize = outcome.stale_folders.values().sum();
134        let mut t = Table::new(&["Folder", "Analyses"]).align(1, Align::Right);
135        for (folder, count) in &outcome.stale_folders {
136            t.push(vec![folder.clone(), count.to_string()]);
137        }
138        print!("{}", t.render());
139        if outcome.stale > shown {
140            println!(
141                "  {}",
142                ui::dim(&format!(
143                    "… and {} elsewhere, not shown",
144                    ui::plural(outcome.stale - shown, "analysis")
145                ))
146            );
147        }
148        println!(
149            "  {}",
150            ui::yellow("run FlacCompagnon again on the folders above")
151        );
152    }
153    if !outcome.waiting_folders.is_empty() {
154        // Folders, written out whole — the same reasoning as `--pending`, and
155        // the same bug before it: a path cut to a column width loses its head,
156        // which is the half that says *which* folder. See [`list_pending`].
157        println!("{}", ui::section("Waiting for a scan"));
158        let shown: usize = outcome.waiting_folders.values().sum();
159        let mut t = Table::new(&["Folder", "Analyses"]).align(1, Align::Right);
160        for (folder, count) in &outcome.waiting_folders {
161            t.push(vec![folder.clone(), count.to_string()]);
162        }
163        print!("{}", t.render());
164        if outcome.waiting > shown {
165            println!(
166                "  {}",
167                ui::dim(&format!(
168                    "… and {} elsewhere — aede import --pending lists every folder",
169                    ui::plural(outcome.waiting - shown, "analysis")
170                ))
171            );
172        }
173        // Kept rather than refused: these are measurements of real files, and
174        // the usual reason they match nothing is that their folder has not
175        // been scanned yet.
176        println!(
177            "  {}",
178            ui::dim("they are stored; scan the folders above and they attach themselves")
179        );
180    }
181    Ok(())
182}

184/// `true` when a waiting record is one the user asked about.
185///
186/// The folders are compared as **strings**, never canonicalized: a waiting
187/// record is precisely one whose path the catalog does not hold, and the
188/// commonest reason is that the folder is not there any more. Asking the
189/// filesystem to resolve it — as `check` legitimately does for folders it is
190/// about to read — would refuse the one scope that matters here.
191fn selected(record: &FileAnalysis, source: Option<&str>, folders: &[String]) -> bool {
192    if source.is_some_and(|s| record.source != s) {
193        return false;
194    }
195    folders.is_empty()
196        || folders
197            .iter()
198            .any(|folder| aede_core::text::is_under(&record.path, folder))
199}

201/// Lists the imported analyses that describe no file the catalog holds,
202/// **grouped by the folder they name**.
203///
204/// `doctor` only ever says how many are waiting, which answers "how many" and
205/// not "which" — and "which" is the whole question, because a count cannot
206/// tell "not scanned yet" from "will never match".
207///
208/// Grouped, and with the folder written out in full, for two reasons that both
209/// came from the listing that did neither. One report of a hundred and forty
210/// tracks is *one* decision — scan that folder, or drop it — and a hundred and
211/// forty rows bury it. And a path cut to fit a column is cut at the wrong end:
212/// `…/1980 Blizzard of Ozz/01 I Don't Know.flac` hides the only part that
213/// distinguishes a drive that is merely unplugged from a folder that was
214/// renamed. The file name identifies a file, but here nobody is looking for a
215/// file — they are looking for the folder to scan, and that is the head of the
216/// path, not its tail. So the column is left unbounded: this is the one
217/// listing whose whole content is the path.
218fn list_pending(catalog: &Catalog, args: &Args) -> Res {
219    let source = args.value("source");
220    let folders = scope(args);
221    let pending: Vec<&FileAnalysis> = catalog
222        .pending_analyses_list()
223        .into_iter()
224        .filter(|a| selected(a, source, &folders))
225        .collect();
226
227    println!("{}", ui::section("Waiting for a scan"));
228    if pending.is_empty() {
229        // Narrowed to nothing and holding nothing are different answers, and
230        // "nothing is waiting" said under a filter is the wrong one: it reads
231        // as a clean catalog when it may only mean the filter missed.
232        let narrowed = source.is_some() || !folders.is_empty();
233        println!(
234            "  {}",
235            match narrowed {
236                true => ui::yellow("nothing waiting matches that"),
237                false => ui::green("nothing is waiting"),
238            }
239        );
240        return Ok(());
241    }
242
243    // One row per folder and per source: the unit a user acts on. Sorted by
244    // folder, so the same catalog lists the same way twice.
245    let mut by_folder: BTreeMap<(&str, &str), usize> = BTreeMap::new();
246    for record in &pending {
247        *by_folder
248            .entry((
249                aede_core::text::folder(&record.path),
250                record.source.as_str(),
251            ))
252            .or_insert(0) += 1;
253    }
254
255    let window = args.window(25)?;
256    let mut t = Table::new(&["Folder", "Analyses", "Source"]).align(1, Align::Right);
257    for ((folder, source), count) in by_folder.iter().skip(window.offset).take(window.limit) {
258        t.push(vec![
259            folder.to_string(),
260            count.to_string(),
261            (*source).to_string(),
262        ]);
263    }
264    print!("{}", t.render());
265    announce_window(window, by_folder.len(), "folder");
266    println!(
267        "  {}",
268        ui::dim(&format!(
269            "{} in all",
270            ui::plural(pending.len(), "waiting analysis")
271        ))
272    );
273    println!(
274        "  {}",
275        ui::dim("scan a folder to attach its analyses, or drop one that is gone for good:")
276    );
277    println!("  {}", ui::dim("aede import --forget --pending <folder>"));
278    Ok(())
279}

283    args.positionals.clone()

286/// What became of the analyses that name one folder.
287#[derive(Default)]
288struct Fate {
289    /// Attached to a file the catalog holds, and still describing its bytes.
290    attached: usize,
291    /// Attached to a file that has changed since the report was written.
292    stale: usize,
293    /// Naming a path the catalog does not hold.
294    waiting: usize,
295}
296
297impl Fate {
298    /// `21 attached`, `19 attached, 2 stale`, `4 waiting`.
299    fn wording(&self) -> String {
300        let mut parts = Vec::new();
301        for (count, word) in [
302            (self.attached, "attached"),
303            (self.stale, "stale"),
304            (self.waiting, "waiting"),
305        ] {
306            if count > 0 {
307                parts.push(format!("{count} {word}"));
308            }
309        }
310        parts.join(", ")
311    }
312}

314/// Lists every analysis held, grouped by the folder it names, and says what
315/// became of each group.
316///
317/// The counterpart of `--pending`, and it was missing: the catalog could say
318/// what had *failed* to attach and nothing at all about what had succeeded. So
319/// a report imported over an artist whose files are clean produced 311 records,
320/// no waiting line, no `doctor` entry — every symptom of having done nothing —
321/// and the only way to see otherwise was to open a track page and hope to land
322/// on a file the report covered. **A store that can only show its failures
323/// cannot be trusted about its successes**, which is the whole reason to look.
324///
325/// Three fates rather than two: attached, waiting, and *stale* — attached to a
326/// file whose bytes have changed since. The third is invisible everywhere else
327/// and is the one that silently voids a verdict.
328fn list_all(catalog: &Catalog, args: &Args) -> Res {
329    let source = args.value("source");
330    let folders = scope(args);
331    let held: BTreeMap<&str, &aede_core::model::AudioFile> =
332        catalog.files.iter().map(|f| (f.path.as_str(), f)).collect();
333
334    let mut by_folder: BTreeMap<(&str, &str), Fate> = BTreeMap::new();
335    let mut total = Fate::default();
336    for record in catalog
337        .analyses
338        .iter()
339        .filter(|a| selected(a, source, &folders))
340    {
341        let fate = by_folder
342            .entry((
343                aede_core::text::folder(&record.path),
344                record.source.as_str(),
345            ))
346            .or_default();
347        let (here, whole) = match held.get(record.path.as_str()) {
348            None => (&mut fate.waiting, &mut total.waiting),
349            Some(file) if record.still_applies(file.size, file.mtime) => {
350                (&mut fate.attached, &mut total.attached)
351            }
352            Some(_) => (&mut fate.stale, &mut total.stale),
353        };
354        *here += 1;
355        *whole += 1;
356    }
357
358    println!("{}", ui::section("Imported analyses"));
359    if by_folder.is_empty() {
360        // Holding nothing and being narrowed to nothing are different answers.
361        let narrowed = source.is_some() || !folders.is_empty();
362        println!(
363            "  {}",
364            match narrowed {
365                true => ui::yellow("nothing imported matches that"),
366                false => ui::dim("nothing imported — aede import <report.json>"),
367            }
368        );
369        return Ok(());
370    }
371
372    let window = args.window(25)?;
373    let mut t = Table::new(&["Folder", "Analyses", "State", "Source"]).align(1, Align::Right);
374    for ((folder, source), fate) in by_folder.iter().skip(window.offset).take(window.limit) {
375        t.push(vec![
376            folder.to_string(),
377            (fate.attached + fate.stale + fate.waiting).to_string(),
378            fate.wording(),
379            (*source).to_string(),
380        ]);
381    }
382    print!("{}", t.render());
383    announce_window(window, by_folder.len(), "folder");
384    println!("  {}", ui::dim(&format!("in all: {}", total.wording())));
385    println!(
386        "  {}",
387        ui::dim("what one of them holds: aede track \"<title>\"")
388    );
389    Ok(())
390}

392/// Removes imported analyses: all of them, one source's, only the ones still
393/// waiting for a scan, or any narrowing of those by folder.
394///
395/// `--pending` is what makes this safe to reach for. A plain `--forget` cannot
396/// drop a report that will never attach without also losing every analysis
397/// that *did* match a file — an hour of somebody else's decoding, and the one
398/// thing in the catalog a re-scan cannot rebuild.
399fn forget(catalog: &mut Catalog, catalog_file: &Path, args: &Args) -> Res {
400    let source = args.value("source");
401    let pending_only = args.has("pending");
402    let folders = scope(args);
403
404    // A folder narrows *which* records are dropped, and only `--pending` gives
405    // it a meaning here — without it, `aede import --forget <folder>` reads
406    // like "import that folder and forget it", which is not a thing. Refused
407    // rather than ignored: a swallowed argument on a destructive command is
408    // the worst place for one.
409    if !folders.is_empty() && !pending_only {
410        return Err(format!(
411            "--forget takes no folder: \"{}\" was ignored.\n\
412             To drop only what is waiting under it: aede import --forget --pending {}",
413            folders.join(" "),
414            folders[0]
415        )
416        .into());
417    }
418
419    let before = catalog.analyses.len();
420    // Collected before the retain: a record only knows it is pending by
421    // comparison with the files the catalog holds right now, and that
422    // comparison cannot be made from inside a closure that is, at the same
423    // time, mutating the very list being compared against.
424    let known: std::collections::BTreeSet<String> =
425        catalog.files.iter().map(|f| f.path.clone()).collect();
426    catalog.analyses.retain(|a| {
427        !(selected(a, source, &folders) && (!pending_only || !known.contains(&a.path)))
428    });
429    let removed = before - catalog.analyses.len();
430    store::save(catalog, catalog_file)?;
431    println!("{}", ui::section("Import"));
432    println!(
433        "  {} removed, {} left",
434        ui::plural(removed, "analysis"),
435        catalog.analyses.len()
436    );
437    // Which catalog, because "311 removed" and a report that still shows them
438    // is a contradiction the user cannot investigate without knowing what was
439    // written. `scan` ends this way for the same reason; a command that
440    // *destroys* something has more need of it, not less. `--data` and
441    // `$AEDE_HOME` both move this file, and the second is easy to forget.
442    println!(
443        "{}",
444        ui::dim(&format!("  catalog: {}", catalog_file.display()))
445    );
446    Ok(())
447}

449/// Gathers the reports named, walking a folder and everything under it.
450///
451/// Recursive, because reports are kept the way the albums they describe are:
452/// one folder per artist, one per album. A walk that only looked at the top
453/// level would find nothing in the folder a user actually points at.
454fn collect_reports(path: &Path, out: &mut Vec<PathBuf>) -> Result<(), Box<dyn std::error::Error>> {
455    if path.is_file() {
456        out.push(path.to_path_buf());
457        return Ok(());
458    }
459    if !path.is_dir() {
460        return Err(format!("\"{}\" does not exist", path.display()).into());
461    }
462    let found = walk_for_reports(path);
463    if found.is_empty() {
464        return Err(format!("no .json report under \"{}\"", path.display()).into());
465    }
466    out.extend(found);
467    Ok(())
468}

470/// Every `.json` file under a folder, sorted.
471///
472/// Sorted so that importing the same folder twice does the same thing twice:
473/// when two reports describe the same file, the same one wins both times.
474fn walk_for_reports(dir: &Path) -> Vec<PathBuf> {
475    let mut found = Vec::new();
476    let mut stack = vec![dir.to_path_buf()];
477    while let Some(current) = stack.pop() {
478        let Ok(entries) = std::fs::read_dir(&current) else {
479            // An unreadable folder must not stop the rest of the walk.
480            continue;
481        };
482        for entry in entries.flatten() {
483            let path = entry.path();
484            let name = entry.file_name().to_string_lossy().to_string();
485            if name.starts_with('.') {
486                continue;
487            }
488            match entry.file_type() {
489                Ok(kind) if kind.is_dir() => stack.push(path),
490                Ok(kind) if kind.is_file() && name.to_ascii_lowercase().ends_with(".json") => {
491                    found.push(path);
492                }
493                _ => {}
494            }
495        }
496    }
497    found.sort();
498    found
