1//! `aede extract` (`aede artwork`): write out the picture your files carry.
2//!
3//! The counterpart of `fetch --covers`, and the one to reach for first. Where
4//! that command downloads a cover for an album that has none anywhere, this one
5//! writes out an image the library **already holds** — so it needs no network,
6//! cannot fetch the wrong edition's artwork, and is instant.
7//!
8//! # It is named for what it does
9//!
10//! `artwork` named the subject and not the act, and a reader who ran it and saw
11//! `nothing to write` above a list of counts took it for a report — something
12//! `doctor` would print — rather than a command that had just declined to write
13//! because there was nothing to write. `extract` is the name now; `artwork`
14//! stays as an alias, so nothing anybody typed before stops working. **A
15//! command that writes must be named for the writing.**
16//!
17//! It exists because of a question that has an unobvious answer: an album can
18//! have no `cover.jpg` in its folder and still not want one, because the
19//! picture is inside the audio files. Players and file managers differ on which
20//! they read, and a library where the artwork is only ever in one of the two
21//! places will look empty in half of them.
22//!
23//! # It works on every format, not on FLAC alone
24//!
25//! Extraction goes through [`aede_core::artwork`], which reads the picture the
26//! same way whatever carries it — a FLAC metadata block, an ID3 `APIC` frame,
27//! an MP4 `covr` atom, a base64-wrapped block inside an Ogg comment. The
28//! reasoning for that single path, rather than four hand-written extractors,
29//! is written on that module.
30//!
31//! # Per folder, not per album
32//!
33//! A double album in `CD1` and `CD2` is one release and two folders, and a
34//! cover image belongs to a folder — that is where every player looks for it.
35//! So this walks folders, and each one that lacks an image gets the picture its
36//! own files carry.
37//!
38//! # `--images`: everything else the files carry
39//!
40//! A tagged file often holds more than the front cover — the back of the
41//! sleeve, the pages of a booklet, the label printed on the disc. Without
42//! `--images` those are ignored, because the cover is the one image anything
43//! else in the library reads. With it, they are written into an `artwork/`
44//! subfolder — **not** beside the music, for the reason set out on
45//! [`aede_core::coverart::EXTRAS`]: any image next to the tracks is taken for
46//! the album's cover, and a back sleeve promoted to cover is worse than none.
47//!
48//! # It never writes into an audio file
49//!
50//! Not with `--images`, not with anything. Extraction reads a file and writes
51//! next to it. Putting a cover *into* files that lack one is the obvious
52//! symmetrical feature and it does not exist here on purpose: a program that
53//! rewrites a music library's audio files is a program that can destroy one.
54

63/// A folder to write into, and the file the pictures come out of.
64struct Target {
65    folder: String,
66    source: String,
67    /// Whether the cover is to be written: `false` when the folder has an image
68    /// already and only the other pictures are wanted.
69    cover: bool,
70}
71
72/// What the command would do, and why it would leave the rest alone.
73struct Survey {
74    targets: Vec<Target>,
75    /// Skipped: an image is already in the folder.
76    has_image: usize,
77    /// Skipped: no file in the folder carries a picture.
78    nothing_inside: usize,
79}
80
81pub fn artwork(args: &Args) -> Res {
82    let catalog = super::load(args)?;
83    let scope = super::scope_of(args, &catalog)?;
84    let images = args.has("images");
85    let survey = survey(&catalog, &scope, images);
86
87    println!("{}", ui::section("Extract"));
88    if survey.targets.is_empty() {
89        println!("  {}", ui::dim("nothing to extract"));
90        skipped(&survey, images);
91        return Ok(());
92    }
93    let covers = survey.targets.iter().filter(|t| t.cover).count();
94    if covers > 0 {
95        // What is about to happen, before it happens. The old wording named a
96        // state and left the reader to infer the action from it.
97        println!(
98            "  {} whose files carry a picture and whose folder has none: \
99             the picture is written into each",
100            ui::plural(covers, "folder")
101        );
102    }
103    if images {
104        println!(
105            "  {} to look through for images other than the cover",
106            ui::plural(survey.targets.len(), "folder")
107        );
108        println!(
109            "  {}",
110            ui::dim(&format!(
111                "--images: the back, the booklet and the rest go into {}/ \
112                 in each folder, so that nothing but the cover sits beside the music",
113                coverart::EXTRAS
114            ))
115        );
116    }
117    skipped(&survey, images);
118
119    if args.has("dry-run") {
120        for target in &survey.targets {
121            println!("  {}", ui::dim(&target.folder));
122        }
123        println!("  {}", ui::dim("nothing was written: --dry-run"));
124        return Ok(());
125    }
126
127    let (mut written, mut extras, mut failed) = (0usize, 0usize, 0usize);
128    for target in &survey.targets {
129        if target.cover {
130            match write_one(target) {
131                Ok(path) => {
132                    written += 1;
133                    println!("  {} {path}", ui::green("→"));
134                }
135                Err(why) => {
136                    failed += 1;
137                    eprintln!("  {} {}: {why}", ui::red("×"), target.folder);
138                }
139            }
140        }
141        if !images {
142            continue;
143        }
144        match write_others(target) {
145            Ok(paths) => {
146                extras += paths.len();
147                for path in paths {
148                    println!("  {} {path}", ui::green("→"));
149                }
150            }
151            Err(why) => {
152                failed += 1;
153                eprintln!("  {} {}: {why}", ui::red("×"), target.folder);
154            }
155        }
156    }
157
158    // `plural` puts the count in the phrase, which keeps the summary out of
159    // the agreement trap the skipped lines have to write both ways around.
160    match images {
161        true => println!(
162            "{} {} and {} extracted, {failed} failed",
163            ui::green("→"),
164            ui::plural(written, "cover"),
165            ui::plural(extras, "other image")
166        ),
167        false => println!(
168            "{} {} extracted, {failed} failed",
169            ui::green("→"),
170            ui::plural(written, "cover")
171        ),
172    }
173    if written > 0 {
174        println!(
175            "  {}",
176            ui::dim("aede scan picks them up — nothing registers a file, the scan finds it")
177        );
178    }
179    Ok(())
180}

182/// Pulls the picture out of one file and writes it beside the music.
183///
184/// The work itself is [`artwork::extract_into`], which is where the guards live
185/// and where real containers are exercised. What is here is the command's
186/// wording for the result.
187fn write_one(target: &Target) -> Result<String, String> {
188    artwork::extract_into(
189        std::path::Path::new(&target.source),
190        std::path::Path::new(&target.folder),
191    )
192    .map(|path| path.display().to_string())
193}

195/// Writes out the pictures that are not the cover, and names the ones it wrote.
196///
197/// A picture already on disk is not in the answer and not a failure: running
198/// this twice over a library should say nothing the second time, not report
199/// every folder as an error for having worked.
200fn write_others(target: &Target) -> Result<Vec<String>, String> {
201    let out = artwork::extract_extras_into(
202        std::path::Path::new(&target.source),
203        std::path::Path::new(&target.folder),
204    )
205    .map_err(|why| why.to_string())?;
206    let mut wrote = Vec::new();
207    for one in out {
208        match one.wrote {
209            Ok(Written::New(path)) => wrote.push(path.display().to_string()),
210            Ok(Written::Already(_)) => {}
211            Err(why) => return Err(format!("{}: {why}", one.kind.stem())),
212        }
213    }
214    Ok(wrote)
215}

217/// Sorts every folder of the library into one of the three.
218///
219/// `images` widens the net: without it a folder that already has a cover is
220/// finished, with it that folder may still hold a booklet nobody has seen.
221///
222/// Unlike `fetch --covers --images`, an `artwork/` folder that is already there
223/// does **not** take a folder out of the list. Nothing here costs a request, so
224/// opening a folder again is free — and it is how a picture added to the files
225/// since the last run gets written out. Nothing is overwritten either way.
226///
227/// The pictures are read out of **one** file per folder — the first that
228/// carries any. A tagger writes the same set into every track of a release, so
229/// reading all of them would be a hundred times the work for the same images,
230/// and the folders where that assumption is wrong are ones where the extra
231/// pictures differ per track, which is not a thing a sleeve does.
232fn survey(catalog: &Catalog, scope: &[String], images: bool) -> Survey {
233    let mut out = Survey {
234        targets: Vec::new(),
235        has_image: 0,
236        nothing_inside: 0,
237    };
238    // Folders in the order the catalog holds their files, so a run reads as a
239    // walk through the library rather than in whatever order a map returns.
240    let mut seen: Vec<&str> = Vec::new();
241    for file in &catalog.files {
242        if !super::in_scope(&file.path, scope) {
243            continue;
244        }
245        let folder = text::folder(&file.path);
246        if folder.is_empty() || seen.contains(&folder) {
247            continue;
248        }
249        seen.push(folder);
250
251        // The disk, not the catalog: `cover_path` answers for a release, this
252        // question is about a folder, and the two differ for a double album.
253        let has_image = scan::cover_in(std::path::Path::new(folder)).is_some();
254        if has_image && !images {
255            out.has_image += 1;
256            continue;
257        }
258        match catalog
259            .files
260            .iter()
261            .find(|f| f.has_embedded_art && text::folder(&f.path) == folder)
262        {
263            Some(source) => out.targets.push(Target {
264                folder: folder.to_string(),
265                source: source.path.clone(),
266                cover: !has_image,
267            }),
268            None => out.nothing_inside += 1,
269        }
270    }
271    out
272}

274/// Says what was left alone and why — the lesson `fetch --covers` had to learn
275/// twice: a filter the reader cannot see is a trap, and "nothing happened" is
276/// exactly when they most need to know which filter caught their album.
277///
278/// Both forms of every sentence are written out because [`ui::plural`] puts the
279/// count *in* the sentence, and a count that opens a sentence takes the verb
280/// with it: "1 folder have an image" is what happens otherwise, and it has now
281/// happened three times in this program.
282fn skipped(survey: &Survey, images: bool) {
283    let already = match images {
284        // With `--images` a folder that has a cover is still worth opening, so
285        // it was not skipped and must not be reported as though it had been.
286        true => (
287            "already has an image in it",
288            "already have an image in them",
289        ),
290        false => (
291            "already has an image in it: --images writes out the back and the booklet too",
292            "already have an image in them: --images writes out the back and the booklet too",
293        ),
294    };
295    super::print_skipped_counts(
296        &[
297            (survey.has_image, already.0, already.1),
298            (
299                survey.nothing_inside,
300                "holds no picture inside its files: aede fetch --covers downloads one",
301                "hold no picture inside their files: aede fetch --covers downloads one",
302            ),
303        ],
304        "folder",
305    );
306}
