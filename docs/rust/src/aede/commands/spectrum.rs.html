1//! The `spectrum` command: a spectrogram beside every track.
2//!
3//! It reads the *catalog*, not the disk, even though it is given folders: the
4//! catalog already knows which files are audio, what they claim to be, and
5//! when they were last read. Walking the filesystem again would be a second
6//! answer to a question already answered, and the two would drift.
7
8use std::path::PathBuf;
9use std::sync::Mutex;
10use std::sync::atomic::{AtomicUsize, Ordering};
11
12use aede_core::model::Catalog;
13use aede_core::{ffmpeg, spectrum};
14
15use super::{Res, load};
16use crate::args::Args;
17use crate::ui::{self, Align, Table};
18
19/// The size used when `--size` is not given.
20///
21/// Half of FlacCompagnon's plotted graph, with a quarter of its plotted pixels.
22/// The final PNG includes legend margins; its encoded size depends on content
23/// and compression. The smaller graph is the default for library-wide runs,
24/// while `--size full` keeps FlacCompagnon's dimensions for comparisons.
25pub const DEFAULT_SIZE: spectrum::Size = spectrum::Size::Half;
26
27pub fn spectrum(args: &Args) -> Res {
28    let catalog = load(args)?;
29    let scope = super::scope_of(args, &catalog)?;
30    let redraw_everything = args.has("full");
31    let dry_run = args.has("dry-run");
32    let size = match args.value("size") {
33        Some(text) => spectrum::Size::parse(text)
34            .ok_or_else(|| format!("--size takes full or half; \"{text}\" is neither"))?,
35        None => DEFAULT_SIZE,
36    };
37
38    let work = to_draw(&catalog, &scope, redraw_everything);
39    println!("{}", ui::section("Spectrograms"));
40    if work.is_empty() {
41        let held = catalog
42            .files
43            .iter()
44            .any(|f| super::in_scope(&f.path, &scope));
45        println!(
46            "  {}",
47            match held {
48                // The ordinary answer on the second run, and it has to read as
49                // a result rather than as an empty screen: nothing was drawn
50                // *because* nothing needed drawing.
51                true => ui::green("every picture is already there and up to date"),
52                false => ui::yellow("no track of the catalog is in that folder"),
53            }
54        );
55        return Ok(());
56    }
57    println!(
58        "  {}",
59        ui::dim(&format!(
60            "pictures are drawn at {} size; changing --size does not redraw \
61             what is already there — --full does",
62            size.as_str()
63        ))
64    );
65
66    if dry_run {
67        let mut t = Table::new(&["Picture", "Track"]);
68        for (audio, picture, _) in work.iter().take(20) {
69            t.push(vec![
70                picture.display().to_string(),
71                aede_core::text::file_name(&audio.display().to_string()).to_string(),
72            ]);
73        }
74        print!("{}", t.render());
75        if work.len() > 20 {
76            println!("{}", ui::dim(&format!("  … and {} more", work.len() - 20)));
77        }
78        println!("  {}", ui::dim(&format!("{} to draw", work.len())));
79        return Ok(());
80    }
81
82    // Resolved once, before the first file: a thousand tracks would otherwise
83    // mean a thousand failed lookups before the first picture.
84    let program = ffmpeg::find().ok_or_else(|| ffmpeg::missing("spectrum"))?;
85
86    // One ffmpeg per track, several at a time. Drawing a spectrogram decodes
87    // the whole file and runs an FFT over it — seconds per track, hours over a
88    // library — and the work is embarrassingly parallel because no two
89    // pictures share anything. The same answer the scan gives to the same
90    // question, down to `--threads` meaning the same thing.
91    let total = work.len();
92    let threads = aede_core::scan::resolve_threads(args.number_or("threads", 0)?).min(total.max(1));
93    let queue = Mutex::new(work);
94    let failures: Mutex<Vec<(String, String)>> = Mutex::new(Vec::new());
95    let done = AtomicUsize::new(0);
96
97    std::thread::scope(|scope| {
98        for _ in 0..threads {
99            scope.spawn(|| {
100                // A poisoned lock must not abandon the rest of the work:
101                // one ffmpeg that panicked is not a reason to stop drawing.
102                while let Some((audio, picture, caption)) = super::take_queued(&queue) {
103                    if let Err(said) =
104                        spectrum::render(&program, &audio, &picture, size, Some(caption.as_str()))
105                    {
106                        failures
107                            .lock()
108                            .unwrap_or_else(|e| e.into_inner())
109                            .push((audio.display().to_string(), said));
110                    }
111                    done.fetch_add(1, Ordering::Relaxed);
112                }
113            });
114        }
115        // The main thread only reports progress, so no picture waits on a
116        // terminal write.
117        let mut last = usize::MAX;
118        while done.load(Ordering::Relaxed) < total {
119            let current = done.load(Ordering::Relaxed);
120            if current != last {
121                last = current;
122                print!("\r  drawing: {current}/{total}   ");
123                let _ = std::io::Write::flush(&mut std::io::stdout());
124            }
125            std::thread::sleep(std::time::Duration::from_millis(80));
126        }
127    });
128    println!("\r  drawing: {total}/{total}   ");
129
130    let failures = failures.into_inner().unwrap_or_else(|e| e.into_inner());
131    let drawn = total - failures.len();
132
133    let mut t = Table::plain(2).align(1, Align::Right);
134    t.push(vec!["Drawn".into(), drawn.to_string()]);
135    if !failures.is_empty() {
136        t.push(vec!["Failed".into(), failures.len().to_string()]);
137    }
138    print!("{}", t.render());
139
140    if !failures.is_empty() {
141        super::print_failure_table("What ffmpeg could not draw", "File", &failures);
142        if failures.len() > 20 {
143            println!(
144                "{}",
145                ui::dim(&format!("  … and {} more", failures.len() - 20))
146            );
147        }
148    }
149    Ok(())
150}

152/// The pictures that have to be drawn: `(track, picture, caption)`.
153///
154/// Everything already drawn and still current is left out here rather than
155/// skipped later, so the count on screen is the work and not the library.
156fn to_draw(
157    catalog: &Catalog,
158    scope: &[String],
159    everything: bool,
160) -> Vec<(PathBuf, PathBuf, String)> {
161    catalog
162        .files
163        .iter()
164        .filter(|file| super::in_scope(&file.path, scope))
165        .filter_map(|file| {
166            let audio = PathBuf::from(&file.path);
167            let picture = spectrum::picture_for(&audio);
168            let wanted = everything || spectrum::out_of_date(&audio, &picture);
169            wanted.then(|| (audio, picture, spectrum::caption(&file.properties)))
170        })
171        .collect()
172}
