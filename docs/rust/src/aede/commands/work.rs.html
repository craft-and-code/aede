11pub fn show_work(args: &Args) -> Res {
12    let catalog = load(args)?;
13    let query = args.positionals.join(" ");
14    if query.trim().is_empty() {
15        return Err("give a work title or MusicBrainz ID".into());
16    }
17    let held = sources::load(&sources::sources_path(&data_dir(args)))?.unwrap_or_default();
18    let family = held.work_parent_links(&catalog);
19    let found = catalog.find_works(&query);
20    let mut sourced = held.find_sourced_works(&catalog, &query);
21    sourced.retain(|work| !found.iter().any(|local| local.mbid == work.mbid));
22    let wanted = text::normalize(&query);
23    let mut parent_only: BTreeMap<String, String> = BTreeMap::new();
24    for link in family.iter().filter(|link| link.trusted) {
25        let title = text::normalize(&link.parent.title);
26        if link.parent.mbid == query
27            || (!wanted.is_empty() && (title == wanted || title.contains(&wanted)))
28        {
29            let title = parent_only.entry(link.parent.mbid.clone()).or_default();
30            if title.is_empty() && !link.parent.title.is_empty() {
31                *title = link.parent.title.clone();
32            }
33        }
34    }
35    parent_only.retain(|mbid, _| {
36        !found.iter().any(|work| &work.mbid == mbid)
37            && !sourced.iter().any(|work| &work.mbid == mbid)
38    });
39    let count = found.len() + sourced.len() + parent_only.len();
40    if count == 0 {
41        return Err(format!("no work matches \"{query}\"").into());
42    }
43    if count > 1 {
44        return Err(format!("{count} works match \"{query}\"; use its MusicBrainz ID").into());
45    }
46
47    if let Some(work) = found.first() {
48        println!("{}", ui::section(&work.title));
49        println!("  {}", ui::dim(&format!("MusicBrainz work: {}", work.mbid)));
50        let recording_ids: BTreeSet<_> = work.recording_ids.iter().copied().collect();
51        print_recordings(&catalog, &recording_ids);
52        let mut navigation = Navigation::default();
53        for &recording_id in &recording_ids {
54            navigation.entity(&catalog, "Recording", EntityKind::Recording, recording_id);
55        }
56        print_family(&family, &work.mbid, &mut navigation);
57        for link in held
58            .work_links(&catalog)
59            .into_iter()
60            .filter(|link| link.work.mbid == work.mbid)
61        {
62            let confidence = super::source_status(link.confidence, link.review, link.trusted);
63            println!(
64                "  {}",
65                ui::dim(&format!(
66                    "source evidence: {} · {} · {}",
67                    link.source,
68                    confidence,
69                    ui::since(link.fetched_at)
70                ))
71            );
72        }
73        let credits: Vec<_> = held
74            .credit_links(&catalog)
75            .into_iter()
76            .filter(|link| {
77                link.work
78                    .as_ref()
79                    .is_some_and(|linked| linked.mbid == work.mbid)
80            })
81            .collect();
82        for link in &credits {
83            if !link.trusted {
84                continue;
85            }
86            navigation.source_artist(&link.credit.artist_mbid);
87        }
88        super::print_sourced_credits(&catalog, credits);
89        super::panel_for(args, &catalog, EntityKind::Work, work.id);
90        navigation.print();
91        return Ok(());
92    }
93
94    if let Some((mbid, title)) = parent_only.into_iter().next() {
95        println!(
96            "{}",
97            ui::section(if title.is_empty() { &mbid } else { &title })
98        );
99        println!("  {}", ui::dim(&format!("MusicBrainz work: {mbid}")));
100        println!(
101            "  {}",
102            ui::dim("external part-of-work evidence; local tags are unchanged")
103        );
104        let mut navigation = Navigation::default();
105        print_family(&family, &mbid, &mut navigation);
106        navigation.print();
107        return Ok(());
108    }
109
110    let work = &sourced[0];
111    let title = if work.title.is_empty() {
112        "Untitled work"
113    } else {
114        &work.title
115    };
116    println!("{}", ui::section(title));
117    println!("  {}", ui::dim(&format!("MusicBrainz work: {}", work.mbid)));
118    println!(
119        "  {}",
120        ui::dim("external evidence; local tags are unchanged")
121    );
122    let recording_ids: BTreeSet<_> = work.links.iter().map(|link| link.recording_id).collect();
123    print_recordings(&catalog, &recording_ids);
124    let mut navigation = Navigation::default();
125    for &recording_id in &recording_ids {
126        navigation.entity(&catalog, "Recording", EntityKind::Recording, recording_id);
127    }
128    print_family(&family, &work.mbid, &mut navigation);
129    for link in &work.links {
130        let attributes = link
131            .work
132            .attributes
133            .iter()
134            .map(|attribute| attribute.name.as_str())
135            .collect::<Vec<_>>();
136        let qualified = match attributes.is_empty() {
137            true => String::new(),
138            false => format!(" · {}", attributes.join(", ")),
139        };
140        println!(
141            "  {}",
142            ui::dim(&format!(
143                "{} · identified · {}{qualified}",
144                link.source,
145                ui::since(link.fetched_at)
146            ))
147        );
148    }
149    let credits: Vec<_> = held
150        .credit_links(&catalog)
151        .into_iter()
152        .filter(|link| {
153            link.work
154                .as_ref()
155                .is_some_and(|linked| linked.mbid == work.mbid)
156        })
157        .collect();
158    for link in &credits {
159        if link.trusted {
160            navigation.source_artist(&link.credit.artist_mbid);
161        }
162    }
163    super::print_sourced_credits(&catalog, credits);
164    navigation.print();
165    Ok(())
166}

168/// Shows only asserted parent/part identities. A matching title never invents
169/// a work relationship, and a rejected/uncertain source is evidence but not a
170/// navigation target.
171fn print_family(
172    family: &[sources::SourcedWorkParentLink],
173    mbid: &str,
174    navigation: &mut Navigation,
175) {
176    let mut parents: Vec<_> = family
177        .iter()
178        .filter(|link| link.child_mbid == mbid)
179        .collect();
180    parents.sort_by(|left, right| {
181        left.parent
182            .mbid
183            .cmp(&right.parent.mbid)
184            .then_with(|| left.source.cmp(&right.source))
185            .then_with(|| right.trusted.cmp(&left.trusted))
186    });
187    parents.dedup_by(|left, right| {
188        left.parent.mbid == right.parent.mbid && left.source == right.source
189    });
190    if !parents.is_empty() {
191        println!("{}", ui::section("Part of"));
192        let mut rows = Table::new(&["Work", "Position", "Evidence"]);
193        for link in parents {
194            let title = if link.parent.title.is_empty() {
195                &link.parent.mbid
196            } else {
197                &link.parent.title
198            };
199            rows.push(vec![
200                title.clone(),
201                part_position(&link.parent),
202                format!(
203                    "{} · {}",
204                    link.source,
205                    super::source_status(link.confidence, link.review, link.trusted)
206                ),
207            ]);
208            if link.trusted {
209                navigation.add(
210                    "Parent work",
211                    format!(
212                        "aede work {}",
213                        super::navigation::shell_arg(&link.parent.mbid)
214                    ),
215                );
216            }
217        }
218        print!("{}", rows.render());
219    }
220
221    let mut children: Vec<_> = family
222        .iter()
223        .filter(|link| link.parent.mbid == mbid)
224        .collect();
225    children.sort_by(|left, right| {
226        left.child_mbid
227            .cmp(&right.child_mbid)
228            .then_with(|| left.source.cmp(&right.source))
229            .then_with(|| right.trusted.cmp(&left.trusted))
230    });
231    children
232        .dedup_by(|left, right| left.child_mbid == right.child_mbid && left.source == right.source);
233    children.sort_by(|left, right| {
234        left.parent
235            .order
236            .unwrap_or(u32::MAX)
237            .cmp(&right.parent.order.unwrap_or(u32::MAX))
238            .then_with(|| left.child_title.cmp(&right.child_title))
239            .then_with(|| left.child_mbid.cmp(&right.child_mbid))
240    });
241    if !children.is_empty() {
242        println!("{}", ui::section("Parts in the library"));
243        let mut rows = Table::new(&["Part", "Position", "Evidence"]);
244        for link in children {
245            rows.push(vec![
246                if link.child_title.is_empty() {
247                    link.child_mbid.clone()
248                } else {
249                    link.child_title.clone()
250                },
251                part_position(&link.parent),
252                format!(
253                    "{} · {}",
254                    link.source,
255                    super::source_status(link.confidence, link.review, link.trusted)
256                ),
257            ]);
258            if link.trusted {
259                navigation.add(
260                    "Part",
261                    format!(
262                        "aede work {}",
263                        super::navigation::shell_arg(&link.child_mbid)
264                    ),
265                );
266            }
267        }
268        print!("{}", rows.render());
269    }
270}

272fn part_position(parent: &sources::WorkParentLink) -> String {
273    let mut parts: Vec<String> = parent
274        .attributes
275        .iter()
276        .map(|attribute| attribute.name.clone())
277        .collect();
278    if let Some(order) = parent.order {
279        parts.push(format!("#{order}"));
280    }
281    if parts.is_empty() {
282        "—".into()
283    } else {
284        parts.join(" · ")
285    }
286}

288fn print_recordings(catalog: &aede_core::model::Catalog, recording_ids: &BTreeSet<u32>) {
289    let mut rows = Table::new(&["Recording", "MusicBrainz ID", "Placements"]);
290    for &recording_id in recording_ids {
291        if let Some(recording) = catalog.recording(recording_id) {
292            rows.push(vec![
293                recording.title.clone(),
294                recording.mbid.clone().unwrap_or_else(|| "—".into()),
295                recording.track_ids.len().to_string(),
296            ]);
297        }
298    }
299    println!("{}", rows.render());
300}
