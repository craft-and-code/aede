1//! Online fallback for a label without a Wikipedia biography.
2//!
3//! MusicBrainz supplies the Discogs identifier as a URL relationship. Discogs
4//! is never searched by name. Failed refreshes keep the last stored profile.
5
6use aede_core::json::Json;
7use aede_core::model::{Catalog, EntityKind, Id};
8use aede_core::sources::{self, Facts, LabelFacts, SourceRecord, Sources};
9use aede_core::user::EntityRef;
10use aede_core::{clock, discogs, musicbrainz, text};
11
12use super::Res;
13
14#[cfg(feature = "fetch")]
15pub fn refresh_online(
16    catalog: &Catalog,
17    label: Id,
18    held: &mut Sources,
19    path: &std::path::Path,
20) -> Res {
21    use aede_core::http::Client;
22
23    let agent = Client::identify(
24        "Aede",
25        env!("CARGO_PKG_VERSION"),
26        env!("CARGO_PKG_REPOSITORY"),
27    );
28    let mut mb = Client::new(agent.clone(), musicbrainz::REQUEST_INTERVAL);
29    let mut dc = Client::new(agent, discogs::REQUEST_INTERVAL);
30    refresh(
31        catalog,
32        label,
33        held,
34        path,
35        &mut |url| {
36            crate::ui::with_loading("Checking MusicBrainz...", || {
37                mb.get_json(url).map_err(|error| error.to_string())
38            })
39        },
40        &mut |url| {
41            crate::ui::with_loading("Loading Discogs data...", || {
42                dc.get_json(url).map_err(|error| error.to_string())
43            })
44        },
45    )
46}

58#[cfg_attr(not(feature = "fetch"), allow(dead_code))]
59fn refresh(
60    catalog: &Catalog,
61    label_id: Id,
62    held: &mut Sources,
63    path: &std::path::Path,
64    ask_musicbrainz: &mut dyn FnMut(&str) -> Result<Json, String>,
65    ask_discogs: &mut dyn FnMut(&str) -> Result<Json, String>,
66) -> Res {
67    let label = catalog
68        .label(label_id)
69        .ok_or("label is no longer in the catalog")?;
70    let entity = EntityRef::of(catalog, EntityKind::Label, label_id)
71        .ok_or("label has no stable source key")?;
72
73    if matches!(held.get(&entity, "wikipedia").map(|row| &row.facts),
74        Some(Facts::Label(facts)) if facts.summary.is_some())
75    {
76        println!("  Wikipedia already has a biography for this label; Discogs was not asked.");
77        return Ok(());
78    }
79    let previous = held.get(&entity, sources::MUSICBRAINZ).cloned();
80    let confidence = previous
81        .as_ref()
82        .map(|record| record.confidence)
83        .unwrap_or(sources::Confidence::Identified);
84    let mbid = previous
85        .as_ref()
86        .and_then(|row| row.source_id.clone())
87        .or_else(|| label.mbid.clone())
88        .ok_or("this label has no MusicBrainz ID; run aede fetch --labels first")?;
89    let mut link = previous.as_ref().and_then(|row| match &row.facts {
90        Facts::Label(facts) => facts.discogs.clone(),
91        _ => None,
92    });
93    if link.is_none() {
94        let url = format!(
95            "{}/label/{mbid}?fmt=json&inc={}",
96            musicbrainz::WEB_SERVICE,
97            musicbrainz::LABEL_INCLUDES
98        );
99        let answer = ask_musicbrainz(&url).map_err(|why| format!("MusicBrainz: {why}"))?;
100        let candidate = musicbrainz::label(&answer)
101            .filter(|candidate| {
102                candidate.mbid == mbid
103                    && text::normalize(&candidate.name) == text::normalize(&label.name)
104            })
105            .ok_or("MusicBrainz did not confirm this label's identity")?;
106        link = candidate.facts.discogs;
107        if let Some(url) = &link {
108            let mut record = previous.unwrap_or(SourceRecord {
109                key: entity.key.clone(),
110                source: sources::MUSICBRAINZ.to_string(),
111                source_id: Some(mbid),
112                fetched_at: clock::now_seconds(),
113                confidence: sources::Confidence::Identified,
114                facts: Facts::Label(LabelFacts::default()),
115            });
116            if let Facts::Label(facts) = &mut record.facts {
117                facts.discogs = Some(url.clone());
118            }
119            held.set(record);
120            sources::save(held, path)?;
121        }
122    }
123    let url = link.ok_or("MusicBrainz has no Discogs link for this label")?;
124    let id = discogs::label_id(&url).ok_or("MusicBrainz gave an unusable Discogs label link")?;
125    let answer = ask_discogs(&discogs::api_url(id)).map_err(|why| format!("Discogs: {why}"))?;
126    let mut prose = discogs::profile(&answer, id, &label.name)
127        .ok_or("Discogs returned no profile matching this label")?;
128    let markup = prose.text.clone();
129    let cached_text = held.get(&entity, discogs::SOURCE).and_then(|record| {
130        let Facts::Label(facts) = &record.facts else {
131            return None;
132        };
133        (facts.summary_markup.as_deref() == Some(markup.as_str())
134            && facts.summary.as_ref().is_some_and(|summary| {
135                discogs::referenced_labels(&summary.text).is_empty()
136                    && discogs::referenced_artists(&summary.text).is_empty()
137            }))
138        .then(|| facts.summary.as_ref().map(|summary| summary.text.clone()))
139        .flatten()
140    });
141    prose.text = if let Some(text) = cached_text {
142        text
143    } else {
144        let label_ids = discogs::referenced_labels(&markup);
145        let artist_ids = discogs::referenced_artists(&markup);
146        let names_to_resolve = label_ids.len() + artist_ids.len();
147        if names_to_resolve > 0 {
148            println!("  Resolving {names_to_resolve} Discogs names...");
149        }
150        let mut labels = std::collections::BTreeMap::new();
151        for linked_id in label_ids {
152            let answer = ask_discogs(&discogs::api_url(linked_id))
153                .map_err(|why| format!("Discogs linked label {linked_id}: {why}"))?;
154            let name = discogs::label_name(&answer, linked_id)
155                .ok_or_else(|| format!("Discogs did not identify linked label {linked_id}"))?;
156            labels.insert(linked_id, name);
157        }
158        let mut artists = std::collections::BTreeMap::new();
159        for artist_id in artist_ids {
160            let answer = ask_discogs(&discogs::artist_api_url(artist_id))
161                .map_err(|why| format!("Discogs linked artist {artist_id}: {why}"))?;
162            let name = discogs::artist_name(&answer, artist_id)
163                .ok_or_else(|| format!("Discogs did not identify linked artist {artist_id}"))?;
164            artists.insert(artist_id, name);
165        }
166        discogs::render_profile(&markup, &labels, &artists)
167    };
168    held.set(SourceRecord {
169        key: entity.key,
170        source: discogs::SOURCE.to_string(),
171        source_id: Some(id.to_string()),
172        fetched_at: clock::now_seconds(),
173        confidence,
174        facts: Facts::Label(LabelFacts {
175            summary: Some(prose),
176            summary_markup: Some(markup),
177            ..Default::default()
178        }),
179    });
180    sources::save(held, path)?;
181    Ok(())
182}
