45struct PendingTrack {
46    index: usize,
47    path: PathBuf,
48    started: u64,
49    active_started_ms: u64,
50    format: PcmFormat,
51    frames: u64,
52    submitted_bytes: u64,
53    complete: bool,
54    meter: OutputMeter,
55    loudness: Option<gain_plan::LoudnessUpdate>,
56    offset_ms: u64,
57    played_before: Duration,
58    had_seek: bool,
59}
60
61impl PendingTrack {
62    fn submitted_duration(&self) -> Duration {
63        let rate = u64::from(self.format.sample_rate());
64        Duration::from_secs(self.frames / rate)
65            + Duration::from_nanos((self.frames % rate) * 1_000_000_000 / rate)
66    }
67
68    fn interrupted_duration(&self, clock: &PlaybackClock) -> Duration {
69        self.submitted_duration().min(Duration::from_millis(
70            clock.active_ms().saturating_sub(self.active_started_ms),
71        ))
72    }
73
74    fn interrupted_ms(&self, clock: &PlaybackClock) -> u64 {
75        milliseconds(self.interrupted_duration(clock))
76    }
77}

90struct PlaybackRecords<'a> {
91    pending: BTreeMap<usize, PendingTrack>,
92    sender: &'a mpsc::SyncSender<PlaybackRecord>,
93    next_offset_ms: u64,
94    resume: Option<ResumeListen>,
95    last_position: Option<(usize, u64)>,
96}
97
98impl<'a> PlaybackRecords<'a> {
99    fn new(sender: &'a mpsc::SyncSender<PlaybackRecord>) -> Self {
100        Self {
101            pending: BTreeMap::new(),
102            sender,
103            next_offset_ms: 0,
104            resume: None,
105            last_position: None,
106        }
107    }
108
109    fn begin(
110        &mut self,
111        token: usize,
112        index: usize,
113        path: &Path,
114        format: PcmFormat,
115        clock: &PlaybackClock,
116    ) -> Res {
117        if self
118            .resume
119            .as_ref()
120            .is_some_and(|listen| listen.index != index)
121        {
122            return Err("seek listening identity does not match the next source".into());
123        }
124        let meter = match OutputMeter::new(format) {
125            Ok(meter) => meter,
126            Err(error) => {
127                eprintln!("True-peak meter unavailable: {error}");
128                OutputMeter::sample_peak_only(format)
129            }
130        };
131        let resumed = self.resume.take();
132        let offset_ms = self.next_offset_ms;
133        self.next_offset_ms = 0;
134        self.pending.insert(
135            token,
136            PendingTrack {
137                index,
138                path: path.to_path_buf(),
139                started: resumed
140                    .as_ref()
141                    .map_or_else(clock::now_seconds, |listen| listen.started),
142                active_started_ms: clock.active_ms(),
143                format,
144                frames: 0,
145                submitted_bytes: 0,
146                complete: false,
147                meter,
148                loudness: None,
149                offset_ms,
150                played_before: resumed
151                    .as_ref()
152                    .map_or(Duration::ZERO, |listen| listen.played),
153                had_seek: resumed.is_some() || offset_ms > 0,
154            },
155        );
156        Ok(())
157    }
158
159    fn source_finished(&mut self, token: usize, update: Option<gain_plan::LoudnessUpdate>) -> Res {
160        self.pending
161            .get_mut(&token)
162            .ok_or("playback track record is missing")?
163            .loudness = update;
164        Ok(())
165    }
166
167    fn submitted(&mut self, event: Submitted<'_>) -> Res {
168        match event {
169            Submitted::Bytes { token, count } => {
170                let record = self
171                    .pending
172                    .get_mut(&token)
173                    .ok_or("output refers to an unknown playback track")?;
174                record.submitted_bytes = record.submitted_bytes.saturating_add(count as u64);
175                record.frames = record.submitted_bytes / (u64::from(record.format.channels()) * 4);
176                Ok(())
177            }
178            Submitted::Complete { span, samples } => self.accepted(span, samples),
179        }
180    }
181
182    fn accepted(&mut self, span: &TrackSpan, samples: &[f32]) -> Res {
183        let record = self
184            .pending
185            .get_mut(&span.token)
186            .ok_or("output refers to an unknown playback track")?;
187        if !samples.is_empty() {
188            match record.meter.observe(samples, span.stats) {
189                Ok(()) => {}
190                Err(OutputMeterError::Meter(error)) => {
191                    eprintln!("True-peak meter stopped: {error}")
192                }
193                Err(error) => return Err(error.into()),
194            }
195        }
196        record.complete |= span.complete;
197        // The last listen stays pending until the output has actually drained,
198        // preserving Stop/Next's incomplete-history behavior at the final tail.
199        if record.complete && self.pending.keys().next_back().copied() != Some(span.token) {
200            self.publish(span.token, true, None)?;
201        }
202        Ok(())
203    }
204
205    fn position_ms(&self, index: usize, clock: &PlaybackClock) -> u64 {
206        self.pending
207            .values()
208            .rev()
209            .find(|record| record.index == index)
210            .map(|record| {
211                record
212                    .offset_ms
213                    .saturating_add(record.interrupted_ms(clock))
214            })
215            .or_else(|| {
216                self.last_position
217                    .filter(|(last, _)| *last == index)
218                    .map(|(_, ms)| ms)
219            })
220            .unwrap_or(0)
221    }
222
223    fn publish(&mut self, token: usize, completed: bool, clock: Option<&PlaybackClock>) -> Res {
224        let mut record = self
225            .pending
226            .remove(&token)
227            .ok_or("playback track record is missing")?;
228        let played = clock.map_or_else(
229            || record.submitted_duration(),
230            |clock| record.interrupted_duration(clock),
231        );
232        self.last_position = Some((
233            record.index,
234            record.offset_ms.saturating_add(milliseconds(played)),
235        ));
236        let played_ms = milliseconds(played.saturating_add(record.played_before));
237        let completed = completed && !record.had_seek;
238        let measured_frames = record.meter.sample_peak_snapshot().frames;
239        report_meter(&mut record.meter);
240        if measured_frames < record.frames {
241            eprintln!(
242                "Signal diagnostics: {} frames in a partially submitted output block have no complete peak/guard statistics",
243                record.frames - measured_frames,
244            );
245        }
246        if played_ms > 0 {
247            self.sender
248                .send(PlaybackRecord::History(HistoryItem {
249                    path: record.path,
250                    started: record.started,
251                    played_ms,
252                    completed,
253                }))
254                .map_err(|_| "listening history worker stopped")?;
255        }
256        if completed && let Some(update) = record.loudness {
257            self.sender
258                .send(PlaybackRecord::Loudness(update))
259                .map_err(|_| "playback record worker stopped")?;
260        }
261        Ok(())
262    }
263
264    fn publish_completed(&mut self) -> Res {
265        let tokens = self
266            .pending
267            .iter()
268            .filter_map(|(&token, record)| record.complete.then_some(token))
269            .collect::<Vec<_>>();
270        for token in tokens {
271            self.publish(token, true, None)?;
272        }
273        Ok(())
274    }
275
276    fn carry_seek(&mut self, index: usize, clock: &PlaybackClock) -> Res {
277        let current = self
278            .pending
279            .iter()
280            .rev()
281            .find_map(|(&token, record)| (record.index == index).then_some(token));
282        if let Some(token) = current {
283            let mut record = self
284                .pending
285                .remove(&token)
286                .ok_or("seek listening record is missing")?;
287            report_meter(&mut record.meter);
288            let played = record
289                .played_before
290                .saturating_add(record.interrupted_duration(clock));
291            self.last_position = Some((
292                index,
293                record
294                    .offset_ms
295                    .saturating_add(record.interrupted_ms(clock)),
296            ));
297            self.resume = Some(ResumeListen {
298                index,
299                path: record.path,
300                started: record.started,
301                played,
302            });
303        }
304        let tokens = self.pending.keys().copied().collect::<Vec<_>>();
305        for token in tokens {
306            let complete = self
307                .pending
308                .get(&token)
309                .is_some_and(|record| record.complete);
310            self.publish(token, complete, (!complete).then_some(clock))?;
311        }
312        Ok(())
313    }
314
315    fn finish(&mut self, completed: bool, clock: &PlaybackClock) -> Res {
316        let tokens = self.pending.keys().copied().collect::<Vec<_>>();
317        for token in tokens {
318            let natural = completed
319                && self
320                    .pending
321                    .get(&token)
322                    .is_some_and(|record| record.complete);
323            self.publish(token, natural, (!natural).then_some(clock))?;
324        }
325        self.finish_resume()
326    }
327
328    fn finish_resume(&mut self) -> Res {
329        if let Some(listen) = self.resume.take()
330            && milliseconds(listen.played) > 0
331        {
332            self.sender
333                .send(PlaybackRecord::History(HistoryItem {
334                    path: listen.path,
335                    started: listen.started,
336                    played_ms: milliseconds(listen.played),
337                    completed: false,
338                }))
339                .map_err(|_| "listening history worker stopped")?;
340        }
341        Ok(())
342    }
343}

345pub(super) enum Submitted<'a> {
346    // Byte progress preserves complete-frame history even if a later write
347    // fails halfway through a span. Peak/guard stats require the entire span.
348    Bytes {
349        token: usize,
350        count: usize,
351    },
352    Complete {
353        span: &'a TrackSpan,
354        samples: &'a [f32],
355    },
356}

358/// Submit attributed portions, reporting byte progress immediately and only
359/// publishing a span's aggregate meter statistics once it is fully accepted.
360pub(super) fn submit_block(
361    block: SessionBlock<'_>,
362    output: &mut impl PlaybackOutput,
363    controls: Option<&Controls>,
364    clock: &mut PlaybackClock,
365    mut submitted: impl FnMut(Submitted<'_>) -> Res,
366) -> Result<PlaybackEnd, Box<dyn Error>> {
367    for span in block.spans {
368        let samples = &block.samples[span.samples.clone()];
369        let bytes = &block.f32le[span.samples.start * 4..span.samples.end * 4];
370        let mut byte_offset = 0;
371        while byte_offset < bytes.len() {
372            if let Some(action) = control_action(controls, output, clock)? {
373                return Ok(action);
374            }
375            match output.write_pcm(samples, bytes, byte_offset) {
376                Ok(0) => return Err("audio output closed before accepting PCM".into()),
377                Ok(count) if count <= bytes.len() - byte_offset => {
378                    let active_ms = clock.active_ms();
379                    if let Some(timeline) = &mut clock.timeline {
380                        timeline.submitted(span.token, count, active_ms)?;
381                    }
382                    submitted(Submitted::Bytes {
383                        token: span.token,
384                        count,
385                    })?;
386                    clock.observe_visualizer(samples, byte_offset, byte_offset + count);
387                    byte_offset += count;
388                }
389                Ok(_) => return Err("audio output reported an invalid write length".into()),
390                Err(error) if error.kind() == std::io::ErrorKind::Interrupted => {}
391                Err(error) if error.kind() == std::io::ErrorKind::WouldBlock => {
392                    std::thread::sleep(Duration::from_millis(5));
393                }
394                Err(error) => return Err(format!("audio output closed: {error}").into()),
395            }
396        }
397        // Completed occurrence callbacks can publish signal diagnostics. Clear
398        // the temporary drawing first so its next redraw cannot erase them.
399        if span.complete {
400            clock.clear_visualizer();
401        }
402        submitted(Submitted::Complete { span, samples })?;
403        if span.complete
404            && let Some(timeline) = &mut clock.timeline
405        {
406            timeline.completed(span.token)?;
407        }
408    }
409    Ok(PlaybackEnd::Natural)
410}

412fn flush_group(
413    processing: &mut Option<PcmSession>,
414    records: &mut PlaybackRecords<'_>,
415    output: &mut impl PlaybackOutput,
416    controls: Option<&Controls>,
417    clock: &mut PlaybackClock,
418) -> Result<PlaybackEnd, Box<dyn Error>> {
419    let Some(session) = processing.as_mut() else {
420        return Ok(PlaybackEnd::Natural);
421    };
422    let result = submit_block(session.finish()?, output, controls, clock, |event| {
423        records.submitted(event)
424    })?;
425    *processing = None;
426    Ok(result)
427}

443fn drain_output_with<O: SessionOutput>(
444    output: &mut O,
445    clock: &mut PlaybackClock,
446    mut control: impl FnMut(&O, &mut PlaybackClock) -> Result<Option<PlaybackEnd>, Box<dyn Error>>,
447    mut active_ms: impl FnMut(&PlaybackClock) -> u64,
448    mut wait: impl FnMut(),
449) -> Result<PlaybackEnd, Box<dyn Error>> {
450    output.close_input();
451    let mut drain = DrainWait::new(active_ms(clock));
452    loop {
453        if let Some(action) = control(output, clock)? {
454            return Ok(action);
455        }
456        if drain.advance(
457            active_ms(clock),
458            output.drain_progress()?,
459            output.host_tail(),
460        )? {
461            output.finish()?;
462            return Ok(PlaybackEnd::Natural);
463        }
464        wait();
465    }
466}

468// A stalled native callback must not trap noninteractive playback forever.
469// This is a progress timeout, independent of the 500 ms queue capacity and
470// deliberately tolerant of coarse host callbacks. Pauses use no active time.
471const DRAIN_STALL_MILLISECONDS: u64 = 5_000;
472
473struct DrainWait {
474    last_consumed: Option<u64>,
475    last_progress_ms: u64,
476    tail_started_ms: Option<u64>,
477}
478
479impl DrainWait {
480    fn new(active_ms: u64) -> Self {
481        Self {
482            last_consumed: None,
483            last_progress_ms: active_ms,
484            tail_started_ms: None,
485        }
486    }
487
488    fn advance(
489        &mut self,
490        active_ms: u64,
491        progress: output::DrainProgress,
492        host_tail: Duration,
493    ) -> Result<bool, Box<dyn Error>> {
494        if progress.drained {
495            let tail_started = *self.tail_started_ms.get_or_insert(active_ms);
496            return Ok(u128::from(active_ms.saturating_sub(tail_started)) >= host_tail.as_millis());
497        }
498        self.tail_started_ms = None;
499        // ffplay cannot report consumed frames: a timeout with no progress
500        // evidence would wrongly assume its buffered audio has stopped.
501        if let Some(consumed) = progress.consumed_frames {
502            if self.last_consumed != Some(consumed) {
503                self.last_consumed = Some(consumed);
504                self.last_progress_ms = active_ms;
505            } else if active_ms.saturating_sub(self.last_progress_ms) >= DRAIN_STALL_MILLISECONDS {
506                return Err(
507                    "native output drain stalled: no consumed-frame progress for 5 seconds".into(),
508                );
509            }
510        }
511        Ok(false)
512    }
513}

515fn prepare_output_with<O: SessionOutput>(
516    output: &mut O,
517    format: PcmFormat,
518    clock: &mut PlaybackClock,
519    control: impl FnMut(&O, &mut PlaybackClock) -> Result<Option<PlaybackEnd>, Box<dyn Error>>,
520    active_ms: impl FnMut(&PlaybackClock) -> u64,
521    wait: impl FnMut(),
522) -> Result<PreparedOutput, Box<dyn Error>> {
523    if output.needs_reopen(format)? {
524        let end = drain_output_with(output, clock, control, active_ms, wait)?;
525        if end != PlaybackEnd::Natural {
526            return Ok(PreparedOutput::Interrupted(end));
527        }
528        if let Some(lyrics) = &mut clock.lyrics {
529            lyrics.reset_output();
530        }
531        if let Some(timeline) = &mut clock.timeline {
532            timeline.reset_output();
533        }
534        clock.visualizer = None;
535    }
536    output.prepare(format).map(PreparedOutput::Ready)
537}

546pub(super) fn play_selection(
547    paths: &[PathBuf],
548    catalog: Option<&Catalog>,
549    normalization: &mut ReadyNormalization<'_>,
550    controls: Option<&Controls>,
551    output: &mut LocalOutput,
552    sender: &mpsc::SyncSender<PlaybackRecord>,
553    settings: SelectionSettings<'_>,
554) -> Res {
555    let SelectionSettings {
556        normalization_mode,
557        tone,
558        order,
559        options,
560    } = settings;
561    let mut records = PlaybackRecords::new(sender);
562    let mut processing: Option<PcmSession> = None;
563    let mut playback_clock = PlaybackClock::new();
564    playback_clock.repeat = options.repeat;
565    playback_clock.shuffle = options.shuffle;
566    playback_clock.smart_available = catalog.is_some();
567    playback_clock.lyrics = options.lyrics.then(lyrics::PlaybackLyrics::new);
568    playback_clock.timeline = crate::ui::is_interactive().then(timeline::PlaybackTimeline::new);
569    // Repeats reuse display metadata within this invocation; successful EOF
570    // replaces the estimate with the source's decoded duration.
571    let mut durations = std::collections::HashMap::new();
572    // Labels are stable within this invocation; one scan avoids catalog-sized
573    // searches during every repeat and speculative source preparation.
574    let labels = playing_labels(paths, catalog);
575    // Index catalog evidence once, restricted to this selection. Repeated
576    // occurrences must not rescan the whole catalog to read their lyrics.
577    let lyric_files = if options.lyrics {
578        let selected: std::collections::HashSet<_> = paths.iter().map(PathBuf::as_path).collect();
579        catalog.map(|catalog| {
580            catalog
581                .files
582                .iter()
583                .filter(|file| selected.contains(Path::new(&file.path)))
584                .map(|file| (Path::new(&file.path), file))
585                .collect::<std::collections::HashMap<_, _>>()
586        })
587    } else {
588        None
589    };
590    let mut cursor = order.current().unwrap_or(0);
591    let mut seek_ms = options.seek_ms;
592    let mut initial_seek_pending = true;
593    let mut token = 0usize;
594    let result = (|| -> Res {
595        while let Some(index) = order.current() {
596            playback_clock.clear_visualizer();
597            order.sync(&playback_clock, catalog, paths)?;
598            let path = &paths[index];
599            let mut seeking = false;
600            let mut seek_reached_eof = false;
601            let end = (|| -> Result<PlaybackEnd, Box<dyn Error>> {
602                if let Some(action) = control_action(controls, output, &mut playback_clock)? {
603                    return Ok(action);
604                }
605                // Extremely short sources can fill the display lookahead before
606                // the SRC emits its delayed frames. Flush that processing tail
607                // so waiting for consumption cannot starve the converter.
608                if playback_clock
609                    .timeline
610                    .as_ref()
611                    .is_some_and(|timeline| !timeline.has_capacity())
612                {
613                    let end = flush_group(
614                        &mut processing,
615                        &mut records,
616                        output,
617                        controls,
618                        &mut playback_clock,
619                    )?;
620                    if end != PlaybackEnd::Natural {
621                        return Ok(end);
622                    }
623                }
624                while playback_clock
625                    .timeline
626                    .as_ref()
627                    .is_some_and(|timeline| !timeline.has_capacity())
628                {
629                    if let Some(action) = control_action(controls, output, &mut playback_clock)? {
630                        return Ok(action);
631                    }
632                    std::thread::sleep(Duration::from_millis(5));
633                }
634                let prepared = (|| -> Result<_, Box<dyn Error>> {
635                    let gain = normalization.prepare(index)?;
636                    let track = PcmTrack::open_stereo(path)?;
637                    Ok((gain, track))
638                })();
639                let (selected_gain, mut track) = match prepared {
640                    Ok(prepared) => prepared,
641                    Err(error) => {
642                        // Preserve a preceding natural source's filter tail even
643                        // when preparing the following file fails.
644                        let end = flush_group(
645                            &mut processing,
646                            &mut records,
647                            output,
648                            controls,
649                            &mut playback_clock,
650                        )?;
651                        let end = if end == PlaybackEnd::Natural {
652                            drain_output(output, controls, &mut playback_clock)?
653                        } else {
654                            end
655                        };
656                        if end == PlaybackEnd::Natural {
657                            playback_clock.clear_visualizer();
658                            records.finish(true, &playback_clock)?;
659                            return Err(error);
660                        }
661                        playback_clock.clear_visualizer();
662                        eprintln!("Warning: could not prepare the next track: {error}");
663                        return Ok(end);
664                    }
665                };
666                if seek_ms > 0 {
667                    seeking = true;
668                    normalization.finish_track(index, false)?;
669                    let mut interrupted = None;
670                    let seek = track.seek_from_start(Duration::from_millis(seek_ms), || {
671                        match control_action(controls, output, &mut playback_clock) {
672                            Ok(None) => false,
673                            result => {
674                                interrupted = Some(result);
675                                true
676                            }
677                        }
678                    });
679                    if let Some(interrupted) = interrupted {
680                        return interrupted?
681                            .ok_or_else(|| "seek interruption has no action".into());
682                    }
683                    let seek = seek?;
684                    seek_ms = seek.frames.saturating_mul(1000)
685                        / u64::from(track.source_format().sample_rate());
686                    println!("Position: {}.{:03} s", seek_ms / 1000, seek_ms % 1000);
687                    if seek.reached_eof {
688                        cursor = index;
689                        initial_seek_pending = false;
690                        seek_reached_eof = true;
691                        order.activated();
692                        records.finish_resume()?;
693                        return Ok(PlaybackEnd::Natural);
694                    }
695                }
696                seeking = false;
697                let input_format = track.format();
698                if processing
699                    .as_ref()
700                    .is_some_and(|session| session.input_format() != input_format)
701                {
702                    let end = flush_group(
703                        &mut processing,
704                        &mut records,
705                        output,
706                        controls,
707                        &mut playback_clock,
708                    )?;
709                    if end != PlaybackEnd::Natural {
710                        return Ok(end);
711                    }
712                }
713                let format = if let Some(session) = &processing {
714                    session.output_format()
715                } else {
716                    let format = match prepare_output_with(
717                        output,
718                        input_format,
719                        &mut playback_clock,
720                        |output, clock| control_action(controls, output, clock),
721                        PlaybackClock::active_ms,
722                        || std::thread::sleep(Duration::from_millis(25)),
723                    )? {
724                        PreparedOutput::Ready(format) => format,
725                        PreparedOutput::Interrupted(end) => return Ok(end),
726                    };
727                    processing = Some(PcmSession::new(input_format, format.sample_rate(), tone)?);
728                    format
729                };
730                playback_clock.clear_visualizer();
731                if playback_clock.visualizer.is_none() {
732                    playback_clock.visualizer = TerminalVisualizer::new(format, !options.lyrics);
733                }
734                let settings = PlaybackSettings {
735                    normalization_mode,
736                    selected_gain,
737                    tone,
738                };
739                let gain_db = describe_track(
740                    &track,
741                    format,
742                    output,
743                    &settings,
744                    normalization.is_measuring(),
745                    &labels[index],
746                    !options.lyrics && playback_clock.visualizer.is_some(),
747                )?;
748                token = token
749                    .checked_add(1)
750                    .ok_or("playback track token exhausted")?;
751                cursor = index;
752                records.publish_completed()?;
753                records.next_offset_ms = seek_ms;
754                records.begin(token, index, path, format, &playback_clock)?;
755                if let Some(timeline) = &mut playback_clock.timeline {
756                    let duration_ms =
757                        *durations
758                            .entry(index)
759                            .or_insert_with(|| match tags::read(path) {
760                                Ok(tags) => tags.properties.duration_ms,
761                                Err(error) => {
762                                    eprintln!("Playback duration unavailable: {error}");
763                                    None
764                                }
765                            });
766                    timeline.begin(token, format, seek_ms, duration_ms, &labels[index])?;
767                }
768                if let Some(lyrics) = &mut playback_clock.lyrics {
769                    let current = catalog
770                        .zip(lyric_files.as_ref())
771                        .and_then(|(catalog, files)| {
772                            files.get(path.as_path()).and_then(|file| {
773                                catalog.file_mtime_subseconds.get(&file.path).map(
774                                    |&mtime_subseconds| aede_core::lyrics::CurrentTrack {
775                                        path,
776                                        size: file.size,
777                                        mtime: file.mtime,
778                                        mtime_subseconds,
779                                        tag: file.first_tag("lyrics"),
780                                        sidecar: file.lyrics_path.as_deref().map(Path::new),
781                                    },
782                                )
783                            })
784                        });
785                    lyrics.begin(token, path, current)?;
786                }
787                initial_seek_pending = false;
788                order.activated();
789                order.describe_transition(&labels);
790                let session = processing
791                    .as_mut()
792                    .ok_or("PCM processing session is missing")?;
793                session.begin_track(token, gain_db)?;
794                let mut source_frames = 0u64;
795                loop {
796                    if let Some(action) = control_action(controls, output, &mut playback_clock)? {
797                        return Ok(action);
798                    }
799                    let raw = track.read_block_observed(
800                        |format, samples| {
801                            if let Err(error) = normalization.observe_source(format, samples) {
802                                playback_clock.clear_visualizer();
803                                eprintln!("Source loudness meter stopped: {error}");
804                            }
805                        },
806                        |_| Ok::<(), std::convert::Infallible>(()),
807                    )?;
808                    let end = if let Some(raw) = raw {
809                        source_frames = source_frames.saturating_add(raw.frames as u64);
810                        submit_block(
811                            session.push_source(raw.samples)?,
812                            output,
813                            controls,
814                            &mut playback_clock,
815                            |event| records.submitted(event),
816                        )?
817                    } else {
818                        order.sync(&playback_clock, catalog, paths)?;
819                        if source_frames == 0 && seek_ms == 0 && order.repeats() {
820                            return Err("cannot repeat an empty audio track".into());
821                        }
822                        let update = match normalization.finish_track(index, seek_ms == 0) {
823                            Ok(update) => update,
824                            Err(error) => {
825                                playback_clock.clear_visualizer();
826                                eprintln!("Warning: source loudness was not cached: {error}");
827                                None
828                            }
829                        };
830                        records.source_finished(token, update)?;
831                        if let Some(timeline) = &mut playback_clock.timeline {
832                            let duration = u128::from(source_frames) * 1000
833                                / u128::from(track.source_format().sample_rate());
834                            let duration_ms =
835                                seek_ms.saturating_add(u64::try_from(duration).unwrap_or(u64::MAX));
836                            timeline.set_duration(token, duration_ms)?;
837                            durations.insert(index, Some(duration_ms));
838                        }
839                        return submit_block(
840                            session.end_track()?,
841                            output,
842                            controls,
843                            &mut playback_clock,
844                            |event| records.submitted(event),
845                        );
846                    };
847                    if end != PlaybackEnd::Natural {
848                        return Ok(end);
849                    }
850                }
851            })()?;
852            if end != PlaybackEnd::Natural {
853                order.focus(cursor)?;
854            }
855            if end != PlaybackEnd::Stop {
856                order.sync(&playback_clock, catalog, paths)?;
857            }
858            let end = if end == PlaybackEnd::Natural {
859                if !seek_reached_eof && output.stopped_early()? {
860                    return Err("audio output stopped before the selection ended".into());
861                }
862                if order.continues() {
863                    // Natural repeat/advance keeps compatible PCM state; manual
864                    // transport and seeking below discard it explicitly.
865                    order.advance(
866                        PlaybackEnd::Natural,
867                        records.position_ms(cursor, &playback_clock),
868                    )?;
869                    seek_ms = 0;
870                    continue;
871                }
872                let end = flush_group(
873                    &mut processing,
874                    &mut records,
875                    output,
876                    controls,
877                    &mut playback_clock,
878                )?;
879                let end = if end == PlaybackEnd::Natural {
880                    drain_output(output, controls, &mut playback_clock)?
881                } else {
882                    end
883                };
884                if end != PlaybackEnd::Stop {
885                    order.sync(&playback_clock, catalog, paths)?;
886                }
887                if end == PlaybackEnd::Natural {
888                    playback_clock.clear_visualizer();
889                    if order.continues() {
890                        records.finish(true, &playback_clock)?;
891                        order.advance(end, 0)?;
892                        seek_ms = 0;
893                        if let Some(lyrics) = &mut playback_clock.lyrics {
894                            lyrics.reset_output();
895                        }
896                        if let Some(timeline) = &mut playback_clock.timeline {
897                            timeline.reset_output();
898                        }
899                        playback_clock.visualizer = None;
900                        playback_clock.reset_elapsed();
901                        continue;
902                    }
903                    records.finish(true, &playback_clock)?;
904                    order.advance(end, 0)?;
905                    return Ok(());
906                }
907                end
908            } else {
909                end
910            };
911            let position_ms = if seeking {
912                seek_ms
913            } else {
914                records.position_ms(cursor, &playback_clock)
915            };
916            playback_clock.clear_visualizer();
917            output.abort()?;
918            if let Some(lyrics) = &mut playback_clock.lyrics {
919                lyrics.reset_output();
920            }
921            if let Some(timeline) = &mut playback_clock.timeline {
922                timeline.reset_output();
923            }
924            processing = None;
925            playback_clock.visualizer = None;
926            normalization.finish_track(index, false)?;
927            order.focus(cursor)?;
928            if let PlaybackEnd::SeekRelative(delta) = end {
929                initial_seek_pending = false;
930                records.carry_seek(cursor, &playback_clock)?;
931                seek_ms = position_ms.saturating_add_signed(delta);
932                records.last_position = Some((cursor, seek_ms));
933                println!("Seeking: {}.{:03} s", seek_ms / 1000, seek_ms % 1000);
934            } else {
935                records.finish(false, &playback_clock)?;
936                let Some(next) = order.advance(end, position_ms)? else {
937                    return Ok(());
938                };
939                // Manual transport has discarded the old output already. A
940                // second key during preparation belongs to this new selection,
941                // unlike a speculative advance after a natural source end.
942                cursor = next;
943                seek_ms = if initial_seek_pending {
944                    options.seek_ms
945                } else {
946                    0
947                };
948            }
949            playback_clock.reset_elapsed();
950        }
951        Ok(())
952    })();
953    if result.is_err() {
954        playback_clock.clear_visualizer();
955        if let Err(error) = output.abort() {
956            eprintln!("Warning: could not stop output: {error}");
957        }
958        if let Some(index) = order.current()
959            && let Err(error) = normalization.finish_track(index, false)
960        {
961            eprintln!("Warning: could not discard loudness capture: {error}");
962        }
963        records.finish(false, &playback_clock)?;
964    }
965    result
966}

968fn report_meter(meter: &mut OutputMeter) {
969    let measured = match meter.snapshot() {
970        Ok(snapshot) => snapshot,
971        Err(error) => {
972            eprintln!("True-peak meter unavailable: {error}");
973            meter.sample_peak_snapshot()
974        }
975    };
976    if measured.frames > 0 {
977        let true_peak = measured
978            .output_true_peak
979            .map(|peak| format!("{} dBTP", peak_db(peak)))
980            .unwrap_or_else(|| "unavailable".to_string());
981        println!(
982            "Submitted signal (before dither/device): sample peak {} dBFS; estimated true peak {true_peak}; pre-guard peak {} dBFS; guarded samples {}",
983            peak_db(measured.output_sample_peak),
984            peak_db(measured.pre_guard_sample_peak),
985            measured.guarded_samples
986        );
987    }
988    if measured.guarded_samples > 0 {
989        eprintln!(
990            "Warning: {} PCM samples were hard-clamped at full scale before output; peak metadata or filter transients may explain this",
991            measured.guarded_samples
992        );
993    }
994}

996fn describe_track(
997    track: &PcmTrack,
998    format: PcmFormat,
999    output: &LocalOutput,
1000    settings: &PlaybackSettings,
1001    measuring: bool,
1002    label: &str,
1003    spectrum: bool,
1004) -> Result<f32, Box<dyn Error>> {
1005    if track.source_format().channels() > 2 {
1006        println!(
1007            "Channels: {} → stereo (LFE omitted, peak-safe downmix)",
1008            track.source_format().layout().name()
1009        );
1010    }
1011    if let Some(description) = output.integer_description() {
1012        println!("Output: {description} with TPDF dither");
1013    }
1014    let resample = format.sample_rate() != track.format().sample_rate();
1015    if resample {
1016        println!(
1017            "Sample rate: {} → {} Hz (device conversion)",
1018            track.format().sample_rate(),
1019            format.sample_rate()
1020        );
1021    }
1022    let mut normalization_gain_db = 0.0;
1023    let mut normalization_headroom_db = 0.0;
1024    if let Some(selection) = settings.selected_gain {
1025        normalization_gain_db = gain_with_headroom_db(selection.gain_db, selection.source_peak)?;
1026        normalization_headroom_db = (selection.gain_db - normalization_gain_db).max(0.0);
1027        println!(
1028            "Normalization: {normalization_gain_db:+.2} dB ({})",
1029            selection.label
1030        );
1031        if normalization_headroom_db > 0.0 {
1032            println!(
1033                "Headroom: requested {:+.2} dB reduced to {normalization_gain_db:+.2} dB ({})",
1034                selection.gain_db, selection.peak_label
1035            );
1036        }
1037    } else if settings.normalization_mode != NormalizationMode::Off {
1038        println!("Normalization: no stored loudness or matching gain; decoded level retained");
1039    }
1040    if measuring {
1041        println!(
1042            "Loudness: measuring source during playback for a future listen; current gain stays fixed"
1043        );
1044    }
1045    let preamp_db = settings.tone.safe_preamp_db();
1046    if !settings.tone.is_flat() {
1047        println!(
1048            "Tone: bass {:+.1} dB, treble {:+.1} dB; headroom preamp {preamp_db:+.1} dB",
1049            settings.tone.bass_db(),
1050            settings.tone.treble_db()
1051        );
1052    }
1053    println!("Playing: {label}");
1054    let normalization_stage = match settings.selected_gain {
1055        Some(selection) => format!("{} {normalization_gain_db:+.2} dB", selection.label),
1056        None if settings.normalization_mode == NormalizationMode::Off => "off".to_string(),
1057        None => "no gain available".to_string(),
1058    };
1059    println!(
1060        "DSP stages: downmix {}; resample {}; normalization {}; tone {}; output guard on; limiter off; output {}; spectrum {}",
1061        if track.source_format().channels() > 2 {
1062            "to stereo"
1063        } else {
1064            "bypass"
1065        },
1066        if resample { "on" } else { "bypass" },
1067        normalization_stage,
1068        if settings.tone.is_flat() {
1069            "bypass"
1070        } else {
1071            "on"
1072        },
1073        output.stage_description(),
1074        if spectrum { "display on" } else { "off" }
1075    );
1076    println!(
1077        "Headroom reserve: normalization {:.2} dB; tone {:.2} dB; dynamic gain reduction unavailable (limiter off)",
1078        normalization_headroom_db, -preamp_db
1079    );
1080    Ok(normalization_gain_db + preamp_db)
1081}
