1//! Getting the catalog out: JSON, CSV, and playlists.
2//!
3//! Three formats for three different questions.
4//!
5//! **JSON** is the faithful dump: nine linked tables, one per table of the
6//! model, which is what allows the catalog to be rebuilt or fed to another
7//! program.
8//!
9//! **CSV** is flat by nature, so it cannot hold that graph. It answers another
10//! question — "let me sort my library in a spreadsheet" — and for that a single
11//! denormalized table beats nine faithful ones. Its values are therefore raw:
12//! milliseconds and bytes, not `4:20` and `31.2 MB`, because a column one
13//! cannot add up is a column one cannot use.
14//!
15//! **M3U** is not an export of the catalog but of a *selection*: whatever is on
16//! screen, handed to a player.
17

29pub fn export(args: &Args) -> Res {
30    // `export` is about the whole catalog and takes no argument. Ignoring one
31    // would answer a question that was not asked, with a file that looks right.
32    if let Some(stray) = args.positionals.first() {
33        return Err(format!(
34            "export covers the whole catalog and takes no argument.\n\
35             To export one album or one artist, ask the command that knows it:\n\
36             \taede album \"{stray}\" --csv\n\
37             \taede artist \"{stray}\" --csv"
38        )
39        .into());
40    }
41
42    let catalog = load(args)?;
43    if args.has("graph") {
44        if args.has("csv") || args.has("tracks") {
45            return Err("--graph is a linked JSON document; drop --csv and --tracks".into());
46        }
47        return export_graph(args, &catalog);
48    }
49    // `--csv` flattens the graph into one table; without it the dump is the
50    // faithful one, and `--json` on `export` asks for exactly that.
51    if args.has("csv") {
52        return if args.has("tracks") {
53            let all: Vec<Id> = catalog.tracks.iter().map(|t| t.id).collect();
54            tracks_table(&catalog, &all, args)
55        } else {
56            let all: Vec<Id> = catalog.releases.iter().map(|r| r.id).collect();
57            albums_table(&catalog, &all, args)
58        };
59    }
60    emit(args, &store::to_json(&catalog).to_string_pretty())
61}

63/// Exports the complete local-first graph: derived catalog, attributed source
64/// evidence, personal statements, and one materialized edge list joining the
65/// three without flattening their provenance.
66fn export_graph(args: &Args, catalog: &Catalog) -> Res {
67    let directory = data_dir(args);
68    let sources = aede_core::sources::load(&aede_core::sources::sources_path(&directory))?
69        .unwrap_or_default();
70    let user = aede_core::user::load(&aede_core::user::user_path(&directory))?.unwrap_or_default();
71    let edges = aede_core::graph::edges(catalog, &sources);
72
73    let mut root = Json::obj();
74    root.set("format", "aede-graph".into());
75    root.set("format_version", 1u32.into());
76    root.set("exported_at", aede_core::clock::now_seconds().into());
77    root.set("catalog", store::to_json(catalog));
78    let conclusions =
79        aede_core::conclusions::load(&aede_core::conclusions::conclusions_path(&directory))?
80            .unwrap_or_else(|| aede_core::conclusions::Conclusions::from_catalog(catalog));
81    root.set("conclusions", aede_core::conclusions::to_json(&conclusions));
82    root.set("sources", aede_core::sources::to_json(&sources));
83    root.set("user", aede_core::user::to_json(&user));
84    root.set(
85        "relations",
86        Json::Arr(
87            edges
88                .iter()
89                .map(|edge| {
90                    let annotation =
91                        user.find_relation(aede_core::user::LOCAL_USER, &edge.reference);
92                    super::relation::edge_json(edge, annotation)
93                })
94                .collect(),
95        ),
96    );
97    emit(args, &root.to_string_pretty())
98}

109/// Writes to the file given by `--output`, or to standard output.
110pub fn emit(args: &Args, text: &str) -> Res {
111    preflight_output(args)?;
112    match args.value("output") {
113        Some(path) => {
114            aede_core::playlist::write_atomic(Path::new(path), text)?;
115            println!(
116                "{} {}",
117                ui::green("→"),
118                format_args!(
119                    "written to {} ({})",
120                    ui::literal(path),
121                    aede_core::text::format_size(text.len() as u64)
122                )
123            );
124        }
125        None => print!("{text}"),
126    }
127    Ok(())
128}

130/// Checks an explicit export destination before a command changes local data.
131///
132/// Active stores and their writer locks are reserved, including paths reached
133/// through directory aliases. Existing audio, final symbolic links and other
134/// non-regular outputs are refused. This reads no catalog and creates no files.
135pub fn preflight_output(args: &Args) -> Res {
136    let Some(raw) = args.value("output") else {
137        return Ok(());
138    };
139    preflight_path(args, Path::new(raw), "export")
140}

142/// Applies the same file protection to exports and positional backup outputs.
143pub(super) fn preflight_path(args: &Args, path: &Path, role: &str) -> Res {
144    let raw = path.display();
145    let subject = if role == "export" {
146        "an export"
147    } else {
148        "a backup"
149    };
150    let existing = match std::fs::symlink_metadata(path) {
151        Ok(metadata) if !metadata.is_file() || metadata.file_type().is_symlink() => {
152            return Err(format!(
153                "{raw}: refusing a symbolic link or non-regular {role} destination"
154            )
155            .into());
156        }
157        Ok(_) => true,
158        Err(error) if error.kind() == std::io::ErrorKind::NotFound => false,
159        Err(error) => {
160            return Err(format!("{raw}: cannot inspect {role} destination: {error}").into());
161        }
162    };
163    let destination = resolved_path(path)?;
164    let directory = resolved_path(&data_dir(args))?;
165    for name in [
166        store::CATALOG_FILE,
167        aede_core::user::USER_FILE,
168        aede_core::sources::SOURCES_FILE,
169        aede_core::conclusions::CONCLUSIONS_FILE,
170        aede_core::accounts::ACCOUNTS_FILE,
171        ".aede.lock",
172        ".aede-server.lock",
173    ] {
174        let protected = resolved_path(&directory.join(name))?;
175        if same_path(&destination, &protected) {
176            return Err(format!(
177                "{raw}: {subject} cannot replace an active Aède store or writer lock"
178            )
179            .into());
180        }
181    }
182    if existing {
183        if aede_core::tags::is_audio_path(path) {
184            return Err(format!("{raw}: {subject} cannot replace an audio file").into());
185        }
186        match aede_core::tags::read(path) {
187            Ok(_) | Err(aede_core::tags::TagError::Malformed(_)) => {
188                return Err(format!("{raw}: {subject} cannot replace an audio file").into());
189            }
190            Err(aede_core::tags::TagError::UnrecognizedFormat) => {}
191            Err(error) => {
192                return Err(format!("{raw}: cannot inspect {role} destination: {error}").into());
193            }
194        }
195    }
196    Ok(())
197}

199fn same_path(left: &Path, right: &Path) -> bool {
200    #[cfg(windows)]
201    {
202        left.as_os_str()
203            .as_encoded_bytes()
204            .eq_ignore_ascii_case(right.as_os_str().as_encoded_bytes())
205    }
206    #[cfg(not(windows))]
207    {
208        left == right
209    }
210}

212/// Resolves aliases in existing ancestors without creating missing directories.
213fn resolved_path(path: &Path) -> std::io::Result<PathBuf> {
214    let absolute = std::path::absolute(path)?;
215    let mut ancestor = absolute.as_path();
216    let mut missing = Vec::new();
217    loop {
218        match ancestor.canonicalize() {
219            Ok(mut resolved) => {
220                if !missing.is_empty() && !resolved.is_dir() {
221                    return Err(std::io::Error::new(
222                        std::io::ErrorKind::NotADirectory,
223                        "export parent is not a directory",
224                    ));
225                }
226                for part in missing.iter().rev() {
227                    match part {
228                        Component::Normal(name) => resolved.push(name),
229                        Component::ParentDir => {
230                            resolved.pop();
231                        }
232                        Component::CurDir => {}
233                        _ => {
234                            return Err(std::io::Error::new(
235                                std::io::ErrorKind::InvalidInput,
236                                "invalid export path",
237                            ));
238                        }
239                    }
240                }
241                return Ok(resolved);
242            }
243            Err(error) if error.kind() == std::io::ErrorKind::NotFound => {
244                // A dangling directory link is not a missing directory that
245                // this command may create in another tree.
246                if std::fs::symlink_metadata(ancestor).is_ok() {
247                    return Err(error);
248                }
249                let Some(part) = ancestor.components().next_back() else {
250                    return Err(error);
251                };
252                let Some(parent) = ancestor.parent() else {
253                    return Err(error);
254                };
255                missing.push(part);
256                ancestor = parent;
257            }
258            Err(error) => return Err(error),
259        }
260    }
261}

263/// Field separator, `,` unless asked otherwise.
264///
265/// Excel in a French or German locale splits on `;` and would show a one-column
266/// sheet otherwise. The alternative — a `sep=;` line at the top of the file —
267/// is understood by Excel alone and corrupts the file for every other reader.
268pub(super) fn separator(args: &Args) -> Result<char, Box<dyn std::error::Error>> {
269    match args.value("separator") {
270        None => Ok(','),
271        Some("tab") => Ok('\t'),
272        Some(value) => {
273            let mut chars = value.chars();
274            match (chars.next(), chars.next()) {
275                (Some(c), None) => Ok(c),
276                _ => Err(
277                    format!("--separator takes one character, or \"tab\": got \"{value}\"").into(),
278                ),
279            }
280        }
281    }
282}

290/// Writes any listing as a table: a header, then the rows as given.
291///
292/// The listings differ too much to share a shape — a year is not an artist —
293/// but they share the quoting, the separator, the JSON typing and where the
294/// text ends up. **One function for both formats**, so that an option honoured
295/// by one and not the other cannot happen again: `--json` was accepted by every
296/// listing and read by none, which printed the ordinary table and dropped the
297/// word.
298pub fn rows_table(header: &[&str], rows: &[Vec<String>], args: &Args) -> Res {
299    // Every listing that can be written out as a table — `album`, `artist`,
300    // `facet`, `annotate`'s and `browse`'s own listings among them — ends up
301    // here, which makes this the one place `--csv` and `--json` together can
302    // be caught for all of them at once, rather than in each caller that
303    // happens to offer both.
304    if let Some(message) = args.output_conflict(&["csv", "json"]) {
305        return Err(message.into());
306    }
307    if args.has("json") {
308        return emit(args, &rows_json(header, rows));
309    }
310    let separator = separator(args)?;
311    let mut out = String::new();
312    push_row(
313        &mut out,
314        &header.iter().map(|h| h.to_string()).collect::<Vec<_>>(),
315        separator,
316    );
317    for row in rows {
318        push_row(&mut out, row, separator);
319    }
320    emit(args, &out)
321}

323/// Columns whose value is a number, and must reach JSON as one.
324///
325/// Named rather than guessed: a cell that merely *looks* like a number is not
326/// one. An album called "1999" is a title, and turning it into an integer
327/// because it happens to parse is a wrong answer no reader could detect.
328const NUMERIC_COLUMNS: &[&str] = &[
329    "tracks",
330    "albums",
331    "artists",
332    "discs",
333    "year",
334    "duration_ms",
335    "size_bytes",
336    "track_no",
337    "disc_no",
338    "bitrate_kbps",
339    "sample_rate_hz",
340    "bit_depth",
341    "channels",
342];
343
344/// Columns holding a yes/no, which JSON has a type for.
345const BOOLEAN_COLUMNS: &[&str] = &["lossless", "compilation"];
346
347/// The same rows as the CSV, as an array of objects.
348///
349/// An empty cell becomes `null` rather than `""`: "this album has no year" and
350/// "this album has an empty year" are different claims, and only the first is
351/// true. The CSV cannot make that distinction; JSON can, so it does.
352pub fn rows_json(header: &[&str], rows: &[Vec<String>]) -> String {
353    let list: Vec<Json> = rows
354        .iter()
355        .map(|row| {
356            let mut o = Json::obj();
357            for (name, cell) in header.iter().zip(row) {
358                o.set(name, cell_json(name, cell));
359            }
360            o
361        })
362        .collect();
363    Json::Arr(list).to_string_pretty()
364}
365
366fn cell_json(name: &str, cell: &str) -> Json {
367    if cell.is_empty() {
368        return Json::Null;
369    }
370    if BOOLEAN_COLUMNS.contains(&name) {
371        match cell {
372            "true" => return Json::Bool(true),
373            "false" => return Json::Bool(false),
374            _ => {}
375        }
376    }
377    if NUMERIC_COLUMNS.contains(&name)
378        && let Ok(number) = cell.parse::<f64>()
379    {
380        return Json::Num(number);
381    }
382    Json::Str(cell.to_string())
383}

385/// One row per album: what a library looks like from above.
386///
387/// Built once, so that the CSV and the JSON can never drift apart.
388fn album_rows(catalog: &Catalog, releases: &[Id]) -> (Vec<&'static str>, Vec<Vec<String>>) {
389    let mut rows: Vec<Vec<String>> = Vec::new();
390    let header = vec![
391        "album_artist",
392        "album",
393        "year",
394        "date",
395        "tracks",
396        "discs",
397        "duration_ms",
398        "size_bytes",
399        "formats",
400        "sample_rates_hz",
401        "bit_depths",
402        "lossless",
403        "compilation",
404        "label",
405        "catalog_number",
406        "barcode",
407        "media",
408        "genres",
409        "integrity",
410        "musicbrainz_albumid",
411        "folder",
412    ];
413    for release in releases.iter().filter_map(|&id| catalog.release(id)) {
414        let tracks: Vec<&aede_core::model::Track> = release
415            .track_ids
416            .iter()
417            .filter_map(|&id| catalog.track(id))
418            .collect();
419        let files: Vec<&aede_core::model::AudioFile> = tracks
420            .iter()
421            .filter_map(|t| catalog.file(t.file_id))
422            .collect();
423
424        let discs: BTreeSet<u32> = tracks.iter().map(|t| t.disc_no.unwrap_or(1)).collect();
425        let duration: u64 = tracks.iter().filter_map(|t| t.duration_ms).sum();
426        let size: u64 = files.iter().map(|f| f.size).sum();
427        let formats: BTreeSet<&str> = files.iter().map(|f| f.properties.codec.as_str()).collect();
428        let rates: BTreeSet<u32> = files
429            .iter()
430            .filter_map(|f| f.properties.sample_rate)
431            .collect();
432        let depths: BTreeSet<u16> = files
433            .iter()
434            .filter_map(|f| f.properties.bit_depth)
435            .collect();
436        let genres: Vec<String> = catalog
437            .genres_of(EntityKind::Release, release.id)
438            .into_iter()
439            .map(|g| g.name.clone())
440            .collect();
441        let labels: Vec<String> = release
442            .label_ids
443            .iter()
444            .filter_map(|&id| catalog.label(id))
445            .map(|l| l.name.clone())
446            .collect();
447
448        let row = [
449            release
450                .album_artist_id
451                .and_then(|id| catalog.artist(id))
452                .map(|a| a.name.clone())
453                .unwrap_or_else(|| "Various Artists".into()),
454            release.title.clone(),
455            release.year.map(|y| y.to_string()).unwrap_or_default(),
456            release.date.clone().unwrap_or_default(),
457            tracks.len().to_string(),
458            discs.len().to_string(),
459            duration.to_string(),
460            size.to_string(),
461            join(formats.iter().map(|c| c.to_string())),
462            join(rates.iter().map(|r| r.to_string())),
463            join(depths.iter().map(|d| d.to_string())),
464            bool_cell(files.iter().all(|f| f.properties.lossless)),
465            bool_cell(release.is_compilation),
466            labels.join(" / "),
467            release.catalog_number.clone().unwrap_or_default(),
468            release.barcode.clone().unwrap_or_default(),
469            release.media.clone().unwrap_or_default(),
470            genres.join(" / "),
471            integrity_of(&files).into(),
472            release.mbid.clone().unwrap_or_default(),
473            release.folder.clone(),
474        ];
475        rows.push(row.to_vec());
476    }
477    (header, rows)
478}

480/// One row per track, for when the album view is too coarse.
481/// The track table, once, for the same reason.
482fn track_rows(catalog: &Catalog, tracks: &[Id]) -> (Vec<&'static str>, Vec<Vec<String>>) {
483    let mut rows: Vec<Vec<String>> = Vec::new();
484    let header = vec![
485        "artist",
486        "album_artist",
487        "album",
488        "year",
489        "disc_no",
490        "track_no",
491        "title",
492        "duration_ms",
493        "size_bytes",
494        "codec",
495        "container",
496        "sample_rate_hz",
497        "bit_depth",
498        "channels",
499        "bitrate_kbps",
500        "lossless",
501        "integrity",
502        "isrc",
503        "musicbrainz_recordingid",
504        "path",
505    ];
506    for track in tracks.iter().filter_map(|&id| catalog.track(id)) {
507        let release = track.release_id.and_then(|id| catalog.release(id));
508        let file = catalog.file(track.file_id);
509        let performers: Vec<String> = catalog
510            .credits_on(EntityKind::Track, track.id)
511            .into_iter()
512            .filter(|(_, role)| *role == "main")
513            .map(|(a, _)| a.name.clone())
514            .collect();
515
516        let row = [
517            performers.join(" / "),
518            release
519                .and_then(|r| r.album_artist_id)
520                .and_then(|id| catalog.artist(id))
521                .map(|a| a.name.clone())
522                .unwrap_or_default(),
523            release.map(|r| r.title.clone()).unwrap_or_default(),
524            release
525                .and_then(|r| r.year)
526                .map(|y| y.to_string())
527                .unwrap_or_default(),
528            track.disc_no.map(|d| d.to_string()).unwrap_or_default(),
529            track.track_no.map(|n| n.to_string()).unwrap_or_default(),
530            track.title.clone(),
531            track.duration_ms.map(|d| d.to_string()).unwrap_or_default(),
532            file.map(|f| f.size.to_string()).unwrap_or_default(),
533            file.map(|f| f.properties.codec.clone()).unwrap_or_default(),
534            file.map(|f| f.properties.container.clone())
535                .unwrap_or_default(),
536            number(file.and_then(|f| f.properties.sample_rate)),
537            number(file.and_then(|f| f.properties.bit_depth)),
538            number(file.and_then(|f| f.properties.channels)),
539            number(file.and_then(|f| f.properties.bitrate_kbps)),
540            bool_cell(file.map(|f| f.properties.lossless).unwrap_or(false)),
541            integrity_of(&file.into_iter().collect::<Vec<_>>()).into(),
542            track.isrc.clone().unwrap_or_default(),
543            track.mbid.clone().unwrap_or_default(),
544            file.map(|f| f.path.clone()).unwrap_or_default(),
545        ];
546        rows.push(row.to_vec());
547    }
548    (header, rows)
549}

551/// Integrity of a set of files, in one word.
552///
553/// The worst verdict wins: an album holding one damaged track is a damaged
554/// album, and one unverified track is enough for the album not to be vouched
555/// for. Anything else would let a problem hide behind an average.
556fn integrity_of(files: &[&aede_core::model::AudioFile]) -> &'static str {
557    use aede_core::audit::integrity::Verdict;
558    if files.is_empty() {
559        return "";
560    }
561    let mut all_nothing = true;
562    let mut any_unverified = false;
563    for file in files {
564        match file.integrity.as_ref().map(|r| &r.verdict) {
565            Some(Verdict::Damaged { .. }) => return "damaged",
566            Some(Verdict::Intact) => all_nothing = false,
567            Some(Verdict::NothingToCheck) => {}
568            None => {
569                any_unverified = true;
570                all_nothing = false;
571            }
572        }
573    }
574    match (any_unverified, all_nothing) {
575        (true, _) => "not_verified",
576        (false, true) => "nothing_to_check",
577        (false, false) => "intact",
578    }
579}

585/// `true`/`false` rather than 1/0: a spreadsheet filters on words, and 1/0
586/// invites arithmetic on something that is not a quantity.
587fn bool_cell(value: bool) -> String {
588    if value { "true" } else { "false" }.to_string()
589}

595fn push_row(out: &mut String, cells: &[String], separator: char) {
596    let line: Vec<String> = cells.iter().map(|c| escape(c, separator)).collect();
597    out.push_str(&line.join(&separator.to_string()));
598    // CRLF, as RFC 4180 asks: it is what Excel expects, and every other reader
599    // copes with it.
600    out.push_str("\r\n");
601}

603/// Quotes a field when it has to be, doubling the quotes inside it.
604///
605/// Album titles carry commas and quotation marks, and a few carry line breaks
606/// left by a tagger. Getting this wrong shifts every following column.
607fn escape(value: &str, separator: char) -> String {
608    let needs_quotes = value.contains(separator)
609        || value.contains('"')
610        || value.contains('\n')
611        || value.contains('\r');
612    if !needs_quotes {
613        return value.to_string();
614    }
615    format!("\"{}\"", value.replace('"', "\"\""))
616}

618/// Renders a track list as an extended M3U playlist.
619///
620/// Paths are absolute, which is what makes the file work wherever it is saved.
621/// The rendering itself lives in [`aede_core::playlist`], shared with `aede
622/// playlist` — that one writes into the album folder and therefore names its
623/// tracks relatively, and two writers agreeing about `#EXTINF` today would
624/// disagree about it in six months.
625pub fn m3u(catalog: &Catalog, tracks: &[Id]) -> Result<String, String> {
626    aede_core::playlist::try_render(catalog, tracks, None, aede_core::playlist::Style::Extended)
627}
