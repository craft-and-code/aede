12pub(super) fn print(
13    report: &ScanReport,
14    catalog: &Catalog,
15    path: &Path,
16    dry_run: bool,
17    machine: bool,
18) {
19    if machine {
20        println!("{}", as_json(report, catalog, dry_run).to_string_compact());
21        return;
22    }
23    println!(
24        "{}",
25        ui::section(if dry_run {
26            "Scan preview"
27        } else {
28            "Scan complete"
29        })
30    );
31    let mut table = Table::plain(2).align(1, Align::Right);
32    table.push(vec!["Files found".into(), report.found.to_string()]);
33    table.push(vec!["Read from disk".into(), report.read.to_string()]);
34    table.push(vec![
35        "Reused from previous scan".into(),
36        report.reused.to_string(),
37    ]);
38    if report.preserved > 0 {
39        table.push(vec![
40            "Kept from inaccessible paths".into(),
41            report.preserved.to_string(),
42        ]);
43    }
44    if report.removed > 0 {
45        table.push(vec![
46            "Gone since last scan".into(),
47            report.removed.to_string(),
48        ]);
49    }
50    if report.reports > 0 {
51        table.push(vec![
52            "Analyses imported".into(),
53            format!(
54                "{} from {}",
55                report.analyses,
56                ui::plural(report.reports, "report")
57            ),
58        ]);
59    }
60    if report.attached > 0 {
61        table.push(vec![
62            "Analyses now attached".into(),
63            report.attached.to_string(),
64        ]);
65    }
66    table.push(vec!["Elapsed".into(), ui::elapsed(report.elapsed_ms)]);
67    print!("{}", table.render());
68
69    if dry_run {
70        paths("Would add", &report.added_paths);
71        paths("Would reread changed files", &report.changed_paths);
72        paths("Would remove", &report.removed_paths);
73    }
74    if !report.failures.is_empty() {
75        println!("{}", ui::section("Unreadable paths"));
76        let mut table = Table::new(&["Path", "Reason"]).path_limit(0, 60);
77        for (path, reason) in report.failures.iter().take(20) {
78            table.push(vec![path.clone(), reason.clone()]);
79        }
80        print!("{}", table.render());
81        remaining(report.failures.len());
82    }
83    let totals = stats::compute(catalog);
84    println!(
85        "\n{} {} · {} · {} · {}",
86        ui::green("→"),
87        ui::plural(totals.tracks, "track"),
88        ui::plural(totals.releases, "album"),
89        ui::plural(totals.artists, "artist"),
90        ui::long_duration(totals.total_duration_ms)
91    );
92    if dry_run {
93        println!(
94            "{}",
95            ui::dim("  preview only: catalog, watched folders and personal data were not changed")
96        );
97    }
98    println!("{}", ui::dim(&format!("  catalog: {}", path.display())));
99}

101fn paths(title: &str, paths: &[String]) {
102    if paths.is_empty() {
103        return;
104    }
105    println!("{}", ui::section(&format!("{title} ({})", paths.len())));
106    let mut table = Table::new(&["File"]).path_limit(0, 80);
107    for path in paths.iter().take(20) {
108        table.push(vec![path.clone()]);
109    }
110    print!("{}", table.render());
111    remaining(paths.len());
112}
113
114fn remaining(count: usize) {
115    if count > 20 {
116        println!("{}", ui::dim(&format!("  … and {} more", count - 20)));
117    }
118}

120pub(super) fn as_json(report: &ScanReport, catalog: &Catalog, dry_run: bool) -> Json {
121    let mut result = Json::obj();
122    result.set("dry_run", dry_run.into());
123    result.set("roots", strings(&catalog.roots));
124    result.set("added", strings(&report.added_paths));
125    result.set("changed", strings(&report.changed_paths));
126    result.set("removed", strings(&report.removed_paths));
127    result.set(
128        "unreadable",
129        Json::Arr(
130            report
131                .failures
132                .iter()
133                .map(|(path, reason)| {
134                    let mut failure = Json::obj();
135                    failure.set("path", path.as_str().into());
136                    failure.set("reason", reason.as_str().into());
137                    failure
138                })
139                .collect(),
140        ),
141    );
142    let mut counts = Json::obj();
143    for (name, count) in [
144        ("found", report.found),
145        ("read", report.read),
146        ("reused", report.reused),
147        ("preserved", report.preserved),
148        ("added", report.added_paths.len()),
149        ("changed", report.changed_paths.len()),
150        ("removed", report.removed),
151        ("catalogued", catalog.files.len()),
152    ] {
153        counts.set(name, (count as u64).into());
154    }
155    result.set("counts", counts);
156    result
157}
158
159fn strings(paths: &[String]) -> Json {
160    Json::Arr(paths.iter().map(|path| path.as_str().into()).collect())
161}
