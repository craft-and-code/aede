14fn arguments(request: &JobRequest) -> Vec<String> {
15    let mut args = Vec::new();
16    let targets = match request {
17        JobRequest::Scan(scan) => {
18            args.push("scan".into());
19            for (enabled, flag) in [
20                (scan.full, "full"),
21                (scan.replace, "replace"),
22                (scan.follow_symlinks, "follow-symlinks"),
23                (scan.include_hidden, "include-hidden"),
24            ] {
25                if enabled {
26                    args.push(format!("--{flag}"));
27                }
28            }
29            if let Some(threads) = scan.threads {
30                args.push(format!("--threads={threads}"));
31            }
32            &scan.folders
33        }
34        JobRequest::Fetch(fetch) => {
35            args.push("fetch".into());
36            for (enabled, flag) in [
37                (fetch.summaries, "summaries"),
38                (fetch.discography, "discography"),
39                (fetch.covers, "covers"),
40                (fetch.lyrics, "lyrics"),
41                (fetch.identify, "identify"),
42                (fetch.credits, "credits"),
43                (fetch.recordings, "recordings"),
44                (fetch.portraits, "portraits"),
45                (fetch.logos, "logos"),
46                (fetch.labels, "labels"),
47                (fetch.fanart, "fanart"),
48                (fetch.banners, "banners"),
49                (fetch.images, "images"),
50                (fetch.full, "full"),
51                (fetch.dry_run, "dry-run"),
52                (fetch.yes, "yes"),
53                (fetch.no_logo, "no-logo"),
54                (fetch.no_label_logo, "no-label-logo"),
55                (fetch.no_portrait, "no-portrait"),
56                (fetch.no_background, "no-background"),
57                (fetch.no_banner, "no-banner"),
58                (fetch.no_album_cover, "no-album-cover"),
59                (fetch.no_cdart, "no-cdart"),
60            ] {
61                if enabled {
62                    args.push(format!("--{flag}"));
63                }
64            }
65            for (name, value) in [("size", &fetch.size), ("lang", &fetch.lang)] {
66                if let Some(value) = value {
67                    args.push(format!("--{name}={value}"));
68                }
69            }
70            &fetch.targets
71        }
72    };
73    // Every value remains one OS argument. The separator prevents a target
74    // such as "--data=/elsewhere" from becoming a command option.
75    args.push("--".into());
76    args.extend(targets.iter().cloned());
77    args
78}

80pub(super) fn run(
81    data_dir: &Path,
82    request: JobRequest,
83    cancellation: Arc<AtomicBool>,
84) -> Result<JobOutput, String> {
85    let data_dir = std::fs::canonicalize(data_dir).map_err(|error| error.to_string())?;
86    let exe = std::env::current_exe().map_err(|error| error.to_string())?;
87    let mut command = Command::new(exe);
88    command
89        .args(arguments(&request))
90        .env("AEDE_DELEGATED_CHILD", "1")
91        .env("AEDE_DELEGATED_DATA_DIR", data_dir)
92        .env("NO_COLOR", "1")
93        .env_remove("AEDE_ADMIN_TOKEN")
94        .env_remove("AEDE_DELEGATED_STDIN_TTY")
95        .env_remove("AEDE_DELEGATED_STDOUT_TTY")
96        .stdin(Stdio::null())
97        .stdout(Stdio::piped())
98        .stderr(Stdio::piped());
99    let secrets: Vec<String> = ["AEDE_ACOUSTID_KEY", "AEDE_FANARTTV_KEY", "AEDE_ADMIN_TOKEN"]
100        .into_iter()
101        .filter_map(|key| std::env::var(key).ok())
102        .filter(|value| !value.is_empty())
103        .collect();
104    execute(command, cancellation, &secrets)
105}

107fn execute(
108    mut command: Command,
109    cancellation: Arc<AtomicBool>,
110    secrets: &[String],
111) -> Result<JobOutput, String> {
112    if cancellation.load(Ordering::Acquire) {
113        return Ok(JobOutput {
114            exit_code: 130,
115            ..Default::default()
116        });
117    }
118    let mut child = command
119        .spawn()
120        .map_err(|error| format!("cannot start administrative command: {error}"))?;
121    let stdout = child.stdout.take().ok_or("missing command stdout")?;
122    let stderr = child.stderr.take().ok_or("missing command stderr")?;
123    let out = std::thread::spawn(move || capture(stdout));
124    let err = std::thread::spawn(move || capture(stderr));
125    let mut cancelled = false;
126    let status = loop {
127        match child.try_wait() {
128            Ok(Some(status)) => break Ok(status),
129            Ok(None) => {}
130            Err(error) => {
131                let _ = child.kill();
132                let _ = child.wait();
133                break Err(format!("cannot wait for administrative command: {error}"));
134            }
135        }
136        if cancellation.load(Ordering::Acquire) && child.kill().is_ok() {
137            cancelled = true;
138        }
139        std::thread::sleep(Duration::from_millis(50));
140    };
141    let stdout = out.join().map_err(|_| "command output reader failed")??;
142    let stderr = err.join().map_err(|_| "command error reader failed")??;
143    let status = status?;
144    let (stdout, stdout_truncated) = redact_capture(stdout, secrets);
145    let (stderr, stderr_truncated) = redact_capture(stderr, secrets);
146    Ok(JobOutput {
147        exit_code: if cancelled {
148            130
149        } else {
150            status.code().unwrap_or(1)
151        },
152        stdout,
153        stderr,
154        output_truncated: stdout_truncated || stderr_truncated,
155    })
156}

158fn capture(mut input: impl Read) -> Result<(Vec<u8>, bool), String> {
159    let mut captured = Vec::new();
160    let mut truncated = false;
161    let mut buffer = [0; 8192];
162    loop {
163        match input.read(&mut buffer) {
164            Ok(0) => return Ok((captured, truncated)),
165            Ok(size) => {
166                let keep = size.min(MAX_OUTPUT.saturating_sub(captured.len()));
167                captured.extend_from_slice(&buffer[..keep]);
168                truncated |= keep < size;
169            }
170            Err(error) if error.kind() == std::io::ErrorKind::Interrupted => {}
171            Err(error) => return Err(format!("cannot read command output: {error}")),
172        }
173    }
174}

176fn redact_capture((bytes, truncated): (Vec<u8>, bool), secrets: &[String]) -> (String, bool) {
177    // Match bytes before UTF-8 decoding: a credential can be cut in the middle
178    // of a character or a percent escape, not only between whole characters.
179    let suffix = if truncated {
180        secret_variants(secrets)
181            .iter()
182            .filter_map(|secret| {
183                (1..=secret.len().min(bytes.len()))
184                    .rev()
185                    .find(|&length| bytes.ends_with(&secret.as_bytes()[..length]))
186            })
187            .max()
188            .unwrap_or(0)
189    } else {
190        0
191    };
192    let mut text = redact(
193        String::from_utf8_lossy(&bytes[..bytes.len() - suffix]).into_owned(),
194        secrets,
195    );
196    let marker = if suffix == 0 { "" } else { "[redacted]" };
197    let limit = MAX_OUTPUT - marker.len();
198    let shortened = text.len() > limit;
199    if shortened {
200        let mut boundary = limit;
201        while !text.is_char_boundary(boundary) {
202            boundary -= 1;
203        }
204        text.truncate(boundary);
205    }
206    // Reserve the marker's bytes so redaction cannot exceed the capture bound
207    // or leave an ambiguous, partially truncated redaction marker itself.
208    text.push_str(marker);
209    (text, truncated || shortened)
210}
211
212fn redact(mut text: String, secrets: &[String]) -> String {
213    for secret in secret_variants(secrets) {
214        text = text.replace(&secret, "[redacted]");
215    }
216    text
217}
218
219fn secret_variants(secrets: &[String]) -> Vec<String> {
220    let mut variants = Vec::new();
221    for secret in secrets {
222        for value in [secret.as_str(), secret.trim()] {
223            if value.is_empty() {
224                continue;
225            }
226            variants.push(value.to_string());
227            // Service keys can also appear in an error's query string. The
228            // source adapters trim configuration and percent-encode AcoustID keys.
229            let mut encoded = String::new();
230            let mut lower_encoded = String::new();
231            for byte in value.bytes() {
232                if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'_' | b'.' | b'~') {
233                    encoded.push(char::from(byte));
234                    lower_encoded.push(char::from(byte));
235                } else {
236                    encoded.push_str(&format!("%{byte:02X}"));
237                    lower_encoded.push_str(&format!("%{byte:02x}"));
238                }
239            }
240            variants.push(encoded);
241            variants.push(lower_encoded);
242        }
243    }
244    variants
245}
