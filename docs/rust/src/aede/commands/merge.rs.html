1//! The `merge` command: saying that two spellings are one musician.
2//!
3//! The half of artist identity no authority can settle. `MUSICBRAINZ_ARTISTID`
4//! answers for the files that went through Picard, and `model::identity` uses
5//! it without a heuristic; for an old rip, a download or a friend's drive there
6//! is nothing to consult, and comparing the strings is exactly what must not
7//! happen — a fragment match merges Angus Young with Neil Young. **Nobody on
8//! earth knows that your `O. Osbourne` is Ozzy except you**, so the program
9//! asks.
10//!
11//! Three properties hold everywhere in here.
12//!
13//! **Nothing on disk is touched.** Not the audio files, not their tags, not
14//! what any source said. A merge is a statement about how the shelf is *read*,
15//! it lives in `user.json` with everything else that was said rather than
16//! derived, and `--forget` takes it back.
17//!
18//! **It takes effect on the next scan**, because the spelling a track is filed
19//! under is decided as the artist is interned and there is no later moment at
20//! which one row can become another without rebuilding what points at it. The
21//! command says so rather than leaving the reader to notice.
22//!
23//! **Two identifiers are two people.** Where both spellings carry a
24//! MusicBrainz identifier and the identifiers differ, the merge is refused: the
25//! disagreement is with MusicBrainz, it is a fact somebody else can check, and
26//! the place to fix it is there. That is the one case in this file where the
27//! program knows better than the person typing.
28

37/// What the owner has said, ready for a scan.
38///
39/// Read straight from `user.json` rather than from a loaded [`UserData`],
40/// because the scan runs before anything else is loaded and needs no more than
41/// this. A file that cannot be read gives an empty list: a merge nobody can
42/// read is a merge that has not been stated, and refusing to scan over it would
43/// make one unreadable line cost the whole library.
44pub fn stated(dir: &std::path::Path) -> Vec<aede_core::model::identity::Chosen> {
45    let path = user::user_path(dir);
46    let Ok(Some(data)) = user::load(&path) else {
47        return Vec::new();
48    };
49    chosen(&data)
50}
51
52/// The same list, from data already in hand.
53fn chosen(data: &UserData) -> Vec<aede_core::model::identity::Chosen> {
54    data.same_artist
55        .iter()
56        .map(|m| aede_core::model::identity::Chosen {
57            spelling: m.spelling.clone(),
58            filed_as: m.filed_as.clone(),
59        })
60        .collect()
61}

63pub fn merge(args: &Args) -> Res {
64    let dir = data_dir(args);
65    let path = user::user_path(&dir);
66    let mut data = user::load(&path)?.unwrap_or_default();
67
68    if args.has("list") {
69        // The catalog is wanted but not required: a statement can be read back
70        // before anything has ever been scanned, and refusing to show one for
71        // want of a shelf would hide exactly the statements that have not been
72        // applied yet.
73        return listed(args, &data, load(args).ok());
74    }
75    if args.has("forget") {
76        return forget(args, &mut data, &path);
77    }
78
79    let catalog = load(args)?;
80    let [spelling, filed_as] = named(args)?;
81    state(&catalog, &mut data, &path, &spelling, &filed_as)
82}
83
84/// The two names a merge is about, in the order they were typed.
85///
86/// **The order is the statement**: the first gives way to the second, which is
87/// the way it reads aloud and the way `--list` shows it back. Two positionals
88/// exactly, because "merge these" with one name is not a statement and with
89/// three it is two statements the command would have to guess how to pair.
90fn named(args: &Args) -> Result<[String; 2], Box<dyn std::error::Error>> {
91    match args.positionals.as_slice() {
92        [one, other] => Ok([one.clone(), other.clone()]),
93        _ => Err("name both spellings, the one that gives way first: \
94                  aede merge \"O. Osbourne\" \"Ozzy Osbourne\""
95            .into()),
96    }
97}

99/// `aede merge <spelling> <filed as>`: record that the two are one artist.
100fn state(
101    catalog: &Catalog,
102    data: &mut UserData,
103    path: &std::path::Path,
104    spelling: &str,
105    filed_as: &str,
106) -> Res {
107    let (from, to) = (text::normalize(spelling), text::normalize(filed_as));
108    if from.is_empty() || to.is_empty() {
109        return Err("a spelling with no letters in it names nobody".into());
110    }
111    if from == to {
112        return Err(format!(
113            "\"{spelling}\" and \"{filed_as}\" are already one name here — \
114             the shelf matches on {from}"
115        )
116        .into());
117    }
118
119    // Both sides are looked up, and neither has to exist. A spelling the shelf
120    // has never held is not a mistake: it is what a merge stated *before* the
121    // folder holding it is scanned looks like, and refusing it would make the
122    // order of two commands matter for no reason. What is worth saying is which
123    // of the two the shelf does not know, so a typo is visible at once.
124    let known = |key: &str| catalog.artists.iter().find(|a| a.key == key);
125    let (left, right) = (known(&from), known(&to));
126
127    // The one refusal. Two identifiers are two people, said so by the only
128    // authority on the question, and a statement to the contrary belongs on
129    // MusicBrainz rather than in a file on this disk.
130    if let (Some(one), Some(other)) = (left, right)
131        && let (Some(a), Some(b)) = (&one.mbid, &other.mbid)
132        && a != b
133    {
134        return Err(format!(
135            "MusicBrainz says these are two people, so they were not merged:\n\
136             \t{} · musicbrainz {a}\n\
137             \t{} · musicbrainz {b}\n\
138             If that is wrong, it is wrong at the source — correcting it there \
139             fixes it for everybody, and a later fetch brings the correction back.",
140            one.name, other.name
141        )
142        .into());
143    }
144
145    let existing = data
146        .same_artist
147        .iter_mut()
148        .find(|m| m.owner == user::LOCAL_USER && m.spelling == from);
149    match existing {
150        Some(already) if already.filed_as == to => {
151            println!("{}", ui::section("Merge"));
152            println!("  {}", ui::dim(&format!("{from} → {to} was already said")));
153            return Ok(());
154        }
155        // One spelling gives way to one artist. Saying it again with another
156        // destination is a correction, not a second statement, and keeping
157        // both would leave the shelf to arbitrate between them.
158        Some(already) => {
159            let was = already.filed_as.clone();
160            already.filed_as = to.clone();
161            already.created_at = aede_core::clock::now_seconds();
162            user::save(data, path)?;
163            println!("{}", ui::section("Merge"));
164            println!("  {} {from} → {to}", ui::green("→"));
165            println!("  {}", ui::dim(&format!("it used to be filed under {was}")));
166            println!("  {}", ui::dim(RESCAN));
167            return Ok(());
168        }
169        None => {}
170    }
171
172    data.same_artist.push(SameArtist {
173        owner: user::LOCAL_USER.to_string(),
174        spelling: from.clone(),
175        filed_as: to.clone(),
176        created_at: aede_core::clock::now_seconds(),
177    });
178    user::save(data, path)?;
179
180    println!("{}", ui::section("Merge"));
181    println!("  {} {from} → {to}", ui::green("→"));
182    for (key, found, typed) in [(&from, left, spelling), (&to, right, filed_as)] {
183        if found.is_none() {
184            println!(
185                "  {}",
186                ui::yellow(&format!(
187                    "no artist called {key} on the shelf — \"{typed}\" may be a typo, \
188                     or a folder that has not been scanned yet"
189                ))
190            );
191        }
192    }
193    println!("  {}", ui::dim(NOTHING_TOUCHED));
194    println!("  {}", ui::dim(RESCAN));
195    Ok(())
196}

198/// What every path in this command has to say once, and says in one place.
199const NOTHING_TOUCHED: &str =
200    "nothing in your files changed: this is how the shelf is read, not what it holds";
201const RESCAN: &str = "run aede scan to rebuild the shelf around it";

203/// `aede merge --list`: the statements on file, so they can be undone.
204///
205/// Each row says whether it is **in effect**. A merge is applied when the graph
206/// is built, so one stated after the last scan is a statement the shelf has not
207/// heard yet — and a listing that showed it as done would be a listing nobody
208/// could trust. The test for it is exact and costs nothing: if the catalog still
209/// holds an artist under the spelling that was to give way, the scan has not
210/// happened.
211fn listed(args: &Args, data: &UserData, catalog: Option<Catalog>) -> Res {
212    println!("{}", ui::section("Merged artists"));
213    if data.same_artist.is_empty() {
214        println!("  {}", ui::dim("nobody has been merged"));
215        println!(
216            "  {}",
217            ui::dim("aede doctor names the pairs worth looking at")
218        );
219        return Ok(());
220    }
221
222    // The words typed are matched against **both** halves, because a person
223    // hunting for a statement remembers the artist, not which side of the arrow
224    // they wrote them on. The name is never swallowed: `aede merge --list ozzy`
225    // said nothing about the word in the fourth command that did it.
226    let wanted: Vec<String> = args
227        .positionals
228        .iter()
229        .map(|n| text::normalize(n))
230        .filter(|n| !n.is_empty())
231        .collect();
232    let rows: Vec<&SameArtist> = data
233        .same_artist
234        .iter()
235        .filter(|m| super::fetch::reaches(&wanted, &[&m.spelling, &m.filed_as]))
236        .collect();
237    if rows.is_empty() {
238        println!(
239            "  {}",
240            ui::dim(&format!(
241                "no merge matches {} — {} in all",
242                wanted.join(", "),
243                ui::plural(data.same_artist.len(), "statement")
244            ))
245        );
246        return Ok(());
247    }
248
249    let waiting = |row: &SameArtist| {
250        catalog
251            .as_ref()
252            .is_some_and(|c| c.artists.iter().any(|a| a.key == row.spelling))
253    };
254    let mut table = Table::new(&["Spelling", "Filed as", "Said", "State"]).align(2, Align::Right);
255    for row in &rows {
256        table.push(vec![
257            row.spelling.clone(),
258            row.filed_as.clone(),
259            ui::since(row.created_at),
260            match waiting(row) {
261                true => "waiting for a scan".to_string(),
262                false => "in effect".to_string(),
263            },
264        ]);
265    }
266    println!("{}", table.render());
267    if rows.iter().any(|row| waiting(row)) {
268        println!("  {}", ui::yellow(RESCAN));
269    }
270    println!(
271        "  {}",
272        ui::dim("aede merge --forget \"<spelling>\" takes one back")
273    );
274    Ok(())
275}

277/// `aede merge --forget <spelling>`: take a statement back.
278///
279/// Keyed on the spelling that gives way, which is the half a person can say
280/// unambiguously: several spellings may be filed under one artist, so naming
281/// the destination would be naming a group rather than a statement.
282fn forget(args: &Args, data: &mut UserData, path: &std::path::Path) -> Res {
283    let wanted: Vec<String> = args
284        .positionals
285        .iter()
286        .map(|n| text::normalize(n))
287        .filter(|n| !n.is_empty())
288        .collect();
289    if wanted.is_empty() {
290        return Err("name the spelling to unmerge: aede merge --forget \"O. Osbourne\"".into());
291    }
292
293    let matched = |m: &SameArtist| wanted.iter().any(|w| m.spelling.contains(w.as_str()));
294    let taken: Vec<String> = data
295        .same_artist
296        .iter()
297        .filter(|m| matched(m))
298        .map(|m| format!("{} → {}", m.spelling, m.filed_as))
299        .collect();
300
301    println!("{}", ui::section("Merged artists"));
302    if taken.is_empty() {
303        println!("  {}", ui::dim("no merge answers to that"));
304        return Ok(());
305    }
306    data.same_artist.retain(|m| !matched(m));
307    user::save(data, path)?;
308    for row in &taken {
309        println!("  {} {row}", ui::green("→"));
310    }
311    println!("  {}", ui::dim("taken back"));
312    println!("  {}", ui::dim(RESCAN));
313    Ok(())
314}
