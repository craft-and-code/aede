1//! `fetch --discography`, and the report it feeds: what is missing from the shelf.
2//!
3//! # Why this is a fact about the artist
4//!
5//! A release key is `artist|title|folder`. **An album you do not own has no
6//! folder**, so a record missing from the library cannot be an entity of this
7//! layer at all — there is nothing to key it on. It is stored instead as a fact
8//! about the artist: *MusicBrainz credits fourteen albums to this name*, with
9//! their identifiers, titles and dates.
10//!
11//! Which of them you are missing is then **derived on read**, by comparing that
12//! list against the catalog. This is the same rule the layer already follows
13//! for agreement with a tag, and for the same reason: a stored "missing" goes
14//! stale the moment the record is bought, and the catalog would be left holding
15//! a claim it had stopped being able to justify. Derived, it corrects itself.
16//!
17//! # Why it is asked for rather than assumed
18//!
19//! It is one more request per artist — more for the prolific, who need a second
20//! page — on top of a run that already costs one per artist and one per album.
21//! Nobody needs a wish list in order to file their music.
22
23// Compiled in every build, for the reason `fetch` is.
24#![cfg_attr(not(feature = "fetch"), allow(dead_code))]

43/// How many pages one artist may cost, at most.
44///
45/// A hundred release groups per page, so this covers an artist with a thousand
46/// of them — beyond any real discography, and the point is not the number but
47/// that the loop has an end. A miscounted total from the service must not turn
48/// into a request per second for ever.
49const MAX_PAGES: usize = 10;
50
51/// The pass, over artists MusicBrainz has already identified.
52pub fn run(
53    catalog: &Catalog,
54    transport: &mut dyn Ask,
55    backoff: &[std::time::Duration],
56    held: &mut sources::Sources,
57    path: &std::path::Path,
58    asked: &super::fetch::Asked,
59) -> Res {
60    let (wanted, again) = (asked.names, asked.again);
61    let targets = targets(catalog, held, wanted, asked.scope, again);
62    println!("{}", ui::section("Discography"));
63    if targets.is_empty() {
64        if !wanted.is_empty() || !asked.scope.is_empty() {
65            println!(
66                "  {}",
67                ui::dim(&super::fetch::nothing_named(
68                    wanted,
69                    asked.scope,
70                    self::targets(catalog, held, wanted, asked.scope, true).len()
71                ))
72            );
73            return Ok(());
74        }
75        println!(
76            "  {}",
77            ui::dim(
78                "nothing to browse: run fetch first, and note that only artists \
79                 with an album of their own here are browsed"
80            )
81        );
82        return Ok(());
83    }
84
85    // At least one request each, and a second page only for the prolific — so
86    // the estimate is a floor rather than a promise, and says so.
87    let total_ms = targets.len() as u64 * musicbrainz::REQUEST_INTERVAL.as_millis() as u64;
88    println!(
89        "  {}, at least one request each, about {} — more for the prolific",
90        ui::plural(targets.len(), "artist"),
91        ui::long_duration(total_ms)
92    );
93
94    if super::fetch::asked_nothing(
95        asked,
96        &targets.iter().map(|t| t.name.clone()).collect::<Vec<_>>(),
97    ) {
98        return Ok(());
99    }
100
101    let (mut stored, mut empty, mut failed) = (0usize, 0usize, 0usize);
102    let mut pending = queue(&targets);
103    let mut done = 0usize;
104    let total = targets.len();
105    while let Some((target, retried)) = pending.pop_front() {
106        print!("\r  browsing: {}/{}", done + 1, total);
107        let _ = std::io::Write::flush(&mut std::io::stdout());
108
109        match browse(transport, backoff, &target.mbid) {
110            Ok(known) => {
111                match known.is_empty() {
112                    true => empty += 1,
113                    false => stored += 1,
114                }
115                store(held, target, known);
116                sources::save(held, path)?;
117            }
118            Err(why) if defer(&mut pending, target, retried, &why) => {
119                continue;
120            }
121            Err(why) => {
122                failed += 1;
123                eprintln!("\r  {} {}: {why}", ui::red("×"), target.name);
124            }
125        }
126        done += 1;
127    }
128    println!();
129
130    println!(
131        "{} {stored} stored, {empty} with nothing credited, {failed} failed",
132        ui::green("→")
133    );
134    println!(
135        "  {}",
136        ui::dim("aede missing lists the records this catalog does not hold")
137    );
138    Ok(())
139}

141/// Every page of one artist's discography, in order.
142fn browse(
143    transport: &mut dyn Ask,
144    backoff: &[std::time::Duration],
145    mbid: &str,
146) -> Result<Vec<KnownRelease>, Refusal> {
147    let mut all: Vec<KnownRelease> = Vec::new();
148    for page in 0..MAX_PAGES {
149        let url = musicbrainz::discography_url(mbid, page * musicbrainz::BROWSE_LIMIT);
150        let answer = ask_with_backoff(transport, &url, backoff)?;
151        let (rows, total) = musicbrainz::discography(&answer);
152        // A page that came back empty ends the walk whatever the total says:
153        // asking again for the same nothing is how a miscount becomes a loop.
154        let arrived = rows.len();
155        all.extend(rows);
156        if arrived == 0 || all.len() >= total {
157            break;
158        }
159    }
160    Ok(all)
161}

163/// Files the discography **into the artist's existing MusicBrainz record**.
164///
165/// Read, add the list, write back — rather than a record of its own. The layer
166/// is keyed on (entity, source), so a second record from the same source about
167/// the same artist is not a second opinion but a lost one: `set` would replace
168/// the first, and the area, the dates and the links would vanish.
169fn store(held: &mut sources::Sources, target: &Target, known: Vec<KnownRelease>) {
170    let mut facts = match held.get(&target.entity, sources::MUSICBRAINZ) {
171        Some(SourceRecord {
172            facts: Facts::Artist(existing),
173            ..
174        }) => existing.clone(),
175        _ => ArtistFacts::default(),
176    };
177    facts.discography = known;
178    held.set(SourceRecord {
179        key: target.entity.key.clone(),
180        source: sources::MUSICBRAINZ.to_string(),
181        source_id: Some(target.mbid.clone()),
182        fetched_at: clock::now_seconds(),
183        // Browsed by identifier: nothing here was matched by name.
184        confidence: sources::Confidence::Identified,
185        facts: Facts::Artist(facts),
186    });
187}

189/// Artists worth browsing: identified, and with a shelf of their own.
190///
191/// Two conditions, and the second is the one that matters. The catalog holds an
192/// artist for every credit it reads, so a guest on one track of a compilation is
193/// an artist of this library in exactly the way a musician whose records fill a
194/// folder is not. Browsing them costs a request a second for a discography
195/// [`absent`] will never report — the two must agree on who has a shelf, or the
196/// pass spends minutes fetching answers nothing can use.
197fn targets(
198    catalog: &Catalog,
199    held: &sources::Sources,
200    wanted: &[String],
201    scope: &super::fetch::Scope,
202    again: bool,
203) -> Vec<Target> {
204    let mut targets = Vec::new();
205    for record in &held.records {
206        if record.source != sources::MUSICBRAINZ {
207            continue;
208        }
209        if !super::fetch::reaches(wanted, &[record.key.as_str()]) {
210            continue;
211        }
212        // A record's key *is* the artist's catalog key, which is what makes a
213        // folder answerable here at all: this pass reads the attributed layer
214        // and never the catalog, so the folder was turned into a set of keys
215        // once, in `fetch`, before any pass ran.
216        if !scope.has_artist(&record.key) {
217            continue;
218        }
219        let Facts::Artist(artist) = &record.facts else {
220            continue;
221        };
222        let Some(mbid) = record.source_id.clone() else {
223            continue;
224        };
225        if !again && !artist.discography.is_empty() {
226            continue;
227        }
228        let entity = record.entity();
229        if !has_shelf(catalog, &entity) {
230            continue;
231        }
232        targets.push(Target {
233            entity,
234            name: record.key.clone(),
235            mbid,
236        });
237    }
238    targets
239}

241/// How many artists a `--discography` pass would browse, if it ran now.
242pub fn waiting(catalog: &Catalog, held: &sources::Sources) -> usize {
243    targets(catalog, held, &[], &super::fetch::EVERYTHING, false).len()
244}

246/// `true` when this artist is the album artist of something in the library.
247///
248/// The one question both halves of this module have to answer the same way:
249/// who has a shelf here. Being in the catalog is not it — a name credited on
250/// one track of a compilation is in the catalog and has no place of its own.
251fn has_shelf(catalog: &Catalog, entity: &EntityRef) -> bool {
252    let Some(id) = entity.resolve(catalog) else {
253        return false;
254    };
255    catalog
256        .releases
257        .iter()
258        .any(|release| release.album_artist_id == Some(id))
259}

265/// One record credited to an artist that this catalog does not hold.
266pub struct Absent<'a> {
267    /// The artist, as the catalog spells them.
268    pub artist: String,
269    /// Why this row is not normally on the report, and `None` when it is.
270    ///
271    /// Two quite different reasons, in one field because they answer one
272    /// question — *why am I not being shown this* — and because a record can
273    /// be both: a live album the reader also set aside. The words for the
274    /// first are MusicBrainz's own; see [`KnownRelease::stated_type`].
275    pub held_back: Option<String>,
276    /// `true` when the reader set this one aside and asked to see it anyway.
277    pub set_aside: bool,
278    /// Which artist, for a caller asking about one rather than about all.
279    pub artist_id: aede_core::model::Id,
280    /// The record MusicBrainz credits to them.
281    pub known: &'a KnownRelease,
282}

284/// What is credited to your artists and not on your shelf.
285///
286/// **Derived, never stored.** The comparison happens here, at the moment of
287/// reading, so the day an album is added to the library it stops being listed —
288/// with nothing to update and nothing to go stale.
289///
290/// Two ways of recognising a record you already have, in this order:
291///
292/// 1. the **release-group identifier**, when your tags carry one. Exact, and
293///    the only comparison that cannot be wrong.
294/// 2. the **normalised title**, otherwise. `text::normalize` is the same
295///    function that decides two spellings are one name everywhere else in this
296///    program, so a shelf that says "Kind Of Blue" is not told it is missing
297///    "Kind of Blue".
298///
299/// Two things are held back, and `everything` lifts both.
300///
301/// Only studio albums are reported — see [`KnownRelease::is_studio_album`] for
302/// why a complete discography would be a true and useless answer — and only
303/// records the reader has not set aside. Those are the report's two filters,
304/// and `--all` is one word for "hold nothing back", so it lifts both and every
305/// row that came back this way carries the reason it would not normally be
306/// here. **A filter is only honest when the reader can turn it off**: saying
307/// what is left out and offering no way to see it is a locked door with a
308/// label on it.
309pub fn absent<'a>(
310    catalog: &Catalog,
311    held: &'a sources::Sources,
312    aside: &[aede_core::user::SetAside],
313    everything: bool,
314) -> Vec<Absent<'a>> {
315    let mut out: Vec<Absent<'a>> = Vec::new();
316    for record in &held.records {
317        if record.source != sources::MUSICBRAINZ {
318            continue;
319        }
320        if !held.is_trusted(catalog, record) {
321            continue;
322        }
323        let Facts::Artist(facts) = &record.facts else {
324            continue;
325        };
326        if facts.discography.is_empty() {
327            continue;
328        }
329        let entity = record.entity();
330        // An artist the catalog cannot place has no shelf to be missing from.
331        let Some(artist_id) = entity.resolve(catalog) else {
332            continue;
333        };
334        let Some(artist) = catalog.artist(artist_id) else {
335            continue;
336        };
337
338        // What is on the shelf, in both spellings the comparison uses.
339        let mut ids: Vec<&str> = Vec::new();
340        let mut titles: Vec<String> = Vec::new();
341        for release in &catalog.releases {
342            if release.album_artist_id != Some(artist_id) {
343                continue;
344            }
345            if let Some(group) = release.release_group_mbid.as_deref() {
346                ids.push(group);
347            }
348            titles.push(text::normalize(&release.title));
349        }
350
351        // **No album of their own, no shelf to have gaps in.** The catalog
352        // holds an artist for every credit it reads — a guest on one track, a
353        // composer, one name on a compilation — and being in the catalog is
354        // not the same as having a place in the library. The first version
355        // missed that distinction and answered a question nobody asked: one
356        // Rolling Stones track on a compilation produced their entire studio
357        // discography as "missing", which is true and worthless, and it did
358        // so for every passing credit at once until the report was mostly
359        // that. What this reports is an *incomplete* discography, which means
360        // one that was started.
361        //
362        // The same rule decides who is browsed at all — see [`has_shelf`].
363        if titles.is_empty() {
364            continue;
365        }
366
367        for known in &facts.discography {
368            // Set aside by hand. Filtered here rather than by dropping it from
369            // the layer, because the two are different claims and this program
370            // keeps them apart everywhere else: MusicBrainz still says this is
371            // an album, and the user still says it is not one they want listed.
372            // Deleting the record would lose the first to record the second.
373            let put_aside = aside.iter().any(|a| a.release_group == known.mbid);
374
375            // Both reasons, in the order a reader meets them: what it is, then
376            // what they decided about it. A record can be both, and naming only
377            // one would answer half the question.
378            let mut why: Vec<String> = Vec::new();
379            if !known.is_studio_album() {
380                why.push(known.stated_type());
381            }
382            if put_aside {
383                why.push("set aside".to_string());
384            }
385            let held_back = (!why.is_empty()).then(|| why.join(", "));
386            if held_back.is_some() && !everything {
387                continue;
388            }
389
390            let have = ids.iter().any(|id| *id == known.mbid)
391                || titles.contains(&text::normalize(&known.title));
392            if !have {
393                out.push(Absent {
394                    artist: artist.name.clone(),
395                    held_back,
396                    set_aside: put_aside,
397                    artist_id,
398                    known,
399                });
400            }
401        }
402    }
403    // Oldest first within an artist, so a discography reads as one.
404    out.sort_by(|a, b| {
405        a.artist
406            .cmp(&b.artist)
407            .then_with(|| a.known.first_released.cmp(&b.known.first_released))
408    });
409    out
410}

412/// How many records one artist has that this shelf does not.
413///
414/// For a page to say so at the moment somebody is looking at that artist —
415/// which is the moment they wonder. `aede missing` was in the help, in the
416/// README and in the manual, and still could not be found, because it was
417/// named nowhere near the question it answers. The same fault `aede extract`
418/// had, and the same fix.
419///
420/// Derived like everything else here: nothing is stored, so the number falls
421/// the day an album is added.
422pub fn absent_for(
423    catalog: &Catalog,
424    held: &sources::Sources,
425    aside: &[aede_core::user::SetAside],
426    artist_id: aede_core::model::Id,
427) -> usize {
428    absent(catalog, held, aside, false)
429        .iter()
430        .filter(|record| record.artist_id == artist_id)
431        .count()
432}

434/// The `missing` command: the records your artists made and you do not have.
435pub fn missing(args: &crate::args::Args) -> Res {
436    let catalog = super::load(args)?;
437    let held = super::sources_held(args)?;
438    let user_path = aede_core::user::user_path(&super::data_dir(args));
439    let mut user = aede_core::user::load(&user_path)?.unwrap_or_default();
440
441    if args.has("forget") {
442        return set_aside(args, &catalog, &held, &mut user, &user_path);
443    }
444    let wanted = super::fetch::names_given(args);
445
446    if args.has("list") {
447        return listed(&catalog, &held, &user, &wanted);
448    }
449
450    // "Hold nothing back", the same word as on every other listing, and here
451    // it lifts everything this report holds back at once: the row limit, the
452    // studio-album filter, and what the reader set aside. Three things, one
453    // intent — *show me all of what the fetch brought back* — and a separate
454    // option for each would be three words to learn for one question.
455    let everything = args.has("all");
456    let window = args.window(50)?;
457
458    let browsed = held.records.iter().any(|r| {
459        held.is_trusted(&catalog, r)
460            && matches!(&r.facts, Facts::Artist(a) if !a.discography.is_empty())
461    });
462    if !browsed {
463        println!("{}", ui::section("Missing"));
464        // Nothing browsed is not an empty shelf, and a reader who cannot tell
465        // which will conclude the wrong one — the same distinction `sources`
466        // draws when nothing has been fetched at all.
467        println!(
468            "  {}",
469            ui::dim(
470                "no discography has been fetched, so nothing can be missing yet: \
471                 aede fetch --discography asks for one"
472            )
473        );
474        return Ok(());
475    }
476
477    let all = absent(&catalog, &held, &user.set_aside, everything);
478    let rows: Vec<&Absent> = all.iter().filter(|a| matches(a, &wanted)).collect();
479    // What the ordinary run is leaving out, counted over the rows this run is
480    // about rather than over the whole layer: a reader who typed a name is owed
481    // the number for that name. Derived from the same walk that produces the
482    // table, so the note and the list cannot disagree.
483    let held_back = match everything {
484        true => Vec::new(),
485        false => absent(&catalog, &held, &user.set_aside, true)
486            .into_iter()
487            .filter(|a| a.held_back.is_some() && matches(a, &wanted))
488            .collect(),
489    };
490
491    println!("{}", ui::section("Missing"));
492    if rows.is_empty() {
493        println!(
494            "  {}",
495            ui::dim(match (wanted.is_empty(), everything) {
496                (true, true) => "every record MusicBrainz credits to your artists is here",
497                (true, false) => "every studio album MusicBrainz credits to your artists is here",
498                (false, _) => "nothing matching that is missing",
499            })
500        );
501        left_out(&held_back, everything);
502        return Ok(());
503    }
504
505    // The last column exists only when there is something to put in it: a blank
506    // column on every ordinary run is a question the reader has to ask and
507    // answer for themselves.
508    let mut headings = vec!["Artist", "Album", "Year"];
509    if everything {
510        headings.push("Left out");
511    }
512    let mut table = crate::ui::Table::new(&headings).limit(1, 46);
513    let total = rows.len();
514    for row in rows.iter().skip(window.offset).take(window.limit) {
515        let mut cells = vec![
516            row.artist.clone(),
517            row.known.title.clone(),
518            row.known.year().unwrap_or("").to_string(),
519        ];
520        if everything {
521            cells.push(row.held_back.clone().unwrap_or_default());
522        }
523        table.push(cells);
524    }
525    print!("{}", table.render());
526    println!(
527        "  {}",
528        ui::plural(
529            total,
530            match everything {
531                true => "record",
532                false => "studio album",
533            }
534        )
535    );
536    super::announce_window(window, total, "record");
537    left_out(&held_back, everything);
538    Ok(())
539}

541/// `true` when a row answers to one of the words the reader typed.
542fn matches(row: &Absent, wanted: &[String]) -> bool {
543    let artist = text::normalize(&row.artist);
544    let title = text::normalize(&row.known.title);
545    wanted.is_empty()
546        || wanted
547            .iter()
548            .any(|w| artist.contains(w.as_str()) || title.contains(w.as_str()))
549}
550
551/// Says what this run held back, how much of it, and how to see it.
552///
553/// **A filter the reader cannot see is a trap, and one they cannot turn off is
554/// only half an answer.** The count is of the rows *this* run held back, so it
555/// falls with the name typed and cannot drift from the table above it; the way
556/// to see them is named in the same breath, because a report that says "some
557/// records are not shown" and stops there has told the reader they are missing
558/// something and left them no move.
559///
560/// The two reasons are named separately even though `--all` lifts both: one is
561/// this program's editorial decision about what a wish list is for, the other
562/// is a decision the reader took themselves, and only the second has a listing
563/// of its own to undo it from.
564fn left_out(rows: &[Absent], everything: bool) {
565    if everything {
566        println!(
567            "  {}",
568            ui::dim(
569                "nothing held back: everything the discography pass brought back, \
570                 and the last column is what MusicBrainz calls each one"
571            )
572        );
573        return;
574    }
575    if rows.is_empty() {
576        // Still stated with nothing to state it about: it is the shape of the
577        // report rather than an incident, and a reader who is not told will
578        // read the list as everything MusicBrainz knows.
579        println!(
580            "  {}",
581            ui::dim("singles, live records and compilations are left out")
582        );
583        return;
584    }
585    println!(
586        "  {}",
587        ui::dim(&format!(
588            "{} left out — --all lists them here, with the reason for each: \
589             singles, live records, compilations, demos, and anything you set aside",
590            ui::plural(rows.len(), "record")
591        ))
592    );
593    let aside = rows.iter().filter(|row| row.set_aside).count();
594    if aside > 0 {
595        println!(
596            "  {}",
597            ui::dim(&format!(
598                "{} among them you set aside yourself — --list shows those on their own",
599                ui::plural(aside, "record")
600            ))
601        );
602    }
603}

605/// Which artist a set-aside record belongs to, as the stored discographies say.
606///
607/// Not kept on the record itself: a set-aside is keyed on the release group,
608/// which is globally unique, so no artist is needed to tell two apart. But a
609/// listing that shows only titles is a listing nobody can act on, and the
610/// answer is already in the layer — derived at the moment of reading, like
611/// everything else here, so it cannot go stale.
612///
613/// Empty when the discography that named it has since been forgotten: the row
614/// is still shown, because a decision the reader took must not disappear
615/// because a fetch was undone.
616fn whose(catalog: &Catalog, held: &sources::Sources, release_group: &str) -> String {
617    for record in &held.records {
618        if record.source != sources::MUSICBRAINZ {
619            continue;
620        }
621        let Facts::Artist(facts) = &record.facts else {
622            continue;
623        };
624        if !facts.discography.iter().any(|k| k.mbid == release_group) {
625            continue;
626        }
627        // The catalog's spelling when it holds the artist, the layer's key
628        // otherwise — a name is better than nothing, and this row exists to
629        // be recognised.
630        return record
631            .entity()
632            .resolve(catalog)
633            .and_then(|id| catalog.artist(id))
634            .map(|artist| artist.name.clone())
635            .unwrap_or_else(|| record.key.clone());
636    }
637    String::new()
638}

640/// `aede missing --forget <text>`: take a record off the report, or put it back.
641fn set_aside(
642    args: &crate::args::Args,
643    catalog: &Catalog,
644    held: &sources::Sources,
645    user: &mut aede_core::user::UserData,
646    path: &std::path::Path,
647) -> Res {
648    let wanted: Vec<String> = args
649        .positionals
650        .iter()
651        .map(|name| text::normalize(name))
652        .filter(|name| !name.is_empty())
653        .collect();
654    if wanted.is_empty() {
655        return Err("name the album to set aside: aede missing --forget \"<title>\"".into());
656    }
657
658    // Putting one back is answered from the list of decisions, not from the
659    // report: a record that has been set aside is by definition absent from the
660    // report, so looking there would find nothing and say so.
661    if args.has("remove") {
662        let before = user.set_aside.len();
663        let put_back: Vec<String> = user
664            .set_aside
665            .iter()
666            .filter(|a| {
667                let title = text::normalize(&a.title);
668                wanted.iter().any(|w| title.contains(w.as_str()))
669            })
670            .map(|a| a.title.clone())
671            .collect();
672        user.set_aside.retain(|a| {
673            let title = text::normalize(&a.title);
674            !wanted.iter().any(|w| title.contains(w.as_str()))
675        });
676        println!("{}", ui::section("Missing"));
677        if user.set_aside.len() == before {
678            println!("  {}", ui::dim("nothing set aside answers to that"));
679            return Ok(());
680        }
681        aede_core::user::save(user, path)?;
682        for title in &put_back {
683            println!("  {} {title}", ui::green("→"));
684        }
685        println!("  {}", ui::dim("back on the list"));
686        return Ok(());
687    }
688
689    let all = absent(catalog, held, &user.set_aside, false);
690    let found: Vec<&Absent> = all.iter().filter(|a| matches(a, &wanted)).collect();
691    match found.len() {
692        0 => Err("nothing missing answers to that".into()),
693        // Several equally good answers are refused rather than arbitrated —
694        // the rule `best_match` and `find_releases` both follow. Setting aside
695        // the wrong record is a decision nobody would see being taken.
696        n if n > 1 => {
697            let names: Vec<String> = found
698                .iter()
699                .map(|a| format!("{} — {}", a.artist, a.known.title))
700                .collect();
701            Err(format!(
702                "several records answer to that, so none was set aside:\n  {}",
703                names.join("\n  ")
704            )
705            .into())
706        }
707        _ => {
708            let one = found[0];
709            user.set_aside.push(aede_core::user::SetAside {
710                owner: aede_core::user::LOCAL_USER.to_string(),
711                release_group: one.known.mbid.clone(),
712                title: one.known.title.clone(),
713                created_at: aede_core::clock::now_seconds(),
714            });
715            aede_core::user::save(user, path)?;
716            println!("{}", ui::section("Missing"));
717            println!("  {} {} — {}", ui::green("→"), one.artist, one.known.title);
718            println!(
719                "  {}",
720                ui::dim(
721                    "set aside; --forget --remove puts it back. What MusicBrainz \
722                     says is untouched — only what you are shown changed"
723                )
724            );
725            Ok(())
726        }
727    }
728}
729
730/// The set-aside decisions a run is about, each with the artist it belongs to.
731///
732/// The names typed used to be **swallowed**: `aede missing "MIKA" --list`
733/// listed every decision on file and said nothing about the word. The fourth
734/// time a command in this program has quietly dropped its argument, and the
735/// reason it keeps happening is that each listing wrote its own matching. This
736/// one calls [`super::fetch::reaches`], which is where that decision lives.
737///
738/// Split out of [`listed`] so the narrowing can be tested without reading what
739/// was printed: a filter is a claim about which rows survive, and that is the
740/// part worth pinning down.
741fn aside_rows<'a>(
742    catalog: &Catalog,
743    held: &sources::Sources,
744    user: &'a aede_core::user::UserData,
745    wanted: &[String],
746) -> Vec<(&'a aede_core::user::SetAside, String)> {
747    user.set_aside
748        .iter()
749        .map(|aside| (aside, whose(catalog, held, &aside.release_group)))
750        .filter(|(aside, artist)| super::fetch::reaches(wanted, &[&aside.title, artist]))
751        .collect()
752}
753
754/// `aede missing --list`: the decisions taken, so they can be undone.
755fn listed(
756    catalog: &Catalog,
757    held: &sources::Sources,
758    user: &aede_core::user::UserData,
759    wanted: &[String],
760) -> Res {
761    println!("{}", ui::section("Set aside"));
762    if user.set_aside.is_empty() {
763        println!("  {}", ui::dim("nothing has been set aside"));
764        return Ok(());
765    }
766
767    let rows = aside_rows(catalog, held, user, wanted);
768    if rows.is_empty() {
769        println!(
770            "  {}",
771            ui::dim(&format!(
772                "nothing set aside matches {} — {} in all",
773                wanted.join(", "),
774                ui::plural(user.set_aside.len(), "record")
775            ))
776        );
777        return Ok(());
778    }
779
780    // The artist was missing from this table, and "Sweet Dreams" on its own
781    // tells a reader nothing about whose decision they are looking at. It is
782    // not stored on the record — a set-aside is keyed on the release group,
783    // which is globally unique — so it is resolved from the discographies
784    // that named it, the way everything else here is derived rather than kept.
785    let mut table = crate::ui::Table::new(&["Artist", "Album", "Identifier"]).limit(1, 40);
786    for (aside, artist) in &rows {
787        table.push(vec![
788            artist.clone(),
789            aside.title.clone(),
790            format!(
791                "https://musicbrainz.org/release-group/{}",
792                aside.release_group
793            ),
794        ]);
795    }
796    print!("{}", table.render());
797    println!(
798        "  {}",
799        ui::dim(
800            "aede missing --forget --remove \"<title>\" puts one back; the address \
801             is where its type is corrected for everybody"
802        )
803    );
804    Ok(())
805}
