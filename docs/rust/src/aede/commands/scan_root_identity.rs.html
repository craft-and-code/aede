1//! Scan-only resolution of declared unavailable roots without losing catalog identity.
2
3use aede_core::model::Catalog;
4use std::{
5    collections::{BTreeMap, BTreeSet, VecDeque},
6    error::Error,
7    ffi::OsString,
8    fs, io,
9    path::{Component, Path, PathBuf},
10};
11
12const MAX_LINKS: usize = 40;
13const MAX_COMPONENTS: usize = 4096;
14
15pub(super) fn resolve(path: &Path) -> Result<PathBuf, Box<dyn Error>> {
16    resolve_path(path).map_err(|error| identity_error(path, &error.to_string()))
17}
18
19fn identity_error(path: &Path, reason: &str) -> Box<dyn Error> {
20    format!(
21        "cannot establish root identity for {path:?}: {reason}; reconnect the folder or select its real path before replacing the catalog"
22    )
23    .into()
24}
25
26fn resolve_path(path: &Path) -> io::Result<PathBuf> {
27    // Some platform canonicalizers erase file/.. or missing/.. if the final target exists.
28    // Inspect those ancestors before allowing a parent component to change identity.
29    if !path.components().any(|part| part == Component::ParentDir)
30        && let Ok(resolved) = path.canonicalize()
31    {
32        return utf8_path(resolved);
33    }
34    let mut pending = absolute_components(path)?;
35    let mut resolved = PathBuf::new();
36    let mut missing = false;
37    let mut links = 0;
38    let mut components = 0;
39    while let Some(part) = pending.pop_front() {
40        components += 1;
41        if components > MAX_COMPONENTS {
42            return Err(io::Error::other("too many path components"));
43        }
44        match Path::new(&part).components().next() {
45            Some(Component::Prefix(_) | Component::RootDir) => resolved.push(&part),
46            Some(Component::CurDir) => {}
47            Some(Component::ParentDir) => {
48                // An absent component could previously have been a link. Its
49                // parent cannot be inferred by flattening the spelling.
50                if missing {
51                    return Err(io::Error::other("parent of an absent component is unknown"));
52                }
53                resolved.pop();
54            }
55            Some(Component::Normal(_)) => {
56                if missing {
57                    resolved.push(&part);
58                    continue;
59                }
60                resolved.push(&part);
61                match fs::symlink_metadata(&resolved) {
62                    Ok(metadata) if metadata.file_type().is_symlink() => {
63                        links += 1;
64                        if links > MAX_LINKS {
65                            return Err(io::Error::other("too many symbolic links"));
66                        }
67                        let target = fs::read_link(&resolved)?;
68                        if target.is_absolute() {
69                            resolved.clear();
70                        } else {
71                            resolved.pop();
72                        }
73                        for part in target.components().rev() {
74                            pending.push_front(part.as_os_str().to_owned());
75                        }
76                    }
77                    Ok(metadata) if !metadata.is_dir() && !pending.is_empty() => {
78                        return Err(io::Error::other("path ancestor is not a directory"));
79                    }
80                    Ok(_) => {}
81                    Err(error) if error.kind() == io::ErrorKind::NotFound => {
82                        resolved.pop();
83                        // Preserve the platform's canonical spelling of every
84                        // existing ancestor before appending missing names.
85                        resolved = resolved.canonicalize()?;
86                        resolved.push(&part);
87                        missing = true;
88                    }
89                    Err(error) => return Err(error),
90                }
91            }
92            None => {}
93        }
94    }
95    utf8_path(if missing {
96        resolved
97    } else {
98        resolved.canonicalize()?
99    })
100}

102fn absolute_components(path: &Path) -> io::Result<VecDeque<OsString>> {
103    if path.as_os_str().is_empty() {
104        return Err(io::Error::new(
105            io::ErrorKind::InvalidInput,
106            "empty root path",
107        ));
108    }
109    let mut parts = path.components();
110    if path.is_absolute() {
111        return Ok(parts.map(|part| part.as_os_str().to_owned()).collect());
112    }
113    // Resolve only the base, never the unvalidated tail: Windows absolute/join
114    // can erase `..`. Prefix-only resolution retains per-drive/root-relative semantics.
115    let base = match parts.clone().next() {
116        Some(Component::Prefix(prefix)) => {
117            parts.next();
118            std::path::absolute(prefix.as_os_str())?
119        }
120        Some(Component::RootDir) => {
121            parts.next();
122            std::path::absolute(std::path::MAIN_SEPARATOR_STR)?
123        }
124        _ => std::env::current_dir()?,
125    };
126    Ok(base
127        .components()
128        .map(|part| part.as_os_str().to_owned())
129        .chain(parts.map(|part| part.as_os_str().to_owned()))
130        .collect())
131}
132
133fn utf8_path(path: PathBuf) -> io::Result<PathBuf> {
134    if path.to_str().is_none() {
135        return Err(io::Error::other("path cannot be represented as UTF-8"));
136    }
137    Ok(path)
138}

140pub(super) fn validate_selection(
141    previous: Option<&Catalog>,
142    roots: &[PathBuf],
143    unavailable: &[(PathBuf, PathBuf)],
144) -> Result<(), Box<dyn Error>> {
145    let Some(previous) = previous else {
146        return Ok(());
147    };
148    if previous
149        .roots
150        .iter()
151        .all(|old| roots.iter().any(|root| Path::new(old).starts_with(root)))
152    {
153        return Ok(());
154    }
155    for (raw, identity) in unavailable {
156        // The catalog retains canonical paths, not a map of configured
157        // aliases. A retargeted or removed link cannot identify which old root
158        // was intentionally dropped, even if its new target is also known.
159        if raw != identity || !known_identity(previous, identity) {
160            return Err(identity_error(
161                raw,
162                "the previous root association is unknown",
163            ));
164        }
165    }
166    Ok(())
167}
168
169fn known_identity(previous: &Catalog, identity: &Path) -> bool {
170    previous.roots.iter().any(|old| Path::new(old) == identity)
171        || previous
172            .files
173            .iter()
174            .any(|file| Path::new(&file.path).starts_with(identity))
175}

177pub(super) fn validate_result(
178    previous: Option<&Catalog>,
179    current: &Catalog,
180    unavailable: &[(PathBuf, PathBuf)],
181    excluded: &[PathBuf],
182) -> Result<(), Box<dyn Error>> {
183    let Some(previous) = previous else {
184        return Ok(());
185    };
186    let Some((raw, _)) = unavailable
187        .iter()
188        .find(|(raw, identity)| raw != identity || !known_identity(previous, identity))
189    else {
190        return Ok(());
191    };
192    let present: BTreeSet<_> = current
193        .files
194        .iter()
195        .map(|file| file.path.as_str())
196        .collect();
197    let mut excluded_directories = BTreeMap::new();
198    // Root deduplication can hide an alias's former target beneath another
199    // watched parent. Root coverage alone cannot establish that its files
200    // survived; refuse before publishing any graph or private-store changes.
201    if previous.files.iter().any(|old| {
202        !present.contains(old.path.as_str())
203            && !excluded_file(Path::new(&old.path), excluded, &mut excluded_directories)
204    }) {
205        return Err(identity_error(
206            raw,
207            "the scan would remove files while the previous root association is unknown",
208        ));
209    }
210    Ok(())
211}

213fn excluded_file(
214    path: &Path,
215    excluded: &[PathBuf],
216    directories: &mut BTreeMap<PathBuf, Option<PathBuf>>,
217) -> bool {
218    if excluded.is_empty() {
219        return false;
220    }
221    if excluded.iter().any(|root| path.starts_with(root)) {
222        return true;
223    }
224    let (Some(directory), Some(name)) = (path.parent(), path.file_name()) else {
225        return false;
226    };
227    let Some(identity) = directories
228        .entry(directory.to_path_buf())
229        .or_insert_with(|| resolve_path(directory).ok())
230        .as_ref()
231    else {
232        return false;
233    };
234    // Only an established exclusion can waive the missing-file guard. Unknown
235    // historical aliases still require recovery before any store is replaced.
236    let identity = identity.join(name);
237    if excluded.iter().any(|root| identity.starts_with(root)) {
238        return true;
239    }
240    if fs::symlink_metadata(&identity).is_ok_and(|metadata| metadata.file_type().is_symlink()) {
241        return resolve_path(&identity)
242            .is_ok_and(|identity| excluded.iter().any(|root| identity.starts_with(root)));
243    }
244    false
245}
