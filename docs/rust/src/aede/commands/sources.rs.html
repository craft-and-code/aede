1//! The `sources` command: what somebody else said about this library.
2//!
3//! The catalog holds what the files say, the annotation file holds what you
4//! say, and `sources.json` holds what a source says — MusicBrainz first, a
5//! plugin later. This is the command that shows it, takes it in, and drops it.
6//!
7//! It exists before anything fetches, on purpose. A store that can only be
8//! filled over the network can only be tested over the network, and the layer
9//! is worth more than that: `--import` reads a document in exactly the shape
10//! the file has, so the whole of it — attachment, agreement, removal — can be
11//! exercised offline, and so can a bug report.
12//!
13//! The shape follows `aede import`, which does the same job for another tool's
14//! analyses: `--list` says what is held, `--forget` removes it, `--source`
15//! narrows to one. Two stores of what somebody else said should not be learned
16//! twice.
17

26pub fn sources(args: &Args) -> Res {
27    let operations: Vec<&str> = ["import", "template", "export", "forget", "list"]
28        .into_iter()
29        .filter(|option| args.has(option))
30        .collect();
31    if operations.len() > 1 {
32        return Err(
33            "choose one sources operation: --import, --template, --export, --forget, or --list"
34                .into(),
35        );
36    }
37    if args.has("source") {
38        if !args.has("list") && !args.has("forget") && !args.has("template") {
39            return Err("--source is used with --list, --forget, or --template".into());
40        }
41        if args
42            .value("source")
43            .is_none_or(|value| value.trim().is_empty())
44        {
45            return Err("--source expects a non-empty source name".into());
46        }
47    }
48    if args.has("output") && !args.has("export") && !args.has("template") {
49        return Err("--output is used with --export or --template".into());
50    }
51    if !args.positionals.is_empty() && !args.has("template") {
52        return Err("names select entities only with --template on sources".into());
53    }
54    let dir = data_dir(args);
55    let path = sources::sources_path(&dir);
56
57    if args.has("import") {
58        return import(args, &path);
59    }
60    if args.has("template") {
61        return template(args);
62    }
63    if args.has("export") {
64        return export(args, &path);
65    }
66    if args.has("forget") {
67        return forget(args, &path);
68    }
69
70    let held = sources::load(&path)?.unwrap_or_default();
71    if held.records.is_empty() {
72        println!("{}", ui::section("Sources"));
73        println!("  {}", ui::dim("nothing has been fetched yet"));
74        // The store being empty is not the same as the feature being absent,
75        // and a reader who cannot tell which will conclude the wrong one.
76        println!(
77            "  {}",
78            ui::dim("aede sources --import <file> takes in a document by hand")
79        );
80        return Ok(());
81    }
82
83    let catalog = load(args)?;
84    match args.has("list") {
85        true => list(&held, &catalog, args),
86        false => summary(&held, &catalog),
87    }
88}

90/// One line per source: how much it said, and how much of it lands.
91fn summary(held: &Sources, catalog: &Catalog) -> Res {
92    println!("{}", ui::section("Sources"));
93
94    let mut names: Vec<&str> = held.records.iter().map(|r| r.source.as_str()).collect();
95    names.sort_unstable();
96    names.dedup();
97
98    let mut table = Table::new(&["Source", "Records", "Attached", "Waiting", "Last fetch"])
99        .align(1, Align::Right)
100        .align(2, Align::Right)
101        .align(3, Align::Right);
102
103    for name in &names {
104        let mine = Sources {
105            records: held
106                .records
107                .iter()
108                .filter(|r| r.source == *name)
109                .cloned()
110                .collect(),
111            reviews: held
112                .reviews
113                .iter()
114                .filter(|review| review.source == *name)
115                .cloned()
116                .collect(),
117            credit_exclusions: held
118                .credit_exclusions
119                .iter()
120                .filter(|decision| decision.relation.provenance == *name)
121                .cloned()
122                .collect(),
123        };
124        let reach = sources::attachment(&mine, catalog);
125        let last = mine.records.iter().map(|r| r.fetched_at).max().unwrap_or(0);
126        table.push(vec![
127            (*name).to_string(),
128            mine.records.len().to_string(),
129            reach.attached.to_string(),
130            reach.waiting.to_string(),
131            ui::since(last),
132        ]);
133    }
134    println!("{}", table.render());
135
136    // A count of what is waiting means nothing without the reason, and the
137    // reason is always the same one: the catalog does not hold that entity
138    // yet. Saying so here is what stops it reading as a failure.
139    let all = sources::attachment(held, catalog);
140    if all.waiting > 0 {
141        // Both verbs agree with the count, because `ui::plural` puts the count
142        // in the sentence and the rest of the sentence has to follow it.
143        let (name, kept) = match all.waiting {
144            1 => ("names", "is"),
145            _ => ("name", "are"),
146        };
147        println!(
148            "  {}",
149            ui::dim(&format!(
150                "{} {name} nothing this catalog holds yet, and {kept} kept until it does",
151                ui::plural(all.waiting, "record")
152            ))
153        );
154    }
155    Ok(())
156}

158/// Every record, with what it says and whether the catalog can place it.
159fn list(held: &Sources, catalog: &Catalog, args: &Args) -> Res {
160    let only = args.value("source");
161    let rows: Vec<&SourceRecord> = held
162        .records
163        .iter()
164        .filter(|r| only.is_none_or(|s| r.source == s))
165        .collect();
166
167    if rows.is_empty() {
168        println!("{}", ui::section("Sources"));
169        println!(
170            "  {}",
171            ui::dim(&ui::literal(&match only {
172                Some(name) => format!("nothing was fetched from \"{name}\""),
173                None => "nothing has been fetched yet".to_string(),
174            }))
175        );
176        return Ok(());
177    }
178
179    println!("{}", ui::section("What sources say"));
180    let mut table = Table::new(&["Entity", "Source", "Confidence", "Says", "Where"])
181        .limit(0, 42)
182        .limit(3, 46);
183    for record in &rows {
184        let entity = record.entity();
185        table.push(vec![
186            format!("{} {}", entity.kind.as_str(), entity.key),
187            record.source.clone(),
188            super::source_status(
189                record.confidence,
190                held.review_for(record).map(|review| review.decision),
191                held.is_trusted(catalog, record),
192            ),
193            says(&record.facts),
194            match entity.resolve(catalog).is_some() {
195                true => "in the catalog".to_string(),
196                false => "waiting".to_string(),
197            },
198        ]);
199    }
200    println!("{}", table.render());
201    println!("  {}", ui::dim(&ui::plural(rows.len(), "record")));
202    Ok(())
203}

205/// What a record says, short enough for a column.
206fn says(facts: &Facts) -> String {
207    let mut parts: Vec<String> = Vec::new();
208    match facts {
209        Facts::Track(t) => {
210            // What it heard, in the order somebody reads it: who, then what.
211            // The score leads nothing — it qualifies the rest, and a line
212            // opening with a number reads as a measurement of the file.
213            if !t.artists.is_empty() {
214                parts.push(t.artists.join(", "));
215            }
216            if let Some(title) = &t.title {
217                parts.push(title.clone());
218            }
219            if let Some(album) = &t.album {
220                parts.push(album.clone());
221            }
222            if !t.works.is_empty() {
223                parts.push(format!("{} work link(s)", t.works.len()));
224            }
225            if !t.credits.is_empty() {
226                parts.push(ui::plural(t.credits.len(), "recording credit"));
227            }
228            if let Some(recording) = &t.recording {
229                parts.push(format!("recording: {recording}"));
230            }
231            if let Some(score) = t.score {
232                parts.push(format!("{score}% sure"));
233            }
234        }
235        Facts::Artist(a) => {
236            if let Some(kind) = &a.kind {
237                parts.push(kind.clone());
238            }
239            if let Some(known) = &a.disambiguation {
240                parts.push(known.clone());
241            }
242            if let Some(area) = a.area.as_ref().or(a.country_code.as_ref()) {
243                parts.push(area.clone());
244            }
245            match (&a.began, &a.ended) {
246                (Some(from), Some(to)) => parts.push(format!("{from}–{to}")),
247                (Some(from), None) => parts.push(format!("{from}–")),
248                (None, Some(to)) => parts.push(format!("ended {to}")),
249                _ => {}
250            }
251            if let Some(active) = a.active {
252                parts.push(match active {
253                    true => "still active".to_string(),
254                    false => "ended".to_string(),
255                });
256            }
257            if a.portrait.is_some() {
258                parts.push("has a portrait".to_string());
259            }
260            if a.logo.is_some() {
261                parts.push("has a logo".to_string());
262            }
263            if !a.genres.is_empty() {
264                parts.push(a.genres.join(", "));
265            }
266            if !a.aliases.is_empty() {
267                parts.push(format!("also called {}", a.aliases.join(", ")));
268            }
269            if !a.members.is_empty() {
270                parts.push(ui::plural(a.members.len(), "membership"));
271            }
272            if !a.discography.is_empty() {
273                parts.push(ui::plural(a.discography.len(), "release"));
274            }
275            for (present, description) in [
276                (a.wikidata.is_some(), "has a Wikidata link"),
277                (a.discogs.is_some(), "has a Discogs link"),
278                (a.homepage.is_some(), "has a homepage"),
279            ] {
280                if present {
281                    parts.push(description.to_string());
282                }
283            }
284            // The detail card prints the prose with its complete credit. This
285            // short table cannot return an extract without that attribution.
286            if a.summary.is_some() {
287                parts.push("has a biography".to_string());
288            }
289        }
290        Facts::Release(r) => {
291            if let Some(primary) = &r.primary_type {
292                parts.push(primary.clone());
293            }
294            parts.extend(r.secondary_types.iter().cloned());
295            if let Some(date) = &r.first_released {
296                parts.push(date.clone());
297            }
298            if let Some(label) = &r.label {
299                parts.push(label.clone());
300            }
301            if r.label.is_none()
302                && let Some(label) = &r.label_mbid
303            {
304                parts.push(format!("label: {label}"));
305            }
306            if r.cover_art.is_some() {
307                parts.push("has a cover".to_string());
308            }
309            if !r.credits.is_empty() {
310                parts.push(ui::plural(r.credits.len(), "edition credit"));
311            }
312            if let Some(edition) = &r.edition_mbid {
313                parts.push(format!("edition: {edition}"));
314            }
315        }
316        Facts::Label(LabelFacts {
317            logo,
318            summary,
319            wikidata,
320            discogs,
321            ..
322        }) => {
323            if let Some(url) = wikidata {
324                parts.push(url.clone());
325            }
326            if let Some(url) = discogs {
327                parts.push(url.clone());
328            }
329            if summary.is_some() {
330                parts.push("has a profile".to_string());
331            }
332            if let Some(logo) = logo {
333                parts.push(format!("logo: {}", logo.url));
334            }
335        }
336    }
337    match parts.is_empty() {
338        // An answer holding nothing is a real answer, and printing an empty
339        // cell would make it look like a missing row instead.
340        true => ui::dim("nothing"),
341        false => parts.join(" · "),
342    }
343}

345/// Takes in a document written in the shape of `sources.json`.
346fn import(args: &Args, path: &std::path::Path) -> Res {
347    let Some(file) = args.value("import") else {
348        return Err("--import expects a file: --import=<file.json>".into());
349    };
350    let text = std::fs::read_to_string(file).map_err(|e| format!("cannot read \"{file}\": {e}"))?;
351    let value =
352        aede_core::json::parse(&text).map_err(|e| format!("\"{file}\" is not JSON: {e}"))?;
353    let incoming = sources::from_json(&value)?;
354
355    let mut held = sources::load_all(path)?.unwrap_or_default();
356    let (mut updated, mut added) = (0usize, 0usize);
357    for record in incoming.records {
358        match held.set(record) {
359            true => updated += 1,
360            false => added += 1,
361        }
362    }
363    for review in incoming.reviews {
364        held.set_review(review);
365    }
366    for decision in incoming.credit_exclusions {
367        held.exclude_credit(decision.relation, decision.excluded_at);
368    }
369    sources::save(&held, path)?;
370
371    println!("{}", ui::section("Imported"));
372    println!(
373        "{} {} added, {} updated, {} held",
374        ui::green("→"),
375        added,
376        updated,
377        held.records.len()
378    );
379    // Naming the file it wrote, for the same reason `import --forget` does: a
380    // command that changed a store the user cannot see should say which one.
381    println!("  {}", ui::dim(&ui::literal(&path.display().to_string())));
382    Ok(())
383}

385/// Writes the layer out, in the shape `--import` reads.
386///
387/// The symmetry matters more than the feature: a store that can be filled and
388/// not emptied out again is a store whose contents you have to trust somebody
389/// else about. `aede notes --export` already draws the same round trip for
390/// what the user writes.
391fn export(args: &Args, path: &std::path::Path) -> Res {
392    let held = sources::load(path)?.unwrap_or_default();
393    if held.records.is_empty() && held.reviews.is_empty() && held.credit_exclusions.is_empty() {
394        // Writing an empty document and reporting success would look like the
395        // export worked and the layer was empty — two different things, and
396        // the second is not what has just been established.
397        return Err(
398            "nothing has been fetched yet, so there is nothing to export.\n\
399             aede sources --template=<file> writes a document you can fill in"
400                .into(),
401        );
402    }
403    super::export::emit(args, &sources::to_json(&held).to_string_pretty())
404}

406/// Writes a document with the right keys and nothing filled in.
407///
408/// The keys are the one thing a person cannot guess: they are how an entity
409/// names itself — a normalised artist name, or an album's artist, title and
410/// folder joined — and until something fetches, there is no way to see one.
411/// Without this, the layer could be read and emptied but never filled by hand,
412/// which makes it untestable by the person who most needs to test it.
413///
414/// Naming entities narrows it:
415/// `aede sources --template --output=x.json "Kind of Blue"` writes that album
416/// alone rather than a skeleton of the whole library.
417fn template(args: &Args) -> Res {
418    let catalog = load(args)?;
419    let source = args.value("source").unwrap_or(sources::MUSICBRAINZ);
420    let now = aede_core::clock::now_seconds();
421
422    let wanted: Vec<String> = args
423        .positionals
424        .iter()
425        .map(|name| aede_core::text::normalize(name))
426        .filter(|name| !name.is_empty())
427        .collect();
428    let matches = |name: &str| -> bool {
429        let key = aede_core::text::normalize(name);
430        wanted.is_empty() || wanted.iter().any(|w| key.contains(w.as_str()))
431    };
432
433    let mut out = Sources::default();
434    let blank = |key: String, facts: Facts| SourceRecord {
435        key,
436        source: source.to_string(),
437        source_id: None,
438        fetched_at: now,
439        confidence: Confidence::Identified,
440        facts,
441    };
442    for artist in &catalog.artists {
443        if !matches(&artist.name) {
444            continue;
445        }
446        if let Some(entity) = EntityRef::of(&catalog, EntityKind::Artist, artist.id) {
447            out.set(blank(entity.key, Facts::Artist(Default::default())));
448        }
449    }
450    for release in &catalog.releases {
451        if !matches(&release.title) {
452            continue;
453        }
454        if let Some(entity) = EntityRef::of(&catalog, EntityKind::Release, release.id) {
455            out.set(blank(entity.key, Facts::Release(Default::default())));
456        }
457    }
458
459    if out.records.is_empty() {
460        return Err(match wanted.is_empty() {
461            true => "this catalog holds nothing to write a template for".to_string(),
462            false => format!(
463                "nothing in this catalog matches \"{}\"",
464                args.positionals.join(" ")
465            ),
466        }
467        .into());
468    }
469
470    super::export::emit(args, &sources::to_json(&out).to_string_pretty())?;
471    // Only when it went to a file: a template printed to the terminal is on
472    // screen, and telling somebody to import what they can see is noise.
473    if args.value("output").is_some() {
474        println!(
475            "  {} {}",
476            ui::plural(out.records.len(), "empty record"),
477            ui::dim("— fill in what you want, then: aede sources --import=<file>")
478        );
479    }
480    Ok(())
481}

483/// Drops what one source said, or everything.
484fn forget(args: &Args, path: &std::path::Path) -> Res {
485    let mut held = sources::load_all(path)?.unwrap_or_default();
486    let before = held.records.len();
487    let decisions_before = held.reviews.len() + held.credit_exclusions.len();
488    if before == 0 && decisions_before == 0 {
489        println!("{}", ui::section("Sources"));
490        println!("  {}", ui::dim("there was nothing to forget"));
491        return Ok(());
492    }
493
494    let removed = match args.value("source") {
495        Some(name) => {
496            let gone = held.forget(name);
497            if gone == 0 && decisions_before == held.reviews.len() + held.credit_exclusions.len() {
498                return Err(format!(
499                    "no source named \"{name}\" — aede sources says which there are"
500                )
501                .into());
502            }
503            gone
504        }
505        None => {
506            let removed = std::mem::take(&mut held.records).len();
507            held.reviews.clear();
508            held.credit_exclusions.clear();
509            removed
510        }
511    };
512
513    sources::save(&held, path)?;
514    println!(
515        "{} {} removed, {} left",
516        ui::green("→"),
517        ui::plural(removed, "record"),
518        held.records.len()
519    );
520    let decisions_removed = decisions_before - held.reviews.len() - held.credit_exclusions.len();
521    if decisions_removed > 0 {
522        println!("  {} removed", ui::plural(decisions_removed, "decision"));
523    }
524    println!("  {}", ui::dim(&ui::literal(&path.display().to_string())));
525    Ok(())
526}

532/// The block an entity page prints when a source has said something about it.
533///
534/// One function called from every page rather than one block written per page:
535/// what a source says has to read the same way on an album and on an artist,
536/// and a rule copied into four pages is a rule that will be right in three of
537/// them.
538///
539/// Silent when nothing was fetched. A page that announced an empty section for
540/// a feature nobody has used yet would be noise on every album in the library.
541pub fn panel_for(args: &Args, catalog: &Catalog, kind: EntityKind, id: Id) {
542    let Ok(held) = super::sources_held(args) else {
543        return;
544    };
545    panel_for_held(catalog, kind, id, &held, false);
546}

548/// The label page may deliberately show its retained Discogs profile when
549/// offline or when a refresh failed. Other pages use the filtered loader.
550pub fn panel_for_held(
551    catalog: &Catalog,
552    kind: EntityKind,
553    id: Id,
554    held: &Sources,
555    refresh_failed: bool,
556) {
557    let Some(entity) = EntityRef::of(catalog, kind, id) else {
558        return;
559    };
560    let records: Vec<&SourceRecord> = held.about(&entity).collect();
561    if records.is_empty() {
562        return;
563    }
564
565    println!("{}", ui::section("What sources say"));
566
567    // Prose first, and outside the table. A paragraph does not fit a column,
568    // and it is the one thing here meant to be read rather than compared with
569    // a tag — there is nothing in a file to compare it against.
570    for record in &records {
571        let prose = match &record.facts {
572            Facts::Artist(artist) => artist.summary.as_ref(),
573            Facts::Label(label) => label.summary.as_ref(),
574            _ => None,
575        };
576        if let Some(prose) = prose {
577            if record.source == aede_core::discogs::SOURCE
578                && (refresh_failed
579                    || !aede_core::discogs::fresh(
580                        record.fetched_at,
581                        aede_core::clock::now_seconds(),
582                    ))
583            {
584                println!(
585                    "  {}",
586                    ui::yellow(
587                        "Stored Discogs profile: update needed; the latest version could not be confirmed."
588                    )
589                );
590            }
591            let display = if record.source == aede_core::discogs::SOURCE {
592                aede_core::discogs::render_profile(
593                    &prose.text,
594                    &Default::default(),
595                    &Default::default(),
596                )
597            } else {
598                prose.text.clone()
599            };
600            let lines = if record.source == aede_core::discogs::SOURCE {
601                discogs_profile_lines(&display, 72)
602            } else {
603                ui::wrap(&ui::literal(&display), 72)
604            };
605            for line in lines {
606                if line.is_empty() {
607                    println!();
608                } else {
609                    println!("  {line}");
610                }
611            }
612            if record.source == aede_core::discogs::SOURCE {
613                println!(
614                    "  {}",
615                    ui::dim(
616                        "Discogs profile: original text (usually English; no French translation supplied)"
617                    )
618                );
619                println!(
620                    "  {}",
621                    ui::dim(&ui::literal(&format!(
622                        "Data provided by Discogs. {} — CC0",
623                        prose.url
624                    )))
625                );
626                println!(
627                    "  {}",
628                    ui::dim(
629                        "This application uses Discogs’ API but is not affiliated with, sponsored or endorsed by Discogs. ‘Discogs’ is a trademark of Zink Media, LLC."
630                    )
631                );
632                println!(
633                    "  {}",
634                    ui::dim(&ui::literal(&format!(
635                        "aede label \"{}\" checks this profile whenever the page opens",
636                        record.key
637                    )))
638                );
639            } else {
640                println!("  {}", ui::dim(&ui::literal(&prose.credit())));
641                println!(
642                    "  {}",
643                    ui::dim(&ui::literal(&format!(
644                        "aede fetch --summaries --full --lang=<code> \"{}\" asks for another",
645                        record.key
646                    )))
647                );
648            }
649            println!();
650        }
651    }
652
653    let mut table = Table::new(&["Source", "Field", "Says", "Your tags"]).limit(2, 34);
654
655    for record in &records {
656        let age = ui::since(record.fetched_at);
657        let attribution = format!(
658            "{} · {} · {age}",
659            record.source,
660            super::source_status(
661                record.confidence,
662                held.review_for(record).map(|review| review.decision),
663                held.is_trusted(catalog, record),
664            )
665        );
666        for (field, theirs, verdict) in compared(catalog, &entity, &record.facts) {
667            table.push(vec![
668                attribution.clone(),
669                field.to_string(),
670                theirs,
671                // The verdict is the point of the column: a value that matches
672                // the tags is not the same news as one that contradicts them,
673                // and neither is one the tags say nothing about.
674                match verdict {
675                    Some(Verdict::Agrees) => ui::green("matches your tags").to_string(),
676                    Some(Verdict::Differs { yours, .. }) => ui::yellow(&yours).to_string(),
677                    // The tags could have answered and did not.
678                    Some(Verdict::NothingToCompare) => ui::dim("nothing in your tags").to_string(),
679                    // There is no tag for this at all — an artist's country is
680                    // not something the files have an opinion about. Saying
681                    // "nothing in your tags" would suggest they were expected
682                    // to, and send the reader looking for a field to fill.
683                    None => String::new(),
684                },
685            ]);
686        }
687    }
688    // A source that was asked and holds nothing is not an empty table. The
689    // distinction is the one this whole layer exists to keep — "checked, and
690    // it knows nothing" against "never checked" — and a table rendering
691    // "(no results)" says neither of them.
692    if table.is_empty() {
693        for record in &records {
694            // A record whose only content is the paragraph just printed has
695            // no row here, and saying it "holds nothing" directly under its
696            // own text would be plainly false.
697            if matches!(&record.facts, Facts::Artist(a) if a.summary.is_some())
698                || matches!(&record.facts, Facts::Label(l) if l.summary.is_some())
699            {
700                continue;
701            }
702            println!(
703                "  {}",
704                ui::dim(&ui::literal(&format!(
705                    "{} was asked and holds nothing about this ({})",
706                    record.source,
707                    ui::since(record.fetched_at)
708                )))
709            );
710        }
711        whence(&records);
712        return;
713    }
714
715    print!("{}", table.render());
716    whence(&records);
717    println!(
718        "{}",
719        ui::dim("  beside your tags, never on top of them — aede sources --forget removes them")
720    );
721}

723/// Discogs separates identity notes from the biography with hard line breaks.
724/// Wrap each source line independently and apply emphasis after measuring it,
725/// so terminal escape sequences do not count toward the available width.
726fn discogs_profile_lines(text: &str, width: usize) -> Vec<String> {
727    text.lines()
728        .flat_map(|line| {
729            if line.trim().is_empty() {
730                return vec![String::new()];
731            }
732            let label_code = line
733                .trim_start()
734                .to_ascii_lowercase()
735                .starts_with("label code:");
736            ui::wrap(&ui::literal(line), width)
737                .into_iter()
738                .map(|line| if label_code { ui::bold(&line) } else { line })
739                .collect()
740        })
741        .collect()
742}

744/// Where each of these answers came from, as an address a reader can follow.
745///
746/// **The identifier was stored and never shown**, which is the one thing a
747/// reader needs to check anything by hand: to open the page, to ask the service
748/// the same question, or to correct the data at its source. Asked for the
749/// MusicBrainz identifier of an artist, there was nowhere in this program to
750/// find it — and the nearest thing on screen was the Wikidata link, which is a
751/// different identifier that looks enough like an answer to waste an afternoon.
752///
753/// Printed as the URL rather than the bare identifier: the identifier is inside
754/// it, so it can still be copied for a query, and the page it opens is where a
755/// wrong type or a missing date is actually fixed — for everybody, not just
756/// here.
757fn whence(records: &[&SourceRecord]) {
758    for record in records {
759        if let Some(address) = address_of(record) {
760            println!(
761                "  {}",
762                ui::dim(&ui::literal(&format!("{}: {address}", record.source)))
763            );
764        }
765    }
766}

768/// Where one record's answer can be looked at, or its bare identifier.
769///
770/// Split from the printing so the decision is testable: which path a
771/// MusicBrainz album takes is the sort of thing that is wrong once and then
772/// wrong for a year, because a link that 404s looks like the service's fault.
773pub(super) fn address_of(record: &SourceRecord) -> Option<String> {
774    let id = record.source_id.as_deref()?;
775    let kind = match record.facts.kind() {
776        EntityKind::Artist => "artist",
777        // Album snapshots normally name the group. A lookup without a group
778        // instead retains the edition ID in both fields — see
779        // `musicbrainz::release`. Do not send that fallback to the group route.
780        EntityKind::Release
781            if matches!(
782                record.musicbrainz_release_identity(),
783                Some(aede_core::sources::MusicBrainzReleaseIdentity::Edition(_))
784            ) =>
785        {
786            "release"
787        }
788        EntityKind::Release => "release-group",
789        EntityKind::Label => "label",
790        // Other entity kinds have no known MusicBrainz URL here.
791        _ => return Some(id.to_string()),
792    };
793    Some(match record.source.as_str() {
794        sources::MUSICBRAINZ => format!("https://musicbrainz.org/{kind}/{id}"),
795        aede_core::discogs::SOURCE if kind == "label" => {
796            format!("https://www.discogs.com/label/{id}")
797        }
798        "wikipedia" => format!("https://www.wikidata.org/wiki/{id}"),
799        // A source this build knows nothing about still has an identifier, and
800        // the identifier is the useful half. Inventing an address for it would
801        // be worse than showing none.
802        _ => id.to_string(),
803    })
804}

806/// The first value of a tag on any file credited to this artist.
807///
808/// A genre lives on the tracks, not on the artist, so "what do my files call
809/// this artist" is answered by the files they appear on.
810/// What one track's own file says under a tag.
811///
812/// Its own file, not its release's: a fingerprint answers for one recording,
813/// and comparing it with an album-level tag would put the wrong two things
814/// side by side.
815fn tags_of_track(catalog: &Catalog, entity: &EntityRef, tag: &str) -> Vec<String> {
816    let Some(id) = entity.resolve(catalog) else {
817        return Vec::new();
818    };
819    catalog
820        .track(id)
821        .and_then(|track| catalog.file(track.file_id))
822        .and_then(|file| file.tags.get(tag))
823        .cloned()
824        .unwrap_or_default()
825}
826
827fn tags_of_artist(catalog: &Catalog, entity: &EntityRef, tag: &str) -> Vec<String> {
828    let Some(id) = entity.resolve(catalog) else {
829        return Vec::new();
830    };
831    // **Every** value, not the first: a genre tag holds a list, whether the
832    // file writes it as several values or as one string with commas in it.
833    // Reading only the first turned "Rock, Pop" into "Rock" on some files and
834    // reported a disagreement with a genre the tags do carry.
835    catalog
836        .tracks_of_artist(id)
837        .into_iter()
838        .filter_map(|t| catalog.track(t))
839        .find_map(|t| {
840            catalog
841                .file(t.file_id)
842                .map(|f| f.tags.get(tag).cloned().unwrap_or_default())
843                .filter(|values: &Vec<String>| !values.is_empty())
844        })
845        .unwrap_or_default()
846}

848/// Each field a record carries, with the tag it can be judged against.
849///
850/// `None` where the catalog holds no counterpart at all: an artist's country
851/// is not something the tags have an opinion about, and printing "nothing in
852/// your tags" against it would suggest the tags were expected to.
853fn compared(
854    catalog: &Catalog,
855    entity: &EntityRef,
856    facts: &Facts,
857) -> Vec<(&'static str, String, Option<Verdict>)> {
858    let mut rows: Vec<(&'static str, String, Option<Verdict>)> = Vec::new();
859    match facts {
860        Facts::Track(t) => {
861            // Compared with the tags, which is the whole use of this: a file
862            // whose audio and whose tags disagree is what fingerprinting is
863            // for, and the panel is where the two sit side by side.
864            if !t.artists.is_empty() {
865                rows.push((
866                    "heard artist",
867                    t.artists.join(", "),
868                    Some(sources::verdict_set(
869                        &t.artists,
870                        &tags_of_track(catalog, entity, "artist"),
871                    )),
872                ));
873            }
874            if let Some(title) = &t.title {
875                rows.push((
876                    "heard title",
877                    title.clone(),
878                    Some(sources::verdict(
879                        title,
880                        tags_of_track(catalog, entity, "title")
881                            .first()
882                            .map(String::as_str),
883                    )),
884                ));
885            }
886            for (field, value) in [("heard on", &t.album), ("recording", &t.recording)] {
887                if let Some(value) = value {
888                    rows.push((field, value.clone(), None));
889                }
890            }
891            for work in &t.works {
892                rows.push(("work", format!("{} ({})", work.title, work.mbid), None));
893                for credit in &work.credits {
894                    rows.push(("work credit", credit_summary(credit), None));
895                }
896            }
897            for credit in &t.credits {
898                rows.push(("recording credit", credit_summary(credit), None));
899            }
900            if let Some(score) = t.score {
901                // Shown, never hidden behind a threshold: a reader deciding
902                // whether to believe this needs the number the service gave,
903                // and a program that quietly dropped the weak matches would
904                // be making that decision for them.
905                rows.push((("how sure"), format!("{score}%"), None));
906            }
907        }
908        Facts::Artist(a) => {
909            // "country" was wrong: MusicBrainz calls this an *area*, and an
910            // area is a country, a city or a region — "Seattle" is a valid
911            // answer. "from" is true of all three.
912            //
913            // "known as" was wrong too, and worse: `disambiguation` is written
914            // to separate two artists who share a name, not to describe one,
915            // which is how a perfectly correct answer came out as
916            // "known as: the band".
917            for (field, value) in [
918                ("type", &a.kind),
919                ("from", &a.area),
920                ("country code", &a.country_code),
921                ("formed", &a.began),
922                ("ended", &a.ended),
923                ("note", &a.disambiguation),
924                ("wikidata", &a.wikidata),
925                ("discogs", &a.discogs),
926                ("homepage", &a.homepage),
927            ] {
928                if let Some(value) = value {
929                    rows.push((field, value.clone(), None));
930                }
931            }
932            if !a.genres.is_empty() {
933                // Beside the genre tag, never over it: the last column is what
934                // your files say, and these two answers are both allowed to
935                // exist.
936                //
937                // Compared as **sets**, which the first version did not do: it
938                // took MusicBrainz's top genre and compared it, as a string, to
939                // the whole tag. `pop` against `Rock, Pop` came out as a
940                // disagreement, when the tag plainly says pop. Genres are not
941                // exclusive and the two sides are not answering at the same
942                // granularity — one is what a crowd voted, the other is what
943                // one person typed — so a shared name is agreement.
944                let theirs = a.genres.join(", ");
945                let yours = tags_of_artist(catalog, entity, "genre");
946                let verdict = aede_core::sources::verdict_set(&a.genres, &yours);
947                rows.push(("genres", theirs, Some(verdict)));
948            }
949            if !a.aliases.is_empty() {
950                rows.push(("also called", a.aliases.join(", "), None));
951            }
952            if !a.members.is_empty() {
953                rows.push((
954                    "memberships",
955                    ui::plural(a.members.len(), "membership"),
956                    None,
957                ));
958            }
959            if !a.discography.is_empty() {
960                rows.push((
961                    "discography",
962                    ui::plural(a.discography.len(), "release"),
963                    None,
964                ));
965            }
966            // Shown even when there is no end date, because that is the case
967            // it answers: a band with no end may have never stopped, or may
968            // be one nobody filled in, and the reader cannot tell otherwise.
969            if let Some(active) = a.active {
970                rows.push((
971                    "status",
972                    match active {
973                        true => "still active".to_string(),
974                        false => "ended".to_string(),
975                    },
976                    None,
977                ));
978            }
979            // No verdict to compare it with — a tag has no field for a
980            // picture — so this is shown, not judged, the same as the
981            // discography and the members above it.
982            if let Some(picture) = &a.portrait {
983                rows.push(("portrait", picture.url.clone(), None));
984            }
985            if let Some(picture) = &a.logo {
986                rows.push(("logo", picture.url.clone(), None));
987            }
988        }
989        Facts::Release(r) => {
990            let release = entity
991                .resolve(catalog)
992                .and_then(|id| catalog.release(id))
993                .cloned();
994            let tag = |name: &str| -> Option<String> {
995                release.as_ref().and_then(|rel| {
996                    rel.track_ids
997                        .first()
998                        .and_then(|&t| catalog.track(t))
999                        .and_then(|t| catalog.file(t.file_id))
1000                        .and_then(|f| f.first_tag(name).map(str::to_string))
1001                })
1002            };
1003
1004            if let Some(primary) = &r.primary_type {
1005                let mut shown = primary.clone();
1006                if !r.secondary_types.is_empty() {
1007                    shown = format!("{shown} · {}", r.secondary_types.join(" · "));
1008                }
1009                rows.push((
1010                    "release type",
1011                    shown,
1012                    Some(aede_core::sources::verdict(
1013                        primary,
1014                        tag("releasetype").as_deref(),
1015                    )),
1016                ));
1017            } else if !r.secondary_types.is_empty() {
1018                rows.push(("release type", r.secondary_types.join(" · "), None));
1019            }
1020            if let Some(date) = &r.first_released {
1021                let yours = release.as_ref().and_then(|rel| rel.date.clone());
1022                rows.push((
1023                    "first released",
1024                    date.clone(),
1025                    Some(aede_core::sources::verdict_date(date, yours.as_deref())),
1026                ));
1027            }
1028            if let Some(label) = &r.label {
1029                let yours = release.as_ref().and_then(|rel| {
1030                    rel.label_ids
1031                        .first()
1032                        .and_then(|&id| catalog.label(id))
1033                        .map(|l| l.name.clone())
1034                });
1035                rows.push((
1036                    "label",
1037                    label.clone(),
1038                    Some(aede_core::sources::verdict(label, yours.as_deref())),
1039                ));
1040            }
1041            for (field, value) in [
1042                ("label ID", &r.label_mbid),
1043                ("edition", &r.edition_mbid),
1044                ("cover", &r.cover_art),
1045            ] {
1046                if let Some(value) = value {
1047                    rows.push((field, value.clone(), None));
1048                }
1049            }
1050            for credit in &r.credits {
1051                rows.push(("edition credit", credit_summary(credit), None));
1052            }
1053        }
1054        // Label links and artwork have no audio-tag counterpart to judge.
1055        Facts::Label(label) => {
1056            if let Some(url) = &label.wikidata {
1057                rows.push(("wikidata", url.clone(), None));
1058            }
1059            if let Some(url) = &label.discogs {
1060                rows.push(("discogs", url.clone(), None));
1061            }
1062            if let Some(picture) = &label.logo {
1063                rows.push(("logo", picture.url.clone(), None));
1064            }
1065        }
1066    }
1067    rows
1068}

1070fn credit_summary(credit: &sources::CreditLink) -> String {
1071    let credited = credit
1072        .credited_as
1073        .as_deref()
1074        .filter(|name| *name != credit.artist_name)
1075        .map(|name| format!(" as {name}"))
1076        .unwrap_or_default();
1077    let attributes = credit
1078        .attributes
1079        .iter()
1080        .map(|attribute| {
1081            attribute
1082                .credited_as
1083                .as_deref()
1084                .or(attribute.value.as_deref())
1085                .unwrap_or(&attribute.name)
1086                .to_string()
1087        })
1088        .collect::<Vec<_>>();
1089    let details = match attributes.is_empty() {
1090        true => String::new(),
1091        false => format!(" ({})", attributes.join(", ")),
1092    };
1093    format!(
1094        "{} — {}{credited}{details}",
1095        credit.role, credit.artist_name
1096    )
1097}
