1//! The `sources` command: what somebody else said about this library.
2//!
3//! The catalog holds what the files say, the annotation file holds what you
4//! say, and `sources.json` holds what a source says — MusicBrainz first, a
5//! plugin later. This is the command that shows it, takes it in, and drops it.
6//!
7//! It exists before anything fetches, on purpose. A store that can only be
8//! filled over the network can only be tested over the network, and the layer
9//! is worth more than that: `--import` reads a document in exactly the shape
10//! the file has, so the whole of it — attachment, agreement, removal — can be
11//! exercised offline, and so can a bug report.
12//!
13//! The shape follows `aede import`, which does the same job for another tool's
14//! analyses: `--list` says what is held, `--forget` removes it, `--source`
15//! narrows to one. Two stores of what somebody else said should not be learned
16//! twice.
17

26pub fn sources(args: &Args) -> Res {
27    let dir = data_dir(args);
28    let path = sources::sources_path(&dir);
29
30    if args.has("import") {
31        return import(args, &path);
32    }
33    if args.has("template") {
34        return template(args);
35    }
36    if args.has("export") {
37        return export(args, &path);
38    }
39    if args.has("forget") {
40        return forget(args, &path);
41    }
42
43    let held = sources::load(&path)?.unwrap_or_default();
44    if held.records.is_empty() {
45        println!("{}", ui::section("Sources"));
46        println!("  {}", ui::dim("nothing has been fetched yet"));
47        // The store being empty is not the same as the feature being absent,
48        // and a reader who cannot tell which will conclude the wrong one.
49        println!(
50            "  {}",
51            ui::dim("aede sources --import <file> takes in a document by hand")
52        );
53        return Ok(());
54    }
55
56    let catalog = load(args)?;
57    match args.has("list") {
58        true => list(&held, &catalog, args),
59        false => summary(&held, &catalog),
60    }
61}

63/// One line per source: how much it said, and how much of it lands.
64fn summary(held: &Sources, catalog: &Catalog) -> Res {
65    println!("{}", ui::section("Sources"));
66
67    let mut names: Vec<&str> = held.records.iter().map(|r| r.source.as_str()).collect();
68    names.sort_unstable();
69    names.dedup();
70
71    let mut table = Table::new(&["Source", "Records", "Attached", "Waiting", "Last fetch"])
72        .align(1, Align::Right)
73        .align(2, Align::Right)
74        .align(3, Align::Right);
75
76    for name in &names {
77        let mine = Sources {
78            records: held
79                .records
80                .iter()
81                .filter(|r| r.source == *name)
82                .cloned()
83                .collect(),
84            reviews: held
85                .reviews
86                .iter()
87                .filter(|review| review.source == *name)
88                .cloned()
89                .collect(),
90            credit_exclusions: held
91                .credit_exclusions
92                .iter()
93                .filter(|decision| decision.relation.provenance == *name)
94                .cloned()
95                .collect(),
96        };
97        let reach = sources::attachment(&mine, catalog);
98        let last = mine.records.iter().map(|r| r.fetched_at).max().unwrap_or(0);
99        table.push(vec![
100            (*name).to_string(),
101            mine.records.len().to_string(),
102            reach.attached.to_string(),
103            reach.waiting.to_string(),
104            ui::since(last),
105        ]);
106    }
107    println!("{}", table.render());
108
109    // A count of what is waiting means nothing without the reason, and the
110    // reason is always the same one: the catalog does not hold that entity
111    // yet. Saying so here is what stops it reading as a failure.
112    let all = sources::attachment(held, catalog);
113    if all.waiting > 0 {
114        // Both verbs agree with the count, because `ui::plural` puts the count
115        // in the sentence and the rest of the sentence has to follow it.
116        let (name, kept) = match all.waiting {
117            1 => ("names", "is"),
118            _ => ("name", "are"),
119        };
120        println!(
121            "  {}",
122            ui::dim(&format!(
123                "{} {name} nothing this catalog holds yet, and {kept} kept until it does",
124                ui::plural(all.waiting, "record")
125            ))
126        );
127    }
128    Ok(())
129}

131/// Every record, with what it says and whether the catalog can place it.
132fn list(held: &Sources, catalog: &Catalog, args: &Args) -> Res {
133    let only = args.value("source");
134    let rows: Vec<&SourceRecord> = held
135        .records
136        .iter()
137        .filter(|r| only.is_none_or(|s| r.source == s))
138        .collect();
139
140    if rows.is_empty() {
141        println!("{}", ui::section("Sources"));
142        println!(
143            "  {}",
144            ui::dim(&match only {
145                Some(name) => format!("nothing was fetched from \"{name}\""),
146                None => "nothing has been fetched yet".to_string(),
147            })
148        );
149        return Ok(());
150    }
151
152    println!("{}", ui::section("What sources say"));
153    let mut table = Table::new(&["Entity", "Source", "Confidence", "Says", "Where"])
154        .limit(0, 42)
155        .limit(3, 46);
156    for record in &rows {
157        let entity = record.entity();
158        table.push(vec![
159            format!("{} {}", entity.kind.as_str(), entity.key),
160            record.source.clone(),
161            super::source_status(
162                record.confidence,
163                held.review_for(record).map(|review| review.decision),
164                held.is_trusted(catalog, record),
165            ),
166            says(&record.facts),
167            match entity.resolve(catalog).is_some() {
168                true => "in the catalog".to_string(),
169                false => "waiting".to_string(),
170            },
171        ]);
172    }
173    println!("{}", table.render());
174    println!("  {}", ui::dim(&ui::plural(rows.len(), "record")));
175    Ok(())
176}

187/// What a record says, short enough for a column.
188fn says(facts: &Facts) -> String {
189    let mut parts: Vec<String> = Vec::new();
190    match facts {
191        Facts::Track(t) => {
192            // What it heard, in the order somebody reads it: who, then what.
193            // The score leads nothing — it qualifies the rest, and a line
194            // opening with a number reads as a measurement of the file.
195            if !t.artists.is_empty() {
196                parts.push(t.artists.join(", "));
197            }
198            if let Some(title) = &t.title {
199                parts.push(title.clone());
200            }
201            if let Some(album) = &t.album {
202                parts.push(album.clone());
203            }
204            if !t.works.is_empty() {
205                parts.push(format!("{} work link(s)", t.works.len()));
206            }
207            if let Some(score) = t.score {
208                parts.push(format!("{score}% sure"));
209            }
210        }
211        Facts::Artist(a) => {
212            if let Some(kind) = &a.kind {
213                parts.push(kind.clone());
214            }
215            if let Some(known) = &a.disambiguation {
216                parts.push(known.clone());
217            }
218            if let Some(area) = &a.area {
219                parts.push(area.clone());
220            }
221            match (&a.began, &a.ended) {
222                (Some(from), Some(to)) => parts.push(format!("{from}–{to}")),
223                (Some(from), None) => parts.push(format!("{from}–")),
224                _ => {}
225            }
226            if let Some(active) = a.active {
227                parts.push(match active {
228                    true => "still active".to_string(),
229                    false => "ended".to_string(),
230                });
231            }
232            if a.portrait.is_some() {
233                parts.push("has a portrait".to_string());
234            }
235            // Last, and left to the column's own limit. A truncated opening
236            // line says more about what was stored than the word "summary"
237            // would, and the whole of it is on the artist's own card, where it
238            // is wrapped and credited.
239            if let Some(prose) = &a.summary {
240                parts.push(prose.text.clone());
241            }
242        }
243        Facts::Release(r) => {
244            if let Some(primary) = &r.primary_type {
245                parts.push(primary.clone());
246            }
247            parts.extend(r.secondary_types.iter().cloned());
248            if let Some(date) = &r.first_released {
249                parts.push(date.clone());
250            }
251            if let Some(label) = &r.label {
252                parts.push(label.clone());
253            }
254        }
255        Facts::Label(LabelFacts {
256            logo,
257            summary,
258            wikidata,
259            discogs,
260            ..
261        }) => {
262            if let Some(url) = wikidata {
263                parts.push(url.clone());
264            }
265            if let Some(url) = discogs {
266                parts.push(url.clone());
267            }
268            if let Some(prose) = summary {
269                parts.push(prose.text.clone());
270            }
271            if let Some(logo) = logo {
272                parts.push(format!("logo: {}", logo.url));
273            }
274        }
275    }
276    match parts.is_empty() {
277        // An answer holding nothing is a real answer, and printing an empty
278        // cell would make it look like a missing row instead.
279        true => ui::dim("nothing"),
280        false => parts.join(" · "),
281    }
282}

284/// Takes in a document written in the shape of `sources.json`.
285fn import(args: &Args, path: &std::path::Path) -> Res {
286    let Some(file) = args.value("import") else {
287        return Err("--import expects a file: --import=<file.json>".into());
288    };
289    let text = std::fs::read_to_string(file).map_err(|e| format!("cannot read \"{file}\": {e}"))?;
290    let value =
291        aede_core::json::parse(&text).map_err(|e| format!("\"{file}\" is not JSON: {e}"))?;
292    let incoming = sources::from_json(&value)?;
293
294    let mut held = sources::load_all(path)?.unwrap_or_default();
295    let (mut updated, mut added) = (0usize, 0usize);
296    for record in incoming.records {
297        match held.set(record) {
298            true => updated += 1,
299            false => added += 1,
300        }
301    }
302    for review in incoming.reviews {
303        held.set_review(review);
304    }
305    for decision in incoming.credit_exclusions {
306        held.exclude_credit(decision.relation, decision.excluded_at);
307    }
308    sources::save(&held, path)?;
309
310    println!("{}", ui::section("Imported"));
311    println!(
312        "{} {} added, {} updated, {} held",
313        ui::green("→"),
314        added,
315        updated,
316        held.records.len()
317    );
318    // Naming the file it wrote, for the same reason `import --forget` does: a
319    // command that changed a store the user cannot see should say which one.
320    println!("  {}", ui::dim(&path.display().to_string()));
321    Ok(())
322}

324/// Writes the layer out, in the shape `--import` reads.
325///
326/// The symmetry matters more than the feature: a store that can be filled and
327/// not emptied out again is a store whose contents you have to trust somebody
328/// else about. `aede notes --export` already draws the same round trip for
329/// what the user writes.
330fn export(args: &Args, path: &std::path::Path) -> Res {
331    let held = sources::load(path)?.unwrap_or_default();
332    if held.records.is_empty() {
333        // Writing an empty document and reporting success would look like the
334        // export worked and the layer was empty — two different things, and
335        // the second is not what has just been established.
336        return Err(
337            "nothing has been fetched yet, so there is nothing to export.\n\
338             aede sources --template=<file> writes a document you can fill in"
339                .into(),
340        );
341    }
342    super::export::emit(args, &sources::to_json(&held).to_string_pretty())
343}

345/// Writes a document with the right keys and nothing filled in.
346///
347/// The keys are the one thing a person cannot guess: they are how an entity
348/// names itself — a normalised artist name, or an album's artist, title and
349/// folder joined — and until something fetches, there is no way to see one.
350/// Without this, the layer could be read and emptied but never filled by hand,
351/// which makes it untestable by the person who most needs to test it.
352///
353/// Naming entities narrows it:
354/// `aede sources --template --output=x.json "Kind of Blue"` writes that album
355/// alone rather than a skeleton of the whole library.
356fn template(args: &Args) -> Res {
357    let catalog = load(args)?;
358    let source = args.value("source").unwrap_or(sources::MUSICBRAINZ);
359    let now = aede_core::clock::now_seconds();
360
361    let wanted: Vec<String> = args
362        .positionals
363        .iter()
364        .map(|name| aede_core::text::normalize(name))
365        .filter(|name| !name.is_empty())
366        .collect();
367    let matches = |name: &str| -> bool {
368        let key = aede_core::text::normalize(name);
369        wanted.is_empty() || wanted.iter().any(|w| key.contains(w.as_str()))
370    };
371
372    let mut out = Sources::default();
373    let blank = |key: String, facts: Facts| SourceRecord {
374        key,
375        source: source.to_string(),
376        source_id: None,
377        fetched_at: now,
378        confidence: Confidence::Identified,
379        facts,
380    };
381    for artist in &catalog.artists {
382        if !matches(&artist.name) {
383            continue;
384        }
385        if let Some(entity) = EntityRef::of(&catalog, EntityKind::Artist, artist.id) {
386            out.set(blank(entity.key, Facts::Artist(Default::default())));
387        }
388    }
389    for release in &catalog.releases {
390        if !matches(&release.title) {
391            continue;
392        }
393        if let Some(entity) = EntityRef::of(&catalog, EntityKind::Release, release.id) {
394            out.set(blank(entity.key, Facts::Release(Default::default())));
395        }
396    }
397
398    if out.records.is_empty() {
399        return Err(match wanted.is_empty() {
400            true => "this catalog holds nothing to write a template for".to_string(),
401            false => format!(
402                "nothing in this catalog matches \"{}\"",
403                args.positionals.join(" ")
404            ),
405        }
406        .into());
407    }
408
409    super::export::emit(args, &sources::to_json(&out).to_string_pretty())?;
410    // Only when it went to a file: a template printed to the terminal is on
411    // screen, and telling somebody to import what they can see is noise.
412    if args.value("output").is_some() {
413        println!(
414            "  {} {}",
415            ui::plural(out.records.len(), "empty record"),
416            ui::dim("— fill in what you want, then: aede sources --import=<file>")
417        );
418    }
419    Ok(())
420}

422/// Drops what one source said, or everything.
423fn forget(args: &Args, path: &std::path::Path) -> Res {
424    let mut held = sources::load_all(path)?.unwrap_or_default();
425    let before = held.records.len();
426    if before == 0 {
427        println!("{}", ui::section("Sources"));
428        println!("  {}", ui::dim("there was nothing to forget"));
429        return Ok(());
430    }
431
432    let removed = match args.value("source") {
433        Some(name) => {
434            let gone = held.forget(name);
435            if gone == 0 {
436                return Err(format!(
437                    "no source named \"{name}\" — aede sources says which there are"
438                )
439                .into());
440            }
441            gone
442        }
443        None => {
444            let removed = std::mem::take(&mut held.records).len();
445            held.reviews.clear();
446            held.credit_exclusions.clear();
447            removed
448        }
449    };
450
451    sources::save(&held, path)?;
452    println!(
453        "{} {} removed, {} left",
454        ui::green("→"),
455        ui::plural(removed, "record"),
456        held.records.len()
457    );
458    println!("  {}", ui::dim(&path.display().to_string()));
459    Ok(())
460}

466/// The block an entity page prints when a source has said something about it.
467///
468/// One function called from every page rather than one block written per page:
469/// what a source says has to read the same way on an album and on an artist,
470/// and a rule copied into four pages is a rule that will be right in three of
471/// them.
472///
473/// Silent when nothing was fetched. A page that announced an empty section for
474/// a feature nobody has used yet would be noise on every album in the library.
475pub fn panel_for(args: &Args, catalog: &Catalog, kind: EntityKind, id: Id) {
476    let Ok(held) = super::sources_held(args) else {
477        return;
478    };
479    panel_for_held(catalog, kind, id, &held, false);
480}

482/// The label page may deliberately show its retained Discogs profile when
483/// offline or when a refresh failed. Other pages use the filtered loader.
484pub fn panel_for_held(
485    catalog: &Catalog,
486    kind: EntityKind,
487    id: Id,
488    held: &Sources,
489    refresh_failed: bool,
490) {
491    let Some(entity) = EntityRef::of(catalog, kind, id) else {
492        return;
493    };
494    let records: Vec<&SourceRecord> = held.about(&entity).collect();
495    if records.is_empty() {
496        return;
497    }
498
499    println!("{}", ui::section("What sources say"));
500
501    // Prose first, and outside the table. A paragraph does not fit a column,
502    // and it is the one thing here meant to be read rather than compared with
503    // a tag — there is nothing in a file to compare it against.
504    for record in &records {
505        let prose = match &record.facts {
506            Facts::Artist(artist) => artist.summary.as_ref(),
507            Facts::Label(label) => label.summary.as_ref(),
508            _ => None,
509        };
510        if let Some(prose) = prose {
511            if record.source == aede_core::discogs::SOURCE
512                && (refresh_failed
513                    || !aede_core::discogs::fresh(
514                        record.fetched_at,
515                        aede_core::clock::now_seconds(),
516                    ))
517            {
518                println!(
519                    "  {}",
520                    ui::yellow(
521                        "Stored Discogs profile: update needed; the latest version could not be confirmed."
522                    )
523                );
524            }
525            let display = if record.source == aede_core::discogs::SOURCE {
526                aede_core::discogs::render_profile(
527                    &prose.text,
528                    &Default::default(),
529                    &Default::default(),
530                )
531            } else {
532                prose.text.clone()
533            };
534            let lines = if record.source == aede_core::discogs::SOURCE {
535                discogs_profile_lines(&display, 72)
536            } else {
537                ui::wrap(&display, 72)
538            };
539            for line in lines {
540                if line.is_empty() {
541                    println!();
542                } else {
543                    println!("  {line}");
544                }
545            }
546            if record.source == aede_core::discogs::SOURCE {
547                println!(
548                    "  {}",
549                    ui::dim(
550                        "Discogs profile: original text (usually English; no French translation supplied)"
551                    )
552                );
553                println!(
554                    "  {}",
555                    ui::dim(&format!("Data provided by Discogs. {} — CC0", prose.url))
556                );
557                println!(
558                    "  {}",
559                    ui::dim(
560                        "This application uses Discogs’ API but is not affiliated with, sponsored or endorsed by Discogs. ‘Discogs’ is a trademark of Zink Media, LLC."
561                    )
562                );
563                println!(
564                    "  {}",
565                    ui::dim(&format!(
566                        "aede label \"{}\" checks this profile whenever the page opens",
567                        record.key
568                    ))
569                );
570            } else {
571                println!("  {}", ui::dim(&prose.credit()));
572                println!(
573                    "  {}",
574                    ui::dim(&format!(
575                        "aede fetch --summaries --full --lang=<code> \"{}\" asks for another",
576                        record.key
577                    ))
578                );
579            }
580            println!();
581        }
582    }
583
584    let mut table = Table::new(&["Source", "Field", "Says", "Your tags"]).limit(2, 34);
585
586    for record in &records {
587        let age = ui::since(record.fetched_at);
588        let attribution = format!(
589            "{} · {} · {age}",
590            record.source,
591            super::source_status(
592                record.confidence,
593                held.review_for(record).map(|review| review.decision),
594                held.is_trusted(catalog, record),
595            )
596        );
597        for (field, theirs, verdict) in compared(catalog, &entity, &record.facts) {
598            table.push(vec![
599                attribution.clone(),
600                field.to_string(),
601                theirs,
602                // The verdict is the point of the column: a value that matches
603                // the tags is not the same news as one that contradicts them,
604                // and neither is one the tags say nothing about.
605                match verdict {
606                    Some(Verdict::Agrees) => ui::green("matches your tags").to_string(),
607                    Some(Verdict::Differs { yours, .. }) => ui::yellow(&yours).to_string(),
608                    // The tags could have answered and did not.
609                    Some(Verdict::NothingToCompare) => ui::dim("nothing in your tags").to_string(),
610                    // There is no tag for this at all — an artist's country is
611                    // not something the files have an opinion about. Saying
612                    // "nothing in your tags" would suggest they were expected
613                    // to, and send the reader looking for a field to fill.
614                    None => String::new(),
615                },
616            ]);
617        }
618    }
619    // A source that was asked and holds nothing is not an empty table. The
620    // distinction is the one this whole layer exists to keep — "checked, and
621    // it knows nothing" against "never checked" — and a table rendering
622    // "(no results)" says neither of them.
623    if table.is_empty() {
624        for record in &records {
625            // A record whose only content is the paragraph just printed has
626            // no row here, and saying it "holds nothing" directly under its
627            // own text would be plainly false.
628            if matches!(&record.facts, Facts::Artist(a) if a.summary.is_some())
629                || matches!(&record.facts, Facts::Label(l) if l.summary.is_some())
630            {
631                continue;
632            }
633            println!(
634                "  {}",
635                ui::dim(&format!(
636                    "{} was asked and holds nothing about this ({})",
637                    record.source,
638                    ui::since(record.fetched_at)
639                ))
640            );
641        }
642        whence(&records);
643        return;
644    }
645
646    print!("{}", table.render());
647    whence(&records);
648    println!(
649        "{}",
650        ui::dim("  beside your tags, never on top of them — aede sources --forget removes them")
651    );
652}

654/// Discogs separates identity notes from the biography with hard line breaks.
655/// Wrap each source line independently and apply emphasis after measuring it,
656/// so terminal escape sequences do not count toward the available width.
657fn discogs_profile_lines(text: &str, width: usize) -> Vec<String> {
658    text.lines()
659        .flat_map(|line| {
660            if line.trim().is_empty() {
661                return vec![String::new()];
662            }
663            let label_code = line
664                .trim_start()
665                .to_ascii_lowercase()
666                .starts_with("label code:");
667            ui::wrap(line, width)
668                .into_iter()
669                .map(|line| if label_code { ui::bold(&line) } else { line })
670                .collect()
671        })
672        .collect()
673}

675/// Where each of these answers came from, as an address a reader can follow.
676///
677/// **The identifier was stored and never shown**, which is the one thing a
678/// reader needs to check anything by hand: to open the page, to ask the service
679/// the same question, or to correct the data at its source. Asked for the
680/// MusicBrainz identifier of an artist, there was nowhere in this program to
681/// find it — and the nearest thing on screen was the Wikidata link, which is a
682/// different identifier that looks enough like an answer to waste an afternoon.
683///
684/// Printed as the URL rather than the bare identifier: the identifier is inside
685/// it, so it can still be copied for a query, and the page it opens is where a
686/// wrong type or a missing date is actually fixed — for everybody, not just
687/// here.
688fn whence(records: &[&SourceRecord]) {
689    for record in records {
690        if let Some(address) = address_of(record) {
691            println!("  {}", ui::dim(&format!("{}: {address}", record.source)));
692        }
693    }
694}

696/// Where one record's answer can be looked at, or its bare identifier.
697///
698/// Split from the printing so the decision is testable: which path a
699/// MusicBrainz album takes is the sort of thing that is wrong once and then
700/// wrong for a year, because a link that 404s looks like the service's fault.
701fn address_of(record: &SourceRecord) -> Option<String> {
702    let id = record.source_id.as_deref()?;
703    let kind = match record.facts.kind() {
704        EntityKind::Artist => "artist",
705        // The identifier kept for an album is the release group's, not the
706        // edition's — see `musicbrainz::release`. Naming the wrong path here
707        // would produce a link that 404s on the one page a reader came to open.
708        EntityKind::Release => "release-group",
709        EntityKind::Label => "label",
710        // Other entity kinds have no known MusicBrainz URL here.
711        _ => return Some(id.to_string()),
712    };
713    Some(match record.source.as_str() {
714        sources::MUSICBRAINZ => format!("https://musicbrainz.org/{kind}/{id}"),
715        aede_core::discogs::SOURCE if kind == "label" => {
716            format!("https://www.discogs.com/label/{id}")
717        }
718        "wikipedia" => format!("https://www.wikidata.org/wiki/{id}"),
719        // A source this build knows nothing about still has an identifier, and
720        // the identifier is the useful half. Inventing an address for it would
721        // be worse than showing none.
722        _ => id.to_string(),
723    })
724}

726/// The first value of a tag on any file credited to this artist.
727///
728/// A genre lives on the tracks, not on the artist, so "what do my files call
729/// this artist" is answered by the files they appear on.
730/// What one track's own file says under a tag.
731///
732/// Its own file, not its release's: a fingerprint answers for one recording,
733/// and comparing it with an album-level tag would put the wrong two things
734/// side by side.
735fn tags_of_track(catalog: &Catalog, entity: &EntityRef, tag: &str) -> Vec<String> {
736    let Some(id) = entity.resolve(catalog) else {
737        return Vec::new();
738    };
739    catalog
740        .track(id)
741        .and_then(|track| catalog.file(track.file_id))
742        .and_then(|file| file.tags.get(tag))
743        .cloned()
744        .unwrap_or_default()
745}
746
747fn tags_of_artist(catalog: &Catalog, entity: &EntityRef, tag: &str) -> Vec<String> {
748    let Some(id) = entity.resolve(catalog) else {
749        return Vec::new();
750    };
751    // **Every** value, not the first: a genre tag holds a list, whether the
752    // file writes it as several values or as one string with commas in it.
753    // Reading only the first turned "Rock, Pop" into "Rock" on some files and
754    // reported a disagreement with a genre the tags do carry.
755    catalog
756        .tracks_of_artist(id)
757        .into_iter()
758        .filter_map(|t| catalog.track(t))
759        .find_map(|t| {
760            catalog
761                .file(t.file_id)
762                .map(|f| f.tags.get(tag).cloned().unwrap_or_default())
763                .filter(|values: &Vec<String>| !values.is_empty())
764        })
765        .unwrap_or_default()
766}

768/// Each field a record carries, with the tag it can be judged against.
769///
770/// `None` where the catalog holds no counterpart at all: an artist's country
771/// is not something the tags have an opinion about, and printing "nothing in
772/// your tags" against it would suggest the tags were expected to.
773fn compared(
774    catalog: &Catalog,
775    entity: &EntityRef,
776    facts: &Facts,
777) -> Vec<(&'static str, String, Option<Verdict>)> {
778    let mut rows: Vec<(&'static str, String, Option<Verdict>)> = Vec::new();
779    match facts {
780        Facts::Track(t) => {
781            // Compared with the tags, which is the whole use of this: a file
782            // whose audio and whose tags disagree is what fingerprinting is
783            // for, and the panel is where the two sit side by side.
784            if !t.artists.is_empty() {
785                rows.push((
786                    "heard artist",
787                    t.artists.join(", "),
788                    Some(sources::verdict_set(
789                        &t.artists,
790                        &tags_of_track(catalog, entity, "artist"),
791                    )),
792                ));
793            }
794            if let Some(title) = &t.title {
795                rows.push((
796                    "heard title",
797                    title.clone(),
798                    Some(sources::verdict(
799                        title,
800                        tags_of_track(catalog, entity, "title")
801                            .first()
802                            .map(String::as_str),
803                    )),
804                ));
805            }
806            for (field, value) in [("heard on", &t.album), ("recording", &t.recording)] {
807                if let Some(value) = value {
808                    rows.push((field, value.clone(), None));
809                }
810            }
811            for work in &t.works {
812                rows.push(("work", format!("{} ({})", work.title, work.mbid), None));
813                for credit in &work.credits {
814                    rows.push(("work credit", credit_summary(credit), None));
815                }
816            }
817            for credit in &t.credits {
818                rows.push(("recording credit", credit_summary(credit), None));
819            }
820            if let Some(score) = t.score {
821                // Shown, never hidden behind a threshold: a reader deciding
822                // whether to believe this needs the number the service gave,
823                // and a program that quietly dropped the weak matches would
824                // be making that decision for them.
825                rows.push((("how sure"), format!("{score}%"), None));
826            }
827        }
828        Facts::Artist(a) => {
829            // "country" was wrong: MusicBrainz calls this an *area*, and an
830            // area is a country, a city or a region — "Seattle" is a valid
831            // answer. "from" is true of all three.
832            //
833            // "known as" was wrong too, and worse: `disambiguation` is written
834            // to separate two artists who share a name, not to describe one,
835            // which is how a perfectly correct answer came out as
836            // "known as: the band".
837            for (field, value) in [
838                ("type", &a.kind),
839                ("from", &a.area),
840                ("formed", &a.began),
841                ("ended", &a.ended),
842                ("note", &a.disambiguation),
843                ("wikidata", &a.wikidata),
844                ("discogs", &a.discogs),
845                ("homepage", &a.homepage),
846            ] {
847                if let Some(value) = value {
848                    rows.push((field, value.clone(), None));
849                }
850            }
851            if !a.genres.is_empty() {
852                // Beside the genre tag, never over it: the last column is what
853                // your files say, and these two answers are both allowed to
854                // exist.
855                //
856                // Compared as **sets**, which the first version did not do: it
857                // took MusicBrainz's top genre and compared it, as a string, to
858                // the whole tag. `pop` against `Rock, Pop` came out as a
859                // disagreement, when the tag plainly says pop. Genres are not
860                // exclusive and the two sides are not answering at the same
861                // granularity — one is what a crowd voted, the other is what
862                // one person typed — so a shared name is agreement.
863                let theirs = a.genres.join(", ");
864                let yours = tags_of_artist(catalog, entity, "genre");
865                let verdict = aede_core::sources::verdict_set(&a.genres, &yours);
866                rows.push(("genres", theirs, Some(verdict)));
867            }
868            if !a.aliases.is_empty() {
869                rows.push(("also called", a.aliases.join(", "), None));
870            }
871            // Shown even when there is no end date, because that is the case
872            // it answers: a band with no end may have never stopped, or may
873            // be one nobody filled in, and the reader cannot tell otherwise.
874            if let Some(active) = a.active {
875                rows.push((
876                    "status",
877                    match active {
878                        true => "still active".to_string(),
879                        false => "ended".to_string(),
880                    },
881                    None,
882                ));
883            }
884            // No verdict to compare it with — a tag has no field for a
885            // picture — so this is shown, not judged, the same as the
886            // discography and the members above it.
887            if let Some(picture) = &a.portrait {
888                rows.push(("portrait", picture.url.clone(), None));
889            }
890        }
891        Facts::Release(r) => {
892            let release = entity
893                .resolve(catalog)
894                .and_then(|id| catalog.release(id))
895                .cloned();
896            let tag = |name: &str| -> Option<String> {
897                release.as_ref().and_then(|rel| {
898                    rel.track_ids
899                        .first()
900                        .and_then(|&t| catalog.track(t))
901                        .and_then(|t| catalog.file(t.file_id))
902                        .and_then(|f| f.first_tag(name).map(str::to_string))
903                })
904            };
905
906            if let Some(primary) = &r.primary_type {
907                let mut shown = primary.clone();
908                if !r.secondary_types.is_empty() {
909                    shown = format!("{shown} · {}", r.secondary_types.join(" · "));
910                }
911                rows.push((
912                    "release type",
913                    shown,
914                    Some(aede_core::sources::verdict(
915                        primary,
916                        tag("releasetype").as_deref(),
917                    )),
918                ));
919            }
920            if let Some(date) = &r.first_released {
921                let yours = release.as_ref().and_then(|rel| rel.date.clone());
922                rows.push((
923                    "first released",
924                    date.clone(),
925                    Some(aede_core::sources::verdict_date(date, yours.as_deref())),
926                ));
927            }
928            if let Some(label) = &r.label {
929                let yours = release.as_ref().and_then(|rel| {
930                    rel.label_ids
931                        .first()
932                        .and_then(|&id| catalog.label(id))
933                        .map(|l| l.name.clone())
934                });
935                rows.push((
936                    "label",
937                    label.clone(),
938                    Some(aede_core::sources::verdict(label, yours.as_deref())),
939                ));
940            }
941        }
942        // Nothing to show — see `LabelFacts`. There is no tag for a label's
943        // own identifier to be compared against, the same reason an
944        // artist's row above compares no name either.
945        Facts::Label(label) => {
946            if let Some(url) = &label.wikidata {
947                rows.push(("wikidata", url.clone(), None));
948            }
949            if let Some(url) = &label.discogs {
950                rows.push(("discogs", url.clone(), None));
951            }
952        }
953    }
954    rows
955}

957fn credit_summary(credit: &sources::CreditLink) -> String {
958    let credited = credit
959        .credited_as
960        .as_deref()
961        .filter(|name| *name != credit.artist_name)
962        .map(|name| format!(" as {name}"))
963        .unwrap_or_default();
964    let attributes = credit
965        .attributes
966        .iter()
967        .map(|attribute| {
968            attribute
969                .credited_as
970                .as_deref()
971                .or(attribute.value.as_deref())
972                .unwrap_or(&attribute.name)
973                .to_string()
974        })
975        .collect::<Vec<_>>();
976    let details = match attributes.is_empty() {
977        true => String::new(),
978        false => format!(" ({})", attributes.join(", ")),
979    };
980    format!(
981        "{} — {}{credited}{details}",
982        credit.role, credit.artist_name
983    )
984}
