1//! The `sources` command: what somebody else said about this library.
2//!
3//! The catalog holds what the files say, the annotation file holds what you
4//! say, and `sources.json` holds what a source says — MusicBrainz first, a
5//! plugin later. This is the command that shows it, takes it in, and drops it.
6//!
7//! It exists before anything fetches, on purpose. A store that can only be
8//! filled over the network can only be tested over the network, and the layer
9//! is worth more than that: `--import` reads a document in exactly the shape
10//! the file has, so the whole of it — attachment, agreement, removal — can be
11//! exercised offline, and so can a bug report.
12//!
13//! The shape follows `aede import`, which does the same job for another tool's
14//! analyses: `--list` says what is held, `--forget` removes it, `--source`
15//! narrows to one. Two stores of what somebody else said should not be learned
16//! twice.
17

26pub fn sources(args: &Args) -> Res {
27    if args.has("yes") && !args.has("forget") {
28        return Err("--yes requires --forget for sources".into());
29    }
30    let operations: Vec<&str> = ["import", "template", "export", "forget", "list"]
31        .into_iter()
32        .filter(|option| args.has(option))
33        .collect();
34    if operations.len() > 1 {
35        return Err(
36            "choose one sources operation: --import, --template, --export, --forget, or --list"
37                .into(),
38        );
39    }
40    if args.has("source") {
41        if !args.has("list") && !args.has("forget") && !args.has("template") {
42            return Err("--source is used with --list, --forget, or --template".into());
43        }
44        if args
45            .value("source")
46            .is_none_or(|value| value.trim().is_empty())
47        {
48            return Err("--source expects a non-empty source name".into());
49        }
50    }
51    if args.has("output") && !args.has("export") && !args.has("template") {
52        return Err("--output is used with --export or --template".into());
53    }
54    if !args.positionals.is_empty() && !args.has("template") {
55        return Err("names select entities only with --template on sources".into());
56    }
57    let dir = data_dir(args);
58    let path = sources::sources_path(&dir);
59
60    if args.has("import") {
61        return import(args, &path);
62    }
63    if args.has("template") {
64        return template(args);
65    }
66    if args.has("export") {
67        return export(args, &path);
68    }
69    if args.has("forget") {
70        return forget(args, &path);
71    }
72
73    let held = sources::load(&path)?.unwrap_or_default();
74    if held.records.is_empty() {
75        println!("{}", ui::section("Sources"));
76        println!("  {}", ui::dim("nothing has been fetched yet"));
77        // The store being empty is not the same as the feature being absent,
78        // and a reader who cannot tell which will conclude the wrong one.
79        println!(
80            "  {}",
81            ui::dim("aede sources --import <file> takes in a document by hand")
82        );
83        return Ok(());
84    }
85
86    let catalog = load(args)?;
87    match args.has("list") {
88        true => list(&held, &catalog, args),
89        false => summary(&held, &catalog),
90    }
91}

93/// One line per source: how much it said, and how much of it lands.
94fn summary(held: &Sources, catalog: &Catalog) -> Res {
95    println!("{}", ui::section("Sources"));
96
97    let mut names: Vec<&str> = held.records.iter().map(|r| r.source.as_str()).collect();
98    names.sort_unstable();
99    names.dedup();
100
101    let mut table = Table::new(&["Source", "Records", "Attached", "Waiting", "Last fetch"])
102        .align(1, Align::Right)
103        .align(2, Align::Right)
104        .align(3, Align::Right);
105
106    for name in &names {
107        let mine = Sources {
108            records: held
109                .records
110                .iter()
111                .filter(|r| r.source == *name)
112                .cloned()
113                .collect(),
114            reviews: held
115                .reviews
116                .iter()
117                .filter(|review| review.source == *name)
118                .cloned()
119                .collect(),
120            credit_exclusions: held
121                .credit_exclusions
122                .iter()
123                .filter(|decision| decision.relation.provenance == *name)
124                .cloned()
125                .collect(),
126        };
127        let reach = sources::attachment(&mine, catalog);
128        let last = mine.records.iter().map(|r| r.fetched_at).max().unwrap_or(0);
129        table.push(vec![
130            (*name).to_string(),
131            mine.records.len().to_string(),
132            reach.attached.to_string(),
133            reach.waiting.to_string(),
134            ui::since(last),
135        ]);
136    }
137    println!("{}", table.render());
138
139    // A count of what is waiting means nothing without the reason, and the
140    // reason is always the same one: the catalog does not hold that entity
141    // yet. Saying so here is what stops it reading as a failure.
142    let all = sources::attachment(held, catalog);
143    if all.waiting > 0 {
144        // Both verbs agree with the count, because `ui::plural` puts the count
145        // in the sentence and the rest of the sentence has to follow it.
146        let (name, kept) = match all.waiting {
147            1 => ("names", "is"),
148            _ => ("name", "are"),
149        };
150        println!(
151            "  {}",
152            ui::dim(&format!(
153                "{} {name} nothing this catalog holds yet, and {kept} kept until it does",
154                ui::plural(all.waiting, "record")
155            ))
156        );
157    }
158    Ok(())
159}

161/// Every record, with what it says and whether the catalog can place it.
162fn list(held: &Sources, catalog: &Catalog, args: &Args) -> Res {
163    let only = args.value("source");
164    let rows: Vec<&SourceRecord> = held
165        .records
166        .iter()
167        .filter(|r| only.is_none_or(|s| r.source == s))
168        .collect();
169
170    if rows.is_empty() {
171        println!("{}", ui::section("Sources"));
172        println!(
173            "  {}",
174            ui::dim(&ui::literal(&match only {
175                Some(name) => format!("nothing was fetched from \"{name}\""),
176                None => "nothing has been fetched yet".to_string(),
177            }))
178        );
179        return Ok(());
180    }
181
182    println!("{}", ui::section("What sources say"));
183    let mut table = Table::new(&["Entity", "Source", "Confidence", "Says", "Where"])
184        .limit(0, 42)
185        .limit(3, 46);
186    for record in &rows {
187        let entity = record.entity();
188        table.push(vec![
189            format!("{} {}", entity.kind.as_str(), entity.key),
190            record.source.clone(),
191            super::source_status(
192                record.confidence,
193                held.review_for(record).map(|review| review.decision),
194                held.is_trusted(catalog, record),
195            ),
196            says(&record.facts),
197            match entity.resolve(catalog).is_some() {
198                true => "in the catalog".to_string(),
199                false => "waiting".to_string(),
200            },
201        ]);
202    }
203    println!("{}", table.render());
204    println!("  {}", ui::dim(&ui::plural(rows.len(), "record")));
205    Ok(())
206}

208/// What a record says, short enough for a column.
209fn says(facts: &Facts) -> String {
210    let mut parts: Vec<String> = Vec::new();
211    match facts {
212        Facts::Track(t) => {
213            // What it heard, in the order somebody reads it: who, then what.
214            // The score leads nothing — it qualifies the rest, and a line
215            // opening with a number reads as a measurement of the file.
216            if !t.artists.is_empty() {
217                parts.push(t.artists.join(", "));
218            }
219            if let Some(title) = &t.title {
220                parts.push(title.clone());
221            }
222            if let Some(album) = &t.album {
223                parts.push(album.clone());
224            }
225            if !t.works.is_empty() {
226                parts.push(format!("{} work link(s)", t.works.len()));
227            }
228            if !t.credits.is_empty() {
229                parts.push(ui::plural(t.credits.len(), "recording credit"));
230            }
231            if let Some(recording) = &t.recording {
232                parts.push(format!("recording: {recording}"));
233            }
234            if let Some(score) = t.score {
235                parts.push(format!("{score}% sure"));
236            }
237        }
238        Facts::Artist(a) => {
239            if let Some(kind) = &a.kind {
240                parts.push(kind.clone());
241            }
242            if let Some(known) = &a.disambiguation {
243                parts.push(known.clone());
244            }
245            if let Some(area) = a.area.as_ref().or(a.country_code.as_ref()) {
246                parts.push(area.clone());
247            }
248            match (&a.began, &a.ended) {
249                (Some(from), Some(to)) => parts.push(format!("{from}–{to}")),
250                (Some(from), None) => parts.push(format!("{from}–")),
251                (None, Some(to)) => parts.push(format!("ended {to}")),
252                _ => {}
253            }
254            if let Some(active) = a.active {
255                parts.push(match active {
256                    true => "still active".to_string(),
257                    false => "ended".to_string(),
258                });
259            }
260            if a.portrait.is_some() {
261                parts.push("has a portrait".to_string());
262            }
263            if a.logo.is_some() {
264                parts.push("has a logo".to_string());
265            }
266            if !a.genres.is_empty() {
267                parts.push(a.genres.join(", "));
268            }
269            if !a.aliases.is_empty() {
270                parts.push(format!("also called {}", a.aliases.join(", ")));
271            }
272            if !a.members.is_empty() {
273                parts.push(ui::plural(a.members.len(), "membership"));
274            }
275            if !a.discography.is_empty() {
276                parts.push(ui::plural(a.discography.len(), "release"));
277            }
278            for (present, description) in [
279                (a.wikidata.is_some(), "has a Wikidata link"),
280                (a.discogs.is_some(), "has a Discogs link"),
281                (a.homepage.is_some(), "has a homepage"),
282            ] {
283                if present {
284                    parts.push(description.to_string());
285                }
286            }
287            // The detail card prints the prose with its complete credit. This
288            // short table cannot return an extract without that attribution.
289            if a.summary.is_some() {
290                parts.push("has a biography".to_string());
291            }
292        }
293        Facts::Release(r) => {
294            if let Some(primary) = &r.primary_type {
295                parts.push(primary.clone());
296            }
297            parts.extend(r.secondary_types.iter().cloned());
298            if let Some(date) = &r.first_released {
299                parts.push(date.clone());
300            }
301            if let Some(label) = &r.label {
302                parts.push(label.clone());
303            }
304            if r.label.is_none()
305                && let Some(label) = &r.label_mbid
306            {
307                parts.push(format!("label: {label}"));
308            }
309            if r.cover_art.is_some() {
310                parts.push("has a cover".to_string());
311            }
312            if !r.credits.is_empty() {
313                parts.push(ui::plural(r.credits.len(), "edition credit"));
314            }
315            if let Some(edition) = &r.edition_mbid {
316                parts.push(format!("edition: {edition}"));
317            }
318        }
319        Facts::Label(LabelFacts {
320            logo,
321            summary,
322            wikidata,
323            discogs,
324            ..
325        }) => {
326            if let Some(url) = wikidata {
327                parts.push(url.clone());
328            }
329            if let Some(url) = discogs {
330                parts.push(url.clone());
331            }
332            if summary.is_some() {
333                parts.push("has a profile".to_string());
334            }
335            if let Some(logo) = logo {
336                parts.push(format!("logo: {}", logo.url));
337            }
338        }
339    }
340    match parts.is_empty() {
341        // An answer holding nothing is a real answer, and printing an empty
342        // cell would make it look like a missing row instead.
343        true => ui::dim("nothing"),
344        false => parts.join(" · "),
345    }
346}

348/// Takes in a document written in the shape of `sources.json`.
349fn import(args: &Args, path: &std::path::Path) -> Res {
350    let Some(file) = args.value("import") else {
351        return Err("--import expects a file: --import=<file.json>".into());
352    };
353    let text = std::fs::read_to_string(file).map_err(|e| format!("cannot read \"{file}\": {e}"))?;
354    let value =
355        aede_core::json::parse(&text).map_err(|e| format!("\"{file}\" is not JSON: {e}"))?;
356    let incoming = sources::from_json(&value)?;
357
358    let mut held = sources::load_all(path)?.unwrap_or_default();
359    let (mut updated, mut added) = (0usize, 0usize);
360    for record in incoming.records {
361        match held.set(record) {
362            true => updated += 1,
363            false => added += 1,
364        }
365    }
366    for review in incoming.reviews {
367        held.set_review(review);
368    }
369    for decision in incoming.credit_exclusions {
370        held.exclude_credit(decision.relation, decision.excluded_at);
371    }
372    sources::save(&held, path)?;
373
374    println!("{}", ui::section("Imported"));
375    println!(
376        "{} {} added, {} updated, {} held",
377        ui::green("→"),
378        added,
379        updated,
380        held.records.len()
381    );
382    // Naming the file it wrote, for the same reason `import --forget` does: a
383    // command that changed a store the user cannot see should say which one.
384    println!("  {}", ui::dim(&ui::literal(&path.display().to_string())));
385    Ok(())
386}

388/// Writes the layer out, in the shape `--import` reads.
389///
390/// The symmetry matters more than the feature: a store that can be filled and
391/// not emptied out again is a store whose contents you have to trust somebody
392/// else about. `aede notes --export` already draws the same round trip for
393/// what the user writes.
394fn export(args: &Args, path: &std::path::Path) -> Res {
395    let held = sources::load(path)?.unwrap_or_default();
396    if held.records.is_empty() && held.reviews.is_empty() && held.credit_exclusions.is_empty() {
397        // Writing an empty document and reporting success would look like the
398        // export worked and the layer was empty — two different things, and
399        // the second is not what has just been established.
400        return Err(
401            "nothing has been fetched yet, so there is nothing to export.\n\
402             aede sources --template=<file> writes a document you can fill in"
403                .into(),
404        );
405    }
406    super::export::emit(args, &sources::to_json(&held).to_string_pretty())
407}

409/// Writes a document with the right keys and nothing filled in.
410///
411/// The keys are the one thing a person cannot guess: they are how an entity
412/// names itself — a normalised artist name, or an album's artist, title and
413/// folder joined — and until something fetches, there is no way to see one.
414/// Without this, the layer could be read and emptied but never filled by hand,
415/// which makes it untestable by the person who most needs to test it.
416///
417/// Naming entities narrows it:
418/// `aede sources --template --output=x.json "Kind of Blue"` writes that album
419/// alone rather than a skeleton of the whole library.
420fn template(args: &Args) -> Res {
421    let catalog = load(args)?;
422    let source = args.value("source").unwrap_or(sources::MUSICBRAINZ);
423    let now = aede_core::clock::now_seconds();
424
425    let wanted: Vec<String> = args
426        .positionals
427        .iter()
428        .map(|name| aede_core::text::normalize(name))
429        .filter(|name| !name.is_empty())
430        .collect();
431    let matches = |name: &str| -> bool {
432        let key = aede_core::text::normalize(name);
433        wanted.is_empty() || wanted.iter().any(|w| key.contains(w.as_str()))
434    };
435
436    let mut out = Sources::default();
437    let blank = |key: String, facts: Facts| SourceRecord {
438        key,
439        source: source.to_string(),
440        source_id: None,
441        fetched_at: now,
442        confidence: Confidence::Identified,
443        facts,
444    };
445    for artist in &catalog.artists {
446        if !matches(&artist.name) {
447            continue;
448        }
449        if let Some(entity) = EntityRef::of(&catalog, EntityKind::Artist, artist.id) {
450            out.set(blank(entity.key, Facts::Artist(Default::default())));
451        }
452    }
453    for release in &catalog.releases {
454        if !matches(&release.title) {
455            continue;
456        }
457        if let Some(entity) = EntityRef::of(&catalog, EntityKind::Release, release.id) {
458            out.set(blank(entity.key, Facts::Release(Default::default())));
459        }
460    }
461
462    if out.records.is_empty() {
463        return Err(match wanted.is_empty() {
464            true => "this catalog holds nothing to write a template for".to_string(),
465            false => format!(
466                "nothing in this catalog matches \"{}\"",
467                args.positionals.join(" ")
468            ),
469        }
470        .into());
471    }
472
473    super::export::emit(args, &sources::to_json(&out).to_string_pretty())?;
474    // Only when it went to a file: a template printed to the terminal is on
475    // screen, and telling somebody to import what they can see is noise.
476    if args.value("output").is_some() {
477        println!(
478            "  {} {}",
479            ui::plural(out.records.len(), "empty record"),
480            ui::dim("— fill in what you want, then: aede sources --import=<file>")
481        );
482    }
483    Ok(())
484}

486/// Drops what one source said, or everything.
487fn forget(args: &Args, path: &std::path::Path) -> Res {
488    let mut held = sources::load_all(path)?.unwrap_or_default();
489    let before = held.records.len();
490    let reviews_before = held.reviews.len();
491    let exclusions_before = held.credit_exclusions.len();
492    let decisions_before = reviews_before + exclusions_before;
493    if before == 0 && decisions_before == 0 {
494        println!("{}", ui::section("Sources"));
495        println!("  {}", ui::dim("there was nothing to forget"));
496        return Ok(());
497    }
498
499    let removed = match args.value("source") {
500        Some(name) => {
501            let gone = held.forget(name);
502            if gone == 0 && decisions_before == held.reviews.len() + held.credit_exclusions.len() {
503                return Err(format!(
504                    "no source named \"{name}\" — aede sources says which there are"
505                )
506                .into());
507            }
508            gone
509        }
510        None => {
511            let removed = std::mem::take(&mut held.records).len();
512            held.reviews.clear();
513            held.credit_exclusions.clear();
514            removed
515        }
516    };
517
518    println!("{}", ui::section("Forget sourced data"));
519    println!(
520        "  data: {}",
521        ui::literal(&super::canonical(&data_dir(args)).display().to_string())
522    );
523    println!("  store: {}", ui::literal(&path.display().to_string()));
524    println!(
525        "  source: {}",
526        ui::literal(args.value("source").unwrap_or("all sources"))
527    );
528    let mut table = Table::plain(2).align(1, Align::Right);
529    table.push(vec!["Records to remove".into(), removed.to_string()]);
530    table.push(vec![
531        "Reviews to remove".into(),
532        (reviews_before - held.reviews.len()).to_string(),
533    ]);
534    table.push(vec![
535        "Credit exclusions to remove".into(),
536        (exclusions_before - held.credit_exclusions.len()).to_string(),
537    ]);
538    print!("{}", table.render());
539    println!(
540        "  A new fetch cannot recreate personal decisions or manually authored source records. Keep a backup."
541    );
542    if !super::confirmed(args, "remove the selected source records and decisions")? {
543        println!("  {}", ui::green("nothing was removed"));
544        return Ok(());
545    }
546    sources::save(&held, path)?;
547    println!(
548        "{} {} removed, {} left",
549        ui::green("→"),
550        ui::plural(removed, "record"),
551        held.records.len()
552    );
553    let decisions_removed = decisions_before - held.reviews.len() - held.credit_exclusions.len();
554    if decisions_removed > 0 {
555        println!("  {} removed", ui::plural(decisions_removed, "decision"));
556    }
557    println!("  {}", ui::dim(&ui::literal(&path.display().to_string())));
558    Ok(())
559}

565/// The block an entity page prints when a source has said something about it.
566///
567/// One function called from every page rather than one block written per page:
568/// what a source says has to read the same way on an album and on an artist,
569/// and a rule copied into four pages is a rule that will be right in three of
570/// them.
571///
572/// Silent when nothing was fetched. A page that announced an empty section for
573/// a feature nobody has used yet would be noise on every album in the library.
574pub fn panel_for(args: &Args, catalog: &Catalog, kind: EntityKind, id: Id) {
575    let Ok(held) = super::sources_held(args) else {
576        return;
577    };
578    panel_for_held(catalog, kind, id, &held, false);
579}

581/// The label page may deliberately show its retained Discogs profile when
582/// offline or when a refresh failed. Other pages use the filtered loader.
583pub fn panel_for_held(
584    catalog: &Catalog,
585    kind: EntityKind,
586    id: Id,
587    held: &Sources,
588    refresh_failed: bool,
589) {
590    let Some(entity) = EntityRef::of(catalog, kind, id) else {
591        return;
592    };
593    let records: Vec<&SourceRecord> = held.about(&entity).collect();
594    if records.is_empty() {
595        return;
596    }
597
598    println!("{}", ui::section("What sources say"));
599
600    // Prose first, and outside the table. A paragraph does not fit a column,
601    // and it is the one thing here meant to be read rather than compared with
602    // a tag — there is nothing in a file to compare it against.
603    for record in &records {
604        let prose = match &record.facts {
605            Facts::Artist(artist) => artist.summary.as_ref(),
606            Facts::Label(label) => label.summary.as_ref(),
607            _ => None,
608        };
609        if let Some(prose) = prose {
610            if record.source == aede_core::discogs::SOURCE
611                && (refresh_failed
612                    || !aede_core::discogs::fresh(
613                        record.fetched_at,
614                        aede_core::clock::now_seconds(),
615                    ))
616            {
617                println!(
618                    "  {}",
619                    ui::yellow(
620                        "Stored Discogs profile: update needed; the latest version could not be confirmed."
621                    )
622                );
623            }
624            let display = if record.source == aede_core::discogs::SOURCE {
625                aede_core::discogs::render_profile(
626                    &prose.text,
627                    &Default::default(),
628                    &Default::default(),
629                )
630            } else {
631                prose.text.clone()
632            };
633            let lines = if record.source == aede_core::discogs::SOURCE {
634                discogs_profile_lines(&display, 72)
635            } else {
636                ui::wrap(&ui::literal(&display), 72)
637            };
638            for line in lines {
639                if line.is_empty() {
640                    println!();
641                } else {
642                    println!("  {line}");
643                }
644            }
645            if record.source == aede_core::discogs::SOURCE {
646                println!(
647                    "  {}",
648                    ui::dim(
649                        "Discogs profile: original text (usually English; no French translation supplied)"
650                    )
651                );
652                println!(
653                    "  {}",
654                    ui::dim(&ui::literal(&format!(
655                        "Data provided by Discogs. {} — CC0",
656                        prose.url
657                    )))
658                );
659                println!(
660                    "  {}",
661                    ui::dim(
662                        "This application uses Discogs’ API but is not affiliated with, sponsored or endorsed by Discogs. ‘Discogs’ is a trademark of Zink Media, LLC."
663                    )
664                );
665                println!(
666                    "  {}",
667                    ui::dim(&ui::literal(&format!(
668                        "aede label \"{}\" checks this profile whenever the page opens",
669                        record.key
670                    )))
671                );
672            } else {
673                println!("  {}", ui::dim(&ui::literal(&prose.credit())));
674                println!(
675                    "  {}",
676                    ui::dim(&ui::literal(&format!(
677                        "aede fetch --summaries --full --lang=<code> \"{}\" asks for another",
678                        record.key
679                    )))
680                );
681            }
682            println!();
683        }
684    }
685
686    let mut table = Table::new(&["Source", "Field", "Says", "Your tags"]).limit(2, 34);
687
688    for record in &records {
689        let age = ui::since(record.fetched_at);
690        let attribution = format!(
691            "{} · {} · {age}",
692            record.source,
693            super::source_status(
694                record.confidence,
695                held.review_for(record).map(|review| review.decision),
696                held.is_trusted(catalog, record),
697            )
698        );
699        for (field, theirs, verdict) in compared(catalog, &entity, &record.facts) {
700            table.push(vec![
701                attribution.clone(),
702                field.to_string(),
703                theirs,
704                // The verdict is the point of the column: a value that matches
705                // the tags is not the same news as one that contradicts them,
706                // and neither is one the tags say nothing about.
707                match verdict {
708                    Some(Verdict::Agrees) => ui::green("matches your tags").to_string(),
709                    Some(Verdict::Differs { yours, .. }) => ui::yellow(&yours).to_string(),
710                    // The tags could have answered and did not.
711                    Some(Verdict::NothingToCompare) => ui::dim("nothing in your tags").to_string(),
712                    // There is no tag for this at all — an artist's country is
713                    // not something the files have an opinion about. Saying
714                    // "nothing in your tags" would suggest they were expected
715                    // to, and send the reader looking for a field to fill.
716                    None => String::new(),
717                },
718            ]);
719        }
720    }
721    // A source that was asked and holds nothing is not an empty table. The
722    // distinction is the one this whole layer exists to keep — "checked, and
723    // it knows nothing" against "never checked" — and a table rendering
724    // "(no results)" says neither of them.
725    if table.is_empty() {
726        for record in &records {
727            // A record whose only content is the paragraph just printed has
728            // no row here, and saying it "holds nothing" directly under its
729            // own text would be plainly false.
730            if matches!(&record.facts, Facts::Artist(a) if a.summary.is_some())
731                || matches!(&record.facts, Facts::Label(l) if l.summary.is_some())
732            {
733                continue;
734            }
735            println!(
736                "  {}",
737                ui::dim(&ui::literal(&format!(
738                    "{} was asked and holds nothing about this ({})",
739                    record.source,
740                    ui::since(record.fetched_at)
741                )))
742            );
743        }
744        whence(&records);
745        return;
746    }
747
748    print!("{}", table.render());
749    whence(&records);
750    println!(
751        "{}",
752        ui::dim("  beside your tags, never on top of them — aede sources --forget removes them")
753    );
754}

756/// Discogs separates identity notes from the biography with hard line breaks.
757/// Wrap each source line independently and apply emphasis after measuring it,
758/// so terminal escape sequences do not count toward the available width.
759fn discogs_profile_lines(text: &str, width: usize) -> Vec<String> {
760    text.lines()
761        .flat_map(|line| {
762            if line.trim().is_empty() {
763                return vec![String::new()];
764            }
765            let label_code = line
766                .trim_start()
767                .to_ascii_lowercase()
768                .starts_with("label code:");
769            ui::wrap(&ui::literal(line), width)
770                .into_iter()
771                .map(|line| if label_code { ui::bold(&line) } else { line })
772                .collect()
773        })
774        .collect()
775}

777/// Where each of these answers came from, as an address a reader can follow.
778///
779/// **The identifier was stored and never shown**, which is the one thing a
780/// reader needs to check anything by hand: to open the page, to ask the service
781/// the same question, or to correct the data at its source. Asked for the
782/// MusicBrainz identifier of an artist, there was nowhere in this program to
783/// find it — and the nearest thing on screen was the Wikidata link, which is a
784/// different identifier that looks enough like an answer to waste an afternoon.
785///
786/// Printed as the URL rather than the bare identifier: the identifier is inside
787/// it, so it can still be copied for a query, and the page it opens is where a
788/// wrong type or a missing date is actually fixed — for everybody, not just
789/// here.
790fn whence(records: &[&SourceRecord]) {
791    for record in records {
792        if let Some(address) = address_of(record) {
793            println!(
794                "  {}",
795                ui::dim(&ui::literal(&format!("{}: {address}", record.source)))
796            );
797        }
798    }
799}

801/// Where one record's answer can be looked at, or its bare identifier.
802///
803/// Split from the printing so the decision is testable: which path a
804/// MusicBrainz album takes is the sort of thing that is wrong once and then
805/// wrong for a year, because a link that 404s looks like the service's fault.
806pub(super) fn address_of(record: &SourceRecord) -> Option<String> {
807    let id = record.source_id.as_deref()?;
808    let kind = match record.facts.kind() {
809        EntityKind::Artist => "artist",
810        // Album snapshots normally name the group. A lookup without a group
811        // instead retains the edition ID in both fields — see
812        // `musicbrainz::release`. Do not send that fallback to the group route.
813        EntityKind::Release
814            if matches!(
815                record.musicbrainz_release_identity(),
816                Some(aede_core::sources::MusicBrainzReleaseIdentity::Edition(_))
817            ) =>
818        {
819            "release"
820        }
821        EntityKind::Release => "release-group",
822        EntityKind::Label => "label",
823        // Other entity kinds have no known MusicBrainz URL here.
824        _ => return Some(id.to_string()),
825    };
826    Some(match record.source.as_str() {
827        sources::MUSICBRAINZ => format!("https://musicbrainz.org/{kind}/{id}"),
828        aede_core::discogs::SOURCE if kind == "label" => {
829            format!("https://www.discogs.com/label/{id}")
830        }
831        "wikipedia" => format!("https://www.wikidata.org/wiki/{id}"),
832        // A source this build knows nothing about still has an identifier, and
833        // the identifier is the useful half. Inventing an address for it would
834        // be worse than showing none.
835        _ => id.to_string(),
836    })
837}

839/// The first value of a tag on any file credited to this artist.
840///
841/// A genre lives on the tracks, not on the artist, so "what do my files call
842/// this artist" is answered by the files they appear on.
843/// What one track's own file says under a tag.
844///
845/// Its own file, not its release's: a fingerprint answers for one recording,
846/// and comparing it with an album-level tag would put the wrong two things
847/// side by side.
848fn tags_of_track(catalog: &Catalog, entity: &EntityRef, tag: &str) -> Vec<String> {
849    let Some(id) = entity.resolve(catalog) else {
850        return Vec::new();
851    };
852    catalog
853        .track(id)
854        .and_then(|track| catalog.file(track.file_id))
855        .and_then(|file| file.tags.get(tag))
856        .cloned()
857        .unwrap_or_default()
858}
859
860fn tags_of_artist(catalog: &Catalog, entity: &EntityRef, tag: &str) -> Vec<String> {
861    let Some(id) = entity.resolve(catalog) else {
862        return Vec::new();
863    };
864    // **Every** value, not the first: a genre tag holds a list, whether the
865    // file writes it as several values or as one string with commas in it.
866    // Reading only the first turned "Rock, Pop" into "Rock" on some files and
867    // reported a disagreement with a genre the tags do carry.
868    catalog
869        .tracks_of_artist(id)
870        .into_iter()
871        .filter_map(|t| catalog.track(t))
872        .find_map(|t| {
873            catalog
874                .file(t.file_id)
875                .map(|f| f.tags.get(tag).cloned().unwrap_or_default())
876                .filter(|values: &Vec<String>| !values.is_empty())
877        })
878        .unwrap_or_default()
879}

881/// Each field a record carries, with the tag it can be judged against.
882///
883/// `None` where the catalog holds no counterpart at all: an artist's country
884/// is not something the tags have an opinion about, and printing "nothing in
885/// your tags" against it would suggest the tags were expected to.
886fn compared(
887    catalog: &Catalog,
888    entity: &EntityRef,
889    facts: &Facts,
890) -> Vec<(&'static str, String, Option<Verdict>)> {
891    let mut rows: Vec<(&'static str, String, Option<Verdict>)> = Vec::new();
892    match facts {
893        Facts::Track(t) => {
894            // Compared with the tags, which is the whole use of this: a file
895            // whose audio and whose tags disagree is what fingerprinting is
896            // for, and the panel is where the two sit side by side.
897            if !t.artists.is_empty() {
898                rows.push((
899                    "heard artist",
900                    t.artists.join(", "),
901                    Some(sources::verdict_set(
902                        &t.artists,
903                        &tags_of_track(catalog, entity, "artist"),
904                    )),
905                ));
906            }
907            if let Some(title) = &t.title {
908                rows.push((
909                    "heard title",
910                    title.clone(),
911                    Some(sources::verdict(
912                        title,
913                        tags_of_track(catalog, entity, "title")
914                            .first()
915                            .map(String::as_str),
916                    )),
917                ));
918            }
919            for (field, value) in [("heard on", &t.album), ("recording", &t.recording)] {
920                if let Some(value) = value {
921                    rows.push((field, value.clone(), None));
922                }
923            }
924            for work in &t.works {
925                rows.push(("work", format!("{} ({})", work.title, work.mbid), None));
926                for credit in &work.credits {
927                    rows.push(("work credit", credit_summary(credit), None));
928                }
929            }
930            for credit in &t.credits {
931                rows.push(("recording credit", credit_summary(credit), None));
932            }
933            if let Some(score) = t.score {
934                // Shown, never hidden behind a threshold: a reader deciding
935                // whether to believe this needs the number the service gave,
936                // and a program that quietly dropped the weak matches would
937                // be making that decision for them.
938                rows.push((("how sure"), format!("{score}%"), None));
939            }
940        }
941        Facts::Artist(a) => {
942            // "country" was wrong: MusicBrainz calls this an *area*, and an
943            // area is a country, a city or a region — "Seattle" is a valid
944            // answer. "from" is true of all three.
945            //
946            // "known as" was wrong too, and worse: `disambiguation` is written
947            // to separate two artists who share a name, not to describe one,
948            // which is how a perfectly correct answer came out as
949            // "known as: the band".
950            for (field, value) in [
951                ("type", &a.kind),
952                ("from", &a.area),
953                ("country code", &a.country_code),
954                ("formed", &a.began),
955                ("ended", &a.ended),
956                ("note", &a.disambiguation),
957                ("wikidata", &a.wikidata),
958                ("discogs", &a.discogs),
959                ("homepage", &a.homepage),
960            ] {
961                if let Some(value) = value {
962                    rows.push((field, value.clone(), None));
963                }
964            }
965            if !a.genres.is_empty() {
966                // Beside the genre tag, never over it: the last column is what
967                // your files say, and these two answers are both allowed to
968                // exist.
969                //
970                // Compared as **sets**, which the first version did not do: it
971                // took MusicBrainz's top genre and compared it, as a string, to
972                // the whole tag. `pop` against `Rock, Pop` came out as a
973                // disagreement, when the tag plainly says pop. Genres are not
974                // exclusive and the two sides are not answering at the same
975                // granularity — one is what a crowd voted, the other is what
976                // one person typed — so a shared name is agreement.
977                let theirs = a.genres.join(", ");
978                let yours = tags_of_artist(catalog, entity, "genre");
979                let verdict = aede_core::sources::verdict_set(&a.genres, &yours);
980                rows.push(("genres", theirs, Some(verdict)));
981            }
982            if !a.aliases.is_empty() {
983                rows.push(("also called", a.aliases.join(", "), None));
984            }
985            if !a.members.is_empty() {
986                rows.push((
987                    "memberships",
988                    ui::plural(a.members.len(), "membership"),
989                    None,
990                ));
991            }
992            if !a.discography.is_empty() {
993                rows.push((
994                    "discography",
995                    ui::plural(a.discography.len(), "release"),
996                    None,
997                ));
998            }
999            // Shown even when there is no end date, because that is the case
1000            // it answers: a band with no end may have never stopped, or may
1001            // be one nobody filled in, and the reader cannot tell otherwise.
1002            if let Some(active) = a.active {
1003                rows.push((
1004                    "status",
1005                    match active {
1006                        true => "still active".to_string(),
1007                        false => "ended".to_string(),
1008                    },
1009                    None,
1010                ));
1011            }
1012            // No verdict to compare it with — a tag has no field for a
1013            // picture — so this is shown, not judged, the same as the
1014            // discography and the members above it.
1015            if let Some(picture) = &a.portrait {
1016                rows.push(("portrait", picture.url.clone(), None));
1017            }
1018            if let Some(picture) = &a.logo {
1019                rows.push(("logo", picture.url.clone(), None));
1020            }
1021        }
1022        Facts::Release(r) => {
1023            let release = entity
1024                .resolve(catalog)
1025                .and_then(|id| catalog.release(id))
1026                .cloned();
1027            let tag = |name: &str| -> Option<String> {
1028                release.as_ref().and_then(|rel| {
1029                    rel.track_ids
1030                        .first()
1031                        .and_then(|&t| catalog.track(t))
1032                        .and_then(|t| catalog.file(t.file_id))
1033                        .and_then(|f| f.first_tag(name).map(str::to_string))
1034                })
1035            };
1036
1037            if let Some(primary) = &r.primary_type {
1038                let mut shown = primary.clone();
1039                if !r.secondary_types.is_empty() {
1040                    shown = format!("{shown} · {}", r.secondary_types.join(" · "));
1041                }
1042                rows.push((
1043                    "release type",
1044                    shown,
1045                    Some(aede_core::sources::verdict(
1046                        primary,
1047                        tag("releasetype").as_deref(),
1048                    )),
1049                ));
1050            } else if !r.secondary_types.is_empty() {
1051                rows.push(("release type", r.secondary_types.join(" · "), None));
1052            }
1053            if let Some(date) = &r.first_released {
1054                let yours = release.as_ref().and_then(|rel| rel.date.clone());
1055                rows.push((
1056                    "first released",
1057                    date.clone(),
1058                    Some(aede_core::sources::verdict_date(date, yours.as_deref())),
1059                ));
1060            }
1061            if let Some(label) = &r.label {
1062                let yours = release.as_ref().and_then(|rel| {
1063                    rel.label_ids
1064                        .first()
1065                        .and_then(|&id| catalog.label(id))
1066                        .map(|l| l.name.clone())
1067                });
1068                rows.push((
1069                    "label",
1070                    label.clone(),
1071                    Some(aede_core::sources::verdict(label, yours.as_deref())),
1072                ));
1073            }
1074            for (field, value) in [
1075                ("label ID", &r.label_mbid),
1076                ("edition", &r.edition_mbid),
1077                ("cover", &r.cover_art),
1078            ] {
1079                if let Some(value) = value {
1080                    rows.push((field, value.clone(), None));
1081                }
1082            }
1083            for credit in &r.credits {
1084                rows.push(("edition credit", credit_summary(credit), None));
1085            }
1086        }
1087        // Label links and artwork have no audio-tag counterpart to judge.
1088        Facts::Label(label) => {
1089            if let Some(url) = &label.wikidata {
1090                rows.push(("wikidata", url.clone(), None));
1091            }
1092            if let Some(url) = &label.discogs {
1093                rows.push(("discogs", url.clone(), None));
1094            }
1095            if let Some(picture) = &label.logo {
1096                rows.push(("logo", picture.url.clone(), None));
1097            }
1098        }
1099    }
1100    rows
1101}

1103fn credit_summary(credit: &sources::CreditLink) -> String {
1104    let credited = credit
1105        .credited_as
1106        .as_deref()
1107        .filter(|name| *name != credit.artist_name)
1108        .map(|name| format!(" as {name}"))
1109        .unwrap_or_default();
1110    let attributes = credit
1111        .attributes
1112        .iter()
1113        .map(|attribute| {
1114            attribute
1115                .credited_as
1116                .as_deref()
1117                .or(attribute.value.as_deref())
1118                .unwrap_or(&attribute.name)
1119                .to_string()
1120        })
1121        .collect::<Vec<_>>();
1122    let details = match attributes.is_empty() {
1123        true => String::new(),
1124        false => format!(" ({})", attributes.join(", ")),
1125    };
1126    format!(
1127        "{} — {}{credited}{details}",
1128        credit.role, credit.artist_name
1129    )
1130}
