1//! The `sources` command: what somebody else said about this library.
2//!
3//! The catalog holds what the files say, the annotation file holds what you
4//! say, and `sources.json` holds what a source says — MusicBrainz first, a
5//! plugin later. This is the command that shows it, takes it in, and drops it.
6//!
7//! It exists before anything fetches, on purpose. A store that can only be
8//! filled over the network can only be tested over the network, and the layer
9//! is worth more than that: `--import` reads a document in exactly the shape
10//! the file has, so the whole of it — attachment, agreement, removal — can be
11//! exercised offline, and so can a bug report.
12//!
13//! The shape follows `aede import`, which does the same job for another tool's
14//! analyses: `--list` says what is held, `--forget` removes it, `--source`
15//! narrows to one. Two stores of what somebody else said should not be learned
16//! twice.
17

26pub fn sources(args: &Args) -> Res {
27    let operations: Vec<&str> = ["import", "template", "export", "forget", "list"]
28        .into_iter()
29        .filter(|option| args.has(option))
30        .collect();
31    if operations.len() > 1 {
32        return Err(
33            "choose one sources operation: --import, --template, --export, --forget, or --list"
34                .into(),
35        );
36    }
37    if args.has("source") {
38        if !args.has("list") && !args.has("forget") && !args.has("template") {
39            return Err("--source is used with --list, --forget, or --template".into());
40        }
41        if args
42            .value("source")
43            .is_none_or(|value| value.trim().is_empty())
44        {
45            return Err("--source expects a non-empty source name".into());
46        }
47    }
48    if args.has("output") && !args.has("export") && !args.has("template") {
49        return Err("--output is used with --export or --template".into());
50    }
51    if !args.positionals.is_empty() && !args.has("template") {
52        return Err("names select entities only with --template on sources".into());
53    }
54    let dir = data_dir(args);
55    let path = sources::sources_path(&dir);
56
57    if args.has("import") {
58        return import(args, &path);
59    }
60    if args.has("template") {
61        return template(args);
62    }
63    if args.has("export") {
64        return export(args, &path);
65    }
66    if args.has("forget") {
67        return forget(args, &path);
68    }
69
70    let held = sources::load(&path)?.unwrap_or_default();
71    if held.records.is_empty() {
72        println!("{}", ui::section("Sources"));
73        println!("  {}", ui::dim("nothing has been fetched yet"));
74        // The store being empty is not the same as the feature being absent,
75        // and a reader who cannot tell which will conclude the wrong one.
76        println!(
77            "  {}",
78            ui::dim("aede sources --import <file> takes in a document by hand")
79        );
80        return Ok(());
81    }
82
83    let catalog = load(args)?;
84    match args.has("list") {
85        true => list(&held, &catalog, args),
86        false => summary(&held, &catalog),
87    }
88}

90/// One line per source: how much it said, and how much of it lands.
91fn summary(held: &Sources, catalog: &Catalog) -> Res {
92    println!("{}", ui::section("Sources"));
93
94    let mut names: Vec<&str> = held.records.iter().map(|r| r.source.as_str()).collect();
95    names.sort_unstable();
96    names.dedup();
97
98    let mut table = Table::new(&["Source", "Records", "Attached", "Waiting", "Last fetch"])
99        .align(1, Align::Right)
100        .align(2, Align::Right)
101        .align(3, Align::Right);
102
103    for name in &names {
104        let mine = Sources {
105            records: held
106                .records
107                .iter()
108                .filter(|r| r.source == *name)
109                .cloned()
110                .collect(),
111            reviews: held
112                .reviews
113                .iter()
114                .filter(|review| review.source == *name)
115                .cloned()
116                .collect(),
117            credit_exclusions: held
118                .credit_exclusions
119                .iter()
120                .filter(|decision| decision.relation.provenance == *name)
121                .cloned()
122                .collect(),
123        };
124        let reach = sources::attachment(&mine, catalog);
125        let last = mine.records.iter().map(|r| r.fetched_at).max().unwrap_or(0);
126        table.push(vec![
127            (*name).to_string(),
128            mine.records.len().to_string(),
129            reach.attached.to_string(),
130            reach.waiting.to_string(),
131            ui::since(last),
132        ]);
133    }
134    println!("{}", table.render());
135
136    // A count of what is waiting means nothing without the reason, and the
137    // reason is always the same one: the catalog does not hold that entity
138    // yet. Saying so here is what stops it reading as a failure.
139    let all = sources::attachment(held, catalog);
140    if all.waiting > 0 {
141        // Both verbs agree with the count, because `ui::plural` puts the count
142        // in the sentence and the rest of the sentence has to follow it.
143        let (name, kept) = match all.waiting {
144            1 => ("names", "is"),
145            _ => ("name", "are"),
146        };
147        println!(
148            "  {}",
149            ui::dim(&format!(
150                "{} {name} nothing this catalog holds yet, and {kept} kept until it does",
151                ui::plural(all.waiting, "record")
152            ))
153        );
154    }
155    Ok(())
156}

158/// Every record, with what it says and whether the catalog can place it.
159fn list(held: &Sources, catalog: &Catalog, args: &Args) -> Res {
160    let only = args.value("source");
161    let rows: Vec<&SourceRecord> = held
162        .records
163        .iter()
164        .filter(|r| only.is_none_or(|s| r.source == s))
165        .collect();
166
167    if rows.is_empty() {
168        println!("{}", ui::section("Sources"));
169        println!(
170            "  {}",
171            ui::dim(&ui::literal(&match only {
172                Some(name) => format!("nothing was fetched from \"{name}\""),
173                None => "nothing has been fetched yet".to_string(),
174            }))
175        );
176        return Ok(());
177    }
178
179    println!("{}", ui::section("What sources say"));
180    let mut table = Table::new(&["Entity", "Source", "Confidence", "Says", "Where"])
181        .limit(0, 42)
182        .limit(3, 46);
183    for record in &rows {
184        let entity = record.entity();
185        table.push(vec![
186            format!("{} {}", entity.kind.as_str(), entity.key),
187            record.source.clone(),
188            super::source_status(
189                record.confidence,
190                held.review_for(record).map(|review| review.decision),
191                held.is_trusted(catalog, record),
192            ),
193            says(&record.facts),
194            match entity.resolve(catalog).is_some() {
195                true => "in the catalog".to_string(),
196                false => "waiting".to_string(),
197            },
198        ]);
199    }
200    println!("{}", table.render());
201    println!("  {}", ui::dim(&ui::plural(rows.len(), "record")));
202    Ok(())
203}

205/// What a record says, short enough for a column.
206fn says(facts: &Facts) -> String {
207    let mut parts: Vec<String> = Vec::new();
208    match facts {
209        Facts::Track(t) => {
210            // What it heard, in the order somebody reads it: who, then what.
211            // The score leads nothing — it qualifies the rest, and a line
212            // opening with a number reads as a measurement of the file.
213            if !t.artists.is_empty() {
214                parts.push(t.artists.join(", "));
215            }
216            if let Some(title) = &t.title {
217                parts.push(title.clone());
218            }
219            if let Some(album) = &t.album {
220                parts.push(album.clone());
221            }
222            if !t.works.is_empty() {
223                parts.push(format!("{} work link(s)", t.works.len()));
224            }
225            if let Some(score) = t.score {
226                parts.push(format!("{score}% sure"));
227            }
228        }
229        Facts::Artist(a) => {
230            if let Some(kind) = &a.kind {
231                parts.push(kind.clone());
232            }
233            if let Some(known) = &a.disambiguation {
234                parts.push(known.clone());
235            }
236            if let Some(area) = &a.area {
237                parts.push(area.clone());
238            }
239            match (&a.began, &a.ended) {
240                (Some(from), Some(to)) => parts.push(format!("{from}–{to}")),
241                (Some(from), None) => parts.push(format!("{from}–")),
242                _ => {}
243            }
244            if let Some(active) = a.active {
245                parts.push(match active {
246                    true => "still active".to_string(),
247                    false => "ended".to_string(),
248                });
249            }
250            if a.portrait.is_some() {
251                parts.push("has a portrait".to_string());
252            }
253            // Last, and left to the column's own limit. A truncated opening
254            // line says more about what was stored than the word "summary"
255            // would, and the whole of it is on the artist's own card, where it
256            // is wrapped and credited.
257            if let Some(prose) = &a.summary {
258                parts.push(prose.text.clone());
259            }
260        }
261        Facts::Release(r) => {
262            if let Some(primary) = &r.primary_type {
263                parts.push(primary.clone());
264            }
265            parts.extend(r.secondary_types.iter().cloned());
266            if let Some(date) = &r.first_released {
267                parts.push(date.clone());
268            }
269            if let Some(label) = &r.label {
270                parts.push(label.clone());
271            }
272        }
273        Facts::Label(LabelFacts {
274            logo,
275            summary,
276            wikidata,
277            discogs,
278            ..
279        }) => {
280            if let Some(url) = wikidata {
281                parts.push(url.clone());
282            }
283            if let Some(url) = discogs {
284                parts.push(url.clone());
285            }
286            if let Some(prose) = summary {
287                parts.push(prose.text.clone());
288            }
289            if let Some(logo) = logo {
290                parts.push(format!("logo: {}", logo.url));
291            }
292        }
293    }
294    match parts.is_empty() {
295        // An answer holding nothing is a real answer, and printing an empty
296        // cell would make it look like a missing row instead.
297        true => ui::dim("nothing"),
298        false => parts.join(" · "),
299    }
300}

302/// Takes in a document written in the shape of `sources.json`.
303fn import(args: &Args, path: &std::path::Path) -> Res {
304    let Some(file) = args.value("import") else {
305        return Err("--import expects a file: --import=<file.json>".into());
306    };
307    let text = std::fs::read_to_string(file).map_err(|e| format!("cannot read \"{file}\": {e}"))?;
308    let value =
309        aede_core::json::parse(&text).map_err(|e| format!("\"{file}\" is not JSON: {e}"))?;
310    let incoming = sources::from_json(&value)?;
311
312    let mut held = sources::load_all(path)?.unwrap_or_default();
313    let (mut updated, mut added) = (0usize, 0usize);
314    for record in incoming.records {
315        match held.set(record) {
316            true => updated += 1,
317            false => added += 1,
318        }
319    }
320    for review in incoming.reviews {
321        held.set_review(review);
322    }
323    for decision in incoming.credit_exclusions {
324        held.exclude_credit(decision.relation, decision.excluded_at);
325    }
326    sources::save(&held, path)?;
327
328    println!("{}", ui::section("Imported"));
329    println!(
330        "{} {} added, {} updated, {} held",
331        ui::green("→"),
332        added,
333        updated,
334        held.records.len()
335    );
336    // Naming the file it wrote, for the same reason `import --forget` does: a
337    // command that changed a store the user cannot see should say which one.
338    println!("  {}", ui::dim(&ui::literal(&path.display().to_string())));
339    Ok(())
340}

342/// Writes the layer out, in the shape `--import` reads.
343///
344/// The symmetry matters more than the feature: a store that can be filled and
345/// not emptied out again is a store whose contents you have to trust somebody
346/// else about. `aede notes --export` already draws the same round trip for
347/// what the user writes.
348fn export(args: &Args, path: &std::path::Path) -> Res {
349    let held = sources::load(path)?.unwrap_or_default();
350    if held.records.is_empty() && held.reviews.is_empty() && held.credit_exclusions.is_empty() {
351        // Writing an empty document and reporting success would look like the
352        // export worked and the layer was empty — two different things, and
353        // the second is not what has just been established.
354        return Err(
355            "nothing has been fetched yet, so there is nothing to export.\n\
356             aede sources --template=<file> writes a document you can fill in"
357                .into(),
358        );
359    }
360    super::export::emit(args, &sources::to_json(&held).to_string_pretty())
361}

363/// Writes a document with the right keys and nothing filled in.
364///
365/// The keys are the one thing a person cannot guess: they are how an entity
366/// names itself — a normalised artist name, or an album's artist, title and
367/// folder joined — and until something fetches, there is no way to see one.
368/// Without this, the layer could be read and emptied but never filled by hand,
369/// which makes it untestable by the person who most needs to test it.
370///
371/// Naming entities narrows it:
372/// `aede sources --template --output=x.json "Kind of Blue"` writes that album
373/// alone rather than a skeleton of the whole library.
374fn template(args: &Args) -> Res {
375    let catalog = load(args)?;
376    let source = args.value("source").unwrap_or(sources::MUSICBRAINZ);
377    let now = aede_core::clock::now_seconds();
378
379    let wanted: Vec<String> = args
380        .positionals
381        .iter()
382        .map(|name| aede_core::text::normalize(name))
383        .filter(|name| !name.is_empty())
384        .collect();
385    let matches = |name: &str| -> bool {
386        let key = aede_core::text::normalize(name);
387        wanted.is_empty() || wanted.iter().any(|w| key.contains(w.as_str()))
388    };
389
390    let mut out = Sources::default();
391    let blank = |key: String, facts: Facts| SourceRecord {
392        key,
393        source: source.to_string(),
394        source_id: None,
395        fetched_at: now,
396        confidence: Confidence::Identified,
397        facts,
398    };
399    for artist in &catalog.artists {
400        if !matches(&artist.name) {
401            continue;
402        }
403        if let Some(entity) = EntityRef::of(&catalog, EntityKind::Artist, artist.id) {
404            out.set(blank(entity.key, Facts::Artist(Default::default())));
405        }
406    }
407    for release in &catalog.releases {
408        if !matches(&release.title) {
409            continue;
410        }
411        if let Some(entity) = EntityRef::of(&catalog, EntityKind::Release, release.id) {
412            out.set(blank(entity.key, Facts::Release(Default::default())));
413        }
414    }
415
416    if out.records.is_empty() {
417        return Err(match wanted.is_empty() {
418            true => "this catalog holds nothing to write a template for".to_string(),
419            false => format!(
420                "nothing in this catalog matches \"{}\"",
421                args.positionals.join(" ")
422            ),
423        }
424        .into());
425    }
426
427    super::export::emit(args, &sources::to_json(&out).to_string_pretty())?;
428    // Only when it went to a file: a template printed to the terminal is on
429    // screen, and telling somebody to import what they can see is noise.
430    if args.value("output").is_some() {
431        println!(
432            "  {} {}",
433            ui::plural(out.records.len(), "empty record"),
434            ui::dim("— fill in what you want, then: aede sources --import=<file>")
435        );
436    }
437    Ok(())
438}

440/// Drops what one source said, or everything.
441fn forget(args: &Args, path: &std::path::Path) -> Res {
442    let mut held = sources::load_all(path)?.unwrap_or_default();
443    let before = held.records.len();
444    let decisions_before = held.reviews.len() + held.credit_exclusions.len();
445    if before == 0 && decisions_before == 0 {
446        println!("{}", ui::section("Sources"));
447        println!("  {}", ui::dim("there was nothing to forget"));
448        return Ok(());
449    }
450
451    let removed = match args.value("source") {
452        Some(name) => {
453            let gone = held.forget(name);
454            if gone == 0 && decisions_before == held.reviews.len() + held.credit_exclusions.len() {
455                return Err(format!(
456                    "no source named \"{name}\" — aede sources says which there are"
457                )
458                .into());
459            }
460            gone
461        }
462        None => {
463            let removed = std::mem::take(&mut held.records).len();
464            held.reviews.clear();
465            held.credit_exclusions.clear();
466            removed
467        }
468    };
469
470    sources::save(&held, path)?;
471    println!(
472        "{} {} removed, {} left",
473        ui::green("→"),
474        ui::plural(removed, "record"),
475        held.records.len()
476    );
477    let decisions_removed = decisions_before - held.reviews.len() - held.credit_exclusions.len();
478    if decisions_removed > 0 {
479        println!("  {} removed", ui::plural(decisions_removed, "decision"));
480    }
481    println!("  {}", ui::dim(&ui::literal(&path.display().to_string())));
482    Ok(())
483}

489/// The block an entity page prints when a source has said something about it.
490///
491/// One function called from every page rather than one block written per page:
492/// what a source says has to read the same way on an album and on an artist,
493/// and a rule copied into four pages is a rule that will be right in three of
494/// them.
495///
496/// Silent when nothing was fetched. A page that announced an empty section for
497/// a feature nobody has used yet would be noise on every album in the library.
498pub fn panel_for(args: &Args, catalog: &Catalog, kind: EntityKind, id: Id) {
499    let Ok(held) = super::sources_held(args) else {
500        return;
501    };
502    panel_for_held(catalog, kind, id, &held, false);
503}

505/// The label page may deliberately show its retained Discogs profile when
506/// offline or when a refresh failed. Other pages use the filtered loader.
507pub fn panel_for_held(
508    catalog: &Catalog,
509    kind: EntityKind,
510    id: Id,
511    held: &Sources,
512    refresh_failed: bool,
513) {
514    let Some(entity) = EntityRef::of(catalog, kind, id) else {
515        return;
516    };
517    let records: Vec<&SourceRecord> = held.about(&entity).collect();
518    if records.is_empty() {
519        return;
520    }
521
522    println!("{}", ui::section("What sources say"));
523
524    // Prose first, and outside the table. A paragraph does not fit a column,
525    // and it is the one thing here meant to be read rather than compared with
526    // a tag — there is nothing in a file to compare it against.
527    for record in &records {
528        let prose = match &record.facts {
529            Facts::Artist(artist) => artist.summary.as_ref(),
530            Facts::Label(label) => label.summary.as_ref(),
531            _ => None,
532        };
533        if let Some(prose) = prose {
534            if record.source == aede_core::discogs::SOURCE
535                && (refresh_failed
536                    || !aede_core::discogs::fresh(
537                        record.fetched_at,
538                        aede_core::clock::now_seconds(),
539                    ))
540            {
541                println!(
542                    "  {}",
543                    ui::yellow(
544                        "Stored Discogs profile: update needed; the latest version could not be confirmed."
545                    )
546                );
547            }
548            let display = if record.source == aede_core::discogs::SOURCE {
549                aede_core::discogs::render_profile(
550                    &prose.text,
551                    &Default::default(),
552                    &Default::default(),
553                )
554            } else {
555                prose.text.clone()
556            };
557            let lines = if record.source == aede_core::discogs::SOURCE {
558                discogs_profile_lines(&display, 72)
559            } else {
560                ui::wrap(&ui::literal(&display), 72)
561            };
562            for line in lines {
563                if line.is_empty() {
564                    println!();
565                } else {
566                    println!("  {line}");
567                }
568            }
569            if record.source == aede_core::discogs::SOURCE {
570                println!(
571                    "  {}",
572                    ui::dim(
573                        "Discogs profile: original text (usually English; no French translation supplied)"
574                    )
575                );
576                println!(
577                    "  {}",
578                    ui::dim(&ui::literal(&format!(
579                        "Data provided by Discogs. {} — CC0",
580                        prose.url
581                    )))
582                );
583                println!(
584                    "  {}",
585                    ui::dim(
586                        "This application uses Discogs’ API but is not affiliated with, sponsored or endorsed by Discogs. ‘Discogs’ is a trademark of Zink Media, LLC."
587                    )
588                );
589                println!(
590                    "  {}",
591                    ui::dim(&ui::literal(&format!(
592                        "aede label \"{}\" checks this profile whenever the page opens",
593                        record.key
594                    )))
595                );
596            } else {
597                println!("  {}", ui::dim(&ui::literal(&prose.credit())));
598                println!(
599                    "  {}",
600                    ui::dim(&ui::literal(&format!(
601                        "aede fetch --summaries --full --lang=<code> \"{}\" asks for another",
602                        record.key
603                    )))
604                );
605            }
606            println!();
607        }
608    }
609
610    let mut table = Table::new(&["Source", "Field", "Says", "Your tags"]).limit(2, 34);
611
612    for record in &records {
613        let age = ui::since(record.fetched_at);
614        let attribution = format!(
615            "{} · {} · {age}",
616            record.source,
617            super::source_status(
618                record.confidence,
619                held.review_for(record).map(|review| review.decision),
620                held.is_trusted(catalog, record),
621            )
622        );
623        for (field, theirs, verdict) in compared(catalog, &entity, &record.facts) {
624            table.push(vec![
625                attribution.clone(),
626                field.to_string(),
627                theirs,
628                // The verdict is the point of the column: a value that matches
629                // the tags is not the same news as one that contradicts them,
630                // and neither is one the tags say nothing about.
631                match verdict {
632                    Some(Verdict::Agrees) => ui::green("matches your tags").to_string(),
633                    Some(Verdict::Differs { yours, .. }) => ui::yellow(&yours).to_string(),
634                    // The tags could have answered and did not.
635                    Some(Verdict::NothingToCompare) => ui::dim("nothing in your tags").to_string(),
636                    // There is no tag for this at all — an artist's country is
637                    // not something the files have an opinion about. Saying
638                    // "nothing in your tags" would suggest they were expected
639                    // to, and send the reader looking for a field to fill.
640                    None => String::new(),
641                },
642            ]);
643        }
644    }
645    // A source that was asked and holds nothing is not an empty table. The
646    // distinction is the one this whole layer exists to keep — "checked, and
647    // it knows nothing" against "never checked" — and a table rendering
648    // "(no results)" says neither of them.
649    if table.is_empty() {
650        for record in &records {
651            // A record whose only content is the paragraph just printed has
652            // no row here, and saying it "holds nothing" directly under its
653            // own text would be plainly false.
654            if matches!(&record.facts, Facts::Artist(a) if a.summary.is_some())
655                || matches!(&record.facts, Facts::Label(l) if l.summary.is_some())
656            {
657                continue;
658            }
659            println!(
660                "  {}",
661                ui::dim(&ui::literal(&format!(
662                    "{} was asked and holds nothing about this ({})",
663                    record.source,
664                    ui::since(record.fetched_at)
665                )))
666            );
667        }
668        whence(&records);
669        return;
670    }
671
672    print!("{}", table.render());
673    whence(&records);
674    println!(
675        "{}",
676        ui::dim("  beside your tags, never on top of them — aede sources --forget removes them")
677    );
678}

680/// Discogs separates identity notes from the biography with hard line breaks.
681/// Wrap each source line independently and apply emphasis after measuring it,
682/// so terminal escape sequences do not count toward the available width.
683fn discogs_profile_lines(text: &str, width: usize) -> Vec<String> {
684    text.lines()
685        .flat_map(|line| {
686            if line.trim().is_empty() {
687                return vec![String::new()];
688            }
689            let label_code = line
690                .trim_start()
691                .to_ascii_lowercase()
692                .starts_with("label code:");
693            ui::wrap(&ui::literal(line), width)
694                .into_iter()
695                .map(|line| if label_code { ui::bold(&line) } else { line })
696                .collect()
697        })
698        .collect()
699}

701/// Where each of these answers came from, as an address a reader can follow.
702///
703/// **The identifier was stored and never shown**, which is the one thing a
704/// reader needs to check anything by hand: to open the page, to ask the service
705/// the same question, or to correct the data at its source. Asked for the
706/// MusicBrainz identifier of an artist, there was nowhere in this program to
707/// find it — and the nearest thing on screen was the Wikidata link, which is a
708/// different identifier that looks enough like an answer to waste an afternoon.
709///
710/// Printed as the URL rather than the bare identifier: the identifier is inside
711/// it, so it can still be copied for a query, and the page it opens is where a
712/// wrong type or a missing date is actually fixed — for everybody, not just
713/// here.
714fn whence(records: &[&SourceRecord]) {
715    for record in records {
716        if let Some(address) = address_of(record) {
717            println!(
718                "  {}",
719                ui::dim(&ui::literal(&format!("{}: {address}", record.source)))
720            );
721        }
722    }
723}

725/// Where one record's answer can be looked at, or its bare identifier.
726///
727/// Split from the printing so the decision is testable: which path a
728/// MusicBrainz album takes is the sort of thing that is wrong once and then
729/// wrong for a year, because a link that 404s looks like the service's fault.
730fn address_of(record: &SourceRecord) -> Option<String> {
731    let id = record.source_id.as_deref()?;
732    let kind = match record.facts.kind() {
733        EntityKind::Artist => "artist",
734        // The identifier kept for an album is the release group's, not the
735        // edition's — see `musicbrainz::release`. Naming the wrong path here
736        // would produce a link that 404s on the one page a reader came to open.
737        EntityKind::Release => "release-group",
738        EntityKind::Label => "label",
739        // Other entity kinds have no known MusicBrainz URL here.
740        _ => return Some(id.to_string()),
741    };
742    Some(match record.source.as_str() {
743        sources::MUSICBRAINZ => format!("https://musicbrainz.org/{kind}/{id}"),
744        aede_core::discogs::SOURCE if kind == "label" => {
745            format!("https://www.discogs.com/label/{id}")
746        }
747        "wikipedia" => format!("https://www.wikidata.org/wiki/{id}"),
748        // A source this build knows nothing about still has an identifier, and
749        // the identifier is the useful half. Inventing an address for it would
750        // be worse than showing none.
751        _ => id.to_string(),
752    })
753}

755/// The first value of a tag on any file credited to this artist.
756///
757/// A genre lives on the tracks, not on the artist, so "what do my files call
758/// this artist" is answered by the files they appear on.
759/// What one track's own file says under a tag.
760///
761/// Its own file, not its release's: a fingerprint answers for one recording,
762/// and comparing it with an album-level tag would put the wrong two things
763/// side by side.
764fn tags_of_track(catalog: &Catalog, entity: &EntityRef, tag: &str) -> Vec<String> {
765    let Some(id) = entity.resolve(catalog) else {
766        return Vec::new();
767    };
768    catalog
769        .track(id)
770        .and_then(|track| catalog.file(track.file_id))
771        .and_then(|file| file.tags.get(tag))
772        .cloned()
773        .unwrap_or_default()
774}
775
776fn tags_of_artist(catalog: &Catalog, entity: &EntityRef, tag: &str) -> Vec<String> {
777    let Some(id) = entity.resolve(catalog) else {
778        return Vec::new();
779    };
780    // **Every** value, not the first: a genre tag holds a list, whether the
781    // file writes it as several values or as one string with commas in it.
782    // Reading only the first turned "Rock, Pop" into "Rock" on some files and
783    // reported a disagreement with a genre the tags do carry.
784    catalog
785        .tracks_of_artist(id)
786        .into_iter()
787        .filter_map(|t| catalog.track(t))
788        .find_map(|t| {
789            catalog
790                .file(t.file_id)
791                .map(|f| f.tags.get(tag).cloned().unwrap_or_default())
792                .filter(|values: &Vec<String>| !values.is_empty())
793        })
794        .unwrap_or_default()
795}

797/// Each field a record carries, with the tag it can be judged against.
798///
799/// `None` where the catalog holds no counterpart at all: an artist's country
800/// is not something the tags have an opinion about, and printing "nothing in
801/// your tags" against it would suggest the tags were expected to.
802fn compared(
803    catalog: &Catalog,
804    entity: &EntityRef,
805    facts: &Facts,
806) -> Vec<(&'static str, String, Option<Verdict>)> {
807    let mut rows: Vec<(&'static str, String, Option<Verdict>)> = Vec::new();
808    match facts {
809        Facts::Track(t) => {
810            // Compared with the tags, which is the whole use of this: a file
811            // whose audio and whose tags disagree is what fingerprinting is
812            // for, and the panel is where the two sit side by side.
813            if !t.artists.is_empty() {
814                rows.push((
815                    "heard artist",
816                    t.artists.join(", "),
817                    Some(sources::verdict_set(
818                        &t.artists,
819                        &tags_of_track(catalog, entity, "artist"),
820                    )),
821                ));
822            }
823            if let Some(title) = &t.title {
824                rows.push((
825                    "heard title",
826                    title.clone(),
827                    Some(sources::verdict(
828                        title,
829                        tags_of_track(catalog, entity, "title")
830                            .first()
831                            .map(String::as_str),
832                    )),
833                ));
834            }
835            for (field, value) in [("heard on", &t.album), ("recording", &t.recording)] {
836                if let Some(value) = value {
837                    rows.push((field, value.clone(), None));
838                }
839            }
840            for work in &t.works {
841                rows.push(("work", format!("{} ({})", work.title, work.mbid), None));
842                for credit in &work.credits {
843                    rows.push(("work credit", credit_summary(credit), None));
844                }
845            }
846            for credit in &t.credits {
847                rows.push(("recording credit", credit_summary(credit), None));
848            }
849            if let Some(score) = t.score {
850                // Shown, never hidden behind a threshold: a reader deciding
851                // whether to believe this needs the number the service gave,
852                // and a program that quietly dropped the weak matches would
853                // be making that decision for them.
854                rows.push((("how sure"), format!("{score}%"), None));
855            }
856        }
857        Facts::Artist(a) => {
858            // "country" was wrong: MusicBrainz calls this an *area*, and an
859            // area is a country, a city or a region — "Seattle" is a valid
860            // answer. "from" is true of all three.
861            //
862            // "known as" was wrong too, and worse: `disambiguation` is written
863            // to separate two artists who share a name, not to describe one,
864            // which is how a perfectly correct answer came out as
865            // "known as: the band".
866            for (field, value) in [
867                ("type", &a.kind),
868                ("from", &a.area),
869                ("formed", &a.began),
870                ("ended", &a.ended),
871                ("note", &a.disambiguation),
872                ("wikidata", &a.wikidata),
873                ("discogs", &a.discogs),
874                ("homepage", &a.homepage),
875            ] {
876                if let Some(value) = value {
877                    rows.push((field, value.clone(), None));
878                }
879            }
880            if !a.genres.is_empty() {
881                // Beside the genre tag, never over it: the last column is what
882                // your files say, and these two answers are both allowed to
883                // exist.
884                //
885                // Compared as **sets**, which the first version did not do: it
886                // took MusicBrainz's top genre and compared it, as a string, to
887                // the whole tag. `pop` against `Rock, Pop` came out as a
888                // disagreement, when the tag plainly says pop. Genres are not
889                // exclusive and the two sides are not answering at the same
890                // granularity — one is what a crowd voted, the other is what
891                // one person typed — so a shared name is agreement.
892                let theirs = a.genres.join(", ");
893                let yours = tags_of_artist(catalog, entity, "genre");
894                let verdict = aede_core::sources::verdict_set(&a.genres, &yours);
895                rows.push(("genres", theirs, Some(verdict)));
896            }
897            if !a.aliases.is_empty() {
898                rows.push(("also called", a.aliases.join(", "), None));
899            }
900            // Shown even when there is no end date, because that is the case
901            // it answers: a band with no end may have never stopped, or may
902            // be one nobody filled in, and the reader cannot tell otherwise.
903            if let Some(active) = a.active {
904                rows.push((
905                    "status",
906                    match active {
907                        true => "still active".to_string(),
908                        false => "ended".to_string(),
909                    },
910                    None,
911                ));
912            }
913            // No verdict to compare it with — a tag has no field for a
914            // picture — so this is shown, not judged, the same as the
915            // discography and the members above it.
916            if let Some(picture) = &a.portrait {
917                rows.push(("portrait", picture.url.clone(), None));
918            }
919        }
920        Facts::Release(r) => {
921            let release = entity
922                .resolve(catalog)
923                .and_then(|id| catalog.release(id))
924                .cloned();
925            let tag = |name: &str| -> Option<String> {
926                release.as_ref().and_then(|rel| {
927                    rel.track_ids
928                        .first()
929                        .and_then(|&t| catalog.track(t))
930                        .and_then(|t| catalog.file(t.file_id))
931                        .and_then(|f| f.first_tag(name).map(str::to_string))
932                })
933            };
934
935            if let Some(primary) = &r.primary_type {
936                let mut shown = primary.clone();
937                if !r.secondary_types.is_empty() {
938                    shown = format!("{shown} · {}", r.secondary_types.join(" · "));
939                }
940                rows.push((
941                    "release type",
942                    shown,
943                    Some(aede_core::sources::verdict(
944                        primary,
945                        tag("releasetype").as_deref(),
946                    )),
947                ));
948            }
949            if let Some(date) = &r.first_released {
950                let yours = release.as_ref().and_then(|rel| rel.date.clone());
951                rows.push((
952                    "first released",
953                    date.clone(),
954                    Some(aede_core::sources::verdict_date(date, yours.as_deref())),
955                ));
956            }
957            if let Some(label) = &r.label {
958                let yours = release.as_ref().and_then(|rel| {
959                    rel.label_ids
960                        .first()
961                        .and_then(|&id| catalog.label(id))
962                        .map(|l| l.name.clone())
963                });
964                rows.push((
965                    "label",
966                    label.clone(),
967                    Some(aede_core::sources::verdict(label, yours.as_deref())),
968                ));
969            }
970        }
971        // Nothing to show — see `LabelFacts`. There is no tag for a label's
972        // own identifier to be compared against, the same reason an
973        // artist's row above compares no name either.
974        Facts::Label(label) => {
975            if let Some(url) = &label.wikidata {
976                rows.push(("wikidata", url.clone(), None));
977            }
978            if let Some(url) = &label.discogs {
979                rows.push(("discogs", url.clone(), None));
980            }
981        }
982    }
983    rows
984}

986fn credit_summary(credit: &sources::CreditLink) -> String {
987    let credited = credit
988        .credited_as
989        .as_deref()
990        .filter(|name| *name != credit.artist_name)
991        .map(|name| format!(" as {name}"))
992        .unwrap_or_default();
993    let attributes = credit
994        .attributes
995        .iter()
996        .map(|attribute| {
997            attribute
998                .credited_as
999                .as_deref()
1000                .or(attribute.value.as_deref())
1001                .unwrap_or(&attribute.name)
1002                .to_string()
1003        })
1004        .collect::<Vec<_>>();
1005    let details = match attributes.is_empty() {
1006        true => String::new(),
1007        false => format!(" ({})", attributes.join(", ")),
1008    };
1009    format!(
1010        "{} — {}{credited}{details}",
1011        credit.role, credit.artist_name
1012    )
1013}
