1//! Starts the local API from the command line.
2
3use super::Res;
4use crate::args::Args;
5
6#[path = "server_jobs.rs"]
7mod server_jobs;
8
9const DEFAULT_PORT: u16 = 8787;
10
11pub fn serve(args: &Args) -> Res {
12    let raw_port = args.number_or("port", DEFAULT_PORT as usize)?;
13    let port = u16::try_from(raw_port).map_err(|_| "--port must be between 0 and 65535")?;
14    let admin_token = match std::env::var("AEDE_ADMIN_TOKEN") {
15        Ok(value) if value.len() >= 32 && value.is_ascii() => Some(value),
16        Ok(_) => return Err("AEDE_ADMIN_TOKEN must contain at least 32 ASCII characters".into()),
17        Err(std::env::VarError::NotPresent) => None,
18        Err(std::env::VarError::NotUnicode(_)) => {
19            return Err("AEDE_ADMIN_TOKEN must be valid ASCII".into());
20        }
21    };
22    let scan_args = args.clone();
23    let job_data = super::data_dir(args);
24    aede_server::serve(
25        &super::data_dir(args),
26        port,
27        admin_token.clone(),
28        move |progress| {
29            super::scan::rescan_with_progress(&scan_args, progress)
30                .map_err(|error| error.to_string())
31        },
32        crate::delegation::allowed,
33        move |request, cancellation| server_jobs::run(&job_data, request, cancellation),
34        |address| {
35            println!("Aède API: http://{address}/api/v1/status");
36            if admin_token.is_some() {
37                println!("Administrative jobs: POST http://{address}/api/admin/v1/scan or /fetch");
38                println!("Server route guide: crates/aede-server/README.md");
39            }
40        },
41    )
42}
