12const DEFAULT_PORT: u16 = 8787;
13
14fn server_options(args: &Args) -> Result<aede_server::ServerOptions, String> {
15    let raw_port = args.number_or("port", DEFAULT_PORT as usize)?;
16    let port = u16::try_from(raw_port).map_err(|_| "--port must be between 0 and 65535")?;
17    let bind = args
18        .value("bind")
19        .unwrap_or("127.0.0.1")
20        .parse::<IpAddr>()
21        .map_err(|_| "--bind expects one literal IPv4 or IPv6 address")?;
22    let bind = SocketAddr::new(bind, port);
23    let certificate = args.value("tls-cert");
24    let private_key = args.value("tls-key");
25    let authority = args.value("authority");
26    let tls = match (certificate, private_key, authority) {
27        (None, None, None) => {
28            if !bind.ip().is_loopback() {
29                return Err(
30                    "a non-loopback --bind requires --tls-cert, --tls-key and --authority".into(),
31                );
32            }
33            None
34        }
35        (Some(certificate), Some(private_key), Some(authority)) => {
36            if port == 0 {
37                return Err("--port=0 cannot be used with TLS; choose a fixed HTTPS port".into());
38            }
39            let certificate = path_option("tls-cert", certificate)?;
40            let private_key = path_option("tls-key", private_key)?;
41            validate_authority(authority)?;
42            Some(aede_server::TlsOptions {
43                certificate,
44                private_key,
45                authority: authority.into(),
46            })
47        }
48        _ => {
49            return Err("--tls-cert, --tls-key and --authority must be provided together".into());
50        }
51    };
52    Ok(aede_server::ServerOptions { bind, tls })
53}
54
55fn path_option(name: &str, value: &str) -> Result<PathBuf, String> {
56    if value.is_empty() {
57        return Err(format!("--{name} needs a nonempty path"));
58    }
59    Ok(PathBuf::from(value))
60}
61
62/// Add command-line context to the server's canonical Host/Origin rule.
63fn validate_authority(value: &str) -> Result<(), String> {
64    aede_server::TlsOptions::validate_authority(value).map_err(|failure| {
65        format!("--authority expects an exact HOST:PORT without a scheme or path: {failure}")
66    })
67}

69pub fn serve(args: &Args) -> Res {
70    let options = server_options(args)?;
71    let admin_token = match std::env::var("AEDE_ADMIN_TOKEN") {
72        Ok(value) if value.len() >= 32 && value.is_ascii() => Some(value),
73        Ok(_) => return Err("AEDE_ADMIN_TOKEN must contain at least 32 ASCII characters".into()),
74        Err(std::env::VarError::NotPresent) => None,
75        Err(std::env::VarError::NotUnicode(_)) => {
76            return Err("AEDE_ADMIN_TOKEN must be valid ASCII".into());
77        }
78    };
79    let scan_args = args.clone();
80    let data_dir = super::data_dir(args);
81    let job_data = data_dir.clone();
82    let tls_authority = options.tls.as_ref().map(|tls| tls.authority.clone());
83    let local_administration = tls_authority.is_none() && admin_token.is_some();
84    aede_server::serve_with_options(
85        &data_dir,
86        options,
87        admin_token,
88        move |progress| {
89            super::scan::rescan_with_progress(&scan_args, progress)
90                .map_err(|error| error.to_string())
91        },
92        crate::delegation::allowed,
93        move |request, cancellation| server_jobs::run(&job_data, request, cancellation),
94        |address| {
95            if let Some(authority) = tls_authority {
96                println!("Aède API: https://{authority}/api/v1/status");
97                println!("Server route guide: crates/aede-server/README.md");
98            } else {
99                println!("Aède API: http://{address}/api/v1/status");
100            }
101            if local_administration {
102                println!("Administrative jobs: POST http://{address}/api/admin/v1/scan or /fetch");
103                println!("Server route guide: crates/aede-server/README.md");
104            }
105        },
106    )
107}
