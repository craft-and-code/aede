20/// Rescans the watched folders on the user's behalf, after a command changed
21/// what "watched" means.
22///
23/// Same scan a bare `aede scan` runs, with one difference that is the whole
24/// reason this exists rather than a call to [`scan()`]: the folders named on the
25/// command line are **not** taken as roots. `aede roots --remove ~/Music` has
26/// `~/Music` as its positional, and feeding that to the scan would add back the
27/// folder that was just dropped.
28pub fn rescan(args: &Args) -> Res {
29    run_scan(args, Watched::Only, &mut |_| {})
30}
31
32/// Runs the watched-root scan and forwards its real traversal/read progress.
33pub fn rescan_with_progress(args: &Args, progress: &mut (dyn FnMut(Progress) + Send)) -> Res {
34    run_scan(args, Watched::Only, progress)
35}

37/// Where the roots of a run come from.
38#[derive(PartialEq)]
39enum Watched {
40    /// A user typing `aede scan <folder>` is also asking for it to be watched.
41    AndWhateverWasNamed,
42    /// A scan nobody typed: the catalog's own list, and nothing else.
43    Only,
44}
45
46fn run_scan(args: &Args, watched: Watched, on_progress: &mut (dyn FnMut(Progress) + Send)) -> Res {
47    let dir = data_dir(args);
48    let catalog_file = store::catalog_path(&dir);
49
50    // A catalog from an earlier version must not block the scan: we start over
51    // and say so.
52    let stored = match store::load(&catalog_file) {
53        Ok(value) => value,
54        Err(store::StoreError::Version { .. }) => {
55            println!(
56                "{}",
57                ui::yellow("Catalog from an earlier version: full scan.")
58            );
59            None
60        }
61        Err(e) => return Err(e.into()),
62    };
63
64    let roots = match watched {
65        Watched::AndWhateverWasNamed => resolve_roots(args, stored.as_ref())?,
66        Watched::Only => stored
67            .as_ref()
68            .map(|c| c.roots.iter().map(PathBuf::from).collect())
69            .unwrap_or_default(),
70    };
71    // `--full` only disables the tag cache; the watched folders are kept.
72    let previous = if args.has("full") {
73        None
74    } else {
75        stored.as_ref()
76    };
77
78    let options = ScanOptions {
79        threads: args.number_or("threads", 0)?,
80        follow_symlinks: args.has("follow-symlinks"),
81        skip_hidden: !args.has("include-hidden"),
82        // Read from the catalog rather than from this command line: a plain
83        // `aede scan` re-reads every root, and an exclusion that had to be
84        // retyped would be forgotten exactly when it mattered.
85        excluded: stored
86            .as_ref()
87            .map(|c| c.excluded.iter().map(PathBuf::from).collect())
88            .unwrap_or_default(),
89        // And from `user.json` for the same reason. A merge is applied as the
90        // artists are interned, so every scan has to be told about it again;
91        // a statement that only held until the next scan would be forgotten
92        // exactly when the shelf was rebuilt around it.
93        same_artist: super::merge::stated(&dir),
94    };
95
96    println!("{}", ui::bold("Scanning folders…"));
97    let mut discovered = 0usize;
98    let (mut catalog, mut report) = scan::scan(&roots, previous, &options, |progress| {
99        on_progress(progress);
100        match progress {
101            Progress::Discovered(count) => {
102                discovered = count;
103                println!("  {count} audio files spotted");
104            }
105            Progress::Read { done, total } => {
106                if total > 0 {
107                    print!("\r  reading: {done}/{total}   ");
108                    use std::io::Write;
109                    let _ = std::io::stdout().flush();
110                }
111            }
112        }
113    })?;
114    // A full scan has no cached catalog for the core to carry exclusions from.
115    // Persist the same watched-folder policy used for this traversal.
116    catalog.excluded = options
117        .excluded
118        .iter()
119        .map(|path| path.to_string_lossy().into_owned())
120        .collect();
121    if report.read > 0 {
122        println!();
123    }
124
125    // A full scan (or a catalog-format change) has no previous catalog, but
126    // conclusions have their own lifetime and are matched by path and bytes.
127    if previous.is_none() {
128        let conclusions_file = aede_core::conclusions::conclusions_path(&dir);
129        let gathered = aede_core::conclusions::load(&conclusions_file)?.or_else(|| {
130            // Reading a legacy catalog is pure. A full scan must carry its
131            // embedded conclusions until the protected save migrates them.
132            stored
133                .as_ref()
134                .map(aede_core::conclusions::Conclusions::from_catalog)
135        });
136        if let Some(gathered) = gathered {
137            // The scan may just have imported reports found beside the audio.
138            // Reapply them after older stored analyses so the fresh report wins.
139            let fresh_analyses = std::mem::take(&mut catalog.analyses);
140            gathered.attach(&mut catalog);
141            aede_core::analysis::merge_into(
142                &mut catalog,
143                fresh_analyses,
144                aede_core::clock::now_seconds(),
145            );
146            report.attached += aede_core::analysis::reconcile(&mut catalog);
147        }
148    }
149    report.analyses = catalog.analyses.len();
150    store::save(&catalog, &catalog_file)?;
151
152    println!("{}", ui::section("Scan complete"));
153    let mut table = Table::plain(2).align(1, Align::Right);
154    table.push(vec!["Files found".into(), report.found.to_string()]);
155    table.push(vec!["Read from disk".into(), report.read.to_string()]);
156    table.push(vec![
157        "Reused from previous scan".into(),
158        report.reused.to_string(),
159    ]);
160    if report.removed > 0 {
161        table.push(vec![
162            "Gone since last scan".into(),
163            report.removed.to_string(),
164        ]);
165    }
166    // Only shown when a report was actually met: on a library holding none, a
167    // line reading zero would answer a question nobody asked.
168    if report.reports > 0 {
169        table.push(vec![
170            "Analyses imported".into(),
171            format!(
172                "{} from {}",
173                report.analyses,
174                ui::plural(report.reports, "report")
175            ),
176        ]);
177    }
178    if report.attached > 0 {
179        table.push(vec![
180            "Analyses now attached".into(),
181            report.attached.to_string(),
182        ]);
183    }
184    table.push(vec!["Elapsed".into(), ui::elapsed(report.elapsed_ms)]);
185    print!("{}", table.render());
186
187    if !report.failures.is_empty() {
188        println!("{}", ui::section("Unreadable files"));
189        let mut t = Table::new(&["File", "Reason"]).path_limit(0, 60);
190        for (path, reason) in report.failures.iter().take(20) {
191            t.push(vec![path.clone(), reason.clone()]);
192        }
193        print!("{}", t.render());
194        if report.failures.len() > 20 {
195            println!(
196                "{}",
197                ui::dim(&format!("  … and {} more", report.failures.len() - 20))
198            );
199        }
200    }
201
202    let s = stats::compute(&catalog);
203    println!(
204        "\n{} {} · {} · {} · {}",
205        ui::green("→"),
206        ui::plural(s.tracks, "track"),
207        ui::plural(s.releases, "album"),
208        ui::plural(s.artists, "artist"),
209        ui::long_duration(s.total_duration_ms)
210    );
211    println!(
212        "{}",
213        ui::dim(&format!("  catalog: {}", catalog_file.display()))
214    );
215    Ok(())
216}

222/// Works out which folders to walk.
223///
224/// The watched folders accumulate across runs. Scanning a second library must
225/// add to the catalog, not silently replace it — that mistake costs the user
226/// their whole catalog with no warning. `--replace` asks for the old
227/// behaviour explicitly.
228fn resolve_roots(args: &Args, stored: Option<&Catalog>) -> Result<Vec<PathBuf>, Box<dyn Error>> {
229    let mut roots: Vec<PathBuf> = Vec::new();
230
231    if !args.has("replace")
232        && let Some(catalog) = stored
233    {
234        for root in &catalog.roots {
235            let path = PathBuf::from(root);
236            if !path.is_dir() {
237                return Err(format!(
238                    "watched folder \"{root}\" is unreachable.\n\
239                         Plug the drive back in, or drop it with: aede roots --remove \"{root}\"",
240                )
241                .into());
242            }
243            roots.push(path);
244        }
245    }
246
247    for raw in &args.positionals {
248        let path = PathBuf::from(raw);
249        if !path.is_dir() {
250            return Err(format!("\"{raw}\" is not a readable folder").into());
251        }
252        roots.push(super::canonical(&path));
253    }
254
255    if roots.is_empty() {
256        // An empty list is legitimate once a catalog exists: it means the last
257        // watched folder was dropped, and this scan is what empties the
258        // catalog — exactly what `roots --remove` said to run. Refusing here
259        // would leave those files in the catalog with no way out.
260        if stored.is_none() {
261            return Err("give at least one folder: aede scan ~/Music".into());
262        }
263        println!(
264            "{}",
265            ui::yellow("No folder is watched any more: the catalog will be emptied.")
266        );
267    }
268
269    Ok(dedupe_roots(roots))
270}
271
272/// Removes duplicates and folders already covered by another root, so a file
273/// is never walked twice.
274fn dedupe_roots(mut roots: Vec<PathBuf>) -> Vec<PathBuf> {
275    roots.sort();
276    roots.dedup();
277    let mut kept: Vec<PathBuf> = Vec::new();
278    for root in roots {
279        if kept.iter().any(|k| root.starts_with(k)) {
280            continue;
281        }
282        kept.retain(|k| !k.starts_with(&root));
283        kept.push(root);
284    }
285    kept
286}

288/// Lists the watched folders, or drops one from the list.
289/// Finishes a command that changed what the library *is*, by running the scan
290/// it just made necessary.
291///
292/// Three lines of every such command used to be an instruction: "run `aede
293/// scan` to drop them". An instruction the program could carry out itself is a
294/// chore handed back to the user, and one they can forget — leaving a catalog
295/// that describes a library nobody has any more, with nothing on screen saying
296/// so. The command that creates the need is the command that satisfies it.
297///
298/// `--no-scan` keeps the old behaviour, and it is not decoration: dropping four
299/// folders one after another would otherwise rescan four times, and a library
300/// on a slow drive makes that minutes. The message then says what is pending,
301/// so the state is never silent.
302///
303/// **`reset` deliberately does not come through here.** It destroys what a scan
304/// cannot rebuild, it is the one command that asks for confirmation, and
305/// rebuilding a catalog somebody has just chosen to throw away would answer a
306/// question they did not ask.
307fn take_effect(args: &Args, pending: &str) -> Res {
308    if args.has("no-scan") {
309        println!("{}", ui::dim(&format!("  {pending}")));
310        println!("{}", ui::dim("  run `aede scan` when you want it applied"));
311        return Ok(());
312    }
313    println!("{}", ui::dim("  rescanning so that it takes effect…"));
314    rescan(args)
315}

317pub fn roots(args: &Args) -> Res {
318    let dir = data_dir(args);
319    let catalog_file = store::catalog_path(&dir);
320    let mut catalog = load(args)?;
321
322    // Exclusions live here rather than in a command of their own: what is
323    // watched and what is deliberately not watched are one question, and a
324    // user asking "what does Aède look at" should get one screen.
325    if let Some(folder) = args.value("exclude") {
326        let wanted = super::canonical(Path::new(folder))
327            .to_string_lossy()
328            .to_string();
329        if args.has("remove") {
330            let before = catalog.excluded.len();
331            catalog.excluded.retain(|f| f != &wanted && f != folder);
332            if catalog.excluded.len() == before {
333                return Err(format!("\"{folder}\" is not excluded").into());
334            }
335            store::save(&catalog, &catalog_file)?;
336            println!("{} {folder} will be read again", ui::green("->"));
337            return take_effect(args, "it will be read again at the next scan");
338        }
339        if catalog.excluded.contains(&wanted) {
340            return Err(format!("\"{folder}\" is already excluded").into());
341        }
342        catalog.excluded.push(wanted);
343        catalog.excluded.sort();
344        store::save(&catalog, &catalog_file)?;
345        println!("{} {folder} will not be read", ui::green("->"));
346        // The files already in the catalog do not vanish because the folder
347        // stopped being read: the same promise `--remove` on a root makes, and
348        // the same way of keeping it.
349        return take_effect(args, "its files stay in the catalog until the next scan");
350    }
351
352    // `--remove` is a flag, and the folder is the argument. It used to take
353    // its value, which made one option mean two things — a flag on `love`, a
354    // value here — and an option with two meanings is one nobody can predict.
355    if args.has("remove") {
356        let target = &args.positionals.join(" ");
357        if target.trim().is_empty() {
358            return Err("which folder? aede roots --remove ~/Music".into());
359        }
360        let wanted = super::canonical(Path::new(target))
361            .to_string_lossy()
362            .to_string();
363        let before = catalog.roots.len();
364        catalog.roots.retain(|r| r != &wanted && r != target);
365        if catalog.roots.len() == before {
366            return Err(format!("\"{target}\" is not a watched folder").into());
367        }
368        store::save(&catalog, &catalog_file)?;
369        println!("{} no longer watching {target}", ui::green("->"));
370        if catalog.roots.is_empty() && !args.has("no-scan") {
371            println!(
372                "{}",
373                ui::yellow("  no folder is watched any more: the catalog will be emptied")
374            );
375        }
376        // Removing a folder from the list does not empty the catalog by
377        // itself: the files stay until a scan rebuilds it from the folders
378        // still watched. Naming this one again on that scan would simply watch
379        // it anew.
380        return take_effect(args, "its files stay in the catalog until the next scan");
381    }
382
383    println!("{}", ui::section("Watched folders"));
384    // The same three measures every listing and every page carries. A folder
385    // count without a weight is the one figure a user cannot act on: knowing a
386    // drive holds four thousand tracks says nothing about whether it will fit
387    // anywhere.
388    let mut t = Table::new(&["Folder", "Tracks", "Duration", "Size"])
389        .align(1, Align::Right)
390        .align(2, Align::Right)
391        .align(3, Align::Right)
392        .path_limit(0, 60);
393    let mut counted: std::collections::BTreeSet<Id> = Default::default();
394    for root in &catalog.roots {
395        let tracks = tracks_under(&catalog, root);
396        counted.extend(tracks.iter().copied());
397        let (duration, size) = totals(&catalog, &tracks);
398        t.push(vec![
399            root.clone(),
400            tracks.len().to_string(),
401            text::format_duration(duration),
402            text::format_size(size),
403        ]);
404    }
405
406    // Files left over from a folder that was dropped and not yet rescanned.
407    // The removal message promises they are still there; a table that does not
408    // show them makes that promise unverifiable.
409    let orphans: Vec<Id> = catalog
410        .tracks
411        .iter()
412        .map(|t| t.id)
413        .filter(|id| !counted.contains(id))
414        .collect();
415    if !orphans.is_empty() {
416        let (duration, size) = totals(&catalog, &orphans);
417        t.push(vec![
418            "(no longer watched)".into(),
419            orphans.len().to_string(),
420            text::format_duration(duration),
421            text::format_size(size),
422        ]);
423    }
424    print!("{}", t.render());
425
426    if catalog.roots.len() + usize::from(!orphans.is_empty()) > 1 {
427        let all: Vec<Id> = catalog.tracks.iter().map(|t| t.id).collect();
428        let (duration, size) = totals(&catalog, &all);
429        println!(
430            "  {}",
431            ui::dim(&format!(
432                "{} · {} · {} in all",
433                ui::plural(all.len(), "track"),
434                ui::long_duration(duration),
435                text::format_size(size)
436            ))
437        );
438    }
439    if !orphans.is_empty() {
440        println!(
441            "  {}",
442            ui::dim("run `aede scan` to drop what is no longer watched")
443        );
444    }
445
446    // Shown on the same screen as the roots, because "what does Aède read" is
447    // one question and an exclusion nobody can see is one nobody remembers
448    // setting — which is how a folder goes missing from a library for months.
449    if !catalog.excluded.is_empty() {
450        println!("{}", ui::section("Never read"));
451        let mut t = Table::new(&["Folder"]).path_limit(0, 70);
452        for folder in &catalog.excluded {
453            t.push(vec![folder.clone()]);
454        }
455        print!("{}", t.render());
456        println!(
457            "  {}",
458            ui::dim("aede roots --exclude <folder> --remove reads it again")
459        );
460    }
461    // Where the answers themselves are kept. `roots` is the command someone
462    // runs to ask where their things are, and the catalog file was the one
463    // location it never named — only `scan` mentioned it, once, in passing.
464    println!(
465        "  {}",
466        ui::dim(&format!("catalog: {}", catalog_file.display()))
467    );
468    Ok(())
469}

471/// Tracks whose file sits in a folder, or in something under it.
472///
473/// On the path boundary, never on the bare string: `/music/Rock` would
474/// otherwise claim every file of `/music/Rockabilly`, and the count of a
475/// watched folder would quietly include a neighbour's.
476fn tracks_under(catalog: &Catalog, root: &str) -> Vec<Id> {
477    catalog
478        .tracks
479        .iter()
480        .filter(|t| {
481            catalog
482                .file(t.file_id)
483                .is_some_and(|f| aede_core::text::is_under(&f.path, root))
484        })
485        .map(|t| t.id)
486        .collect()
487}
