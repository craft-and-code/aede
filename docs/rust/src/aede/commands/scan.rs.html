22/// Rescans the watched folders on the user's behalf, after a command changed
23/// what "watched" means.
24///
25/// Same scan a bare `aede scan` runs, with one difference that is the whole
26/// reason this exists rather than a call to [`scan()`]: the folders named on the
27/// command line are **not** taken as roots. `aede roots --remove ~/Music` has
28/// `~/Music` as its positional, and feeding that to the scan would add back the
29/// folder that was just dropped.
30pub fn rescan(args: &Args) -> Res {
31    run_scan(args, Watched::Only, &mut |_| {})
32}

34/// Runs the watched-root scan and forwards its real traversal/read progress.
35pub fn rescan_with_progress(args: &Args, progress: &mut (dyn FnMut(Progress) + Send)) -> Res {
36    run_scan(args, Watched::Only, progress)
37}

39/// Where the roots of a run come from.
40#[derive(PartialEq)]
41enum Watched {
42    /// A user typing `aede scan <folder>` is also asking for it to be watched.
43    AndWhateverWasNamed,
44    /// A scan nobody typed: the catalog's own list, and nothing else.
45    Only,
46}
47
48fn run_scan(args: &Args, watched: Watched, on_progress: &mut (dyn FnMut(Progress) + Send)) -> Res {
49    let dir = data_dir(args);
50    let catalog_file = store::catalog_path(&dir);
51
52    // A catalog from an earlier version must not block the scan: we start over
53    // and say so.
54    let stored = match store::load(&catalog_file) {
55        Ok(value) => value,
56        Err(store::StoreError::Version { .. }) => {
57            eprintln!(
58                "{}",
59                ui::yellow("Catalog from an earlier version: full scan.")
60            );
61            None
62        }
63        Err(e) => return Err(e.into()),
64    };
65
66    let roots = match watched {
67        Watched::AndWhateverWasNamed => resolve_roots(args, stored.as_ref())?,
68        Watched::Only => stored
69            .as_ref()
70            .map(|c| c.roots.iter().map(PathBuf::from).collect())
71            .unwrap_or_default(),
72    };
73    // A full read still needs the previous catalog as evidence for paths the
74    // filesystem cannot currently expose. An unavailable folder is not empty.
75    let previous = stored.as_ref();
76
77    let options = ScanOptions {
78        force_read: args.has("full"),
79        threads: args.number_or("threads", 0)?,
80        follow_symlinks: args.has("follow-symlinks"),
81        skip_hidden: !args.has("include-hidden"),
82        // Read from the catalog rather than from this command line: a plain
83        // `aede scan` re-reads every root, and an exclusion that had to be
84        // retyped would be forgotten exactly when it mattered.
85        excluded: stored
86            .as_ref()
87            .map(|c| c.excluded.iter().map(PathBuf::from).collect())
88            .unwrap_or_default(),
89        // And from `user.json` for the same reason. A merge is applied as the
90        // artists are interned, so every scan has to be told about it again;
91        // a statement that only held until the next scan would be forgotten
92        // exactly when the shelf was rebuilt around it.
93        same_artist: super::merge::stated(&dir),
94    };
95
96    let machine = args.has("json");
97    if !machine {
98        println!(
99            "{}",
100            ui::bold(if args.has("dry-run") {
101                "Previewing scan…"
102            } else {
103                "Scanning folders…"
104            })
105        );
106    }
107    let (mut catalog, mut report) = scan::scan(&roots, previous, &options, |progress| {
108        on_progress(progress);
109        if machine {
110            return;
111        }
112        match progress {
113            Progress::Discovered(count) => {
114                println!("  {count} audio files spotted");
115            }
116            Progress::Read { done, total } => {
117                if total > 0 {
118                    print!("\r  reading: {done}/{total}   ");
119                    use std::io::Write;
120                    let _ = std::io::stdout().flush();
121                }
122            }
123        }
124    })?;
125    // Persist the watched-folder policy used for this traversal.
126    catalog.excluded = options
127        .excluded
128        .iter()
129        .map(|path| path.to_string_lossy().into_owned())
130        .collect();
131    if report.read > 0 && !machine {
132        println!();
133    }
134
135    // Freshly read files have no inherited verdicts. The independent store
136    // can reattach results when the complete file identity still matches.
137    if options.force_read || previous.is_none() {
138        let conclusions_file = aede_core::conclusions::conclusions_path(&dir);
139        let gathered = aede_core::conclusions::load(&conclusions_file)?.or_else(|| {
140            // Reading a legacy catalog is pure. A full scan must carry its
141            // embedded conclusions until the protected save migrates them.
142            stored
143                .as_ref()
144                .map(aede_core::conclusions::Conclusions::from_catalog)
145        });
146        if let Some(gathered) = gathered {
147            // The scan may just have imported reports found beside the audio.
148            // Reapply them after older stored analyses so the fresh report wins.
149            let fresh_analyses = std::mem::take(&mut catalog.analyses);
150            gathered.attach(&mut catalog);
151            aede_core::analysis::merge_into(
152                &mut catalog,
153                fresh_analyses,
154                aede_core::clock::now_seconds(),
155            );
156            report.attached += aede_core::analysis::reconcile(&mut catalog);
157        }
158    }
159    report.analyses = catalog.analyses.len();
160    if args.has("dry-run") {
161        report_output::print(&report, &catalog, &catalog_file, true, machine);
162        return Ok(());
163    }
164
165    // Persist evidence before replacing the graph. If the final reconciliation
166    // cannot be saved, old references still carry enough evidence to recover
167    // safely on the next read rather than guessing from a filename.
168    let user_file = aede_core::user::user_path(&dir);
169    let mut personal = aede_core::user::load(&user_file)?;
170    if let (Some(data), Some(previous)) = (&mut personal, previous) {
171        let before = aede_core::user::to_json(data);
172        aede_core::user::reconcile(data, previous);
173        if aede_core::user::to_json(data) != before {
174            aede_core::user::save(data, &user_file)?;
175        }
176    }
177    store::save(&catalog, &catalog_file)?;
178    if let Some(data) = &mut personal {
179        let before = aede_core::user::to_json(data);
180        aede_core::user::reconcile(data, &catalog);
181        if aede_core::user::to_json(data) != before {
182            aede_core::user::save(data, &user_file)?;
183        }
184    }
185    report_output::print(&report, &catalog, &catalog_file, false, machine);
186    Ok(())
187}

193/// Works out which folders to walk.
194///
195/// The watched folders accumulate across runs. Scanning a second library must
196/// add to the catalog, not silently replace it — that mistake costs the user
197/// their whole catalog with no warning. `--replace` asks for the old
198/// behaviour explicitly.
199fn resolve_roots(args: &Args, stored: Option<&Catalog>) -> Result<Vec<PathBuf>, Box<dyn Error>> {
200    let mut roots: Vec<PathBuf> = Vec::new();
201
202    if !args.has("replace")
203        && let Some(catalog) = stored
204    {
205        for root in &catalog.roots {
206            let path = PathBuf::from(root);
207            // A saved root may be temporarily unmounted. Let the core report
208            // its failure and preserve its entries while other roots advance.
209            roots.push(path);
210        }
211    }
212
213    for raw in &args.positionals {
214        let path = PathBuf::from(raw);
215        if !path.is_dir() {
216            return Err(format!("\"{raw}\" is not a readable folder").into());
217        }
218        roots.push(super::canonical(&path));
219    }
220
221    if roots.is_empty() {
222        // An empty list is legitimate once a catalog exists: it means the last
223        // watched folder was dropped, and this scan is what empties the
224        // catalog — exactly what `roots --remove` said to run. Refusing here
225        // would leave those files in the catalog with no way out.
226        if stored.is_none() {
227            return Err("give at least one folder: aede scan ~/Music".into());
228        }
229        eprintln!(
230            "{}",
231            ui::yellow("No folder is watched any more: the catalog will be emptied.")
232        );
233    }
234
235    Ok(dedupe_roots(roots))
236}

238/// Removes duplicates and folders already covered by another root, so a file
239/// is never walked twice.
240fn dedupe_roots(mut roots: Vec<PathBuf>) -> Vec<PathBuf> {
241    roots.sort();
242    roots.dedup();
243    let mut kept: Vec<PathBuf> = Vec::new();
244    for root in roots {
245        if kept.iter().any(|k| root.starts_with(k)) {
246            continue;
247        }
248        kept.retain(|k| !k.starts_with(&root));
249        kept.push(root);
250    }
251    kept
252}

254/// Lists the watched folders, or drops one from the list.
255/// Finishes a command that changed what the library *is*, by running the scan
256/// it just made necessary.
257///
258/// Three lines of every such command used to be an instruction: "run `aede
259/// scan` to drop them". An instruction the program could carry out itself is a
260/// chore handed back to the user, and one they can forget — leaving a catalog
261/// that describes a library nobody has any more, with nothing on screen saying
262/// so. The command that creates the need is the command that satisfies it.
263///
264/// `--no-scan` keeps the old behaviour, and it is not decoration: dropping four
265/// folders one after another would otherwise rescan four times, and a library
266/// on a slow drive makes that minutes. The message then says what is pending,
267/// so the state is never silent.
268///
269/// **`reset` deliberately does not come through here.** It destroys what a scan
270/// cannot rebuild, it is the one command that asks for confirmation, and
271/// rebuilding a catalog somebody has just chosen to throw away would answer a
272/// question they did not ask.
273fn take_effect(args: &Args, pending: &str) -> Res {
274    if args.has("no-scan") {
275        println!("{}", ui::dim(&format!("  {pending}")));
276        println!("{}", ui::dim("  run `aede scan` when you want it applied"));
277        return Ok(());
278    }
279    println!("{}", ui::dim("  rescanning so that it takes effect…"));
280    rescan(args)
281}

283pub fn roots(args: &Args) -> Res {
284    let dir = data_dir(args);
285    let catalog_file = store::catalog_path(&dir);
286    let mut catalog = load(args)?;
287
288    // Exclusions live here rather than in a command of their own: what is
289    // watched and what is deliberately not watched are one question, and a
290    // user asking "what does Aède look at" should get one screen.
291    if let Some(folder) = args.value("exclude") {
292        let wanted = super::path_key(&super::canonical(Path::new(folder)))?;
293        if args.has("remove") {
294            let before = catalog.excluded.len();
295            catalog.excluded.retain(|f| f != &wanted && f != folder);
296            if catalog.excluded.len() == before {
297                return Err(format!("\"{folder}\" is not excluded").into());
298            }
299            store::save(&catalog, &catalog_file)?;
300            println!("{} {folder} will be read again", ui::green("->"));
301            return take_effect(args, "it will be read again at the next scan");
302        }
303        if catalog.excluded.contains(&wanted) {
304            return Err(format!("\"{folder}\" is already excluded").into());
305        }
306        catalog.excluded.push(wanted);
307        catalog.excluded.sort();
308        store::save(&catalog, &catalog_file)?;
309        println!("{} {folder} will not be read", ui::green("->"));
310        // The files already in the catalog do not vanish because the folder
311        // stopped being read: the same promise `--remove` on a root makes, and
312        // the same way of keeping it.
313        return take_effect(args, "its files stay in the catalog until the next scan");
314    }
315
316    // `--remove` is a flag, and the folder is the argument. It used to take
317    // its value, which made one option mean two things — a flag on `love`, a
318    // value here — and an option with two meanings is one nobody can predict.
319    if args.has("remove") {
320        let target = &args.positionals.join(" ");
321        if target.trim().is_empty() {
322            return Err("which folder? aede roots --remove ~/Music".into());
323        }
324        let wanted = super::path_key(&super::canonical(Path::new(target)))?;
325        let before = catalog.roots.len();
326        catalog.roots.retain(|r| r != &wanted && r != target);
327        if catalog.roots.len() == before {
328            return Err(format!("\"{target}\" is not a watched folder").into());
329        }
330        store::save(&catalog, &catalog_file)?;
331        println!("{} no longer watching {target}", ui::green("->"));
332        if catalog.roots.is_empty() && !args.has("no-scan") {
333            println!(
334                "{}",
335                ui::yellow("  no folder is watched any more: the catalog will be emptied")
336            );
337        }
338        // Removing a folder from the list does not empty the catalog by
339        // itself: the files stay until a scan rebuilds it from the folders
340        // still watched. Naming this one again on that scan would simply watch
341        // it anew.
342        return take_effect(args, "its files stay in the catalog until the next scan");
343    }
344
345    println!("{}", ui::section("Watched folders"));
346    // The same three measures every listing and every page carries. A folder
347    // count without a weight is the one figure a user cannot act on: knowing a
348    // drive holds four thousand tracks says nothing about whether it will fit
349    // anywhere.
350    let mut t = Table::new(&["Folder", "Tracks", "Duration", "Size"])
351        .align(1, Align::Right)
352        .align(2, Align::Right)
353        .align(3, Align::Right)
354        .path_limit(0, 60);
355    let mut counted: std::collections::BTreeSet<Id> = Default::default();
356    for root in &catalog.roots {
357        let tracks = tracks_under(&catalog, root);
358        counted.extend(tracks.iter().copied());
359        let (duration, size) = totals(&catalog, &tracks);
360        t.push(vec![
361            root.clone(),
362            tracks.len().to_string(),
363            text::format_duration(duration),
364            text::format_size(size),
365        ]);
366    }
367
368    // Files left over from a folder that was dropped and not yet rescanned.
369    // The removal message promises they are still there; a table that does not
370    // show them makes that promise unverifiable.
371    let orphans: Vec<Id> = catalog
372        .tracks
373        .iter()
374        .map(|t| t.id)
375        .filter(|id| !counted.contains(id))
376        .collect();
377    if !orphans.is_empty() {
378        let (duration, size) = totals(&catalog, &orphans);
379        t.push(vec![
380            "(no longer watched)".into(),
381            orphans.len().to_string(),
382            text::format_duration(duration),
383            text::format_size(size),
384        ]);
385    }
386    print!("{}", t.render());
387
388    if catalog.roots.len() + usize::from(!orphans.is_empty()) > 1 {
389        let all: Vec<Id> = catalog.tracks.iter().map(|t| t.id).collect();
390        let (duration, size) = totals(&catalog, &all);
391        println!(
392            "  {}",
393            ui::dim(&format!(
394                "{} · {} · {} in all",
395                ui::plural(all.len(), "track"),
396                ui::long_duration(duration),
397                text::format_size(size)
398            ))
399        );
400    }
401    if !orphans.is_empty() {
402        println!(
403            "  {}",
404            ui::dim("run `aede scan` to drop what is no longer watched")
405        );
406    }
407
408    // Shown on the same screen as the roots, because "what does Aède read" is
409    // one question and an exclusion nobody can see is one nobody remembers
410    // setting — which is how a folder goes missing from a library for months.
411    if !catalog.excluded.is_empty() {
412        println!("{}", ui::section("Never read"));
413        let mut t = Table::new(&["Folder"]).path_limit(0, 70);
414        for folder in &catalog.excluded {
415            t.push(vec![folder.clone()]);
416        }
417        print!("{}", t.render());
418        println!(
419            "  {}",
420            ui::dim("aede roots --exclude <folder> --remove reads it again")
421        );
422    }
423    // Where the answers themselves are kept. `roots` is the command someone
424    // runs to ask where their things are, and the catalog file was the one
425    // location it never named — only `scan` mentioned it, once, in passing.
426    println!(
427        "  {}",
428        ui::dim(&format!("catalog: {}", catalog_file.display()))
429    );
430    Ok(())
431}

433/// Tracks whose file sits in a folder, or in something under it.
434///
435/// On the path boundary, never on the bare string: `/music/Rock` would
436/// otherwise claim every file of `/music/Rockabilly`, and the count of a
437/// watched folder would quietly include a neighbour's.
438fn tracks_under(catalog: &Catalog, root: &str) -> Vec<Id> {
439    catalog
440        .tracks
441        .iter()
442        .filter(|t| {
443            catalog
444                .file(t.file_id)
445                .is_some_and(|f| aede_core::text::is_under(&f.path, root))
446        })
447        .map(|t| t.id)
448        .collect()
449}
