1//! Explicit full decoding without audio output or persisted container verdicts.
2use super::*;
3use aede_core::playback::decoder::{FileDecoder, FlacMd5Status};
4use std::path::Path;
5use std::process::{Command, Stdio};

7struct Verification {
8    method: &'static str,
9    hash: bool,
10}
11struct FileResult<'a> {
12    path: &'a str,
13    outcome: Result<Verification, String>,
14}
15
16pub(super) fn run(catalog: &Catalog, scope: &[String], json: bool) -> Res {
17    let rows = catalog
18        .files
19        .iter()
20        .filter(|file| super::super::in_scope(&file.path, scope))
21        .map(|file| FileResult {
22            path: &file.path,
23            outcome: verify(
24                Path::new(&file.path),
25                file.properties.codec == "flac",
26                Some(Path::new("flac")),
27            ),
28        });
29    let failed = if json {
30        write_json(&mut std::io::stdout().lock(), rows)?
31    } else {
32        let rows: Vec<_> = rows.collect();
33        let failed = rows.iter().filter(|row| row.outcome.is_err()).count();
34        let mut table = Table::new(&["File", "Result", "Verifier"]).path_limit(0, 60);
35        for row in &rows {
36            match &row.outcome {
37                Ok(value) => {
38                    table.push(vec![row.path.into(), "Decoded".into(), value.method.into()])
39                }
40                Err(error) => table.push(vec![row.path.into(), error.clone(), "Failed".into()]),
41            }
42        }
43        ui::section("Full decode test");
44        print!("{}", table.render());
45        failed
46    };
47    if failed > 0 {
48        return Err(format!("{failed} file(s) failed the complete decode test").into());
49    }
50    Ok(())
51}

53// Flush after each completed row: the final document remains ordinary version-1 JSON.
54fn write_json<'a>(
55    writer: &mut impl std::io::Write,
56    rows: impl Iterator<Item = FileResult<'a>>,
57) -> std::io::Result<usize> {
58    writeln!(writer, "{{\"version\":1,\"files\":[")?;
59    let mut failed = 0;
60    for (index, row) in rows.enumerate() {
61        failed += usize::from(row.outcome.is_err());
62        writeln!(
63            writer,
64            "{}{}",
65            if index == 0 { "" } else { "," },
66            structured_row(&row).to_string_compact()
67        )?;
68        writer.flush()?;
69    }
70    writeln!(writer, "]}}")?;
71    writer.flush()?;
72    Ok(failed)
73}
74fn structured_row(row: &FileResult<'_>) -> aede_core::json::Json {
75    use aede_core::json::Json;
76    let mut value = Json::obj();
77    value.set("path", Json::Str(row.path.into()));
78    match &row.outcome {
79        Ok(result) => {
80            value.set(
81                "status",
82                Json::Str(if result.hash { "ok" } else { "no_hash" }.into()),
83            );
84            value.set("verifier", Json::Str(result.method.into()));
85        }
86        Err(error) => {
87            value.set("status", Json::Str("ko".into()));
88            value.set("verifier", Json::Str("Failed".into()));
89            value.set("message", Json::Str(error.clone()));
90        }
91    }
92    value
93}

106fn verify(path: &Path, flac: bool, program: Option<&Path>) -> Result<Verification, String> {
107    let before = std::fs::symlink_metadata(path).map_err(|e| e.to_string())?;
108    if !before.is_file() {
109        return Err("source must be a regular file, not a link".into());
110    }
111    let method = if flac && let Some(program) = program {
112        match external(program, path) {
113            Ok((status, _)) if status.success() => {
114                let status = FileDecoder::open(path)
115                    .map_err(|error| error.to_string())?
116                    .flac_md5_status();
117                let hash = match status {
118                    Some(FlacMd5Status::Pending | FlacMd5Status::Verified) => true,
119                    Some(FlacMd5Status::NoSignature) => false,
120                    _ => return Err("FLAC MD5 verification state is invalid".into()),
121                };
122                Verification {
123                    method: if hash {
124                        "flac -t"
125                    } else {
126                        "flac -t · decoded, no FLAC MD5"
127                    },
128                    hash,
129                }
130            }
131            Ok((_, diagnostic)) => {
132                return Err(if diagnostic.is_empty() {
133                    "flac -t rejected the complete stream".into()
134                } else {
135                    format!("flac -t rejected the complete stream:\n{diagnostic}")
136                });
137            }
138            Err(error) if error.kind() == std::io::ErrorKind::NotFound => native(path)?,
139            Err(error) => return Err(error.to_string()),
140        }
141    } else {
142        native(path)?
143    };
144    let after = std::fs::symlink_metadata(path).map_err(|e| e.to_string())?;
145    if !after.is_file()
146        || before.len() != after.len()
147        || before.modified().ok() != after.modified().ok()
148    {
149        return Err("source changed during verification".into());
150    }
151    Ok(method)
152}
153// stdout is discarded, so draining this single pipe before wait cannot fill a second pipe.
154fn external(program: &Path, path: &Path) -> std::io::Result<(std::process::ExitStatus, String)> {
155    let mut child = Command::new(program)
156        .args(["--test", "--silent", "--"])
157        .arg(path)
158        .stdin(Stdio::null())
159        .stdout(Stdio::null())
160        .stderr(Stdio::piped())
161        .spawn()?;
162    let diagnostic = child.stderr.take().map(decode_diagnostic).transpose();
163    let status = child.wait()?;
164    Ok((status, diagnostic?.unwrap_or_default()))
165}

167// Retain a small prefix but consume all bytes; control escapes stay within the client limit.
168fn decode_diagnostic(mut input: impl std::io::Read) -> std::io::Result<String> {
169    use std::io::Read;
170    let mut bytes = Vec::new();
171    input.by_ref().take(3000).read_to_end(&mut bytes)?;
172    let hidden = std::io::copy(&mut input, &mut std::io::sink())? > 0;
173    let mut text = ui::literal(String::from_utf8_lossy(&bytes).trim());
174    let truncated = hidden || text.len() > 3500;
175    if text.len() > 3500 {
176        let mut end = 3500;
177        while !text.is_char_boundary(end) {
178            end -= 1;
179        }
180        text.truncate(end);
181    }
182    if truncated {
183        text.push_str("\n[diagnostic truncated]");
184    }
185    Ok(text)
186}
187fn native(path: &Path) -> Result<Verification, String> {
188    let mut decoder = FileDecoder::open(path).map_err(|e| e.to_string())?;
189    let channels = usize::from(decoder.channels());
190    if channels == 0 || channels > 64 {
191        return Err("unsupported channel count".into());
192    }
193    let mut buffer = vec![0.0; 4096 * channels];
194    while decoder
195        .read_frames(&mut buffer)
196        .map_err(|e| e.to_string())?
197        != 0
198    {}
199    match decoder.flac_md5_status() {
200        Some(FlacMd5Status::Verified) => Ok(Verification {
201            method: "Rust · FLAC MD5 verified",
202            hash: true,
203        }),
204        Some(FlacMd5Status::NoSignature) => Ok(Verification {
205            method: "Rust · decoded, no FLAC MD5",
206            hash: false,
207        }),
208        Some(_) => Err("FLAC MD5 verification did not complete".into()),
209        None => Ok(Verification {
210            method: "Rust · decoded, no FLAC MD5",
211            hash: false,
212        }),
213    }
